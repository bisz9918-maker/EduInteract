    <script>
    (function() {
        let currentStep = 0;
        const maxStep = 5;
        
        const btnPrev = document.getElementById('scene1-btn-prev');
        const btnNext = document.getElementById('scene1-btn-next');
        
        // Elements
        const rectABCD = document.getElementById('scene1-rect-abcd');
        const rhombusPQCB = document.getElementById('scene1-rhombus-pqcb');
        const auxGroup = document.getElementById('scene1-aux-group');
        const triPBH = document.getElementById('scene1-tri-pbh');
        const trapBPEC = document.getElementById('scene1-trap-bpec');
        
        // Labels
        const labelH = document.getElementById('scene1-label-h');
        const valAD = document.getElementById('scene1-val-ad');
        const valCD = document.getElementById('scene1-val-cd');
        const valBP = document.getElementById('scene1-val-bp');
        const valPH = document.getElementById('scene1-val-ph');
        const valBH = document.getElementById('scene1-val-bh');
        const valPE = document.getElementById('scene1-val-pe');
        const valCE = document.getElementById('scene1-val-ce');
        
        // Texts
        const textSteps = [
            null, // step 0
            document.getElementById('scene1-text-step1'),
            document.getElementById('scene1-text-step2'),
            document.getElementById('scene1-text-step3'),
            document.getElementById('scene1-text-step4'),
            document.getElementById('scene1-text-step5')
        ];

        function show(el) { if(el) el.classList.remove('scene1-hidden'); }
        function hide(el) { if(el) el.classList.add('scene1-hidden'); }
        
        function updateStep() {
            btnPrev.disabled = currentStep === 0;
            btnNext.disabled = currentStep === maxStep;
            
            // Reset all highlights
            rectABCD.classList.remove('scene1-highlight-blue');
            rhombusPQCB.classList.remove('scene1-highlight-blue');
            hide(triPBH);
            hide(trapBPEC);
            
            // Step 1
            if (currentStep >= 1) {
                if (currentStep === 1) rectABCD.classList.add('scene1-highlight-blue');
                show(valAD);
                if (currentStep < 4) show(valCD); else hide(valCD);
                textSteps[1].classList.add('active');
            } else {
                hide(valAD);
                hide(valCD);
                textSteps[1].classList.remove('active');
            }
            
            // Step 2
            if (currentStep >= 2) {
                if (currentStep === 2) rhombusPQCB.classList.add('scene1-highlight-blue');
                show(auxGroup);
                show(labelH);
                show(valBP);
                show(valPH);
                textSteps[2].classList.add('active');
            } else {
                hide(auxGroup);
                hide(labelH);
                hide(valBP);
                hide(valPH);
                textSteps[2].classList.remove('active');
            }
            
            // Step 3
            if (currentStep >= 3) {
                if (currentStep === 3) {
                    show(triPBH);
                    triPBH.classList.add('scene1-highlight-red', 'scene1-fill-red');
                }
                show(valBH);
                textSteps[3].classList.add('active');
            } else {
                hide(valBH);
                textSteps[3].classList.remove('active');
            }
            
            // Step 4
            if (currentStep >= 4) {
                if (currentStep === 4) {
                    show(trapBPEC);
                    trapBPEC.classList.add('scene1-highlight-blue');
                }
                show(valPE);
                show(valCE);
                textSteps[4].classList.add('active');
            } else {
                hide(valPE);
                hide(valCE);
                textSteps[4].classList.remove('active');
            }
            
            // Step 5
            if (currentStep >= 5) {
                textSteps[5].classList.add('active');
            } else {
                textSteps[5].classList.remove('active');
            }
        }
        
        btnPrev.addEventListener('click', () => {
            if (currentStep > 0) {
                currentStep--;
                updateStep();
            }
        });
        
        btnNext.addEventListener('click', () => {
            if (currentStep < maxStep) {
                currentStep++;
                updateStep();
            }
        });
        
        // Initial render
        updateStep();
    })();
    </script>
