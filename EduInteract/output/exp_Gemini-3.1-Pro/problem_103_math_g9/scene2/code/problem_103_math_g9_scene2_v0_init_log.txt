    <script>
    (function() {
        let currentStep = 0;
        const maxStep = 4;
        
        const btnPrev = document.getElementById('scene2-btn-prev');
        const btnNext = document.getElementById('scene2-btn-next');
        
        const shadowAFED = document.getElementById('scene2-shadow-afed');
        const shadowBFP = document.getElementById('scene2-shadow-bfp');
        const linePF = document.getElementById('scene2-line-pf');
        const markF = document.getElementById('scene2-mark-f');
        const labelF = document.getElementById('scene2-label-f');
        
        const dimAF = document.getElementById('scene2-dim-af');
        const dimAD = document.getElementById('scene2-dim-ad');
        const dimFB = document.getElementById('scene2-dim-fb');
        const dimFP = document.getElementById('scene2-dim-fp');
        
        function updateState() {
            // Update buttons
            btnPrev.disabled = currentStep === 0;
            btnNext.disabled = currentStep === maxStep;
            
            // Update texts
            for (let i = 0; i <= maxStep; i++) {
                const textEl = document.getElementById('scene2-text-' + i);
                if (i === currentStep) {
                    textEl.classList.add('active');
                } else {
                    textEl.classList.remove('active');
                }
            }
            
            // Update visuals
            // Step 0: Initial
            if (currentStep === 0) {
                linePF.style.opacity = '0';
                markF.style.opacity = '0';
                labelF.style.opacity = '0';
                
                shadowAFED.setAttribute('fill', '#888888');
                shadowAFED.setAttribute('fill-opacity', '0.3');
                dimAF.style.opacity = '0';
                dimAD.style.opacity = '0';
                
                shadowBFP.setAttribute('fill', '#888888');
                shadowBFP.setAttribute('fill-opacity', '0.3');
                dimFB.style.opacity = '0';
                dimFP.style.opacity = '0';
            }
            // Step 1: Auxiliary line
            if (currentStep >= 1) {
                linePF.style.opacity = '1';
                markF.style.opacity = '1';
                labelF.style.opacity = '1';
            } else {
                linePF.style.opacity = '0';
                markF.style.opacity = '0';
                labelF.style.opacity = '0';
            }
            
            // Step 2: Rectangle
            if (currentStep >= 2) {
                shadowAFED.setAttribute('fill', '#1565c0');
                shadowAFED.setAttribute('fill-opacity', '0.4');
                dimAF.style.opacity = '1';
                dimAD.style.opacity = '1';
            } else if (currentStep < 2) {
                shadowAFED.setAttribute('fill', '#888888');
                shadowAFED.setAttribute('fill-opacity', '0.3');
                dimAF.style.opacity = '0';
                dimAD.style.opacity = '0';
            }
            
            // Step 3 & 4: Triangle
            if (currentStep >= 3) {
                shadowBFP.setAttribute('fill', '#2e7d32');
                shadowBFP.setAttribute('fill-opacity', '0.4');
                dimFB.style.opacity = '1';
                dimFP.style.opacity = '1';
            } else if (currentStep < 3) {
                shadowBFP.setAttribute('fill', '#888888');
                shadowBFP.setAttribute('fill-opacity', '0.3');
                dimFB.style.opacity = '0';
                dimFP.style.opacity = '0';
            }
        }
        
        btnPrev.addEventListener('click', () => {
            if (currentStep > 0) {
                currentStep--;
                updateState();
            }
        });
        
        btnNext.addEventListener('click', () => {
            if (currentStep < maxStep) {
                currentStep++;
                updateState();
            }
        });
        
        // Initialize
        updateState();
    })();
    </script>
