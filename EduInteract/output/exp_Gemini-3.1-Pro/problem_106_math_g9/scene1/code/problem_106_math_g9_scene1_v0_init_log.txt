    <script>
    (function() {
        let currentStep = 0;
        const maxStep = 4;
        
        const btnPrev = document.getElementById('scene1-btn-prev');
        const btnNext = document.getElementById('scene1-btn-next');
        
        // Elements
        const lineAB = document.getElementById('scene1-line-ab');
        const angleACB = document.getElementById('scene1-angle-acb');
        const angleADB = document.getElementById('scene1-angle-adb');
        
        const arcCD = document.getElementById('scene1-arc-cd');
        const angleDAE = document.getElementById('scene1-angle-dae');
        const angleCBE = document.getElementById('scene1-angle-cbe');
        const fillADE = document.getElementById('scene1-fill-ade');
        const fillBCE = document.getElementById('scene1-fill-bce');
        
        const labelAE = document.getElementById('scene1-label-ae');
        const labelCE = document.getElementById('scene1-label-ce');
        const labelDE = document.getElementById('scene1-label-de');
        
        const hlADE = document.getElementById('scene1-hl-ade');
        const eqText = document.getElementById('scene1-equation-text');
        
        function updateScene() {
            // Reset all
            lineAB.classList.remove('scene1-hl-blue');
            angleACB.classList.add('scene1-hidden');
            angleADB.classList.add('scene1-hidden');
            
            arcCD.classList.add('scene1-hidden');
            angleDAE.classList.add('scene1-hidden');
            angleCBE.classList.add('scene1-hidden');
            fillADE.classList.add('scene1-hidden');
            fillBCE.classList.add('scene1-hidden');
            
            labelAE.classList.add('scene1-hidden');
            labelCE.classList.add('scene1-hidden');
            labelDE.classList.add('scene1-hidden');
            
            hlADE.classList.add('scene1-hidden');
            eqText.classList.add('scene1-hidden');
            
            // Apply steps
            if (currentStep >= 1) {
                lineAB.classList.add('scene1-hl-blue');
                angleACB.classList.remove('scene1-hidden');
                angleADB.classList.remove('scene1-hidden');
            }
            if (currentStep >= 2) {
                arcCD.classList.remove('scene1-hidden');
                angleDAE.classList.remove('scene1-hidden');
                angleCBE.classList.remove('scene1-hidden');
                fillADE.classList.remove('scene1-hidden');
                fillBCE.classList.remove('scene1-hidden');
            }
            if (currentStep >= 3) {
                labelAE.classList.remove('scene1-hidden');
                labelCE.classList.remove('scene1-hidden');
                labelDE.classList.remove('scene1-hidden');
            }
            if (currentStep >= 4) {
                hlADE.classList.remove('scene1-hidden');
                eqText.classList.remove('scene1-hidden');
            }
            
            // Update buttons
            btnPrev.disabled = currentStep === 0;
            btnNext.disabled = currentStep === maxStep;
        }
        
        btnPrev.addEventListener('click', () => {
            if (currentStep > 0) {
                currentStep--;
                updateScene();
            }
        });
        
        btnNext.addEventListener('click', () => {
            if (currentStep < maxStep) {
                currentStep++;
                updateScene();
            }
        });
        
        // Initial state
        updateScene();
    })();
    </script>
