    <script>
    (function() {
        function describeArc(x, y, radius, startAngle, endAngle) {
            var start = polarToCartesian(x, y, radius, endAngle);
            var end = polarToCartesian(x, y, radius, startAngle);
            var largeArcFlag = endAngle - startAngle <= 180 ? "0" : "1";
            var d = [
                "M", start.x, start.y, 
                "A", radius, radius, 0, largeArcFlag, 0, end.x, end.y
            ].join(" ");
            return d;       
        }

        function polarToCartesian(centerX, centerY, radius, angleInDegrees) {
            var angleInRadians = (angleInDegrees) * Math.PI / 180.0;
            return {
                x: centerX + (radius * Math.cos(angleInRadians)),
                y: centerY + (radius * Math.sin(angleInRadians))
            };
        }

        document.getElementById('scene1-arc-aoc').setAttribute('d', describeArc(300, 350, 40, -100, -30));
        document.getElementById('scene1-arc-a').setAttribute('d', describeArc(435, 272, 20, 50.2, 150));
        document.getElementById('scene1-arc-c').setAttribute('d', describeArc(273, 197, 20, -20.2, 80));
        document.getElementById('scene1-arc-d').setAttribute('d', describeArc(368, 162, 20, 109.9, 159.8));
        document.getElementById('scene1-arc-cod').setAttribute('d', describeArc(300, 350, 70, -100, -70));
        document.getElementById('scene1-arc-aod').setAttribute('d', describeArc(300, 350, 90, -70, -30));

        let currentStep = 0;
        const maxStep = 4;
        
        const btnPrev = document.getElementById('scene1-btn-prev');
        const btnNext = document.getElementById('scene1-btn-next');
        
        function show(id) {
            const el = document.getElementById(id);
            if(el) {
                el.classList.remove('scene1-hidden');
                el.classList.add('scene1-visible');
            }
        }
        
        function hide(id) {
            const el = document.getElementById(id);
            if(el) {
                el.classList.remove('scene1-visible');
                el.classList.add('scene1-hidden');
            }
        }

        function showText(id) {
            const el = document.getElementById(id);
            if(el) {
                el.style.display = 'block';
                // trigger reflow
                void el.offsetWidth;
                el.style.opacity = 1;
            }
        }

        function hideText(id) {
            const el = document.getElementById(id);
            if(el) {
                el.style.opacity = 0;
                el.style.display = 'none';
            }
        }
        
        function updateScene() {
            btnPrev.disabled = currentStep === 0;
            btnNext.disabled = currentStep === maxStep;
            
            hide('scene1-rot-tri');
            document.getElementById('scene1-rot-tri').style.transform = 'rotate(0deg)';
            document.getElementById('scene1-rot-tri').style.transformOrigin = '300px 350px';
            
            hide('scene1-arc-aoc'); hide('scene1-val-aoc');
            hide('scene1-arc-a'); hide('scene1-val-a');
            hide('scene1-arc-c'); hide('scene1-val-c');
            hide('scene1-arc-d'); hide('scene1-val-d');
            hide('scene1-tri-ocd-highlight');
            hide('scene1-arc-cod'); hide('scene1-val-cod');
            hide('scene1-arc-aod'); hide('scene1-val-aod');
            
            hideText('scene1-text-step1');
            hideText('scene1-text-step2');
            hideText('scene1-text-step3');
            hideText('scene1-text-step4');
            
            if (currentStep >= 1) {
                show('scene1-rot-tri');
                setTimeout(() => {
                    document.getElementById('scene1-rot-tri').style.transform = 'rotate(-70deg)';
                }, 50);
                show('scene1-arc-aoc'); show('scene1-val-aoc');
                showText('scene1-text-step1');
            }
            
            if (currentStep >= 2) {
                hide('scene1-rot-tri');
                show('scene1-arc-a'); show('scene1-val-a');
                show('scene1-arc-c'); show('scene1-val-c');
                show('scene1-arc-d'); show('scene1-val-d');
                showText('scene1-text-step2');
            }
            
            if (currentStep >= 3) {
                show('scene1-tri-ocd-highlight');
                show('scene1-arc-cod'); show('scene1-val-cod');
                showText('scene1-text-step3');
            }
            
            if (currentStep >= 4) {
                hide('scene1-tri-ocd-highlight');
                show('scene1-arc-aod'); show('scene1-val-aod');
                showText('scene1-text-step4');
            }
        }
        
        btnPrev.addEventListener('click', () => {
            if (currentStep > 0) {
                currentStep--;
                updateScene();
            }
        });
        
        btnNext.addEventListener('click', () => {
            if (currentStep < maxStep) {
                currentStep++;
                updateScene();
            }
        });
        
        updateScene();
    })();
    </script>
