    <script>
    (function() {
        let currentStep = 0;
        const maxStep = 5;
        
        const btnPrev = document.getElementById('scene2-btn-prev');
        const btnNext = document.getElementById('scene2-btn-next');
        
        function show(el) {
            if (!el) return;
            el.style.display = '';
            setTimeout(() => { el.style.opacity = '1'; }, 10);
        }
        
        function hide(el) {
            if (!el) return;
            el.style.opacity = '0';
            setTimeout(() => { el.style.display = 'none'; }, 300);
        }
        
        function updateScene() {
            btnPrev.disabled = currentStep === 0;
            btnNext.disabled = currentStep === maxStep;
            
            // Step 1: Add N/S, E/W, bold 0 lines
            if (currentStep >= 1) {
                document.getElementById('scene2-lbl-lon1').textContent = '20°W';
                document.getElementById('scene2-lbl-lon3').textContent = '20°E';
                document.getElementById('scene2-lbl-lon4').textContent = '40°E';
                
                document.getElementById('scene2-lbl-lat1').textContent = '66.5° N';
                document.getElementById('scene2-lbl-lat2').textContent = '23.5° N';
                document.getElementById('scene2-lbl-lat4').textContent = '23.5° S';
                
                document.getElementById('scene2-lon-0').style.strokeWidth = '3';
                document.getElementById('scene2-lat-0').style.strokeWidth = '3';
            } else {
                document.getElementById('scene2-lbl-lon1').textContent = '20°';
                document.getElementById('scene2-lbl-lon3').textContent = '20°';
                document.getElementById('scene2-lbl-lon4').textContent = '40°';
                
                document.getElementById('scene2-lbl-lat1').textContent = '66.5° ';
                document.getElementById('scene2-lbl-lat2').textContent = '23.5° ';
                document.getElementById('scene2-lbl-lat4').textContent = '23.5° ';
                
                document.getElementById('scene2-lon-0').style.strokeWidth = '1';
                document.getElementById('scene2-lat-0').style.strokeWidth = '1';
            }
            
            // Step 2: Highlight F and E, draw arrows
            const arrows = document.getElementById('scene2-arrows');
            if (currentStep === 2) {
                document.getElementById('scene2-pt-e').style.fill = '#1565c0';
                document.getElementById('scene2-pt-f').style.fill = '#1565c0';
                show(arrows);
            } else {
                document.getElementById('scene2-pt-e').style.fill = '#000';
                document.getElementById('scene2-pt-f').style.fill = '#000';
                hide(arrows);
            }
            
            // Step 3 & 4: Masks
            const masksHtml = document.getElementById('scene2-masks-html');
            const masksText = document.getElementById('scene2-masks-text');
            
            const maskWest = document.getElementById('scene2-mask-west');
            const maskWestText = document.getElementById('scene2-mask-west-text');
            const maskEast = document.getElementById('scene2-mask-east');
            const maskEastText = document.getElementById('scene2-mask-east-text');
            
            const maskNorth = document.getElementById('scene2-mask-north');
            const maskNorthText = document.getElementById('scene2-mask-north-text');
            const maskTropics = document.getElementById('scene2-mask-tropics');
            const maskTropicsText = document.getElementById('scene2-mask-tropics-text');

            if (currentStep >= 3) {
                show(masksHtml);
                show(masksText);
            } else {
                hide(masksHtml);
                hide(masksText);
            }

            if (currentStep >= 3) {
                document.getElementById('scene2-lon-20w').style.stroke = '#c62828';
                document.getElementById('scene2-lon-20w').style.strokeWidth = '3';
                maskWest.style.display = ''; maskWestText.style.display = '';
                maskEast.style.display = ''; maskEastText.style.display = '';
            } else {
                document.getElementById('scene2-lon-20w').style.stroke = '#000';
                document.getElementById('scene2-lon-20w').style.strokeWidth = '1';
                maskWest.style.display = 'none'; maskWestText.style.display = 'none';
                maskEast.style.display = 'none'; maskEastText.style.display = 'none';
            }
            
            if (currentStep >= 4) {
                document.getElementById('scene2-lat-235n').style.stroke = '#2e7d32';
                document.getElementById('scene2-lat-235n').style.strokeWidth = '3';
                maskNorth.style.display = ''; maskNorthText.style.display = '';
                maskTropics.style.display = ''; maskTropicsText.style.display = '';
            } else {
                document.getElementById('scene2-lat-235n').style.stroke = '#000';
                document.getElementById('scene2-lat-235n').style.strokeWidth = '1';
                maskNorth.style.display = 'none'; maskNorthText.style.display = 'none';
                maskTropics.style.display = 'none'; maskTropicsText.style.display = 'none';
            }
            
            // Step 5: Lock intersection
            const ptYi = document.getElementById('scene2-pt-yi');
            if (currentStep === 5) {
                ptYi.style.fill = '#c62828';
                ptYi.classList.add('scene2-pulse-anim');
            } else {
                ptYi.style.fill = '#000';
                ptYi.classList.remove('scene2-pulse-anim');
            }
        }
        
        // Add pointer-events: auto to buttons so they can be clicked
        // since SVG has pointer-events: none
        document.getElementById('scene2-controls').style.pointerEvents = 'auto';
        
        btnPrev.addEventListener('click', () => {
            if (currentStep > 0) {
                currentStep--;
                updateScene();
            }
        });
        
        btnNext.addEventListener('click', () => {
            if (currentStep < maxStep) {
                currentStep++;
                updateScene();
            }
        });
        
        // Init
        updateScene();
    })();
    </script>
