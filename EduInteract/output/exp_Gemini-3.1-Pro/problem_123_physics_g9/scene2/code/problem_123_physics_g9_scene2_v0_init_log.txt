    <script>
    (function() {
        const slider = document.getElementById('scene2-slider');
        
        // Elements
        const waterJiaLeft = document.getElementById('scene2-water-jia-left');
        const waterJiaRight = document.getElementById('scene2-water-jia-right');
        const waterLineJiaLeft = document.getElementById('scene2-water-line-jia-left');
        const waterLineJiaRight = document.getElementById('scene2-water-line-jia-right');
        
        const waterYiLeft = document.getElementById('scene2-water-yi-left');
        const waterYiRight = document.getElementById('scene2-water-yi-right');
        const waterLineYiLeft = document.getElementById('scene2-water-line-yi-left');
        const waterLineYiRight = document.getElementById('scene2-water-line-yi-right');
        
        const waterBing = document.getElementById('scene2-water-bing');
        const waterLineBing = document.getElementById('scene2-water-line-bing');
        
        const waterCenter = document.getElementById('scene2-water-center');
        const waterLineCenter = document.getElementById('scene2-water-line-center');
        
        const hlYiLeft = document.getElementById('scene2-hl-yi-left');
        const hlYiRight = document.getElementById('scene2-hl-yi-right');
        const hlBing = document.getElementById('scene2-hl-bing');
        
        const valve = document.getElementById('scene2-valve');
        
        const labelYi = document.getElementById('scene2-label-yi');
        const labelBing = document.getElementById('scene2-label-bing');
        
        const hLineTop = document.getElementById('scene2-h-line-top');
        const hLineBottom = document.getElementById('scene2-h-line-bottom');
        const hArrow = document.getElementById('scene2-h-arrow');
        const hLabel = document.getElementById('scene2-h-label');

        function updateScene() {
            const t = parseFloat(slider.value);
            
            // Calculate y positions
            const yJia = 80 + 20 * t;
            const yYi = 210 + 30 * t;
            const yBing = 360 - 20 * t;
            const yCenter = 70 - 40 * t;
            
            // Update 甲
            waterJiaLeft.setAttribute('y', yJia);
            waterJiaLeft.setAttribute('height', 140 - yJia);
            waterJiaRight.setAttribute('y', yJia);
            waterJiaRight.setAttribute('height', 140 - yJia);
            waterLineJiaLeft.setAttribute('y1', yJia);
            waterLineJiaLeft.setAttribute('y2', yJia);
            waterLineJiaRight.setAttribute('y1', yJia);
            waterLineJiaRight.setAttribute('y2', yJia);
            
            // Update 乙
            waterYiLeft.setAttribute('y', yYi);
            waterYiLeft.setAttribute('height', 270 - yYi);
            waterYiRight.setAttribute('y', yYi);
            waterYiRight.setAttribute('height', 270 - yYi);
            waterLineYiLeft.setAttribute('y1', yYi);
            waterLineYiLeft.setAttribute('y2', yYi);
            waterLineYiRight.setAttribute('y1', yYi);
            waterLineYiRight.setAttribute('y2', yYi);
            
            // Update 丙
            waterBing.setAttribute('y', yBing);
            waterBing.setAttribute('height', 400 - yBing);
            waterLineBing.setAttribute('y1', yBing);
            waterLineBing.setAttribute('y2', yBing);
            
            // Update Center
            waterCenter.setAttribute('y', yCenter);
            waterCenter.setAttribute('height', 260 - yCenter);
            waterLineCenter.setAttribute('y1', yCenter);
            waterLineCenter.setAttribute('y2', yCenter);
            
            // Update Highlights
            hlYiLeft.setAttribute('height', yYi - 210);
            hlYiRight.setAttribute('height', yYi - 210);
            
            hlBing.setAttribute('y', yBing);
            hlBing.setAttribute('height', 360 - yBing);
            
            // Update Valve
            if (t > 0) {
                valve.setAttribute('fill', '#ffffff');
            } else {
                valve.setAttribute('fill', '#000000');
            }
            
            // Update Labels opacity
            if (t > 0.1) {
                labelYi.setAttribute('opacity', Math.min(1, (t - 0.1) * 2));
                labelBing.setAttribute('opacity', Math.min(1, (t - 0.1) * 2));
            } else {
                labelYi.setAttribute('opacity', 0);
                labelBing.setAttribute('opacity', 0);
            }
            
            // Update h marker
            hLineTop.setAttribute('y1', yJia);
            hLineTop.setAttribute('y2', yJia);
            hLineBottom.setAttribute('y1', yBing);
            hLineBottom.setAttribute('y2', yBing);
            hArrow.setAttribute('y1', yJia);
            hArrow.setAttribute('y2', yBing);
            hLabel.setAttribute('y', (yJia + yBing) / 2 + 5);
        }

        slider.addEventListener('input', updateScene);
        updateScene();
    })();
    </script>
