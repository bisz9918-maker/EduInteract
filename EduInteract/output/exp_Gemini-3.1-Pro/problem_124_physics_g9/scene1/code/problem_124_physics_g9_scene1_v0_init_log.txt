    <script>
    (function() {
        let currentStep = 0;
        const maxStep = 4;
        
        const btnPrev = document.getElementById('scene1-btn-prev');
        const btnNext = document.getElementById('scene1-btn-next');
        
        const switchBlade = document.getElementById('scene1-switch-blade');
        const batteryLabels = document.getElementById('scene1-battery-labels');
        const currentArrows = document.getElementById('scene1-current-arrows');
        const rightHand = document.getElementById('scene1-right-hand');
        const poleSJia = document.getElementById('scene1-pole-s-jia');
        const poleNJia = document.getElementById('scene1-pole-n-jia');
        const poleNYi = document.getElementById('scene1-pole-n-yi');
        const poleSYi = document.getElementById('scene1-pole-s-yi');
        const magneticLines = document.getElementById('scene1-magnetic-lines');
        const hintText = document.getElementById('scene1-hint-text');
        
        function updateState() {
            // 按钮状态
            btnPrev.disabled = currentStep === 0;
            btnNext.disabled = currentStep === maxStep;
            
            // 步骤 0: 初始状态
            if (currentStep < 1) {
                switchBlade.setAttribute('x2', '190');
                switchBlade.setAttribute('y2', '335');
                batteryLabels.style.opacity = '0';
                currentArrows.style.opacity = '0';
            }
            
            // 步骤 1: 标出电流
            if (currentStep >= 1) {
                switchBlade.setAttribute('x2', '200');
                switchBlade.setAttribute('y2', '360');
                batteryLabels.style.opacity = '1';
                currentArrows.style.opacity = '1';
            }
            
            // 步骤 2: 安培定则
            if (currentStep < 2) {
                rightHand.style.opacity = '0';
                poleSJia.style.opacity = '0';
                poleNJia.style.opacity = '0';
            } else {
                rightHand.style.opacity = '1';
                poleSJia.style.opacity = '1';
                poleNJia.style.opacity = '1';
            }
            
            // 步骤 3: 磁感线分析
            if (currentStep < 3) {
                magneticLines.setAttribute('stroke', '#888888');
                magneticLines.setAttribute('stroke-width', '1.5');
                hintText.style.opacity = '0';
            } else {
                magneticLines.setAttribute('stroke', '#2e7d32');
                magneticLines.setAttribute('stroke-width', '3');
                hintText.style.opacity = '1';
            }
            
            // 步骤 4: 得出结论
            if (currentStep < 4) {
                poleNYi.style.opacity = '0';
                poleSYi.style.opacity = '0';
                poleNJia.style.transform = 'scale(1)';
                poleNYi.style.transform = 'scale(1)';
            } else {
                poleNYi.style.opacity = '1';
                poleSYi.style.opacity = '1';
                // 强调同名磁极
                poleNJia.style.transform = 'scale(1)';
                poleNJia.style.transformOrigin = '290px 290px';
                poleNYi.style.transform = 'scale(1)';
                poleNYi.style.transformOrigin = '460px 290px';
            }
        }
        
        btnPrev.addEventListener('click', () => {
            if (currentStep > 0) {
                currentStep--;
                updateState();
            }
        });
        
        btnNext.addEventListener('click', () => {
            if (currentStep < maxStep) {
                currentStep++;
                updateState();
            }
        });
        
        // 初始化
        updateState();
    })();
    </script>
