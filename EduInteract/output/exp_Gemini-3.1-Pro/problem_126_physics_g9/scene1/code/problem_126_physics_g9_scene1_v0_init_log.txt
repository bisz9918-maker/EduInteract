    <script>
    (function() {
        let currentStep = 0;
        const maxStep = 4;
        
        const btnPrev = document.getElementById('scene1-btn-prev');
        const btnNext = document.getElementById('scene1-btn-next');
        
        let currentS = 0;
        let targetS = 0;
        let animationFrame;
        
        const movableGroup = document.getElementById('scene1-movable-group');
        const rope1 = document.getElementById('scene1-rope1');
        const rope2 = document.getElementById('scene1-rope2');
        const rope3 = document.getElementById('scene1-rope3');
        const ropeFree = document.getElementById('scene1-rope-free');
        const freeEndGroup = document.getElementById('scene1-free-end-group');
        
        const step1Vectors = document.getElementById('scene1-step1-vectors');
        const step1S = document.getElementById('scene1-step1-s');
        const step2Vectors = document.getElementById('scene1-step2-vectors');
        const step23S = document.getElementById('scene1-step2-3s');
        const step2N = document.getElementById('scene1-step2-n');
        
        const fVector = document.getElementById('scene1-f-vector');
        const fText = document.getElementById('scene1-f-text');
        
        function updatePositions() {
            // Movable group translates by currentS
            movableGroup.setAttribute('transform', `translate(${currentS}, 0)`);
            
            // Ropes update
            rope1.setAttribute('x1', 400 + currentS);
            rope2.setAttribute('x1', 400 + currentS);
            rope3.setAttribute('x1', 400 + currentS);
            
            // Free end rope
            const freeEndX = 500 - currentS * 2.9;
            const freeEndY = 210 - currentS * 0.72;
            ropeFree.setAttribute('x2', freeEndX);
            ropeFree.setAttribute('y2', freeEndY);
            
            // Free end group translates
            const dx = freeEndX - 500;
            const dy = freeEndY - 210;
            freeEndGroup.setAttribute('transform', `translate(${dx}, ${dy})`);
        }
        
        function animate() {
            const diff = targetS - currentS;
            if (Math.abs(diff) > 0.1) {
                currentS += diff * 0.1;
                updatePositions();
                animationFrame = requestAnimationFrame(animate);
            } else {
                currentS = targetS;
                updatePositions();
            }
        }
        
        function show(el) {
            el.style.display = 'block';
            // small delay to allow display:block to apply before opacity transition
            setTimeout(() => el.style.opacity = '1', 10);
        }
        
        function hide(el) {
            el.style.opacity = '0';
            setTimeout(() => el.style.display = 'none', 300);
        }
        
        function updateState() {
            btnPrev.disabled = currentStep === 0;
            btnNext.disabled = currentStep === maxStep;
            
            // Step 0 -> 1: Move
            if (currentStep >= 1) {
                targetS = 50;
            } else {
                targetS = 0;
            }
            cancelAnimationFrame(animationFrame);
            animate();
            
            // Text areas
            for (let i = 0; i <= maxStep; i++) {
                const textEl = document.getElementById(`scene1-text-step${i}`);
                if (i === currentStep) {
                    show(textEl);
                } else {
                    hide(textEl);
                }
            }
            
            // Step 1 elements
            if (currentStep >= 1) {
                show(step1Vectors);
                show(step1S);
            } else {
                hide(step1Vectors);
                hide(step1S);
            }
            
            // Step 2 elements
            if (currentStep >= 2) {
                show(step2Vectors);
                show(step23S);
                show(step2N);
                rope1.setAttribute('stroke', '#2e7d32');
                rope2.setAttribute('stroke', '#2e7d32');
                rope3.setAttribute('stroke', '#2e7d32');
                rope1.setAttribute('stroke-width', '3');
                rope2.setAttribute('stroke-width', '3');
                rope3.setAttribute('stroke-width', '3');
            } else {
                hide(step2Vectors);
                hide(step23S);
                hide(step2N);
                rope1.setAttribute('stroke', '#000000');
                rope2.setAttribute('stroke', '#000000');
                rope3.setAttribute('stroke', '#000000');
                rope1.setAttribute('stroke-width', '2');
                rope2.setAttribute('stroke-width', '2');
                rope3.setAttribute('stroke-width', '2');
            }
            
            // Step 3 elements
            if (currentStep >= 3) {
                fVector.setAttribute('stroke-width', '4');
                fText.setAttribute('font-size', '20');
                fText.setAttribute('font-weight', 'bold');
            } else {
                fVector.setAttribute('stroke-width', '2');
                fText.setAttribute('font-size', '16');
                fText.setAttribute('font-weight', 'normal');
            }
        }
        
        btnPrev.addEventListener('click', () => {
            if (currentStep > 0) {
                currentStep--;
                updateState();
            }
        });
        
        btnNext.addEventListener('click', () => {
            if (currentStep < maxStep) {
                currentStep++;
                updateState();
            }
        });
        
        // Initial setup
        updatePositions();
    })();
    </script>
