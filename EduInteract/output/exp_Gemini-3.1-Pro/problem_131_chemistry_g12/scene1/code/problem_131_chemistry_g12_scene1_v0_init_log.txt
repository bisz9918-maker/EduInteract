    <script>
    (function() {
        let currentStep = 0;
        const maxStep = 4;
        
        const btnPrev = document.getElementById('scene1-btn-prev');
        const btnNext = document.getElementById('scene1-btn-next');
        
        const stepGroups = [
            null, // step 0 is base
            document.getElementById('scene1-step1-group'),
            document.getElementById('scene1-step2-group'),
            document.getElementById('scene1-step3-group'),
            document.getElementById('scene1-step4-group')
        ];

        function updateStep(step) {
            // Update buttons
            btnPrev.disabled = step === 0;
            btnNext.disabled = step === maxStep;
            
            // Update visibility
            for (let i = 1; i <= maxStep; i++) {
                if (i <= step) {
                    stepGroups[i].classList.remove('hidden');
                } else {
                    stepGroups[i].classList.add('hidden');
                }
            }
        }

        btnPrev.addEventListener('click', () => {
            if (currentStep > 0) {
                currentStep--;
                updateStep(currentStep);
            }
        });

        btnNext.addEventListener('click', () => {
            if (currentStep < maxStep) {
                currentStep++;
                updateStep(currentStep);
            }
        });

        // Initialize
        updateStep(0);
    })();
    </script>
