    <script>
    (function() {
        let currentStep = 0;
        const maxStep = 4;
        const btnPrev = document.getElementById('scene1-btn-prev');
        const btnNext = document.getElementById('scene1-btn-next');

        function updateScene() {
            for (let i = 1; i <= maxStep; i++) {
                const group = document.getElementById(`scene1-step${i}-group`);
                if (i <= currentStep) {
                    group.style.display = 'block';
                    setTimeout(() => {
                        group.classList.remove('hidden');
                        group.classList.add('visible');
                    }, 10);
                } else {
                    group.classList.remove('visible');
                    group.classList.add('hidden');
                    setTimeout(() => {
                        if (currentStep < i) group.style.display = 'none';
                    }, 300);
                }
            }
            
            const formula = document.getElementById('scene1-formula');
            if (formula) {
                if (currentStep === 3) {
                    formula.style.display = 'block';
                    setTimeout(() => {
                        formula.classList.remove('hidden');
                        formula.classList.add('visible');
                    }, 10);
                } else {
                    formula.classList.remove('visible');
                    formula.classList.add('hidden');
                    setTimeout(() => {
                        if (currentStep !== 3) formula.style.display = 'none';
                    }, 300);
                }
            }

            btnPrev.disabled = currentStep === 0;
            btnNext.disabled = currentStep === maxStep;
        }

        btnPrev.addEventListener('click', () => {
            if (currentStep > 0) {
                currentStep--;
                updateScene();
            }
        });

        btnNext.addEventListener('click', () => {
            if (currentStep < maxStep) {
                currentStep++;
                updateScene();
            }
        });

        updateScene();
    })();
    </script>
