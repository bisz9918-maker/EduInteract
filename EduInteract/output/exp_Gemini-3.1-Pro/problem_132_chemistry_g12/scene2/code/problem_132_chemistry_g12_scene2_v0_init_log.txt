    <script>
    (function() {
        let currentStep = 0;
        const maxStep = 4;
        
        const btnPrev = document.getElementById('scene2-btn-prev');
        const btnNext = document.getElementById('scene2-btn-next');
        
        const steps = [
            { svg: null, html: null },
            { svg: document.getElementById('scene2-step1-svg'), html: document.getElementById('scene2-step1-html') },
            { svg: document.getElementById('scene2-step2-svg'), html: document.getElementById('scene2-step2-html') },
            { svg: document.getElementById('scene2-step3-svg'), html: document.getElementById('scene2-step3-html') },
            { svg: null, html: document.getElementById('scene2-step4-html') }
        ];
        
        function show(el) {
            if (!el) return;
            el.style.display = 'block';
            void el.offsetWidth;
            el.style.opacity = '1';
        }
        
        function hide(el) {
            if (!el) return;
            el.style.opacity = '0';
            setTimeout(() => {
                if (el.style.opacity === '0') {
                    el.style.display = 'none';
                }
            }, 500);
        }
        
        function updateState() {
            btnPrev.disabled = currentStep === 0;
            btnNext.disabled = currentStep === maxStep;
            
            for (let i = 1; i <= maxStep; i++) {
                if (currentStep >= i) {
                    show(steps[i].svg);
                    show(steps[i].html);
                } else {
                    hide(steps[i].svg);
                    hide(steps[i].html);
                }
            }
        }
        
        btnPrev.addEventListener('click', () => {
            if (currentStep > 0) {
                currentStep--;
                updateState();
            }
        });
        
        btnNext.addEventListener('click', () => {
            if (currentStep < maxStep) {
                currentStep++;
                updateState();
            }
        });
        
        updateState();
    })();
    </script>
