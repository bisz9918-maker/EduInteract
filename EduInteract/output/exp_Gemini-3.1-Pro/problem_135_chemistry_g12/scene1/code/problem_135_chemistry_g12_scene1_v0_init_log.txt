    <script>
    (function() {
        const slider = document.getElementById('scene1-slider');
        const glow = document.getElementById('scene1-glow');
        const dot = document.getElementById('scene1-dot');
        const curveBg = document.getElementById('scene1-curve-bg');
        const particlesContainer = document.getElementById('scene1-particles-container');
        
        const cx = 560, cy = 135, r = 80;
        
        // 预生成曲线路径
        let pathD = "M 480 380 ";
        for (let v = 1; v <= 100; v++) {
            let x = 480 + v * 2.4;
            let y;
            if (v <= 50) {
                y = 380 - 110 * Math.pow(v/50, 0.8);
            } else {
                y = 270 + 50 * Math.pow((v-50)/50, 1.2);
            }
            pathD += `L ${x} ${y} `;
        }
        curveBg.setAttribute('d', pathD);

        // 粒子数据结构
        const particles = {
            ch3cooh: [],
            ch3coo: [],
            nh4: [],
            nh3h2o: []
        };

        // 随机生成粒子位置
        function generateParticles(type, count, className) {
            for (let i = 0; i < count; i++) {
                // 在圆内随机分布
                const angle = Math.random() * Math.PI * 2;
                const radius = Math.random() * (r - 10); // 留出边缘
                const x = cx + radius * Math.cos(angle);
                const y = cy + radius * Math.sin(angle);
                
                const circle = document.createElementNS("http://www.w3.org/2000/svg", "circle");
                circle.setAttribute("cx", x);
                circle.setAttribute("cy", y);
                circle.setAttribute("r", 5);
                circle.setAttribute("class", className);
                
                // 保存初始相对坐标，用于后续稀释动画
                circle.dataset.dx = radius * Math.cos(angle);
                circle.dataset.dy = radius * Math.sin(angle);
                
                particlesContainer.appendChild(circle);
                particles[type].push(circle);
            }
        }

        // 初始化粒子
        generateParticles('ch3cooh', 15, 'particle-ch3cooh');
        generateParticles('ch3coo', 15, 'particle-ch3coo');
        generateParticles('nh4', 15, 'particle-nh4');
        generateParticles('nh3h2o', 8, 'particle-nh3h2o');

        function updateScene() {
            const v = parseInt(slider.value);
            
            // 1. 更新曲线动点
            const dotX = 480 + v * 2.4;
            let dotY;
            if (v <= 50) {
                dotY = 380 - 110 * Math.pow(v/50, 0.8);
            } else {
                dotY = 270 + 50 * Math.pow((v-50)/50, 1.2);
            }
            dot.setAttribute('cx', dotX);
            dot.setAttribute('cy', dotY);

            // 2. 更新灯泡光晕
            let opacity;
            if (v <= 50) {
                opacity = 0.2 + 0.6 * (v / 50); // 0.2 -> 0.8
            } else {
                opacity = 0.8 - 0.3 * ((v - 50) / 50); // 0.8 -> 0.5
            }
            glow.setAttribute('opacity', opacity);

            // 3. 更新微观粒子
            // 稀释系数 (v > 50 时生效)
            const scale = v > 50 ? 1 + ((v - 50) / 100) * 0.5 : 1;

            // CH3COOH: 0-50 逐渐消失
            const ch3coohOpacity = v <= 50 ? 1 - (v / 50) : 0;
            particles.ch3cooh.forEach(p => {
                p.style.opacity = ch3coohOpacity;
                p.style.display = ch3coohOpacity > 0 ? 'block' : 'none';
                // 位置不随稀释改变，因为已经消失了
            });

            // CH3COO- 和 NH4+: 0-50 逐渐出现，50-100 稀释扩散
            const ionOpacity = v <= 50 ? (v / 50) : 1;
            particles.ch3coo.forEach(p => {
                p.style.opacity = ionOpacity;
                p.style.display = ionOpacity > 0 ? 'block' : 'none';
                p.setAttribute('cx', cx + parseFloat(p.dataset.dx) * scale);
                p.setAttribute('cy', cy + parseFloat(p.dataset.dy) * scale);
            });
            particles.nh4.forEach(p => {
                p.style.opacity = ionOpacity;
                p.style.display = ionOpacity > 0 ? 'block' : 'none';
                p.setAttribute('cx', cx + parseFloat(p.dataset.dx) * scale);
                p.setAttribute('cy', cy + parseFloat(p.dataset.dy) * scale);
            });

            // NH3·H2O: 50-100 逐渐出现
            const nh3h2oOpacity = v > 50 ? (v - 50) / 50 : 0;
            particles.nh3h2o.forEach(p => {
                p.style.opacity = nh3h2oOpacity;
                p.style.display = nh3h2oOpacity > 0 ? 'block' : 'none';
                p.setAttribute('cx', cx + parseFloat(p.dataset.dx) * scale);
                p.setAttribute('cy', cy + parseFloat(p.dataset.dy) * scale);
            });
        }

        slider.addEventListener('input', updateScene);
        
        // 初始化状态
        updateScene();
    })();
    </script>
