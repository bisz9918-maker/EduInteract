    <script>
    (function() {
        let currentStep = 0;
        const maxStep = 4;
        
        const btnPrev = document.getElementById('scene1-btn-prev');
        const btnNext = document.getElementById('scene1-btn-next');
        
        const unreactMols = document.querySelectorAll('.scene1-unreact');
        const step1Hl = document.getElementById('scene1-step1-hl');
        const step3Hl = document.getElementById('scene1-step3-hl');
        const step4Eq = document.getElementById('scene1-step4-eq');

        function show(el) {
            el.style.display = '';
            // trigger reflow
            void el.offsetWidth;
            el.style.opacity = '1';
        }

        function hide(el) {
            el.style.opacity = '0';
            setTimeout(() => {
                if (el.style.opacity === '0') {
                    el.style.display = 'none';
                }
            }, 500);
        }

        function updateStep() {
            btnPrev.disabled = currentStep === 0;
            btnNext.disabled = currentStep === maxStep;

            // Step 1: Highlight unreacted
            if (currentStep === 1) {
                show(step1Hl);
            } else {
                hide(step1Hl);
            }

            // Step 2: Fade unreacted
            if (currentStep >= 2) {
                unreactMols.forEach(el => el.style.opacity = '0.2');
            } else {
                unreactMols.forEach(el => el.style.opacity = '1');
            }

            // Step 3: Highlight products
            if (currentStep >= 3) {
                show(step3Hl);
            } else {
                hide(step3Hl);
            }

            // Step 4: Show equation
            if (currentStep >= 4) {
                show(step4Eq);
            } else {
                hide(step4Eq);
            }
        }

        btnPrev.addEventListener('click', () => {
            if (currentStep > 0) {
                currentStep--;
                updateStep();
            }
        });

        btnNext.addEventListener('click', () => {
            if (currentStep < maxStep) {
                currentStep++;
                updateStep();
            }
        });

        updateStep();
    })();
    </script>
