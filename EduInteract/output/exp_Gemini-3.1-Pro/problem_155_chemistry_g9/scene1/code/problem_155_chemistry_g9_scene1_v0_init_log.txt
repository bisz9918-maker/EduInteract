    <script>
    (function() {
        // Generate particles
        function createParticles(groupId, color, count, minX, maxX, minY, maxY) {
            const group = document.getElementById(groupId);
            for (let i = 0; i < count; i++) {
                const x = minX + Math.random() * (maxX - minX);
                const y = minY + Math.random() * (maxY - minY);
                const circle = document.createElementNS("http://www.w3.org/2000/svg", "circle");
                circle.setAttribute("cx", x);
                circle.setAttribute("cy", y);
                circle.setAttribute("r", 2.5);
                circle.setAttribute("fill", color);
                group.appendChild(circle);
            }
        }
        
        createParticles("scene1-particles-co1", "#1565c0", 80, 170, 330, 270, 295);
        createParticles("scene1-particles-impurity", "#888888", 40, 170, 330, 270, 295);
        createParticles("scene1-particles-co2", "#1565c0", 80, 470, 630, 270, 295);

        let currentStep = 0;
        const maxStep = 3;
        
        const btnPrev = document.getElementById("scene1-btn-prev");
        const btnNext = document.getElementById("scene1-btn-next");
        
        function updateState() {
            btnPrev.disabled = currentStep === 0;
            btnNext.disabled = currentStep === maxStep;
            
            const setOpacity = (id, op) => document.getElementById(id).style.opacity = op;
            
            if (currentStep === 0) {
                setOpacity("scene1-label-900", 0);
                setOpacity("scene1-particles-co1", 1);
                setOpacity("scene1-flow-mid", 0);
                setOpacity("scene1-label-gas-mid", 0);
                
                setOpacity("scene1-label-600", 0);
                setOpacity("scene1-particles-co2", 0);
                setOpacity("scene1-label-pure-co", 0);
                setOpacity("scene1-flow-right", 0);
                
                setOpacity("scene1-label-solid-x", 1);
                setOpacity("scene1-label-impurity", 0);
                setOpacity("scene1-label-gas-right", 0);
                
                document.getElementById("scene1-flow-left").style.animationPlayState = "paused";
            } else if (currentStep === 1) {
                setOpacity("scene1-label-900", 1);
                setOpacity("scene1-particles-co1", 0.3);
                setOpacity("scene1-flow-mid", 1);
                setOpacity("scene1-label-gas-mid", 1);
                
                setOpacity("scene1-label-600", 0);
                setOpacity("scene1-particles-co2", 0);
                setOpacity("scene1-label-pure-co", 0);
                setOpacity("scene1-flow-right", 0);
                
                setOpacity("scene1-label-solid-x", 1);
                setOpacity("scene1-label-impurity", 0);
                setOpacity("scene1-label-gas-right", 0);
                
                document.getElementById("scene1-flow-left").style.animationPlayState = "running";
                document.getElementById("scene1-flow-mid").style.animationPlayState = "running";
            } else if (currentStep === 2) {
                setOpacity("scene1-label-900", 1);
                setOpacity("scene1-particles-co1", 0.3);
                setOpacity("scene1-flow-mid", 1);
                setOpacity("scene1-label-gas-mid", 1);
                
                setOpacity("scene1-label-600", 1);
                setOpacity("scene1-particles-co2", 1);
                setOpacity("scene1-label-pure-co", 1);
                setOpacity("scene1-flow-right", 1);
                
                setOpacity("scene1-label-solid-x", 1);
                setOpacity("scene1-label-impurity", 0);
                setOpacity("scene1-label-gas-right", 0);
                
                document.getElementById("scene1-flow-left").style.animationPlayState = "running";
                document.getElementById("scene1-flow-mid").style.animationPlayState = "running";
                document.getElementById("scene1-flow-right").style.animationPlayState = "running";
            } else if (currentStep === 3) {
                setOpacity("scene1-label-900", 1);
                setOpacity("scene1-particles-co1", 0);
                setOpacity("scene1-flow-mid", 1);
                setOpacity("scene1-label-gas-mid", 1);
                
                setOpacity("scene1-label-600", 1);
                setOpacity("scene1-particles-co2", 1);
                setOpacity("scene1-label-pure-co", 1);
                setOpacity("scene1-flow-right", 1);
                
                setOpacity("scene1-label-solid-x", 0);
                setOpacity("scene1-label-impurity", 1);
                setOpacity("scene1-label-gas-right", 1);
            }
        }
        
        btnPrev.addEventListener("click", () => {
            if (currentStep > 0) {
                currentStep--;
                updateState();
            }
        });
        
        btnNext.addEventListener("click", () => {
            if (currentStep < maxStep) {
                currentStep++;
                updateState();
            }
        });
        
        // Initial state
        document.getElementById("scene1-flow-left").style.animationPlayState = "paused";
        document.getElementById("scene1-flow-mid").style.animationPlayState = "paused";
        document.getElementById("scene1-flow-right").style.animationPlayState = "paused";
        updateState();
    })();
    </script>
