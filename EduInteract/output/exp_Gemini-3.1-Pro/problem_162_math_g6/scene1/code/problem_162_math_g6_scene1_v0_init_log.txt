<!DOCTYPE html>
<html lang="zh">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <script src="https://d3js.org/d3.v7.min.js"></script>
    <style>
        body { margin: 0; padding: 0; background: #ffffff; font-family: sans-serif; }
        #scene1-container { position: relative; width: 800px; height: 500px; background: #ffffff; margin: 0 auto; overflow: hidden; }
        #scene1-controls { position: absolute; bottom: 20px; left: 0; right: 0; display: flex; justify-content: center; gap: 20px; z-index: 10; }
        button { padding: 8px 16px; font-size: 16px; cursor: pointer; border: 1px solid #ccc; background: #f9f9f9; border-radius: 4px; }
        button:disabled { opacity: 0.5; cursor: not-allowed; }
        button:hover:not(:disabled) { background: #e9e9e9; }
    </style>
</head>
<body>
    <div id="scene1-container">
        <svg id="scene1-svg" width="800" height="500"></svg>
        <div id="scene1-controls">
            <button id="scene1-btn-prev" disabled>◀ 上一步</button>
            <button id="scene1-btn-next">下一步 ▶</button>
        </div>
    </div>
    <script>
    (function() {
        const svg = d3.select("#scene1-svg");
        
        // --- 坐标与数据 ---
        // 图形A
        const rectA = { x: 95, y: 180, w: 150, h: 90 };
        // 图形B
        const rectB = { x: 325, y: 180, w: 150, h: 90 };
        // 图形C
        const rectC = { x: 555, y: 180, w: 150, h: 90 };

        // 组
        const gDashed = svg.append("g").attr("id", "scene1-dashed-group");
        const gShapes = svg.append("g").attr("id", "scene1-shapes-group");
        const gLines = svg.append("g").attr("id", "scene1-lines-group");
        const gLabels = svg.append("g").attr("id", "scene1-labels-group");

        // --- 绘制静态图形 ---
        // 图形A
        const shapeA = gShapes.append("rect")
            .attr("id", "scene1-shape-a")
            .attr("x", rectA.x).attr("y", rectA.y)
            .attr("width", rectA.w).attr("height", rectA.h)
            .attr("fill", "rgba(0,0,0,0.1)")
            .attr("stroke", "#000000")
            .attr("stroke-width", 2);

        // 图形B
        const ptsB = [
            [325,270], [475,270], [475,252], [445,252], [445,234], [415,234],
            [415,216], [385,216], [385,198], [355,198], [355,180], [325,180]
        ];
        const shapeB = gShapes.append("polygon")
            .attr("id", "scene1-shape-b")
            .attr("points", ptsB.map(p => p.join(",")).join(" "))
            .attr("fill", "rgba(0,0,0,0.1)")
            .attr("stroke", "#000000")
            .attr("stroke-width", 2);

        // 图形C
        const ptsC = [
            [555,270], [705,270], [705,180], [675,180], [675,225], [585,225], [585,180], [555,180]
        ];
        const shapeC = gShapes.append("polygon")
            .attr("id", "scene1-shape-c")
            .attr("points", ptsC.map(p => p.join(",")).join(" "))
            .attr("fill", "rgba(0,0,0,0.1)")
            .attr("stroke", "#000000")
            .attr("stroke-width", 2);

        // --- 绘制虚线框 (初始隐藏) ---
        const dashA = gDashed.append("rect")
            .attr("x", rectA.x).attr("y", rectA.y).attr("width", rectA.w).attr("height", rectA.h)
            .attr("fill", "none").attr("stroke", "#888888").attr("stroke-width", 2).attr("stroke-dasharray", "5,5")
            .style("opacity", 0);
        const dashB = gDashed.append("rect")
            .attr("x", rectB.x).attr("y", rectB.y).attr("width", rectB.w).attr("height", rectB.h)
            .attr("fill", "none").attr("stroke", "#888888").attr("stroke-width", 2).attr("stroke-dasharray", "5,5")
            .style("opacity", 0);
        const dashC = gDashed.append("rect")
            .attr("x", rectC.x).attr("y", rectC.y).attr("width", rectC.w).attr("height", rectC.h)
            .attr("fill", "none").attr("stroke", "#888888").attr("stroke-width", 2).attr("stroke-dasharray", "5,5")
            .style("opacity", 0);

        // --- 绘制动态线段 (初始隐藏) ---
        // B水平段 (4段)
        const bHorizLines = [];
        for(let i=0; i<4; i++) {
            let y = 252 - i*18;
            let x1 = 445 - i*30;
            let x2 = 475 - i*30;
            bHorizLines.push(
                gLines.append("line")
                    .attr("x1", x1).attr("y1", y).attr("x2", x2).attr("y2", y)
                    .attr("stroke", "#1565c0").attr("stroke-width", 4)
                    .style("opacity", 0)
            );
        }
        // B竖直段 (4段)
        const bVertLines = [];
        for(let i=0; i<4; i++) {
            let x = 445 - i*30;
            let y1 = 234 - i*18;
            let y2 = 252 - i*18;
            bVertLines.push(
                gLines.append("line")
                    .attr("x1", x).attr("y1", y1).attr("x2", x).attr("y2", y2)
                    .attr("stroke", "#2e7d32").attr("stroke-width", 4)
                    .style("opacity", 0)
            );
        }
        // C水平段
        const cHorizLine = gLines.append("line")
            .attr("x1", 585).attr("y1", 225).attr("x2", 675).attr("y2", 225)
            .attr("stroke", "#1565c0").attr("stroke-width", 4)
            .style("opacity", 0);
        
        // C多余竖直段
        const cVertLine1 = gLines.append("line")
            .attr("x1", 585).attr("y1", 180).attr("x2", 585).attr("y2", 225)
            .attr("stroke", "#c62828").attr("stroke-width", 4)
            .style("opacity", 0);
        const cVertLine2 = gLines.append("line")
            .attr("x1", 675).attr("y1", 180).attr("x2", 675).attr("y2", 225)
            .attr("stroke", "#c62828").attr("stroke-width", 4)
            .style("opacity", 0);

        // --- 绘制标签 ---
        const labels = [
            { text: "A.", x: 65, y: 225, anchor: "end" },
            { text: "10", x: 170, y: 295, anchor: "middle" },
            { text: "6", x: 250, y: 230, anchor: "start" },
            { text: "B.", x: 290, y: 225, anchor: "end" },
            { text: "10", x: 400, y: 295, anchor: "middle" },
            { text: "6", x: 315, y: 230, anchor: "end" },
            { text: "C.", x: 525, y: 225, anchor: "end" },
            { text: "10", x: 630, y: 295, anchor: "middle" },
            { text: "6", x: 715, y: 230, anchor: "start" }
        ];
        labels.forEach(l => {
            gLabels.append("text")
                .attr("x", l.x).attr("y", l.y)
                .attr("text-anchor", l.anchor)
                .attr("font-size", "16px")
                .attr("font-family", "sans-serif")
                .attr("fill", "#000000")
                .text(l.text);
        });

        // --- 交互逻辑 ---
        let currentStep = 0;
        const maxStep = 4;
        const btnPrev = document.getElementById("scene1-btn-prev");
        const btnNext = document.getElementById("scene1-btn-next");

        function updateButtons() {
            btnPrev.disabled = currentStep === 0;
            btnNext.disabled = currentStep === maxStep;
        }

        let blinkInterval;

        function renderStep(step, direction) {
            const t = d3.transition().duration(800);
            
            // 清理闪烁
            if (blinkInterval) {
                clearInterval(blinkInterval);
                blinkInterval = null;
            }
            cVertLine1.transition().duration(0).style("opacity", step === 4 ? 1 : 0);
            cVertLine2.transition().duration(0).style("opacity", step === 4 ? 1 : 0);

            if (step === 0) {
                dashA.transition(t).style("opacity", 0);
                shapeA.transition(t).attr("stroke-width", 2);
                
                dashB.transition(t).style("opacity", 0);
                bHorizLines.forEach((l, i) => {
                    let y = 252 - i*18;
                    l.transition(t).attr("y1", y).attr("y2", y).style("opacity", 0);
                });
                bVertLines.forEach((l, i) => {
                    let x = 445 - i*30;
                    l.transition(t).attr("x1", x).attr("x2", x).style("opacity", 0);
                });

                dashC.transition(t).style("opacity", 0);
                cHorizLine.transition(t).attr("y1", 225).attr("y2", 225).style("opacity", 0);
            }
            else if (step === 1) {
                dashA.transition(t).style("opacity", 1);
                shapeA.transition(t).attr("stroke-width", 4);
                
                dashB.transition(t).style("opacity", 0);
                bHorizLines.forEach((l, i) => {
                    let y = 252 - i*18;
                    l.transition(t).attr("y1", y).attr("y2", y).style("opacity", 0);
                });
                bVertLines.forEach((l, i) => {
                    let x = 445 - i*30;
                    l.transition(t).attr("x1", x).attr("x2", x).style("opacity", 0);
                });
            }
            else if (step === 2) {
                dashA.transition(t).style("opacity", 1);
                shapeA.transition(t).attr("stroke-width", 4);
                
                dashB.transition(t).style("opacity", 1);
                bHorizLines.forEach((l, i) => {
                    let origY = 252 - i*18;
                    if (direction === 'forward') {
                        l.attr("y1", origY).attr("y2", origY).style("opacity", 1)
                         .transition(t).attr("y1", 180).attr("y2", 180);
                    } else {
                        l.transition(t).attr("y1", 180).attr("y2", 180).style("opacity", 1);
                    }
                });
                bVertLines.forEach((l, i) => {
                    let origX = 445 - i*30;
                    if (direction === 'forward') {
                        l.attr("x1", origX).attr("x2", origX).style("opacity", 1)
                         .transition(t).attr("x1", 475).attr("x2", 475);
                    } else {
                        l.transition(t).attr("x1", 475).attr("x2", 475).style("opacity", 1);
                    }
                });

                dashC.transition(t).style("opacity", 0);
                cHorizLine.transition(t).attr("y1", 225).attr("y2", 225).style("opacity", 0);
            }
            else if (step === 3) {
                dashB.transition(t).style("opacity", 1);
                bHorizLines.forEach(l => l.transition(t).attr("y1", 180).attr("y2", 180).style("opacity", 1));
                bVertLines.forEach(l => l.transition(t).attr("x1", 475).attr("x2", 475).style("opacity", 1));

                dashC.transition(t).style("opacity", 1);
                if (direction === 'forward') {
                    cHorizLine.attr("y1", 225).attr("y2", 225).style("opacity", 1)
                              .transition(t).attr("y1", 180).attr("y2", 180);
                } else {
                    cHorizLine.transition(t).attr("y1", 180).attr("y2", 180).style("opacity", 1);
                }
            }
            else if (step === 4) {
                dashC.transition(t).style("opacity", 1);
                cHorizLine.transition(t).attr("y1", 180).attr("y2", 180).style("opacity", 1);
                
                cVertLine1.style("opacity", 1);
                cVertLine2.style("opacity", 1);
                
                // 闪烁效果
                let blinkState = false;
                blinkInterval = setInterval(() => {
                    blinkState = !blinkState;
                    cVertLine1.transition().duration(400).style("opacity", blinkState ? 0.2 : 1);
                    cVertLine2.transition().duration(400).style("opacity", blinkState ? 0.2 : 1);
                }, 500);
            }
        }

        btnPrev.addEventListener("click", () => {
            if (currentStep > 0) {
                currentStep--;
                updateButtons();
                renderStep(currentStep, 'backward');
            }
        });

        btnNext.addEventListener("click", () => {
            if (currentStep < maxStep) {
                currentStep++;
                updateButtons();
                renderStep(currentStep, 'forward');
            }
        });

        // 初始化
        updateButtons();
        renderStep(0, 'forward');

    })();
    </script>
</body>
</html>
