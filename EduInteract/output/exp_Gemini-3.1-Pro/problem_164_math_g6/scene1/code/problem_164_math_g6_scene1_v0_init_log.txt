    <script>
    (function() {
        let currentStep = 0;
        const maxStep = 3;

        const btnPrev = document.getElementById('scene1-btn-prev');
        const btnNext = document.getElementById('scene1-btn-next');

        function show(id) {
            const el = document.getElementById(id);
            if (el) {
                el.classList.remove('hidden');
                el.classList.add('visible');
            }
        }

        function hide(id) {
            const el = document.getElementById(id);
            if (el) {
                el.classList.remove('visible');
                el.classList.add('hidden');
            }
        }

        function addClass(id, cls) {
            const el = document.getElementById(id);
            if (el) el.classList.add(cls);
        }

        function removeClass(id, cls) {
            const el = document.getElementById(id);
            if (el) el.classList.remove(cls);
        }

        function updateScene() {
            // Reset all
            hide('scene1-proj-top');
            hide('scene1-view-top1'); hide('scene1-view-top2'); hide('scene1-view-top3');
            hide('scene1-mark-top1'); hide('scene1-mark-top2'); hide('scene1-mark-top3');
            removeClass('scene1-poly-top1', 'wrong'); removeClass('scene1-poly-top2', 'correct'); removeClass('scene1-poly-top3', 'wrong');

            hide('scene1-proj-front');
            hide('scene1-view-front1'); hide('scene1-view-front2'); hide('scene1-view-front3');
            hide('scene1-mark-front1'); hide('scene1-mark-front2'); hide('scene1-mark-front3');
            removeClass('scene1-poly-front1', 'wrong'); removeClass('scene1-line-front1', 'wrong');
            removeClass('scene1-poly-front2', 'correct'); removeClass('scene1-poly-front3', 'correct');

            hide('scene1-proj-side');
            hide('scene1-view-side2');
            hide('scene1-mark-side2');
            removeClass('scene1-poly-side2', 'correct'); removeClass('scene1-line-side2', 'correct');

            hide('scene1-conclusion');

            if (currentStep >= 1) {
                show('scene1-proj-top');
                show('scene1-view-top1'); show('scene1-view-top2'); show('scene1-view-top3');
                
                // Delay marks and highlights for better animation effect
                setTimeout(() => {
                    if (currentStep >= 1) {
                        addClass('scene1-poly-top1', 'wrong'); show('scene1-mark-top1');
                        addClass('scene1-poly-top2', 'correct'); show('scene1-mark-top2');
                        addClass('scene1-poly-top3', 'wrong'); show('scene1-mark-top3');
                    }
                }, 300);
            }

            if (currentStep >= 2) {
                show('scene1-proj-front');
                show('scene1-view-front1'); show('scene1-view-front2'); show('scene1-view-front3');
                
                setTimeout(() => {
                    if (currentStep >= 2) {
                        addClass('scene1-poly-front1', 'wrong'); addClass('scene1-line-front1', 'wrong'); show('scene1-mark-front1');
                        addClass('scene1-poly-front2', 'correct'); show('scene1-mark-front2');
                        addClass('scene1-poly-front3', 'correct'); show('scene1-mark-front3');
                    }
                }, 300);
            }

            if (currentStep >= 3) {
                show('scene1-proj-side');
                show('scene1-view-side2');
                
                setTimeout(() => {
                    if (currentStep >= 3) {
                        addClass('scene1-poly-side2', 'correct'); addClass('scene1-line-side2', 'correct'); show('scene1-mark-side2');
                        show('scene1-conclusion');
                    }
                }, 300);
            }

            btnPrev.disabled = currentStep === 0;
            btnNext.disabled = currentStep === maxStep;
        }

        btnPrev.addEventListener('click', () => {
            if (currentStep > 0) {
                currentStep--;
                updateScene();
            }
        });

        btnNext.addEventListener('click', () => {
            if (currentStep < maxStep) {
                currentStep++;
                updateScene();
            }
        });

        // Initialize
        updateScene();
    })();
    </script>
