    <script>
    (function() {
        let currentStep = 0;
        const maxStep = 4;

        const btnPrev = document.getElementById('scene1-btn-prev');
        const btnNext = document.getElementById('scene1-btn-next');
        const stepIndicator = document.getElementById('scene1-step-indicator');

        // Elements
        const elJia = document.getElementById('scene1-jia');
        const elYiGroup = document.getElementById('scene1-yi-group');
        const elYiRight = document.getElementById('scene1-yi-right');
        const elGreenTri = document.getElementById('scene1-green-tri');
        
        const elJiaBase = document.getElementById('scene1-jia-base');
        const elJiaHeight = document.getElementById('scene1-jia-height');
        const elAuxLine = document.getElementById('scene1-aux-line');
        
        const elLabelBase = document.getElementById('scene1-label-base');
        const elLabelHeight = document.getElementById('scene1-label-height');
        const elLabelJiaArea = document.getElementById('scene1-label-jia-area');
        const elLabelYiArea = document.getElementById('scene1-label-yi-area');
        const elLabelConclusion = document.getElementById('scene1-label-conclusion');

        function updateState() {
            stepIndicator.textContent = `${currentStep} / ${maxStep}`;
            btnPrev.disabled = currentStep === 0;
            btnNext.disabled = currentStep === maxStep;

            // Reset all
            elJia.setAttribute('opacity', '0.2');
            elYiGroup.setAttribute('opacity', '0.2');
            elYiRight.setAttribute('opacity', '1');
            elGreenTri.setAttribute('opacity', '0');
            elGreenTri.style.transform = 'translateX(0px)';
            
            elJiaBase.setAttribute('opacity', '0');
            elJiaHeight.setAttribute('opacity', '0');
            elAuxLine.setAttribute('opacity', '0');
            
            elLabelBase.setAttribute('opacity', '0');
            elLabelHeight.setAttribute('opacity', '0');
            elLabelJiaArea.setAttribute('opacity', '0');
            elLabelYiArea.setAttribute('opacity', '0');
            elLabelConclusion.setAttribute('opacity', '0');
            
            elLabelJiaArea.setAttribute('fill', '#000000');
            elLabelJiaArea.setAttribute('font-weight', 'normal');
            elLabelYiArea.setAttribute('fill', '#000000');
            elLabelYiArea.setAttribute('font-weight', 'normal');

            if (currentStep >= 1) {
                elJia.setAttribute('opacity', '0.4');
                elJiaBase.setAttribute('opacity', '1');
                elJiaHeight.setAttribute('opacity', '1');
                elAuxLine.setAttribute('opacity', '1');
                elLabelBase.setAttribute('opacity', '1');
                elLabelHeight.setAttribute('opacity', '1');
                elLabelJiaArea.setAttribute('opacity', '1');
            }

            if (currentStep >= 2) {
                elJia.setAttribute('opacity', '0.2');
                elJiaBase.setAttribute('opacity', '0');
                elJiaHeight.setAttribute('opacity', '0');
                elAuxLine.setAttribute('opacity', '0');
                elLabelBase.setAttribute('opacity', '0');
                elLabelHeight.setAttribute('opacity', '0');
                
                elYiGroup.setAttribute('opacity', '0.4');
                elYiRight.setAttribute('opacity', '0');
                elGreenTri.setAttribute('opacity', '0.6');
            }

            if (currentStep >= 3) {
                elGreenTri.style.transform = 'translateX(-160px)';
            }

            if (currentStep >= 4) {
                elLabelYiArea.setAttribute('opacity', '1');
                elLabelConclusion.setAttribute('opacity', '1');
                
                elLabelJiaArea.setAttribute('fill', '#c62828');
                elLabelJiaArea.setAttribute('font-weight', 'bold');
                elLabelYiArea.setAttribute('fill', '#c62828');
                elLabelYiArea.setAttribute('font-weight', 'bold');
            }
        }

        btnPrev.addEventListener('click', () => {
            if (currentStep > 0) {
                currentStep--;
                updateState();
            }
        });

        btnNext.addEventListener('click', () => {
            if (currentStep < maxStep) {
                currentStep++;
                updateState();
            }
        });

        // Initialize
        updateState();
    })();
    </script>
