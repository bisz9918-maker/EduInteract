    <script>
    (function() {
        let currentStep = 0;
        const maxStep = 5;
        
        const btnPrev = document.getElementById('scene1-btn-prev');
        const btnNext = document.getElementById('scene1-btn-next');
        const panel = document.getElementById('scene1-panel');
        
        const els = {
            line3_1: document.getElementById('scene1-line3-1'),
            line3_2: document.getElementById('scene1-line3-2'),
            line3_3: document.getElementById('scene1-line3-3'),
            f1White: document.getElementById('scene1-f1-white'),
            f2Yellow2: document.getElementById('scene1-f2-yellow2'),
            f2White: document.getElementById('scene1-f2-white'),
            dom: document.getElementById('scene1-dom'),
            rec: document.getElementById('scene1-rec'),
            pYGeno: document.getElementById('scene1-p-y-geno'),
            pWGeno: document.getElementById('scene1-p-w-geno'),
            f1YGeno: document.getElementById('scene1-f1-y-geno'),
            f1WGeno: document.getElementById('scene1-f1-w-geno'),
            f2Y1Geno: document.getElementById('scene1-f2-y1-geno'),
            f2Y2Geno: document.getElementById('scene1-f2-y2-geno'),
            testcross: document.getElementById('scene1-testcross'),
            f1YProb: document.getElementById('scene1-f1-y-prob'),
            f1WProb: document.getElementById('scene1-f1-w-prob'),
            calc2: document.getElementById('scene1-calc2'),
            calc3_1: document.getElementById('scene1-calc3-1'),
            calc3_2: document.getElementById('scene1-calc3-2'),
            f2Y1Full: document.getElementById('scene1-f2-y1-full'),
            f2Y2Full: document.getElementById('scene1-f2-y2-full'),
            f2WFull: document.getElementById('scene1-f2-w-full'),
            f2Summary: document.getElementById('scene1-f2-summary'),
            boxF1: document.getElementById('scene1-box-f1'),
            boxF2: document.getElementById('scene1-box-f2')
        };

        const panelContents = [
            "<b>初始状态</b><br><br>请点击「下一步」开始推导。",
            "<b>1. 显隐性判断</b><br><br>观察过程③：F₁的白花自交，后代出现了黄花。<br><br>根据“无中生有为隐性”，可以判断<b>黄花为隐性性状(a)</b>，<b>白花为显性性状(A)</b>。",
            "<b>2. 基因型推导</b><br><br>隐性性状(黄花)的基因型一定是 <b>aa</b>。<br><br>F₁白花自交后代有隐性纯合子(aa)，说明F₁白花一定是杂合子 <b>Aa</b>。<br><br>P代黄花(aa)与白花杂交产生F₁黄花(aa)，说明P代白花也必须提供a基因，因此P代白花也是 <b>Aa</b>。",
            "<b>3. F₁比例计算</b><br><br>P代杂交：aa × Aa<br>这是一个典型的<b>测交</b>。<br><br>后代F₁的比例为：<br>黄花(aa) : 白花(Aa) = 1 : 1<br><br>即各占 <b>1/2</b>。",
            "<b>4. F₂比例计算</b><br><br>过程②：1/2的F₁黄花(aa)自交，后代全为黄花(aa)，占总数的 <b>1/2</b>。<br><br>过程③：1/2的F₁白花(Aa)自交，后代基因型比例为 1/4 AA, 2/4 Aa, 1/4 aa。<br>乘以前提概率1/2，得到：<br><b>1/8 AA, 1/4 Aa, 1/8 aa</b>。<br><br>汇总F₂：<br>黄花 = 1/2 + 1/8 = <b>5/8</b><br>白花 = 1/8 + 1/4 = <b>3/8</b>",
            "<b>5. 结论对比</b><br><br>对比F₁与F₂中的白花个体：<br><br>F₁白花基因型全是 <b>Aa</b>。<br><br>F₂白花基因型包含 <b>AA</b> (占1/3) 和 <b>Aa</b> (占2/3)。<br><br>因此，两者的基因型不完全相同。"
        ];

        function show(el) {
            if(!el) return;
            el.style.opacity = '1';
            el.style.display = '';
        }

        function hide(el) {
            if(!el) return;
            el.style.opacity = '0';
        }

        function setHighlight(el, isHighlight) {
            if(!el) return;
            if (isHighlight) {
                el.classList.add('scene1-highlight');
            } else {
                el.classList.remove('scene1-highlight');
            }
        }

        function setLineHighlight(el, isHighlight) {
            if(!el) return;
            if (isHighlight) {
                el.classList.add('scene1-highlight-stroke');
                el.setAttribute('marker-end', el.getAttribute('marker-end') ? 'url(#scene1-arrowhead-red)' : '');
            } else {
                el.classList.remove('scene1-highlight-stroke');
                el.setAttribute('marker-end', el.getAttribute('marker-end') ? 'url(#scene1-arrowhead)' : '');
            }
        }

        function updateView() {
            btnPrev.disabled = currentStep === 0;
            btnNext.disabled = currentStep === maxStep;
            
            panel.innerHTML = panelContents[currentStep];
            if (window.MathJax) {
                MathJax.typesetPromise([panel]).catch(function (err) { console.log(err.message); });
            }

            // Reset all
            hide(els.dom); hide(els.rec);
            hide(els.pYGeno); hide(els.pWGeno); hide(els.f1YGeno); hide(els.f1WGeno);
            hide(els.f2Y1Geno); hide(els.f2Y2Geno);
            hide(els.testcross); hide(els.f1YProb); hide(els.f1WProb);
            hide(els.calc2); hide(els.calc3_1); hide(els.calc3_2);
            hide(els.f2Y1Full); hide(els.f2Y2Full); hide(els.f2WFull);
            hide(els.f2Summary);
            hide(els.boxF1); hide(els.boxF2);
            
            setHighlight(els.f1White, false);
            setHighlight(els.f2Yellow2, false);
            setHighlight(els.f2White, false);
            
            setLineHighlight(els.line3_1, false);
            setLineHighlight(els.line3_2, false);
            setLineHighlight(els.line3_3, false);

            if (currentStep >= 1) {
                if (currentStep === 1) {
                    setHighlight(els.f1White, true);
                    setHighlight(els.f2Yellow2, true);
                    setHighlight(els.f2White, true);
                    setLineHighlight(els.line3_1, true);
                    setLineHighlight(els.line3_2, true);
                    setLineHighlight(els.line3_3, true);
                    show(els.dom);
                    show(els.rec);
                }
            }
            
            if (currentStep >= 2) {
                show(els.pYGeno);
                show(els.pWGeno);
                show(els.f1YGeno);
                show(els.f1WGeno);
                if (currentStep < 4) {
                    show(els.f2Y1Geno);
                    show(els.f2Y2Geno);
                }
            }
            
            if (currentStep >= 3) {
                show(els.testcross);
                show(els.f1YProb);
                show(els.f1WProb);
            }
            
            if (currentStep >= 4) {
                show(els.calc2);
                show(els.calc3_1);
                show(els.calc3_2);
                show(els.f2Y1Full);
                show(els.f2Y2Full);
                show(els.f2WFull);
                show(els.f2Summary);
            }
            
            if (currentStep >= 5) {
                show(els.boxF1);
                show(els.boxF2);
            }
        }

        btnPrev.addEventListener('click', () => {
            if (currentStep > 0) {
                currentStep--;
                updateView();
            }
        });

        btnNext.addEventListener('click', () => {
            if (currentStep < maxStep) {
                currentStep++;
                updateView();
            }
        });

        updateView();
    })();
    </script>
</body>
