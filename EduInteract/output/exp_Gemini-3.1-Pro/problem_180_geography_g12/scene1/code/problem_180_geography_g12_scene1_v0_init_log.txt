    <script>
    (function() {
        let currentStep = 0;
        const maxStep = 4;

        const btnPrev = document.getElementById('scene1-btn-prev');
        const btnNext = document.getElementById('scene1-btn-next');
        const stepIndicator = document.getElementById('scene1-step-indicator');
        const infoContent = document.getElementById('scene1-info-content');

        // SVG Elements
        const terminator = document.getElementById('scene1-terminator');
        const sunrays = document.getElementById('scene1-sunrays');
        const sunrayLines = document.querySelectorAll('.scene1-ray');
        const meridianCenter = document.getElementById('scene1-meridian-center');
        const meridianLeft1 = document.getElementById('scene1-meridian-left1');
        const southHighlight = document.getElementById('scene1-south-highlight');
        const label73 = document.getElementById('scene1-label-73');
        
        const label6Group = document.getElementById('scene1-label-6-group');
        const label6 = document.getElementById('scene1-label-6');
        
        const label12Group = document.getElementById('scene1-label-12-group');
        const label12 = document.getElementById('scene1-label-12');
        
        const label4Group = document.getElementById('scene1-label-4-group');
        const label4 = document.getElementById('scene1-label-4');
        
        const lonDiff = document.getElementById('scene1-lon-diff');
        const rotation = document.getElementById('scene1-rotation');

        const stepData = [
            {
                text: "<p>请点击「下一步」开始解析光照图。</p>"
            },
            {
                text: "<p><b>1. 晨昏线与A点</b></p><p>左侧为夜半球，右侧为昼半球，顺着自转方向（自西向东），由夜进入昼的界线为<b>晨线</b>。</p><p>A点位于赤道与晨线的交点，地方时为 <b>6:00</b>，此时太阳高度为 <b>0°</b>。</p>"
            },
            {
                text: "<p><b>2. 直射点与C点</b></p><p>晨昏线与 73°41′N 相切，推导直射点纬度为 90° - 73°41′ = <b>16°19′N</b>。</p><p>C点在平分昼半球的经线上，地方时为 <b>12:00</b>，正午太阳高度为 <b>73°41′</b>。</p>"
            },
            {
                text: "<p><b>3. B点昼长</b></p><p>相邻经线经度差为 30°。B点在A点以西 30°，时间晚 2 小时。</p><p>推导B点日出时间为 <b>4:00</b>，计算昼长 = (12 - 4) × 2 = <b>16小时</b>。</p>"
            },
            {
                text: "<p><b>4. A点夜长与南半球</b></p><p>A点在赤道上，全年昼夜平分，夜长 <b>12小时</b>。</p><p>南半球阴影面积大于高亮(昼)面积，结论为南半球 <b>昼短夜长</b>。</p>"
            }
        ];

        function updateScene(step) {
            // Reset all
            terminator.setAttribute('stroke', '#000000');
            terminator.setAttribute('stroke-width', '1');
            
            sunrays.setAttribute('stroke', '#000000');
            sunrays.setAttribute('stroke-width', '1');
            sunrayLines.forEach(line => line.setAttribute('marker-end', 'url(#scene1-arrow-black)'));
            
            meridianCenter.setAttribute('stroke', '#000000');
            meridianCenter.setAttribute('stroke-width', '1');
            meridianLeft1.setAttribute('stroke', '#000000');
            meridianLeft1.setAttribute('stroke-width', '1');
            
            label73.setAttribute('fill', '#000000');
            label73.setAttribute('font-weight', 'normal');
            
            label6Group.classList.add('scene1-hidden');
            label6.setAttribute('fill', '#c62828');
            
            label12Group.classList.add('scene1-hidden');
            label12.setAttribute('fill', '#f57f17');
            
            label4Group.classList.add('scene1-hidden');
            label4.setAttribute('fill', '#1565c0');
            
            lonDiff.classList.add('scene1-hidden');
            rotation.classList.add('scene1-hidden');
            southHighlight.classList.add('scene1-hidden');

            // Apply step specific styles
            if (step >= 1) {
                label6Group.classList.remove('scene1-hidden');
                if (step === 1) {
                    terminator.setAttribute('stroke', '#c62828');
                    terminator.setAttribute('stroke-width', '3');
                    rotation.classList.remove('scene1-hidden');
                } else {
                    label6.setAttribute('fill', '#000000');
                }
            }

            if (step >= 2) {
                label12Group.classList.remove('scene1-hidden');
                if (step === 2) {
                    sunrays.setAttribute('stroke', '#f57f17');
                    sunrays.setAttribute('stroke-width', '2');
                    sunrayLines.forEach(line => line.setAttribute('marker-end', 'url(#scene1-arrow-orange)'));
                    label73.setAttribute('fill', '#f57f17');
                    label73.setAttribute('font-weight', 'bold');
                } else {
                    label12.setAttribute('fill', '#000000');
                }
            }

            if (step >= 3) {
                label4Group.classList.remove('scene1-hidden');
                if (step === 3) {
                    meridianCenter.setAttribute('stroke', '#1565c0');
                    meridianCenter.setAttribute('stroke-width', '3');
                    meridianLeft1.setAttribute('stroke', '#1565c0');
                    meridianLeft1.setAttribute('stroke-width', '3');
                    lonDiff.classList.remove('scene1-hidden');
                } else {
                    label4.setAttribute('fill', '#000000');
                }
            }

            if (step >= 4) {
                southHighlight.classList.remove('scene1-hidden');
            }

            // Update UI
            infoContent.innerHTML = stepData[step].text;
            stepIndicator.textContent = `步骤 ${step} / ${maxStep}`;
            btnPrev.disabled = step === 0;
            btnNext.disabled = step === maxStep;
        }

        btnPrev.addEventListener('click', () => {
            if (currentStep > 0) {
                currentStep--;
                updateScene(currentStep);
            }
        });

        btnNext.addEventListener('click', () => {
            if (currentStep < maxStep) {
                currentStep++;
                updateScene(currentStep);
            }
        });

        // Initialize
        updateScene(0);
    })();
    </script>
