    <script>
    (function() {
        let currentStep = 0;
        const maxStep = 4;
        
        const btnPrev = document.getElementById('scene2-btn-prev');
        const btnNext = document.getElementById('scene2-btn-next');
        const beijingTimeText = document.getElementById('scene2-beijing-time');
        const spanArrow = document.getElementById('scene2-span-arrow');
        
        const stepElements = document.querySelectorAll('.step-element');

        function updateScene() {
            // Update buttons
            btnPrev.disabled = currentStep === 0;
            btnNext.disabled = currentStep === maxStep;
            
            // Update elements visibility
            stepElements.forEach(el => {
                const step = parseInt(el.getAttribute('data-step'));
                if (currentStep >= step) {
                    el.classList.add('visible');
                    if (el.id === 'scene2-step3-elements') {
                        spanArrow.classList.add('visible');
                    }
                } else {
                    el.classList.remove('visible');
                    if (el.id === 'scene2-step3-elements') {
                        spanArrow.classList.remove('visible');
                    }
                }
            });
            
            // Update Beijing time text
            if (currentStep >= 4) {
                beijingTimeText.textContent = '20:00';
            } else {
                beijingTimeText.textContent = '?';
            }
        }

        btnPrev.addEventListener('click', () => {
            if (currentStep > 0) {
                currentStep--;
                updateScene();
            }
        });

        btnNext.addEventListener('click', () => {
            if (currentStep < maxStep) {
                currentStep++;
                updateScene();
            }
        });

        // Initialize
        updateScene();
    })();
    </script>
