    <script>
    (function() {
        let currentStep = 0;
        const maxStep = 2;
        
        const btnPrev = document.getElementById('scene2-btn-prev');
        const btnNext = document.getElementById('scene2-btn-next');
        
        const step1 = document.getElementById('scene2-step1');
        const step2 = document.getElementById('scene2-step2');
        
        function updateScene() {
            btnPrev.disabled = currentStep === 0;
            btnNext.disabled = currentStep === maxStep;
            
            if (currentStep >= 1) {
                step1.classList.remove('hidden');
                step1.classList.add('visible');
            } else {
                step1.classList.remove('visible');
                step1.classList.add('hidden');
            }
            
            if (currentStep >= 2) {
                step2.classList.remove('hidden');
                step2.classList.add('visible');
            } else {
                step2.classList.remove('visible');
                step2.classList.add('hidden');
            }
        }
        
        btnPrev.addEventListener('click', () => {
            if (currentStep > 0) {
                currentStep--;
                updateScene();
            }
        });
        
        btnNext.addEventListener('click', () => {
            if (currentStep < maxStep) {
                currentStep++;
                updateScene();
            }
        });
        
        // Initialize
        updateScene();
    })();
    </script>
