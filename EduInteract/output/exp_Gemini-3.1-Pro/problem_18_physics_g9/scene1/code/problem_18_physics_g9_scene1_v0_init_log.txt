    <script>
    (function() {
        let currentStep = 0;
        const maxStep = 4;
        
        const btnPrev = document.getElementById('scene1-btn-prev');
        const btnNext = document.getElementById('scene1-btn-next');
        const textDynamic = document.getElementById('scene1-text-dynamic');
        
        const objB = document.getElementById('scene1-obj-b');
        const objA = document.getElementById('scene1-obj-a');
        
        const vecGb = document.getElementById('scene1-vec-gb');
        const vecTb = document.getElementById('scene1-vec-tb');
        const vecTprime = document.getElementById('scene1-vec-tprime');
        const vecGa = document.getElementById('scene1-vec-ga');
        const vecTa = document.getElementById('scene1-vec-ta');
        const vecFbuoy = document.getElementById('scene1-vec-fbuoy');
        
        const stepTexts = [
            "", // Step 0
            "<strong>分析物体B：</strong><br>$G_B = m_B g = 19.6	ext{N}$<br>因静止，受力平衡：<br>$T = G_B = 19.6	ext{N}$<br>测力计示数为 $19.6	ext{N}$", // Step 1
            "<strong>辨析平衡力：</strong><br>B受到的重力 $G_B$ 与<br>B对绳子的拉力 $T'$<br>作用在不同物体上，<br><strong>不是</strong>平衡力。", // Step 2
            "<strong>分析物体A状态：</strong><br>整体处于静止状态，<br>静止状态 = 平衡状态。", // Step 3
            "<strong>计算浮力与体积：</strong><br>$G_A = m_A g = 29.4	ext{N}$<br>$F_浮 = G_A - T = 9.8	ext{N}$<br>$V_排 = rac{F_浮}{ho_水 g} = 1000	ext{cm}^3$" // Step 4
        ];
        
        function updateScene() {
            // 按钮状态
            btnPrev.disabled = currentStep === 0;
            btnNext.disabled = currentStep === maxStep;
            
            // 文本更新
            textDynamic.innerHTML = stepTexts[currentStep];
            if (window.MathJax) {
                MathJax.typesetPromise([textDynamic]).catch(function (err) {
                    console.error(err.message);
                });
            }
            
            // 元素状态重置
            objB.classList.remove('highlight');
            objA.classList.remove('highlight');
            vecGb.style.opacity = 0;
            vecTb.style.opacity = 0;
            vecTprime.style.opacity = 0;
            vecGa.style.opacity = 0;
            vecTa.style.opacity = 0;
            vecFbuoy.style.opacity = 0;
            
            // 根据步骤设置状态
            if (currentStep === 1) {
                objB.classList.add('highlight');
                vecGb.style.opacity = 1;
                vecTb.style.opacity = 1;
            } else if (currentStep === 2) {
                objB.classList.add('highlight');
                vecGb.style.opacity = 1;
                vecTprime.style.opacity = 1;
            } else if (currentStep === 3) {
                objA.classList.add('highlight');
                vecGb.style.opacity = 1;
                vecTb.style.opacity = 1;
            } else if (currentStep === 4) {
                objA.classList.add('highlight');
                vecGb.style.opacity = 1;
                vecTb.style.opacity = 1;
                vecGa.style.opacity = 1;
                vecTa.style.opacity = 1;
                vecFbuoy.style.opacity = 1;
            }
        }
        
        btnPrev.addEventListener('click', () => {
            if (currentStep > 0) {
                currentStep--;
                updateScene();
            }
        });
        
        btnNext.addEventListener('click', () => {
            if (currentStep < maxStep) {
                currentStep++;
                updateScene();
            }
        });
        
        // 初始化
        updateScene();
    })();
    </script>
