    <script>
    (function() {
        let currentStep = 0;
        const maxStep = 3;
        
        const btnPrev = document.getElementById('scene1-btn-prev');
        const btnNext = document.getElementById('scene1-btn-next');
        
        const hlBoundary = document.getElementById('scene1-hl-boundary');
        const hlOp = document.getElementById('scene1-hl-op');
        const arrowAdd = document.getElementById('scene1-arrow-add');
        const arrowSub = document.getElementById('scene1-arrow-sub');
        
        const formula1 = document.getElementById('scene1-formula-1');
        const formula2 = document.getElementById('scene1-formula-2');
        const formula3 = document.getElementById('scene1-formula-3');

        function show(el) {
            if (!el) return;
            el.style.opacity = '1';
            el.style.display = '';
        }

        function hide(el) {
            if (!el) return;
            el.style.opacity = '0';
            el.style.display = 'none';
        }

        function renderStep() {
            btnPrev.disabled = currentStep === 0;
            btnNext.disabled = currentStep === maxStep;
            
            // Step 1
            if (currentStep >= 1) {
                if (currentStep === 1) show(hlBoundary); else hide(hlBoundary);
                show(formula1);
            } else {
                hide(hlBoundary);
                hide(formula1);
            }
            
            // Step 2
            if (currentStep >= 2) {
                show(hlOp);
                if (currentStep === 2) show(arrowAdd); else hide(arrowAdd);
                show(formula2);
            } else {
                hide(hlOp);
                hide(arrowAdd);
                hide(formula2);
            }
            
            // Step 3
            if (currentStep >= 3) {
                show(arrowSub);
                show(formula3);
            } else {
                hide(arrowSub);
                hide(formula3);
            }
        }

        btnPrev.addEventListener('click', () => {
            if (currentStep > 0) {
                currentStep--;
                renderStep();
            }
        });

        btnNext.addEventListener('click', () => {
            if (currentStep < maxStep) {
                currentStep++;
                renderStep();
            }
        });

        // Init
        renderStep();
    })();
    </script>
