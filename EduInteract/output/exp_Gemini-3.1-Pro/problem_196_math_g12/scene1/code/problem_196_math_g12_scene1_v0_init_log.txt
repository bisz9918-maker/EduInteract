    <script>
    (function() {
        let currentStep = 0;
        const maxStep = 4;
        
        const btnPrev = document.getElementById('scene1-btn-prev');
        const btnNext = document.getElementById('scene1-btn-next');
        
        function updateStep() {
            btnPrev.disabled = currentStep === 0;
            btnNext.disabled = currentStep === maxStep;
            
            const setOpacity = (id, opacity) => {
                const el = document.getElementById(id);
                if (el) el.style.opacity = opacity;
            };
            
            const setPointStyle = (id, isRed) => {
                const el = document.getElementById(id);
                if (el) {
                    el.setAttribute('fill', isRed ? '#c62828' : '#000000');
                    el.setAttribute('r', isRed ? '5' : '3');
                }
            };
            
            // Step 1
            const step1Opacity = currentStep >= 1 ? 1 : 0;
            setOpacity('scene1-line-OP', step1Opacity);
            setOpacity('scene1-line-OM', step1Opacity);
            setOpacity('scene1-angle-OPA', step1Opacity);
            setOpacity('scene1-angle-OMA', step1Opacity);
            
            // Step 2
            const step2Opacity = currentStep >= 2 ? 1 : 0;
            setOpacity('scene1-circle-aux', step2Opacity);
            setOpacity('scene1-line-PM', step2Opacity);
            
            const isRed = currentStep >= 2;
            setPointStyle('scene1-point-A', isRed);
            setPointStyle('scene1-point-P', isRed);
            setPointStyle('scene1-point-O', isRed);
            setPointStyle('scene1-point-M', isRed);
            
            // Step 3
            const step3Opacity = currentStep >= 3 ? 0.2 : 0;
            setOpacity('scene1-arc-AM', currentStep >= 3 ? 1 : 0);
            setOpacity('scene1-sector-APM', step3Opacity);
            setOpacity('scene1-sector-AOM', step3Opacity);
            
            // Step 4
            const step4Opacity = currentStep >= 4 ? 1 : 0;
            setOpacity('scene1-triangle-AOM', currentStep >= 4 ? 0.5 : 0);
            setOpacity('scene1-sector-OAM', currentStep >= 4 ? 0.2 : 0);
            setOpacity('scene1-text-conclusion', step4Opacity);
        }
        
        btnPrev.addEventListener('click', () => {
            if (currentStep > 0) {
                currentStep--;
                updateStep();
            }
        });
        
        btnNext.addEventListener('click', () => {
            if (currentStep < maxStep) {
                currentStep++;
                updateStep();
            }
        });
        
        updateStep();
    })();
    </script>
