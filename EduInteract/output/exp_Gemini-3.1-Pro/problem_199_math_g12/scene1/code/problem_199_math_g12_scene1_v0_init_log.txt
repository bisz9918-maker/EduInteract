    <script>
    (function() {
        let currentStep = 0;
        const maxStep = 3;
        
        const btnPrev = document.getElementById('scene1-btn-prev');
        const btnNext = document.getElementById('scene1-btn-next');
        
        const step1Elements = document.getElementById('scene1-step1-elements');
        const step2Lines = document.getElementById('scene1-step2-lines');
        const pointO = document.getElementById('scene1-point-o');
        const lineMM = document.getElementById('scene1-line-mm');
        const lineNN = document.getElementById('scene1-line-nn');
        const triOMN = document.getElementById('scene1-tri-omn');
        const triOMPNP = document.getElementById('scene1-tri-ompnp');
        const mathConclusion = document.getElementById('scene1-math-conclusion');
        
        function show(el) {
            el.style.display = '';
            // Force reflow
            void el.offsetWidth;
            el.style.opacity = '1';
        }
        
        function hide(el) {
            el.style.opacity = '0';
            setTimeout(() => {
                if (el.style.opacity === '0') {
                    el.style.display = 'none';
                }
            }, 500);
        }
        
        function updateScene() {
            // Update buttons
            btnPrev.disabled = currentStep === 0;
            btnNext.disabled = currentStep === maxStep;
            
            // Step 1: Show L', M', N', △OM'N'
            if (currentStep >= 1) {
                show(step1Elements);
            } else {
                hide(step1Elements);
            }
            
            // Step 2: Highlight O, draw dashed lines
            if (currentStep >= 2) {
                show(step2Lines);
                pointO.style.fill = '#2e7d32';
                pointO.setAttribute('r', '6');
                
                // Trigger drawing animation if just entered step 2
                if (currentStep === 2) {
                    lineMM.classList.remove('scene1-draw-dash');
                    lineNN.classList.remove('scene1-draw-dash');
                    void lineMM.offsetWidth; // reflow
                    lineMM.classList.add('scene1-draw-dash');
                    lineNN.classList.add('scene1-draw-dash');
                }
            } else {
                hide(step2Lines);
                pointO.style.fill = '#000000';
                pointO.setAttribute('r', '4');
                lineMM.classList.remove('scene1-draw-dash');
                lineNN.classList.remove('scene1-draw-dash');
            }
            
            // Step 3: Blinking triangles, show conclusion
            if (currentStep >= 3) {
                triOMN.classList.add('scene1-blinking');
                triOMPNP.classList.add('scene1-blinking');
                mathConclusion.style.opacity = '1';
            } else {
                triOMN.classList.remove('scene1-blinking');
                triOMPNP.classList.remove('scene1-blinking');
                mathConclusion.style.opacity = '0';
            }
        }
        
        btnPrev.addEventListener('click', () => {
            if (currentStep > 0) {
                currentStep--;
                updateScene();
            }
        });
        
        btnNext.addEventListener('click', () => {
            if (currentStep < maxStep) {
                currentStep++;
                updateScene();
            }
        });
        
        // Initial setup
        updateScene();
    })();
    </script>
