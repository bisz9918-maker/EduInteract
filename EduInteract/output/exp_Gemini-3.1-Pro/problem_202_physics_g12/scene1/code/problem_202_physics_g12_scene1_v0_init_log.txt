    <script>
    (function() {
        // Generate B field crosses
        const bFieldGroup = document.getElementById('scene1-bfield');
        const xs = [540, 590, 640, 690, 740];
        const ys = [100, 150, 200, 300, 350, 400];
        
        xs.forEach(x => {
            ys.forEach(y => {
                const path = document.createElementNS('http://www.w3.org/2000/svg', 'path');
                path.setAttribute('d', `M ${x-4} ${y-4} L ${x+4} ${y+4} M ${x-4} ${y+4} L ${x+4} ${y-4}`);
                path.setAttribute('stroke', '#888');
                path.setAttribute('stroke-width', '1.5');
                bFieldGroup.appendChild(path);
            });
        });
        
        // State management
        let currentStep = 0;
        const maxStep = 2;
        
        const btnPrev = document.getElementById('scene1-btn-prev');
        const btnNext = document.getElementById('scene1-btn-next');
        
        const step1 = document.getElementById('scene1-step1');
        const step2 = document.getElementById('scene1-step2');
        
        function updateState() {
            btnPrev.disabled = currentStep === 0;
            btnNext.disabled = currentStep === maxStep;
            
            if (currentStep >= 1) {
                step1.style.display = 'block';
                // Force reflow
                void step1.offsetWidth;
                step1.style.opacity = '1';
            } else {
                step1.style.opacity = '0';
                setTimeout(() => { if(currentStep < 1) step1.style.display = 'none'; }, 500);
            }
            
            if (currentStep >= 2) {
                step2.style.display = 'block';
                // Force reflow
                void step2.offsetWidth;
                step2.style.opacity = '1';
            } else {
                step2.style.opacity = '0';
                setTimeout(() => { if(currentStep < 2) step2.style.display = 'none'; }, 500);
            }
        }
        
        btnPrev.addEventListener('click', () => {
            if (currentStep > 0) {
                currentStep--;
                updateState();
            }
        });
        
        btnNext.addEventListener('click', () => {
            if (currentStep < maxStep) {
                currentStep++;
                updateState();
            }
        });
        
        // MathJax rendering
        if (window.MathJax) {
            MathJax.typesetPromise();
        }
    })();
    </script>
