    <script>
    (function() {
        let currentStep = 0;
        const maxStep = 3;

        const btnPrev = document.getElementById('scene1-btn-prev');
        const btnNext = document.getElementById('scene1-btn-next');
        const stepText = document.getElementById('scene1-step-text');

        const step1Group = document.getElementById('scene1-step1-group');
        const step2Group = document.getElementById('scene1-step2-group');
        const step3Group = document.getElementById('scene1-step3-group');
        const tri1 = document.getElementById('scene1-tri1');
        const tri2 = document.getElementById('scene1-tri2');
        const particleStart = document.getElementById('scene1-particle-start');

        function updateScene() {
            // Update buttons
            btnPrev.disabled = currentStep === 0;
            btnNext.disabled = currentStep === maxStep;
            stepText.textContent = `${currentStep + 1} / ${maxStep + 1}`;

            // Step 1: Parabola
            if (currentStep >= 1) {
                step1Group.style.opacity = '1';
                particleStart.style.opacity = '0'; // Hide initial particle to avoid duplicate
            } else {
                step1Group.style.opacity = '0';
                particleStart.style.opacity = '1';
            }

            // Step 2: Velocity decomposition
            if (currentStep >= 2) {
                step2Group.style.opacity = '1';
            } else {
                step2Group.style.opacity = '0';
            }

            // Step 3: Linear trajectory and similar triangles
            if (currentStep >= 3) {
                step3Group.style.opacity = '1';
                tri1.style.opacity = '0.15';
                tri2.style.opacity = '0.15';
            } else {
                step3Group.style.opacity = '0';
                tri1.style.opacity = '0';
                tri2.style.opacity = '0';
            }
        }

        btnPrev.addEventListener('click', () => {
            if (currentStep > 0) {
                currentStep--;
                updateScene();
            }
        });

        btnNext.addEventListener('click', () => {
            if (currentStep < maxStep) {
                currentStep++;
                updateScene();
            }
        });

        // Initialize
        updateScene();
    })();
    </script>
