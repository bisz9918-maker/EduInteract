            <div id="scene1-text-step1" class="scene1-step-text">
                <strong>1. 正方形性质</strong><br>
                因为四边形 $ABCD$ 是正方形，<br>
                所以 $AD = AB$，<br>
                且 $\angle DAB = 90^\circ$。
            </div>
            <div id="scene1-text-step2" class="scene1-step-text" style="top: 80px;">
                <strong>2. 角度转换</strong><br>
                因为 $\angle DAB = 90^\circ$，<br>
                所以 $\angle DAE + \angle BAF = 90^\circ$。<br>
                又因为 $DE \perp a$，即 $\angle DEA = 90^\circ$，<br>
                所以 $\angle DAE + \angle ADE = 90^\circ$。<br>
                因此，$\angle ADE = \angle BAF$。
            </div>
            <div id="scene1-text-step3" class="scene1-step-text" style="top: 200px;">
                <strong>3. 全等证明</strong><br>
                在 $	riangle ADE$ 和 $	riangle BAF$ 中：<br>
                $egin{cases} \angle DEA = \angle AFB = 90^\circ \ \angle ADE = \angle BAF \ AD = AB \end{cases}$<br>
                所以 $	riangle ADE \cong 	riangle BAF$ (AAS)。
            </div>
            <div id="scene1-text-step4" class="scene1-step-text" style="top: 320px;">
                <strong>4. 计算求解</strong><br>
                由全等可知：<br>
                $AE = BF = 3$<br>
                $AF = DE = 4$<br>
                所以 $EF = AE + AF = 3 + 4 = 7$。
            </div>
        </div>

    <script>
    (function() {
        let currentStep = 0;
        const maxStep = 4;
        
        const btnPrev = document.getElementById('scene1-btn-prev');
        const btnNext = document.getElementById('scene1-btn-next');
        
        function updateStep() {
            btnPrev.disabled = currentStep === 0;
            btnNext.disabled = currentStep === maxStep;
            
            const setOpacity = (id, opacity) => {
                const el = document.getElementById(id);
                if (el) {
                    el.style.opacity = opacity;
                    if (opacity > 0) {
                        el.style.display = '';
                    }
                }
            };
            
            // Step 1
            setOpacity('scene1-hl-ad', currentStep >= 1 ? 1 : 0);
            setOpacity('scene1-hl-ab', currentStep >= 1 ? 1 : 0);
            setOpacity('scene1-tick-ad', currentStep >= 1 ? 1 : 0);
            setOpacity('scene1-tick-ab', currentStep >= 1 ? 1 : 0);
            setOpacity('scene1-rt-a', currentStep >= 1 ? 1 : 0);
            setOpacity('scene1-text-step1', currentStep >= 1 ? 1 : 0);
            
            // Step 2
            setOpacity('scene1-arc-dae', currentStep >= 2 ? 1 : 0);
            setOpacity('scene1-arc-baf', currentStep >= 2 ? 1 : 0);
            setOpacity('scene1-arc-ade', currentStep >= 2 ? 1 : 0);
            setOpacity('scene1-tick-baf', currentStep >= 2 ? 1 : 0);
            setOpacity('scene1-tick-ade', currentStep >= 2 ? 1 : 0);
            setOpacity('scene1-text-step2', currentStep >= 2 ? 1 : 0);
            
            if (currentStep >= 2) {
                document.getElementById('scene1-arc-ade').setAttribute('stroke', '#c62828');
            } else {
                document.getElementById('scene1-arc-ade').setAttribute('stroke', '#1565c0');
            }
            
            // Step 3
            setOpacity('scene1-tri-ade', currentStep >= 3 ? 0.2 : 0);
            setOpacity('scene1-tri-baf', currentStep >= 3 ? 0.2 : 0);
            setOpacity('scene1-text-step3', currentStep >= 3 ? 1 : 0);
            
            const blinkElements = ['scene1-rt-e', 'scene1-rt-f', 'scene1-arc-ade', 'scene1-arc-baf', 'scene1-hl-ad', 'scene1-hl-ab'];
            if (currentStep === 3) {
                blinkElements.forEach(id => document.getElementById(id).classList.add('scene1-blinking'));
            } else {
                blinkElements.forEach(id => document.getElementById(id).classList.remove('scene1-blinking'));
            }
            
            // Step 4
            setOpacity('scene1-label-ae', currentStep >= 4 ? 1 : 0);
            setOpacity('scene1-label-af', currentStep >= 4 ? 1 : 0);
            setOpacity('scene1-hl-ef', currentStep >= 4 ? 1 : 0);
            setOpacity('scene1-text-step4', currentStep >= 4 ? 1 : 0);
        }
        
        btnPrev.addEventListener('click', () => {
            if (currentStep > 0) {
                currentStep--;
                updateStep();
            }
        });
        
        btnNext.addEventListener('click', () => {
            if (currentStep < maxStep) {
                currentStep++;
                updateStep();
            }
        });
        
        updateStep();
    })();
    </script>
