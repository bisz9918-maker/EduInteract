    <script>
    (function() {
        let currentStep = 0;
        const totalSteps = 5;
        
        const texts = [
            "请点击「下一步」逐一验证结论",
            "① 抛物线与x轴有两个交点 ⇒ \(b^2 - 4ac > 0\) ⇒ \(4ac < b^2\)。结论①正确",
            "⑤ 对称轴为 \(x=1\)，左交点为 \(x=-1\)。由对称性，右交点为 \(x = 1 + (1 - (-1)) = 3\)。结论⑤正确",
            "④ 当 \(y > 0\) 时，图象在x轴上方，对应的x取值范围是 \(-1 < x < 3\)。结论④正确",
            "③ 当 \(x > 0\) 时，在 \(0 < x < 1\) 区间，y随x增大而增大；在 \(x > 1\) 区间，y随x增大而减小。结论③错误",
            "② 图象过\((-1,0)\) ⇒ \(a-b+c=0\)；对称轴\(x=1\) ⇒ \(b=-2a\)。代入得：\(a-(-2a)+c=0\) ⇒ \(3a+c=0\)。结论②错误"
        ];

        const btnPrev = document.getElementById('scene1-btn-prev');
        const btnNext = document.getElementById('scene1-btn-next');
        const textPanel = document.getElementById('scene1-text-panel');

        function updateScene() {
            // Update buttons
            btnPrev.disabled = currentStep === 0;
            btnNext.disabled = currentStep === totalSteps;

            // Update text
            textPanel.innerHTML = texts[currentStep];
            if (window.MathJax) {
                MathJax.typesetPromise([textPanel]);
            }

            // Hide all step elements
            for (let i = 1; i <= totalSteps; i++) {
                const el = document.getElementById(`scene1-step${i}-elements`);
                if (el) {
                    el.classList.remove('active');
                    // Reset animation for step 2
                    if (i === 2) {
                        const animEl = document.getElementById('scene1-step2-anim');
                        if (animEl) animEl.style.transform = 'translate(0,0)';
                    }
                }
            }

            // Show current step element
            if (currentStep > 0) {
                const el = document.getElementById(`scene1-step${currentStep}-elements`);
                if (el) {
                    el.classList.add('active');
                    if (currentStep === 2) {
                        setTimeout(() => {
                            const animEl = document.getElementById('scene1-step2-anim');
                            if (animEl) animEl.style.transform = 'translate(100px,0)';
                        }, 50);
                    }
                }
            }
        }

        btnPrev.addEventListener('click', () => {
            if (currentStep > 0) {
                currentStep--;
                updateScene();
            }
        });

        btnNext.addEventListener('click', () => {
            if (currentStep < totalSteps) {
                currentStep++;
                updateScene();
            }
        });

        // Initial render
        updateScene();
    })();
    </script>
