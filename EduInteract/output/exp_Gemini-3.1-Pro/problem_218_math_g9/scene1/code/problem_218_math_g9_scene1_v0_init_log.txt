    <script>
    (function() {
        let currentStep = 0;
        const maxStep = 4;

        const btnPrev = document.getElementById('scene1-btn-prev');
        const btnNext = document.getElementById('scene1-btn-next');
        const formulaBox = document.getElementById('scene1-formula');

        const elements = {
            polyAbc: document.getElementById('scene1-poly-abc'),
            polyDef: document.getElementById('scene1-poly-def'),
            polyCdf: document.getElementById('scene1-poly-cdf'),
            lineEf: document.getElementById('scene1-line-ef'),
            angleAcb: document.getElementById('scene1-angle-acb'),
            angleAcf: document.getElementById('scene1-angle-acf'),
            angleF: document.getElementById('scene1-angle-f'),
            angleCdf: document.getElementById('scene1-angle-cdf'),
            textAcb: document.getElementById('scene1-text-acb'),
            textAcf: document.getElementById('scene1-text-acf'),
            textF: document.getElementById('scene1-text-f'),
            textCdf: document.getElementById('scene1-text-cdf')
        };

        const formulas = [
            "已知：<br>$\angle A = \angle EDF = 90^\circ$<br>$AB = AC$<br>$\angle E = 30^\circ$<br>$\angle BCE = 40^\circ$",
            "在等腰 $Rt\triangle ABC$ 中：<br>$\because AB = AC, \angle A = 90^\circ$<br>$\therefore \angle ACB = 45^\circ$",
            "平角计算：<br>$\angle ACF = 180^\circ - \angle BCE - \angle ACB$<br>$\angle ACF = 180^\circ - 40^\circ - 45^\circ = 95^\circ$",
            "在 $Rt\triangle DEF$ 中：<br>$\because \angle EDF = 90^\circ, \angle E = 30^\circ$<br>$\therefore \angle F = 90^\circ - 30^\circ = 60^\circ$",
            "在 $\triangle CDF$ 中：<br>$\angle CDF = 180^\circ - \angle ACF - \angle F$<br>$\angle CDF = 180^\circ - 95^\circ - 60^\circ = 25^\circ$"
        ];

        function updateScene() {
            btnPrev.disabled = currentStep === 0;
            btnNext.disabled = currentStep === maxStep;

            // Reset all
            Object.values(elements).forEach(el => {
                if (el === elements.lineEf) {
                    el.setAttribute('stroke-width', '2');
                    el.setAttribute('stroke', '#000');
                } else {
                    el.style.opacity = '0';
                }
            });

            // Apply step logic
            if (currentStep >= 1) {
                if (currentStep === 1) elements.polyAbc.style.opacity = '0.2';
                elements.angleAcb.style.opacity = '1';
                elements.textAcb.style.opacity = '1';
            }
            if (currentStep >= 2) {
                if (currentStep === 2) {
                    elements.lineEf.setAttribute('stroke-width', '4');
                    elements.lineEf.setAttribute('stroke', '#888');
                }
                elements.angleAcf.style.opacity = '1';
                elements.textAcf.style.opacity = '1';
            }
            if (currentStep >= 3) {
                if (currentStep === 3) elements.polyDef.style.opacity = '0.2';
                elements.angleF.style.opacity = '1';
                elements.textF.style.opacity = '1';
            }
            if (currentStep >= 4) {
                elements.polyCdf.style.opacity = '0.2';
                elements.angleCdf.style.opacity = '1';
                elements.textCdf.style.opacity = '1';
            }

            formulaBox.innerHTML = formulas[currentStep];
            if (window.MathJax) {
                MathJax.typesetPromise([formulaBox]).catch(function (err) {
                    console.error(err.message);
                });
            }
        }

        btnPrev.addEventListener('click', () => {
            if (currentStep > 0) {
                currentStep--;
                updateScene();
            }
        });

        btnNext.addEventListener('click', () => {
            if (currentStep < maxStep) {
                currentStep++;
                updateScene();
            }
        });

        // Initial render
        updateScene();
    })();
    </script>
