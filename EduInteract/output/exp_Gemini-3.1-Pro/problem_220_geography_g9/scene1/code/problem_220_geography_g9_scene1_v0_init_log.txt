    <script>
    (function() {
        let currentStep = 0;
        const maxStep = 4;

        const btnPrev = document.getElementById('scene1-btn-prev');
        const btnNext = document.getElementById('scene1-btn-next');
        const stepText = document.getElementById('scene1-step-text');
        const title = document.getElementById('scene1-title');

        // Elements
        const arrow1 = document.getElementById('scene1-arrow1');
        const arrow2 = document.getElementById('scene1-arrow2');
        const arrow3 = document.getElementById('scene1-arrow3');
        const arrow4 = document.getElementById('scene1-arrow4');

        const labelCool = document.getElementById('scene1-label-cool');
        const labelHeat = document.getElementById('scene1-label-heat');

        const labelAH = document.getElementById('scene1-label-a-h');
        const labelBL = document.getElementById('scene1-label-b-l');
        const labelCH = document.getElementById('scene1-label-c-h');
        const labelDL = document.getElementById('scene1-label-d-l');

        const label30N = document.getElementById('scene1-label-30n');
        const label60N = document.getElementById('scene1-label-60n');
        const labelDynamicDown = document.getElementById('scene1-label-dynamic-down');

        const arrowPolar = document.getElementById('scene1-arrow-polar');
        const labelPolar = document.getElementById('scene1-label-polar');
        const labelWesterlies = document.getElementById('scene1-label-westerlies');
        const polarFront = document.getElementById('scene1-polar-front');
        const labelFront = document.getElementById('scene1-label-front');

        function show(el) {
            el.classList.remove('hidden');
            el.classList.add('visible');
        }

        function hide(el) {
            el.classList.remove('visible');
            el.classList.add('hidden');
        }

        function updateScene() {
            stepText.textContent = `步骤 ${currentStep} / 4`;
            btnPrev.disabled = currentStep === 0;
            btnNext.disabled = currentStep === maxStep;

            // Reset all dynamic elements
            hide(labelCool); hide(labelHeat);
            hide(labelAH); hide(labelBL); hide(labelCH); hide(labelDL);
            hide(label30N); hide(label60N); hide(labelDynamicDown);
            hide(arrowPolar); hide(labelPolar); hide(labelWesterlies);
            hide(polarFront); hide(labelFront);

            // Reset arrows
            arrow1.setAttribute('stroke', '#000000'); arrow1.setAttribute('stroke-width', '2'); arrow1.setAttribute('marker-end', 'url(#arrow-black)');
            arrow2.setAttribute('stroke', '#000000'); arrow2.setAttribute('stroke-width', '2'); arrow2.setAttribute('marker-end', 'url(#arrow-black)');
            arrow3.setAttribute('stroke', '#000000'); arrow3.setAttribute('stroke-width', '2'); arrow3.setAttribute('marker-end', 'url(#arrow-black)');
            arrow4.setAttribute('stroke', '#000000'); arrow4.setAttribute('stroke-width', '2'); arrow4.setAttribute('marker-end', 'url(#arrow-black)');

            if (currentStep === 0) {
                title.textContent = "请点击下一步，逐步推导环流模型";
            } else if (currentStep === 1) {
                title.textContent = "若为热力环流：b处受热膨胀上升，a处冷却收缩下沉。近地面温度：b > a（选项A错误）";
                show(labelCool); show(labelHeat);
                arrow2.setAttribute('stroke', '#c62828'); arrow2.setAttribute('marker-end', 'url(#arrow-red)');
                arrow4.setAttribute('stroke', '#1565c0'); arrow4.setAttribute('marker-end', 'url(#arrow-blue)');
            } else if (currentStep === 2) {
                title.textContent = "若为热力环流：近地面受热处形成低压，冷却处形成高压；高空相反。气压：a > b > c > d（选项B错误）";
                show(labelCool); show(labelHeat);
                show(labelAH); show(labelBL); show(labelCH); show(labelDL);
                arrow2.setAttribute('stroke', '#c62828'); arrow2.setAttribute('marker-end', 'url(#arrow-red)');
                arrow4.setAttribute('stroke', '#1565c0'); arrow4.setAttribute('marker-end', 'url(#arrow-blue)');
                arrow1.setAttribute('stroke-width', '4');
                arrow3.setAttribute('stroke-width', '4');
            } else if (currentStep === 3) {
                title.textContent = "若为北半球中纬环流：a处下沉为30°N，b处上升为60°N。④为动力原因下沉（选项D错误）";
                show(label30N); show(label60N); show(labelDynamicDown);
            } else if (currentStep === 4) {
                title.textContent = "若为北半球中纬环流：60°N处冷暖气流相遇，暖气流爬升形成极锋（选项C正确）";
                show(label30N); show(label60N); show(labelDynamicDown);
                show(arrowPolar); show(labelPolar); show(labelWesterlies);
                show(polarFront); show(labelFront);
                arrow3.setAttribute('stroke', '#c62828'); arrow3.setAttribute('marker-end', 'url(#arrow-red)');
                arrow2.setAttribute('stroke', '#c62828'); arrow2.setAttribute('marker-end', 'url(#arrow-red)');
            }
        }

        btnPrev.addEventListener('click', () => {
            if (currentStep > 0) {
                currentStep--;
                updateScene();
            }
        });

        btnNext.addEventListener('click', () => {
            if (currentStep < maxStep) {
                currentStep++;
                updateScene();
            }
        });

        // Initialize
        updateScene();
    })();
    </script>
