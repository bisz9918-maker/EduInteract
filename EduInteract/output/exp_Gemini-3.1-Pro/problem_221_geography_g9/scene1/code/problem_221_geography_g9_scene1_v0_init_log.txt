    <script>
    (function() {
        let currentStep = 0;
        const maxStep = 4;
        
        const btnPrev = document.getElementById('scene1-btn-prev');
        const btnNext = document.getElementById('scene1-btn-next');
        
        function show(el) {
            if (!el) return;
            el.style.display = '';
            // trigger reflow
            void el.offsetWidth;
            el.style.opacity = '1';
        }
        
        function hide(el) {
            if (!el) return;
            el.style.opacity = '0';
            setTimeout(() => {
                if (el.style.opacity === '0') {
                    el.style.display = 'none';
                }
            }, 300);
        }
        
        function updateScene() {
            btnPrev.disabled = currentStep === 0;
            btnNext.disabled = currentStep === maxStep;
            
            for (let i = 1; i <= maxStep; i++) {
                const el = document.getElementById(`scene1-step${i}`);
                if (i <= currentStep) {
                    show(el);
                } else {
                    hide(el);
                }
            }
        }
        
        btnPrev.addEventListener('click', () => {
            if (currentStep > 0) {
                currentStep--;
                updateScene();
            }
        });
        
        btnNext.addEventListener('click', () => {
            if (currentStep < maxStep) {
                currentStep++;
                updateScene();
            }
        });
        
        // Initialize
        updateScene();
    })();
    </script>
