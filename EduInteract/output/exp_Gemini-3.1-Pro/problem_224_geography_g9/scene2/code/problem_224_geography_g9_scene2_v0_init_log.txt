    <script>
    (function() {
        let currentStep = 0;
        const maxStep = 3;
        
        const btnPrev = document.getElementById('scene2-btn-prev');
        const btnNext = document.getElementById('scene2-btn-next');
        
        const steps = [
            null,
            document.getElementById('scene2-step1'),
            document.getElementById('scene2-step2'),
            document.getElementById('scene2-step3')
        ];

        function updateScene() {
            btnPrev.disabled = currentStep === 0;
            btnNext.disabled = currentStep === maxStep;
            
            for (let i = 1; i <= maxStep; i++) {
                if (steps[i]) {
                    if (i <= currentStep) {
                        steps[i].classList.add('step-active');
                    } else {
                        steps[i].classList.remove('step-active');
                    }
                }
            }
        }

        btnPrev.addEventListener('click', () => {
            if (currentStep > 0) {
                currentStep--;
                updateScene();
            }
        });

        btnNext.addEventListener('click', () => {
            if (currentStep < maxStep) {
                currentStep++;
                updateScene();
            }
        });

        updateScene();
    })();
    </script>
