    <script>
    (function() {
        const slider = document.getElementById('scene3-slider');
        const rayN = document.getElementById('scene3-ray-n');
        const rayM = document.getElementById('scene3-ray-m');
        const rayS = document.getElementById('scene3-ray-s');
        const rayLabel = document.getElementById('scene3-ray-label');
        const angleArc = document.getElementById('scene3-angle-arc');
        const angleText = document.getElementById('scene3-angle-text');
        const infoLat = document.getElementById('scene3-info-lat');
        const infoH = document.getElementById('scene3-info-h');

        // 常量定义
        const R = 160;
        const CX = 200;
        const CY = 220;
        const RAY_LENGTH = 400;
        const A_LAT = -23.5; // A点纬度 23.5°S
        const A_X = 346.7;
        const A_Y = 283.8;
        
        // 地平面倾角 (A点切线角度)
        // A点在圆上的角度为 23.5° (向下为正)
        // 切线角度为 23.5° - 90° = -66.5°
        const PLANE_ANGLE_RAD = (A_LAT - 90) * Math.PI / 180;

        function updateScene() {
            const val = parseInt(slider.value);
            // val: 0 -> 23.5°N, 100 -> 23.5°S
            // lat: 23.5 -> -23.5
            const lat = 23.5 - (val / 100) * 47;
            const latRad = lat * Math.PI / 180;

            // 更新信息面板
            const latStr = lat > 0 ? `${lat.toFixed(1)}°N` : (lat < 0 ? `${Math.abs(lat).toFixed(1)}°S` : `0.0°`);
            infoLat.textContent = `太阳直射点纬度：${latStr}`;

            // 计算光线方向向量 (指向地球)
            // 直射点纬度为 lat，光线从右向左射入
            // 太阳光线与赤道面的夹角为 lat
            // 光线方向向量: dx = -cos(lat), dy = sin(lat) (SVG y轴向下)
            const dx = -Math.cos(latRad);
            const dy = Math.sin(latRad);

            // 更新光线起点
            // 北光线终点: (346.7, 156.2)
            rayN.setAttribute('x1', 346.7 - RAY_LENGTH * dx);
            rayN.setAttribute('y1', 156.2 - RAY_LENGTH * dy);
            
            // 中光线终点: (360, 220)
            const mX1 = 360 - RAY_LENGTH * dx;
            const mY1 = 220 - RAY_LENGTH * dy;
            rayM.setAttribute('x1', mX1);
            rayM.setAttribute('y1', mY1);
            
            // 太阳光线标签位置，根据光线倾角动态调整，避免与光线重叠
            // 放在中光线起点附近，向上偏移
            rayLabel.setAttribute('x', mX1 - 20);
            rayLabel.setAttribute('y', mY1 - 20);

            // 南光线终点: (346.7, 283.8)
            rayS.setAttribute('x1', A_X - RAY_LENGTH * dx);
            rayS.setAttribute('y1', A_Y - RAY_LENGTH * dy);

            // 计算高度角 H
            // H = 90° - |A点纬度 - 直射点纬度|
            const H = 90 - Math.abs(A_LAT - lat);
            infoH.textContent = `A点正午太阳高度角：${H.toFixed(1)}°`;
            angleText.textContent = `H = ${H.toFixed(1)}°`;

            // 绘制高度角圆弧
            // 起点：地平面方向 (向右上方)
            // 终点：光线反方向 (向右上方)
            const arcRadius = 40;
            
            // 地平面方向向量 (向右上方)
            // 角度为 -66.5°
            const planeDx = Math.cos(PLANE_ANGLE_RAD);
            const planeDy = Math.sin(PLANE_ANGLE_RAD);
            
            const startX = A_X + arcRadius * planeDx;
            const startY = A_Y + arcRadius * planeDy;

            // 光线反方向向量
            // 角度为 -lat
            const rayRevRad = -latRad;
            const rayRevDx = Math.cos(rayRevRad);
            const rayRevDy = Math.sin(rayRevRad);
            
            const endX = A_X + arcRadius * rayRevDx;
            const endY = A_Y + arcRadius * rayRevDy;

            // 绘制圆弧
            // sweep-flag: 1 (顺时针，从地平面到光线)
            angleArc.setAttribute('d', `M ${startX} ${startY} A ${arcRadius} ${arcRadius} 0 0 1 ${endX} ${endY}`);

            // 更新文本位置 (角平分线方向)
            const midRad = (PLANE_ANGLE_RAD + rayRevRad) / 2;
            const textDist = 105; // 进一步增加距离避免与光线重叠
            angleText.setAttribute('x', A_X + textDist * Math.cos(midRad) - 20);
            angleText.setAttribute('y', A_Y + textDist * Math.sin(midRad) + 5);
        }

        slider.addEventListener('input', updateScene);
        updateScene(); // 初始化
    })();
    </script>
