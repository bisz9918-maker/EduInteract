    <script>
    (function() {
        let currentStep = 0;
        const maxStep = 4;

        const btnPrev = document.getElementById('scene2-btn-prev');
        const btnNext = document.getElementById('scene2-btn-next');

        const jiaFill = document.getElementById('scene2-jia-fill');
        const jiaPointer = document.getElementById('scene2-jia-pointer');
        const jiaText = document.getElementById('scene2-jia-text');

        const yiFill = document.getElementById('scene2-yi-fill');
        const yiPiston = document.getElementById('scene2-yi-piston');
        const yiWeight = document.getElementById('scene2-yi-weight');
        const yiText = document.getElementById('scene2-yi-text');

        const step2Group = document.getElementById('scene2-step2-group');
        const step3Group = document.getElementById('scene2-step3-group');
        const step4Group = document.getElementById('scene2-step4-group');

        function updateStep(step) {
            // 步骤0：初始状态
            if (step === 0) {
                jiaFill.setAttribute('fill', '#c62828');
                jiaFill.setAttribute('opacity', '0.1');
                jiaPointer.setAttribute('transform', 'rotate(0, 240, 70)');
                jiaText.textContent = '0.2 mol NO₂';

                yiFill.setAttribute('y', '98');
                yiFill.setAttribute('height', '152');
                yiFill.setAttribute('fill', '#c62828');
                yiFill.setAttribute('opacity', '0.1');
                
                yiPiston.setAttribute('y', '90');
                yiWeight.setAttribute('y', '74');
                
                yiText.style.top = '170px';
                yiText.textContent = '0.2 mol NO₂';

                step2Group.style.opacity = '0';
                step3Group.style.opacity = '0';
                step4Group.style.opacity = '0';
            }
            // 步骤1：反应进行
            if (step >= 1) {
                jiaFill.setAttribute('fill', '#1565c0');
                jiaFill.setAttribute('opacity', '0.1');
                jiaPointer.setAttribute('transform', 'rotate(-30, 240, 70)');
                jiaText.textContent = '平衡状态 A';

                yiFill.setAttribute('y', '138');
                yiFill.setAttribute('height', '112');
                yiFill.setAttribute('fill', '#1565c0');
                yiFill.setAttribute('opacity', '0.15');
                
                yiPiston.setAttribute('y', '130');
                yiWeight.setAttribute('y', '114');
                
                yiText.style.top = '194px';
                yiText.textContent = '平衡状态 B';
            }
            
            // 步骤2：等效加压
            if (step >= 2) {
                step2Group.style.opacity = '1';
            } else {
                step2Group.style.opacity = '0';
            }

            // 步骤3：平衡移动
            if (step >= 3) {
                step3Group.style.opacity = '1';
            } else {
                step3Group.style.opacity = '0';
            }

            // 步骤4：浓度对比
            if (step >= 4) {
                step4Group.style.opacity = '1';
            } else {
                step4Group.style.opacity = '0';
            }

            btnPrev.disabled = step === 0;
            btnNext.disabled = step === maxStep;
        }

        btnPrev.addEventListener('click', () => {
            if (currentStep > 0) {
                currentStep--;
                updateStep(currentStep);
            }
        });

        btnNext.addEventListener('click', () => {
            if (currentStep < maxStep) {
                currentStep++;
                updateStep(currentStep);
            }
        });

        // 初始化
        updateStep(0);
    })();
    </script>
