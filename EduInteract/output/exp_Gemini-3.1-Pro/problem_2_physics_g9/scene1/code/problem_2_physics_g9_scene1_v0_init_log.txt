    <script>
    (function() {
        const btnPrev = document.getElementById('scene1-btn-prev');
        const btnNext = document.getElementById('scene1-btn-next');
        
        let currentStep = 0;
        const maxStep = 3;
        
        const cartGroup = document.getElementById('scene1-cart-group');
        const cartLabel = document.getElementById('scene1-cart-label');
        const cartV = document.getElementById('scene1-cart-v');
        const blockGroup = document.getElementById('scene1-block-group');
        const blockV = document.getElementById('scene1-block-v');
        const forces = document.getElementById('scene1-forces');
        const smoothLabel = document.getElementById('scene1-smooth-label');
        
        const barPe = document.getElementById('scene1-bar-pe');
        const barKe = document.getElementById('scene1-bar-ke');
        const barMe = document.getElementById('scene1-bar-me');
        
        const hint1 = document.getElementById('scene1-hint1');
        const hint2 = document.getElementById('scene1-hint2');
        const hint3 = document.getElementById('scene1-hint3');
        
        function show(el) {
            el.style.opacity = 1;
            el.style.display = '';
        }
        function hide(el) {
            el.style.opacity = 0;
            el.style.display = 'none';
        }
        
        let animFrame;
        function animate(duration, draw, onComplete) {
            let start = performance.now();
            cancelAnimationFrame(animFrame);
            function step(time) {
                let progress = (time - start) / duration;
                if (progress > 1) progress = 1;
                draw(progress);
                if (progress < 1) {
                    animFrame = requestAnimationFrame(step);
                } else if (onComplete) {
                    onComplete();
                }
            }
            animFrame = requestAnimationFrame(step);
        }
        
        function updateState(animateTransition = true) {
            btnPrev.disabled = currentStep === 0;
            btnNext.disabled = currentStep === maxStep;
            
            hide(hint1); hide(hint2); hide(hint3);
            hide(forces); hide(smoothLabel);
            hide(cartV); hide(blockV);
            
            if (currentStep === 0) {
                cancelAnimationFrame(animFrame);
                cartGroup.setAttribute('transform', 'translate(100, 170) rotate(45)');
                cartLabel.setAttribute('transform', 'rotate(-45, 0, -75)');
                blockGroup.setAttribute('transform', 'translate(380, 320)');
                barPe.setAttribute('y', '140'); barPe.setAttribute('height', '180');
                barKe.setAttribute('y', '320'); barKe.setAttribute('height', '0');
                barMe.setAttribute('y', '140'); barMe.setAttribute('height', '180');
            } else if (currentStep === 1) {
                if (animateTransition) {
                    animate(1000, p => {
                        let x = 100 + 150 * p;
                        let y = 170 + 150 * p;
                        let angle = p < 1 ? 45 : 0;
                        cartGroup.setAttribute('transform', `translate(${x}, ${y}) rotate(${angle})`);
                        cartLabel.setAttribute('transform', `rotate(${-angle}, 0, -75)`);
                        
                        let peH = 180 * (1 - p);
                        let keH = 160 * p;
                        let meH = 180 - 20 * p;
                        barPe.setAttribute('y', 320 - peH); barPe.setAttribute('height', peH);
                        barKe.setAttribute('y', 320 - keH); barKe.setAttribute('height', keH);
                        barMe.setAttribute('y', 320 - meH); barMe.setAttribute('height', meH);
                    }, () => {
                        show(cartV);
                        show(hint1);
                    });
                } else {
                    cartGroup.setAttribute('transform', 'translate(250, 320) rotate(0)');
                    cartLabel.setAttribute('transform', 'rotate(0, 0, -75)');
                    barPe.setAttribute('y', '320'); barPe.setAttribute('height', '0');
                    barKe.setAttribute('y', '160'); barKe.setAttribute('height', '160');
                    barMe.setAttribute('y', '160'); barMe.setAttribute('height', '160');
                    show(cartV);
                    show(hint1);
                }
                blockGroup.setAttribute('transform', 'translate(380, 320)');
            } else if (currentStep === 2) {
                cancelAnimationFrame(animFrame);
                cartGroup.setAttribute('transform', 'translate(250, 320) rotate(0)');
                cartLabel.setAttribute('transform', 'rotate(0, 0, -75)');
                blockGroup.setAttribute('transform', 'translate(380, 320)');
                barPe.setAttribute('y', '320'); barPe.setAttribute('height', '0');
                barKe.setAttribute('y', '160'); barKe.setAttribute('height', '160');
                barMe.setAttribute('y', '160'); barMe.setAttribute('height', '160');
                show(forces);
                show(hint2);
            } else if (currentStep === 3) {
                barPe.setAttribute('y', '320'); barPe.setAttribute('height', '0');
                barKe.setAttribute('y', '160'); barKe.setAttribute('height', '160');
                barMe.setAttribute('y', '160'); barMe.setAttribute('height', '160');
                show(smoothLabel);
                
                if (animateTransition) {
                    animate(1500, p => {
                        if (p < 0.33) {
                            let p1 = p / 0.33;
                            let cx = 250 + 70 * p1;
                            cartGroup.setAttribute('transform', `translate(${cx}, 320) rotate(0)`);
                            cartLabel.setAttribute('transform', 'rotate(0, 0, -75)');
                            blockGroup.setAttribute('transform', 'translate(380, 320)');
                        } else {
                            let p2 = (p - 0.33) / 0.67;
                            cartGroup.setAttribute('transform', 'translate(320, 320) rotate(0)');
                            let bx = 380 + 150 * p2;
                            blockGroup.setAttribute('transform', `translate(${bx}, 320)`);
                        }
                    }, () => {
                        show(blockV);
                        show(hint3);
                    });
                } else {
                    cartGroup.setAttribute('transform', 'translate(320, 320) rotate(0)');
                    cartLabel.setAttribute('transform', 'rotate(0, 0, -75)');
                    blockGroup.setAttribute('transform', 'translate(530, 320)');
                    show(blockV);
                    show(hint3);
                }
            }
        }
        
        btnPrev.addEventListener('click', () => {
            if (currentStep > 0) {
                currentStep--;
                updateState(true);
            }
        });
        
        btnNext.addEventListener('click', () => {
            if (currentStep < maxStep) {
                currentStep++;
                updateState(true);
            }
        });
        
        updateState(false);
    })();
    </script>
