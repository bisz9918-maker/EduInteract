    <script>
    (function() {
        const btnPrev = document.getElementById('scene1-btn-prev');
        const btnNext = document.getElementById('scene1-btn-next');
        const valveC = document.getElementById('scene1-valve-c');
        const valveA = document.getElementById('scene1-valve-a');
        const smoke = document.getElementById('scene1-smoke');
        const smokeLabel = document.getElementById('scene1-smoke-label');
        const smokeLine = document.getElementById('scene1-smoke-line');
        const hclContainer = document.getElementById('scene1-molecules-hcl');
        const nh3Container = document.getElementById('scene1-molecules-nh3');
        
        let currentStep = 0;
        const maxStep = 3;
        
        // 分子数据
        const numMolecules = 15;
        const hclMolecules = [];
        const nh3Molecules = [];
        
        function initMolecules() {
            for (let i = 0; i < numMolecules; i++) {
                // HCl
                const hcl = document.createElementNS('http://www.w3.org/2000/svg', 'circle');
                hcl.setAttribute('r', '4');
                hcl.setAttribute('fill', '#2e7d32');
                hclContainer.appendChild(hcl);
                hclMolecules.push({ el: hcl, type: 'hcl', state: 'flask', x: 250, y: 120, targetX: 250, targetY: 120, pathIndex: 0 });
                
                // NH3
                const nh3 = document.createElementNS('http://www.w3.org/2000/svg', 'circle');
                nh3.setAttribute('r', '4');
                nh3.setAttribute('fill', '#1565c0');
                nh3Container.appendChild(nh3);
                nh3Molecules.push({ el: nh3, type: 'nh3', state: 'flask', x: 550, y: 120, targetX: 550, targetY: 120, pathIndex: 0 });
            }
            resetMolecules();
        }
        
        function resetMolecules() {
            hclMolecules.forEach(m => {
                m.state = 'flask';
                const r = Math.random() * 50;
                const theta = Math.random() * 2 * Math.PI;
                m.x = 250 + r * Math.cos(theta);
                m.y = 120 + r * Math.sin(theta);
                m.el.setAttribute('cx', m.x);
                m.el.setAttribute('cy', m.y);
            });
            nh3Molecules.forEach(m => {
                m.state = 'flask';
                const r = Math.random() * 50;
                const theta = Math.random() * 2 * Math.PI;
                m.x = 550 + r * Math.cos(theta);
                m.y = 120 + r * Math.sin(theta);
                m.el.setAttribute('cx', m.x);
                m.el.setAttribute('cy', m.y);
            });
        }
        
        initMolecules();
        
        function updateScene() {
            btnPrev.disabled = currentStep === 0;
            btnNext.disabled = currentStep === maxStep;
            
            // 步骤1：活塞旋转
            if (currentStep >= 1) {
                valveC.style.transform = 'rotate(90deg)';
                valveA.style.transform = 'rotate(90deg)';
            } else {
                valveC.style.transform = 'rotate(0deg)';
                valveA.style.transform = 'rotate(0deg)';
            }
            
            // 步骤2：分子扩散 (动画逻辑在 requestAnimationFrame 中处理)
            if (currentStep === 2) {
                startDiffusion();
            } else if (currentStep < 2) {
                resetMolecules();
            }
            
            // 步骤3：白烟
            if (currentStep >= 3) {
                smoke.style.opacity = '1';
                smokeLabel.style.opacity = '1';
                smokeLine.style.opacity = '1';
            } else {
                smoke.style.opacity = '0';
                smokeLabel.style.opacity = '0';
                smokeLine.style.opacity = '0';
            }
        }
        
        function startDiffusion() {
            // 设置分子的目标路径
            hclMolecules.forEach(m => {
                m.state = 'diffusing';
                m.path = [
                    {x: 250, y: 280},
                    {x: 350, y: 280}
                ];
                m.pathIndex = 0;
            });
            nh3Molecules.forEach(m => {
                m.state = 'diffusing';
                m.path = [
                    {x: 540, y: 280},
                    {x: 350, y: 280}
                ];
                m.pathIndex = 0;
            });
        }
        
        function animate() {
            if (currentStep < 2) {
                // 布朗运动
                hclMolecules.forEach(m => {
                    if (m.state === 'flask') {
                        m.x += (Math.random() - 0.5) * 2;
                        m.y += (Math.random() - 0.5) * 2;
                        // 限制在圆内
                        const dx = m.x - 250;
                        const dy = m.y - 120;
                        if (dx*dx + dy*dy > 2500) {
                            m.x = 250 + dx * 0.9;
                            m.y = 120 + dy * 0.9;
                        }
                        m.el.setAttribute('cx', m.x);
                        m.el.setAttribute('cy', m.y);
                    }
                });
                nh3Molecules.forEach(m => {
                    if (m.state === 'flask') {
                        m.x += (Math.random() - 0.5) * 2;
                        m.y += (Math.random() - 0.5) * 2;
                        const dx = m.x - 550;
                        const dy = m.y - 120;
                        if (dx*dx + dy*dy > 2500) {
                            m.x = 550 + dx * 0.9;
                            m.y = 120 + dy * 0.9;
                        }
                        m.el.setAttribute('cx', m.x);
                        m.el.setAttribute('cy', m.y);
                    }
                });
            } else if (currentStep === 2) {
                // 扩散运动
                const hclSpeed = 1;
                const nh3Speed = 1.5;
                
                hclMolecules.forEach(m => {
                    if (m.state === 'diffusing' && m.pathIndex < m.path.length) {
                        const target = m.path[m.pathIndex];
                        const dx = target.x - m.x;
                        const dy = target.y - m.y;
                        const dist = Math.sqrt(dx*dx + dy*dy);
                        if (dist < hclSpeed) {
                            m.x = target.x;
                            m.y = target.y;
                            m.pathIndex++;
                        } else {
                            m.x += (dx / dist) * hclSpeed;
                            m.y += (dy / dist) * hclSpeed;
                        }
                        m.el.setAttribute('cx', m.x);
                        m.el.setAttribute('cy', m.y);
                    }
                });
                
                nh3Molecules.forEach(m => {
                    if (m.state === 'diffusing' && m.pathIndex < m.path.length) {
                        const target = m.path[m.pathIndex];
                        const dx = target.x - m.x;
                        const dy = target.y - m.y;
                        const dist = Math.sqrt(dx*dx + dy*dy);
                        if (dist < nh3Speed) {
                            m.x = target.x;
                            m.y = target.y;
                            m.pathIndex++;
                        } else {
                            m.x += (dx / dist) * nh3Speed;
                            m.y += (dy / dist) * nh3Speed;
                        }
                        m.el.setAttribute('cx', m.x);
                        m.el.setAttribute('cy', m.y);
                    }
                });
            }
            
            requestAnimationFrame(animate);
        }
        
        btnPrev.addEventListener('click', () => {
            if (currentStep > 0) {
                currentStep--;
                updateScene();
            }
        });
        
        btnNext.addEventListener('click', () => {
            if (currentStep < maxStep) {
                currentStep++;
                updateScene();
            }
        });
        
        updateScene();
        animate();
    })();
    </script>
