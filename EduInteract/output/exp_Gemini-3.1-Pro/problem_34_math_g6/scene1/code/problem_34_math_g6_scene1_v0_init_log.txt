    <script>
    (function() {
        let currentStep = 0;
        const maxStep = 4;
        
        const btnPrev = document.getElementById('scene1-btn-prev');
        const btnNext = document.getElementById('scene1-btn-next');
        
        const perimeterHighlight = document.getElementById('scene1-perimeter-highlight');
        const movingLine = document.getElementById('scene1-moving-line');
        const dashedOrig = document.getElementById('scene1-dashed-orig');
        const dashedPath1 = document.getElementById('scene1-dashed-path1');
        const dashedPath2 = document.getElementById('scene1-dashed-path2');
        const rectHighlight = document.getElementById('scene1-rect-highlight');
        const extraLine1 = document.getElementById('scene1-extra-line1');
        const extraLine2 = document.getElementById('scene1-extra-line2');
        const formula1 = document.getElementById('scene1-formula1');
        const formula2 = document.getElementById('scene1-formula2');
        const formula3 = document.getElementById('scene1-formula3');
        
        function updateScene() {
            btnPrev.disabled = currentStep === 0;
            btnNext.disabled = currentStep === maxStep;
            
            // Reset all
            perimeterHighlight.style.opacity = '0';
            movingLine.style.opacity = '0';
            movingLine.style.transform = 'translateX(0)';
            dashedOrig.style.opacity = '0';
            dashedPath1.style.opacity = '0';
            dashedPath2.style.opacity = '0';
            rectHighlight.style.opacity = '0';
            extraLine1.style.opacity = '0';
            extraLine2.style.opacity = '0';
            formula1.style.opacity = '0';
            formula2.style.opacity = '0';
            formula3.style.opacity = '0';
            
            if (currentStep >= 1) {
                if (currentStep === 1) {
                    perimeterHighlight.style.opacity = '1';
                }
            }
            
            if (currentStep >= 2) {
                movingLine.style.opacity = '1';
                dashedOrig.style.opacity = '1';
                dashedPath1.style.opacity = '1';
                dashedPath2.style.opacity = '1';
                
                // Trigger translation
                setTimeout(() => {
                    if (currentStep >= 2) {
                        movingLine.style.transform = 'translateX(100px)';
                    }
                }, 50);
            }
            
            if (currentStep >= 3) {
                movingLine.style.opacity = '0';
                rectHighlight.style.opacity = '1';
                extraLine1.style.opacity = '1';
                extraLine2.style.opacity = '1';
            }
            
            if (currentStep >= 4) {
                formula1.style.opacity = '1';
                setTimeout(() => { if (currentStep >= 4) formula2.style.opacity = '1'; }, 500);
                setTimeout(() => { if (currentStep >= 4) formula3.style.opacity = '1'; }, 1000);
            }
        }
        
        btnPrev.addEventListener('click', () => {
            if (currentStep > 0) {
                currentStep--;
                updateScene();
            }
        });
        
        btnNext.addEventListener('click', () => {
            if (currentStep < maxStep) {
                currentStep++;
                updateScene();
            }
        });
        
        updateScene();
    })();
    </script>
