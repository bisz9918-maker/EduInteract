    <script>
    (function() {
        // Generate grid
        const gridGroup = document.getElementById('scene1-grid');
        for (let i = 0; i <= 12; i++) {
            const x = 160 + i * 40;
            const line = document.createElementNS('http://www.w3.org/2000/svg', 'line');
            line.setAttribute('x1', x);
            line.setAttribute('y1', 70);
            line.setAttribute('x2', x);
            line.setAttribute('y2', 430);
            gridGroup.appendChild(line);
        }
        for (let i = 0; i <= 9; i++) {
            const y = 430 - i * 40;
            const line = document.createElementNS('http://www.w3.org/2000/svg', 'line');
            line.setAttribute('x1', 160);
            line.setAttribute('y1', y);
            line.setAttribute('x2', 640);
            line.setAttribute('y2', y);
            gridGroup.appendChild(line);
        }

        // Generate axis numbers
        const axisNumbersGroup = document.getElementById('scene1-axis-numbers');
        for (let i = 1; i <= 12; i++) {
            const x = 160 + i * 40;
            const text = document.createElementNS('http://www.w3.org/2000/svg', 'text');
            text.setAttribute('x', x);
            text.setAttribute('y', 445);
            text.setAttribute('text-anchor', 'middle');
            text.textContent = i;
            axisNumbersGroup.appendChild(text);
        }
        for (let i = 1; i <= 8; i++) { // Changed to 8 to avoid overlapping with 'y'
            const y = 430 - i * 40;
            const text = document.createElementNS('http://www.w3.org/2000/svg', 'text');
            text.setAttribute('x', 145);
            text.setAttribute('y', y + 5);
            text.setAttribute('text-anchor', 'end');
            text.textContent = i;
            axisNumbersGroup.appendChild(text);
        }

        // State management
        let currentStep = 0;
        const maxStep = 3;
        
        const btnPrev = document.getElementById('scene1-btn-prev');
        const btnNext = document.getElementById('scene1-btn-next');
        
        const flagGroup = document.getElementById('scene1-flag-group');
        const flagpole = document.getElementById('scene1-flagpole');
        const flagSurface = document.getElementById('scene1-flag-surface');
        const pointO = document.getElementById('scene1-point-o');
        const labelO = document.getElementById('scene1-label-o');
        const labelOCoord = document.getElementById('scene1-label-o-coord');
        const lengthGroup = document.getElementById('scene1-length-group');
        const flagShadow = document.getElementById('scene1-flag-shadow');
        const rotationGroup = document.getElementById('scene1-rotation-group');
        const pointAPrimeGroup = document.getElementById('scene1-point-a-prime-group');
        
        let rotationTimeout = null;
        let aPrimeTimeout = null;

        function updateScene() {
            btnPrev.disabled = currentStep === 0;
            btnNext.disabled = currentStep === maxStep;
            
            clearTimeout(rotationTimeout);
            clearTimeout(aPrimeTimeout);

            // Step 0: Initial
            if (currentStep === 0) {
                flagGroup.style.transform = 'rotate(0deg)';
                flagpole.setAttribute('stroke', '#000000');
                flagSurface.setAttribute('stroke', '#000000');
                flagSurface.setAttribute('fill', '#000000');
                pointO.setAttribute('fill', '#000000');
                labelO.classList.remove('scene1-hidden');
                labelOCoord.classList.add('scene1-hidden');
                lengthGroup.classList.add('scene1-hidden');
                flagShadow.classList.add('scene1-hidden');
                rotationGroup.classList.add('scene1-hidden');
                pointAPrimeGroup.classList.add('scene1-hidden');
            }

            // Step 1: Highlight O and length
            if (currentStep === 1) {
                flagGroup.style.transform = 'rotate(0deg)';
                flagpole.setAttribute('stroke', '#c62828');
                flagSurface.setAttribute('stroke', '#000000');
                flagSurface.setAttribute('fill', '#000000');
                pointO.setAttribute('fill', '#c62828');
                labelO.classList.add('scene1-hidden');
                labelOCoord.classList.remove('scene1-hidden');
                lengthGroup.classList.remove('scene1-hidden');
                flagShadow.classList.add('scene1-hidden');
                rotationGroup.classList.add('scene1-hidden');
                pointAPrimeGroup.classList.add('scene1-hidden');
            }

            // Step 2: Rotate
            if (currentStep === 2) {
                flagShadow.classList.remove('scene1-hidden');
                flagGroup.style.transform = 'rotate(90deg)';
                flagpole.setAttribute('stroke', '#1565c0');
                flagSurface.setAttribute('stroke', '#1565c0');
                flagSurface.setAttribute('fill', '#1565c0');
                pointO.setAttribute('fill', '#c62828');
                labelO.classList.add('scene1-hidden');
                labelOCoord.classList.remove('scene1-hidden');
                lengthGroup.classList.remove('scene1-hidden');
                pointAPrimeGroup.classList.add('scene1-hidden');
                
                rotationTimeout = setTimeout(() => {
                    rotationGroup.classList.remove('scene1-hidden');
                }, 1000);
            }

            // Step 3: Show A'
            if (currentStep === 3) {
                flagShadow.classList.remove('scene1-hidden');
                flagGroup.style.transform = 'rotate(90deg)';
                flagpole.setAttribute('stroke', '#1565c0');
                flagSurface.setAttribute('stroke', '#1565c0');
                flagSurface.setAttribute('fill', '#1565c0');
                pointO.setAttribute('fill', '#c62828');
                labelO.classList.add('scene1-hidden');
                labelOCoord.classList.remove('scene1-hidden');
                lengthGroup.classList.remove('scene1-hidden');
                rotationGroup.classList.remove('scene1-hidden');
                
                aPrimeTimeout = setTimeout(() => {
                    pointAPrimeGroup.classList.remove('scene1-hidden');
                }, 300); // Small delay for visual effect
            }
        }

        btnPrev.addEventListener('click', () => {
            if (currentStep > 0) {
                currentStep--;
                updateScene();
            }
        });

        btnNext.addEventListener('click', () => {
            if (currentStep < maxStep) {
                currentStep++;
                updateScene();
            }
        });

        // Initialize
        updateScene();
    })();
    </script>
