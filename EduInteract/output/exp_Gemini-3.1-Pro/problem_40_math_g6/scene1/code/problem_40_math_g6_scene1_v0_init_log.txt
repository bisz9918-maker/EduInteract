    <script>
    (function() {
        let currentStep = 0;
        const maxStep = 4;

        const btnPrev = document.getElementById('scene1-btn-prev');
        const btnNext = document.getElementById('scene1-btn-next');
        const netGroup = document.getElementById('scene1-net-group');
        const title = document.getElementById('scene1-title');
        const conclusion = document.getElementById('scene1-conclusion');
        const lineTop = document.getElementById('scene1-line-top');
        const lineBottom = document.getElementById('scene1-line-bottom');

        const sq0 = document.getElementById('scene1-sq0');
        const sq1 = document.getElementById('scene1-sq1');
        const sq2 = document.getElementById('scene1-sq2');
        const sq3 = document.getElementById('scene1-sq3');
        const sq4 = document.getElementById('scene1-sq4');
        const sq5 = document.getElementById('scene1-sq5');

        const hl0 = document.getElementById('scene1-hl-0');
        const hl1 = document.getElementById('scene1-hl-1');
        const hl2 = document.getElementById('scene1-hl-2');
        const hl3 = document.getElementById('scene1-hl-3');
        const hl4 = document.getElementById('scene1-hl-4');
        const hl5 = document.getElementById('scene1-hl-5');

        const patHeart = '#scene1-pat-heart';
        const patSmile = '#scene1-pat-smile';
        const patStar = '#scene1-pat-star';

        function updateScene() {
            btnPrev.disabled = currentStep === 0;
            btnNext.disabled = currentStep === maxStep;

            // Reset highlights and lines
            [hl0, hl1, hl2, hl3, hl4, hl5].forEach(hl => hl.style.opacity = '0');
            lineTop.style.display = 'none';
            lineBottom.style.display = 'none';

            if (currentStep === 0) {
                netGroup.style.opacity = '0';
                conclusion.textContent = '点击下一步开始验证选项';
                conclusion.setAttribute('fill', '#333');
            } else {
                netGroup.style.opacity = '1';
                
                if (currentStep === 1) {
                    title.textContent = '选项 A';
                    sq0.setAttribute('href', patStar);
                    sq1.setAttribute('href', patHeart);
                    sq2.setAttribute('href', patSmile);
                    sq3.setAttribute('href', patHeart);
                    sq4.setAttribute('href', patSmile);
                    sq5.setAttribute('href', patStar);
                    
                    hl1.style.opacity = '0.3'; hl3.style.opacity = '0.3';
                    hl2.style.opacity = '0.3'; hl4.style.opacity = '0.3';
                    hl0.style.opacity = '0.3'; hl5.style.opacity = '0.3';
                    
                    conclusion.textContent = '结论：A选项三组相对面图案均相同，符合题意。';
                    conclusion.setAttribute('fill', '#2e7d32');
                } else if (currentStep === 2) {
                    title.textContent = '选项 B';
                    sq0.setAttribute('href', patSmile);
                    sq1.setAttribute('href', patHeart);
                    sq2.setAttribute('href', patHeart);
                    sq3.setAttribute('href', patStar);
                    sq4.setAttribute('href', patStar);
                    sq5.setAttribute('href', patSmile);
                    
                    hl1.style.opacity = '0.3'; hl3.style.opacity = '0.3';
                    lineTop.style.display = 'block';
                    
                    conclusion.textContent = '结论：心形和星形相对，图案不同，不符合题意。';
                    conclusion.setAttribute('fill', '#d32f2f');
                } else if (currentStep === 3) {
                    title.textContent = '选项 C';
                    sq0.setAttribute('href', patSmile);
                    sq1.setAttribute('href', patHeart);
                    sq2.setAttribute('href', patSmile);
                    sq3.setAttribute('href', patHeart);
                    sq4.setAttribute('href', patStar);
                    sq5.setAttribute('href', patStar);
                    
                    hl2.style.opacity = '0.3'; hl4.style.opacity = '0.3';
                    lineBottom.style.display = 'block';
                    
                    conclusion.textContent = '结论：笑脸和星形相对，图案不同，不符合题意。';
                    conclusion.setAttribute('fill', '#d32f2f');
                } else if (currentStep === 4) {
                    title.textContent = '选项 D';
                    sq0.setAttribute('href', patSmile);
                    sq1.setAttribute('href', patHeart);
                    sq2.setAttribute('href', patStar);
                    sq3.setAttribute('href', patSmile);
                    sq4.setAttribute('href', patStar);
                    sq5.setAttribute('href', patHeart);
                    
                    hl1.style.opacity = '0.3'; hl3.style.opacity = '0.3';
                    lineTop.style.display = 'block';
                    
                    conclusion.textContent = '结论：心形和笑脸相对，图案不同，不符合题意。';
                    conclusion.setAttribute('fill', '#d32f2f');
                }
            }
        }

        btnPrev.addEventListener('click', () => {
            if (currentStep > 0) {
                currentStep--;
                updateScene();
            }
        });

        btnNext.addEventListener('click', () => {
            if (currentStep < maxStep) {
                currentStep++;
                updateScene();
            }
        });

        // Initialize
        updateScene();
    })();
    </script>
