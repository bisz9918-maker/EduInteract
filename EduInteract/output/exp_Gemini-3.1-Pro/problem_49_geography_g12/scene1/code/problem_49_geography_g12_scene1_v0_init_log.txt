    <script>
    (function() {
        let currentStep = 0;
        const maxStep = 4;
        
        const btnPrev = document.getElementById('scene1-btn-prev');
        const btnNext = document.getElementById('scene1-btn-next');
        
        const lineB = document.getElementById('scene1-line-B');
        const line180 = document.getElementById('scene1-line-180');
        const line60E = document.getElementById('scene1-line-60E');
        const lineMA = document.getElementById('scene1-line-MA');
        const lineAA = document.getElementById('scene1-line-AA');
        const nightMask = document.getElementById('scene1-night-mask');
        const polarDay = document.getElementById('scene1-polar-day');
        const pointM = document.getElementById('scene1-point-M');
        const pointSubsolar = document.getElementById('scene1-point-subsolar');
        const rotation = document.getElementById('scene1-rotation');
        
        const labelB = document.getElementById('scene1-label-B');
        const label180 = document.getElementById('scene1-label-180');
        const longLabels = document.getElementById('scene1-long-labels');
        const label12h = document.getElementById('scene1-label-12h');
        const labelPolarDay = document.getElementById('scene1-label-polar-day');
        const labelSubsolar = document.getElementById('scene1-label-subsolar');
        
        function updateState() {
            btnPrev.disabled = currentStep === 0;
            btnNext.disabled = currentStep === maxStep;
            
            // Reset all
            line180.setAttribute('stroke', '#000000');
            line180.setAttribute('stroke-width', '1');
            lineB.setAttribute('stroke', '#000000');
            lineB.setAttribute('stroke-width', '1');
            line60E.setAttribute('stroke', '#000000');
            line60E.setAttribute('stroke-width', '1');
            line60E.removeAttribute('stroke-dasharray');
            lineMA.classList.remove('scene1-hidden');
            
            rotation.classList.add('scene1-hidden');
            label180.classList.add('scene1-hidden');
            
            pointM.classList.add('scene1-hidden');
            pointM.classList.remove('scene1-pulsing');
            labelB.textContent = 'B';
            labelB.setAttribute('fill', '#000000');
            
            longLabels.classList.add('scene1-hidden');
            label12h.classList.add('scene1-hidden');
            
            lineAA.classList.add('scene1-hidden');
            nightMask.style.opacity = '0';
            polarDay.style.opacity = '0';
            labelPolarDay.classList.add('scene1-hidden');
            pointSubsolar.classList.add('scene1-hidden');
            labelSubsolar.classList.add('scene1-hidden');
            
            if (currentStep >= 1) {
                rotation.classList.remove('scene1-hidden');
                line180.setAttribute('stroke', '#1565c0');
                line180.setAttribute('stroke-width', '3');
                lineB.setAttribute('stroke', '#1565c0');
                lineB.setAttribute('stroke-width', '3');
                label180.classList.remove('scene1-hidden');
                labelB.textContent = '0时';
                labelB.setAttribute('fill', '#1565c0');
            }
            
            if (currentStep >= 2) {
                pointM.classList.remove('scene1-hidden');
                pointM.classList.add('scene1-pulsing');
                labelB.textContent = '0时(120°W)';
            }
            
            if (currentStep >= 3) {
                longLabels.classList.remove('scene1-hidden');
                line60E.setAttribute('stroke', '#c62828');
                line60E.setAttribute('stroke-width', '3');
                line60E.setAttribute('stroke-dasharray', '5,5');
                label12h.classList.remove('scene1-hidden');
            }
            
            if (currentStep >= 4) {
                lineMA.classList.add('scene1-hidden');
                lineAA.classList.remove('scene1-hidden');
                nightMask.style.opacity = '0.2';
                polarDay.style.opacity = '0.2';
                labelPolarDay.classList.remove('scene1-hidden');
                pointSubsolar.classList.remove('scene1-hidden');
                labelSubsolar.classList.remove('scene1-hidden');
            }
        }
        
        btnPrev.addEventListener('click', () => {
            if (currentStep > 0) {
                currentStep--;
                updateState();
            }
        });
        
        btnNext.addEventListener('click', () => {
            if (currentStep < maxStep) {
                currentStep++;
                updateState();
            }
        });
        
        // Initial state
        updateState();
    })();
    </script>
