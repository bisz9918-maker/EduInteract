    <script>
    (function() {
        let currentStep = 0;
        const maxStep = 3;

        const btnPrev = document.getElementById('scene1-btn-prev');
        const btnNext = document.getElementById('scene1-btn-next');
        
        const switchBlade = document.getElementById('scene1-switch-blade');
        const wireB = document.getElementById('scene1-wire-b');
        const wireM = document.getElementById('scene1-wire-m');
        const labelM = document.getElementById('scene1-label-m');
        const statusText = document.getElementById('scene1-status');
        const l1Glow = document.getElementById('scene1-l1-glow');
        const l2Glow = document.getElementById('scene1-l2-glow');

        const hlNeg = document.getElementById('scene1-hl-neg');
        const hlPos = document.getElementById('scene1-hl-pos');
        const hlA = document.getElementById('scene1-hl-a');
        const hlB = document.getElementById('scene1-hl-b');
        const hlM = document.getElementById('scene1-hl-m');
        const hlSwitch = document.getElementById('scene1-hl-switch');
        const hlL1 = document.getElementById('scene1-hl-l1');
        const hlL2 = document.getElementById('scene1-hl-l2');

        function updateScene() {
            btnPrev.disabled = currentStep === 0;
            btnNext.disabled = currentStep === maxStep;

            // Reset all highlights
            const allHl = [hlNeg, hlPos, hlA, hlB, hlM, hlSwitch, hlL1, hlL2];
            allHl.forEach(hl => {
                hl.style.opacity = '0';
                hl.style.stroke = 'none';
            });
            l1Glow.style.opacity = '0';
            l2Glow.style.opacity = '0';
            statusText.style.opacity = '0';

            // Reset wire shapes and styles
            wireB.style.strokeDasharray = 'none';
            wireB.style.stroke = '#000';
            wireM.setAttribute('d', 'M 300 350 C 400 420, 600 420, 650 300');
            hlM.setAttribute('d', 'M 300 350 C 400 420, 600 420, 650 300');
            labelM.setAttribute('x', '450');
            labelM.setAttribute('y', '415');

            if (currentStep === 0) {
                switchBlade.setAttribute('x2', '280');
                switchBlade.setAttribute('y2', '310');
            } else if (currentStep === 1) {
                switchBlade.setAttribute('x2', '300');
                switchBlade.setAttribute('y2', '350');
                
                // Short circuit
                [hlPos, hlB, hlM, hlSwitch, hlNeg].forEach(hl => {
                    hl.style.opacity = '0.5';
                    hl.style.stroke = '#c62828';
                });
                
                statusText.textContent = '电源短路';
                statusText.style.fill = '#c62828';
                statusText.style.opacity = '1';
            } else if (currentStep === 2) {
                switchBlade.setAttribute('x2', '300');
                switchBlade.setAttribute('y2', '350');
                
                // Remove wire b
                wireB.style.strokeDasharray = '5,5';
                wireB.style.stroke = '#888';
                
                // Series circuit
                [hlPos, hlL2, hlA, hlL1, hlM, hlSwitch, hlNeg].forEach(hl => {
                    hl.style.opacity = '0.5';
                    hl.style.stroke = '#1565c0';
                });
                
                l1Glow.style.opacity = '0.5';
                l2Glow.style.opacity = '0.5';
                
                statusText.textContent = '串联';
                statusText.style.fill = '#1565c0';
                statusText.style.opacity = '1';
            } else if (currentStep === 3) {
                switchBlade.setAttribute('x2', '300');
                switchBlade.setAttribute('y2', '350');
                
                // Move wire M
                wireM.setAttribute('d', 'M 300 350 C 400 380, 480 380, 550 300');
                hlM.setAttribute('d', 'M 300 350 C 400 380, 480 380, 550 300');
                labelM.setAttribute('x', '430');
                labelM.setAttribute('y', '395');
                
                // Parallel circuit
                // Main circuit: green
                [hlPos, hlM, hlSwitch, hlNeg].forEach(hl => {
                    hl.style.opacity = '0.5';
                    hl.style.stroke = '#2e7d32';
                });
                // Branches: blue
                [hlA, hlL2, hlB, hlL1].forEach(hl => {
                    hl.style.opacity = '0.5';
                    hl.style.stroke = '#1565c0';
                });
                
                l1Glow.style.opacity = '0.5';
                l2Glow.style.opacity = '0.5';
                
                statusText.textContent = '并联';
                statusText.style.fill = '#2e7d32';
                statusText.style.opacity = '1';
            }
        }

        btnPrev.addEventListener('click', () => {
            if (currentStep > 0) {
                currentStep--;
                updateScene();
            }
        });

        btnNext.addEventListener('click', () => {
            if (currentStep < maxStep) {
                currentStep++;
                updateScene();
            }
        });

        // Initialize
        updateScene();
    })();
    </script>
