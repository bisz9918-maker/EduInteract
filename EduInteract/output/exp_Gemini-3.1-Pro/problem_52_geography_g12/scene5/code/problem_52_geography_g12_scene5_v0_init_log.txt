    <script>
    (function() {
        let currentStep = 0;
        const maxStep = 4;
        
        const btnPrev = document.getElementById('scene5-btn-prev');
        const btnNext = document.getElementById('scene5-btn-next');
        const stepText = document.getElementById('scene5-step-text');
        
        const phase1 = document.getElementById('scene5-phase1');
        const phase2 = document.getElementById('scene5-phase2');
        const step1Elements = document.getElementById('scene5-step1-elements');
        const formula = document.getElementById('scene5-formula');
        const step4Elements = document.getElementById('scene5-step4-elements');

        function updateScene() {
            // Update buttons and text
            btnPrev.disabled = currentStep === 0;
            btnNext.disabled = currentStep === maxStep;
            stepText.textContent = `步骤 ${currentStep} / ${maxStep}`;

            if (currentStep < 3) {
                // Phase 1 is visible
                phase1.style.opacity = '1';
                phase1.style.visibility = 'visible';
                phase1.style.pointerEvents = 'auto';
                
                phase2.style.opacity = '0';
                phase2.style.visibility = 'hidden';
                phase2.style.pointerEvents = 'none';
                
                if (currentStep >= 1) {
                    step1Elements.style.opacity = '1';
                    step1Elements.style.visibility = 'visible';
                } else {
                    step1Elements.style.opacity = '0';
                    step1Elements.style.visibility = 'hidden';
                }
                
                if (currentStep >= 2) {
                    formula.style.opacity = '1';
                    formula.style.visibility = 'visible';
                } else {
                    formula.style.opacity = '0';
                    formula.style.visibility = 'hidden';
                }
            } else {
                // Phase 2 is visible
                phase1.style.opacity = '0';
                phase1.style.visibility = 'hidden';
                phase1.style.pointerEvents = 'none';
                
                phase2.style.opacity = '1';
                phase2.style.visibility = 'visible';
                phase2.style.pointerEvents = 'auto';
                
                formula.style.opacity = '0';
                formula.style.visibility = 'hidden';
                
                if (currentStep >= 4) {
                    step4Elements.style.opacity = '1';
                    step4Elements.style.visibility = 'visible';
                } else {
                    step4Elements.style.opacity = '0';
                    step4Elements.style.visibility = 'hidden';
                }
            }
        }

        btnPrev.addEventListener('click', () => {
            if (currentStep > 0) {
                currentStep--;
                updateScene();
            }
        });

        btnNext.addEventListener('click', () => {
            if (currentStep < maxStep) {
                currentStep++;
                updateScene();
            }
        });

        // Initialize
        updateScene();
    })();
    </script>
