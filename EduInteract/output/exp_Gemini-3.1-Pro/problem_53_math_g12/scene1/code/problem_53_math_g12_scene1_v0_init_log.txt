    <script>
    (function() {
        let currentStep = 0;
        const maxStep = 4;
        
        const btnPrev = document.getElementById('scene1-btn-prev');
        const btnNext = document.getElementById('scene1-btn-next');
        
        const planePac = document.getElementById('scene1-plane-pac');
        const planePad = document.getElementById('scene1-plane-pad');
        const linePa = document.getElementById('scene1-line-pa');
        const lineMn = document.getElementById('scene1-line-mn');
        const arrows = document.getElementById('scene1-arrows');
        
        const bgLines = ['scene1-line-pd', 'scene1-line-ad', 'scene1-line-cd'];
        const fgLines = ['scene1-line-pb', 'scene1-line-ab', 'scene1-line-bc'];
        const otherPoints = [
            document.querySelector('circle[cx="500"]'), // B
            document.querySelector('circle[cx="300"]')  // D
        ];
        const otherLabels = [
            document.querySelector('text[x="500"]'), // B
            document.querySelector('text[x="280"]')  // D
        ];
        
        const labelCn = document.getElementById('scene1-label-cn');
        const labelNp = document.getElementById('scene1-label-np');
        const labelConclusion = document.getElementById('scene1-label-conclusion');
        
        function updateScene() {
            btnPrev.disabled = currentStep === 0;
            btnNext.disabled = currentStep === maxStep;
            
            // Step 0: Initial
            if (currentStep === 0) {
                planePac.setAttribute('fill-opacity', '0');
                planePad.setAttribute('fill-opacity', '0');
                linePa.setAttribute('stroke', '#000000');
                linePa.setAttribute('stroke-width', '2');
                lineMn.setAttribute('stroke', '#1565c0');
                lineMn.setAttribute('stroke-width', '2');
                arrows.setAttribute('opacity', '0');
                
                resetDimming();
                
                labelCn.setAttribute('opacity', '0');
                labelNp.setAttribute('opacity', '0');
                labelConclusion.setAttribute('opacity', '0');
            }
            // Step 1: Highlight planes and PA
            else if (currentStep === 1) {
                planePac.setAttribute('fill-opacity', '0.2');
                planePad.setAttribute('fill-opacity', '0.2');
                linePa.setAttribute('stroke', '#c62828');
                linePa.setAttribute('stroke-width', '4');
                lineMn.setAttribute('stroke', '#1565c0');
                lineMn.setAttribute('stroke-width', '2');
                arrows.setAttribute('opacity', '0');
                
                resetDimming();
                
                labelCn.setAttribute('opacity', '0');
                labelNp.setAttribute('opacity', '0');
                labelConclusion.setAttribute('opacity', '0');
            }
            // Step 2: Highlight MN and show arrows
            else if (currentStep === 2) {
                planePac.setAttribute('fill-opacity', '0.2');
                planePad.setAttribute('fill-opacity', '0.2');
                linePa.setAttribute('stroke', '#c62828');
                linePa.setAttribute('stroke-width', '4');
                lineMn.setAttribute('stroke', '#c62828');
                lineMn.setAttribute('stroke-width', '4');
                arrows.setAttribute('opacity', '1');
                
                resetDimming();
                
                labelCn.setAttribute('opacity', '0');
                labelNp.setAttribute('opacity', '0');
                labelConclusion.setAttribute('opacity', '0');
            }
            // Step 3: Dim background
            else if (currentStep === 3) {
                planePac.setAttribute('fill-opacity', '0.2');
                planePad.setAttribute('fill-opacity', '0');
                linePa.setAttribute('stroke', '#c62828');
                linePa.setAttribute('stroke-width', '4');
                lineMn.setAttribute('stroke', '#c62828');
                lineMn.setAttribute('stroke-width', '4');
                arrows.setAttribute('opacity', '1');
                
                applyDimming();
                
                labelCn.setAttribute('opacity', '0');
                labelNp.setAttribute('opacity', '0');
                labelConclusion.setAttribute('opacity', '0');
            }
            // Step 4: Show conclusion
            else if (currentStep === 4) {
                planePac.setAttribute('fill-opacity', '0.2');
                planePad.setAttribute('fill-opacity', '0');
                linePa.setAttribute('stroke', '#c62828');
                linePa.setAttribute('stroke-width', '4');
                lineMn.setAttribute('stroke', '#c62828');
                lineMn.setAttribute('stroke-width', '4');
                arrows.setAttribute('opacity', '1');
                
                applyDimming();
                
                labelCn.setAttribute('opacity', '1');
                labelNp.setAttribute('opacity', '1');
                labelConclusion.setAttribute('opacity', '1');
            }
        }
        
        function resetDimming() {
            bgLines.forEach(id => {
                const el = document.getElementById(id);
                el.setAttribute('stroke', '#000000');
                el.setAttribute('opacity', '1');
            });
            fgLines.forEach(id => {
                const el = document.getElementById(id);
                el.setAttribute('stroke', '#000000');
                el.setAttribute('opacity', '1');
            });
            otherPoints.forEach(el => {
                if (el) {
                    el.setAttribute('fill', '#000000');
                    el.setAttribute('opacity', '1');
                }
            });
            otherLabels.forEach(el => {
                if (el) {
                    el.setAttribute('fill', '#000000');
                    el.setAttribute('opacity', '1');
                }
            });
        }
        
        function applyDimming() {
            bgLines.forEach(id => {
                const el = document.getElementById(id);
                el.setAttribute('stroke', '#888888');
                el.setAttribute('opacity', '0.2');
            });
            fgLines.forEach(id => {
                const el = document.getElementById(id);
                el.setAttribute('stroke', '#888888');
                el.setAttribute('opacity', '0.2');
            });
            otherPoints.forEach(el => {
                if (el) {
                    el.setAttribute('fill', '#888888');
                    el.setAttribute('opacity', '0.2');
                }
            });
            otherLabels.forEach(el => {
                if (el) {
                    el.setAttribute('fill', '#888888');
                    el.setAttribute('opacity', '0.2');
                }
            });
        }
        
        btnPrev.addEventListener('click', () => {
            if (currentStep > 0) {
                currentStep--;
                updateScene();
            }
        });
        
        btnNext.addEventListener('click', () => {
            if (currentStep < maxStep) {
                currentStep++;
                updateScene();
            }
        });
        
        // Initialize
        updateScene();
    })();
    </script>
