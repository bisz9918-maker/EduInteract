    <script>
    (function() {
        let currentStep = 0;
        const maxStep = 4;
        
        const btnPrev = document.getElementById('scene1-btn-prev');
        const btnNext = document.getElementById('scene1-btn-next');
        const stepText = document.getElementById('scene1-step-text');
        
        const stepTexts = [
            "初始状态：长方体及各边长",
            "步骤 1/4：前右面展开",
            "步骤 2/4：底右面展开",
            "步骤 3/4：底后面展开",
            "结论：最短路径对比"
        ];
        
        function show(id) {
            const el = document.getElementById(id);
            if (el) {
                el.classList.remove('hidden');
                // trigger reflow
                void el.offsetWidth;
                el.style.opacity = 1;
            }
        }
        
        function hide(id) {
            const el = document.getElementById(id);
            if (el) {
                el.style.opacity = 0;
                setTimeout(() => {
                    if (el.style.opacity == 0) {
                        el.classList.add('hidden');
                    }
                }, 300);
            }
        }
        
        function updateScene() {
            btnPrev.disabled = currentStep === 0;
            btnNext.disabled = currentStep === maxStep;
            stepText.textContent = stepTexts[currentStep];
            
            // Hide all dynamic elements first
            ['scene1-fill-front', 'scene1-fill-right', 'scene1-fill-bottom', 'scene1-fill-back',
             'scene1-path-3d-1', 'scene1-path-3d-2', 'scene1-path-3d-3',
             'scene1-2d-step1', 'scene1-2d-step2', 'scene1-2d-step3', 'scene1-conclusion'].forEach(hide);
             
            if (currentStep === 1) {
                show('scene1-fill-front');
                show('scene1-fill-right');
                show('scene1-path-3d-1');
                show('scene1-2d-step1');
            } else if (currentStep === 2) {
                show('scene1-fill-bottom');
                show('scene1-fill-right');
                show('scene1-path-3d-2');
                show('scene1-2d-step2');
            } else if (currentStep === 3) {
                show('scene1-fill-bottom');
                show('scene1-fill-back');
                show('scene1-path-3d-3');
                show('scene1-2d-step3');
            } else if (currentStep === 4) {
                show('scene1-path-3d-1'); // Show the shortest path
                show('scene1-conclusion');
            }
            
            // Re-render MathJax if needed
            if (window.MathJax) {
                MathJax.typesetPromise();
            }
        }
        
        btnPrev.addEventListener('click', () => {
            if (currentStep > 0) {
                currentStep--;
                updateScene();
            }
        });
        
        btnNext.addEventListener('click', () => {
            if (currentStep < maxStep) {
                currentStep++;
                updateScene();
            }
        });
        
        // Initial setup
        updateScene();
    })();
    </script>
