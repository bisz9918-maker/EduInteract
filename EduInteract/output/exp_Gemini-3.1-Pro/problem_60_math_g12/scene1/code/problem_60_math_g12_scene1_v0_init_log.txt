    <script>
    (function() {
        let currentStep = 0;
        const maxStep = 3;
        
        const descriptions = [
            "初始状态：圆O，直径AB，切线AC、BD、CD，切点为P。",
            "连接辅助线：连接OC、OD、OP。",
            "由切线长定理可知：AC=PC，BD=PD，且OC平分∠AOP，OD平分∠BOP。",
            "推导结论：∠COD = 90°，且OP⊥CD，由射影定理得 PC·PD = OP²。"
        ];

        const btnPrev = document.getElementById('scene1-btn-prev');
        const btnNext = document.getElementById('scene1-btn-next');
        const descText = document.getElementById('scene1-desc');
        
        const step1Els = document.getElementById('scene1-step1-elements');
        const step2Els = document.getElementById('scene1-step2-elements');
        const step3Els = document.getElementById('scene1-step3-elements');

        function updateScene() {
            // Update buttons
            btnPrev.disabled = currentStep === 0;
            btnNext.disabled = currentStep === maxStep;
            
            // Update description
            descText.textContent = descriptions[currentStep];
            
            // Update visibility
            if (currentStep >= 1) {
                step1Els.classList.remove('scene1-hidden');
                step1Els.classList.add('scene1-visible');
            } else {
                step1Els.classList.remove('scene1-visible');
                step1Els.classList.add('scene1-hidden');
            }
            
            if (currentStep >= 2) {
                step2Els.classList.remove('scene1-hidden');
                step2Els.classList.add('scene1-visible');
            } else {
                step2Els.classList.remove('scene1-visible');
                step2Els.classList.add('scene1-hidden');
            }
            
            if (currentStep >= 3) {
                step3Els.classList.remove('scene1-hidden');
                step3Els.classList.add('scene1-visible');
            } else {
                step3Els.classList.remove('scene1-visible');
                step3Els.classList.add('scene1-hidden');
            }
        }

        btnPrev.addEventListener('click', () => {
            if (currentStep > 0) {
                currentStep--;
                updateScene();
            }
        });

        btnNext.addEventListener('click', () => {
            if (currentStep < maxStep) {
                currentStep++;
                updateScene();
            }
        });

        // Initialize
        updateScene();
    })();
    </script>
