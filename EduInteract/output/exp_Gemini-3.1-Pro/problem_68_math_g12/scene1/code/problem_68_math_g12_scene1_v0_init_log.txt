    <script>
    (function() {
        let currentStep = 0;
        const maxStep = 4;

        const btnPrev = document.getElementById('scene1-btn-prev');
        const btnNext = document.getElementById('scene1-btn-next');

        // 获取需要控制的元素
        const els = {
            lineOc: document.getElementById('scene1-line-oc'),
            hlOb: document.getElementById('scene1-hl-ob'),
            hlBp: document.getElementById('scene1-hl-bp'),
            hlOc: document.getElementById('scene1-hl-oc'),
            hlCd: document.getElementById('scene1-hl-cd'),
            linePc: document.getElementById('scene1-line-pc'),
            angleC: document.getElementById('scene1-angle-c'),
            angleO: document.getElementById('scene1-angle-o'),
            areaOcp: document.getElementById('scene1-area-ocp'),
            areaOcd: document.getElementById('scene1-area-ocd'),
            braceOb: document.getElementById('scene1-brace-ob'),
            braceBp: document.getElementById('scene1-brace-bp'),
            braceOp: document.getElementById('scene1-brace-op'),
            labelOc: document.getElementById('scene1-label-oc'),
            labelCd: document.getElementById('scene1-label-cd'),
            textAngle: document.getElementById('scene1-text-angle')
        };

        function setOpacity(el, val) {
            if(el) el.style.opacity = val;
        }

        function updateScene() {
            // 按钮状态
            btnPrev.disabled = currentStep === 0;
            btnNext.disabled = currentStep === maxStep;

            // 步骤 0: 初始状态
            if (currentStep === 0) {
                setOpacity(els.lineOc, 0);
                setOpacity(els.hlOc, 0);
                setOpacity(els.labelOc, 0);
                
                setOpacity(els.angleC, 0);
                setOpacity(els.textAngle, 0);
                els.linePc.setAttribute('stroke-width', '2');
                
                setOpacity(els.areaOcp, 0);
                setOpacity(els.braceOp, 0);
                setOpacity(els.angleO, 0);
                
                setOpacity(els.areaOcd, 0);
                setOpacity(els.hlCd, 0);
                setOpacity(els.labelCd, 0);
            }
            // 步骤 1: 辅助线
            else if (currentStep === 1) {
                setOpacity(els.lineOc, 1);
                setOpacity(els.hlOc, 1);
                setOpacity(els.labelOc, 1);
                
                setOpacity(els.angleC, 0);
                setOpacity(els.textAngle, 0);
                els.linePc.setAttribute('stroke-width', '2');
                
                setOpacity(els.areaOcp, 0);
                setOpacity(els.braceOp, 0);
                setOpacity(els.angleO, 0);
                
                setOpacity(els.areaOcd, 0);
                setOpacity(els.hlCd, 0);
                setOpacity(els.labelCd, 0);
            }
            // 步骤 2: 切线性质
            else if (currentStep === 2) {
                setOpacity(els.lineOc, 1);
                setOpacity(els.hlOc, 1);
                setOpacity(els.labelOc, 1);
                
                setOpacity(els.angleC, 1);
                setOpacity(els.textAngle, 1);
                els.linePc.setAttribute('stroke-width', '4');
                
                setOpacity(els.areaOcp, 0);
                setOpacity(els.braceOp, 0);
                setOpacity(els.angleO, 0);
                
                setOpacity(els.areaOcd, 0);
                setOpacity(els.hlCd, 0);
                setOpacity(els.labelCd, 0);
            }
            // 步骤 3: 解直角三角形OCP
            else if (currentStep === 3) {
                setOpacity(els.lineOc, 1);
                setOpacity(els.hlOc, 1);
                setOpacity(els.labelOc, 1);
                
                setOpacity(els.angleC, 1);
                setOpacity(els.textAngle, 1);
                els.linePc.setAttribute('stroke-width', '4');
                
                setOpacity(els.areaOcp, 1);
                setOpacity(els.braceOp, 1);
                setOpacity(els.angleO, 1);
                
                setOpacity(els.areaOcd, 0);
                setOpacity(els.hlCd, 0);
                setOpacity(els.labelCd, 0);
            }
            // 步骤 4: 求解CD
            else if (currentStep === 4) {
                setOpacity(els.lineOc, 1);
                setOpacity(els.hlOc, 1);
                setOpacity(els.labelOc, 1);
                
                setOpacity(els.angleC, 1);
                setOpacity(els.textAngle, 1);
                els.linePc.setAttribute('stroke-width', '4');
                
                setOpacity(els.areaOcp, 0); // 取消 OCP 高亮
                setOpacity(els.braceOp, 1);
                setOpacity(els.angleO, 1);
                
                setOpacity(els.areaOcd, 1); // 高亮 OCD
                setOpacity(els.hlCd, 1);
                setOpacity(els.labelCd, 1);
            }
        }

        btnPrev.addEventListener('click', () => {
            if (currentStep > 0) {
                currentStep--;
                updateScene();
            }
        });

        btnNext.addEventListener('click', () => {
            if (currentStep < maxStep) {
                currentStep++;
                updateScene();
            }
        });

        // 初始化
        updateScene();
    })();
    </script>
