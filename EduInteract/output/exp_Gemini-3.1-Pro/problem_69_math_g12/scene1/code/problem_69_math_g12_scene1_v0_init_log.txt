    <script>
    (function() {
        let currentStep = 0;
        const maxStep = 4;
        
        const btnPrev = document.getElementById('scene1-btn-prev');
        const btnNext = document.getElementById('scene1-btn-next');
        
        function show(el) {
            if (!el) return;
            el.style.display = el.tagName === 'g' ? 'inline' : 'block';
            // Force reflow
            void el.offsetWidth;
            el.style.opacity = '1';
        }
        
        function hide(el) {
            if (!el) return;
            el.style.opacity = '0';
            setTimeout(() => {
                if (el.style.opacity === '0') {
                    el.style.display = 'none';
                }
            }, 500);
        }
        
        function updateScene() {
            // Update SVG steps
            for (let i = 1; i <= maxStep; i++) {
                const el = document.getElementById('scene1-step' + i);
                const labelEl = document.getElementById('scene1-labels-step' + i);
                if (i <= currentStep) {
                    show(el);
                    show(labelEl);
                } else {
                    hide(el);
                    hide(labelEl);
                }
            }
            
            // Update Text steps
            for (let i = 0; i <= maxStep; i++) {
                const el = document.getElementById('scene1-text-step' + i);
                if (i === currentStep) {
                    show(el);
                } else {
                    hide(el);
                }
            }
            
            btnPrev.disabled = currentStep === 0;
            btnNext.disabled = currentStep === maxStep;
        }
        
        btnPrev.addEventListener('click', () => {
            if (currentStep > 0) {
                currentStep--;
                updateScene();
            }
        });
        
        btnNext.addEventListener('click', () => {
            if (currentStep < maxStep) {
                currentStep++;
                updateScene();
            }
        });
        
        // Initial state
        updateScene();
    })();
    </script>
