    <script>
    (function() {
        let currentStep = 0;
        const maxStep = 2;
        
        const btnPrev = document.getElementById('scene1-btn-prev');
        const btnNext = document.getElementById('scene1-btn-next');
        const statusText = document.getElementById('scene1-status');
        
        const groupA = document.getElementById('scene1-group-A');
        const groupB = document.getElementById('scene1-group-B');
        const groupC = document.getElementById('scene1-group-C');
        
        const box0 = document.getElementById('scene1-box0');
        const box2 = document.getElementById('scene1-box2');
        
        const AvA = document.getElementById('scene1-A-vA');
        const BvA = document.getElementById('scene1-B-vA');
        const Bv = document.getElementById('scene1-B-v');
        const Cv0 = document.getElementById('scene1-C-v0');
        const CvC = document.getElementById('scene1-C-vC');
        const Cv = document.getElementById('scene1-C-v');
        
        const steps = [
            {
                name: "初始状态",
                pos: { A: 100, B: 260, C: 100 },
                show: [box0, Cv0],
                hide: [box2, AvA, BvA, Bv, CvC, Cv]
            },
            {
                name: "C滑离A瞬间",
                pos: { A: 180, B: 340, C: 340 },
                show: [AvA, BvA, CvC],
                hide: [box0, box2, Cv0, Bv, Cv]
            },
            {
                name: "最终状态",
                pos: { A: 260, B: 500, C: 560 },
                show: [box2, AvA, Bv, Cv],
                hide: [box0, BvA, Cv0, CvC]
            }
        ];
        
        function setOpacity(el, val) {
            el.style.opacity = val;
            if (val === 0) {
                // Wait for transition to finish before setting display none
                setTimeout(() => {
                    if (el.style.opacity == 0) el.style.display = 'none';
                }, 500);
            } else {
                el.style.display = '';
                // Force reflow
                void el.offsetWidth;
                el.style.opacity = 1;
            }
        }
        
        function updateScene() {
            const step = steps[currentStep];
            
            // Update positions
            groupA.style.transform = `translate(${step.pos.A}px, 0)`;
            groupB.style.transform = `translate(${step.pos.B}px, 0)`;
            groupC.style.transform = `translate(${step.pos.C}px, 0)`;
            
            // Update visibility
            step.show.forEach(el => setOpacity(el, 1));
            step.hide.forEach(el => setOpacity(el, 0));
            
            // Update controls
            statusText.textContent = `状态 ${currentStep + 1} / 3：${step.name}`;
            btnPrev.disabled = currentStep === 0;
            btnNext.disabled = currentStep === maxStep;
        }
        
        btnPrev.addEventListener('click', () => {
            if (currentStep > 0) {
                currentStep--;
                updateScene();
            }
        });
        
        btnNext.addEventListener('click', () => {
            if (currentStep < maxStep) {
                currentStep++;
                updateScene();
            }
        });
        
        // Initialize
        updateScene();
    })();
    </script>
