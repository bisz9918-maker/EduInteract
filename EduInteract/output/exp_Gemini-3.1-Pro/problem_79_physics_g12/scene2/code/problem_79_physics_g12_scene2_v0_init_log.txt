    <script>
    (function() {
        const states = [
            {
                // Step 0: Initial
                M_x: 100,
                m_x: 230,
                F_opacity: 0,
                path_vM: "M 100 420 L 100 420 L 100 420",
                path_vm: "M 100 420 L 100 420 L 100 420",
                area_points: "100,420 100,420 100,420 100,420",
                area_opacity: 0,
                grid_opacity: 1,
                t08_opacity: 0,
                t20_opacity: 0,
                v24_opacity: 0,
                v20_opacity: 0,
                v08_opacity: 0,
                label_vM_opacity: 0,
                label_vm_opacity: 0,
                label_vm_x: 320,
                label_vm_y: 372,
                cospeed_opacity: 0,
                area_text_opacity: 0
            },
            {
                // Step 1: Accelerate
                M_x: 196,
                m_x: 262,
                F_opacity: 1,
                path_vM: "M 100 420 L 300 276 L 300 276",
                path_vm: "M 100 420 L 300 372 L 300 372",
                area_points: "100,420 300,276 300,372 100,420",
                area_opacity: 0.2,
                grid_opacity: 1,
                t08_opacity: 0,
                t20_opacity: 0,
                v24_opacity: 0,
                v20_opacity: 0,
                v08_opacity: 0,
                label_vM_opacity: 1,
                label_vm_opacity: 1,
                label_vm_x: 320,
                label_vm_y: 372,
                cospeed_opacity: 0,
                area_text_opacity: 0
            },
            {
                // Step 2: Critical point (F removed)
                M_x: 196,
                m_x: 262,
                F_opacity: 0,
                path_vM: "M 100 420 L 300 276 L 300 276",
                path_vm: "M 100 420 L 300 372 L 300 372",
                area_points: "100,420 300,276 300,372 100,420",
                area_opacity: 0.2,
                grid_opacity: 1,
                t08_opacity: 1,
                t20_opacity: 0,
                v24_opacity: 1,
                v20_opacity: 0,
                v08_opacity: 1,
                label_vM_opacity: 1,
                label_vm_opacity: 1,
                label_vm_x: 320,
                label_vm_y: 372,
                cospeed_opacity: 0,
                area_text_opacity: 0
            },
            {
                // Step 3: Decelerate and catch up
                M_x: 460,
                m_x: 430,
                F_opacity: 0,
                path_vM: "M 100 420 L 300 276 L 600 300",
                path_vm: "M 100 420 L 300 372 L 600 300",
                area_points: "100,420 300,276 600,300 300,372",
                area_opacity: 0.2,
                grid_opacity: 1,
                t08_opacity: 1,
                t20_opacity: 0,
                v24_opacity: 1,
                v20_opacity: 0,
                v08_opacity: 1,
                label_vM_opacity: 1,
                label_vm_opacity: 1,
                label_vm_x: 620,
                label_vm_y: 325,
                cospeed_opacity: 0,
                area_text_opacity: 0
            },
            {
                // Step 4: Co-speed
                M_x: 460,
                m_x: 430,
                F_opacity: 0,
                path_vM: "M 100 420 L 300 276 L 600 300",
                path_vm: "M 100 420 L 300 372 L 600 300",
                area_points: "100,420 300,276 600,300 300,372",
                area_opacity: 0.2,
                grid_opacity: 1,
                t08_opacity: 1,
                t20_opacity: 1,
                v24_opacity: 1,
                v20_opacity: 1,
                v08_opacity: 1,
                label_vM_opacity: 1,
                label_vm_opacity: 1,
                label_vm_x: 620,
                label_vm_y: 325,
                cospeed_opacity: 1,
                area_text_opacity: 1
            }
        ];

        let currentStep = 0;

        const btnPrev = document.getElementById('scene2-btn-prev');
        const btnNext = document.getElementById('scene2-btn-next');

        function updateScene(step) {
            const state = states[step];
            const t = d3.transition().duration(1000);

            // Update Model
            d3.select('#scene2-model-M').transition(t).attr('transform', `translate(${state.M_x}, 140)`);
            d3.select('#scene2-model-m').transition(t).attr('transform', `translate(${state.m_x}, 110)`);
            
            // Force F moves with M
            d3.select('#scene2-force-F')
                .transition(t)
                .attr('transform', `translate(${state.M_x + 160}, 160)`)
                .style('opacity', state.F_opacity);

            // Update Paths
            d3.select('#scene2-path-vM').transition(t).attr('d', state.path_vM);
            d3.select('#scene2-path-vm').transition(t).attr('d', state.path_vm);

            // Update Area
            d3.select('#scene2-area').transition(t)
                .attr('points', state.area_points)
                .style('opacity', state.area_opacity);

            // Update Grid and Labels
            d3.select('#scene2-grid-lines').transition(t).style('opacity', state.grid_opacity);
            
            d3.select('#scene2-line-t08').transition(t).style('opacity', state.t08_opacity);
            d3.select('#scene2-label-t08').transition(t).style('opacity', state.t08_opacity);
            
            d3.select('#scene2-line-t20').transition(t).style('opacity', state.t20_opacity);
            d3.select('#scene2-label-t20').transition(t).style('opacity', state.t20_opacity);
            
            d3.select('#scene2-line-v24').transition(t).style('opacity', state.v24_opacity);
            d3.select('#scene2-label-v24').transition(t).style('opacity', state.v24_opacity);
            
            d3.select('#scene2-line-v20').transition(t).style('opacity', state.v20_opacity);
            d3.select('#scene2-label-v20').transition(t).style('opacity', state.v20_opacity);
            
            d3.select('#scene2-line-v08').transition(t).style('opacity', state.v08_opacity);
            d3.select('#scene2-label-v08').transition(t).style('opacity', state.v08_opacity);

            d3.select('#scene2-label-vM').transition(t).style('opacity', state.label_vM_opacity);
            
            d3.select('#scene2-label-vm')
                .transition(t)
                .attr('x', state.label_vm_x)
                .attr('y', state.label_vm_y)
                .style('opacity', state.label_vm_opacity);
            
            d3.select('#scene2-label-cospeed').transition(t).style('opacity', state.cospeed_opacity);
            d3.select('#scene2-label-area').transition(t).style('opacity', state.area_text_opacity);

            // Update Buttons
            btnPrev.disabled = step === 0;
            btnNext.disabled = step === states.length - 1;
        }

        btnPrev.addEventListener('click', () => {
            if (currentStep > 0) {
                currentStep--;
                updateScene(currentStep);
            }
        });

        btnNext.addEventListener('click', () => {
            if (currentStep < states.length - 1) {
                currentStep++;
                updateScene(currentStep);
            }
        });

        // Initialize
        updateScene(0);
    })();
    </script>
