    <script>
    (function() {
        const slider = document.getElementById('scene2-slider');
        const blockA = document.getElementById('scene2-block-a');
        const blockAText = document.getElementById('scene2-block-a-text');
        const blockB = document.getElementById('scene2-block-b');
        const blockBText = document.getElementById('scene2-block-b-text');
        const spring = document.getElementById('scene2-spring');
        const vbArrow = document.getElementById('scene2-vb-arrow');
        const vbLabel = document.getElementById('scene2-vb-label');
        const statusT = document.getElementById('scene2-status-t');
        const statusV = document.getElementById('scene2-status-v');
        const statusBox = document.getElementById('scene2-status');

        function drawSpring(startX, startY, endX, endY) {
            const segments = 16;
            const width = endX - startX;
            const segWidth = width / segments;
            const amp = 10;
            let d = `M ${startX} ${startY} `;
            for (let i = 1; i <= segments; i++) {
                const x = startX + i * segWidth;
                const y = startY + (i % 2 === 0 ? 0 : (i % 4 === 1 ? -amp : amp));
                d += `L ${x} ${y} `;
            }
            spring.setAttribute('d', d);
        }

        function updateScene(s) {
            // s from 0 to 1
            // A position: x_A = 250 + 80 * cos(pi * s)
            const xA = 250 + 80 * Math.cos(Math.PI * s);
            blockA.setAttribute('x', xA - 20);
            blockAText.setAttribute('x', xA);
            
            // Spring
            drawSpring(100, 150, xA - 20, 150);

            // B position: y_B = 150 + 240 * s^2
            const yB = 150 + 240 * s * s;
            blockB.setAttribute('y', yB - 20);
            blockBText.setAttribute('y', yB - 35);

            // Velocity B
            const vLength = 50 * s;
            if (vLength > 2) {
                vbArrow.setAttribute('opacity', '1');
                vbLabel.setAttribute('opacity', '1');
                vbArrow.setAttribute('y1', yB);
                vbArrow.setAttribute('y2', yB + vLength);
                vbLabel.setAttribute('y', yB);
            } else {
                vbArrow.setAttribute('opacity', '0');
                vbLabel.setAttribute('opacity', '0');
            }

            // Status
            if (s === 0) {
                statusT.innerHTML = '\(t = 0\)';
                statusV.innerHTML = '\(v_B = 0\)';
                statusBox.style.color = '#000';
                statusBox.style.fontWeight = 'normal';
            } else if (s === 1) {
                statusT.innerHTML = '\(t = rac{T}{2}\)';
                statusV.innerHTML = '\(v_B = rac{gT}{2}\)';
                statusBox.style.color = '#c62828';
                statusBox.style.fontWeight = 'bold';
            } else {
                statusT.innerHTML = `\(t = ${(s/2).toFixed(2)}T\)`;
                statusV.innerHTML = `\(v_B = ${(s/2).toFixed(2)}gT\)`;
                statusBox.style.color = '#000';
                statusBox.style.fontWeight = 'normal';
            }
            
            if (window.MathJax && MathJax.typesetPromise) {
                MathJax.typesetPromise([statusBox]).catch(err => console.log(err));
            }
        }

        slider.addEventListener('input', (e) => {
            const s = parseFloat(e.target.value) / 100;
            updateScene(s);
        });

        // Initial draw
        updateScene(0);
    })();
    </script>
