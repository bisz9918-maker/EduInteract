    <script>
    (function() {
        // 生成磁场点阵
        const bField = document.getElementById('scene1-b-field');
        for (let x = 100; x <= 700; x += 50) {
            for (let y = 100; y <= 400; y += 50) {
                const circle = document.createElementNS('http://www.w3.org/2000/svg', 'circle');
                circle.setAttribute('cx', x);
                circle.setAttribute('cy', y);
                circle.setAttribute('r', 2);
                bField.appendChild(circle);
            }
        }

        // 状态数据
        const steps = [
            {
                x: 600,
                vLen: 0,
                faLen: 0,
                showV: 0,
                showFA: 0,
                showFric: 0,
                showI: 0,
                text: "初始状态：金属棒在恒力F作用下静止，即将开始运动。"
            },
            {
                x: 520,
                vLen: 30,
                faLen: 0,
                showV: 1,
                showFA: 0,
                showFric: 0,
                showI: 1,
                text: "运动与感应：棒向左切割磁感线，产生感应电动势。回路中产生逆时针感应电流。"
            },
            {
                x: 440,
                vLen: 50,
                faLen: 20,
                showV: 1,
                showFA: 1,
                showFric: 1,
                showI: 1,
                text: "受力分析：棒受恒力F、滑动摩擦力f和安培力F_A。安培力向右，阻碍相对运动。"
            },
            {
                x: 340,
                vLen: 70,
                faLen: 40,
                showV: 1,
                showFA: 1,
                showFric: 1,
                showI: 1,
                text: "动态变化：加速阶段。速度 v 增大 → 电流 I 增大 → 安培力 F_A 增大。合力逐渐减小。"
            },
            {
                x: 220,
                vLen: 90,
                faLen: 60,
                showV: 1,
                showFA: 1,
                showFric: 1,
                showI: 1,
                text: "匀速稳定：当 F_A 增大到使合力为零时，棒达到最大速度并做匀速直线运动。此时 F = F_A + f。"
            }
        ];

        let currentStep = 0;

        const rodGroup = document.getElementById('scene1-rod-group');
        const vGroup = document.getElementById('scene1-v-group');
        const vArrow = document.getElementById('scene1-v-arrow');
        const vText = document.getElementById('scene1-v-text');
        
        const faGroup = document.getElementById('scene1-fa-group');
        const faArrow = document.getElementById('scene1-fa-arrow');
        const faText = document.getElementById('scene1-fa-text');
        
        const fGroup = document.getElementById('scene1-f-group');
        const rodIGroup = document.getElementById('scene1-rod-i-group');
        const railCurrents = document.getElementById('scene1-rail-currents');
        
        const railITop = document.getElementById('scene1-rail-i-top');
        const railITopText = document.getElementById('scene1-rail-i-top-text');
        const railIBottom = document.getElementById('scene1-rail-i-bottom');
        const railIBottomText = document.getElementById('scene1-rail-i-bottom-text');

        const topText = document.getElementById('scene1-top-text');
        const btnPrev = document.getElementById('scene1-btn-prev');
        const btnNext = document.getElementById('scene1-btn-next');

        function updateScene() {
            const data = steps[currentStep];
            
            // 棒位置
            rodGroup.setAttribute('transform', `translate(${data.x}, 0)`);
            
            // 速度 v
            vGroup.style.opacity = data.showV;
            if (data.vLen > 0) {
                vArrow.setAttribute('x2', -data.vLen);
                vText.setAttribute('x', -data.vLen - 5);
            }
            
            // 安培力 F_A
            faGroup.style.opacity = data.showFA;
            if (data.faLen > 0) {
                faArrow.setAttribute('x2', data.faLen);
                faText.setAttribute('x', data.faLen + 5);
            }
            
            // 摩擦力 f
            fGroup.style.opacity = data.showFric;
            
            // 电流 I
            rodIGroup.style.opacity = data.showI;
            railCurrents.style.opacity = data.showI;
            
            // 导轨电流位置更新 (跟随棒)
            railITop.setAttribute('x1', data.x + 60);
            railITop.setAttribute('x2', data.x + 30);
            railITopText.setAttribute('x', data.x + 40);
            
            railIBottom.setAttribute('x1', data.x + 60);
            railIBottom.setAttribute('x2', data.x + 90);
            railIBottomText.setAttribute('x', data.x + 70);

            // 文本
            topText.textContent = data.text;
            
            // 按钮状态
            btnPrev.disabled = currentStep === 0;
            btnNext.disabled = currentStep === steps.length - 1;
        }

        btnPrev.addEventListener('click', () => {
            if (currentStep > 0) {
                currentStep--;
                updateScene();
            }
        });

        btnNext.addEventListener('click', () => {
            if (currentStep < steps.length - 1) {
                currentStep++;
                updateScene();
            }
        });

        // 初始化
        updateScene();
    })();
    </script>
