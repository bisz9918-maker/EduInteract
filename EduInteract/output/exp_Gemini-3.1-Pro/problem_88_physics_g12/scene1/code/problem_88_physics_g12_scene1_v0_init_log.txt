    <script>
    (function() {
        const slider = document.getElementById('scene1-slider-theta');
        const sliderText = document.getElementById('scene1-slider-text');
        
        const rod3d = document.getElementById('scene1-rod-3d');
        const rod2d = document.getElementById('scene1-rod-2d');
        const opLine = document.getElementById('scene1-op-line');
        
        const vecV0 = document.getElementById('scene1-vec-v0');
        const vecVx = document.getElementById('scene1-vec-vx');
        const vecVy = document.getElementById('scene1-vec-vy');
        const vecRect = document.getElementById('scene1-vec-rect');
        
        const labelV0 = document.getElementById('scene1-label-v0');
        const labelVx = document.getElementById('scene1-label-vx');
        const labelVy = document.getElementById('scene1-label-vy');
        
        const angleO = document.getElementById('scene1-angle-o');
        const angleP = document.getElementById('scene1-angle-p');
        const labelThetaP = document.getElementById('scene1-label-theta-p');

        function updateScene(thetaDeg) {
            sliderText.textContent = `拖动滑块改变金属棒位置 (θ = ${thetaDeg}°)`;
            const theta = thetaDeg * Math.PI / 180;
            
            // 3D Rod
            const r3d = 140;
            const p1x = 120 + r3d * Math.sin(theta);
            const p1y = 120 + r3d * Math.cos(theta);
            const p2x = 180 + r3d * Math.sin(theta);
            const p2y = 170 + r3d * Math.cos(theta);
            
            // Extend rod by 10px
            const dx = p2x - p1x;
            const dy = p2y - p1y;
            const len = Math.sqrt(dx*dx + dy*dy);
            const ux = dx / len;
            const uy = dy / len;
            
            rod3d.setAttribute('x1', p1x - 10 * ux);
            rod3d.setAttribute('y1', p1y - 10 * uy);
            rod3d.setAttribute('x2', p2x + 10 * ux);
            rod3d.setAttribute('y2', p2y + 10 * uy);
            
            // 2D Rod
            const r2d = 200;
            const px = 500 + r2d * Math.sin(theta);
            const py = 150 + r2d * Math.cos(theta);
            
            rod2d.setAttribute('cx', px);
            rod2d.setAttribute('cy', py);
            
            opLine.setAttribute('x2', px);
            opLine.setAttribute('y2', py);
            
            // 2D Angle at O
            const arcOx = 500 + 40 * Math.sin(theta);
            const arcOy = 150 + 40 * Math.cos(theta);
            angleO.setAttribute('d', `M 500 190 A 40 40 0 0 0 ${arcOx} ${arcOy}`);
            
            // 2D Vectors
            const v0Len = 100;
            const vxLen = v0Len * Math.cos(theta);
            const vyLen = v0Len * Math.sin(theta);
            
            const v0x = px + vxLen;
            const v0y = py - vyLen;
            
            vecV0.setAttribute('x1', px);
            vecV0.setAttribute('y1', py);
            vecV0.setAttribute('x2', v0x);
            vecV0.setAttribute('y2', v0y);
            
            vecVx.setAttribute('x1', px);
            vecVx.setAttribute('y1', py);
            vecVx.setAttribute('x2', px + vxLen);
            vecVx.setAttribute('y2', py);
            
            vecVy.setAttribute('x1', px);
            vecVy.setAttribute('y1', py);
            vecVy.setAttribute('x2', px);
            vecVy.setAttribute('y2', py - vyLen);
            
            vecRect.setAttribute('d', `M ${px + vxLen} ${py} L ${v0x} ${v0y} L ${px} ${py - vyLen}`);
            
            // 2D Angle at P
            if (thetaDeg > 0 && thetaDeg < 90) {
                const arcPx = px + 30 * Math.cos(theta);
                const arcPy = py - 30 * Math.sin(theta);
                angleP.setAttribute('d', `M ${px + 30} ${py} A 30 30 0 0 0 ${arcPx} ${arcPy}`);
                angleP.style.display = 'block';
                labelThetaP.style.display = 'block';
                labelThetaP.setAttribute('x', px + 45);
                labelThetaP.setAttribute('y', py - 25);
            } else {
                angleP.style.display = 'none';
                labelThetaP.style.display = 'none';
            }
            
            // Labels
            labelV0.setAttribute('x', v0x + 25);
            labelV0.setAttribute('y', v0y - 5);
            
            labelVx.setAttribute('x', px + vxLen / 2 - 5);
            labelVx.setAttribute('y', py + 35);
            
            labelVy.setAttribute('x', px - 55);
            labelVy.setAttribute('y', py - vyLen / 2 + 5);
            
            if (thetaDeg === 90) {
                labelVx.style.display = 'none';
            } else {
                labelVx.style.display = 'block';
            }
            
            if (thetaDeg === 0) {
                labelVy.style.display = 'none';
            } else {
                labelVy.style.display = 'block';
            }
        }

        slider.addEventListener('input', (e) => {
            updateScene(parseFloat(e.target.value));
        });

        // Initial update
        updateScene(30);
    })();
    </script>
