    <script>
    (function() {
        let currentStep = 0;
        const totalSteps = 5;

        const btnPrev = document.getElementById('scene2-btn-prev');
        const btnNext = document.getElementById('scene2-btn-next');
        const stepIndicator = document.getElementById('scene2-step-indicator');

        // Elements
        const polyDCE = document.getElementById('scene2-poly-dce');
        const polyACG = document.getElementById('scene2-poly-acg');
        const polyABF = document.getElementById('scene2-poly-abf');
        const polyADG = document.getElementById('scene2-poly-adg');
        const polyCHG = document.getElementById('scene2-poly-chg');
        
        const lineCP = document.getElementById('scene2-line-cp');
        const perpCP = document.getElementById('scene2-perp-cp');
        const textCP = document.getElementById('scene2-text-cp');
        
        const textCG = document.getElementById('scene2-text-cg');
        const textDG = document.getElementById('scene2-text-dg');

        const lineAG = document.getElementById('scene2-line-ag');
        const lineGH = document.getElementById('scene2-line-gh');
        const lineHF = document.getElementById('scene2-line-hf');

        function updateScene() {
            // Update buttons
            btnPrev.disabled = currentStep === 0;
            btnNext.disabled = currentStep === totalSteps;
            stepIndicator.textContent = `步骤 ${currentStep} / ${totalSteps}`;

            // Update texts
            for (let i = 0; i <= totalSteps; i++) {
                const textEl = document.getElementById(`scene2-step-text-${i}`);
                if (i === currentStep) {
                    textEl.classList.add('active');
                } else {
                    textEl.classList.remove('active');
                }
            }

            // Reset all highlights
            polyDCE.style.fillOpacity = "0"; polyDCE.style.strokeWidth = "0";
            polyACG.style.fillOpacity = "0"; polyACG.style.strokeWidth = "0";
            polyABF.style.fillOpacity = "0"; polyABF.style.strokeWidth = "0";
            polyADG.style.fillOpacity = "0"; polyADG.style.strokeWidth = "0";
            polyCHG.style.fillOpacity = "0"; polyCHG.style.strokeWidth = "0";
            
            lineCP.style.opacity = "0";
            perpCP.style.opacity = "0";
            textCP.style.opacity = "0";
            
            textCG.style.opacity = "0";
            textDG.style.opacity = "0";

            lineAG.style.strokeWidth = "0";
            lineGH.style.strokeWidth = "0";
            lineHF.style.strokeWidth = "0";

            // Apply step logic
            if (currentStep >= 1) {
                if (currentStep === 1) {
                    polyDCE.style.fillOpacity = "0.2";
                    polyDCE.style.strokeWidth = "3";
                }
            }
            if (currentStep >= 2) {
                lineCP.style.opacity = "1";
                perpCP.style.opacity = "1";
                textCP.style.opacity = "1";
                if (currentStep > 2) {
                    lineCP.style.stroke = "#999";
                    perpCP.style.stroke = "#999";
                } else {
                    lineCP.style.stroke = "#1565c0";
                    perpCP.style.stroke = "#1565c0";
                }
            }
            if (currentStep === 3) {
                polyACG.style.fillOpacity = "0.2";
                polyACG.style.strokeWidth = "3";
                polyABF.style.fillOpacity = "0.2";
                polyABF.style.strokeWidth = "3";
                textCG.style.opacity = "1";
            }
            if (currentStep >= 4) {
                textCG.style.opacity = "1";
                if (currentStep === 4) {
                    polyADG.style.fillOpacity = "0.2";
                    polyADG.style.strokeWidth = "3";
                    polyCHG.style.fillOpacity = "0.2";
                    polyCHG.style.strokeWidth = "3";
                    textDG.style.opacity = "1";
                }
            }
            if (currentStep === 5) {
                textDG.style.opacity = "1";
                lineAG.style.strokeWidth = "4";
                lineGH.style.strokeWidth = "4";
                lineHF.style.strokeWidth = "4";
            }
        }

        btnPrev.addEventListener('click', () => {
            if (currentStep > 0) {
                currentStep--;
                updateScene();
            }
        });

        btnNext.addEventListener('click', () => {
            if (currentStep < totalSteps) {
                currentStep++;
                updateScene();
            }
        });

        // Initialize
        updateScene();
    })();
    </script>
