    <script>
    (function() {
        // Scene 1 实现
        let scene1_step = 0;
        const totalSteps = 4;

        const btnPrev = document.getElementById('scene1-btn-prev');
        const btnNext = document.getElementById('scene1-btn-next');
        const stepIndicator = document.getElementById('scene1-step-indicator');

        // 元素引用
        const solidLarge = document.getElementById('scene1-solid-large');
        const solidSmall = document.getElementById('scene1-solid-small');
        const netGroup = document.getElementById('scene1-net-group');
        const descGroup = document.getElementById('scene1-desc-group');
        const highlightTextGroup = document.getElementById('scene1-highlight-text-group');
        const constraintGroup = document.getElementById('scene1-constraint-group');

        const baseBottomLarge = document.getElementById('scene1-base-bottom-large');
        const baseTopLarge = document.getElementById('scene1-base-top-large');

        // 创建侧面多边形（大版）用于填充高亮
        const sideFrontLarge = createPolygon('330,320 470,320 500,220 360,220', 'scene1-side-front-large');
        const sideRightLarge = createPolygon('470,320 500,220 430,99 400,199', 'scene1-side-right-large');
        const sideLeftLarge = createPolygon('330,320 360,220 430,99 400,199', 'scene1-side-left-large');
        solidLarge.insertBefore(sideLeftLarge, baseBottomLarge.nextSibling);
        solidLarge.insertBefore(sideRightLarge, sideLeftLarge.nextSibling);
        solidLarge.insertBefore(sideFrontLarge, sideRightLarge.nextSibling);

        function createPolygon(points, id) {
            const poly = document.createElementNS('http://www.w3.org/2000/svg', 'polygon');
            poly.setAttribute('points', points);
            poly.setAttribute('id', id);
            poly.setAttribute('fill', 'none');
            poly.setAttribute('stroke', 'none');
            return poly;
        }

        function show(el) {
            el.style.display = 'block';
            el.style.opacity = '1';
        }

        function hide(el) {
            el.style.display = 'none';
            el.style.opacity = '0';
        }

        function setOpacity(el, val) {
            el.style.opacity = val;
        }

        function updateScene() {
            stepIndicator.textContent = '步骤 ' + (scene1_step + 1) + '/' + totalSteps;
            btnPrev.disabled = scene1_step === 0;
            btnNext.disabled = scene1_step === totalSteps - 1;

            // 步骤0: 立体图大版 + 说明文字
            if (scene1_step === 0) {
                show(solidLarge);
                hide(solidSmall);
                hide(netGroup);
                show(descGroup);
                hide(highlightTextGroup);
                hide(constraintGroup);

                baseBottomLarge.setAttribute('fill', 'none');
                baseTopLarge.setAttribute('fill', 'none');
                sideFrontLarge.setAttribute('fill', 'none');
                sideRightLarge.setAttribute('fill', 'none');
                sideLeftLarge.setAttribute('fill', 'none');
            }
            // 步骤1: 高亮底面和侧面
            else if (scene1_step === 1) {
                show(solidLarge);
                hide(solidSmall);
                hide(netGroup);
                show(descGroup);
                show(highlightTextGroup);
                hide(constraintGroup);

                baseBottomLarge.setAttribute('fill', '#1565c0');
                baseBottomLarge.setAttribute('fill-opacity', '0.2');
                baseTopLarge.setAttribute('fill', '#1565c0');
                baseTopLarge.setAttribute('fill-opacity', '0.2');
                sideFrontLarge.setAttribute('fill', '#2e7d32');
                sideFrontLarge.setAttribute('fill-opacity', '0.15');
                sideRightLarge.setAttribute('fill', '#2e7d32');
                sideRightLarge.setAttribute('fill-opacity', '0.15');
                sideLeftLarge.setAttribute('fill', '#2e7d32');
                sideLeftLarge.setAttribute('fill-opacity', '0.15');
            }
            // 步骤2: 立体图小版 + 展开图
            else if (scene1_step === 2) {
                hide(solidLarge);
                show(solidSmall);
                show(netGroup);
                hide(descGroup);
                hide(highlightTextGroup);
                hide(constraintGroup);
            }
            // 步骤3: 约束条件
            else if (scene1_step === 3) {
                hide(solidLarge);
                show(solidSmall);
                setOpacity(solidSmall, '0.3');
                show(netGroup);
                hide(descGroup);
                hide(highlightTextGroup);
                show(constraintGroup);
            }
        }

        btnPrev.addEventListener('click', function() {
            if (scene1_step > 0) {
                scene1_step--;
                updateScene();
            }
        });

        btnNext.addEventListener('click', function() {
            if (scene1_step < totalSteps - 1) {
                scene1_step++;
                updateScene();
            }
        });

        // 初始化
        updateScene();
    })();
    </script>
