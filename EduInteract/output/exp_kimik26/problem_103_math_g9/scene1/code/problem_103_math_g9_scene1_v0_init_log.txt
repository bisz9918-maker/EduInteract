    <script>
    (function() {
        // Scene 1 实现
        let currentStep = 0;
        const totalSteps = 6;

        const btnPrev = document.getElementById('scene1-btn-prev');
        const btnNext = document.getElementById('scene1-btn-next');
        const stepIndicator = document.getElementById('scene1-step-indicator');

        // 获取各步骤元素
        const elLinePf = document.getElementById('scene1-line-pf');
        const elLabelF = document.getElementById('scene1-label-f');
        const elAnnoSide5 = document.getElementById('scene1-anno-side-5');
        const elAnnoPf4 = document.getElementById('scene1-anno-pf-4');
        const elAnnoBc5 = document.getElementById('scene1-anno-bc-5');

        const elTriBfp = document.getElementById('scene1-tri-bfp');
        const elRightF = document.getElementById('scene1-right-f');
        const elAnnoBf3 = document.getElementById('scene1-anno-bf-3');
        const elAnnoBp5 = document.getElementById('scene1-anno-bp-5');
        const elAnnoPq5 = document.getElementById('scene1-anno-pq-5');
        const elAnnoQc5 = document.getElementById('scene1-anno-qc-5');

        const elRectPfce = document.getElementById('scene1-rect-pfce');
        const elRightC = document.getElementById('scene1-right-c');
        const elRightE = document.getElementById('scene1-right-e');
        const elAnnoPe2 = document.getElementById('scene1-anno-pe-2');
        const elAnnoEc4 = document.getElementById('scene1-anno-ec-4');
        const elAnnoFc2 = document.getElementById('scene1-anno-fc-2');

        const elTrapPbce = document.getElementById('scene1-trap-pbce');

        const elSquare = document.getElementById('scene1-square-abcd');
        const elRhombus = document.getElementById('scene1-rhombus-pqcb');
        const elShadow = document.getElementById('scene1-shadow-abped');

        function setActive(el, active) {
            if (!el) return;
            if (active) {
                el.classList.add('active');
                el.style.display = '';
                el.style.opacity = '1';
            } else {
                el.classList.remove('active');
                el.style.display = 'none';
                el.style.opacity = '0';
            }
        }

        function clearHighlights() {
            elSquare.classList.remove('hl-blue-stroke');
            elRhombus.classList.remove('hl-blue-stroke');
            elShadow.classList.remove('hl-blue-stroke');
            elTrapPbce.classList.remove('hl-red-stroke');
        }

        function renderStep() {
            // 更新按钮状态
            btnPrev.disabled = currentStep === 0;
            btnNext.disabled = currentStep === totalSteps - 1;
            stepIndicator.textContent = '步骤 ' + (currentStep + 1) + '/' + totalSteps;

            // 清除所有高亮
            clearHighlights();

            // 隐藏所有步骤专属元素
            setActive(elLinePf, false);
            setActive(elLabelF, false);
            setActive(elAnnoSide5, false);
            setActive(elAnnoPf4, false);
            setActive(elAnnoBc5, false);

            setActive(elTriBfp, false);
            setActive(elRightF, false);
            setActive(elAnnoBf3, false);
            setActive(elAnnoBp5, false);
            setActive(elAnnoPq5, false);
            setActive(elAnnoQc5, false);

            setActive(elRectPfce, false);
            setActive(elRightC, false);
            setActive(elRightE, false);
            setActive(elAnnoPe2, false);
            setActive(elAnnoEc4, false);
            setActive(elAnnoFc2, false);

            setActive(elTrapPbce, false);

            // 更新右侧文字
            for (let i = 0; i < totalSteps; i++) {
                const g = document.getElementById('scene1-text-step' + i);
                if (g) {
                    if (i === currentStep) {
                        g.classList.add('active');
                        g.style.display = 'block';
                    } else {
                        g.classList.remove('active');
                        g.style.display = 'none';
                    }
                }
            }

            // 根据当前步骤显示对应元素
            if (currentStep >= 1) {
                setActive(elLinePf, true);
                setActive(elLabelF, true);
                setActive(elAnnoSide5, true);
                setActive(elAnnoPf4, true);
                setActive(elAnnoBc5, true);
            }

            if (currentStep >= 2) {
                setActive(elTriBfp, true);
                setActive(elRightF, true);
                setActive(elAnnoBf3, true);
                setActive(elAnnoBp5, true);
                setActive(elAnnoPq5, true);
                setActive(elAnnoQc5, true);
            }

            if (currentStep >= 3) {
                setActive(elRectPfce, true);
                setActive(elRightC, true);
                setActive(elRightE, true);
                setActive(elAnnoPe2, true);
                setActive(elAnnoEc4, true);
                setActive(elAnnoFc2, true);
            }

            if (currentStep >= 4) {
                setActive(elTrapPbce, true);
            }

            // 高亮控制
            if (currentStep === 1) {
                elSquare.classList.add('hl-blue-stroke');
            } else if (currentStep === 2) {
                elRhombus.classList.add('hl-blue-stroke');
            } else if (currentStep === 4) {
                elTrapPbce.classList.add('hl-red-stroke');
            } else if (currentStep === 5) {
                elShadow.classList.add('hl-blue-stroke');
            }
        }

        btnPrev.addEventListener('click', function() {
            if (currentStep > 0) {
                currentStep--;
                renderStep();
            }
        });

        btnNext.addEventListener('click', function() {
            if (currentStep < totalSteps - 1) {
                currentStep++;
                renderStep();
            }
        });

        // 初始渲染
        renderStep();
    })();
    </script>
