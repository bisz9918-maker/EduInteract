    <script>
    (function() {
        // Scene 2 实现
        let currentStep = 0; // 0,1,2,3
        let currentView = 'counterexample'; // 'counterexample' or 'special', only valid at step 3

        const btnPrev = document.getElementById('scene2-btn-prev');
        const btnNext = document.getElementById('scene2-btn-next');
        const btnToggle = document.getElementById('scene2-btn-toggle');
        const stepIndicator = document.getElementById('scene2-step-indicator');

        function setOpacity(id, val) {
            const el = document.getElementById(id);
            if (el) el.setAttribute('opacity', val);
        }
        function setDisplay(id, val) {
            const el = document.getElementById(id);
            if (el) el.style.display = val;
        }

        function render() {
            btnPrev.disabled = (currentStep === 0);
            btnNext.disabled = (currentStep === 3);
            stepIndicator.textContent = '步骤 ' + (currentStep + 1) + '/4';

            if (currentStep === 3) {
                setDisplay('scene2-btn-toggle', 'inline-block');
                btnToggle.textContent = (currentView === 'counterexample') ? '查看特例' : '查看反例';
            } else {
                setDisplay('scene2-btn-toggle', 'none');
            }

            // 信息面板 HTML div 显隐
            setDisplay('scene2-info-step0', (currentStep === 0) ? 'block' : 'none');
            setDisplay('scene2-info-step1', (currentStep === 1) ? 'block' : 'none');
            setDisplay('scene2-info-step2', (currentStep === 2) ? 'block' : 'none');
            if (currentStep === 3) {
                setDisplay('scene2-info-step3-counter', (currentView === 'counterexample') ? 'block' : 'none');
                setDisplay('scene2-info-step3-special', (currentView === 'special') ? 'block' : 'none');
            } else {
                setDisplay('scene2-info-step3-counter', 'none');
                setDisplay('scene2-info-step3-special', 'none');
            }

            // 步骤1元素
            if (currentStep >= 1) {
                setOpacity('scene2-arc-bmn', '1');
                setOpacity('scene2-label-x-bmn', '1');
                setOpacity('scene2-arc-amb', '1');
                setOpacity('scene2-label-x-amb', '1');
            } else {
                setOpacity('scene2-arc-bmn', '0');
                setOpacity('scene2-label-x-bmn', '0');
                setOpacity('scene2-arc-amb', '0');
                setOpacity('scene2-label-x-amb', '0');
            }

            // 步骤2元素
            if (currentStep >= 2) {
                setOpacity('scene2-point-o', '1');
                setOpacity('scene2-label-o', '1');
                setOpacity('scene2-line-om', '1');
                setOpacity('scene2-line-on', '1');
                setOpacity('scene2-fill-omn', '1');
                setOpacity('scene2-arc-omn', '1');
                setOpacity('scene2-label-2x', '1');
                setOpacity('scene2-arc-onm', '1');
                setOpacity('scene2-label-2y', '1');
                setOpacity('scene2-arc-mon', '1');
                setOpacity('scene2-label-60', '1');
            } else {
                setOpacity('scene2-point-o', '0');
                setOpacity('scene2-label-o', '0');
                setOpacity('scene2-line-om', '0');
                setOpacity('scene2-line-on', '0');
                setOpacity('scene2-fill-omn', '0');
                setOpacity('scene2-arc-omn', '0');
                setOpacity('scene2-label-2x', '0');
                setOpacity('scene2-arc-onm', '0');
                setOpacity('scene2-label-2y', '0');
                setOpacity('scene2-arc-mon', '0');
                setOpacity('scene2-label-60', '0');
            }

            // 步骤3元素
            if (currentStep >= 3) {
                setOpacity('scene2-line-oa', '1');
                setOpacity('scene2-label-oa', '1');
                setOpacity('scene2-line-ob', '1');
                setOpacity('scene2-label-ob', '1');
                const oaLine = document.getElementById('scene2-line-oa');
                const obLine = document.getElementById('scene2-line-ob');
                const oaLabel = document.getElementById('scene2-label-oa');
                const obLabel = document.getElementById('scene2-label-ob');
                if (currentView === 'counterexample') {
                    oaLine.setAttribute('stroke', '#1565c0');
                    obLine.setAttribute('stroke', '#c62828');
                    oaLabel.setAttribute('fill', '#1565c0');
                    obLabel.setAttribute('fill', '#c62828');
                } else {
                    oaLine.setAttribute('stroke', '#2e7d32');
                    obLine.setAttribute('stroke', '#2e7d32');
                    oaLabel.setAttribute('fill', '#2e7d32');
                    obLabel.setAttribute('fill', '#2e7d32');
                }
            } else {
                setOpacity('scene2-line-oa', '0');
                setOpacity('scene2-label-oa', '0');
                setOpacity('scene2-line-ob', '0');
                setOpacity('scene2-label-ob', '0');
            }
        }

        btnPrev.addEventListener('click', function() {
            if (currentStep > 0) {
                currentStep--;
                render();
            }
        });
        btnNext.addEventListener('click', function() {
            if (currentStep < 3) {
                currentStep++;
                render();
            }
        });
        btnToggle.addEventListener('click', function() {
            if (currentStep === 3) {
                currentView = (currentView === 'counterexample') ? 'special' : 'counterexample';
                render();
            }
        });

        render();
    })();
    </script>
