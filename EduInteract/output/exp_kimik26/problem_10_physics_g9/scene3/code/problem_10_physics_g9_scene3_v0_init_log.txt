    <script>
    (function() {
        // Scene 3 实现
        var currentStep = 0;
        var totalSteps = 6; // 0~5

        var btnPrev = document.getElementById('scene3-btn-prev');
        var btnNext = document.getElementById('scene3-btn-next');
        var indicator = document.getElementById('scene3-step-indicator');
        var labelRho1 = document.getElementById('scene3-label-rho1');

        // 步骤分组元素
        var stepGroups = [
            null, // step0 没有额外分组
            document.getElementById('scene3-step1-group'),
            document.getElementById('scene3-step2-group'),
            document.getElementById('scene3-step3-group'),
            document.getElementById('scene3-step4-group'),
            document.getElementById('scene3-step5-group')
        ];

        function renderStep() {
            // 更新步骤指示器
            indicator.textContent = '步骤 ' + (currentStep + 1) + ' / ' + totalSteps;

            // 更新按钮状态
            btnPrev.disabled = (currentStep === 0);
            btnNext.disabled = (currentStep === totalSteps - 1);

            // 控制 ρ₁ 颜色：步骤1及以上变为蓝色高亮
            if (currentStep >= 1) {
                labelRho1.setAttribute('fill', '#1565c0');
                labelRho1.setAttribute('font-weight', 'bold');
            } else {
                labelRho1.setAttribute('fill', '#000');
                labelRho1.setAttribute('font-weight', 'normal');
            }

            // 控制各步骤分组的显隐 (使用 display:none 彻底移除布局)
            for (var i = 1; i < stepGroups.length; i++) {
                if (stepGroups[i]) {
                    stepGroups[i].style.display = (i <= currentStep) ? 'block' : 'none';
                }
            }
        }

        btnPrev.addEventListener('click', function() {
            if (currentStep > 0) {
                currentStep--;
                renderStep();
            }
        });

        btnNext.addEventListener('click', function() {
            if (currentStep < totalSteps - 1) {
                currentStep++;
                renderStep();
            }
        });

        // 初始渲染
        renderStep();
    })();
    </script>
