    <script>
    (function() {
        // Scene 1 实现 - 台球反弹入袋角度推导
        let currentStep = 0;
        const totalSteps = 4;

        // 元素引用
        const els = {
            incidentPath: document.getElementById('scene1-incident-path'),
            reflectPath: document.getElementById('scene1-reflect-path'),
            targetPath: document.getElementById('scene1-target-path'),
            normalLine: document.getElementById('scene1-normal-line'),
            angle1Arc: document.getElementById('scene1-angle1-arc'),
            angle1Fill: document.getElementById('scene1-angle1-fill'),
            angle2Arc: document.getElementById('scene1-angle2-arc'),
            angle2Fill: document.getElementById('scene1-angle2-fill'),
            angle3Arc: document.getElementById('scene1-angle3-arc'),
            angle3Fill: document.getElementById('scene1-angle3-fill'),
            rightAngle: document.getElementById('scene1-right-angle'),
            labelO: document.getElementById('scene1-label-o'),
            labelNormal: document.getElementById('scene1-label-normal'),
            labelAngle1: document.getElementById('scene1-label-angle1'),
            labelAngle2: document.getElementById('scene1-label-angle2'),
            labelAngle3: document.getElementById('scene1-label-angle3'),
            deriveText: document.getElementById('scene1-derive-text'),
            conclusion1: document.getElementById('scene1-conclusion-1'),
            conclusion2: document.getElementById('scene1-conclusion-2'),
            conclusion3: document.getElementById('scene1-conclusion-3'),
            conclusion4: document.getElementById('scene1-conclusion-4'),
            prevBtn: document.getElementById('scene1-prev-btn'),
            nextBtn: document.getElementById('scene1-next-btn'),
            resetBtn: document.getElementById('scene1-reset-btn'),
            stepIndicator: document.getElementById('scene1-step-indicator')
        };

        function show(el) {
            if (!el) return;
            el.style.display = 'inline';
            el.style.opacity = '1';
        }

        function hide(el) {
            if (!el) return;
            el.style.display = 'none';
            el.style.opacity = '0';
        }

        function setOpacity(el, opacity) {
            if (!el) return;
            el.style.opacity = String(opacity);
            if (opacity > 0) el.style.display = 'inline';
        }

        function setStrokeWidth(el, width) {
            if (!el) return;
            el.setAttribute('stroke-width', String(width));
        }

        function clearConclusions() {
            els.conclusion1.textContent = '';
            els.conclusion2.textContent = '';
            els.conclusion3.textContent = '';
            els.conclusion4.textContent = '';
            els.deriveText.textContent = '';
            hide(els.conclusion1);
            hide(els.conclusion2);
            hide(els.conclusion3);
            hide(els.conclusion4);
            hide(els.deriveText);
        }

        function applyStep(step) {
            clearConclusions();

            // 默认状态：所有动态元素先隐藏
            hide(els.incidentPath);
            hide(els.reflectPath);
            hide(els.normalLine);
            hide(els.angle1Arc);
            hide(els.angle1Fill);
            hide(els.angle2Arc);
            hide(els.angle2Fill);
            hide(els.rightAngle);
            hide(els.labelO);
            hide(els.labelNormal);
            hide(els.labelAngle1);
            hide(els.labelAngle2);

            // 默认显示且高亮的元素
            show(els.targetPath);
            setOpacity(els.targetPath, 1);
            setStrokeWidth(els.targetPath, 2);
            show(els.angle3Arc);
            show(els.angle3Fill);
            setOpacity(els.angle3Arc, 1);
            setOpacity(els.angle3Fill, 1);
            setStrokeWidth(els.angle3Arc, 1.5);
            show(els.labelAngle3);
            els.labelAngle3.textContent = '∠3';

            if (step === 0) {
                // 步骤0：题目条件
                els.labelAngle3.textContent = '∠3 = 30°';
                setStrokeWidth(els.targetPath, 3);
                setStrokeWidth(els.angle3Arc, 3);
            } else if (step === 1) {
                // 步骤1：入射路径与法线
                show(els.incidentPath);
                show(els.normalLine);
                show(els.angle1Arc);
                show(els.angle1Fill);
                show(els.labelO);
                show(els.labelNormal);
                show(els.labelAngle1);
                els.labelAngle3.textContent = '∠3 = 30°';

                setStrokeWidth(els.incidentPath, 3);
                setStrokeWidth(els.normalLine, 2.5);
                setStrokeWidth(els.angle1Arc, 3);
                setOpacity(els.targetPath, 0.4);
                setOpacity(els.angle3Arc, 0.4);
                setOpacity(els.angle3Fill, 0.4);
            } else if (step === 2) {
                // 步骤2：反射定律
                show(els.incidentPath);
                show(els.reflectPath);
                show(els.normalLine);
                show(els.angle1Arc);
                show(els.angle1Fill);
                show(els.angle2Arc);
                show(els.angle2Fill);
                show(els.labelO);
                show(els.labelNormal);
                show(els.labelAngle1);
                show(els.labelAngle2);
                els.labelAngle3.textContent = '∠3 = 30°';
                show(els.deriveText);
                els.deriveText.textContent = '反射定律：∠1 = ∠2';

                setStrokeWidth(els.incidentPath, 3);
                setStrokeWidth(els.reflectPath, 3);
                setStrokeWidth(els.angle1Arc, 3);
                setStrokeWidth(els.angle2Arc, 3);
                setOpacity(els.targetPath, 0.4);
                setOpacity(els.normalLine, 0.4);
                setOpacity(els.angle3Arc, 0.4);
                setOpacity(els.angle3Fill, 0.4);
            } else if (step === 3) {
                // 步骤3：角度关系
                show(els.incidentPath);
                show(els.reflectPath);
                show(els.normalLine);
                show(els.angle1Arc);
                show(els.angle1Fill);
                show(els.angle2Arc);
                show(els.angle2Fill);
                show(els.angle3Arc);
                show(els.angle3Fill);
                show(els.rightAngle);
                show(els.labelO);
                show(els.labelNormal);
                show(els.labelAngle1);
                show(els.labelAngle2);
                els.labelAngle3.textContent = '∠3 = 30°';
                show(els.deriveText);
                els.deriveText.textContent = '∠2 + ∠3 = 90°';

                setStrokeWidth(els.normalLine, 3);
                setStrokeWidth(els.angle2Arc, 3);
                setStrokeWidth(els.angle3Arc, 3);
                setOpacity(els.incidentPath, 0.4);
                setOpacity(els.reflectPath, 0.4);
                setOpacity(els.angle1Arc, 0.4);
                setOpacity(els.angle1Fill, 0.4);
                setOpacity(els.targetPath, 0.4);
            } else if (step === 4) {
                // 步骤4：求解结论
                show(els.incidentPath);
                show(els.reflectPath);
                show(els.targetPath);
                show(els.normalLine);
                show(els.angle1Arc);
                show(els.angle1Fill);
                show(els.angle2Arc);
                show(els.angle2Fill);
                show(els.angle3Arc);
                show(els.angle3Fill);
                show(els.rightAngle);
                show(els.labelO);
                show(els.labelNormal);
                show(els.labelAngle1);
                show(els.labelAngle2);
                els.labelAngle3.textContent = '∠3 = 30°';

                show(els.conclusion1);
                show(els.conclusion2);
                show(els.conclusion3);
                show(els.conclusion4);
                els.conclusion1.textContent = '∵ ∠2 + ∠3 = 90°，∠3 = 30°';
                els.conclusion2.textContent = '∴ ∠2 = 90° − 30° = 60°';
                els.conclusion3.textContent = '由反射定律：∠1 = ∠2 = 60°';
                els.conclusion4.textContent = '答案：B';

                setStrokeWidth(els.targetPath, 2);
                setStrokeWidth(els.angle3Arc, 2);
            }

            // 更新按钮状态
            els.prevBtn.disabled = step === 0;
            els.nextBtn.disabled = step === totalSteps;
            els.stepIndicator.textContent = '步骤 ' + step + '/' + totalSteps;
        }

        function nextStep() {
            if (currentStep < totalSteps) {
                currentStep++;
                applyStep(currentStep);
            }
        }

        function prevStep() {
            if (currentStep > 0) {
                currentStep--;
                applyStep(currentStep);
            }
        }

        function resetStep() {
            currentStep = 0;
            applyStep(currentStep);
        }

        // 事件绑定
        els.nextBtn.addEventListener('click', nextStep);
        els.prevBtn.addEventListener('click', prevStep);
        els.resetBtn.addEventListener('click', resetStep);

        // 初始化
        applyStep(0);
    })();
    </script>
</body>
