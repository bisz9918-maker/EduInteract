    <script>
    (function() {
        // Scene 1 实现
        var currentStep = 0;
        var totalSteps = 5;

        var btnPrev = document.getElementById('scene1-btn-prev');
        var btnNext = document.getElementById('scene1-btn-next');
        var stepText = document.getElementById('scene1-step-text');

        var stepNames = [
            '步骤 0/5：题目条件',
            '步骤 1/5：垂直平分线',
            '步骤 2/5：角度推导',
            '步骤 3/5：相似与等腰',
            '步骤 4/5：设元列方程',
            '步骤 5/5：求解 cosA'
        ];

        // 各步骤显示的元素ID列表（累积显示）
        var stepElements = {
            0: ['scene1-label-ab4', 'scene1-label-ac4', 'scene1-label-c72', 'scene1-arc-c', 'scene1-right-angle'],
            1: ['scene1-line-be', 'scene1-label-eaeb'],
            2: ['scene1-arc-a', 'scene1-label-a36', 'scene1-arc-abc', 'scene1-label-abc72',
                'scene1-arc-abe', 'scene1-label-abe36', 'scene1-arc-ebc', 'scene1-label-ebc36',
                'scene1-arc-bec', 'scene1-label-bec72'],
            3: ['scene1-fill-abc', 'scene1-fill-ebc', 'scene1-label-ebbc', 'scene1-label-similar'],
            4: ['scene1-label-aex', 'scene1-label-ebx', 'scene1-label-ec4x', 'scene1-label-bcx',
                'scene1-label-ratio', 'scene1-label-equation'],
            5: ['scene1-fill-ade', 'scene1-label-ad2', 'scene1-label-cos1', 'scene1-label-cos2',
                'scene1-label-cos3', 'scene1-label-ans']
        };

        // 步骤专属元素：仅在特定步骤显示，后续步骤隐藏（避免右侧推导区重叠）
        var stepOnlyElements = {
            3: ['scene1-label-similar'],
            4: ['scene1-label-ratio', 'scene1-label-equation'],
            5: ['scene1-label-cos1', 'scene1-label-cos2', 'scene1-label-cos3', 'scene1-label-ans']
        };

        // 高亮配置：步骤 -> {元素ID: 颜色类}
        var stepHighlights = {
            1: {
                'scene1-line-de': '#c62828',
                'scene1-right-angle': '#c62828'
            }
        };

        // 原始样式映射（用于恢复）
        var originalStyles = {
            'scene1-line-de': '#000',
            'scene1-right-angle': '#1565c0'
        };

        function show(el) {
            if (!el) return;
            el.classList.remove('scene1-hidden');
            el.classList.add('scene1-visible');
        }

        function hide(el) {
            if (!el) return;
            el.classList.remove('scene1-visible');
            el.classList.add('scene1-hidden');
        }

        function setStroke(el, color) {
            if (!el) return;
            el.setAttribute('stroke', color);
        }

        function renderStep() {
            stepText.textContent = stepNames[currentStep];
            btnPrev.disabled = currentStep === 0;
            btnNext.disabled = currentStep === totalSteps;

            // 处理各步骤元素的显隐（累积显示）
            for (var step = 0; step <= totalSteps; step++) {
                var ids = stepElements[step] || [];
                ids.forEach(function(id) {
                    var el = document.getElementById(id);
                    if (step <= currentStep) {
                        show(el);
                    } else {
                        hide(el);
                    }
                });
            }

            // 处理步骤专属元素：只在对应步骤显示
            Object.keys(stepOnlyElements).forEach(function(stepStr) {
                var step = parseInt(stepStr);
                var ids = stepOnlyElements[step];
                ids.forEach(function(id) {
                    var el = document.getElementById(id);
                    if (step === currentStep) {
                        show(el);
                    } else {
                        hide(el);
                    }
                });
            });

            // 处理高亮
            Object.keys(originalStyles).forEach(function(id) {
                var el = document.getElementById(id);
                if (el) setStroke(el, originalStyles[id]);
            });

            for (var s = 0; s <= currentStep; s++) {
                var hl = stepHighlights[s];
                if (hl) {
                    Object.keys(hl).forEach(function(id) {
                        var el = document.getElementById(id);
                        if (el) setStroke(el, hl[id]);
                    });
                }
            }
        }

        btnPrev.addEventListener('click', function() {
            if (currentStep > 0) {
                currentStep--;
                renderStep();
            }
        });

        btnNext.addEventListener('click', function() {
            if (currentStep < totalSteps) {
                currentStep++;
                renderStep();
            }
        });

        // 初始渲染
        renderStep();
    })();
    </script>
