    <script>
    (function() {
        // Scene 1 实现：极地经纬网方位判读
        // 状态机
        let currentStep = 0;
        const totalSteps = 5;

        // 元素引用
        const els = {
            meridians: {
                '140e': document.getElementById('scene1-meridian-140e'),
                '160e': document.getElementById('scene1-meridian-160e'),
                '180': document.getElementById('scene1-meridian-180'),
                '160w': document.getElementById('scene1-meridian-160w'),
                '140w': document.getElementById('scene1-meridian-140w'),
                '120w': document.getElementById('scene1-meridian-120w')
            },
            parallels: {
                '60n': document.getElementById('scene1-parallel-60n'),
                '40n': document.getElementById('scene1-parallel-40n'),
                '20n': document.getElementById('scene1-parallel-20n')
            },
            pointA: document.getElementById('scene1-point-a'),
            pointB: document.getElementById('scene1-point-b'),
            highlightA: document.getElementById('scene1-highlight-a'),
            highlightB: document.getElementById('scene1-highlight-b'),
            arrowNorth: document.getElementById('scene1-arrow-north'),
            labelNorth: document.getElementById('scene1-label-north'),
            arrowSouth: document.getElementById('scene1-arrow-south'),
            labelSouth: document.getElementById('scene1-label-south'),
            rotationArc: document.getElementById('scene1-rotation-arc'),
            labelRotation: document.getElementById('scene1-label-rotation'),
            arrowWest: document.getElementById('scene1-arrow-west'),
            labelWest: document.getElementById('scene1-label-west'),
            crossN: document.getElementById('scene1-cross-n'),
            crossS: document.getElementById('scene1-cross-s'),
            crossE: document.getElementById('scene1-cross-e'),
            crossW: document.getElementById('scene1-cross-w'),
            crossLabelN: document.getElementById('scene1-cross-label-n'),
            crossLabelS: document.getElementById('scene1-cross-label-s'),
            crossLabelE: document.getElementById('scene1-cross-label-e'),
            crossLabelW: document.getElementById('scene1-cross-label-w'),
            arrowSouthwest: document.getElementById('scene1-arrow-southwest'),
            noteStep1A: document.getElementById('scene1-note-step1-a'),
            noteStep1B: document.getElementById('scene1-note-step1-b'),
            noteStep2: document.getElementById('scene1-note-step2'),
            noteStep3: document.getElementById('scene1-note-step3'),
            noteStep4: document.getElementById('scene1-note-step4'),
            conclusionText: document.getElementById('scene1-conclusion-text'),
            btnPrev: document.getElementById('scene1-btn-prev'),
            btnNext: document.getElementById('scene1-btn-next'),
            stepIndicator: document.getElementById('scene1-step-indicator')
        };

        // 工具函数：设置元素显示/隐藏
        function show(el) {
            if (el) {
                el.style.display = '';
                el.setAttribute('opacity', '1');
            }
        }
        function hide(el) {
            if (el) {
                el.setAttribute('opacity', '0');
            }
        }

        // 工具函数：设置SVG元素的class
        function setClass(el, className) {
            if (el) el.setAttribute('class', className);
        }

        // 渲染当前步骤
        function renderStep() {
            // 更新步骤指示器
            els.stepIndicator.textContent = '步骤 ' + currentStep + '/' + totalSteps;
            els.btnPrev.disabled = currentStep === 0;
            els.btnNext.disabled = currentStep === totalSteps;

            // 重置所有元素到默认状态
            // 经线
            setClass(els.meridians['140e'], 'meridian-normal');
            setClass(els.meridians['160e'], 'meridian-normal');
            setClass(els.meridians['180'], 'meridian-normal');
            setClass(els.meridians['160w'], 'meridian-normal');
            setClass(els.meridians['140w'], 'meridian-normal');
            setClass(els.meridians['120w'], 'meridian-normal');

            // 纬线
            setClass(els.parallels['60n'], 'parallel-normal');
            setClass(els.parallels['40n'], 'parallel-normal');
            setClass(els.parallels['20n'], 'parallel-normal');

            // 点
            setClass(els.pointA, 'point-normal');
            els.pointA.setAttribute('r', '4');
            setClass(els.pointB, 'point-normal');
            els.pointB.setAttribute('r', '4');

            // 隐藏所有步骤专属元素
            hide(els.highlightA);
            hide(els.highlightB);
            hide(els.arrowNorth);
            hide(els.labelNorth);
            hide(els.arrowSouth);
            hide(els.labelSouth);
            hide(els.rotationArc);
            hide(els.labelRotation);
            hide(els.arrowWest);
            hide(els.labelWest);
            hide(els.crossN);
            hide(els.crossS);
            hide(els.crossE);
            hide(els.crossW);
            hide(els.crossLabelN);
            hide(els.crossLabelS);
            hide(els.crossLabelE);
            hide(els.crossLabelW);
            hide(els.arrowSouthwest);
            hide(els.noteStep1A);
            hide(els.noteStep1B);
            hide(els.noteStep2);
            hide(els.noteStep3);
            hide(els.noteStep4);
            hide(els.conclusionText);

            // 根据步骤设置状态
            switch (currentStep) {
                case 0:
                    // 步骤0：初始状态，全部默认
                    break;

                case 1:
                    // 步骤1：定位两点
                    setClass(els.meridians['140e'], 'meridian-dim');
                    setClass(els.meridians['160w'], 'meridian-dim');
                    setClass(els.meridians['140w'], 'meridian-dim');
                    setClass(els.meridians['120w'], 'meridian-dim');
                    setClass(els.meridians['160e'], 'meridian-red');
                    setClass(els.meridians['180'], 'meridian-blue');
                    setClass(els.parallels['40n'], 'parallel-dim');
                    setClass(els.parallels['20n'], 'parallel-red');
                    setClass(els.parallels['60n'], 'parallel-red');
                    setClass(els.pointA, 'point-red');
                    els.pointA.setAttribute('r', '5');
                    setClass(els.pointB, 'point-blue');
                    els.pointB.setAttribute('r', '5');
                    show(els.highlightA);
                    show(els.highlightB);
                    show(els.noteStep1A);
                    show(els.noteStep1B);
                    break;

                case 2:
                    // 步骤2：确定北极方向
                    setClass(els.meridians['140e'], 'meridian-dim');
                    setClass(els.meridians['160w'], 'meridian-dim');
                    setClass(els.meridians['140w'], 'meridian-dim');
                    setClass(els.meridians['120w'], 'meridian-dim');
                    setClass(els.meridians['160e'], 'meridian-red');
                    setClass(els.meridians['180'], 'meridian-blue');
                    setClass(els.parallels['40n'], 'parallel-dim');
                    setClass(els.parallels['20n'], 'parallel-red');
                    setClass(els.parallels['60n'], 'parallel-red');
                    setClass(els.pointA, 'point-red');
                    els.pointA.setAttribute('r', '5');
                    setClass(els.pointB, 'point-blue');
                    els.pointB.setAttribute('r', '5');
                    show(els.highlightA);
                    show(els.highlightB);
                    show(els.noteStep1A);
                    show(els.noteStep1B);
                    setClass(els.parallels['60n'], 'parallel-green');
                    setClass(els.parallels['40n'], 'parallel-green');
                    setClass(els.parallels['20n'], 'parallel-green');
                    show(els.arrowNorth);
                    show(els.labelNorth);
                    show(els.noteStep2);
                    break;

                case 3:
                    // 步骤3：判断南北方向
                    setClass(els.meridians['140e'], 'meridian-dim');
                    setClass(els.meridians['160w'], 'meridian-dim');
                    setClass(els.meridians['140w'], 'meridian-dim');
                    setClass(els.meridians['120w'], 'meridian-dim');
                    setClass(els.meridians['160e'], 'meridian-red');
                    setClass(els.meridians['180'], 'meridian-blue');
                    setClass(els.parallels['40n'], 'parallel-dim');
                    setClass(els.parallels['20n'], 'parallel-red');
                    setClass(els.parallels['60n'], 'parallel-red');
                    setClass(els.pointA, 'point-red');
                    els.pointA.setAttribute('r', '5');
                    setClass(els.pointB, 'point-blue');
                    els.pointB.setAttribute('r', '5');
                    show(els.highlightA);
                    show(els.highlightB);
                    show(els.noteStep1A);
                    show(els.noteStep1B);
                    show(els.noteStep2);
                    show(els.arrowSouth);
                    show(els.labelSouth);
                    show(els.noteStep3);
                    break;

                case 4:
                    // 步骤4：判断东西方向
                    setClass(els.meridians['140e'], 'meridian-dim');
                    setClass(els.meridians['160w'], 'meridian-dim');
                    setClass(els.meridians['140w'], 'meridian-dim');
                    setClass(els.meridians['120w'], 'meridian-dim');
                    setClass(els.meridians['160e'], 'meridian-red');
                    setClass(els.meridians['180'], 'meridian-blue');
                    setClass(els.parallels['40n'], 'parallel-dim');
                    setClass(els.parallels['20n'], 'parallel-red');
                    setClass(els.parallels['60n'], 'parallel-red');
                    setClass(els.pointA, 'point-red');
                    els.pointA.setAttribute('r', '5');
                    setClass(els.pointB, 'point-blue');
                    els.pointB.setAttribute('r', '5');
                    show(els.highlightA);
                    show(els.highlightB);
                    show(els.noteStep1A);
                    show(els.noteStep1B);
                    show(els.noteStep2);
                    show(els.noteStep3);
                    show(els.rotationArc);
                    show(els.labelRotation);
                    show(els.arrowWest);
                    show(els.labelWest);
                    show(els.noteStep4);
                    break;

                case 5:
                    // 步骤5：综合结论
                    setClass(els.meridians['140e'], 'meridian-dim');
                    setClass(els.meridians['160w'], 'meridian-dim');
                    setClass(els.meridians['140w'], 'meridian-dim');
                    setClass(els.meridians['120w'], 'meridian-dim');
                    setClass(els.meridians['160e'], 'meridian-red');
                    setClass(els.meridians['180'], 'meridian-blue');
                    setClass(els.parallels['40n'], 'parallel-dim');
                    setClass(els.parallels['20n'], 'parallel-red');
                    setClass(els.parallels['60n'], 'parallel-red');
                    setClass(els.pointA, 'point-red');
                    els.pointA.setAttribute('r', '5');
                    setClass(els.pointB, 'point-blue');
                    els.pointB.setAttribute('r', '5');
                    show(els.highlightA);
                    show(els.highlightB);
                    show(els.noteStep1A);
                    show(els.noteStep1B);
                    show(els.noteStep2);
                    show(els.noteStep3);
                    show(els.noteStep4);
                    show(els.crossN);
                    show(els.crossS);
                    show(els.crossE);
                    show(els.crossW);
                    show(els.crossLabelN);
                    show(els.crossLabelS);
                    show(els.crossLabelE);
                    show(els.crossLabelW);
                    show(els.arrowSouthwest);
                    show(els.conclusionText);
                    break;
            }
        }

        // 按钮事件
        els.btnPrev.addEventListener('click', function() {
            if (currentStep > 0) {
                currentStep--;
                renderStep();
            }
        });

        els.btnNext.addEventListener('click', function() {
            if (currentStep < totalSteps) {
                currentStep++;
                renderStep();
            }
        });

        // 初始渲染
        renderStep();
    })();
    </script>
</body>
