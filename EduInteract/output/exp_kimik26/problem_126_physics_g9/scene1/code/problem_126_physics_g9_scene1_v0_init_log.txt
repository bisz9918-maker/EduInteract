    <script>
    (function() {
        let currentStep = 0;

        const STEP_ELEMENTS = [
            [],
            ['scene1-num-1', 'scene1-num-2', 'scene1-num-3', 'scene1-label-n'],
            ['scene1-num-1', 'scene1-num-2', 'scene1-num-3', 'scene1-label-n',
             'scene1-label-s', 'scene1-label-3s', 'scene1-guide-1', 'scene1-guide-2', 'scene1-formula-s'],
            ['scene1-num-1', 'scene1-num-2', 'scene1-num-3', 'scene1-label-n',
             'scene1-label-s', 'scene1-label-3s', 'scene1-guide-1', 'scene1-guide-2', 'scene1-formula-s',
             'scene1-label-v', 'scene1-label-3v', 'scene1-formula-v']
        ];

        const ROPE_IDS = [
            'scene1-rope-seg-1', 'scene1-rope-seg-2', 'scene1-rope-seg-3', 'scene1-rope-seg-4',
            'scene1-rope-arc-1', 'scene1-rope-arc-2', 'scene1-rope-arc-3'
        ];

        const ARROW_S_IDS = ['scene1-arrow-s', 'scene1-arrow-3s'];
        const ARROW_V_IDS = ['scene1-arrow-v', 'scene1-arrow-3v'];

        function setDisplay(id, show) {
            const el = document.getElementById(id);
            if (el) {
                el.style.display = show ? (el.tagName.toLowerCase() === 'text' ? 'inline' : 'block') : 'none';
                el.style.opacity = show ? '1' : '0';
            }
        }

        function setStrokeColor(id, color) {
            const el = document.getElementById(id);
            if (el) el.setAttribute('stroke', color);
        }

        function render() {
            document.getElementById('scene1-step-text').textContent = '步骤 ' + (currentStep + 1) + '/4';
            document.getElementById('scene1-btn-prev').disabled = (currentStep === 0);
            document.getElementById('scene1-btn-next').disabled = (currentStep === 3);

            const ropeColor = (currentStep >= 1) ? '#1565c0' : '#000000';
            ROPE_IDS.forEach(id => setStrokeColor(id, ropeColor));

            const allOptional = [
                'scene1-num-1', 'scene1-num-2', 'scene1-num-3', 'scene1-label-n',
                'scene1-label-s', 'scene1-label-3s', 'scene1-guide-1', 'scene1-guide-2', 'scene1-formula-s',
                'scene1-label-v', 'scene1-label-3v', 'scene1-formula-v'
            ];
            allOptional.forEach(id => setDisplay(id, false));

            if (STEP_ELEMENTS[currentStep]) {
                STEP_ELEMENTS[currentStep].forEach(id => setDisplay(id, true));
            }

            ARROW_S_IDS.forEach(id => setDisplay(id, currentStep >= 2));
            ARROW_V_IDS.forEach(id => setDisplay(id, currentStep >= 3));
        }

        document.getElementById('scene1-btn-next').addEventListener('click', function() {
            if (currentStep < 3) { currentStep++; render(); }
        });

        document.getElementById('scene1-btn-prev').addEventListener('click', function() {
            if (currentStep > 0) { currentStep--; render(); }
        });

        render();
    })();
    </script>
