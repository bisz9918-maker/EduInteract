    <script>
    (function() {
        // Scene 2 实现
        let currentStep = 0;
        const totalSteps = 5;

        // 元素引用
        const els = {
            arrowFrictionGroup: document.getElementById('scene2-arrow-f-friction-group'),
            labelFriction: document.getElementById('scene2-label-f-friction'),
            arrowGLine: document.getElementById('scene2-arrow-g-line'),
            arrowGHead: document.getElementById('scene2-arrow-g-head'),
            labelG: document.getElementById('scene2-label-g'),
            rightAngle: document.getElementById('scene2-right-angle'),
            wgZero: document.getElementById('scene2-wg-zero'),
            arrowSGroup: document.getElementById('scene2-arrow-s-group'),
            labelS: document.getElementById('scene2-label-s'),
            arrowVGroup: document.getElementById('scene2-arrow-v-group'),
            labelV: document.getElementById('scene2-label-v'),
            arrowFLine: document.getElementById('scene2-arrow-f-line'),
            arrowFHead: document.getElementById('scene2-arrow-f-head'),
            labelF: document.getElementById('scene2-label-f'),
            arrow3sGroup: document.getElementById('scene2-arrow-3s-group'),
            label3s: document.getElementById('scene2-label-3s'),
            label3v: document.getElementById('scene2-label-3v'),
            formulaWyou: document.getElementById('scene2-formula-wyou'),
            formulaWzong: document.getElementById('scene2-formula-wzong'),
            formulaP1: document.getElementById('scene2-formula-p1'),
            formulaP2: document.getElementById('scene2-formula-p2'),
            formulaEta1: document.getElementById('scene2-formula-eta1'),
            cancelLine: document.getElementById('scene2-cancel-line'),
            formulaEta2: document.getElementById('scene2-formula-eta2'),
            cardA: document.getElementById('scene2-card-a'),
            cardB: document.getElementById('scene2-card-b'),
            cardC: document.getElementById('scene2-card-c'),
            cardD: document.getElementById('scene2-card-d'),
            btnPrev: document.getElementById('scene2-btn-prev'),
            btnNext: document.getElementById('scene2-btn-next'),
            btnReset: document.getElementById('scene2-btn-reset'),
            stepIndicator: document.getElementById('scene2-step-indicator')
        };

        function setOpacity(el, val) {
            if (el) el.setAttribute('opacity', val);
        }

        function setStroke(el, color) {
            if (el) el.setAttribute('stroke', color);
        }

        function setFill(el, color) {
            if (el) el.setAttribute('fill', color);
        }

        function updateStep() {
            // 更新指示器
            els.stepIndicator.textContent = '步骤 ' + currentStep + ' / ' + totalSteps;
            els.btnPrev.disabled = currentStep === 0;
            els.btnNext.disabled = currentStep === totalSteps;

            // 重置所有动态元素到初始状态
            setOpacity(els.arrowFrictionGroup, 0);
            setOpacity(els.labelFriction, 0);
            setStroke(els.arrowGLine, '#000');
            setFill(els.arrowGHead, '#000');
            setFill(els.labelG, '#000');
            setOpacity(els.rightAngle, 0);
            setOpacity(els.wgZero, 0);
            setOpacity(els.arrowSGroup, 1);
            setOpacity(els.labelS, 1);
            setOpacity(els.arrowVGroup, 0);
            setOpacity(els.labelV, 0);
            setStroke(els.arrowFLine, '#000');
            setFill(els.arrowFHead, '#000');
            setFill(els.labelF, '#000');
            setOpacity(els.arrow3sGroup, 0);
            setOpacity(els.label3s, 0);
            setOpacity(els.label3v, 0);
            setOpacity(els.formulaWyou, 0);
            setOpacity(els.formulaWzong, 0);
            setOpacity(els.formulaP1, 0);
            setOpacity(els.formulaP2, 0);
            setOpacity(els.formulaEta1, 0);
            setOpacity(els.cancelLine, 0);
            setOpacity(els.formulaEta2, 0);
            setOpacity(els.cardA, 0);
            setOpacity(els.cardB, 0);
            setOpacity(els.cardC, 0);
            setOpacity(els.cardD, 0);

            // 根据步骤设置状态
            if (currentStep >= 1) {
                // 步骤1：有用功分析
                setOpacity(els.arrowFrictionGroup, 1);
                setOpacity(els.labelFriction, 1);
                setStroke(els.arrowGLine, '#aaaaaa');
                setFill(els.arrowGHead, '#aaaaaa');
                setFill(els.labelG, '#aaaaaa');
                setOpacity(els.rightAngle, 1);
                setOpacity(els.wgZero, 1);
                setOpacity(els.formulaWyou, 1);
            }

            if (currentStep >= 2) {
                // 步骤2：总功分析
                setStroke(els.arrowFLine, '#1565c0');
                setFill(els.arrowFHead, '#1565c0');
                setFill(els.labelF, '#1565c0');
                setOpacity(els.arrow3sGroup, 1);
                setOpacity(els.label3s, 1);
                setOpacity(els.formulaWzong, 1);
                // 淡化W有
                if (currentStep >= 2) setOpacity(els.formulaWyou, 0.6);
            }

            if (currentStep >= 3) {
                // 步骤3：功率分析
                setOpacity(els.arrowSGroup, 0);
                setOpacity(els.labelS, 0);
                setOpacity(els.arrowVGroup, 1);
                setOpacity(els.labelV, 1);
                setOpacity(els.label3v, 1);
                setOpacity(els.formulaP1, 1);
                setOpacity(els.formulaP2, 1);
                // 淡化W总
                if (currentStep >= 3) setOpacity(els.formulaWzong, 0.6);
            }

            if (currentStep >= 4) {
                // 步骤4：机械效率
                setOpacity(els.formulaEta1, 1);
                setOpacity(els.cancelLine, 1);
                setOpacity(els.formulaEta2, 1);
                // 淡化功率公式
                if (currentStep >= 4) {
                    setOpacity(els.formulaP1, 0.6);
                    setOpacity(els.formulaP2, 0.6);
                }
            }

            if (currentStep >= 5) {
                // 步骤5：选项判断
                setOpacity(els.formulaWyou, 0.3);
                setOpacity(els.formulaWzong, 0.3);
                setOpacity(els.formulaP1, 0.3);
                setOpacity(els.formulaP2, 0.3);
                setOpacity(els.formulaEta1, 0.3);
                setOpacity(els.formulaEta2, 0.3);
                setOpacity(els.cancelLine, 0.3);
                setOpacity(els.cardA, 1);
                setOpacity(els.cardB, 1);
                setOpacity(els.cardC, 1);
                setOpacity(els.cardD, 1);
            }
        }

        function nextStep() {
            if (currentStep < totalSteps) {
                currentStep++;
                updateStep();
            }
        }

        function prevStep() {
            if (currentStep > 0) {
                currentStep--;
                updateStep();
            }
        }

        function resetStep() {
            currentStep = 0;
            updateStep();
        }

        // 事件绑定
        els.btnNext.addEventListener('click', nextStep);
        els.btnPrev.addEventListener('click', prevStep);
        els.btnReset.addEventListener('click', resetStep);

        // 初始化
        updateStep();
    })();
    </script>
