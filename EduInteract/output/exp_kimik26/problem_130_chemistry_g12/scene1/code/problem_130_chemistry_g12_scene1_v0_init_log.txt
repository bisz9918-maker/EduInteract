    <script>
    (function() {
        const STEP_COUNT = 5;
        let currentStep = 0;

        const title = document.getElementById('scene1-title');
        const btnPrev = document.getElementById('scene1-btn-prev');
        const btnNext = document.getElementById('scene1-btn-next');
        const indicator = document.getElementById('scene1-step-indicator');

        const cards = [
            document.getElementById('scene1-card-ch3-1'),
            document.getElementById('scene1-card-ch2-1'),
            document.getElementById('scene1-card-ch-1'),
            document.getElementById('scene1-card-ch3-2'),
            document.getElementById('scene1-card-ch2-2'),
            document.getElementById('scene1-card-cl-1')
        ];

        const centerGroupStep1 = document.getElementById('scene1-center-group-step1');
        const centerGroupStep2 = document.getElementById('scene1-center-group-step2');
        const explainStep1 = document.getElementById('scene1-explain-step1');
        const combos = document.getElementById('scene1-combos');
        const marks = document.getElementById('scene1-marks');
        const mask1 = document.getElementById('scene1-mask-1');
        const bubble = document.getElementById('scene1-bubble-invalid');
        const validLabels = document.getElementById('scene1-valid-labels');

        function show(el) {
            if (!el) return;
            el.style.display = 'block';
            requestAnimationFrame(() => { el.style.opacity = '1'; });
        }
        function hide(el) {
            if (!el) return;
            el.style.opacity = '0';
            setTimeout(() => { if (el.style.opacity === '0') el.style.display = 'none'; }, 300);
        }

        function renderStep() {
            btnPrev.disabled = currentStep === 0;
            btnNext.disabled = currentStep === STEP_COUNT - 1;
            indicator.textContent = '步骤 ' + (currentStep + 1) + '/' + STEP_COUNT;

            if (currentStep === 0 || currentStep === 1) {
                title.textContent = '基团与价键';
            } else if (currentStep === 4) {
                title.textContent = '四种有效方案';
            } else {
                title.textContent = '枚举与排除';
            }

            if (currentStep === 0) {
                cards.forEach(c => { c.style.display = 'flex'; c.style.opacity = '1'; });
                cards.forEach(c => {
                    c.style.filter = '';
                    c.style.borderWidth = '2px';
                });
                hide(centerGroupStep1);
                hide(centerGroupStep2);
                hide(explainStep1);
                hide(combos);
                hide(marks);
                hide(mask1);
                hide(bubble);
                hide(validLabels);
            }

            if (currentStep === 1) {
                cards.forEach((c, idx) => {
                    c.style.display = 'flex';
                    if (idx === 2) {
                        c.style.opacity = '1';
                        c.style.borderWidth = '3px';
                        c.style.filter = 'drop-shadow(0 0 8px #ef6c00)';
                    } else {
                        c.style.opacity = '0.3';
                        c.style.borderWidth = '2px';
                        c.style.filter = '';
                    }
                });
                show(centerGroupStep1);
                hide(centerGroupStep2);
                show(explainStep1);
                hide(combos);
                hide(marks);
                hide(mask1);
                hide(bubble);
                hide(validLabels);
            }

            if (currentStep === 2) {
                cards.forEach(c => hide(c));
                hide(centerGroupStep1);
                show(centerGroupStep2);
                hide(explainStep1);
                show(combos);
                hide(marks);
                hide(mask1);
                hide(bubble);
                hide(validLabels);
            }

            if (currentStep === 3) {
                cards.forEach(c => hide(c));
                hide(centerGroupStep1);
                show(centerGroupStep2);
                hide(explainStep1);
                show(combos);
                show(marks);
                show(mask1);
                show(bubble);
                hide(validLabels);
            }

            if (currentStep === 4) {
                cards.forEach(c => hide(c));
                hide(centerGroupStep1);
                centerGroupStep2.style.display = 'block';
                centerGroupStep2.style.opacity = '0.4';
                hide(explainStep1);
                show(combos);
                show(marks);
                hide(mask1);
                hide(bubble);
                show(validLabels);

                const combo1 = document.getElementById('scene1-combo-1');
                const mark1 = document.getElementById('scene1-mark-1');
                if (combo1) combo1.style.opacity = '0';
                if (mark1) mark1.style.opacity = '0';
            } else {
                const combo1 = document.getElementById('scene1-combo-1');
                const mark1 = document.getElementById('scene1-mark-1');
                if (combo1) combo1.style.opacity = '1';
                if (mark1) mark1.style.opacity = '1';
            }
        }

        btnPrev.addEventListener('click', function() {
            if (currentStep > 0) { currentStep--; renderStep(); }
        });
        btnNext.addEventListener('click', function() {
            if (currentStep < STEP_COUNT - 1) { currentStep++; renderStep(); }
        });

        renderStep();
    })();
    </script>
