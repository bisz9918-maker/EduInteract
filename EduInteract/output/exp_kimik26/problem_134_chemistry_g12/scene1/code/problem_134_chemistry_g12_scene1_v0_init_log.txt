<script>
(function() {
var currentStep = 0;
var totalSteps = 5;
var btnPrev = document.getElementById('scene1-btn-prev');
var btnNext = document.getElementById('scene1-btn-next');
var indicator = document.getElementById('scene1-step-indicator');
var groupStep1 = document.getElementById('scene1-group-step1');
var groupStep2 = document.getElementById('scene1-group-step2');
var groupStep3 = document.getElementById('scene1-group-step3');
var groupStep4 = document.getElementById('scene1-group-step4');
function updateIndicator() { indicator.textContent = '步骤 ' + (currentStep + 1) + ' / ' + totalSteps; }
function renderStep() {
btnPrev.disabled = (currentStep === 0);
btnNext.disabled = (currentStep === totalSteps - 1);
if (currentStep >= 1) { groupStep1.classList.add('active'); } else { groupStep1.classList.remove('active'); }
if (currentStep >= 2) { groupStep2.classList.add('active'); } else { groupStep2.classList.remove('active'); }
if (currentStep >= 3) { groupStep3.classList.add('active'); } else { groupStep3.classList.remove('active'); }
if (currentStep >= 4) { groupStep4.classList.add('active'); } else { groupStep4.classList.remove('active'); }
updateIndicator();
}
btnPrev.addEventListener('click', function() { if (currentStep > 0) { currentStep--; renderStep(); } });
btnNext.addEventListener('click', function() { if (currentStep < totalSteps - 1) { currentStep++; renderStep(); } });
renderStep();
})();
</script>
