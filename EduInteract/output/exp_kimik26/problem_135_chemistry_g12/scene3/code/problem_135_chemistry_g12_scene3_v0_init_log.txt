    <script>
    (function() {
        // Scene 3 实现
        let currentStep = 0;
        const totalSteps = 5;

        const btnPrev = document.getElementById('scene3-btn-prev');
        const btnNext = document.getElementById('scene3-btn-next');
        const stepIndicator = document.getElementById('scene3-step-indicator');

        const curves = {
            a: document.getElementById('scene3-curve-a'),
            b: document.getElementById('scene3-curve-b'),
            c: document.getElementById('scene3-curve-c'),
            d: document.getElementById('scene3-curve-d')
        };

        const htmlCards = {
            a: document.getElementById('scene3-html-card-a'),
            b: document.getElementById('scene3-html-card-b'),
            c: document.getElementById('scene3-html-card-c'),
            d: document.getElementById('scene3-html-card-d')
        };

        const colors = {
            a: '#1565c0',
            b: '#c62828',
            c: '#2e7d32',
            d: '#7b1fa2'
        };

        function resetAll() {
            // 重置所有曲线
            Object.values(curves).forEach(curve => {
                curve.classList.remove('highlight', 'dimmed');
                curve.style.strokeWidth = '2px';
                curve.style.opacity = '1';
            });

            // 重置所有HTML卡片
            Object.values(htmlCards).forEach(card => {
                card.classList.remove('expanded', 'dimmed-card', 'highlight-card');
                card.style.opacity = '1';
                card.style.background = '#ffffff';
            });
        }

        function highlightOption(key) {
            // 高亮对应曲线
            Object.keys(curves).forEach(k => {
                if (k === key) {
                    curves[k].classList.add('highlight');
                    curves[k].classList.remove('dimmed');
                } else {
                    curves[k].classList.remove('highlight');
                    curves[k].classList.add('dimmed');
                }
            });

            // 高亮对应HTML卡片
            Object.keys(htmlCards).forEach(k => {
                const card = htmlCards[k];
                if (k === key) {
                    card.classList.add('expanded', 'highlight-card');
                    card.classList.remove('dimmed-card');
                    card.style.opacity = '1';
                    card.style.background = colors[k] + '1A'; // 10% opacity
                } else {
                    card.classList.remove('expanded', 'highlight-card');
                    card.classList.add('dimmed-card');
                    card.style.opacity = '0.3';
                    card.style.background = '#ffffff';
                }
            });
        }

        function renderStep() {
            resetAll();
            stepIndicator.textContent = '步骤 ' + currentStep + '/' + totalSteps;

            // 更新按钮状态
            btnPrev.disabled = currentStep === 0;
            btnNext.disabled = currentStep === totalSteps;

            switch (currentStep) {
                case 0:
                    // 初始状态：所有曲线和卡片正常显示
                    break;
                case 1:
                    highlightOption('a');
                    break;
                case 2:
                    highlightOption('b');
                    break;
                case 3:
                    highlightOption('c');
                    break;
                case 4:
                    highlightOption('d');
                    break;
                case 5:
                    // 总结回顾：同步骤0
                    break;
            }
        }

        btnPrev.addEventListener('click', function() {
            if (currentStep > 0) {
                currentStep--;
                renderStep();
            }
        });

        btnNext.addEventListener('click', function() {
            if (currentStep < totalSteps) {
                currentStep++;
                renderStep();
            }
        });

        // 初始渲染
        renderStep();
    })();
    </script>
