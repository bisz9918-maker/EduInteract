    <script>
    (function() {
        // ===== 状态变量 =====
        let currentStep = 0;
        const totalSteps = 7;
        let isAnimating = false;

        // ===== DOM 元素缓存 =====
        const btnPrev = document.getElementById('scene2-btn-prev');
        const btnNext = document.getElementById('scene2-btn-next');
        const stepIndicator = document.getElementById('scene2-step-indicator');
        const analysisPanel = document.getElementById('scene2-analysis-panel');
        const zoomBox = document.getElementById('scene2-zoom-box');

        // 高亮环
        const highlights = {};
        for (let i = 1; i <= 9; i++) {
            highlights['c' + i] = document.getElementById('scene2-highlight-c' + i);
        }

        // 价键标签
        const valenceLabels = {};
        for (let i = 1; i <= 9; i++) {
            valenceLabels['c' + i] = document.getElementById('scene2-valence-c' + i);
        }

        // 键组
        const bonds = {
            'h-c1': document.getElementById('scene2-bond-h-c1'),
            'c1-c2': document.getElementById('scene2-bond-c1-c2'),
            'c2-c3': document.getElementById('scene2-bond-c2-c3'),
            'c3-c4': document.getElementById('scene2-bond-c3-c4'),
            'c4-c5': document.getElementById('scene2-bond-c4-c5'),
            'c5-c6': document.getElementById('scene2-bond-c5-c6'),
            'c6-c7': document.getElementById('scene2-bond-c6-c7'),
            'c7-c8': document.getElementById('scene2-bond-c7-c8'),
            'c8-c9': document.getElementById('scene2-bond-c8-c9'),
            'c9-x': document.getElementById('scene2-bond-c9-x'),
        };

        // 引线
        const leads = {
            1: document.getElementById('scene2-lead-1'),
            2: document.getElementById('scene2-lead-2'),
            3: document.getElementById('scene2-lead-3'),
        };

        // 选项
        const options = {
            a: document.getElementById('scene2-option-a'),
            b: document.getElementById('scene2-option-b'),
            c: document.getElementById('scene2-option-c'),
            d: document.getElementById('scene2-option-d'),
        };

        // 放大区
        const zoomBondLines = document.querySelectorAll('#scene2-zoom-bond .scene2-zoom-bond-line');
        const zoomLabel = document.getElementById('scene2-zoom-label');
        const zoomLeft = document.getElementById('scene2-zoom-left');
        const zoomRight = document.getElementById('scene2-zoom-right');

        // ===== 工具函数 =====
        function setBondState(bondKey, state) {
            const group = bonds[bondKey];
            if (!group) return;
            const lines = group.querySelectorAll('line');
            lines.forEach(line => {
                line.setAttribute('class', 'scene2-bond-' + state);
            });
        }

        function setHighlight(atomKey, active) {
            const el = highlights[atomKey];
            if (!el) return;
            if (active) el.classList.add('active');
            else el.classList.remove('active');
        }

        function setValence(atomKey, text, color) {
            const el = valenceLabels[atomKey];
            if (!el) return;
            el.textContent = text;
            el.setAttribute('fill', color);
        }

        function setAnalysis(html) {
            analysisPanel.innerHTML = html;
        }

        function setLead(num, active) {
            const el = leads[num];
            if (!el) return;
            if (active) el.classList.add('active');
            else el.classList.remove('active');
        }

        function setZoom(bondType, label, isX) {
            if (bondType === 'single') {
                zoomBondLines[0].style.display = 'block';
                zoomBondLines[1].style.display = 'none';
                zoomBondLines[2].style.display = 'none';
            } else if (bondType === 'triple') {
                zoomBondLines[0].style.display = 'block';
                zoomBondLines[1].style.display = 'block';
                zoomBondLines[2].style.display = 'block';
            }
            zoomLabel.textContent = label;
            zoomLeft.className = 'scene2-zoom-atom';
            zoomRight.className = isX ? 'scene2-zoom-atom x-atom' : 'scene2-zoom-atom';
        }

        function showZoom(active) {
            if (active) zoomBox.classList.add('active');
            else zoomBox.classList.remove('active');
        }

        function setOption(optKey, state) {
            const opt = options[optKey];
            if (!opt) return;
            opt.classList.remove('wrong', 'correct');
            if (state === 'wrong' || state === 'correct') {
                opt.classList.add(state);
            }
        }

        function resetAll() {
            Object.keys(bonds).forEach(k => setBondState(k, 'pending'));
            Object.keys(highlights).forEach(k => setHighlight(k, false));
            for (let i = 1; i <= 9; i++) {
                setValence('c' + i, '待', '#888');
            }
            Object.keys(leads).forEach(k => setLead(k, false));
            showZoom(false);
            Object.keys(options).forEach(k => setOption(k, 'neutral'));
        }

        // ===== 分步渲染 =====
        function renderStep(step) {
            resetAll();
            stepIndicator.textContent = '步骤 ' + step + '/7';
            btnPrev.disabled = step === 0;
            btnNext.textContent = step === totalSteps ? '重新开始' : '下一步 ▶';

            switch (step) {
                case 0:
                    setAnalysis('从分子左端开始，利用碳四价规则<br>逐步推导每个碳原子的键型');
                    break;

                case 1:
                    setHighlight('c1', true);
                    setValence('c1', '剩3', '#1565c0');
                    setBondState('h-c1', 'done');
                    setBondState('c1-c2', 'current');
                    setAnalysis('C₁已连接1个H原子（单键），<br>剩余3个价键 → 与C₂形成三键');
                    break;

                case 2:
                    setBondState('h-c1', 'done');
                    setBondState('c1-c2', 'done');
                    setBondState('c2-c3', 'current');
                    setHighlight('c2', true);
                    setValence('c1', '✓', '#000');
                    setValence('c2', '剩1', '#1565c0');
                    setAnalysis('C₂已用3键（与C₁三键），<br>剩余1个价键 → 与C₃形成单键');
                    break;

                case 3:
                    setBondState('h-c1', 'done');
                    setBondState('c1-c2', 'done');
                    setBondState('c2-c3', 'done');
                    setBondState('c3-c4', 'current');
                    setHighlight('c3', true);
                    setValence('c1', '✓', '#000');
                    setValence('c2', '✓', '#000');
                    setValence('c3', '剩3', '#1565c0');
                    setAnalysis('C₃已用1键（与C₂单键），<br>剩余3个价键 → 与C₄形成三键（①处）');
                    break;

                case 4:
                    setBondState('h-c1', 'done');
                    setBondState('c1-c2', 'done');
                    setBondState('c2-c3', 'done');
                    setBondState('c3-c4', 'done');
                    setBondState('c4-c5', 'done');
                    setBondState('c5-c6', 'done');
                    setBondState('c6-c7', 'done');
                    setBondState('c7-c8', 'done');
                    setBondState('c8-c9', 'done');
                    setBondState('c9-x', 'done');
