    <script>
    (function() {
        let currentStep = 0;
        const totalSteps = 6;

        const btnPrev = document.getElementById('scene1-btn-prev');
        const btnNext = document.getElementById('scene1-btn-next');
        const stepIndicator = document.getElementById('scene1-step-indicator');

        const groupMetabolism = document.getElementById('scene1-group-metabolism');
        const groupTable = document.getElementById('scene1-group-table');
        const groupConclusion = document.getElementById('scene1-group-conclusion');
        const groupHybridLeft = document.getElementById('scene1-group-hybrid-left');
        const groupHybridRight = document.getElementById('scene1-group-hybrid-right');

        const aLabel = document.getElementById('scene1-a-label');
        const inhibitLabel = document.getElementById('scene1-inhibit-label');
        const inhibitH = document.getElementById('scene1-inhibit-h');
        const inhibitV = document.getElementById('scene1-inhibit-v');
        const inhibitT = document.getElementById('scene1-inhibit-t');
        const bLabel = document.getElementById('scene1-b-label');
        const enzyme1Arrow = document.getElementById('scene1-enzyme1-arrow');
        const enzyme1Label = document.getElementById('scene1-enzyme1-label');
        const enzyme1Gene = document.getElementById('scene1-enzyme1-gene');
        const brownBox = document.getElementById('scene1-brown-box');
        const enzyme2Arrow = document.getElementById('scene1-enzyme2-arrow');
        const enzyme2Label = document.getElementById('scene1-enzyme2-label');
        const enzyme2Gene = document.getElementById('scene1-enzyme2-gene');
        const blackBox = document.getElementById('scene1-black-box');

        function setDisplay(el, show) {
            if (!el) return;
            el.style.display = show ? 'block' : 'none';
        }

        function setOpacity(el, val) {
            if (!el) return;
            el.style.opacity = val;
        }

        function resetAllHighlights() {
            aLabel.setAttribute('fill', '#000000');
            inhibitLabel.setAttribute('fill', '#000000');
            inhibitH.setAttribute('stroke', '#000000');
            inhibitH.setAttribute('stroke-width', '2');
            inhibitV.setAttribute('stroke', '#000000');
            inhibitV.setAttribute('stroke-width', '2');
            inhibitT.setAttribute('stroke', '#000000');
            inhibitT.setAttribute('stroke-width', '2');

            bLabel.setAttribute('fill', '#000000');
            enzyme1Label.setAttribute('fill', '#000000');
            enzyme1Gene.setAttribute('fill', '#000000');
            enzyme1Arrow.setAttribute('stroke', '#000000');
            enzyme1Arrow.setAttribute('stroke-width', '2');
            enzyme1Arrow.setAttribute('marker-end', 'url(#scene1-arrowhead)');
            brownBox.setAttribute('stroke', '#000000');
            brownBox.setAttribute('stroke-width', '2');

            enzyme2Label.setAttribute('fill', '#000000');
            enzyme2Gene.setAttribute('fill', '#000000');
            enzyme2Arrow.setAttribute('stroke', '#000000');
            enzyme2Arrow.setAttribute('stroke-width', '2');
            enzyme2Arrow.setAttribute('marker-end', 'url(#scene1-arrowhead)');
            blackBox.setAttribute('stroke', '#000000');
            blackBox.setAttribute('stroke-width', '2');

            setOpacity(aLabel, 1);
            setOpacity(inhibitLabel, 1);
            setOpacity(inhibitH, 1);
            setOpacity(inhibitV, 1);
            setOpacity(inhibitT, 1);
            setOpacity(bLabel, 1);
            setOpacity(enzyme1Label, 1);
            setOpacity(enzyme1Gene, 1);
            setOpacity(enzyme1Arrow, 1);
            setOpacity(brownBox, 1);
            setOpacity(enzyme2Label, 1);
            setOpacity(enzyme2Gene, 1);
            setOpacity(enzyme2Arrow, 1);
            setOpacity(blackBox, 1);
            setOpacity(groupMetabolism, 1);
        }

        function highlightA() {
            aLabel.setAttribute('fill', '#c62828');
            inhibitLabel.setAttribute('fill', '#c62828');
            inhibitH.setAttribute('stroke', '#c62828');
            inhibitH.setAttribute('stroke-width', '3');
            inhibitV.setAttribute('stroke', '#c62828');
            inhibitV.setAttribute('stroke-width', '3');
            inhibitT.setAttribute('stroke', '#c62828');
            inhibitT.setAttribute('stroke-width', '3');

            setOpacity(enzyme1Arrow, 0.3);
            setOpacity(enzyme1Label, 0.3);
            setOpacity(enzyme1Gene, 0.3);
            setOpacity(bLabel, 0.3);
            setOpacity(brownBox, 0.3);
        }

        function highlightB() {
            bLabel.setAttribute('fill', '#2e7d32');
            enzyme1Label.setAttribute('fill', '#2e7d32');
            enzyme1Gene.setAttribute('fill', '#2e7d32');
            enzyme1Arrow.setAttribute('stroke', '#2e7d32');
            enzyme1Arrow.setAttribute('stroke-width', '3');
            enzyme1Arrow.setAttribute('marker-end', 'url(#scene1-arrowhead-green)');
            brownBox.setAttribute('stroke', '#2e7d32');
            brownBox.setAttribute('stroke-width', '3');
        }

        function highlightC() {
            enzyme2Label.setAttribute('fill', '#1565c0');
            enzyme2Gene.setAttribute('fill', '#1565c0');
            enzyme2Arrow.setAttribute('stroke', '#1565c0');
            enzyme2Arrow.setAttribute('stroke-width', '3');
            enzyme2Arrow.setAttribute('marker-end', 'url(#scene1-arrowhead-blue)');
            blackBox.setAttribute('stroke', '#1565c0');
            blackBox.setAttribute('stroke-width', '3');
        }

        function updateStep(step) {
            currentStep = step;
            stepIndicator.textContent = '步骤 ' + (step + 1) + '/' + totalSteps;
            btnPrev.disabled = step === 0;
            btnNext.disabled = step === totalSteps - 1;

            resetAllHighlights();
            setDisplay(groupMetabolism, true);
            setDisplay(groupTable, false);
            setDisplay(groupConclusion, false);
            setDisplay(groupHybridLeft, false);
            setDisplay(groupHybridRight, false);

            switch (step) {
                case 0:
                    break;
                case 1:
                    highlightA();
                    break;
                case 2:
                    highlightB();
                    break;
                case 3:
                    highlightC();
                    break;
                case 4:
                    setOpacity(groupMetabolism, 0.3);
                    setDisplay(groupTable, true);
                    setDisplay(groupConclusion, true);
                    break;
                case 5:
                    setDisplay(groupMetabolism, false);
                    setDisplay(groupTable, false);
                    setDisplay(groupConclusion, false);
                    setDisplay(groupHybridLeft, true);
                    setDisplay(groupHybridRight, true);
                    break;
            }
        }

        btnPrev.addEventListener('click', function() {
            if (currentStep > 0) updateStep(currentStep - 1);
        });

        btnNext.addEventListener('click', function() {
            if (currentStep < totalSteps - 1) updateStep(currentStep + 1);
        });

        updateStep(0);
    })();
    </script>
