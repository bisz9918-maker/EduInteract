<script>
(function(){
var currentStep=0,totalSteps=6;
var allEdges=['grass-antelope','grass-zebra','grass-mouse','antelope-wilddog','antelope-lion','zebra-wilddog','zebra-lion','wilddog-lion','mouse-snake','snake-snakeeater','snakeeater-lion'];
var allLevelIds=['grass','antelope','zebra','mouse','wilddog','snake','snakeeater','lion'];
var stepConfig=[
{edges:[],levels:{},showLionLevels:false},
{edges:['grass-antelope','antelope-wilddog','wilddog-lion'],levels:{grass:'1',antelope:'2',wilddog:'3',lion:'4'},showLionLevels:false},
{edges:['grass-antelope','antelope-lion'],levels:{grass:'1',antelope:'2',lion:'3'},showLionLevels:false},
{edges:['grass-zebra','zebra-wilddog','wilddog-lion'],levels:{grass:'1',zebra:'2',wilddog:'3',lion:'4'},showLionLevels:false},
{edges:['grass-zebra','zebra-lion'],levels:{grass:'1',zebra:'2',lion:'3'},showLionLevels:false},
{edges:['grass-mouse','mouse-snake','snake-snakeeater','snakeeater-lion'],levels:{grass:'1',mouse:'2',snake:'3',snakeeater:'4',lion:'5'},showLionLevels:false},
{edges:[],levels:{},showLionLevels:true}
];
function getEl(id){return document.getElementById('scene1-'+id);}
function showElement(el){if(!el)return;el.style.display='block';el.style.opacity='1';}
function hideElement(el){if(!el)return;el.style.display='none';el.style.opacity='0';}
function renderStep(){
var config=stepConfig[currentStep];
allEdges.forEach(function(edgeId){
var edgeEl=getEl('edge-'+edgeId);if(!edgeEl)return;
if(currentStep===0||currentStep===6){edgeEl.setAttribute('stroke','#000');edgeEl.setAttribute('stroke-width','1.5');edgeEl.setAttribute('marker-end','url(#scene1-arrow-black)');}
else if(config.edges.indexOf(edgeId)!==-1){edgeEl.setAttribute('stroke','#1565c0');edgeEl.setAttribute('stroke-width','3');edgeEl.setAttribute('marker-end','url(#scene1-arrow-blue)');}
else{edgeEl.setAttribute('stroke','#ccc');edgeEl.setAttribute('stroke-width','1');edgeEl.setAttribute('marker-end','url(#scene1-arrow-gray)');}
});
allLevelIds.forEach(function(levelId){
var levelEl=getEl('level-'+levelId);if(!levelEl)return;
if(config.levels[levelId]){if(levelId==='lion'){var t=getEl('level-lion-text');if(t)t.textContent=config.levels[levelId];}showElement(levelEl);}
else{hideElement(levelEl);}
});
var lionLevelsEl=getEl('lion-levels');
if(config.showLionLevels){showElement(lionLevelsEl);}else{hideElement(lionLevelsEl);}
getEl('btn-prev').disabled=(currentStep===0);
getEl('btn-next').disabled=(currentStep===totalSteps);
getEl('step-indicator').textContent='步骤 '+currentStep+'/'+totalSteps;
}
getEl('btn-prev').addEventListener('click',function(){if(currentStep>0){currentStep--;renderStep();}});
getEl('btn-next').addEventListener('click',function(){if(currentStep<totalSteps){currentStep++;renderStep();}});
renderStep();
})();
</script>
