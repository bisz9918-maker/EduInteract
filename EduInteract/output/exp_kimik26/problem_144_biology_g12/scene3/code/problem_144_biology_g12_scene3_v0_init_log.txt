    <script>
    (function() {
        var currentState = 1;
        var exp1Group = document.getElementById('scene3-exp1-group');
        var exp2Group = document.getElementById('scene3-exp2-group');
        var exp1Boxes = document.getElementById('scene3-exp1-boxes');
        var exp2Boxes = document.getElementById('scene3-exp2-boxes');
        var btnExp1 = document.getElementById('scene3-btn-exp1');
        var btnExp2 = document.getElementById('scene3-btn-exp2');
        
        function switchState(state) {
            currentState = state;
            if (state === 1) {
                exp1Group.style.display = 'block';
                exp2Group.style.display = 'none';
                exp1Boxes.style.display = 'block';
                exp2Boxes.style.display = 'none';
                btnExp1.classList.add('active');
                btnExp2.classList.remove('active');
            } else {
                exp1Group.style.display = 'none';
                exp2Group.style.display = 'block';
                exp1Boxes.style.display = 'none';
                exp2Boxes.style.display = 'block';
                btnExp1.classList.remove('active');
                btnExp2.classList.add('active');
            }
        }
        
        btnExp1.addEventListener('click', function() { switchState(1); });
        btnExp2.addEventListener('click', function() { switchState(2); });
    })();
    </script>
