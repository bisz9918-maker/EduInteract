    <script>
    (function() {
        // Scene 3 实现
        const totalSteps = 6;
        let currentStep = 0;

        const stepLabels = [
            "P代亲本",
            "F₁代验证",
            "F₁交配分解",
            "F₂棋盘格",
            "表现型统计",
            "概率计算"
        ];

        const btnPrev = document.getElementById("scene3-btn-prev");
        const btnNext = document.getElementById("scene3-btn-next");
        const stepCounter = document.getElementById("scene3-step-counter");
        const stepLabel = document.getElementById("scene3-step-label");

        function renderStep() {
            // 更新步骤计数器和标签
            stepCounter.textContent = "步骤 " + (currentStep + 1) + " / " + totalSteps;
            stepLabel.textContent = stepLabels[currentStep];

            // 更新按钮状态
            btnPrev.disabled = (currentStep === 0);
            btnNext.disabled = (currentStep === totalSteps - 1);

            // 更新各步骤组的显隐：只有当前步骤显示，其余隐藏
            for (let i = 0; i < totalSteps; i++) {
                const group = document.getElementById("scene3-step" + i);
                if (i === currentStep) {
                    group.classList.add("active");
                } else {
                    group.classList.remove("active");
                }
            }
        }

        btnPrev.addEventListener("click", function() {
            if (currentStep > 0) {
                currentStep--;
                renderStep();
            }
        });

        btnNext.addEventListener("click", function() {
            if (currentStep < totalSteps - 1) {
                currentStep++;
                renderStep();
            }
        });

        // 初始渲染
        renderStep();
    })();
    </script>
