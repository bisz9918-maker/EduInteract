    <script>
    (function() {
        let currentStep = 0;
        const totalSteps = 5;

        const btnPrev = document.getElementById('scene2-btn-prev');
        const btnNext = document.getElementById('scene2-btn-next');
        const stepIndicator = document.getElementById('scene2-step-indicator');

        const cardA = document.getElementById('scene2-card-a');
        const cardB = document.getElementById('scene2-card-b');
        const cardC = document.getElementById('scene2-card-c');
        const cardD = document.getElementById('scene2-card-d');

        const valGroup = document.getElementById('scene2-val-group');
        const arrowGroup = document.getElementById('scene2-arrow-group');
        const contentAGroup = document.getElementById('scene2-content-a-group');
        const contentBGroup = document.getElementById('scene2-content-b-group');
        const contentCGroup = document.getElementById('scene2-content-c-group');
        const contentDGroup = document.getElementById('scene2-content-d-group');
        const markGroup = document.getElementById('scene2-mark-group');
        const summary = document.getElementById('scene2-summary');

        function setOpacity(el, opacity) {
            if (!el) return;
            el.setAttribute('opacity', opacity);
            el.style.display = opacity === 0 ? 'none' : '';
        }

        function setCardStyle(card, type) {
            card.classList.remove('active-blue', 'active-green');
            if (type === 'active-blue') {
                card.classList.add('active-blue');
            } else if (type === 'active-green') {
                card.classList.add('active-green');
            }
        }

        function render() {
            stepIndicator.textContent = '步骤 ' + currentStep + '/' + totalSteps;
            btnPrev.disabled = (currentStep === 0);
            btnNext.disabled = (currentStep === totalSteps);

            setCardStyle(cardA, 'inactive');
            setCardStyle(cardB, 'inactive');
            setCardStyle(cardC, 'inactive');
            setCardStyle(cardD, 'inactive');

            setOpacity(valGroup, 0);
            setOpacity(arrowGroup, 0);
            setOpacity(contentAGroup, 0);
            setOpacity(contentBGroup, 0);
            setOpacity(contentCGroup, 0);
            setOpacity(contentDGroup, 0);
            setOpacity(markGroup, 0);
            setOpacity(summary, 0);

            switch (currentStep) {
                case 0:
                    break;
                case 1:
                    setCardStyle(cardA, 'active-blue');
                    setOpacity(valGroup, 1);
                    setOpacity(arrowGroup, 1);
                    setOpacity(contentAGroup, 1);
                    setOpacity(markGroup, 1);
                    break;
                case 2:
                    setCardStyle(cardB, 'active-blue');
                    setOpacity(contentBGroup, 1);
                    setOpacity(markGroup, 1);
                    break;
                case 3:
                    setCardStyle(cardC, 'active-blue');
                    setOpacity(contentCGroup, 1);
                    setOpacity(markGroup, 1);
                    break;
                case 4:
                    setCardStyle(cardD, 'active-blue');
                    setOpacity(contentDGroup, 1);
                    setOpacity(markGroup, 1);
                    break;
                case 5:
                    setCardStyle(cardC, 'active-green');
                    setCardStyle(cardD, 'active-green');
                    setOpacity(markGroup, 1);
                    setOpacity(summary, 1);
                    break;
            }
        }

        btnPrev.addEventListener('click', function() {
            if (currentStep > 0) {
                currentStep--;
                render();
            }
        });

        btnNext.addEventListener('click', function() {
            if (currentStep < totalSteps) {
                currentStep++;
                render();
            }
        });

        render();
    })();
    </script>
