    <script>
    (function() {
        // Scene 2 实现
        var currentStep = 0; // 0~4
        var totalSteps = 5;
        var stepNames = [
            "认识装置",
            "分析产物与检验目标",
            "错误顺序警示",
            "正确顺序演示",
            "总结与安全"
        ];

        // 获取元素引用
        var els = {
            indicator: document.getElementById("scene2-step-indicator"),
            btnPrev: document.getElementById("scene2-btn-prev"),
            btnNext: document.getElementById("scene2-btn-next"),

            // 上部装置组
            aGroup: document.getElementById("scene2-a-group"),
            bGroup: document.getElementById("scene2-b-group"),
            cGroup: document.getElementById("scene2-c-group"),
            dGroup: document.getElementById("scene2-d-group"),

            // 信息区
            step1Info: document.getElementById("scene2-step1-info"),
            step2Warning: document.getElementById("scene2-step2-warning"),
            step3Labels: document.getElementById("scene2-step3-labels"),
            step4Summary: document.getElementById("scene2-step4-summary"),

            // 流程图
            flowDev1: document.getElementById("scene2-flow-dev1"),
            flowDev2: document.getElementById("scene2-flow-dev2"),
            flowDev3: document.getElementById("scene2-flow-dev3"),
            flowLine1: document.getElementById("scene2-flow-line1"),
            flowLine2: document.getElementById("scene2-flow-line2"),
            flowLine3: document.getElementById("scene2-flow-line3"),
            arrows: document.getElementById("scene2-arrows"),

            // 流程图子元素
            f1Body: document.getElementById("scene2-f1-body"),
            f1Liquid: document.getElementById("scene2-f1-liquid"),
            f1Name: document.getElementById("scene2-f1-name"),
            f2Body: document.getElementById("scene2-f2-body"),
            f2Liquid: document.getElementById("scene2-f2-liquid"),
            f2Name: document.getElementById("scene2-f2-name"),
            f3Flame: document.getElementById("scene2-f3-flame"),
            f3Body: document.getElementById("scene2-f3-body"),

            // 上部装置子元素（用于高亮）
            aBody: document.getElementById("scene2-a-body"),
            bBody: document.getElementById("scene2-b-body"),
            bLong: document.getElementById("scene2-b-long"),
            bShort: document.getElementById("scene2-b-short"),
            cBody: document.getElementById("scene2-c-body"),
            cLong: document.getElementById("scene2-c-long"),
            cShort: document.getElementById("scene2-c-short"),
            dBody: document.getElementById("scene2-d-body"),
            dLong: document.getElementById("scene2-d-long"),
            dShort: document.getElementById("scene2-d-short"),

            // 标签
            labelLong1: document.getElementById("scene2-label-long1"),
            labelLong2: document.getElementById("scene2-label-long2"),
        };

        function show(el) {
            if (!el) return;
            el.style.display = "";
            el.style.opacity = "1";
        }
        function hide(el) {
            if (!el) return;
            el.style.display = "none";
            el.style.opacity = "0";
        }

        function clearHighlight() {
            var items = [els.aBody, els.bBody, els.bLong, els.bShort, els.cBody, els.cLong, els.cShort, els.dBody, els.dLong, els.dShort, els.f1Body, els.f2Body, els.f3Body, els.flowLine1, els.flowLine2, els.flowLine3];
            items.forEach(function(item) {
                if (item) {
                    item.classList.remove("scene2-highlight-red", "scene2-highlight-blue", "scene2-highlight-green");
                }
            });
        }

        function setRedHighlight() {
            clearHighlight();
            var items = [els.dBody, els.dLong, els.dShort, els.cBody, els.cLong, els.cShort, els.aBody, els.f1Body, els.f2Body, els.f3Body, els.flowLine1, els.flowLine2, els.flowLine3];
            items.forEach(function(item) {
                if (item) item.classList.add("scene2-highlight-red");
            });
        }

        function setBlueHighlight() {
            clearHighlight();
            var items = [els.cBody, els.cLong, els.cShort, els.bBody, els.bLong, els.bShort, els.aBody, els.f1Body, els.f2Body, els.f3Body, els.flowLine1, els.flowLine2, els.flowLine3];
            items.forEach(function(item) {
                if (item) item.classList.add("scene2-highlight-blue");
            });
        }

        function setGreenHighlight() {
            clearHighlight();
            var items = [els.cBody, els.cLong, els.cShort, els.bBody, els.bLong, els.bShort, els.aBody, els.f1Body, els.f2Body, els.f3Body, els.flowLine1, els.flowLine2, els.flowLine3];
            items.forEach(function(item) {
                if (item) item.classList.add("scene2-highlight-green");
            });
        }

        function render() {
            // 更新指示器
            els.indicator.textContent = "步骤 " + (currentStep + 1) + " / " + totalSteps + "：" + stepNames[currentStep];
            els.btnPrev.disabled = (currentStep === 0);
            els.btnNext.disabled = (currentStep === totalSteps - 1);

            // 默认隐藏所有动态元素
            hide(els.step1Info);
            hide(els.step2Warning);
            hide(els.step3Labels);
            hide(els.step4Summary);
            hide(els.flowDev1);
            hide(els.flowDev2);
            hide(els.flowDev3);
            hide(els.flowLine1);
            hide(els.flowLine2);
            hide(els.flowLine3);
            hide(els.arrows);
            clearHighlight();

            // 恢复流程图默认液体颜色
            if (els.f1Liquid) els.f1Liquid.setAttribute("fill", "#f5f5f5");
            if (els.f2Liquid) els.f2Liquid.setAttribute("fill", "#f5f5f5");

            switch (currentStep) {
                case 0:
                    // 步骤0：认识装置 - 只显示上部四个装置
                    break;
                case 1:
                    // 步骤1：分析产物与检验目标
                    show(els.step1Info);
                    break;
                case 2:
                    // 步骤2：错误顺序警示（图甲→d→c→a）
                    show(els.flowDev1);
                    show(els.flowDev2);
                    show(els.flowDev3);
                    show(els.flowLine1);
                    show(els.flowLine2);
                    show(els.flowLine3);
                    show(els.step2Warning);

                    // 设置装置为 d, c, a
                    els.f1Name.textContent = "d";
                    els.f2Name.textContent = "c";
                    if (els.f1Liquid) els.f1Liquid.setAttribute("fill", "#e3f2fd");

                    setRedHighlight();
                    break;
                case 3:
                    // 步骤3：正确顺序演示（图甲→c→b→a）
                    show(els.flowDev1);
                    show(els.flowDev2);
                    show(els.flowDev3);
                    show(els.flowLine1);
                    show(els.flowLine2);
                    show(els.flowLine3);
                    show(els.step3Labels);
                    show(els.arrows);

                    // 设置装置为 c, b, a
                    els.f1Name.textContent = "c";
                    els.f2Name.textContent = "b";

                    setBlueHighlight();
                    break;
                case 4:
                    // 步骤4：总结与安全
                    show(els.flowDev1);
                    show(els.flowDev2);
                    show(els.flowDev3);
                    show(els.flowLine1);
                    show(els.flowLine2);
                    show(els.flowLine3);
                    show(els.step4Summary);

                    // 设置装置为 c, b, a
                    els.f1Name.textContent = "c";
                    els.f2Name.textContent = "b";

                    setGreenHighlight();
                    break;
            }
        }

        // 事件绑定
        els.btnPrev.addEventListener("click", function() {
            if (currentStep > 0) {
                currentStep--;
                render();
            }
        });
        els.btnNext.addEventListener("click", function() {
            if (currentStep < totalSteps - 1) {
                currentStep++;
                render();
            }
        });

        // 初始渲染
        render();
    })();
    </script>
