    <script>
    (function() {
        // Scene 1 实现
        const svgNS = "http://www.w3.org/2000/svg";
        let currentStep = 0;
        const totalSteps = 5;

        // 固定seed伪随机
        function seededRandom(seed) {
            let x = Math.sin(seed * 127.1 + 311.7) * 43758.5453;
            return x - Math.floor(x);
        }

        // 生成粒子到指定组
        function createParticles(groupId, count, xMin, xMax, yMin, yMax, color, radius, shape) {
            const g = document.createElementNS(svgNS, "g");
            g.setAttribute("id", groupId);
            g.style.display = "none";
            for (let i = 0; i < count; i++) {
                const x = xMin + seededRandom(groupId + i * 3 + 1) * (xMax - xMin);
                const y = yMin + seededRandom(groupId + i * 3 + 2) * (yMax - yMin);
                let el;
                if (shape === "rect") {
                    el = document.createElementNS(svgNS, "rect");
                    const size = radius * 2;
                    el.setAttribute("x", x - size/2);
                    el.setAttribute("y", y - size/2);
                    el.setAttribute("width", size);
                    el.setAttribute("height", size);
                    el.setAttribute("rx", 2);
                } else {
                    el = document.createElementNS(svgNS, "circle");
                    el.setAttribute("cx", x);
                    el.setAttribute("cy", y);
                    el.setAttribute("r", radius);
                }
                el.setAttribute("fill", color);
                if (shape !== "rect") {
                    el.setAttribute("opacity", "0.85");
                } else {
                    el.setAttribute("opacity", "0.9");
                }
                g.appendChild(el);
            }
            return g;
        }

        const particleContainer = document.getElementById("scene1-particles");

        // 容器1 Co颗粒：15个，棕色，直径12（半径6）
        const co1Group = createParticles("scene1-co1", 15, 210, 290, 360, 390, "#5d4037", 6, "circle");
        particleContainer.appendChild(co1Group);

        // 容器1 杂质：5个，橙色，10x10（半径5用于rect）
        const impurityGroup = createParticles("scene1-impurity", 5, 220, 280, 370, 390, "#ff8f00", 5, "rect");
        particleContainer.appendChild(impurityGroup);

        // 容器2 Co颗粒：10个，棕色
        const co2Group = createParticles("scene1-co2", 10, 510, 570, 360, 390, "#5d4037", 6, "circle");
        particleContainer.appendChild(co2Group);

        // HCl粒子(入口)：20个，蓝色，直径6
        const hclInletGroup = createParticles("scene1-hcl-inlet", 20, 100, 190, 250, 270, "#1565c0", 3, "circle");
        particleContainer.appendChild(hclInletGroup);

        // HCl粒子(容器1)：15个，蓝色
        const hclC1Group = createParticles("scene1-hcl-c1", 15, 200, 300, 300, 350, "#1565c0", 3, "circle");
        particleContainer.appendChild(hclC1Group);

        // CoCl₂粒子(容器1)：15个，红色
        const cocl2C1Group = createParticles("scene1-cocl2-c1", 15, 200, 300, 300, 350, "#c62828", 3, "circle");
        particleContainer.appendChild(cocl2C1Group);

        // H₂粒子(容器1)：15个，绿色
        const h2C1Group = createParticles("scene1-h2-c1", 15, 200, 300, 300, 350, "#2e7d32", 3, "circle");
        particleContainer.appendChild(h2C1Group);

        // 迁移粒子：25个，红绿混合
        const migrateGroup = document.createElementNS(svgNS, "g");
        migrateGroup.setAttribute("id", "scene1-migrate");
        migrateGroup.style.display = "none";
        for (let i = 0; i < 25; i++) {
            const x = 310 + seededRandom(500 + i * 3 + 1) * 180;
            const y = 250 + seededRandom(500 + i * 3 + 2) * 20;
            const color = i % 2 === 0 ? "#c62828" : "#2e7d32";
            const c = document.createElementNS(svgNS, "circle");
            c.setAttribute("cx", x);
            c.setAttribute("cy", y);
            c.setAttribute("r", 3);
            c.setAttribute("fill", color);
            c.setAttribute("opacity", "0.85");
            migrateGroup.appendChild(c);
        }
        particleContainer.appendChild(migrateGroup);

        // CoCl₂粒子(容器2)：15个，红色
        const cocl2C2Group = createParticles("scene1-cocl2-c2", 15, 500, 580, 300, 350, "#c62828", 3, "circle");
        particleContainer.appendChild(cocl2C2Group);

        // H₂粒子(容器2)：15个，绿色
        const h2C2Group = createParticles("scene1-h2-c2", 15, 500, 580, 300, 350, "#2e7d32", 3, "circle");
        particleContainer.appendChild(h2C2Group);

        // HCl粒子(出口)：20个，蓝色
        const hclOutletGroup = createParticles("scene1-hcl-outlet", 20, 590, 700, 250, 270, "#1565c0", 3, "circle");
        particleContainer.appendChild(hclOutletGroup);

        // 获取元素引用
        const els = {
            labelGas: document.getElementById("scene1-label-gas"),
            labelRecycle: document.getElementById("scene1-label-recycle"),
            label900: document.getElementById("scene1-label-900"),
            label600: document.getElementById("scene1-label-600"),
            labelImpure: document.getElementById("scene1-label-impure"),
            labelImpurity: document.getElementById("scene1-label-impurity"),
            labelPure: document.getElementById("scene1-label-pure"),
            highlightC1: document.getElementById("scene1-highlight-c1"),
            highlightC2: document.getElementById("scene1-highlight-c2"),
            flowInlet: document.getElementById("scene1-flow-inlet"),
            flowMigrate: document.getElementById("scene1-flow-migrate"),
            equation: document.getElementById("scene1-equation"),
            co1: document.getElementById("scene1-co1"),
            co2: document.getElementById("scene1-co2"),
            impurity: document.getElementById("scene1-impurity"),
            hclInlet: document.getElementById("scene1-hcl-inlet"),
            hclC1: document.getElementById("scene1-hcl-c1"),
            cocl2C1: document.getElementById("scene1-cocl2-c1"),
            h2C1: document.getElementById("scene1-h2-c1"),
            migrate: document.getElementById("scene1-migrate"),
            cocl2C2: document.getElementById("scene1-cocl2-c2"),
            h2C2: document.getElementById("scene1-h2-c2"),
            hclOutlet: document.getElementById("scene1-hcl-outlet"),
            btnPrev: document.getElementById("scene1-btn-prev"),
            btnNext: document.getElementById("scene1-btn-next"),
            indicator: document.getElementById("scene1-step-indicator")
        };

        // 控制Co1颗粒显示数量
        function setCo1Visible(count) {
            const children = els.co1.children;
            for (let i = 0; i < children.length; i++) {
                children[i].style.display = i < count ? "inline" : "none";
            }
        }

        function show(el) {
            if (!el) return;
            el.style.display = "inline";
            if (el.tagName.toLowerCase() === "g") {
                el.style.display = "block";
            }
        }

        function hide(el) {
            if (!el) return;
            el.style.display = "none";
        }

        function renderStep() {
            const s = currentStep;

            // 方程式/顶部文字
            if (s === 2) {
                els.equation.innerHTML = '<span style="color:#1565c0">Co(固) + 2HCl(气) → CoCl₂(气) + H₂(气)</span>&nbsp;&nbsp;&nbsp;900℃';
                els.equation.style.display = "block";
            } else if (s === 4) {
                els.equation.innerHTML = '<span style="color:#c62828">CoCl₂(气) + H₂(气) → Co(固) + 2HCl(气)</span>&nbsp;&nbsp;&nbsp;600℃';
                els.equation.style.display = "block";
            } else if (s === 5) {
                els.equation.innerHTML = '提纯完成——纯净Co在容器2，HCl可循环';
                els.equation.style.display = "block";
            } else {
                els.equation.style.display = "none";
            }

            // 标签
            s >= 1 && s <= 2 ? show(els.labelGas) : hide(els.labelGas);
            s === 5 ? show(els.labelRecycle) : hide(els.labelRecycle);
            s >= 2 && s <= 3 ? show(els.label900) : hide(els.label900);
            s >= 4 && s <= 5 ? show(els.label600) : hide(els.label600);
            s >= 0 && s <= 1 ? show(els.labelImpure) : hide(els.labelImpure);
            s >= 2 && s <= 5 ? show(els.labelImpurity) : hide(els.labelImpurity);
            s >= 4 && s <= 5 ? show(els.labelPure) : hide(els.labelPure);

            // 高亮
            s === 2 ? show(els.highlightC1) : hide(els.highlightC1);
            s === 4 ? show(els.highlightC2) : hide(els.highlightC2);

            // 流动箭头
            s === 1 ? show(els.flowInlet) : hide(els.flowInlet);
            s === 3 ? show(els.flowMigrate) : hide(els.flowMigrate);

            // 固体颗粒
            if (s <= 1) {
                show(els.co1);
                setCo1Visible(15);
            } else if (s === 2 || s === 3 || s === 4) {
                show(els.co1);
                setCo1Visible(5);
            } else {
                hide(els.co1);
            }

            s >= 0 ? show(els.impurity) : hide(els.impurity);
            s >= 4 && s <= 5 ? show(els.co2) : hide(els.co2);

            // 气体粒子
            s >= 1 && s <= 2 ? show(els.hclInlet) : hide(els.hclInlet);
            s === 2 ? show(els.hclC1) : hide(els.hclC1);
            s === 2 ? show(els.cocl2C1) : hide(els.cocl2C1);
            s === 2 ? show(els.h2C1) : hide(els.h2C1);
            s === 3 ? show(els.migrate) : hide(els.migrate);
            s === 4 ? show(els.cocl2C2) : hide(els.cocl2C2);
            s === 4 ? show(els.h2C2) : hide(els.h2C2);
            s >= 4 && s <= 5 ? show(els.hclOutlet) : hide(els.hclOutlet);

            // 按钮状态
            els.btnPrev.disabled = s === 0;
            els.btnNext.disabled = s === totalSteps;
            els.btnNext.textContent = s === totalSteps ? "完成" : "下一步 ▶";
            els.indicator.textContent = "步骤 " + s + " / " + totalSteps;
        }

        els.btnPrev.addEventListener("click", function() {
            if (currentStep > 0) {
                currentStep--;
                renderStep();
            }
        });

        els.btnNext.addEventListener("click", function() {
            if (currentStep < totalSteps) {
                currentStep++;
                renderStep();
            }
        });

        // 初始渲染
        renderStep();
    })();
    </script>
</body>
