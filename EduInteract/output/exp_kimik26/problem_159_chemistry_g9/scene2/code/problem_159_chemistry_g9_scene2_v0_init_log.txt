    <script>
    (function() {
        // Scene 2 实现：固体加热分解质量变化对比
        // 状态机：1=KMnO₄, 2=CaCO₃, 3=对比模式
        let currentState = 1;

        const btnKmno4 = document.getElementById('scene2-btn-kmno4');
        const btnCaco3 = document.getElementById('scene2-btn-caco3');
        const btnCompare = document.getElementById('scene2-btn-compare');

        // 元素集合
        const els = {
            curveKmno4: document.getElementById('scene2-curve-kmno4'),
            curveCaco3: document.getElementById('scene2-curve-caco3'),
            curveBWrong: document.getElementById('scene2-curve-b-wrong'),
            fillKmno4: document.getElementById('scene2-fill-kmno4'),
            fillCaco3: document.getElementById('scene2-fill-caco3'),
            lineInit: document.getElementById('scene2-line-init'),
            lineKFinal: document.getElementById('scene2-line-k-final'),
            lineCFinal: document.getElementById('scene2-line-c-final'),
            vlineStart: document.getElementById('scene2-vline-start'),
            vlineEnd: document.getElementById('scene2-vline-end'),
            dotStart: document.getElementById('scene2-dot-start'),
            dotKPlat: document.getElementById('scene2-dot-k-plat'),
            dotCStart: document.getElementById('scene2-dot-c-start'),
            dotCEnd: document.getElementById('scene2-dot-c-end'),
            labelMInit: document.getElementById('scene2-label-m-init'),
            labelMKFinal: document.getElementById('scene2-label-m-k-final'),
            labelMCFinal: document.getElementById('scene2-label-m-c-final'),
            labelCStart: document.getElementById('scene2-label-c-start'),
            labelCEnd: document.getElementById('scene2-label-c-end'),
            labelKDesc: document.getElementById('scene2-label-k-desc'),
            labelPlatform: document.getElementById('scene2-label-platform'),
            infoKmno4: document.getElementById('scene2-info-kmno4'),
            infoCaco3: document.getElementById('scene2-info-caco3'),
            infoCompare: document.getElementById('scene2-info-compare')
        };

        function show(el) {
            if (!el) return;
            el.classList.remove('scene2-hidden');
            el.classList.add('scene2-visible');
        }
        function hide(el) {
            if (!el) return;
            el.classList.remove('scene2-visible');
            el.classList.add('scene2-hidden');
        }

        function setButtonState(state) {
            // 重置所有按钮
            btnKmno4.className = 'scene2-btn scene2-btn-default';
            btnCaco3.className = 'scene2-btn scene2-btn-default';
            btnCompare.className = 'scene2-btn scene2-btn-default';

            if (state === 1) {
                btnKmno4.className = 'scene2-btn scene2-btn-active-blue';
            } else if (state === 2) {
                btnCaco3.className = 'scene2-btn scene2-btn-active-green';
            } else if (state === 3) {
                btnCompare.className = 'scene2-btn scene2-btn-active-compare';
            }
        }

        function setInfoPanel(state) {
            els.infoKmno4.classList.remove('active');
            els.infoCaco3.classList.remove('active');
            els.infoCompare.classList.remove('active');
            if (state === 1) els.infoKmno4.classList.add('active');
            else if (state === 2) els.infoCaco3.classList.add('active');
            else if (state === 3) els.infoCompare.classList.add('active');
        }

        function render(state) {
            currentState = state;
            setButtonState(state);
            setInfoPanel(state);

            // 先全部隐藏
            hide(els.curveKmno4);
            hide(els.curveCaco3);
            hide(els.curveBWrong);
            hide(els.fillKmno4);
            hide(els.fillCaco3);
            hide(els.lineKFinal);
            hide(els.lineCFinal);
            hide(els.vlineStart);
            hide(els.vlineEnd);
            hide(els.dotKPlat);
            hide(els.dotCStart);
            hide(els.dotCEnd);
            hide(els.labelMKFinal);
            hide(els.labelMCFinal);
            hide(els.labelCStart);
            hide(els.labelCEnd);
            hide(els.labelKDesc);
            hide(els.labelPlatform);

            // 始终显示
            show(els.lineInit);
            show(els.dotStart);
            show(els.labelMInit);

            if (state === 1) {
                // KMnO₄ 分解
                show(els.curveKmno4);
                show(els.fillKmno4);
                show(els.lineKFinal);
                show(els.dotKPlat);
                show(els.labelMKFinal);
            } else if (state === 2) {
                // CaCO₃ 煅烧
                show(els.curveCaco3);
                show(els.fillCaco3);
                show(els.lineCFinal);
                show(els.dotCStart);
                show(els.dotCEnd);
                show(els.labelMCFinal);
            } else if (state === 3) {
                // 对比模式
                show(els.curveKmno4);
                show(els.curveCaco3);
                show(els.curveBWrong);
                show(els.fillKmno4);
                show(els.fillCaco3);
                show(els.lineKFinal);
                show(els.lineCFinal);
                show(els.vlineStart);
                show(els.vlineEnd);
                show(els.dotKPlat);
                show(els.dotCStart);
                show(els.dotCEnd);
                show(els.labelMKFinal);
                show(els.labelMCFinal);
                show(els.labelCStart);
                show(els.labelCEnd);
                show(els.labelKDesc);
                show(els.labelPlatform);
            }
        }

        // 事件绑定
        btnKmno4.addEventListener('click', function() { render(1); });
        btnCaco3.addEventListener('click', function() { render(2); });
        btnCompare.addEventListener('click', function() { render(3); });

        // 初始渲染
        render(1);
    })();
    </script>
