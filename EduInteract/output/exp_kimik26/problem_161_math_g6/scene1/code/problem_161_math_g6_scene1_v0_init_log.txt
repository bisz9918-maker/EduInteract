    <script>
    (function() {
        // Scene 1 实现
        let currentStep = 0;
        const totalSteps = 5;

        // 元素引用
        const els = {
            zones: [
                document.getElementById('scene1-zone-1'),
                document.getElementById('scene1-zone-2'),
                document.getElementById('scene1-zone-3'),
                document.getElementById('scene1-zone-4')
            ],
            combos2: [
                document.getElementById('scene1-combo-12'),
                document.getElementById('scene1-combo-23'),
                document.getElementById('scene1-combo-34')
            ],
            combos3: [
                document.getElementById('scene1-combo-123'),
                document.getElementById('scene1-combo-234')
            ],
            combo4: document.getElementById('scene1-combo-1234'),
            hint: document.getElementById('scene1-hint'),
            stats: [
                document.getElementById('scene1-stat-1'),
                document.getElementById('scene1-stat-2'),
                document.getElementById('scene1-stat-3'),
                document.getElementById('scene1-stat-4')
            ],
            separator: document.getElementById('scene1-separator'),
            formula: document.getElementById('scene1-formula'),
            answer: document.getElementById('scene1-answer'),
            stepIndicator: document.getElementById('scene1-step-indicator'),
            btnPrev: document.getElementById('scene1-btn-prev'),
            btnNext: document.getElementById('scene1-btn-next')
        };

        function setOpacity(el, val) {
            if (el) el.setAttribute('opacity', val);
        }

        function setFill(el, color) {
            if (el) el.setAttribute('fill', color);
        }

        function setFontWeight(el, weight) {
            if (el) el.setAttribute('font-weight', weight);
        }

        function hideAllHighlights() {
            els.zones.forEach(r => setOpacity(r, 0));
            els.combos2.forEach(r => setOpacity(r, 0));
            els.combos3.forEach(r => setOpacity(r, 0));
            setOpacity(els.combo4, 0);
        }

        function updateScene(step) {
            currentStep = step;
            els.stepIndicator.textContent = '步骤 ' + step + ' / ' + totalSteps;
            els.btnPrev.disabled = (step === 0);
            els.btnNext.disabled = (step === totalSteps);

            // 重置所有统计行样式
            els.stats.forEach(stat => {
                setOpacity(stat, 0);
                setFill(stat, '#000000');
                setFontWeight(stat, 'normal');
            });
            setOpacity(els.separator, 0);
            setOpacity(els.formula, 0);
            setOpacity(els.answer, 0);

            // 提示文字
            setOpacity(els.hint, step === 0 ? 1 : 0);

            hideAllHighlights();

            if (step === 0) {
                // 初始状态，仅显示图形和提示
            } else if (step === 1) {
                // 单块：4个
                els.zones.forEach(r => setOpacity(r, 1));
                setOpacity(els.stats[0], 1);
                setFill(els.stats[0], '#1565c0');
                setFontWeight(els.stats[0], 'bold');
            } else if (step === 2) {
                // 两块：3个
                els.combos2.forEach(r => setOpacity(r, 1));
                setOpacity(els.stats[0], 1);
                setFill(els.stats[0], '#888888');
                setOpacity(els.stats[1], 1);
                setFill(els.stats[1], '#1565c0');
                setFontWeight(els.stats[1], 'bold');
            } else if (step === 3) {
                // 三块：2个
                els.combos3.forEach(r => setOpacity(r, 1));
                setOpacity(els.stats[0], 1);
                setFill(els.stats[0], '#888888');
                setOpacity(els.stats[1], 1);
                setFill(els.stats[1], '#888888');
                setOpacity(els.stats[2], 1);
                setFill(els.stats[2], '#1565c0');
                setFontWeight(els.stats[2], 'bold');
            } else if (step === 4) {
                // 四块：1个
                setOpacity(els.combo4, 1);
                setOpacity(els.stats[0], 1);
                setFill(els.stats[0], '#888888');
                setOpacity(els.stats[1], 1);
                setFill(els.stats[1], '#888888');
                setOpacity(els.stats[2], 1);
                setFill(els.stats[2], '#888888');
                setOpacity(els.stats[3], 1);
                setFill(els.stats[3], '#1565c0');
                setFontWeight(els.stats[3], 'bold');
            } else if (step === 5) {
                // 汇总
                setOpacity(els.stats[0], 1);
                setFill(els.stats[0], '#888888');
                setOpacity(els.stats[1], 1);
                setFill(els.stats[1], '#888888');
                setOpacity(els.stats[2], 1);
                setFill(els.stats[2], '#888888');
                setOpacity(els.stats[3], 1);
                setFill(els.stats[3], '#888888');
                setOpacity(els.separator, 1);
                setOpacity(els.formula, 1);
                setOpacity(els.answer, 1);
            }
        }

        function changeStep(delta) {
            const newStep = currentStep + delta;
            if (newStep >= 0 && newStep <= totalSteps) {
                updateScene(newStep);
            }
        }

        els.btnPrev.addEventListener('click', function() { changeStep(-1); });
        els.btnNext.addEventListener('click', function() { changeStep(1); });

        // 初始化
        updateScene(0);
    })();
    </script>
