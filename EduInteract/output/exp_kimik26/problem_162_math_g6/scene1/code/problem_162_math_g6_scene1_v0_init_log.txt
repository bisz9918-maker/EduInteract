    <script>
    (function() {
        // Scene 1 实现
        var currentStep = 0;
        var totalSteps = 4;

        var btnPrev = document.getElementById('scene1-btn-prev');
        var btnNext = document.getElementById('scene1-btn-next');
        var indicator = document.getElementById('scene1-step-indicator');
        var svg = document.getElementById('scene1-svg');

        // 获取所有带 data-steps 属性的元素
        function getStepElements() {
            return svg.querySelectorAll('[data-steps]');
        }

        // 检查某步骤是否在 data-steps 中
        function shouldShow(el, step) {
            var attr = el.getAttribute('data-steps');
            if (!attr) return false;
            var steps = attr.split(',').map(function(s) { return parseInt(s.trim(), 10); });
            return steps.indexOf(step) !== -1;
        }

        // 渲染指定步骤
        function renderStep(step) {
            var elements = getStepElements();
            elements.forEach(function(el) {
                if (shouldShow(el, step)) {
                    el.style.display = '';
                    el.style.opacity = '1';
                } else {
                    el.style.display = 'none';
                    el.style.opacity = '0';
                }
            });

            // 更新按钮状态
            btnPrev.disabled = (step <= 0);
            btnNext.disabled = (step >= totalSteps);

            // 更新指示器
            indicator.textContent = '步骤 ' + step + '/' + totalSteps;
        }

        // 上一步
        btnPrev.addEventListener('click', function() {
            if (currentStep > 0) {
                currentStep--;
                renderStep(currentStep);
            }
        });

        // 下一步
        btnNext.addEventListener('click', function() {
            if (currentStep < totalSteps) {
                currentStep++;
                renderStep(currentStep);
            }
        });

        // 初始化
        renderStep(currentStep);
    })();
    </script>
