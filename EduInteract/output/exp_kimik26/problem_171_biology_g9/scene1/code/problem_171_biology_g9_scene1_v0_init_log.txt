<script>
(function(){
var currentStep=0,totalSteps=5;
var els={
labelA:document.getElementById('scene1-label-a'),labelB:document.getElementById('scene1-label-b'),labelC:document.getElementById('scene1-label-c'),labelD:document.getElementById('scene1-label-d'),labelE:document.getElementById('scene1-label-e'),
baseList:document.getElementById('scene1-base-list'),dntList:document.getElementById('scene1-dnt-list'),sugarCompare:document.getElementById('scene1-sugar-compare'),
boxD:document.getElementById('scene1-box-d'),
optARow:document.getElementById('scene1-option-a-row'),optBRow:document.getElementById('scene1-option-b-row'),optCRow:document.getElementById('scene1-option-c-row'),optDRow:document.getElementById('scene1-option-d-row'),
optANum:document.getElementById('scene1-opt-a-num'),optCNum:document.getElementById('scene1-opt-c-num'),optDSugar:document.getElementById('scene1-opt-d-sugar'),
optAMark:document.getElementById('scene1-option-a-mark'),optBMark:document.getElementById('scene1-option-b-mark'),optCMark:document.getElementById('scene1-option-c-mark'),optDMark:document.getElementById('scene1-option-d-mark'),
btnPrev:document.getElementById('scene1-btn-prev'),btnNext:document.getElementById('scene1-btn-next'),stepIndicator:document.getElementById('scene1-step-indicator')
};
function show(el){if(!el)return;el.style.display='';requestAnimationFrame(function(){el.style.opacity='1';});}
function hide(el){if(!el)return;el.style.opacity='0';setTimeout(function(){el.style.display='none';},300);}
function setOp(el,val){if(!el)return;el.style.opacity=String(val);if(val>0)el.style.display='';}
function render(){
els.stepIndicator.textContent='步骤 '+currentStep+'/'+totalSteps;
els.btnPrev.disabled=currentStep===0;
els.btnNext.disabled=currentStep===totalSteps;
if(currentStep===0){
setOp(els.labelA,0);els.labelB.textContent='糖';els.labelB.setAttribute('fill','#000');els.labelC.setAttribute('fill','#000');els.labelD.textContent='( )';els.labelD.setAttribute('fill','#000');els.labelE.setAttribute('fill','#000');
hide(els.baseList);hide(els.dntList);hide(els.sugarCompare);
els.optARow.style.background='transparent';els.optBRow.style.background='transparent';els.optCRow.style.background='transparent';els.optDRow.style.background='transparent';
els.optANum.style.color='#333';els.optCNum.style.color='#333';els.optDSugar.style.color='#333';
setOp(els.optAMark,0);setOp(els.optBMark,0);setOp(els.optCMark,0);setOp(els.optDMark,0);
els.boxD.classList.remove('scene1-flash-anim');
}
if(currentStep>=1){
setOp(els.labelA,1);els.labelB.textContent='脱氧核糖';els.labelB.setAttribute('fill','#1565c0');els.labelC.setAttribute('fill','#1565c0');els.labelD.textContent='脱氧核苷酸';els.labelD.setAttribute('fill','#1565c0');els.labelE.setAttribute('fill','#1565c0');
els.optARow.style.background='rgba(198,40,40,0.12)';els.optBRow.style.background='rgba(46,125,50,0.12)';els.optCRow.style.background='rgba(198,40,40,0.12)';els.optDRow.style.background='rgba(198,40,40,0.12)';
setOp(els.optAMark,1);setOp(els.optBMark,1);setOp(els.optCMark,1);setOp(els.optDMark,1);
}
if(currentStep===2){show(els.baseList);els.optANum.style.color='#c62828';}else if(currentStep!==2){hide(els.baseList);if(currentStep<2)els.optANum.style.color='#333';}
if(currentStep===3){els.boxD.classList.remove('scene1-flash-anim');void els.boxD.offsetWidth;els.boxD.classList.add('scene1-flash-anim');}else{els.boxD.classList.remove('scene1-flash-anim');}
if(currentStep===4){show(els.dntList);els.optCNum.style.color='#c62828';}else if(currentStep!==4){hide(els.dntList);if(currentStep<4)els.optCNum.style.color='#333';}
if(currentStep===5){show(els.sugarCompare);els.optDSugar.style.color='#c62828';}else if(currentStep!==5){hide(els.sugarCompare);if(currentStep<5)els.optDSugar.style.color='#333';}
}
function nextStep(){if(currentStep<totalSteps){currentStep++;render();}}
function prevStep(){if(currentStep>0){currentStep--;render();}}
els.btnNext.addEventListener('click',nextStep);
els.btnPrev.addEventListener('click',prevStep);
render();
})();
</script>
