        // Scene 1 实现

        const totalSteps = 6;

        const btnPrev = document.getElementById('scene1-btn-prev');
        const btnNext = document.getElementById('scene1-btn-next');
        const indicator = document.getElementById('scene1-step-indicator');

        // 各步骤需要显示的元素ID列表（SVG图形 + HTML文字包装器）
        const stepElements = {
            0: [],
            1: [
                'scene1-line-root-passive', 'scene1-line-root-active',
                'scene1-passive-box', 'scene1-active-box',
                'scene1-passive-text-wrap', 'scene1-active-text-wrap'
            ],
            2: [
                'scene1-line-root-passive', 'scene1-line-root-active',
                'scene1-passive-box', 'scene1-active-box',
                'scene1-passive-text-wrap', 'scene1-active-text-wrap',

            ],
            3: [
                'scene1-line-root-passive', 'scene1-line-root-active',
                'scene1-passive-box', 'scene1-active-box',
                'scene1-passive-text-wrap', 'scene1-active-text-wrap',
                'scene1-line-passive-free', 'scene1-line-passive-assisted', 'scene1-line-active-ex4',
                'scene1-free-box', 'scene1-assisted-box',
                'scene1-free-text-wrap', 'scene1-assisted-text-wrap',
                'scene1-oxygen-box', 'scene1-ex3-box', 'scene1-ex4-box',
                'scene1-oxygen-text-wrap', 'scene1-ex3-text-wrap', 'scene1-ex4-text-wrap',
                'scene1-feature-free-box', 'scene1-feature-assisted-box', 'scene1-feature-active-box',
                'scene1-feature-free-text-wrap', 'scene1-feature-assisted-text-wrap', 'scene1-feature-active-text-wrap'
            ],
            4: [
                'scene1-line-root-passive', 'scene1-line-root-active',
                'scene1-passive-box', 'scene1-active-box',
                'scene1-passive-text-wrap', 'scene1-active-text-wrap',
                'scene1-line-passive-free', 'scene1-line-passive-assisted', 'scene1-line-active-ex4',
                'scene1-free-box', 'scene1-assisted-box',

                'scene1-oxygen-box', 'scene1-ex3-box', 'scene1-ex4-box',
                'scene1-oxygen-text-wrap', 'scene1-ex3-text-wrap', 'scene1-ex4-text-wrap',
                'scene1-feature-free-box', 'scene1-feature-assisted-box', 'scene1-feature-active-box',
                'scene1-feature-free-text-wrap', 'scene1-feature-assisted-text-wrap', 'scene1-feature-active-text-wrap'

                'scene1-line-root-passive', 'scene1-line-root-active',

            ]
        };

        function updateIndicator() {
            indicator.textContent = '步骤 ' + (currentStep + 1) + ' / ' + totalSteps;
            btnPrev.disabled = currentStep === 0;
            btnNext.disabled = currentStep === totalSteps - 1;
        }

        function showElement(id) {
            const el = document.getElementById(id);
            if (el) el.classList.add('scene1-visible');
        }

        function hideElement(id) {
            const el = document.getElementById(id);
            if (el) el.classList.remove('scene1-visible');
        }

        function renderStep() {
            // 先隐藏所有步骤元素
            const allEls = document.querySelectorAll('.scene1-step-el');
            allEls.forEach(function(el) {
                el.classList.remove('scene1-visible');
            });

            // 显示当前步骤应有的元素
            const ids = stepElements[currentStep] || [];
            ids.forEach(function(id) {
                showElement(id);
            });

            // 步骤4特殊处理：③④文字切换，虚线变实线
            const ex3Box = document.getElementById('scene1-ex3-box');
            const ex3Text = document.getElementById('scene1-ex3-text');
            const ex4Box = document.getElementById('scene1-ex4-box');
            const ex4Text = document.getElementById('scene1-ex4-text');

            if (currentStep >= 4) {
                // ③变为绿色实线，显示"葡萄糖进入红细胞"
                ex3Box.setAttribute('stroke', '#2e7d32');
                ex3Box.classList.remove('scene1-dashed');
                ex3Text.textContent = '葡萄糖进入红细胞';
                // ④变为蓝色实线，显示"小肠上皮细胞吸收葡萄糖"
                ex4Box.setAttribute('stroke', '#1565c0');
                ex4Box.classList.remove('scene1-dashed');
                ex4Text.textContent = '小肠上皮细胞吸收葡萄糖';
            } else {
                // 恢复虚线和占位文字
                ex3Box.setAttribute('stroke', '#888888');
                ex3Box.classList.add('scene1-dashed');
                ex3Text.textContent = '③';
                ex4Box.setAttribute('stroke', '#888888');
                ex4Box.classList.add('scene1-dashed');
                ex4Text.textContent = '④';
            }

            // 步骤5特殊处理：选项高亮
            const optABox = document.getElementById('scene1-option-a-box');
            const optBBox = document.getElementById('scene1-option-b-box');
            const optCBox = document.getElementById('scene1-option-c-box');
            const optDBox = document.getElementById('scene1-option-d-box');

            if (currentStep >= 5) {
                optABox.setAttribute('class', 'scene1-step-el scene1-option-anim scene1-incorrect');
                optBBox.setAttribute('class', 'scene1-step-el scene1-option-anim scene1-correct');
                optCBox.setAttribute('class', 'scene1-step-el scene1-option-anim scene1-incorrect');
                optDBox.setAttribute('class', 'scene1-step-el scene1-option-anim scene1-incorrect');
                // 确保选项文字也显示
                ['scene1-option-a-text-wrap', 'scene1-option-b-text-wrap', 'scene1-option-c-text-wrap', 'scene1-option-d-text-wrap'].forEach(showElement);
            } else {
                optABox.setAttribute('class', 'scene1-step-el scene1-option-anim');
                optBBox.setAttribute('class', 'scene1-step-el scene1-option-anim');
                optCBox.setAttribute('class', 'scene1-step-el scene1-option-anim');
                optDBox.setAttribute('class', 'scene1-step-el scene1-option-anim');
            }

            updateIndicator();
        }

        btnPrev.addEventListener('click', function() {
            if (currentStep > 0) {
                currentStep--;
                renderStep();
            }
        });

        btnNext.addEventListener('click', function() {
            if (currentStep < totalSteps - 1) {
                currentStep++;
                renderStep();
            }
        });

        // 初始渲染
        renderStep();
    })();
    </script>
