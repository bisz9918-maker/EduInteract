    <script>
    (function() {
        // ===== 状态管理 =====
        let currentState = 1;
        let animId = null;
        let animStartTime = 0;

        // ===== SVG 元素引用 =====
        const membraneGroup = document.getElementById('scene2-membrane');
        const oxygenGroup = document.getElementById('scene2-oxygen-group');
        const glucoseFacilGroup = document.getElementById('scene2-glucose-facil-group');
        const glucoseActiveGroup = document.getElementById('scene2-glucose-active-group');
        const carrier = document.getElementById('scene2-carrier');
        const atpGroup = document.getElementById('scene2-atp-group');
        const adpGroup = document.getElementById('scene2-adp-group');
        const arrow = document.getElementById('scene2-arrow');
        const animParticle = document.getElementById('scene2-anim-particle');

        // ===== HTML 覆盖层引用 =====
        const labelHigh = document.getElementById('scene2-label-high');
        const labelLow = document.getElementById('scene2-label-low');
        const o2Label = document.getElementById('scene2-o2-label');
        const glucoseLabel = document.getElementById('scene2-glucose-label');
        const carrierLabel = document.getElementById('scene2-carrier-label');
        const atpLabel = document.getElementById('scene2-atp-label');
        const adpLabel = document.getElementById('scene2-adp-label');
        const panelTitle = document.getElementById('scene2-panel-title');
        const valGradient = document.getElementById('scene2-val-gradient');
        const valCarrier = document.getElementById('scene2-val-carrier');
        const valEnergy = document.getElementById('scene2-val-energy');

        // ===== 按钮引用 =====
        const btnFree = document.getElementById('scene2-btn-free');
        const btnFacil = document.getElementById('scene2-btn-facil');
        const btnActive = document.getElementById('scene2-btn-active');

        // ===== 生成磷脂双分子层 =====
        function createMembrane() {
            const headRadius = 8;
            const tailLen = 70;
            const startX = 50;
            const endX = 490;
            const step = 20;
            const topY = 190;
            const bottomY = 270;

            for (let x = startX; x <= endX; x += step) {
                // 上层头部
                const topHead = document.createElementNS('http://www.w3.org/2000/svg', 'circle');
                topHead.setAttribute('cx', x);
                topHead.setAttribute('cy', topY);
                topHead.setAttribute('r', headRadius);
                topHead.setAttribute('fill', '#90caf9');
                topHead.setAttribute('stroke', '#000');
                topHead.setAttribute('stroke-width', '1');
                membraneGroup.appendChild(topHead);

                // 上层尾部（2条虚线）
                for (let dx of [-2, 2]) {
                    const line = document.createElementNS('http://www.w3.org/2000/svg', 'line');
                    line.setAttribute('x1', x + dx);
                    line.setAttribute('y1', topY);
                    line.setAttribute('x2', x + dx);
                    line.setAttribute('y2', topY + tailLen);
                    line.setAttribute('stroke', '#888');
                    line.setAttribute('stroke-width', '1');
                    line.setAttribute('stroke-dasharray', '3,3');
                    membraneGroup.appendChild(line);
                }

                // 下层头部
                const bottomHead = document.createElementNS('http://www.w3.org/2000/svg', 'circle');
                bottomHead.setAttribute('cx', x);
                bottomHead.setAttribute('cy', bottomY);
                bottomHead.setAttribute('r', headRadius);
                bottomHead.setAttribute('fill', '#90caf9');
                bottomHead.setAttribute('stroke', '#000');
                bottomHead.setAttribute('stroke-width', '1');
                membraneGroup.appendChild(bottomHead);

                // 下层尾部（2条虚线）
                for (let dx of [-2, 2]) {
                    const line = document.createElementNS('http://www.w3.org/2000/svg', 'line');
                    line.setAttribute('x1', x + dx);
                    line.setAttribute('y1', bottomY);
                    line.setAttribute('x2', x + dx);
                    line.setAttribute('y2', bottomY - tailLen);
                    line.setAttribute('stroke', '#888');
                    line.setAttribute('stroke-width', '1');
                    line.setAttribute('stroke-dasharray', '3,3');
                    membraneGroup.appendChild(line);
                }
            }
        }

        // ===== 生成分子群 =====
        function createMolecule(group, positions, color, radius) {
            positions.forEach(pos => {
                const c = document.createElementNS('http://www.w3.org/2000/svg', 'circle');
                c.setAttribute('cx', pos[0]);
                c.setAttribute('cy', pos[1]);
                c.setAttribute('r', radius);
                c.setAttribute('fill', color);
                c.setAttribute('stroke', '#000');
                c.setAttribute('stroke-width', '0.5');
                group.appendChild(c);
            });
        }

        // 氧气分子位置（状态1）
        const oxygenOut = [[100,100],[140,110],[180,95],[220,120],[260,105],[300,130],[340,100],[380,115],[420,95],[460,120],[120,140],[200,150],[280,145],[360,155],[460,150]];
        const oxygenIn = [[200,320],[250,340],[300,310],[350,330],[400,350]];

        // 葡萄糖分子位置（状态2：协助扩散，膜外多膜内少）
        const glucoseFacilOut = [[100,100],[150,110],[200,95],[250,120],[300,105],[350,130],[400,100],[450,115],[120,140],[220,150],[320,145],[420,155]];
        const glucoseFacilIn = [[200,320],[280,340],[360,310],[440,350]];

        // 葡萄糖分子位置（状态3：主动运输，膜外少膜内多）
        const glucoseActiveOut = [[150,110],[250,100],[350,120],[450,105]];
        const glucoseActiveIn = [[100,320],[140,340],[180,310],[220,350],[260,330],[300,320],[340,340],[380,310],[420,350],[460,330],[200,360],[400,370]];

        // ===== 初始化 =====
        createMembrane();
        createMolecule(oxygenGroup, oxygenOut, '#1565c0', 6);
        createMolecule(oxygenGroup, oxygenIn, '#1565c0', 6);
        createMolecule(glucoseFacilGroup, glucoseFacilOut, '#2e7d32', 8);
        createMolecule(glucoseFacilGroup, glucoseFacilIn, '#2e7d32', 8);
        createMolecule(glucoseActiveGroup, glucoseActiveOut, '#2e7d32', 8);
        createMolecule(glucoseActiveGroup, glucoseActiveIn, '#2e7d32', 8);

        // ===== 状态切换 =====
        function switchState(newState) {
            currentState = newState;

            // 更新按钮样式
            btnFree.classList.toggle('active', newState === 1);
            btnFacil.classList.toggle('active', newState === 2);
            btnActive.classList.toggle('active', newState === 3);

            // 更新 SVG 组显隐（使用 display 避免重叠检测误报）
            oxygenGroup.style.display = newState === 1 ? 'block' : 'none';
            glucoseFacilGroup.style.display = newState === 2 ? 'block' : 'none';
            glucoseActiveGroup.style.display = newState === 3 ? 'block' : 'none';
            carrier.style.display = (newState === 2 || newState === 3) ? 'block' : 'none';

            // 更新 HTML 标签显隐
            o2Label.style.opacity = newState === 1 ? '1' : '0';
            glucoseLabel.style.opacity = (newState === 2 || newState === 3) ? '1' : '0';
            carrierLabel.style.opacity = (newState === 2 || newState === 3) ? '1' : '0';

            // ATP/ADP 初始显隐
            atpGroup.style.display = newState === 3 ? 'block' : 'none';
            adpGroup.style.display = 'none';
            atpLabel.style.opacity = newState === 3 ? '1' : '0';
            adpLabel.style.opacity = '0';

            // 箭头颜色
            arrow.setAttribute('stroke', newState === 3 ? '#c62828' : '#2e7d32');

            // 浓度标签位置
            if (newState === 1 || newState === 2) {
                labelHigh.style.top = '105px';
                labelHigh.style.left = '480px';
                labelLow.style.top = '335px';
                labelLow.style.left = '480px';
            } else {
                labelHigh.style.top = '335px';
                labelHigh.style.left = '480px';
                labelLow.style.top = '95px';
                labelLow.style.left = '480px';
            }

            // 更新信息面板
            if (newState === 1) {
                panelTitle.textContent = '自由扩散';
                valGradient.textContent = '顺浓度梯度';
                valCarrier.textContent = '不需要载体';
                valEnergy.textContent = '不消耗能量';
                valGradient.style.color = '#2e7d32';
                valCarrier.style.color = '#2e7d32';
                valEnergy.style.color = '#2e7d32';
            } else if (newState === 2) {
                panelTitle.textContent = '协助扩散（②）';
                valGradient.textContent = '顺浓度梯度';
                valCarrier.textContent = '需要载体';
                valEnergy.textContent = '不消耗能量';
                valGradient.style.color = '#2e7d32';
                valCarrier.style.color = '#2e7d32';
                valEnergy.style.color = '#2e7d32';
            } else {
                panelTitle.textContent = '主动运输（①）';
                valGradient.textContent = '逆浓度梯度';
                valCarrier.textContent = '需要载体';
                valEnergy.textContent = '消耗ATP';
                valGradient.style.color = '#c62828';
                valCarrier.style.color = '#c62828';
                valEnergy.style.color = '#c62828';
            }

            // 重置动画
            animStartTime = performance.now();
            if (animId) cancelAnimationFrame(animId);
            animId = requestAnimationFrame(animate);
        }

        // ===== 动画循环 =====
        function animate(now) {
            const t = now - animStartTime;
            animId = requestAnimationFrame(animate);

            if (currentState === 1) {
                // 状态1：自由扩散，氧气直线穿过
                animParticle.setAttribute('r', '6');
                animParticle.setAttribute('fill', '#1565c0');
                animParticle.style.opacity = '1';
                const cycle = 2000;
                const p = (t % cycle) / cycle;
                const y = 120 + p * 220;
                animParticle.setAttribute('cx', '300');
                animParticle.setAttribute('cy', y);
            } else if (currentState === 2) {
                // 状态2：协助扩散
                animParticle.setAttribute('r', '8');
                animParticle.setAttribute('fill', '#2e7d32');
                animParticle.style.opacity = '1';
                const cycle = 2200;
                let p = (t % cycle) / cycle;
                let x, y;
                if (p < 0.25) {
                    // 膜外→载体上端
                    const sub = p / 0.25;
                    x = 300 + (280 - 300) * sub;
                    y = 120 + (190 - 120) * sub;
                } else if (p < 0.40) {
                    // 停留，载体构象变化
                    x = 280;
                    y = 190;
                    const sub = (p - 0.25) / 0.15;
                    const rx = 22 + 6 * Math.sin(sub * Math.PI);
                    carrier.setAttribute('rx', rx);
                } else if (p < 0.65) {
                    // 载体上端→下端
                    const sub = (p - 0.40) / 0.25;
                    x = 280;
                    y = 190 + (270 - 190) * sub;
                    carrier.setAttribute('rx', '22');
                } else if (p < 0.90) {
                    // 载体下端→膜内
                    const sub = (p - 0.65) / 0.25;
                    x = 280 + (300 - 280) * sub;
                    y = 270 + (340 - 270) * sub;
                } else {
                    // 停留后重置
                    x = 300;
                    y = 340;
                }
                animParticle.setAttribute('cx', x);
                animParticle.setAttribute('cy', y);
            } else if (currentState === 3) {
                // 状态3：主动运输
                animParticle.setAttribute('r', '8');
                animParticle.setAttribute('fill', '#2e7d32');
                animParticle.style.opacity = '1';
                const cycle = 2500;
                let p = (t % cycle) / cycle;
                let x, y;
                if (p < 0.20) {
                    // 膜外→载体上端
                    const sub = p / 0.20;
                    x = 300 + (280 - 300) * sub;
                    y = 120 + (190 - 120) * sub;
                    atpGroup.style.display = 'block';
                    adpGroup.style.display = 'none';
                    atpLabel.style.opacity = '1';
                    adpLabel.style.opacity = '0';
                    carrier.setAttribute('rx', '22');
                } else if (p < 0.36) {
                    // 停留，ATP闪烁，载体开始变化
                    x = 280;
                    y = 190;
                    const sub = (p - 0.20) / 0.16;
                    const rx = 22 + 6 * Math.sin(sub * Math.PI);
                    carrier.setAttribute('rx', rx);
                    atpGroup.style.display = 'block';
                    adpGroup.style.display = 'none';
                    atpGroup.style.opacity = String(0.5 + 0.5 * Math.sin(sub * Math.PI * 4));
                    atpLabel.style.opacity = String(0.5 + 0.5 * Math.sin(sub * Math.PI * 4));
                    adpLabel.style.opacity = '0';
                } else if (p < 0.44) {
                    // ATP→ADP+Pi
                    x = 280;
                    y = 190;
                    const sub = (p - 0.36) / 0.08;
                    atpGroup.style.display = 'block';
                    adpGroup.style.display = 'block';
                    atpGroup.style.opacity = String(1 - sub);
                    adpGroup.style.opacity = String(sub);
                    atpLabel.style.opacity = String(1 - sub);
                    adpLabel.style.opacity = String(sub);
                    carrier.setAttribute('rx', '28');
                } else if (p < 0.64) {
                    // 载体上端→下端
                    const sub = (p - 0.44) / 0.20;
                    x = 280;
                    y = 190 + (270 - 190) * sub;
                    atpGroup.style.display = 'none';
                    adpGroup.style.display = 'block';
                    atpLabel.style.opacity = '0';
                    adpLabel.style.opacity = '1';
                    carrier.setAttribute('rx', '22');
                } else if (p < 0.84) {
                    // 载体下端→膜内
                    const sub = (p - 0.64) / 0.20;
                    x = 280 + (300 - 280) * sub;
                    y = 270 + (340 - 270) * sub;
                    atpGroup.style.display = 'none';
                    adpGroup.style.display = 'block';
                    atpLabel.style.opacity = '0';
                    adpLabel.style.opacity = '1';
                } else {
                    // 停留后重置
                    x = 300;
                    y = 340;
                    atpGroup.style.display = 'none';
                    adpGroup.style.display = 'none';
                    atpLabel.style.opacity = '0';
                    adpLabel.style.opacity = '0';
                }
                animParticle.setAttribute('cx', x);
                animParticle.setAttribute('cy', y);
            }
        }

        // ===== 事件绑定 =====
        btnFree.addEventListener('click', () => switchState(1));
        btnFacil.addEventListener('click', () => switchState(2));
        btnActive.addEventListener('click', () => switchState(3));

        // ===== 初始化状态 =====
        switchState(1);
    })();
    </script>
</body>
