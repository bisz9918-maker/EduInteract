    <script>
    (function() {
        // Scene 2 实现
        const svg = document.getElementById('scene2-svg');
        const dynamicLayer = document.getElementById('scene2-dynamic-layer');
        const pointer = document.getElementById('scene2-ammeter-pointer');
        const btnModeB = document.getElementById('scene2-btn-mode-b');
        const btnModeC = document.getElementById('scene2-btn-mode-c');
        const btnPlay = document.getElementById('scene2-btn-play');

        // 状态
        let state = {
            mode: 'b', // 'b' 或 'c'
            isPlaying: false,
            animationId: null,
            startTime: 0
        };

        // 指针旋转辅助函数
        function setPointerAngle(angle) {
            // angle: 0=右, 90=下, 180=左, 270=上
            const rad = (angle - 90) * Math.PI / 180;
            const len = 28;
            const x2 = 380 + len * Math.cos(rad);
            const y2 = 110 + len * Math.sin(rad);
            pointer.setAttribute('x2', x2);
            pointer.setAttribute('y2', y2);
        }

        // 创建兴奋波
        function createWave(cx, cy, color, id) {
            const circle = document.createElementNS('http://www.w3.org/2000/svg', 'circle');
            circle.setAttribute('cx', cx);
            circle.setAttribute('cy', cy);
            circle.setAttribute('r', 8);
            circle.setAttribute('fill', color);
            circle.setAttribute('fill-opacity', '0.7');
            circle.setAttribute('id', id);
            dynamicLayer.appendChild(circle);
            return circle;
        }

        // 创建闪电符号
        function createFlash(x, y) {
            const g = document.createElementNS('http://www.w3.org/2000/svg', 'g');
            g.setAttribute('id', 'scene2-flash');
            const path = document.createElementNS('http://www.w3.org/2000/svg', 'path');
            path.setAttribute('d', `M${x-4},${y-12} L${x+2},${y-2} L${x-2},${y-2} L${x+4},${y+12} L${x-2},${y+2} L${x+2},${y+2} Z`);
            path.setAttribute('fill', '#ffd700');
            path.setAttribute('stroke', '#e6c200');
            path.setAttribute('stroke-width', '1');
            g.appendChild(path);
            dynamicLayer.appendChild(g);
            return g;
        }

        // 创建禁止符号
        function createBlockSign(x, y) {
            const g = document.createElementNS('http://www.w3.org/2000/svg', 'g');
            g.setAttribute('id', 'scene2-block-sign');
            const circle = document.createElementNS('http://www.w3.org/2000/svg', 'circle');
            circle.setAttribute('cx', x);
            circle.setAttribute('cy', y);
            circle.setAttribute('r', 10);
            circle.setAttribute('fill', 'none');
            circle.setAttribute('stroke', '#c62828');
            circle.setAttribute('stroke-width', '2');
            const line1 = document.createElementNS('http://www.w3.org/2000/svg', 'line');
            line1.setAttribute('x1', x - 6);
            line1.setAttribute('y1', y - 6);
            line1.setAttribute('x2', x + 6);
            line1.setAttribute('y2', y + 6);
            line1.setAttribute('stroke', '#c62828');
            line1.setAttribute('stroke-width', '2');
            const line2 = document.createElementNS('http://www.w3.org/2000/svg', 'line');
            line2.setAttribute('x1', x + 6);
            line2.setAttribute('y1', y - 6);
            line2.setAttribute('x2', x - 6);
            line2.setAttribute('y2', y + 6);
            line2.setAttribute('stroke', '#c62828');
            line2.setAttribute('stroke-width', '2');
            g.appendChild(circle);
            g.appendChild(line1);
            g.appendChild(line2);
            dynamicLayer.appendChild(g);
            return g;
        }

        // 创建结果文字
        function createResultText(text, color) {
            const t = document.createElementNS('http://www.w3.org/2000/svg', 'text');
            t.setAttribute('x', 460);
            t.setAttribute('y', 110);
            t.setAttribute('id', 'scene2-result-text');
            t.setAttribute('font-size', '14');
            t.setAttribute('fill', color);
            t.setAttribute('font-weight', 'bold');
            t.textContent = text;
            dynamicLayer.appendChild(t);
            return t;
        }

        // 创建递质粒子
        function createParticles(synapseX, synapseEndX, y) {
            const particles = [];
            for (let i = 0; i < 5; i++) {
                const p = document.createElementNS('http://www.w3.org/2000/svg', 'circle');
                const offsetY = (Math.random() - 0.5) * 12;
                p.setAttribute('cx', synapseX + Math.random() * 5);
                p.setAttribute('cy', y + offsetY);
                p.setAttribute('r', 2.5);
                p.setAttribute('fill', '#5c6bc0');
                p.setAttribute('fill-opacity', '0.8');
                p.setAttribute('class', 'scene2-particle');
                dynamicLayer.appendChild(p);
                particles.push({
                    el: p,
                    startX: synapseX + Math.random() * 3,
                    endX: synapseEndX - Math.random() * 3,
                    y: y + offsetY,
                    delay: Math.random() * 0.3,
                    speed: 0.5 + Math.random() * 0.5
                });
            }
            return particles;
        }

        // 清除所有动态元素
        function clearDynamic() {
            if (state.animationId) {
                cancelAnimationFrame(state.animationId);
                state.animationId = null;
            }
            dynamicLayer.innerHTML = '';
            setPointerAngle(90);
        }

        // 重置状态
        function reset() {
            clearDynamic();
            state.isPlaying = false;
            btnPlay.disabled = false;
            btnPlay.style.opacity = '1';
        }

        // 线性插值
        function lerp(a, b, t) {
            return a + (b - a) * t;
        }

        // 动画：刺激b点
        function playAnimationB() {
            if (state.isPlaying) return;
            reset();
            state.isPlaying = true;
            btnPlay.disabled = true;
            btnPlay.style.opacity = '0.5';
            state.startTime = performance.now();

            const flash = createFlash(300, 180);
            const waveLeft = createWave(300, 250, '#2e7d32', 'scene2-wave-left');
            const waveRight = createWave(300, 250, '#2e7d32', 'scene2-wave-right');
            let particles = [];
            let waveC = null;
            let resultShown = false;

            function animate(now) {
                const t = (now - state.startTime) / 1000;

                // 1. t=0.0s: 闪电出现，兴奋波出发
                // 2. t=0.5s: 兴奋波1到达a，指针第1次偏转（左偏135°）
                // 3. t=0.5s~1.5s: 指针保持左偏
                // 4. t=1.5s: 兴奋波2到达B-C突触，触发递质释放
                // 5. t=1.5s~2.5s: 递质释放，兴奋波在C内变为红色，传向d
                // 6. t=3.0s: 兴奋波到达d，指针第2次偏转（右偏45°）
                // 7. t=3.0s~4.0s: 指针保持右偏
                // 8. t=4.0s~4.5s: 指针回到垂直
                // 9. t=4.5s: 显示结果

                // 兴奋波1: b(300) -> a(140)
                if (t <= 0.5) {
                    const progress = t / 0.5;
                    waveLeft.setAttribute('cx', lerp(300, 140, progress));
                } else if (waveLeft.parentNode) {
                    waveLeft.remove();
                }

                // 兴奋波2: b(300) -> B-C突触(430)
                if (t <= 1.5) {
                    const progress = Math.min(t / 1.5, 1);
                    waveRight.setAttribute('cx', lerp(300, 430, progress));
                } else if (waveRight.parentNode) {
                    waveRight.remove();
                }

                // 指针第1次偏转 (0.5s, 左偏135°)
                if (t >= 0.5 && t < 0.7) {
                    const p = (t - 0.5) / 0.2;
                    setPointerAngle(lerp(90, 135, p));
                } else if (t >= 0.7 && t < 3.0) {
                    setPointerAngle(135);
                }

                // 递质释放 (1.5s开始)
                if (t >= 1.5 && t < 2.5 && particles.length === 0) {
                    particles = createParticles(430, 445, 250);
                }
                if (particles.length > 0) {
                    particles.forEach(pt => {
                        const ptTime = t - 1.5 - pt.delay;
                        if (ptTime > 0 && ptTime < 1.0) {
                            const p = Math.min(ptTime / 1.0, 1);
                            pt.el.setAttribute('cx', lerp(pt.startX, pt.endX, p));
                        } else if (ptTime >= 1.0) {
                            pt.el.remove();
                        }
                    });
                }

                // C内兴奋波 (1.5s开始，从445到640，持续1.5s)
                if (t >= 1.5 && t < 3.0) {
                    if (!waveC || !waveC.parentNode) {
                        waveC = createWave(445, 250, '#c62828', 'scene2-wave-c');
                    }
                    const progress = Math.min((t - 1.5) / 1.5, 1);
                    waveC.setAttribute('cx', lerp(445, 640, progress));
                } else if (waveC && waveC.parentNode && t >= 3.0) {
                    waveC.remove();
                }

                // 指针第2次偏转 (3.0s, 右偏45°)
                if (t >= 3.0 && t < 3.2) {
                    const p = (t - 3.0) / 0.2;
                    setPointerAngle(lerp(135, 45, p));
                } else if (t >= 3.2 && t < 4.0) {
                    setPointerAngle(45);
                }

                // 指针回中 (4.0s~4.5s)
                if (t >= 4.0 && t < 4.5) {
                    const p = (t - 4.0) / 0.5;
                    setPointerAngle(lerp(45, 90, p));
                }

                // 显示结果
                if (t >= 4.5 && !resultShown) {
                    resultShown = true;
                    if (flash.parentNode) flash.remove();
                    createResultText('偏转2次', '#2e7d32');
                    state.isPlaying = false;
                    btnPlay.disabled = false;
                    btnPlay.style.opacity = '1';
                    return;
                }

                if (state.isPlaying) {
                    state.animationId = requestAnimationFrame(animate);
                }
            }

            state.animationId = requestAnimationFrame(animate);
        }

        // 动画：刺激c点
        function playAnimationC() {
            if (state.isPlaying) return;
            reset();
            state.isPlaying = true;
            btnPlay.disabled = true;
            btnPlay.style.opacity = '0.5';
            state.startTime = performance.now();

            const flash = createFlash(480, 160);
            const waveRight = createWave(480, 250, '#c62828', 'scene2-wave-right');
            const waveLeft = createWave(480, 250, '#c62828', 'scene2-wave-left');
            let blockSign = null;
            let resultShown = false;

            function animate(now) {
                const t = (now - state.startTime) / 1000;

                // 1. t=0.0s: 闪电出现，兴奋波出发
                // 2. t=0.4s: 兴奋波1到达d，指针第1次偏转（右偏45°）
                // 3. t=0.4s~1.2s: 指针保持右偏
                // 4. t=1.2s~1.8s: 兴奋波2继续向左到B-C突触后膜
                // 5. t=1.8s: 出现禁止符号
                // 6. t=1.2s~2.0s: 指针回中
                // 7. t=2.0s: 显示结果

                // 兴奋波1: c(480) -> d(640)
                if (t <= 0.4) {
                    const progress = t / 0.4;
                    waveRight.setAttribute('cx', lerp(480, 640, progress));
                } else if (waveRight.parentNode) {
                    waveRight.remove();
                }

                // 兴奋波2: c(480) -> B-C突触后膜(445)
                if (t <= 1.8) {
                    const progress = Math.min(t / 1.8, 1);
                    waveLeft.setAttribute('cx', lerp(480, 445, progress));
                } else if (waveLeft.parentNode) {
                    waveLeft.remove();
                }

                // 指针第1次偏转 (0.4s, 右偏45°)
                if (t >= 0.4 && t < 0.6) {
                    const p = (t - 0.4) / 0.2;
                    setPointerAngle(lerp(90, 45, p));
                } else if (t >= 0.6 && t < 1.2) {
                    setPointerAngle(45);
                }

                // 指针回中 (1.2s~2.0s)
                if (t >= 1.2 && t < 2.0) {
                    const p = (t - 1.2) / 0.8;
                    setPointerAngle(lerp(45, 90, p));
                }

                // 禁止符号 (1.8s)
                if (t >= 1.8 && !blockSign) {
                    blockSign = createBlockSign(437, 220);
                }

                // 显示结果
                if (t >= 2.0 && !resultShown) {
                    resultShown = true;
                    if (flash.parentNode) flash.remove();
                    if (blockSign && blockSign.parentNode) blockSign.remove();
                    createResultText('仅偏转1次', '#c62828');
                    state.isPlaying = false;
                    btnPlay.disabled = false;
                    btnPlay.style.opacity = '1';
                    return;
                }

                if (state.isPlaying) {
                    state.animationId = requestAnimationFrame(animate);
                }
            }

            state.animationId = requestAnimationFrame(animate);
        }

        // 播放按钮点击
        btnPlay.addEventListener('click', function() {
            if (state.mode === 'b') {
                playAnimationB();
            } else {
                playAnimationC();
            }
        });

        // 模式切换
        btnModeB.addEventListener('click', function() {
            if (state.mode === 'b') return;
            state.mode = 'b';
            btnModeB.classList.add('active');
            btnModeC.classList.remove('active');
            reset();
        });

        btnModeC.addEventListener('click', function() {
            if (state.mode === 'c') return;
            state.mode = 'c';
            btnModeC.classList.add('active');
            btnModeB.classList.remove('active');
            reset();
        });

        // 初始化
        setPointerAngle(90);
    })();
    </script>
