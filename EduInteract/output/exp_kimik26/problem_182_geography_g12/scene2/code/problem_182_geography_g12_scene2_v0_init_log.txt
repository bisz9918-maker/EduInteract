    <script>
    (function() {
        var currentStep = 0;
        var totalSteps = 4;

        var btnPrev = document.getElementById('scene2-btn-prev');
        var btnNext = document.getElementById('scene2-btn-next');
        var indicator = document.getElementById('scene2-step-indicator');

        var stepGroups = [
            document.getElementById('scene2-step0'),
            document.getElementById('scene2-step1'),
            document.getElementById('scene2-step2'),
            document.getElementById('scene2-step3')
        ];

        function updateDisplay() {
            stepGroups[0].style.display = 'block';
            stepGroups[0].style.opacity = '1';

            for (var i = 1; i < totalSteps; i++) {
                if (i <= currentStep) {
                    stepGroups[i].style.display = 'block';
                    stepGroups[i].style.opacity = '1';
                } else {
                    stepGroups[i].style.display = 'none';
                    stepGroups[i].style.opacity = '0';
                }
            }

            indicator.textContent = (currentStep + 1) + ' / ' + totalSteps;
            btnPrev.disabled = currentStep === 0;
            btnNext.disabled = currentStep === totalSteps - 1;
        }

        btnPrev.addEventListener('click', function() {
            if (currentStep > 0) {
                currentStep--;
                updateDisplay();
            }
        });

        btnNext.addEventListener('click', function() {
            if (currentStep < totalSteps - 1) {
                currentStep++;
                updateDisplay();
            }
        });

        updateDisplay();
    })();
    </script>
