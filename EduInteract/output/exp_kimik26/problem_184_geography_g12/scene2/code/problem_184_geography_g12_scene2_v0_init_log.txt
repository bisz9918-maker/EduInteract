    <script>
    (function() {
        // Scene 2 实现
        let currentStep = 0;
        const totalSteps = 7;

        const btnPrev = document.getElementById('scene2-btn-prev');
        const btnNext = document.getElementById('scene2-btn-next');
        const indicator = document.getElementById('scene2-step-indicator');

        // 各步骤需要显示/隐藏的元素映射
        const stepElements = {
            1: ['scene2-polarday', 'scene2-label-polarday', 'scene2-label-summer'],
            2: ['scene2-sunlight', 'scene2-sunpoint', 'scene2-label-direct'],
            3: ['scene2-line-120e', 'scene2-line-30w', 'scene2-label-0time', 'scene2-label-12time',
                'scene2-label-120e', 'scene2-label-30w', 'scene2-label-bjtime'],
            4: [], // 仅面板切换
            5: [], // 仅面板切换
            6: []  // 仅面板切换，三点高亮用样式
        };

        // 步骤3中需要改变颜色的经线
        const line0 = document.getElementById('scene2-line-0');
        const line180 = document.getElementById('scene2-line-180');

        // 三点元素
        const pointE = document.getElementById('scene2-point-e');
        const pointF = document.getElementById('scene2-point-f');
        const pointG = document.getElementById('scene2-point-g');

        function updateScene(step) {
            // 更新面板
            document.querySelectorAll('#scene2-panel .panel-step').forEach(el => {
                el.classList.toggle('active', parseInt(el.dataset.step) === step);
            });

            // 更新指示器
            indicator.textContent = '步骤 ' + (step + 1) + ' / ' + totalSteps;

            // 按钮状态
            btnPrev.disabled = step <= 0;
            btnNext.disabled = step >= totalSteps - 1;

            // 处理各步骤元素的显隐
            for (let s = 1; s <= 6; s++) {
                const ids = stepElements[s] || [];
                const shouldShow = step >= s;
                ids.forEach(id => {
                    const el = document.getElementById(id);
                    if (el) {
                        if (shouldShow) {
                            el.classList.remove('scene2-hidden');
                            el.classList.add('scene2-visible');
                        } else {
                            el.classList.remove('scene2-visible');
                            el.classList.add('scene2-hidden');
                        }
                    }
                });
            }

            // 步骤3：经线高亮
            if (step >= 3) {
                line0.setAttribute('stroke', '#1565c0');
                line0.setAttribute('stroke-width', '2');
                line180.setAttribute('stroke', '#c62828');
                line180.setAttribute('stroke-width', '2');
            } else {
                line0.setAttribute('stroke', '#888888');
                line0.setAttribute('stroke-width', '1');
                line180.setAttribute('stroke', '#888888');
                line180.setAttribute('stroke-width', '1');
            }

            // 步骤6：三点高亮（半径放大）
            if (step >= 6) {
                pointE.setAttribute('r', '6');
                pointF.setAttribute('r', '6');
                pointG.setAttribute('r', '6');
            } else {
                pointE.setAttribute('r', '4');
                pointF.setAttribute('r', '4');
                pointG.setAttribute('r', '4');
            }
        }

        btnPrev.addEventListener('click', function() {
            if (currentStep > 0) {
                currentStep--;
                updateScene(currentStep);
            }
        });

        btnNext.addEventListener('click', function() {
            if (currentStep < totalSteps - 1) {
                currentStep++;
                updateScene(currentStep);
            }
        });

        // 初始化
        updateScene(currentStep);
    })();
    </script>
