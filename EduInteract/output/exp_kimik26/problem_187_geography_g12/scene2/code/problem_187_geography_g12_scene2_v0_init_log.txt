    <script>
    (function() {
        // Scene 2 实现：以地方时正午推算四地经度
        // 状态机：currentStep 范围 0~6，对应步骤 0/7 ~ 6/7
        
        let currentStep = 0;
        const totalSteps = 7; // 0~6
        
        // 控件引用
        const btnPrev = document.getElementById('scene2-btn-prev');
        const btnNext = document.getElementById('scene2-btn-next');
        const stepIndicator = document.getElementById('scene2-step-indicator');
        
        // 计算框文字行引用 (HTML divs inside foreignObject)
        const calcLines = [
            document.getElementById('scene2-calc-line1'),
            document.getElementById('scene2-calc-line2'),
            document.getElementById('scene2-calc-line3'),
            document.getElementById('scene2-calc-line4'),
            document.getElementById('scene2-calc-line5')
        ];
        
        // 元素引用
        const meridian0 = document.getElementById('scene2-meridian-0');
        const meridian180 = document.getElementById('scene2-meridian-180');
        const meridian60e = document.getElementById('scene2-meridian-60e');
        const meridian120e = document.getElementById('scene2-meridian-120e');
        const sectorA = document.getElementById('scene2-fill-sector-a');
        const sectorB = document.getElementById('scene2-fill-sector-b');
        const pointA = document.getElementById('scene2-point-a');
        const pointB = document.getElementById('scene2-point-b');
        const pointC = document.getElementById('scene2-point-c');
        const pointD = document.getElementById('scene2-point-d');
        const labelA = document.getElementById('scene2-label-a');
        const labelB = document.getElementById('scene2-label-b');
        const labelC = document.getElementById('scene2-label-c');
        const labelD = document.getElementById('scene2-label-d');
        const formulaFO = document.getElementById('scene2-formula-fo');
        const calcFO = document.getElementById('scene2-calc-fo');
        const axisMarkA = document.getElementById('scene2-axis-mark-a');
        const axisMarkB = document.getElementById('scene2-axis-mark-b');
        const axisMarkC = document.getElementById('scene2-axis-mark-c');
        const axisMarkD = document.getElementById('scene2-axis-mark-d');
        const axisLabelA = document.getElementById('scene2-axis-label-a');
        const axisLabelB = document.getElementById('scene2-axis-label-b');
        const axisLabelC = document.getElementById('scene2-axis-label-c');
        const axisLabelD = document.getElementById('scene2-axis-label-d');
        
        // 刻度元素
        const ticks = [
            document.getElementById('scene2-tick-180w'),
            document.getElementById('scene2-tick-120w'),
            document.getElementById('scene2-tick-60w'),
            document.getElementById('scene2-tick-60e'),
            document.getElementById('scene2-tick-120e')
        ];
        const tickLabels = [
            document.getElementById('scene2-ticklabel-180w'),
            document.getElementById('scene2-ticklabel-120w'),
            document.getElementById('scene2-ticklabel-60w'),
            document.getElementById('scene2-ticklabel-60e'),
            document.getElementById('scene2-ticklabel-120e')
        ];
        
        /**
         * 统一控制元素显隐
         * 同时设置 display 和 opacity，避免残留属性导致不可见
         */
        function show(el) {
            if (!el) return;
            el.style.display = 'block';
            el.style.opacity = '1';
        }
        
        function hide(el) {
            if (!el) return;
            el.style.display = 'none';
            el.style.opacity = '0';
        }
        
        /**
         * 带淡入效果的显示（用于扇形填充）
         */
        function showWithFade(el, targetOpacity) {
            if (!el) return;
            el.style.display = 'block';
            el.style.opacity = '0';
            requestAnimationFrame(function() {
                el.style.opacity = String(targetOpacity);
            });
        }
        
        /**
         * 更新计算框文字内容
         */
        function updateCalcContent(step) {
            // 清空所有行
            calcLines.forEach(function(line) {
                line.textContent = '';
                line.style.color = '#000';
                line.style.fontWeight = 'normal';
            });
            
            if (step === 2) {
                calcLines[0].textContent = '① A地：当地12:00 = 世界时4:00';
                calcLines[1].textContent = '② 时差 = 12 - 4 = +8小时（早于世界时）';
                calcLines[2].textContent = '③ 经度 = 8 × 15° = 120°E';
                calcLines[3].textContent = '④ 结果：120°E';
                calcLines[3].style.color = '#2e7d32';
                calcLines[3].style.fontWeight = 'bold';
            } else if (step === 3) {
                calcLines[0].textContent = '① B地：当地12:00 = 世界时8:00';
                calcLines[1].textContent = '② 时差 = 12 - 8 = +4小时（早于世界时）';
                calcLines[2].textContent = '③ 经度 = 4 × 15° = 60°E';
                calcLines[3].textContent = '④ 结果：60°E';
                calcLines[3].style.color = '#2e7d32';
                calcLines[3].style.fontWeight = 'bold';
            } else if (step === 4) {
                calcLines[0].textContent = '① C地：当地12:00 = 世界时12:00';
                calcLines[1].textContent = '② 时差 = 0小时';
                calcLines[2].textContent = '③ 经度 = 0°';
                calcLines[3].textContent = '④ 结果：0°（本初子午线）';
                calcLines[3].style.color = '#2e7d32';
                calcLines[3].style.fontWeight = 'bold';
            } else if (step === 5) {
                calcLines[0].textContent = '① D地：当地12:00 = 世界时24:00（次日0:00）';
                calcLines[1].textContent = '② 时差 = +12小时（或 -12小时）';
                calcLines[2].textContent = '③ 经度 = 12 × 15° = 180°';
                calcLines[3].textContent = '④ 结果：180°（国际日期变更线）';
                calcLines[3].style.color = '#c62828';
                calcLines[3].style.fontWeight = 'bold';
                calcLines[4].textContent = '提示：日界线附近，日期需±1天';
                calcLines[4].style.color = '#c62828';
            } else if (step === 6) {
                calcLines[0].textContent = '四地经度汇总：';
                calcLines[0].style.fontWeight = 'bold';
                calcLines[1].textContent = 'A地 → 120°E';
                calcLines[1].style.color = '#1565c0';
                calcLines[2].textContent = 'B地 → 60°E';
                calcLines[2].style.color = '#1565c0';
                calcLines[3].textContent = 'C地 → 0°';
                calcLines[3].style.color = '#2e7d32';
                calcLines[4].textContent = 'D地 → 180°';
                calcLines[4].style.color = '#c62828';
            }
        }
        
        /**
         * 主更新函数：根据 currentStep 更新所有视觉元素
         */
        function updateScene() {
            stepIndicator.textContent = '步骤 ' + currentStep + '/7';
            btnPrev.disabled = currentStep <= 0;
            btnNext.disabled = currentStep >= 6;
            
            // 公式框：步骤1+ 显示
            if (currentStep >= 1) {
                show(formulaFO);
            } else {
                hide(formulaFO);
            }
            
            // 计算框：步骤2+ 显示
            if (currentStep >= 2) {
                show(calcFO);
                updateCalcContent(currentStep);
            } else {
                hide(calcFO);
            }
            
            // 刻度线及标签：步骤1+ 显示全部7个刻度
            ticks.forEach(function(t) {
                if (currentStep >= 1) show(t); else hide(t);
            });
            tickLabels.forEach(function(t) {
                if (currentStep >= 1) show(t); else hide(t);
            });
            
            // 120°E 及 A地：步骤2+ 显示
            if (currentStep >= 2) {
                show(meridian120e);
                show(pointA);
                show(labelA);
                show(axisMarkA);
                show(axisLabelA);
                showWithFade(sectorA, 0.1);
            } else {
                hide(meridian120e);
                hide(pointA);
                hide(labelA);
                hide(axisMarkA);
                hide(axisLabelA);
                hide(sectorA);
            }
            
            // 60°E 及 B地：步骤3+ 显示
            if (currentStep >= 3) {
                show(meridian60e);
                show(pointB);
                show(labelB);
                show(axisMarkB);
                show(axisLabelB);
                showWithFade(sectorB, 0.1);
            } else {
                hide(meridian60e);
                hide(pointB);
                hide(labelB);
                hide(axisMarkB);
                hide(axisLabelB);
                hide(sectorB);
            }
            
            // C地：步骤4+ 显示
            if (currentStep >= 4) {
                show(pointC);
                show(labelC);
                show(axisMarkC);
                show(axisLabelC);
            } else {
                hide(pointC);
                hide(labelC);
                hide(axisMarkC);
                hide(axisLabelC);
            }
            
            // 0°经线高亮：步骤4+ 绿色加粗
            if (currentStep >= 4) {
                meridian0.setAttribute('stroke', '#2e7d32');
                meridian0.setAttribute('stroke-width', '3');
            } else {
                meridian0.setAttribute('stroke', '#000000');
                meridian0.setAttribute('stroke-width', '2');
            }
            
            // D地：步骤5+ 显示
            if (currentStep >= 5) {
                show(pointD);
                show(labelD);
                show(axisMarkD);
                show(axisLabelD);
            } else {
                hide(pointD);
                hide(labelD);
                hide(axisMarkD);
                hide(axisLabelD);
            }
            
            // 180°经线高亮：步骤5+ 红色加粗
            if (currentStep >= 5) {
                meridian180.setAttribute('stroke', '#c62828');
                meridian180.setAttribute('stroke-width', '3');
            } else {
                meridian180.setAttribute('stroke', '#c62828');
                meridian180.setAttribute('stroke-width', '2');
            }
        }
        
        // 事件绑定
        btnPrev.addEventListener('click', function() {
            if (currentStep > 0) {
                currentStep--;
                updateScene();
            }
        });
        
        btnNext.addEventListener('click', function() {
            if (currentStep < 6) {
                currentStep++;
                updateScene();
            }
        });
        
        // 初始渲染
        updateScene();
    })();
    </script>
