    <script>
    (function() {
        // Scene 3 实现
        const STEP_MAX = 5;
        let currentStep = 0;

        const btnPrev = document.getElementById('scene3-btn-prev');
        const btnNext = document.getElementById('scene3-btn-next');
        const stepLabel = document.getElementById('scene3-step-label');

        // 各步骤对应的动态元素组
        const stepGroups = [
            null, // 步骤0无额外动态组（基础元素始终显示）
            document.getElementById('scene3-step1-group'),
            document.getElementById('scene3-step2-group'),
            document.getElementById('scene3-step3-group'),
            document.getElementById('scene3-step4-group'),
            document.getElementById('scene3-step5-group')
        ];

        // 步骤1中需要在步骤2替换的标注
        const label20w = document.getElementById('scene3-label-20w');
        const label160e = document.getElementById('scene3-label-160e');
        const labelDusk = document.getElementById('scene3-label-dusk');
        const labelDawn = document.getElementById('scene3-label-dawn');

        // 步骤3的右侧标签，在步骤5时需要隐藏以避免与时间面板重叠
        const label110w = document.getElementById('scene3-label-110w');
        const labelSunNoon = document.getElementById('scene3-label-sun-noon');
        const labelEPoint = document.getElementById('scene3-label-e-point');

        function setStep(step) {
            if (step < 0 || step > STEP_MAX) return;
            currentStep = step;

            // 更新步骤标签
            stepLabel.textContent = '步骤 ' + (currentStep + 1) + '/6';

            // 更新按钮状态
            btnPrev.disabled = (currentStep === 0);
            btnNext.disabled = (currentStep === STEP_MAX);

            // 隐藏所有动态组
            for (let i = 1; i <= STEP_MAX; i++) {
                if (stepGroups[i]) {
                    stepGroups[i].style.display = 'none';
                }
            }

            // 显示当前步骤及之前的所有动态组（累积显示）
            for (let i = 1; i <= currentStep; i++) {
                if (stepGroups[i]) {
                    stepGroups[i].style.display = 'block';
                }
            }

            // 特殊处理：步骤2时隐藏步骤1的20°W和160°E标注，显示昏线/晨线
            if (currentStep >= 2) {
                if (label20w) label20w.style.display = 'none';
                if (label160e) label160e.style.display = 'none';
                if (labelDusk) labelDusk.style.display = 'block';
                if (labelDawn) labelDawn.style.display = 'block';
            } else {
                if (label20w) label20w.style.display = 'block';
                if (label160e) label160e.style.display = 'block';
                if (labelDusk) labelDusk.style.display = 'none';
                if (labelDawn) labelDawn.style.display = 'none';
            }

            // 步骤5时隐藏右侧标签，避免与时间面板重叠
            if (currentStep >= 5) {
                if (label110w) label110w.style.display = 'none';
                if (labelSunNoon) labelSunNoon.style.display = 'none';
                if (labelEPoint) labelEPoint.style.display = 'none';
            } else {
                if (label110w) label110w.style.display = 'block';
                if (labelSunNoon) labelSunNoon.style.display = 'block';
                if (labelEPoint) labelEPoint.style.display = 'block';
            }
        }

        btnPrev.addEventListener('click', function() {
            setStep(currentStep - 1);
        });

        btnNext.addEventListener('click', function() {
            setStep(currentStep + 1);
        });

        // 初始化
        setStep(0);
    })();
    </script>
