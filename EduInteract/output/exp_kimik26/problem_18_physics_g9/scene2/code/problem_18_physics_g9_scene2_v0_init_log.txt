    <script>
    (function() {
        // Scene 2 实现
        let currentStep = 0;
        const totalSteps = 6;

        const stepGroups = [];
        for (let i = 0; i < totalSteps; i++) {
            stepGroups.push(document.getElementById('scene2-step' + i + '-group'));
        }

        const cardA = document.getElementById('scene2-card-a');
        const cardB = document.getElementById('scene2-card-b');
        const cardC = document.getElementById('scene2-card-c');
        const cardD = document.getElementById('scene2-card-d');

        const iconA = document.getElementById('scene2-icon-a');
        const iconB = document.getElementById('scene2-icon-b');
        const iconC = document.getElementById('scene2-icon-c');
        const iconD = document.getElementById('scene2-icon-d');

        const btnPrev = document.getElementById('scene2-btn-prev');
        const btnNext = document.getElementById('scene2-btn-next');
        const indicator = document.getElementById('scene2-step-indicator');

        const arrowS1 = document.getElementById('scene2-arrow-s1');
        const arrowS2_1 = document.getElementById('scene2-arrow-s2-1');
        const arrowS2_2 = document.getElementById('scene2-arrow-s2-2');
        const arrowS3 = document.getElementById('scene2-arrow-s3');

        function show(el) {
            if (!el) return;
            el.style.display = 'block';
            void el.offsetWidth;
            el.style.opacity = '1';
        }

        function hide(el) {
            if (!el) return;
            el.style.opacity = '0';
            setTimeout(function() {
                if (el.style.opacity === '0') el.style.display = 'none';
            }, 200);
        }

        function setCardStyle(card, type) {
            if (type === 'correct') {
                card.style.border = '2px solid #1565c0';
                card.style.background = 'rgba(21,101,192,0.08)';
            } else if (type === 'wrong') {
                card.style.border = '1px solid #c62828';
                card.style.background = '#ffffff';
            } else {
                card.style.border = '1px solid #000000';
                card.style.background = '#ffffff';
            }
        }

        function updateUI() {
            indicator.textContent = '步骤 ' + (currentStep + 1) + ' / ' + totalSteps;
            btnPrev.disabled = (currentStep === 0);
            btnNext.disabled = (currentStep === totalSteps - 1);

            for (let i = 0; i < totalSteps; i++) {
                if (i === currentStep) {
                    show(stepGroups[i]);
                } else {
                    hide(stepGroups[i]);
                }
            }

            setCardStyle(cardA, 'default');
            setCardStyle(cardB, 'default');
            setCardStyle(cardC, 'default');
            setCardStyle(cardD, 'default');

            iconA.style.display = 'none'; iconA.style.opacity = '0';
            iconB.style.display = 'none'; iconB.style.opacity = '0';
            iconC.style.display = 'none'; iconC.style.opacity = '0';
            iconD.style.display = 'none'; iconD.style.opacity = '0';

            if (currentStep >= 1) {
                setCardStyle(cardA, 'correct');
                show(iconA);
            }
            if (currentStep >= 2) {
                setCardStyle(cardB, 'wrong');
                show(iconB);
            }
            if (currentStep >= 3) {
                setCardStyle(cardC, 'wrong');
                show(iconC);
            }
            if (currentStep >= 4) {
                setCardStyle(cardD, 'wrong');
                show(iconD);
            }

            arrowS1.style.display = (currentStep === 1) ? 'block' : 'none';
            arrowS2_1.style.display = (currentStep === 2) ? 'block' : 'none';
            arrowS2_2.style.display = (currentStep === 2) ? 'block' : 'none';
            arrowS3.style.display = (currentStep === 3) ? 'block' : 'none';
        }

        btnPrev.addEventListener('click', function() {
            if (currentStep > 0) {
                currentStep--;
                updateUI();
            }
        });

        btnNext.addEventListener('click', function() {
            if (currentStep < totalSteps - 1) {
                currentStep++;
                updateUI();
            }
        });

        updateUI();
    })();
    </script>
