(function(){

var btnPrev=document.getElementById('scene1-btn-prev');

var elBaseHighlight=document.getElementById('scene1-base-highlight');

var elLabelStep4a=document.getElementById('scene1-label-step4a');
var elLabelStep4b=document.getElementById('scene1-label-step4b');
function setOpacity(el,val){el.setAttribute('opacity',val);}
function updateButtons(){btnPrev.disabled=currentStep===0;btnNext.disabled=currentStep===totalSteps;stepIndicator.textContent='步骤 '+currentStep+'/'+totalSteps;}
function renderStep(){
if(currentStep===0){
setOpacity(elBaseHighlight,'0');setOpacity(elExtEA,'0');setOpacity(elExtED,'0');setOpacity(elPointE,'0');setOpacity(elPlanePAB,'0');setOpacity(elDCHighlight,'0');
setOpacity(elLabelStep1,'0');setOpacity(elLabelStep2,'0');setOpacity(elLabelStep3,'0');setOpacity(elLabelStep4a,'0');setOpacity(elLabelStep4b,'0');
}else if(currentStep===1){
setOpacity(elBaseHighlight,'1');setOpacity(elExtEA,'0');setOpacity(elExtED,'0');setOpacity(elPointE,'0');setOpacity(elPlanePAB,'0');setOpacity(elDCHighlight,'0');
setOpacity(elLabelStep1,'1');setOpacity(elLabelStep2,'0');setOpacity(elLabelStep3,'0');setOpacity(elLabelStep4a,'0');setOpacity(elLabelStep4b,'0');
}else if(currentStep===2){
setOpacity(elBaseHighlight,'1');setOpacity(elExtEA,'1');setOpacity(elExtED,'1');setOpacity(elPointE,'1');setOpacity(elPlanePAB,'0');setOpacity(elDCHighlight,'0');
setOpacity(elLabelStep1,'1');setOpacity(elLabelStep2,'1');setOpacity(elLabelStep3,'0');setOpacity(elLabelStep4a,'0');setOpacity(elLabelStep4b,'0');
}else if(currentStep===3){
setOpacity(elBaseHighlight,'1');setOpacity(elExtEA,'1');setOpacity(elExtED,'1');setOpacity(elPointE,'1');setOpacity(elPlanePAB,'1');setOpacity(elDCHighlight,'1');

}else if(currentStep===4){
setOpacity(elBaseHighlight,'1');setOpacity(elExtEA,'1');setOpacity(elExtED,'1');setOpacity(elPointE,'1');setOpacity(elPlanePAB,'1');setOpacity(elDCHighlight,'1');
setOpacity(elLabelStep1,'1');setOpacity(elLabelStep2,'1');setOpacity(elLabelStep3,'1');setOpacity(elLabelStep4a,'1');setOpacity(elLabelStep4b,'1');
}
updateButtons();
}
btnPrev.addEventListener('click',function(){if(currentStep>0){currentStep--;renderStep();}});
btnNext.addEventListener('click',function(){if(currentStep<totalSteps){currentStep++;renderStep();}});
renderStep();
})();
