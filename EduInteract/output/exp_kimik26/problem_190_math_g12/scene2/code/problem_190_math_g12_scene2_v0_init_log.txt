    <script>
    (function() {
        // Scene 2 实现：命题②——平面PAB内存在无数条直线与平面PDC平行
        
        let currentStep = 0;
        const totalSteps = 5;
        
        const btnPrev = document.getElementById('scene2-btn-prev');
        const btnNext = document.getElementById('scene2-btn-next');
        const indicator = document.getElementById('scene2-step-indicator');
        
        const groups = [
            document.getElementById('scene2-group-step1'),
            document.getElementById('scene2-group-step2'),
            document.getElementById('scene2-group-step3'),
            document.getElementById('scene2-group-step4')
        ];
        
        function updateUI() {
            // 更新步骤指示器
            indicator.textContent = '步骤 ' + (currentStep + 1) + ' / ' + totalSteps;
            
            // 更新按钮状态
            btnPrev.disabled = (currentStep === 0);
            btnNext.disabled = (currentStep === totalSteps - 1);
            
            // 控制各步骤组的显示/隐藏
            // 步骤0：只显示基础图形（所有group隐藏）
            // 步骤1：显示group-step1
            // 步骤2：显示group-step1, group-step2
            // 步骤3：显示group-step1, group-step2, group-step3
            // 步骤4：显示group-step1, group-step2, group-step3, group-step4
            for (let i = 0; i < groups.length; i++) {
                if (groups[i]) {
                    groups[i].style.display = (i < currentStep) ? 'block' : 'none';
                }
            }
        }
        
        function nextStep() {
            if (currentStep < totalSteps - 1) {
                currentStep++;
                updateUI();
            }
        }
        
        function prevStep() {
            if (currentStep > 0) {
                currentStep--;
                updateUI();
            }
        }
        
        btnPrev.addEventListener('click', prevStep);
        btnNext.addEventListener('click', nextStep);
        
        // 初始化
        updateUI();
    })();
    </script>
