        <div id="scene3-formula-0" class="scene3-formula-box active">
            <div class="scene3-formula-title" style="color: #1565c0;">半圆锥</div>
            <div class="scene3-formula-content">
                $$V_1 = \frac{1}{2} \cdot \frac{1}{3}\pi r^2 h$$
                $$= \frac{1}{2} \cdot \frac{1}{3}\pi \cdot 1^2 \cdot a$$
                $$= \color{#1565c0}{\frac{\pi a}{6}}$$
            </div>
        </div>

        <!-- 步骤1：三棱锥体积公式 -->
        <div id="scene3-formula-1" class="scene3-formula-box">
            <div class="scene3-formula-title" style="color: #c62828;">三棱锥</div>
            <div class="scene3-formula-content">
                $$V_2 = \frac{1}{3} S_{\text{底}} \cdot h$$
                $$= \frac{1}{3} \cdot \left(\frac{1}{2} \cdot 2 \cdot 1\right) \cdot a$$
                $$= \color{#c62828}{\frac{a}{3}}$$
            </div>
        </div>

        <!-- 步骤2：总体积合并 -->
        <div id="scene3-formula-2" class="scene3-formula-box">
            <div class="scene3-formula-title" style="color: #2e7d32;">总体积</div>
            <div class="scene3-formula-content">
                $$V = V_1 + V_2$$
                $$= \frac{\pi a}{6} + \frac{a}{3}$$
                $$= \frac{\pi a}{6} + \frac{2a}{6}$$
                $$= \color{#2e7d32}{\frac{(\pi+2)a}{6}}$$
            </div>
        </div>

        <!-- 步骤3：求解a -->
        <div id="scene3-formula-3" class="scene3-formula-box">
            <div class="scene3-formula-title" style="color: #2e7d32;">求解 a</div>
            <div class="scene3-formula-content">
                $$V = \frac{\pi + 2}{6}$$
                $$\frac{(\pi+2)a}{6} = \frac{\pi+2}{6}$$
                <div class="scene3-answer-box">
                    $$\Rightarrow a = 1$$
                </div>
                <div class="scene3-answer-label">答案</div>
            </div>
        </div>

    <script>
    (function() {
        // Scene 3 实现
        let currentStep = 0;
        const totalSteps = 4;

        const btnPrev = document.getElementById('scene3-btn-prev');
        const btnNext = document.getElementById('scene3-btn-next');
        const stepIndicator = document.getElementById('scene3-step-indicator');

        const graphicBoxes = [
            document.getElementById('scene3-graphic-0'),
            document.getElementById('scene3-graphic-1'),
            document.getElementById('scene3-graphic-2')
        ];

        const formulaBoxes = [
            document.getElementById('scene3-formula-0'),
            document.getElementById('scene3-formula-1'),
            document.getElementById('scene3-formula-2'),
            document.getElementById('scene3-formula-3')
        ];

        function updateUI() {
            // 更新按钮状态
            btnPrev.disabled = currentStep === 0;
            btnNext.disabled = currentStep === totalSteps - 1;
            stepIndicator.textContent = (currentStep + 1) + ' / ' + totalSteps;

            // 更新左侧图形显示
            graphicBoxes.forEach((box, index) => {
                if (index === currentStep) {
                    box.classList.add('active');
                } else {
                    box.classList.remove('active');
                }
            });

            // 更新右侧公式显示
            formulaBoxes.forEach((box, index) => {
                if (index === currentStep) {
                    box.classList.add('active');
                } else {
                    box.classList.remove('active');
                }
            });
        }

        btnPrev.addEventListener('click', function() {
            if (currentStep > 0) {
                currentStep--;
                updateUI();
            }
        });

        btnNext.addEventListener('click', function() {
            if (currentStep < totalSteps - 1) {
                currentStep++;
                updateUI();
            }
        });

        // 初始化
        updateUI();
    })();
    </script>
