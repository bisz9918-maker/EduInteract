    <script>
    (function() {
        let currentStep = 0;
        const totalSteps = 5;
        const stepGroups = [
            document.getElementById('scene1-step0-base'),
            document.getElementById('scene1-step1-trajectory'),
            document.getElementById('scene1-step2-highlight'),
            document.getElementById('scene1-step3-arc'),
            document.getElementById('scene1-step4-return')
        ];
        const btnPrev = document.getElementById('scene1-btn-prev');
        const btnNext = document.getElementById('scene1-btn-next');
        const indicator = document.getElementById('scene1-step-indicator');

        function renderStep() {
            stepGroups[0].style.display = 'block';
            for (let i = 1; i < stepGroups.length; i++) {
                stepGroups[i].style.display = i <= currentStep ? 'block' : 'none';
            }
            btnPrev.disabled = currentStep === 0;
            btnNext.disabled = currentStep === totalSteps - 1;
            indicator.textContent = '步骤 ' + (currentStep + 1) + '/' + totalSteps;
        }

        btnPrev.addEventListener('click', function() {
            if (currentStep > 0) { currentStep--; renderStep(); }
        });

        btnNext.addEventListener('click', function() {
            if (currentStep < totalSteps - 1) { currentStep++; renderStep(); }
        });

        renderStep();
    })();
    </script>
