    <script>
    (function() {
        // 状态机
        let currentStep = 0;
        const stepTitles = [
            '步骤 0/4：坐标系',
            '步骤 1/4：水平分运动',
            '步骤 2/4：竖直分运动',
            '步骤 3/4：合运动轨迹',
            '步骤 4/4：公式总结'
        ];

        // 按步骤分组的元素 ID 列表
        const stepGroups = {
            0: [
                'scene2-axis-x', 'scene2-axis-y',
                'scene2-dot-o', 'scene2-label-o',
                'scene2-label-x', 'scene2-label-y',
                'scene2-arrow-v0', 'scene2-label-v0',
                'scene2-arrow-a', 'scene2-label-a'
            ],
            1: [
                'scene2-dot-b1', 'scene2-dot-b2', 'scene2-dot-b3',
                'scene2-label-t1x', 'scene2-label-t2x', 'scene2-label-t3x',
                'scene2-formula-x'
            ],
            2: [
                'scene2-dot-r1', 'scene2-dot-r2', 'scene2-dot-r3',
                'scene2-label-t1y', 'scene2-label-t2y', 'scene2-label-t3y',
                'scene2-formula-y'
            ],
            3: [
                'scene2-aux-v1', 'scene2-aux-h1',
                'scene2-aux-v2', 'scene2-aux-h2',
                'scene2-aux-v3', 'scene2-aux-h3',
                'scene2-dot-p1', 'scene2-dot-p2', 'scene2-dot-p3',
                'scene2-label-p1', 'scene2-label-p2', 'scene2-label-p3',
                'scene2-trajectory',
                'scene2-arrow-v', 'scene2-label-v'
            ],
            4: [
                'scene2-formula-summary'
            ]
        };

        // 缓存元素引用
        const elCache = {};
        function getEl(id) {
            if (!elCache[id]) {
                elCache[id] = document.getElementById(id);
            }
            return elCache[id];
        }

        // 初始化抛物线路径：y = 100 + 0.002*(x-120)^2
        function initTrajectory() {
            const path = getEl('scene2-trajectory');
            let d = '';
            for (let x = 120; x <= 420; x += 5) {
                const y = 100 + 0.002 * (x - 120) * (x - 120);
                d += (d === '' ? 'M' : 'L') + x + ',' + y;
            }
            path.setAttribute('d', d);
        }

        // 渲染当前步骤
        function renderStep() {
            // 步骤0始终显示
            stepGroups[0].forEach(id => {
                const el = getEl(id);
                if (el) el.style.opacity = '1';
            });

            // 步骤1-4根据 currentStep 显隐
            for (let s = 1; s <= 4; s++) {
                const visible = currentStep >= s ? '1' : '0';
                stepGroups[s].forEach(id => {
                    const el = getEl(id);
                    if (el) el.style.opacity = visible;
                });
            }

            // 更新按钮状态
            getEl('scene2-btn-prev').disabled = (currentStep === 0);
            getEl('scene2-btn-next').disabled = (currentStep === 4);

            // 更新步骤指示文本
            getEl('scene2-step-text').textContent = stepTitles[currentStep];
        }

        // 事件绑定
        getEl('scene2-btn-prev').addEventListener('click', function() {
            if (currentStep > 0) {
                currentStep--;
                renderStep();
            }
        });

        getEl('scene2-btn-next').addEventListener('click', function() {
            if (currentStep < 4) {
                currentStep++;
                renderStep();
            }
        });

        // 初始化
        initTrajectory();
        renderStep();
    })();
    </script>
