    <script>
    (function() {
        // Scene 2 实现：F-ω² 关系图分步揭示
        const MAX_STEP = 5;
        let currentStep = 0;

        const btnPrev = document.getElementById('scene2-btn-prev');
        const btnNext = document.getElementById('scene2-btn-next');
        const counter = document.getElementById('scene2-step-counter');

        // 收集所有带 data-step 的元素
        const allElements = document.querySelectorAll('#scene2-svg [data-step]');

        // 初始化画线动画：设置 stroke-dasharray 和 stroke-dashoffset
        function initLineAnimation(el) {
            if (el.tagName === 'polyline' || el.tagName === 'line') {
                const length = el.getTotalLength ? el.getTotalLength() : 1000;
                el.style.strokeDasharray = length;
                el.style.strokeDashoffset = length;
            }
        }

        // 为所有曲线元素初始化动画
        const lineElements = document.querySelectorAll('#scene2-svg .scene2-animate-line');
        lineElements.forEach(initLineAnimation);

        // 高亮圆点特殊处理：初始 r=0
        const highlightDot = document.getElementById('scene2-highlight-dot');
        if (highlightDot) {
            highlightDot.setAttribute('r', '0');
        }

        function updateButtons() {
            btnPrev.disabled = currentStep <= 0;
            btnNext.disabled = currentStep >= MAX_STEP;
            counter.textContent = '步骤 ' + currentStep + '/' + MAX_STEP;
        }

        function showElement(el) {
            el.style.display = '';
            // 强制重排以触发动画
            void el.offsetWidth;
            el.style.opacity = '1';
            // 对线条元素触发画线动画
            if (el.classList.contains('scene2-animate-line')) {
                el.style.strokeDashoffset = '0';
            }
            // 对高亮圆点触发缩放动画
            if (el.id === 'scene2-highlight-dot') {
                el.setAttribute('r', '5');
            }
        }

        function hideElement(el) {
            el.style.opacity = '0';
            if (el.classList.contains('scene2-animate-line')) {
                const length = el.getTotalLength ? el.getTotalLength() : 1000;
                el.style.strokeDashoffset = length;
            }
            if (el.id === 'scene2-highlight-dot') {
                el.setAttribute('r', '0');
            }
            // 延迟后设置 display:none 以避免动画中断
            setTimeout(() => {
                if (parseFloat(el.style.opacity) === 0) {
                    el.style.display = 'none';
                }
            }, 400);
        }

        function renderStep() {
            allElements.forEach(el => {
                const step = parseInt(el.getAttribute('data-step'), 10);
                if (step <= currentStep) {
                    showElement(el);
                } else {
                    hideElement(el);
                }
            });

            // 图例高亮处理
            const legendMap = [
                { id: 'scene2-legend-fA', step: 1 },
                { id: 'scene2-legend-TAB', step: 2 },
                { id: 'scene2-legend-fB', step: 3 },
                { id: 'scene2-legend-TOB', step: 4 }
            ];
            legendMap.forEach(item => {
                const el = document.getElementById(item.id);
                if (el) {
                    el.style.opacity = currentStep >= item.step ? '1' : '0.3';
                }
            });

            updateButtons();
        }

        btnPrev.addEventListener('click', function() {
            if (currentStep > 0) {
                currentStep--;
                renderStep();
            }
        });

        btnNext.addEventListener('click', function() {
            if (currentStep < MAX_STEP) {
                currentStep++;
                renderStep();
            }
        });

        // 初始渲染
        renderStep();
    })();
    </script>
