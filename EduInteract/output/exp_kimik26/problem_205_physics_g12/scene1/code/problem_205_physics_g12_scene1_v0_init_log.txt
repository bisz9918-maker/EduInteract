    <script>
    (function() {
        // Scene 1 实现
        var currentStep = 0;
        var maxStep = 4;

        var btnPrev = document.getElementById('scene1-btn-prev');
        var btnNext = document.getElementById('scene1-btn-next');
        var btnReset = document.getElementById('scene1-btn-reset');

        // 各步骤需要显示的元素ID列表
        var stepElements = {
            1: ['scene1-track-mn-highlight', 'scene1-label-s', 'scene1-force-qe', 'scene1-label-qe', 'scene1-force-f', 'scene1-label-f'],
            2: ['scene1-arc-nl-highlight', 'scene1-line-2r', 'scene1-label-2r'],
            3: ['scene1-arc-np-highlight', 'scene1-line-v', 'scene1-line-h', 'scene1-label-rv', 'scene1-label-rh'],
            4: ['scene1-circle-l', 'scene1-label-vmin', 'scene1-label-critical']
        };

        // 步骤2需要隐藏步骤1的元素
        var step1Only = ['scene1-track-mn-highlight', 'scene1-label-s', 'scene1-force-qe', 'scene1-label-qe', 'scene1-force-f', 'scene1-label-f'];
        // 步骤3需要隐藏步骤2的元素
        var step2Only = ['scene1-arc-nl-highlight', 'scene1-line-2r', 'scene1-label-2r'];
        // 步骤4需要隐藏步骤3的元素，但保留N→L高亮
        var step3Only = ['scene1-arc-np-highlight', 'scene1-line-v', 'scene1-line-h', 'scene1-label-rv', 'scene1-label-rh'];

        function setOpacity(id, val) {
            var el = document.getElementById(id);
            if (el) el.setAttribute('opacity', val);
        }

        function render() {
            // 先隐藏所有步骤元素
            for (var s = 1; s <= maxStep; s++) {
                var ids = stepElements[s];
                if (ids) {
                    for (var i = 0; i < ids.length; i++) {
                        setOpacity(ids[i], 0);
                    }
                }
            }

            // 根据当前步骤显示对应元素
            if (currentStep >= 1) {
                // 步骤1元素
                var ids1 = stepElements[1];
                for (var i = 0; i < ids1.length; i++) setOpacity(ids1[i], 1);
            }
            if (currentStep >= 2) {
                // 隐藏步骤1-only元素
                for (var i = 0; i < step1Only.length; i++) setOpacity(step1Only[i], 0);
                // 显示步骤2元素
                var ids2 = stepElements[2];
                for (var i = 0; i < ids2.length; i++) setOpacity(ids2[i], 1);
            }
            if (currentStep >= 3) {
                // 隐藏步骤2-only元素
                for (var i = 0; i < step2Only.length; i++) setOpacity(step2Only[i], 0);
                // 显示步骤3元素
                var ids3 = stepElements[3];
                for (var i = 0; i < ids3.length; i++) setOpacity(ids3[i], 1);
            }
            if (currentStep >= 4) {
                // 隐藏步骤3-only元素
                for (var i = 0; i < step3Only.length; i++) setOpacity(step3Only[i], 0);
                // 保留步骤2的N→L高亮
                setOpacity('scene1-arc-nl-highlight', 1);
                // 显示步骤4元素
                var ids4 = stepElements[4];
                for (var i = 0; i < ids4.length; i++) setOpacity(ids4[i], 1);
            }

            // 更新按钮状态
            btnPrev.disabled = currentStep <= 0;
            btnNext.disabled = currentStep >= maxStep;
        }

        btnPrev.addEventListener('click', function() {
            if (currentStep > 0) {
                currentStep--;
                render();
            }
        });

        btnNext.addEventListener('click', function() {
            if (currentStep < maxStep) {
                currentStep++;
                render();
            }
        });

        btnReset.addEventListener('click', function() {
            currentStep = 0;
            render();
        });

        // 初始渲染
        render();
    })();
    </script>
