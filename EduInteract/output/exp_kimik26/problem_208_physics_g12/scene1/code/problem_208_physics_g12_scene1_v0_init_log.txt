    <script>
    (function() {
        // Scene 1 实现
        let scene1_step = 0;

        // 生成磁场点阵
        const dotsGroup = document.getElementById('scene1-dots');
        for (let x = 465; x <= 685; x += 30) {
            for (let y = 195; y <= 375; y += 30) {
                const circle = document.createElementNS('http://www.w3.org/2000/svg', 'circle');
                circle.setAttribute('cx', x);
                circle.setAttribute('cy', y);
                circle.setAttribute('r', 2);
                circle.setAttribute('fill', '#1565c0');
                circle.setAttribute('stroke', '#1565c0');
                dotsGroup.appendChild(circle);
            }
        }

        // 元素引用
        const efield1 = document.getElementById('scene1-efield1');
        const efield2 = document.getElementById('scene1-efield2');
        const efield3 = document.getElementById('scene1-efield3');

        const trajLine = document.getElementById('scene1-trajectory-line');
        const trajParabola = document.getElementById('scene1-trajectory-parabola');
        const forceFe = document.getElementById('scene1-force-fe');
        const forceFb = document.getElementById('scene1-force-fb');
        const deflectionLine = document.getElementById('scene1-deflection-line');
        const velV0 = document.getElementById('scene1-vel-v0');
        const velVy = document.getElementById('scene1-vel-vy');
        const velV = document.getElementById('scene1-vel-v');
        const angleArc = document.getElementById('scene1-angle-arc');

        const labelB0 = document.getElementById('scene1-label-B0');
        const labelFe = document.getElementById('scene1-label-FE');
        const labelFb = document.getElementById('scene1-label-FB');
        const formula1 = document.getElementById('scene1-formula1');
        const labelDeltaY = document.getElementById('scene1-label-delta-y');
        const labelV0m = document.getElementById('scene1-label-v0-m');
        const labelVy = document.getElementById('scene1-label-vy');
        const labelV = document.getElementById('scene1-label-v');
        const label30deg = document.getElementById('scene1-label-30deg');
        const formula2 = document.getElementById('scene1-formula2');
        const formula3 = document.getElementById('scene1-formula3');
        const labelM = document.getElementById('scene1-label-M');
        const labelM2 = document.getElementById('scene1-label-M2');

        const btnPrev = document.getElementById('scene1-btn-prev');
        const btnNext = document.getElementById('scene1-btn-next');
        const stepIndicator = document.getElementById('scene1-step-indicator');

        function setDisplay(el, val) {
            if (el) el.style.display = val;
        }

        function scene1_render() {
            // 更新指示器
            stepIndicator.textContent = '步骤 ' + (scene1_step + 1) + '/4';
            btnPrev.disabled = (scene1_step === 0);
            btnNext.disabled = (scene1_step === 3);

            // 重置所有动态元素
            setDisplay(trajLine, 'none');
            setDisplay(trajParabola, 'none');
            setDisplay(forceFe, 'none');
            setDisplay(forceFb, 'none');
            setDisplay(deflectionLine, 'none');
            setDisplay(velV0, 'none');
            setDisplay(velVy, 'none');
            setDisplay(velV, 'none');
            setDisplay(angleArc, 'none');
            setDisplay(labelB0, 'none');
            setDisplay(labelFe, 'none');
            setDisplay(labelFb, 'none');
            setDisplay(formula1, 'none');
            setDisplay(labelDeltaY, 'none');
            setDisplay(labelV0m, 'none');
            setDisplay(labelVy, 'none');
            setDisplay(labelV, 'none');
            setDisplay(label30deg, 'none');
            setDisplay(formula2, 'none');
            setDisplay(formula3, 'none');
            setDisplay(labelM2, 'none');

            // 默认显示原始 M 标签
            setDisplay(labelM, 'block');

            // 电场线默认宽度
            efield1.setAttribute('stroke-width', '1');
            efield2.setAttribute('stroke-width', '1');
            efield3.setAttribute('stroke-width', '1');

            if (scene1_step === 0) {
                // 步骤0：装置总览，只显示基础元素
            } else if (scene1_step === 1) {
                // 步骤1：匀速直线运动
                setDisplay(trajLine, 'block');
                setDisplay(labelB0, 'block');
                setDisplay(forceFe, 'block');
                setDisplay(forceFb, 'block');
                setDisplay(labelFe, 'block');
                setDisplay(labelFb, 'block');
                setDisplay(formula1, 'block');
                efield1.setAttribute('stroke-width', '2');
                efield2.setAttribute('stroke-width', '2');
                efield3.setAttribute('stroke-width', '2');
            } else if (scene1_step === 2) {
                // 步骤2：类平抛运动
                setDisplay(trajParabola, 'block');
                setDisplay(deflectionLine, 'block');
                setDisplay(labelDeltaY, 'block');
                // 步骤2/3 使用 M2 标签避免与抛物线重叠
                setDisplay(labelM, 'none');
                setDisplay(labelM2, 'block');
            } else if (scene1_step === 3) {
                // 步骤3：速度分解
                setDisplay(trajParabola, 'block');
                setDisplay(deflectionLine, 'block');
                setDisplay(labelDeltaY, 'block');
                setDisplay(velV0, 'block');
                setDisplay(velVy, 'block');
                setDisplay(velV, 'block');
                setDisplay(angleArc, 'block');
                setDisplay(labelV0m, 'block');
                setDisplay(labelVy, 'block');
                setDisplay(labelV, 'block');
                setDisplay(label30deg, 'block');
                setDisplay(formula2, 'block');
                setDisplay(formula3, 'block');
                // 步骤2/3 使用 M2 标签
                setDisplay(labelM, 'none');
                setDisplay(labelM2, 'block');
            }
        }

        window.scene1_prev = function() {
            if (scene1_step > 0) {
                scene1_step--;
                scene1_render();
            }
        };

        window.scene1_next = function() {
            if (scene1_step < 3) {
                scene1_step++;
                scene1_render();
            }
        };

        // 初始渲染
        scene1_render();
    })();
    </script>
