    <script>
    (function() {
        // Scene 1 实现
        let currentStep = 0; // 0~3

        // 获取元素
        const elLineUpHighlight = document.getElementById('scene1-line-up-highlight');
        const elLineDownHighlight = document.getElementById('scene1-line-down-highlight');
        const elFillUp = document.getElementById('scene1-fill-up');
        const elFillDown = document.getElementById('scene1-fill-down');
        const elLabelUpstage = document.getElementById('scene1-label-upstage');
        const elLabelDownstage = document.getElementById('scene1-label-downstage');
        const elFormulaA1 = document.getElementById('scene1-formula-a1');
        const elFormulaA2 = document.getElementById('scene1-formula-a2');
        const elHintNext = document.getElementById('scene1-hint-next');
        const elBtnPrev = document.getElementById('scene1-btn-prev');
        const elBtnNext = document.getElementById('scene1-btn-next');
        const elStepIndicator = document.getElementById('scene1-step-indicator');

        function renderStep() {
            // 更新指示器
            elStepIndicator.textContent = '步骤 ' + (currentStep + 1) + '/4';

            // 更新按钮状态
            elBtnPrev.disabled = (currentStep === 0);
            elBtnNext.disabled = (currentStep === 3);

            // 步骤0：初始状态，只显示基础元素
            if (currentStep === 0) {
                elLineUpHighlight.setAttribute('opacity', '0');
                elLineDownHighlight.setAttribute('opacity', '0');
                elFillUp.setAttribute('opacity', '0');
                elFillDown.setAttribute('opacity', '0');
                elLabelUpstage.setAttribute('opacity', '0');
                elLabelDownstage.setAttribute('opacity', '0');
                elFormulaA1.setAttribute('opacity', '0');
                elFormulaA2.setAttribute('opacity', '0');
                elHintNext.setAttribute('opacity', '0');
            }
            // 步骤1：上滑阶段高亮
            else if (currentStep === 1) {
                elLineUpHighlight.setAttribute('opacity', '1');
                elLineDownHighlight.setAttribute('opacity', '0');
                elFillUp.setAttribute('opacity', '0.15');
                elFillDown.setAttribute('opacity', '0');
                elLabelUpstage.setAttribute('opacity', '1');
                elLabelDownstage.setAttribute('opacity', '0');
                elFormulaA1.setAttribute('opacity', '1');
                elFormulaA2.setAttribute('opacity', '0');
                elHintNext.setAttribute('opacity', '0');
            }
            // 步骤2：下滑阶段高亮
            else if (currentStep === 2) {
                elLineUpHighlight.setAttribute('opacity', '1');
                elLineDownHighlight.setAttribute('opacity', '1');
                elFillUp.setAttribute('opacity', '0.15');
                elFillDown.setAttribute('opacity', '0.15');
                elLabelUpstage.setAttribute('opacity', '1');
                elLabelDownstage.setAttribute('opacity', '1');
                elFormulaA1.setAttribute('opacity', '1');
                elFormulaA2.setAttribute('opacity', '1');
                elHintNext.setAttribute('opacity', '0');
            }
            // 步骤3：汇总结论
            else if (currentStep === 3) {
                elLineUpHighlight.setAttribute('opacity', '1');
                elLineDownHighlight.setAttribute('opacity', '1');
                elFillUp.setAttribute('opacity', '0.15');
                elFillDown.setAttribute('opacity', '0.15');
                elLabelUpstage.setAttribute('opacity', '1');
                elLabelDownstage.setAttribute('opacity', '1');
                elFormulaA1.setAttribute('opacity', '1');
                elFormulaA2.setAttribute('opacity', '1');
                elHintNext.setAttribute('opacity', '1');
            }
        }

        window.prevStep = function() {
            if (currentStep > 0) {
                currentStep--;
                renderStep();
            }
        };

        window.nextStep = function() {
            if (currentStep < 3) {
                currentStep++;
                renderStep();
            }
        };

        // 初始化
        renderStep();
    })();
    </script>
