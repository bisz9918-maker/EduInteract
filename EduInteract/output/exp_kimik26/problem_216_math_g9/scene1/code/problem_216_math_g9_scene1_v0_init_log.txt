    <script>
    (function() {
        var currentStep = 0;
        var totalSteps = 6;

        var btnPrev = document.getElementById('scene1-btn-prev');
        var btnNext = document.getElementById('scene1-btn-next');
        var stepLabel = document.getElementById('scene1-step-label');

        var stepNames = [
            '步骤0：题目条件',
            '步骤1：验证结论⑤',
            '步骤2：验证结论④',
            '步骤3：验证结论①',
            '步骤4：验证结论③',
            '步骤5：验证结论②',
            '步骤6：总结'
        ];

        var stepSvgElements = {
            1: ['scene1-aux-line', 'scene1-point-b', 'scene1-label-b', 'scene1-point-d', 'scene1-label-d', 'scene1-sym-dist'],
            2: ['scene1-fill-region', 'scene1-solution-set'],
            3: ['scene1-highlight-a', 'scene1-highlight-b', 'scene1-delta'],
            4: ['scene1-arrow-up', 'scene1-label-up', 'scene1-arrow-down', 'scene1-label-down'],
            5: ['scene1-formula1', 'scene1-formula2', 'scene1-formula3'],
            6: []
        };

        var stepPanelElements = {
            1: ['scene1-icon5', 'scene1-hint5'],
            2: ['scene1-icon4', 'scene1-hint4'],
            3: ['scene1-icon1', 'scene1-hint1'],
            4: ['scene1-icon3', 'scene1-hint3'],
            5: ['scene1-icon2', 'scene1-hint2'],
            6: ['scene1-summary']
        };

        var borderColors = {
            1: '#2e7d32',
            2: '#c62828',
            3: '#c62828',
            4: '#2e7d32',
            5: '#2e7d32'
        };

        function setDisplay(id, show) {
            var el = document.getElementById(id);
            if (el) {
                el.style.display = show ? 'block' : 'none';
            }
        }

        function setPanelDisplay(id, show) {
            var el = document.getElementById(id);
            if (el) {
                el.style.display = show ? (el.tagName === 'SPAN' ? 'inline' : 'block') : 'none';
            }
        }

        function setBorderColor(itemId, color) {
            var el = document.getElementById(itemId);
            if (el) {
                el.style.borderLeftColor = color;
            }
        }

        function goToStep(step) {
            currentStep = step;
            stepLabel.textContent = stepNames[step];
            btnPrev.disabled = step === 0;
            btnNext.disabled = step === totalSteps;

            for (var s = 1; s <= totalSteps; s++) {
                var ids = stepSvgElements[s];
                if (ids) {
                    for (var i = 0; i < ids.length; i++) {
                        setDisplay(ids[i], false);
                    }
                }
            }

            for (var s = 1; s <= totalSteps; s++) {
                var ids = stepPanelElements[s];
                if (ids) {
                    for (var i = 0; i < ids.length; i++) {
                        setPanelDisplay(ids[i], false);
                    }
                }
            }

            for (var s = 1; s <= step; s++) {
                var ids = stepSvgElements[s];
                if (ids) {
                    for (var i = 0; i < ids.length; i++) {
                        setDisplay(ids[i], true);
                    }
                }
            }

            for (var s = 1; s <= step; s++) {
                var ids = stepPanelElements[s];
                if (ids) {
                    for (var i = 0; i < ids.length; i++) {
                        setPanelDisplay(ids[i], true);
                    }
                }
            }

            if (step === 6) {
                setBorderColor('scene1-item1', borderColors[1]);
                setBorderColor('scene1-item2', borderColors[2]);
                setBorderColor('scene1-item3', borderColors[3]);
                setBorderColor('scene1-item4', borderColors[4]);
                setBorderColor('scene1-item5', borderColors[5]);
            } else {
                setBorderColor('scene1-item1', '#ccc');
                setBorderColor('scene1-item2', '#ccc');
                setBorderColor('scene1-item3', '#ccc');
                setBorderColor('scene1-item4', '#ccc');
                setBorderColor('scene1-item5', '#ccc');
            }
        }

        btnPrev.addEventListener('click', function() {
            if (currentStep > 0) goToStep(currentStep - 1);
        });

        btnNext.addEventListener('click', function() {
            if (currentStep < totalSteps) goToStep(currentStep + 1);
        });

        goToStep(0);
    })();
    </script>
