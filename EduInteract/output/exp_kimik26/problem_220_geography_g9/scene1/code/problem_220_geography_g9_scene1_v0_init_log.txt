    <script>
    (function() {
        const STATE_COUNT = 5;
        let currentState = 0;
        
        const btnPrev = document.getElementById('scene1-btn-prev');
        const btnNext = document.getElementById('scene1-btn-next');
        const stepIndicator = document.getElementById('scene1-step-indicator');
        const arrow2 = document.getElementById('scene1-arrow-2');
        const arrow4 = document.getElementById('scene1-arrow-4');
        const container = document.getElementById('scene1-container');
        
        function updateButtons() {
            btnPrev.classList.toggle('disabled', currentState === 0);
            btnNext.classList.toggle('disabled', currentState === STATE_COUNT - 1);
            btnPrev.disabled = currentState === 0;
            btnNext.disabled = currentState === STATE_COUNT - 1;
            stepIndicator.textContent = '步骤 ' + (currentState + 1) + '/' + STATE_COUNT;
        }
        
        function render() {
            // 处理带 data-step 的元素
            const elements = container.querySelectorAll('.scene1-step-element');
            elements.forEach(function(el) {
                const step = parseInt(el.getAttribute('data-step'), 10);
                if (currentState >= step) {
                    el.classList.remove('scene1-hidden');
                } else {
                    el.classList.add('scene1-hidden');
                }
            });
            
            // 步骤1：箭头高亮
            if (currentState >= 1) {
                arrow2.setAttribute('stroke', '#c62828');
                arrow2.setAttribute('marker-end', 'url(#scene1-arrowhead-red)');
                arrow4.setAttribute('stroke', '#1565c0');
                arrow4.setAttribute('marker-end', 'url(#scene1-arrowhead-blue)');
            } else {
                arrow2.setAttribute('stroke', '#000000');
                arrow2.setAttribute('marker-end', 'url(#scene1-arrowhead)');
                arrow4.setAttribute('stroke', '#000000');
                arrow4.setAttribute('marker-end', 'url(#scene1-arrowhead)');
            }
            
            updateButtons();
        }
        
        btnPrev.addEventListener('click', function() {
            if (currentState > 0) {
                currentState--;
                render();
            }
        });
        
        btnNext.addEventListener('click', function() {
            if (currentState < STATE_COUNT - 1) {
                currentState++;
                render();
            }
        });
        
        // 初始渲染
        render();
    })();
    </script>
