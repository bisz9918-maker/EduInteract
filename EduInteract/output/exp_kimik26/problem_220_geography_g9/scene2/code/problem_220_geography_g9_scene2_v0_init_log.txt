    <script>
    (function() {
        // Scene 2 实现
        let currentStep = 0;
        const totalSteps = 4;

        const btnPrev = document.getElementById('scene2-btn-prev');
        const btnNext = document.getElementById('scene2-btn-next');
        const indicator = document.getElementById('scene2-step-indicator');

        function renderStep() {
            // 隐藏所有步骤元素
            for (let s = 0; s <= totalSteps; s++) {
                const els = document.querySelectorAll('.scene2-step' + s);
                els.forEach(function(el) {
                    el.classList.remove('scene2-visible');
                    el.classList.add('scene2-hidden');
                });
            }

            // 显示当前步骤元素
            const currentEls = document.querySelectorAll('.scene2-step' + currentStep);
            currentEls.forEach(function(el) {
                el.classList.remove('scene2-hidden');
                el.classList.add('scene2-visible');
            });

            // 更新按钮状态
            btnPrev.disabled = (currentStep === 0);
            btnNext.disabled = (currentStep === totalSteps);
            indicator.textContent = '步骤 ' + currentStep + '/' + totalSteps;
        }

        btnPrev.addEventListener('click', function() {
            if (currentStep > 0) {
                currentStep--;
                renderStep();
            }
        });

        btnNext.addEventListener('click', function() {
            if (currentStep < totalSteps) {
                currentStep++;
                renderStep();
            }
        });

        // 初始渲染
        renderStep();
    })();
    </script>
