    <script>
    (function() {
        // Scene 3 实现
        var currentStep = 0;
        var totalSteps = 6;

        // 元素引用
        var btnPrev = document.getElementById('scene3-btn-prev');
        var btnNext = document.getElementById('scene3-btn-next');
        var stepText = document.getElementById('scene3-step-text');

        // 各步骤需要显示的元素ID列表
        var stepElements = {
            0: [], // 步骤0：纯底图，无额外元素
            1: [ // 步骤1：赤道+南北半球
                'scene3-equator-line', 'scene3-equator-label',
                'scene3-north-fill', 'scene3-south-fill',
                'scene3-legend-north', 'scene3-legend-south'
            ],
            2: [ // 步骤2：东西半球（保留步骤1元素，新增步骤2元素）
                'scene3-equator-line', 'scene3-equator-label',
                'scene3-north-fill', 'scene3-south-fill',
                'scene3-legend-north', 'scene3-legend-south',
                'scene3-line-160e', 'scene3-line-20w',
                'scene3-label-20w-red',
                'scene3-east-fill', 'scene3-west-fill',
                'scene3-legend-east', 'scene3-legend-west'
            ],
            3: [ // 步骤3：纬度带（保留步骤1-2元素，新增步骤3元素）
                'scene3-equator-line', 'scene3-equator-label',
                'scene3-north-fill', 'scene3-south-fill',
                'scene3-legend-north', 'scene3-legend-south',
                'scene3-line-160e', 'scene3-line-20w',
                'scene3-label-20w-red',
                'scene3-east-fill', 'scene3-west-fill',
                'scene3-legend-east', 'scene3-legend-west',
                'scene3-line-30n', 'scene3-line-30s', 'scene3-line-60n-purple', 'scene3-line-60s',
                'scene3-label-60n-purple', 'scene3-label-30n-purple', 'scene3-label-30s-purple', 'scene3-label-60s-purple',
                'scene3-lat-fill-low1', 'scene3-lat-fill-low2', 'scene3-lat-fill-mid1', 'scene3-lat-fill-mid2',
                'scene3-lat-fill-high1', 'scene3-lat-fill-high2',
                'scene3-legend-low', 'scene3-legend-mid', 'scene3-legend-high'
            ],
            4: [ // 步骤4：隐藏填充，显示表格，保留分界线
                'scene3-equator-line', 'scene3-equator-label',
                'scene3-line-160e', 'scene3-line-20w',
                'scene3-label-20w-red',
                'scene3-line-30n', 'scene3-line-30s', 'scene3-line-60n-purple', 'scene3-line-60s',
                'scene3-label-60n-purple', 'scene3-label-30n-purple', 'scene3-label-30s-purple', 'scene3-label-60s-purple',
                'scene3-summary-table'
            ],
            5: [ // 步骤5：表格可点击，保留分界线
                'scene3-equator-line', 'scene3-equator-label',
                'scene3-line-160e', 'scene3-line-20w',
                'scene3-label-20w-red',
                'scene3-line-30n', 'scene3-line-30s', 'scene3-line-60n-purple', 'scene3-line-60s',
                'scene3-label-60n-purple', 'scene3-label-30n-purple', 'scene3-label-30s-purple', 'scene3-label-60s-purple',
                'scene3-summary-table'
            ]
        };

        // 所有可切换的元素ID（去重）
        var allToggleIds = [];
        for (var s in stepElements) {
            stepElements[s].forEach(function(id) {
                if (allToggleIds.indexOf(id) === -1) {
                    allToggleIds.push(id);
                }
            });
        }

        // 获取元素
        function getEl(id) {
            return document.getElementById(id);
        }

        // 显示/隐藏元素
        function showEl(id) {
            var el = getEl(id);
            if (el) {
                el.style.display = '';
                el.style.opacity = '1';
            }
        }
        function hideEl(id) {
            var el = getEl(id);
            if (el) {
                el.style.opacity = '0';
                setTimeout(function() {
                    if (el.style.opacity === '0') {
                        el.style.display = 'none';
                    }
                }, 400);
            }
        }

        // 渲染当前步骤
        function renderStep() {
            var showIds = stepElements[currentStep] || [];

            // 先隐藏所有可切换元素
            allToggleIds.forEach(function(id) {
                if (showIds.indexOf(id) === -1) {
                    hideEl(id);
                }
            });

            // 再显示当前步骤需要的元素
            showIds.forEach(function(id) {
                showEl(id);
            });

            // 更新按钮状态
            btnPrev.disabled = currentStep === 0;
            btnNext.disabled = currentStep === totalSteps - 1;
            stepText.textContent = '步骤 ' + (currentStep + 1) + '/' + totalSteps;

            // 步骤5：启用表格行点击
            var tableRows = ['scene3-table-row-a', 'scene3-table-row-b', 'scene3-table-row-c', 'scene3-table-row-d'];
            tableRows.forEach(function(rowId) {
                var row = getEl(rowId);
                if (row) {
                    if (currentStep === 5) {
                        row.style.cursor = 'pointer';
                    } else {
                        row.style.cursor = 'default';
                        row.classList.remove('active');
                    }
                }
            });

            // 清除点高亮
            ['scene3-point-a', 'scene3-point-b', 'scene3-point-c', 'scene3-point-d'].forEach(function(pid) {
                var p = getEl(pid);
                if (p) p.classList.remove('scene3-point-highlight');
            });
        }

        // 按钮事件
        btnPrev.addEventListener('click', function() {
            if (currentStep > 0) {
                currentStep--;
                renderStep();
            }
        });
        btnNext.addEventListener('click', function() {
            if (currentStep < totalSteps - 1) {
                currentStep++;
                renderStep();
            }
        });

        // 表格行点击事件
        function setupTableRow(rowId, pointId) {
            var row = getEl(rowId);
            if (row) {
                row.addEventListener('click', function() {
                    if (currentStep !== 5) return;

                    // 清除所有行激活状态
                    ['scene3-table-row-a', 'scene3-table-row-b', 'scene3-table-row-c', 'scene3-table-row-d'].forEach(function(rid) {
                        var r = getEl(rid);
                        if (r) r.classList.remove('active');
                    });
                    // 清除所有点高亮
                    ['scene3-point-a', 'scene3-point-b', 'scene3-point-c', 'scene3-point-d'].forEach(function(pid) {
                        var p = getEl(pid);
                        if (p) p.classList.remove('scene3-point-highlight');
                    });

                    // 激活当前行和对应点
                    row.classList.add('active');
                    var point = getEl(pointId);
                    if (point) point.classList.add('scene3-point-highlight');
                });
            }
        }

        setupTableRow('scene3-table-row-a', 'scene3-point-a');
        setupTableRow('scene3-table-row-b', 'scene3-point-b');
        setupTableRow('scene3-table-row-c', 'scene3-point-c');
        setupTableRow('scene3-table-row-d', 'scene3-point-d');

        // 初始化：先隐藏所有可切换元素
        allToggleIds.forEach(function(id) {
            var el = getEl(id);
            if (el) {
                el.style.display = 'none';
                el.style.opacity = '0';
            }
        });

        // 渲染初始状态
        renderStep();
    })();
    </script>
