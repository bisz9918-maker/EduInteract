    <script>
    (function() {
        // Scene 2 实现：夏至/冬至状态切换
        var currentState = 'summer';

        var btnSummer = document.getElementById('scene2-btn-summer');
        var btnWinter = document.getElementById('scene2-btn-winter');

        // 状态元素映射
        var summerElements = [
            'scene2-sun-summer', 'scene2-rays-summer', 'scene2-angle-90',
            'scene2-label-90', 'scene2-label-sun-summer'
        ];
        var winterElements = [
            'scene2-sun-winter', 'scene2-rays-winter', 'scene2-angle-43',
            'scene2-label-43', 'scene2-label-sun-winter', 'scene2-point-s',
            'scene2-label-s', 'scene2-label-lat-s'
        ];

        function show(id) {
            var el = document.getElementById(id);
            if (el) { el.style.display = 'block'; el.style.opacity = '1'; }
        }
        function hide(id) {
            var el = document.getElementById(id);
            if (el) { el.style.display = 'none'; el.style.opacity = '0'; }
        }

        function setState(state) {
            currentState = state;
            if (state === 'summer') {
                summerElements.forEach(show);
                winterElements.forEach(hide);
                btnSummer.classList.add('active-summer');
                btnWinter.classList.remove('active-winter');
                document.getElementById('scene2-formula').textContent = 'H = 90° - |23.5° - 23.5°| = 90°';
            } else {
                summerElements.forEach(hide);
                winterElements.forEach(show);
                btnSummer.classList.remove('active-summer');
                btnWinter.classList.add('active-winter');
                document.getElementById('scene2-formula').textContent = 'H = 90° - |23.5° - (-23.5°)| = 43°';
            }
        }

        btnSummer.addEventListener('click', function() { setState('summer'); });
        btnWinter.addEventListener('click', function() { setState('winter'); });

        // 初始状态
        setState('summer');
    })();
    </script>
