    <script>
    (function() {
        // Scene 2 实现
        var currentStep = 0;
        var totalSteps = 4;
        var stepNames = ["晶胞初识", "均摊法计算", "配位数分析", "错误辨析"];

        var btnPrev = document.getElementById("scene2-btn-prev");
        var btnNext = document.getElementById("scene2-btn-next");
        var indicator = document.getElementById("scene2-step-indicator");
        var stepName = document.getElementById("scene2-step-name");

        // 各步骤对应的元素组ID
        var stepGroups = [
            ["scene2-panel-step0"],
            ["scene2-labels-step1", "scene2-panel-step1"],
            ["scene2-stack-group", "scene2-coord-group", "scene2-panel-step2"],
            ["scene2-stack-group", "scene2-coord-group", "scene2-panel-step3"]
        ];

        function setDisplay(id, show) {
            var el = document.getElementById(id);
            if (el) {
                el.setAttribute("display", show ? "inline" : "none");
            }
        }

        function renderStep(step) {
            // 更新控件
            indicator.textContent = (step + 1) + " / " + totalSteps;
            stepName.textContent = stepNames[step];
            btnPrev.disabled = (step === 0);
            if (step === totalSteps - 1) {
                btnNext.textContent = "已完成";
                btnNext.disabled = true;
            } else {
                btnNext.textContent = "下一步 ▶";
                btnNext.disabled = false;
            }

            // 隐藏所有步骤专属组
            for (var s = 0; s < totalSteps; s++) {
                stepGroups[s].forEach(function(id) {
                    setDisplay(id, false);
                });
            }

            // 显示当前步骤的组
            stepGroups[step].forEach(function(id) {
                setDisplay(id, true);
            });
        }

        btnPrev.addEventListener("click", function() {
            if (currentStep > 0) {
                currentStep--;
                renderStep(currentStep);
            }
        });

        btnNext.addEventListener("click", function() {
            if (currentStep < totalSteps - 1) {
                currentStep++;
                renderStep(currentStep);
            }
        });

        // 初始渲染
        renderStep(currentStep);
    })();
    </script>
