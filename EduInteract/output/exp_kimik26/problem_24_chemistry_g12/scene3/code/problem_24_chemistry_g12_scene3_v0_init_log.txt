    <script>
    (function() {
        // Scene 3 实现
        // 状态机
        var state = { step: 0, maxStep: 6 };
        var isAnimating = false;

        // 元素引用
        var elGas甲Red = document.getElementById('scene3-gas-甲-red');
        var elGas甲Blue = document.getElementById('scene3-gas-甲-blue');
        var elLabel甲N = document.getElementById('scene3-label-甲-n');

        var elPiston乙 = document.getElementById('scene3-piston-乙');
        var elGas乙Red = document.getElementById('scene3-gas-乙-red');
        var elGas乙Blue = document.getElementById('scene3-gas-乙-blue');
        var elLabel乙Vp = document.getElementById('scene3-label-乙-Vp');
        var elArrow乙 = document.getElementById('scene3-arrow-乙');
        var elLabel乙N = document.getElementById('scene3-label-乙-n');

        var elCompareRow1 = document.getElementById('scene3-compare-row1');
        var elCompareRow2 = document.getElementById('scene3-compare-row2');
        var elCompareRow3 = document.getElementById('scene3-compare-row3');
        var elOptions = document.getElementById('scene3-options');

        var btnPrev = document.getElementById('scene3-btn-prev');
        var btnNext = document.getElementById('scene3-btn-next');
        var elIndicator = document.getElementById('scene3-step-indicator');

        // 工具函数
        function setOpacity(el, val) {
            if (!el) return;
            el.style.opacity = val;
            el.style.transition = 'opacity 300ms ease';
        }

        // 活塞动画：从 y=46 到 y=116，气体同步压缩
        function animatePiston(duration, callback) {
            var startY = 46;
            var endY = 116;
            var startGasY = 60;
            var endGasY = 130;
            var startGasH = 178;
            var endGasH = 108;
            var startTime = null;

            function easeOut(t) {
                return t * (2 - t);
            }

            function step(timestamp) {
                if (!startTime) startTime = timestamp;
                var progress = Math.min((timestamp - startTime) / duration, 1);
                var eased = easeOut(progress);

                var curPistonY = startY + (endY - startY) * eased;
                var curGasY = startGasY + (endGasY - startGasY) * eased;
                var curGasH = startGasH + (endGasH - startGasH) * eased;

                elPiston乙.setAttribute('y', curPistonY);
                elGas乙Red.setAttribute('y', curGasY);
                elGas乙Red.setAttribute('height', curGasH);
                elGas乙Blue.setAttribute('y', curGasY);
                elGas乙Blue.setAttribute('height', curGasH);

                if (progress < 1) {
                    requestAnimationFrame(step);
                } else {
                    isAnimating = false;
                    if (callback) callback();
                }
            }

            isAnimating = true;
            requestAnimationFrame(step);
        }

        // 反向活塞动画（回退）
        function animatePistonReverse(duration, callback) {
            var startY = 116;
            var endY = 46;
            var startGasY = 130;
            var endGasY = 60;
            var startGasH = 108;
            var endGasH = 178;
            var startTime = null;

            function easeOut(t) {
                return t * (2 - t);
            }

            function step(timestamp) {
                if (!startTime) startTime = timestamp;
                var progress = Math.min((timestamp - startTime) / duration, 1);
                var eased = easeOut(progress);

                var curPistonY = startY + (endY - startY) * eased;
                var curGasY = startGasY + (endGasY - startGasY) * eased;
                var curGasH = startGasH + (endGasH - startGasH) * eased;

                elPiston乙.setAttribute('y', curPistonY);
                elGas乙Red.setAttribute('y', curGasY);
                elGas乙Red.setAttribute('height', curGasH);
                elGas乙Blue.setAttribute('y', curGasY);
                elGas乙Blue.setAttribute('height', curGasH);

                if (progress < 1) {
                    requestAnimationFrame(step);
                } else {
                    isAnimating = false;
                    if (callback) callback();
                }
            }

            isAnimating = true;
            requestAnimationFrame(step);
        }

        // 渲染函数
        function render() {
            // 更新指示器
            elIndicator.textContent = '步骤 ' + state.step + ' / ' + state.maxStep;

            // 按钮状态
            btnPrev.disabled = (state.step === 0) || isAnimating;
            btnNext.disabled = (state.step === state.maxStep) || isAnimating;

            // 步骤0：初始状态
            if (state.step === 0) {
                setOpacity(elGas甲Red, 1);
                setOpacity(elGas甲Blue, 0);
                setOpacity(elLabel甲N, 0);

                // 乙恢复初始
                elPiston乙.setAttribute('y', '46');
                elGas乙Red.setAttribute('y', '60');
                elGas乙Red.setAttribute('height', '178');
                elGas乙Blue.setAttribute('y', '60');
                elGas乙Blue.setAttribute('height', '178');
                setOpacity(elGas乙Red, 1);
                setOpacity(elGas乙Blue, 0);
                setOpacity(elLabel乙Vp, 0);
                setOpacity(elArrow乙, 0);
                setOpacity(elLabel乙N, 0);

                setOpacity(elCompareRow1, 0);
                setOpacity(elCompareRow2, 0);
                setOpacity(elCompareRow3, 0);
                setOpacity(elOptions, 0);
            }

            // 步骤1：甲平衡
            if (state.step >= 1) {
                setOpacity(elGas甲Red, 0.7);
                setOpacity(elGas甲Blue, 1);
                setOpacity(elLabel甲N, 1);
            } else {
                setOpacity(elGas甲Red, 1);
                setOpacity(elGas甲Blue, 0);
                setOpacity(elLabel甲N, 0);
            }

            // 步骤2：乙平衡
            if (state.step >= 2) {
                setOpacity(elGas乙Red, 0.6);
                setOpacity(elGas乙Blue, 1);
                setOpacity(elLabel乙Vp, 1);
                setOpacity(elArrow乙, 1);
                setOpacity(elLabel乙N, 1);
            } else if (state.step < 2) {
                if (!isAnimating) {
                    elPiston乙.setAttribute('y', '46');
                    elGas乙Red.setAttribute('y', '60');
                    elGas乙Red.setAttribute('height', '178');
                    elGas乙Blue.setAttribute('y', '60');
                    elGas乙Blue.setAttribute('height', '178');
                }
                setOpacity(elGas乙Red, 1);
                setOpacity(elGas乙Blue, 0);
                setOpacity(elLabel乙Vp, 0);
                setOpacity(elArrow乙, 0);
                setOpacity(elLabel乙N, 0);
            }

            // 步骤3-6
            if (state.step >= 3) {
                setOpacity(elCompareRow1, 1);
            } else {
                setOpacity(elCompareRow1, 0);
            }

            if (state.step >= 4) {
                setOpacity(elCompareRow2, 1);
            } else {
                setOpacity(elCompareRow2, 0);
            }

            if (state.step >= 5) {
                setOpacity(elCompareRow3, 1);
            } else {
                setOpacity(elCompareRow3, 0);
            }

            if (state.step >= 6) {
                setOpacity(elOptions, 1);
            } else {
                setOpacity(elOptions, 0);
            }
        }

        // 上一步
        function prevStep() {
            if (state.step > 0 && !isAnimating) {
                state.step--;
                if (state.step === 1) {
                    animatePistonReverse(500, function() {
                        render();
                    });
                } else {
                    render();
                }
            }
        }

        // 下一步
        function nextStep() {
            if (state.step < state.maxStep && !isAnimating) {
                state.step++;
                if (state.step === 2) {
                    render();
                    animatePiston(500, function() {
                        // 动画完成
                    });
                } else {
                    render();
                }
            }
        }

        // 事件绑定
        btnPrev.addEventListener('click', prevStep);
        btnNext.addEventListener('click', nextStep);

        // 初始渲染
        render();
    })();
    </script>
