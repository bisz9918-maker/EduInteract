    <script>
    (function() {
        // Scene 2 实现
        let currentStep = 0;
        const totalSteps = 4;
        const titles = [
            "步骤 1 / 4：基因在染色体上的位置",
            "步骤 2 / 4：题干情形（杂合无毛雌 × 无毛雄）",
            "步骤 3 / 4：第2问情形（有毛雌 × 无毛雄）",
            "步骤 4 / 4：对比总结"
        ];

        const titleEl = document.getElementById('scene2-title');
        const indicatorEl = document.getElementById('scene2-step-indicator');
        const prevBtn = document.getElementById('scene2-btn-prev');
        const nextBtn = document.getElementById('scene2-btn-next');

        function renderStep() {
            // 隐藏所有步骤组
            for (let i = 0; i < totalSteps; i++) {
                const group = document.getElementById('scene2-step' + i + '-group');
                if (group) {
                    group.style.display = 'none';
                    group.classList.remove('active');
                }
            }
            // 显示当前步骤组
            const activeGroup = document.getElementById('scene2-step' + currentStep + '-group');
            if (activeGroup) {
                activeGroup.style.display = 'block';
                activeGroup.classList.add('active');
            }
            // 更新标题
            if (titleEl) titleEl.textContent = titles[currentStep];
            // 更新指示器
            if (indicatorEl) indicatorEl.textContent = '步骤 ' + (currentStep + 1) + ' / ' + totalSteps;
            // 更新按钮状态
            if (prevBtn) prevBtn.disabled = (currentStep === 0);
            if (nextBtn) nextBtn.disabled = (currentStep === totalSteps - 1);
        }

        prevBtn.addEventListener('click', function() {
            if (currentStep > 0) {
                currentStep--;
                renderStep();
            }
        });

        nextBtn.addEventListener('click', function() {
            if (currentStep < totalSteps - 1) {
                currentStep++;
                renderStep();
            }
        });

        // 初始渲染
        renderStep();
    })();
    </script>
