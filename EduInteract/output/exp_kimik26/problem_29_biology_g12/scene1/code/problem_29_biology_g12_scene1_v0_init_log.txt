<script>
(function(){
var currentStep=0;
var totalSteps=5;
var stepGroups=[];
var arrowGroups=[];
for(var i=0;i<totalSteps;i++){stepGroups.push(document.getElementById('scene1-step'+i+'-group'));}
for(var i=0;i<4;i++){arrowGroups.push(document.getElementById('scene1-arrow-'+i+'-'+(i+1)));}
var btnPrev=document.getElementById('scene1-btn-prev');
var btnNext=document.getElementById('scene1-btn-next');
var indicator=document.getElementById('scene1-step-indicator');

function setGroupVisible(group,visible){
if(!group)return;
var children=group.getElementsByTagName('*');
for(var i=0;i<children.length;i++){
children[i].style.display=visible?'':'none';
children[i].style.visibility=visible?'visible':'hidden';
}
group.style.display=visible?'block':'none';
group.style.visibility=visible?'visible':'hidden';
group.style.opacity=visible?'1':'0';
}

function renderStep(){
for(var i=0;i<totalSteps;i++){
if(stepGroups[i]){setGroupVisible(stepGroups[i],i===currentStep);}
}
for(var i=0;i<arrowGroups.length;i++){
if(arrowGroups[i]){arrowGroups[i].style.display='block';arrowGroups[i].style.opacity='1';}
}
btnPrev.disabled=(currentStep===0);
btnNext.disabled=(currentStep===totalSteps-1);
indicator.textContent='步骤 '+(currentStep+1)+'/'+totalSteps;
}

btnPrev.addEventListener('click',function(){if(currentStep>0){currentStep--;renderStep();}});
btnNext.addEventListener('click',function(){if(currentStep<totalSteps-1){currentStep++;renderStep();}});
renderStep();
})();
</script>
