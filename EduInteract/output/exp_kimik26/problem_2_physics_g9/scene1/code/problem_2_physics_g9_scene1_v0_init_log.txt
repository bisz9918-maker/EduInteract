    <script>
    (function() {
        // Scene 1 实现
        var currentStep = 0;
        var isAnimating = false;

        // 元素引用
        var carGroup = document.getElementById('scene1-car-group');
        var heightLine = document.getElementById('scene1-height-line');
        var heightTop = document.getElementById('scene1-height-top');
        var heightBottom = document.getElementById('scene1-height-bottom');
        var vArrowSlope = document.getElementById('scene1-v-arrow-slope');
        var vArrowHorizontal = document.getElementById('scene1-v-arrow-horizontal');
        var energyBg = document.getElementById('scene1-energy-bg');
        var epBar = document.getElementById('scene1-ep-bar');
        var ekBar = document.getElementById('scene1-ek-bar');
        var emBar = document.getElementById('scene1-em-bar');
        var transformArrow = document.getElementById('scene1-transform-arrow');
        var frictionArrow = document.getElementById('scene1-friction-arrow');
        var heatWave = document.getElementById('scene1-heat-wave');
        var labelH = document.getElementById('scene1-label-h');
        var labelVSlope = document.getElementById('scene1-label-v-slope');
        var labelVHorizontal = document.getElementById('scene1-label-v-horizontal');
        var labelEp = document.getElementById('scene1-label-ep');
        var labelEk = document.getElementById('scene1-label-ek');
        var labelEm = document.getElementById('scene1-label-em');
        var labelF = document.getElementById('scene1-label-f');
        var labelQ = document.getElementById('scene1-label-q');
        var conclusionText = document.getElementById('scene1-conclusion-text');
        var btnPrev = document.getElementById('scene1-btn-prev');
        var btnNext = document.getElementById('scene1-btn-next');
        var stepIndicator = document.getElementById('scene1-step-indicator');

        // 小车位置
        var carPositions = [
            { x: 150, y: 112 },  // 步骤0/1: 顶端
            { x: 150, y: 112 },  // 步骤1: 顶端
            { x: 225, y: 200 },  // 步骤2: 中点
            { x: 300, y: 287 },  // 步骤3: 底端
            { x: 300, y: 287 }   // 步骤4: 底端
        ];

        // 能量条目标值 (高度百分比, 0-100)
        var energyTargets = [
            { ep: 0, ek: 0, em: 0 },    // 步骤0
            { ep: 100, ek: 0, em: 100 }, // 步骤1
            { ep: 50, ek: 40, em: 95 },  // 步骤2
            { ep: 0, ek: 80, em: 90 },   // 步骤3
            { ep: 0, ek: 80, em: 90 }    // 步骤4
        ];

        // 当前能量条值
        var currentEnergy = { ep: 0, ek: 0, em: 0 };

        function setCarPosition(x, y) {
            carGroup.setAttribute('transform', 'translate(' + x + ',' + y + ')');
        }

        function show(el) {
            if (el) el.style.display = '';
        }
        function hide(el) {
            if (el) el.style.display = 'none';
        }

        function setBarHeight(bar, percent) {
            var maxHeight = 100;
            var height = maxHeight * percent / 100;
            var y = 240 - height;
            bar.setAttribute('y', y);
            bar.setAttribute('height', height);
        }

        function animateBars(target, duration, callback) {
            var start = { ep: currentEnergy.ep, ek: currentEnergy.ek, em: currentEnergy.em };
            var startTime = performance.now();

            function frame(now) {
                var t = Math.min((now - startTime) / duration, 1);
                currentEnergy.ep = start.ep + (target.ep - start.ep) * t;
                currentEnergy.ek = start.ek + (target.ek - start.ek) * t;
                currentEnergy.em = start.em + (target.em - start.em) * t;
                setBarHeight(epBar, currentEnergy.ep);
                setBarHeight(ekBar, currentEnergy.ek);
                setBarHeight(emBar, currentEnergy.em);
                if (t < 1) {
                    requestAnimationFrame(frame);
                } else {
                    if (callback) callback();
                }
            }
            requestAnimationFrame(frame);
        }

        function animateCar(fromX, fromY, toX, toY, duration, callback) {
            var startTime = performance.now();
            function frame(now) {
                var t = Math.min((now - startTime) / duration, 1);
                // ease-in-out
                t = t < 0.5 ? 2 * t * t : -1 + (4 - 2 * t) * t;
                var x = fromX + (toX - fromX) * t;
                var y = fromY + (toY - fromY) * t;
                setCarPosition(x, y);
                if (t < 1) {
                    requestAnimationFrame(frame);
                } else {
                    if (callback) callback();
                }
            }
            requestAnimationFrame(frame);
        }

        function updateUI() {
            stepIndicator.textContent = '步骤 ' + currentStep + '/4';
            btnPrev.disabled = currentStep === 0;
            btnNext.disabled = currentStep === 4;
        }

        function goToStep(step) {
            if (isAnimating || step === currentStep) return;
            isAnimating = true;
            var oldStep = currentStep;
            currentStep = step;
            updateUI();

            // 先隐藏所有步骤专属元素
            hide(heightLine); hide(heightTop); hide(heightBottom);
            hide(vArrowSlope); hide(vArrowHorizontal);
            hide(energyBg); hide(epBar); hide(ekBar); hide(emBar);
            hide(transformArrow); hide(frictionArrow); hide(heatWave);
            hide(labelH); hide(labelVSlope); hide(labelVHorizontal);
            hide(labelEp); hide(labelEk); hide(labelEm);
            hide(labelF); hide(labelQ);
            hide(conclusionText);

            // 根据步骤显示元素
            if (currentStep >= 1) {
                show(heightLine); show(heightTop); show(heightBottom);
                show(energyBg); show(epBar); show(ekBar); show(emBar);
                show(labelH); show(labelEp); show(labelEk); show(labelEm);
            }
            if (currentStep === 2) {
                show(vArrowSlope); show(labelVSlope);
            }
            if (currentStep >= 3) {
                show(vArrowHorizontal); show(labelVHorizontal);
            }
            if (currentStep === 3) {
                show(transformArrow);
                conclusionText.textContent = 'A 错误：重力势能转化为动能';
                show(conclusionText);
            }
            if (currentStep === 4) {
                show(frictionArrow); show(heatWave);
                show(labelF); show(labelQ);
                conclusionText.textContent = 'B 错误：机械能因摩擦而减小';
                show(conclusionText);
            }

            // 小车位置动画
            var fromPos = carPositions[oldStep];
            var toPos = carPositions[currentStep];
            var targetEnergy = energyTargets[currentStep];

            var animCount = 0;
            function checkDone() {
                animCount++;
                if (animCount >= 2) {
                    isAnimating = false;
                }
            }

            animateCar(fromPos.x, fromPos.y, toPos.x, toPos.y, 600, checkDone);
            animateBars(targetEnergy, 600, checkDone);
        }

        function nextStep() {
            if (currentStep < 4) goToStep(currentStep + 1);
        }
        function prevStep() {
            if (currentStep > 0) goToStep(currentStep - 1);
        }

        btnNext.addEventListener('click', nextStep);
        btnPrev.addEventListener('click', prevStep);

        // 初始化
        setCarPosition(carPositions[0].x, carPositions[0].y);
        updateUI();
        isAnimating = false;
    })();
    </script>
