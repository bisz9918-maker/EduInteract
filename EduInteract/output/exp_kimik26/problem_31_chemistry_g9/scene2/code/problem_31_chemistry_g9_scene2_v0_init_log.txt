    <script>
    (function() {
        // Scene 2 状态机
        let currentStep = 0; // 0~3

        // 获取元素
        const bccCell = document.getElementById('scene2-bcc-cell');
        const fccCell = document.getElementById('scene2-fcc-cell');
        const bccDiagonal = document.getElementById('scene2-bcc-body-diagonal');
        const fccDiagonal = document.getElementById('scene2-fcc-face-diagonal');
        const hintText = document.getElementById('scene2-hint-text');
        const bccSection = document.getElementById('scene2-bcc-section');
        const fccSection = document.getElementById('scene2-fcc-section');
        const step3Formula = document.getElementById('scene2-step3-formula');
        const btnPrev = document.getElementById('scene2-btn-prev');
        const btnNext = document.getElementById('scene2-btn-next');
        const stepLabel = document.getElementById('scene2-step-label');
        const bccDiagonalLine = document.getElementById('scene2-bcc-diagonal-line');
        const fccDiagonalLine = document.getElementById('scene2-fcc-diagonal-line');

        const indicators = [
            document.getElementById('scene2-indicator-0'),
            document.getElementById('scene2-indicator-1'),
            document.getElementById('scene2-indicator-2'),
            document.getElementById('scene2-indicator-3')
        ];

        function show(el) {
            if (!el) return;
            el.classList.remove('scene2-hidden');
        }
        function hide(el) {
            if (!el) return;
            el.classList.add('scene2-hidden');
        }
        function dim(el) {
            if (!el) return;
            el.classList.add('scene2-dimmed');
        }
        function undim(el) {
            if (!el) return;
            el.classList.remove('scene2-dimmed');
        }

        function updateIndicators() {
            indicators.forEach((ind, idx) => {
                ind.setAttribute('fill', idx === currentStep ? '#1565c0' : '#cccccc');
            });
        }

        function updateButtons() {
            btnPrev.disabled = currentStep === 0;
            btnNext.disabled = currentStep === 3;
            stepLabel.textContent = '步骤 ' + (currentStep + 1) + '/4';
        }

        function triggerDrawAnimation(lineEl) {
            if (!lineEl) return;
            lineEl.classList.remove('active');
            // 强制重排
            void lineEl.offsetWidth;
            lineEl.classList.add('active');
        }

        function renderStep() {
            // 先重置所有状态
            undim(bccCell);
            undim(fccCell);
            hide(bccDiagonal);
            hide(fccDiagonal);
            hide(bccSection);
            hide(fccSection);
            hide(step3Formula);
            show(hintText);

            // 清除线条动画
            if (bccDiagonalLine) bccDiagonalLine.classList.remove('active');
            if (fccDiagonalLine) fccDiagonalLine.classList.remove('active');

            switch (currentStep) {
                case 0:
                    // 初始状态：两个晶胞正常显示，提示文字可见
                    break;
                case 1:
                    // BCC体对角线分析
                    hide(hintText);
                    show(bccDiagonal);
                    dim(fccCell);
                    show(bccSection);
                    // 延迟触发动画
                    setTimeout(() => triggerDrawAnimation(bccDiagonalLine), 50);
                    break;
                case 2:
                    // FCC面对角线分析
                    hide(hintText);
                    show(fccDiagonal);
                    dim(bccCell);
                    show(fccSection);
                    setTimeout(() => triggerDrawAnimation(fccDiagonalLine), 50);
                    break;
                case 3:
                    // 边长比推导
                    hide(hintText);
                    show(step3Formula);
                    break;
            }

            updateIndicators();
            updateButtons();
        }

        btnPrev.addEventListener('click', function() {
            if (currentStep > 0) {
                currentStep--;
                renderStep();
            }
        });

        btnNext.addEventListener('click', function() {
            if (currentStep < 3) {
                currentStep++;
                renderStep();
            }
        });

        // 初始化
        renderStep();
    })();
    </script>
