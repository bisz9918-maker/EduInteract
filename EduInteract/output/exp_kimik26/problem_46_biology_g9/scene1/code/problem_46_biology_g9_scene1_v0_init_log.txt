<script>
(function(){
'use strict';
let currentStep=0;
const totalSteps=6;
const btnPrev=document.getElementById('scene1-btn-prev');
const btnNext=document.getElementById('scene1-btn-next');
const indicator=document.getElementById('scene1-step-indicator');
const panelLeft=document.getElementById('scene1-panel-left');
const panelRight=document.getElementById('scene1-panel-right');
const step1Phospho=document.getElementById('scene1-step1-phospho-highlight');
const step2Labels=document.getElementById('scene1-step2-labels');
const step2Hbonds=document.getElementById('scene1-step2-hbonds-highlight');
const step3Group=document.getElementById('scene1-step3-group');
const step4Group=document.getElementById('scene1-step4-group');
const step5Group=document.getElementById('scene1-step5-group');
const step5DnaHighlight=document.getElementById('scene1-step5-dna-highlight');
const barC=document.getElementById('scene1-bar-c');
const barG=document.getElementById('scene1-bar-g');
const barA=document.getElementById('scene1-bar-a');
const barT=document.getElementById('scene1-bar-t');
const labelC=document.getElementById('scene1-label-c');
const labelG=document.getElementById('scene1-label-g');
const labelA=document.getElementById('scene1-label-a');
const labelT=document.getElementById('scene1-label-t');

function show(el){ if(el){ el.style.display='block'; el.style.opacity='1'; } }
function hide(el){ if(el){ el.style.display='none'; } }
function setOpacity(el,val){ if(el){ el.style.opacity=String(val); } }

function animateBars(){
const duration=300;
const start=performance.now();
const targets={c:72,g:72,a:48,t:48};
function frame(now){
const elapsed=now-start;
const t=Math.min(elapsed/duration,1);
const ease=t*(2-t);
barC.setAttribute('width',String(targets.c*ease));
barG.setAttribute('width',String(targets.g*ease));
barA.setAttribute('width',String(targets.a*ease));
barT.setAttribute('width',String(targets.t*ease));
if(t>=1){
setOpacity(labelC,1); setOpacity(labelG,1); setOpacity(labelA,1); setOpacity(labelT,1);
return;
}
requestAnimationFrame(frame);
}
requestAnimationFrame(frame);
}

function resetBars(){
barC.setAttribute('width','0'); barG.setAttribute('width','0'); barA.setAttribute('width','0'); barT.setAttribute('width','0');
setOpacity(labelC,0); setOpacity(labelG,0); setOpacity(labelA,0); setOpacity(labelT,0);
}

function renderStep(){
indicator.textContent='步骤 '+(currentStep+1)+' / '+totalSteps;
btnPrev.disabled=currentStep===0;
btnNext.disabled=currentStep===totalSteps-1;
hide(step1Phospho); hide(panelLeft); hide(step2Labels); hide(step2Hbonds);
hide(panelRight); hide(step3Group); hide(step4Group); hide(step5Group); hide(step5DnaHighlight);
resetBars();
switch(currentStep){
case 0: break;
case 1: show(step1Phospho); show(panelLeft); break;
case 2: show(step1Phospho); show(panelLeft); show(step2Labels); show(step2Hbonds); break;
