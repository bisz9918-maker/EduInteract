    <script>
    (function() {
        // Scene 2 实现
        var currentStep = 0;
        var totalSteps = 5;
        var stepNames = ['题目条件', '步骤 1 / 5', '步骤 2 / 5', '步骤 3 / 5', '步骤 4 / 5', '步骤 5 / 5'];
        
        var btnPrev = document.getElementById('scene2-btn-prev');
        var btnNext = document.getElementById('scene2-btn-next');
        var indicator = document.getElementById('scene2-step-indicator');
        
        function showStep(step) {
            // 隐藏所有步骤
            for (var i = 0; i <= totalSteps; i++) {
                var g = document.getElementById('scene2-step-' + i);
                if (g) {
                    g.style.display = 'none';
                }
            }
            // 显示当前步骤
            var currentG = document.getElementById('scene2-step-' + step);
            if (currentG) {
                currentG.style.display = 'block';
            }
            
            // 更新指示器
            indicator.textContent = stepNames[step];
            
            // 更新按钮状态
            btnPrev.disabled = (step === 0);
            if (step === totalSteps) {
                btnNext.textContent = '重新开始';
            } else {
                btnNext.textContent = '下一步 ▶';
            }
        }
        
        btnPrev.addEventListener('click', function() {
            if (currentStep > 0) {
                currentStep--;
                showStep(currentStep);
            }
        });
        
        btnNext.addEventListener('click', function() {
            if (currentStep < totalSteps) {
                currentStep++;
                showStep(currentStep);
            } else {
                // 重新开始
                currentStep = 0;
                showStep(currentStep);
            }
        });
        
        // 初始显示步骤0
        showStep(0);
    })();
    </script>
