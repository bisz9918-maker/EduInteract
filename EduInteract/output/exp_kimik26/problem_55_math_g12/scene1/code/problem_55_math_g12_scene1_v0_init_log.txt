    <script>
    (function() {
        // Scene 1 实现
        const currentStepEl = document.getElementById('scene1-step-title');
        const prevBtn = document.getElementById('scene1-btn-prev');
        const nextBtn = document.getElementById('scene1-btn-next');
        const formulaEl = document.getElementById('scene1-formula');

        let currentStep = 0;
        const totalSteps = 6;

        const stepNames = [
            '步骤 1/6：题图概览',
            '步骤 2/6：标注已知',
            '步骤 3/6：确定点A',
            '步骤 4/6：确定点B',
            '步骤 5/6：发现等边三角形',
            '步骤 6/6：得出结论'
        ];

        const formulas = [
            '',
            '已知：\\(\\angle AFx=60°\\)，圆半径\\(\\frac{p}{2}\\)，\\(NA\\perp y\\)轴',
            '联立\\(\\begin{cases}y=\\sqrt{3}(x-\\frac{p}{2})\\y^2=2px\\end{cases}\\)得\\(A(\\frac{3p}{2},\\sqrt{3}p)\\)，由距离公式\\(AF=2p\\)',
            '\\(B\\)在\\(AF\\)上且\\(FB=\\frac{p}{2}\\)，故\\(AB=AF-FB=\\frac{3p}{2}\\)',
            '\\(NA=\\frac{3p}{2}\\)，\\(NB=\\frac{3p}{2}\\)，\\(AB=\\frac{3p}{2}\\)，\\(\\therefore \\triangle ANB\\)为等边三角形',
            '\\(\\because NA\\perp ON\\ \\therefore \\angle ANO=90°\\)；\\(\\angle ONB=\\angle ANO-\\angle ANB=90°-60°=30°\\)'
        ];

        // SVG元素引用
        const svgEls = {
            circle: document.getElementById('scene1-circle'),
            lineNa: document.getElementById('scene1-line-na'),
            lineAf: document.getElementById('scene1-line-af'),
            parabola1: document.getElementById('scene1-parabola-1'),
            parabola2: document.getElementById('scene1-parabola-2'),
            parabola3: document.getElementById('scene1-parabola-3'),
            parabola4: document.getElementById('scene1-parabola-4'),
            lineFb: document.getElementById('scene1-line-fb'),
            lineAb: document.getElementById('scene1-line-ab'),
            lineNb: document.getElementById('scene1-line-nb'),
            lineOn: document.getElementById('scene1-line-on'),
            angleAfx: document.getElementById('scene1-angle-afx'),
            angleAfxFill: document.getElementById('scene1-angle-afx-fill'),
            angleAnb: document.getElementById('scene1-angle-anb'),
            angleAnbFill: document.getElementById('scene1-angle-anb-fill'),
            rightAngle: document.getElementById('scene1-right-angle'),
            triangleAnb: document.getElementById('scene1-triangle-anb'),
            angleOnb: document.getElementById('scene1-angle-onb'),
            angleOnbFill: document.getElementById('scene1-angle-onb-fill')
        };

        // HTML标注div引用
        const annoEls = {
            a60afx: document.getElementById('scene1-anno-60-afx'),
            radius: document.getElementById('scene1-anno-radius'),
            aCoord: document.getElementById('scene1-anno-a-coord'),
            af: document.getElementById('scene1-anno-af'),
            fb: document.getElementById('scene1-anno-fb'),
            ab: document.getElementById('scene1-anno-ab'),
            bCoord: document.getElementById('scene1-anno-b-coord'),
            na: document.getElementById('scene1-anno-na'),
            nb: document.getElementById('scene1-anno-nb'),
            anb: document.getElementById('scene1-anno-anb'),
            onb: document.getElementById('scene1-anno-onb'),
            answer: document.getElementById('scene1-anno-answer')
        };

        function show(el) {
            if (!el) return;
            el.classList.remove('scene1-hidden');
            if (el.style) { el.style.display = ''; el.style.opacity = '1'; }
        }

        function hide(el) {
            if (!el) return;
            el.classList.add('scene1-hidden');
            if (el.style) { el.style.display = 'none'; el.style.opacity = '0'; }
        }

        function addClass(el, cls) {
            if (el) el.classList.add(cls);
        }

        function removeClass(el, cls) {
            if (el) el.classList.remove(cls);
        }

        function clearHighlights() {
            Object.values(svgEls).forEach(el => {
                if (!el) return;
                removeClass(el, 'scene1-hl-blue');
                removeClass(el, 'scene1-hl-red');
                removeClass(el, 'scene1-hl-green');
            });
            const resetStroke = (el, c, w) => { if (el) { el.setAttribute('stroke', c); el.setAttribute('stroke-width', w); } };
            resetStroke(svgEls.circle, '#000', '1.5');
            resetStroke(svgEls.lineNa, '#000', '1.5');
            resetStroke(svgEls.lineAf, '#000', '1.5');
            resetStroke(svgEls.parabola1, '#000', '1.5');
            resetStroke(svgEls.parabola2, '#000', '1.5');
            resetStroke(svgEls.parabola3, '#000', '1.5');
            resetStroke(svgEls.parabola4, '#000', '1.5');
            resetStroke(svgEls.lineFb, '#000', '1.5');
            resetStroke(svgEls.lineAb, '#000', '1.5');
            resetStroke(svgEls.lineNb, '#000', '1.5');
            resetStroke(svgEls.lineOn, '#000', '1.5');
        }

        function renderStep(step) {
            currentStep = step;
            currentStepEl.textContent = stepNames[step];
            prevBtn.disabled = step === 0;
            nextBtn.disabled = step === totalSteps - 1;

            formulaEl.innerHTML = formulas[step];
            if (window.MathJax && MathJax.typesetPromise) {
                MathJax.typesetPromise([formulaEl]).catch(() => {});
            }

            clearHighlights();

            // 隐藏所有SVG步骤元素
            hide(svgEls.angleAfx); hide(svgEls.angleAfxFill);
            hide(svgEls.angleAnb); hide(svgEls.angleAnbFill);
            hide(svgEls.rightAngle); hide(svgEls.triangleAnb);
            hide(svgEls.angleOnb); hide(svgEls.angleOnbFill);

            // 隐藏所有HTML标注
            Object.values(annoEls).forEach(hide);

            // 确保基础SVG元素可见
            show(svgEls.circle); show(svgEls.lineNa); show(svgEls.lineAf);
            show(svgEls.parabola1); show(svgEls.parabola2); show(svgEls.parabola3); show(svgEls.parabola4);
            show(svgEls.lineFb); show(svgEls.lineAb); show(svgEls.lineNb); show(svgEls.lineOn);

            switch (step) {
                case 0:
                    break;
                case 1:
                    addClass(svgEls.circle, 'scene1-hl-red');
                    addClass(svgEls.lineNa, 'scene1-hl-green');
                    show(svgEls.angleAfx); show(svgEls.angleAfxFill);
                    show(annoEls.a60afx); show(annoEls.radius);
                    break;
                case 2:
                    addClass(svgEls.lineAf, 'scene1-hl-blue');
                    addClass(svgEls.parabola1, 'scene1-hl-blue');
                    addClass(svgEls.parabola2, 'scene1-hl-blue');
                    addClass(svgEls.parabola3, 'scene1-hl-blue');
                    addClass(svgEls.parabola4, 'scene1-hl-blue');
                    show(annoEls.aCoord); show(annoEls.af);
                    show(svgEls.angleAfx); show(svgEls.angleAfxFill);
                    show(annoEls.a60afx); show(annoEls.radius);
                    break;
                case 3:
                    addClass(svgEls.lineFb, 'scene1-hl-red');
                    addClass(svgEls.lineAb, 'scene1-hl-red');
                    show(annoEls.fb); show(annoEls.ab); show(annoEls.bCoord);
                    show(annoEls.aCoord); show(annoEls.af);
                    show(svgEls.angleAfx); show(svgEls.angleAfxFill);
                    show(annoEls.a60afx); show(annoEls.radius);
                    break;
                case 4:
                    addClass(svgEls.lineNa, 'scene1-hl-green');
                    addClass(svgEls.lineNb, 'scene1-hl-green');
                    addClass(svgEls.lineAb, 'scene1-hl-green');
                    show(annoEls.na); show(annoEls.nb);
                    annoEls.ab.classList.remove('scene1-anno-red');
                    annoEls.ab.classList.add('scene1-anno-green');
                    show(annoEls.ab);
                    show(svgEls.angleAnb); show(svgEls.angleAnbFill);
                    show(annoEls.anb);
                    show(svgEls.rightAngle); show(svgEls.triangleAnb);
                    show(annoEls.aCoord); show(annoEls.af); show(annoEls.bCoord);
                    show(svgEls.angleAfx); show(svgEls.angleAfxFill);
                    show(annoEls.a60afx); show(annoEls.radius);
                    break;
                case 5:
                    show(svgEls.angleOnb); show(svgEls.angleOnbFill);
                    show(annoEls.onb); show(annoEls.answer);
                    show(annoEls.na); show(annoEls.nb);
                    annoEls.ab.classList.remove('scene1-anno-red');
                    annoEls.ab.classList.add('scene1-anno-green');
                    show(annoEls.ab);
                    show(svgEls.angleAnb); show(svgEls.angleAnbFill);
                    show(annoEls.anb);
                    show(svgEls.rightAngle); show(svgEls.triangleAnb);
                    show(annoEls.aCoord); show(annoEls.af); show(annoEls.bCoord);
                    show(svgEls.angleAfx); show(svgEls.angleAfxFill);
                    show(annoEls.a60afx); show(annoEls.radius);
                    break;
            }
        }

        prevBtn.addEventListener('click', () => { if (currentStep > 0) renderStep(currentStep - 1); });
        nextBtn.addEventListener('click', () => { if (currentStep < totalSteps - 1) renderStep(currentStep + 1); });

        renderStep(0);
    })();
    </script>
</body>
