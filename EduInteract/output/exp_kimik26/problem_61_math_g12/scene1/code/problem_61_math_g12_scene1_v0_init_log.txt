    <script>
    (function() {
        // Scene 1 实现 - 球体截面与平面ABC几何关系分析
        let currentStep = 0;
        const totalSteps = 5;

        const els = {};
        function getEl(id) {
            if (!els[id]) els[id] = document.getElementById(id);
            return els[id];
        }

        function show(id) {
            const el = getEl(id);
            if (el) { el.style.opacity = '1'; el.style.display = ''; }
        }
        function hide(id) {
            const el = getEl(id);
            if (el) { el.style.opacity = '0'; }
        }
        function setStroke(id, color, width) {
            const el = getEl(id);
            if (el) { el.setAttribute('stroke', color); if (width) el.setAttribute('stroke-width', width); }
        }
        function setFillOpacity(id, val) {
            const el = getEl(id);
            if (el) { el.setAttribute('fill-opacity', val); }
        }
        function setDasharray(id, val) {
            const el = getEl(id);
            if (el) { if (val) el.setAttribute('stroke-dasharray', val); else el.removeAttribute('stroke-dasharray'); }
        }

        function renderStep() {
            getEl('scene1-step-text').textContent = '步骤 ' + currentStep + ' / ' + totalSteps;
            getEl('scene1-btn-prev').disabled = (currentStep === 0);
            getEl('scene1-btn-next').disabled = (currentStep === totalSteps);

            // 先全部隐藏/重置
            hide('scene1-plane-abc'); setFillOpacity('scene1-plane-abc', '0');
            hide('scene1-section-circle'); setFillOpacity('scene1-section-circle', '0');
            setStroke('scene1-section-circle', '#888888');
            getEl('scene1-section-circle').setAttribute('stroke-width', '1');
            hide('scene1-line-oh');
            hide('scene1-point-h'); hide('scene1-label-h');
            hide('scene1-label-d');
            hide('scene1-line-r-small'); hide('scene1-label-r-small');
            hide('scene1-label-ac');
            hide('scene1-label-bc');
            hide('scene1-right-angle');
            hide('scene1-label-angle-abc-red');
            hide('scene1-tick-ab'); hide('scene1-tick-bc');
            hide('scene1-derive-r'); hide('scene1-derive-ac'); hide('scene1-derive-bc');

            // 重置AC样式
            setStroke('scene1-line-ac', '#888888');
            setDasharray('scene1-line-ac', '5,5');
            getEl('scene1-line-ac').setAttribute('stroke-width', '1');
            show('scene1-line-ac');

            // 重置BC样式
            setStroke('scene1-line-bc', '#000000');
            getEl('scene1-line-bc').setAttribute('stroke-width', '1.5');

            // 基础元素始终显示
            show('scene1-ball-outline');
            show('scene1-point-o'); show('scene1-label-o');
            show('scene1-point-a'); show('scene1-label-a');
            show('scene1-point-b'); show('scene1-label-b');
            show('scene1-point-c'); show('scene1-label-c');
            show('scene1-line-ab'); show('scene1-line-bc');
            show('scene1-line-r'); show('scene1-label-r');
            show('scene1-label-babc');

            switch (currentStep) {
                case 0:
                    break;
                case 1:
                    show('scene1-plane-abc'); setFillOpacity('scene1-plane-abc', '0.15');
                    show('scene1-section-circle');
                    break;
                case 2:
                    show('scene1-plane-abc'); setFillOpacity('scene1-plane-abc', '0.15');
                    show('scene1-section-circle');
                    show('scene1-line-oh');
                    show('scene1-point-h'); show('scene1-label-h');
                    show('scene1-label-d');
                    break;
                case 3:
                    show('scene1-plane-abc'); setFillOpacity('scene1-plane-abc', '0.15');
                    show('scene1-section-circle'); setFillOpacity('scene1-section-circle', '0.1');
                    setStroke('scene1-section-circle', '#2e7d32');
                    getEl('scene1-section-circle').setAttribute('stroke-width', '2');
                    show('scene1-line-oh');
                    show('scene1-point-h'); show('scene1-label-h');
                    show('scene1-label-d');
                    show('scene1-line-r-small'); show('scene1-label-r-small');
                    show('scene1-derive-r');
                    break;
                case 4:
                    show('scene1-plane-abc'); setFillOpacity('scene1-plane-abc', '0.15');
                    show('scene1-section-circle'); setFillOpacity('scene1-section-circle', '0.1');
                    setStroke('scene1-section-circle', '#2e7d32');
                    getEl('scene1-section-circle').setAttribute('stroke-width', '2');
                    show('scene1-line-oh');
                    show('scene1-point-h'); show('scene1-label-h');
                    show('scene1-label-d');
                    show('scene1-line-r-small'); show('scene1-label-r-small');
                    show('scene1-derive-r');
                    setStroke('scene1-line-ac', '#c62828');
                    setDasharray('scene1-line-ac', null);
                    getEl('scene1-line-ac').setAttribute('stroke-width', '2');
                    show('scene1-right-angle');
                    show('scene1-label-angle-abc-red');
                    show('scene1-label-ac');
                    show('scene1-derive-ac');
                    break;
                case 5:
                    show('scene1-plane-abc'); setFillOpacity('scene1-plane-abc', '0.15');
                    show('scene1-section-circle'); setFillOpacity('scene1-section-circle', '0.1');
                    setStroke('scene1-section-circle', '#2e7d32');
                    getEl('scene1-section-circle').setAttribute('stroke-width', '2');
                    show('scene1-line-oh');
                    show('scene1-point-h'); show('scene1-label-h');
                    show('scene1-label-d');
                    show('scene1-line-r-small'); show('scene1-label-r-small');
                    show('scene1-derive-r');
                    setStroke('scene1-line-ac', '#c62828');
                    setDasharray('scene1-line-ac', null);
                    getEl('scene1-line-ac').setAttribute('stroke-width', '2');
                    show('scene1-right-angle');
                    show('scene1-label-angle-abc-red');
                    show('scene1-label-ac');
                    show('scene1-derive-ac');
                    setStroke('scene1-line-bc', '#2e7d32');
                    getEl('scene1-line-bc').setAttribute('stroke-width', '2.5');
                    show('scene1-tick-ab'); show('scene1-tick-bc');
                    show('scene1-label-bc');
                    show('scene1-derive-bc');
                    break;
            }
        }

        getEl('scene1-btn-prev').addEventListener('click', function() {
            if (currentStep > 0) { currentStep--; renderStep(); }
        });
        getEl('scene1-btn-next').addEventListener('click', function() {
            if (currentStep < totalSteps) { currentStep++; renderStep(); }
        });

        renderStep();
    })();
    </script>
