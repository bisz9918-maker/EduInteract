    <script>
    (function() {
        // Scene 2 实现
        let currentStep = 0;
        const totalSteps = 6;

        const btnPrev = document.getElementById('scene2-btn-prev');
        const btnNext = document.getElementById('scene2-btn-next');
        const stepIndicator = document.getElementById('scene2-step-indicator');

        const lineOB = document.getElementById('scene2-line-ob');
        const lineOC = document.getElementById('scene2-line-oc');
        const lineBC = document.getElementById('scene2-line-bc');

        // 各步骤对应的元素组
        const stepGroups = {
            1: document.getElementById('scene2-step1-group'),
            2: [
                document.getElementById('scene2-step2-fill-group'),
                document.getElementById('scene2-step2-text-group')
            ],
            3: [
                document.getElementById('scene2-step3-angle-group'),
                document.getElementById('scene2-step3-text-group')
            ],
            4: [
                document.getElementById('scene2-step4-arc-group'),
                document.getElementById('scene2-step4-text-group')
            ],
            5: document.getElementById('scene2-step5-group')
        };

        function setVisibility(el, visible) {
            if (!el) return;
            if (visible) {
                el.classList.remove('scene2-hidden');
                el.classList.add('scene2-visible');
                el.style.display = '';
                el.style.opacity = '1';
            } else {
                el.classList.remove('scene2-visible');
                el.classList.add('scene2-hidden');
                el.style.display = 'none';
                el.style.opacity = '0';
            }
        }

        function renderStep() {
            // 更新步骤指示器
            stepIndicator.textContent = '步骤 ' + (currentStep + 1) + ' / ' + totalSteps;

            // 更新按钮状态
            btnPrev.disabled = (currentStep === 0);
            btnNext.disabled = (currentStep === totalSteps - 1);

            // 步骤1：显示BC=3
            setVisibility(stepGroups[1], currentStep >= 1);

            // 步骤2：显示三角形填充和等边判定文字，高亮三边
            const step2Els = Array.isArray(stepGroups[2]) ? stepGroups[2] : [stepGroups[2]];
            step2Els.forEach(el => setVisibility(el, currentStep >= 2));
            if (currentStep >= 2) {
                lineOB.setAttribute('stroke-width', '2.5');
                lineOC.setAttribute('stroke-width', '2.5');
                lineBC.setAttribute('stroke-width', '2.5');
            } else {
                lineOB.setAttribute('stroke-width', '1.5');
                lineOC.setAttribute('stroke-width', '1.5');
                lineBC.setAttribute('stroke-width', '1.5');
            }

            // 步骤3：显示角度弧线和中心角说明
            const step3Els = Array.isArray(stepGroups[3]) ? stepGroups[3] : [stepGroups[3]];
            step3Els.forEach(el => setVisibility(el, currentStep >= 3));

            // 步骤4：显示大圆劣弧和路径说明
            const step4Els = Array.isArray(stepGroups[4]) ? stepGroups[4] : [stepGroups[4]];
            step4Els.forEach(el => setVisibility(el, currentStep >= 4));

            // 步骤5：显示结论公式
            setVisibility(stepGroups[5], currentStep >= 5);
        }

        btnPrev.addEventListener('click', function() {
            if (currentStep > 0) {
                currentStep--;
                renderStep();
            }
        });

        btnNext.addEventListener('click', function() {
            if (currentStep < totalSteps - 1) {
                currentStep++;
                renderStep();
            }
        });

        // 初始渲染
        renderStep();
    })();
    </script>
