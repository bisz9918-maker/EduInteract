    <script>
    (function() {
        // Scene 3 实现
        // 坐标表（来自实现计划）：
        // A(280,231), B(360,369), C(680,369)
        // D(320,300), E(480,300), F(520,369), G(400,300)
        // A' 步骤0: (340,169), 步骤2: (400,189), 步骤3: (360,172)

        var currentStep = 0;
        var animationId = null;
        var isAnimating = false;

        // 各步骤配置
        var steps = [
            {
                caption: "异面直线 A′E（蓝）与 BD（红）的夹角如何度量？",
                aPrime: [340, 169],
                bdStroke: "#c62828", bdWidth: 3,
                showL: false, showAlpha: false, showVert: false, showAG: false, showRightG: false,
                alphaPos: [425, 225], aprimeLabelPos: [320, 159],
                sideNote: "", showSideNote: false
            },
            {
                caption: "平移 BD 至过点 E 得直线 l，∠(A′E, l) = α 即为异面直线夹角。",
                aPrime: [340, 169],
                bdStroke: "#000000", bdWidth: 1,
                showL: true, showAlpha: true, showVert: false, showAG: false, showRightG: false,
                alphaPos: [425, 225], aprimeLabelPos: [320, 159],
                sideNote: "", showSideNote: false
            },
            {
                caption: "A′ 到体积最大位置（A′G⊥AF）时，α ≈ 8.5° ≠ 90°，但 α 随旋转变化。",
                aPrime: [400, 189],
                bdStroke: "#000000", bdWidth: 1,
                showL: true, showAlpha: true, showVert: false, showAG: true, showRightG: true,
                alphaPos: [445, 235], aprimeLabelPos: [380, 179],
                sideNote: "α ≈ 8.5° ≠ 90°", showSideNote: true
            },
            {
                caption: "cosθ = 1/3 时，A′E ⊥ BD，命题 (4) 错误。",
                aPrime: [360, 172],
                bdStroke: "#000000", bdWidth: 1,
                showL: true, showAlpha: true, showVert: true, showAG: true, showRightG: false,
                alphaPos: [435, 230], aprimeLabelPos: [340, 162],
                sideNote: "cosθ = 1/3 时，A′E ⊥ BD", showSideNote: true
            }
        ];

        // 元素引用
        var el = {
            lineBD: document.getElementById("scene3-line-BD"),
            lineAE: document.getElementById("scene3-line-AE"),
            lineAD: document.getElementById("scene3-line-AD"),
            lineAG: document.getElementById("scene3-line-AG"),
            lineL: document.getElementById("scene3-line-l"),
            arcAlpha: document.getElementById("scene3-arc-alpha"),
            perpSymbol: document.getElementById("scene3-perp-symbol"),
            rightAngleG: document.getElementById("scene3-right-angle-G"),
            pointAprime: document.getElementById("scene3-point-Aprime"),
            labelAprime: document.getElementById("scene3-label-Aprime"),
            labelL: document.getElementById("scene3-label-l"),
            labelAlpha: document.getElementById("scene3-label-alpha"),
            caption: document.getElementById("scene3-caption"),
            sideNote: document.getElementById("scene3-side-note"),
            btnPrev: document.getElementById("scene3-btn-prev"),
            btnNext: document.getElementById("scene3-btn-next"),
            stepIndicator: document.getElementById("scene3-step-indicator")
        };

        // 固定点坐标
        var E = [480, 300];
        var D = [320, 300];
        var G = [400, 300];
        var F = [520, 369];

        function setDisplay(elem, show) {
            if (!elem) return;
            elem.style.display = show ? "block" : "none";
            elem.style.opacity = show ? "1" : "0";
        }

        function setLineEndpoints(line, x1, y1, x2, y2) {
            line.setAttribute("x1", x1);
            line.setAttribute("y1", y1);
            line.setAttribute("x2", x2);
            line.setAttribute("y2", y2);
        }

        // 计算单位向量
        function normalize(v) {
            var len = Math.sqrt(v[0]*v[0] + v[1]*v[1]);
            if (len === 0) return [0, 0];
            return [v[0]/len, v[1]/len];
        }

        // 绘制夹角弧线（圆心E，从EA'到EF方向）
        function drawAlphaArc(aPrime) {
            var r = 28;
            var vEA = [aPrime[0] - E[0], aPrime[1] - E[1]];
            var vEF = [F[0] - E[0], F[1] - E[1]];
            var uEA = normalize(vEA);
            var uEF = normalize(vEF);

            var start = [E[0] + r * uEA[0], E[1] + r * uEA[1]];
            var end = [E[0] + r * uEF[0], E[1] + r * uEF[1]];

            // 计算 sweep 方向（叉积判断）
            var cross = uEA[0]*uEF[1] - uEA[1]*uEF[0];
            var sweep = cross > 0 ? 1 : 0;

            var d = "M " + start[0].toFixed(1) + "," + start[1].toFixed(1) +
                    " A " + r + "," + r + " 0 0 " + sweep + " " + end[0].toFixed(1) + "," + end[1].toFixed(1);
            el.arcAlpha.setAttribute("d", d);
        }

        // 绘制垂直符号（在E点，EA'与l之间）
        function drawPerpSymbol(aPrime) {
            var size = 12;
            var vEA = [aPrime[0] - E[0], aPrime[1] - E[1]];
            var vEF = [F[0] - E[0], F[1] - E[1]];
            var uEA = normalize(vEA);
            var uEF = normalize(vEF);

            var p1 = [E[0] + size * uEA[0], E[1] + size * uEA[1]];
            var p2 = [E[0] + size * uEF[0], E[1] + size * uEF[1]];
            var p3 = [p1[0] + size * uEF[0], p1[1] + size * uEF[1]];

            var d = "M " + p1[0].toFixed(1) + "," + p1[1].toFixed(1) +
                    " L " + p3[0].toFixed(1) + "," + p3[1].toFixed(1) +
                    " L " + p2[0].toFixed(1) + "," + p2[1].toFixed(1);
            el.perpSymbol.setAttribute("d", d);
        }

        // 绘制G处直角符号（A'G ⊥ AF）
        function drawRightAngleG(aPrime) {
            var size = 10;
            var vGA = [aPrime[0] - G[0], aPrime[1] - G[1]];
            var vGF = [F[0] - G[0], F[1] - G[1]];
            var uGA = normalize(vGA);
            var uGF = normalize(vGF);

            var p1 = [G[0] + size * uGA[0], G[1] + size * uGA[1]];
            var p2 = [G[0] + size * uGF[0], G[1] + size * uGF[1]];
            var p3 = [p1[0] + size * uGF[0], p1[1] + size * uGF[1]];

            var d = "M " + p1[0].toFixed(1) + "," + p1[1].toFixed(1) +
                    " L " + p3[0].toFixed(1) + "," + p3[1].toFixed(1) +
                    " L " + p2[0].toFixed(1) + "," + p2[1].toFixed(1);
            el.rightAngleG.setAttribute("d", d);
        }

        // 更新所有依赖A'位置的元素
        function updateAPrimeDependent(aPrime) {
            setLineEndpoints(el.lineAE, aPrime[0], aPrime[1], E[0], E[1]);
            setLineEndpoints(el.lineAD, aPrime[0], aPrime[1], D[0], D[1]);
            setLineEndpoints(el.lineAG, aPrime[0], aPrime[1], G[0], G[1]);
            el.pointAprime.setAttribute("cx", aPrime[0]);
            el.pointAprime.setAttribute("cy", aPrime[1]);

            if (el.labelAlpha.style.display !== "none") {
                drawAlphaArc(aPrime);
            }
            if (el.perpSymbol.style.display !== "none") {
                drawPerpSymbol(aPrime);
            }
            if (el.rightAngleG.style.display !== "none") {
                drawRightAngleG(aPrime);
            }
        }

        // 应用步骤配置（无动画）
        function applyStepConfig(step) {
            var cfg = steps[step];

            // BD样式
            el.lineBD.setAttribute("stroke", cfg.bdStroke);
            el.lineBD.setAttribute("stroke-width", cfg.bdWidth);

            // A'位置及依赖元素
            updateAPrimeDependent(cfg.aPrime);

            // 标签位置
            el.labelAprime.setAttribute("x", cfg.aprimeLabelPos[0]);
            el.labelAprime.setAttribute("y", cfg.aprimeLabelPos[1]);
            el.labelAlpha.setAttribute("x", cfg.alphaPos[0]);
            el.labelAlpha.setAttribute("y", cfg.alphaPos[1]);

            // 显隐控制
            setDisplay(el.lineL, cfg.showL);
            setDisplay(el.labelL, cfg.showL);
            setDisplay(el.arcAlpha, cfg.showAlpha);
            setDisplay(el.labelAlpha, cfg.showAlpha);
            setDisplay(el.perpSymbol, cfg.showVert);
            setDisplay(el.lineAG, cfg.showAG);
            setDisplay(el.rightAngleG, cfg.showRightG);
            setDisplay(el.sideNote, cfg.showSideNote);

            // 文字内容
            el.caption.textContent = cfg.caption;
            el.sideNote.textContent = cfg.sideNote;

            // 按钮状态
            el.btnPrev.disabled = step === 0;
            el.btnNext.disabled = step === steps.length - 1;
            el.stepIndicator.textContent = "步骤 " + (step + 1) + "/" + steps.length;
        }

        // 动画过渡到目标步骤
        function animateToStep(targetStep) {
            if (isAnimating && animationId) {
                cancelAnimationFrame(animationId);
            }
            isAnimating = true;

            var startStep = currentStep;
            var startCfg = steps[startStep];
            var targetCfg = steps[targetStep];
            var startAPrime = startCfg.aPrime;
            var targetAPrime = targetCfg.aPrime;
            var duration = 600;
            var startTime = performance.now();

            // 先应用目标步骤的非位置相关配置
            el.lineBD.setAttribute("stroke", targetCfg.bdStroke);
            el.lineBD.setAttribute("stroke-width", targetCfg.bdWidth);
            setDisplay(el.lineL, targetCfg.showL);
            setDisplay(el.labelL, targetCfg.showL);
            setDisplay(el.lineAG, targetCfg.showAG);
            setDisplay(el.sideNote, targetCfg.showSideNote);
            el.sideNote.textContent = targetCfg.sideNote;
            el.caption.textContent = targetCfg.caption;

            // 对于后退到步骤0/1，需要隐藏某些元素
            if (targetStep < 2) {
                setDisplay(el.rightAngleG, false);
            }
            if (targetStep < 3) {
                setDisplay(el.perpSymbol, false);
            }
            if (targetStep < 1) {
                setDisplay(el.arcAlpha, false);
                setDisplay(el.labelAlpha, false);
            }

            function easeInOut(t) {
                return t < 0.5 ? 2*t*t : -1 + (4 - 2*t)*t;
            }

            function frame(now) {
                var elapsed = now - startTime;
                var t = Math.min(elapsed / duration, 1);
                var et = easeInOut(t);

                var curAPrime = [
                    startAPrime[0] + (targetAPrime[0] - startAPrime[0]) * et,
                    startAPrime[1] + (targetAPrime[1] - startAPrime[1]) * et
                ];

                updateAPrimeDependent(curAPrime);

                // 标签位置插值
                var curLabelX = startCfg.aprimeLabelPos[0] + (targetCfg.aprimeLabelPos[0] - startCfg.aprimeLabelPos[0]) * et;
                var curLabelY = startCfg.aprimeLabelPos[1] + (targetCfg.aprimeLabelPos[1] - startCfg.aprimeLabelPos[1]) * et;
                el.labelAprime.setAttribute("x", curLabelX);
                el.labelAprime.setAttribute("y", curLabelY);

                var curAlphaX = startCfg.alphaPos[0] + (targetCfg.alphaPos[0] - startCfg.alphaPos[0]) * et;
                var curAlphaY = startCfg.alphaPos[1] + (targetCfg.alphaPos[1] - startCfg.alphaPos[1]) * et;
                el.labelAlpha.setAttribute("x", curAlphaX);
                el.labelAlpha.setAttribute("y", curAlphaY);

                // 步骤2/3的alpha和垂直符号在动画中逐步显示
                if (targetStep >= 1 && t > 0.3) {
                    setDisplay(el.arcAlpha, true);
                    setDisplay(el.labelAlpha, true);
                }
                if (targetStep >= 2 && t > 0.3) {
                    setDisplay(el.rightAngleG, targetCfg.showRightG);
                }
                if (targetStep >= 3 && t > 0.3) {
                    setDisplay(el.perpSymbol, targetCfg.showVert);
                }

                if (t < 1) {
                    animationId = requestAnimationFrame(frame);
                } else {
                    isAnimating = false;
                    currentStep = targetStep;
                    applyStepConfig(currentStep);
                }
            }

            animationId = requestAnimationFrame(frame);
        }

        function setState(newStep) {
            if (newStep === currentStep) return;
            if (newStep < 0 || newStep >= steps.length) return;

            // 如果A'位置不同，使用动画；否则直接切换
            var startAPrime = steps[currentStep].aPrime;
            var targetAPrime = steps[newStep].aPrime;
            var needsAnimation = (startAPrime[0] !== targetAPrime[0] || startAPrime[1] !== targetAPrime[1]);

            if (needsAnimation) {
                animateToStep(newStep);
            } else {
                currentStep = newStep;
                applyStepConfig(currentStep);
            }
        }

        // 事件绑定
        el.btnPrev.addEventListener("click", function() {
            setState(currentStep - 1);
        });
        el.btnNext.addEventListener("click", function() {
            setState(currentStep + 1);
        });

        // 初始化
        applyStepConfig(0);

    })();
    </script>
