    <script>
    (function() {
        // Scene 1 实现
        var currentStep = 0;
        var MAX_STEP = 4;

        var btnPrev = document.getElementById('scene1-btn-prev');
        var btnNext = document.getElementById('scene1-btn-next');

        // 元素引用
        var elLineOc = document.getElementById('scene1-line-oc');
        var elTriangleOcp = document.getElementById('scene1-triangle-ocp');
        var elRightAngleOc = document.getElementById('scene1-rightangle-oc');
        var elAnnoOc = document.getElementById('scene1-anno-oc');
        var elAnnoOp = document.getElementById('scene1-anno-op');
        var elAnnoPc = document.getElementById('scene1-anno-pc');
        var elAnnoCd = document.getElementById('scene1-anno-cd');
        var elFormula = document.getElementById('scene1-formula');
        var elLineCd = document.getElementById('scene1-line-cd');

        function show(el) {
            if (!el) return;
            el.classList.remove('scene1-step-hidden');
            el.classList.add('scene1-step-visible');
            el.style.display = '';
            el.style.opacity = '1';
        }

        function hide(el) {
            if (!el) return;
            el.classList.remove('scene1-step-visible');
            el.classList.add('scene1-step-hidden');
            el.style.opacity = '0';
        }

        function setFormula(html) {
            if (!elFormula) return;
            elFormula.innerHTML = html;
        }

        function updateButtons() {
            btnPrev.disabled = currentStep <= 0;
            btnNext.disabled = currentStep >= MAX_STEP;
        }

        function enterStep(step) {
            switch (step) {
                case 0:
                    // 初始状态：基础图形 + 已知标注
                    hide(elLineOc);
                    elLineOc.classList.remove('scene1-oc-animate');
                    hide(elTriangleOcp);
                    hide(elRightAngleOc);
                    hide(elAnnoOc);
                    hide(elAnnoOp);
                    hide(elAnnoPc);
                    hide(elAnnoCd);
                    hide(elFormula);
                    elLineCd.setAttribute('stroke', '#000000');
                    elLineCd.setAttribute('stroke-dasharray', 'none');
                    break;
                case 1:
                    // 切线性质：OC + 直角标记 + OC=2
                    show(elLineOc);
                    elLineOc.classList.add('scene1-oc-animate');
                    show(elRightAngleOc);
                    show(elAnnoOc);
                    break;
                case 2:
                    // 直角三角形三边：填充 + OP=4 + PC=2√3
                    show(elTriangleOcp);
                    // 延迟弹出标注
                    setTimeout(function() { show(elAnnoOp); }, 0);
                    setTimeout(function() { show(elAnnoPc); }, 150);
                    break;
                case 3:
                    // 面积法：CD变虚线 + 公式
                    elLineCd.setAttribute('stroke', '#888888');
                    elLineCd.setAttribute('stroke-dasharray', '5,5');
                    show(elFormula);
                    setFormula('<tspan fill="#1565c0">OC</tspan> · <tspan fill="#c62828">PC</tspan> = <tspan fill="#c62828">OP</tspan> · <tspan fill="#2e7d32">CD</tspan>');
                    break;
                case 4:
                    // 结论：CD=√3 + 公式更新
                    show(elAnnoCd);
                    show(elFormula);
                    setFormula('2 × 2√3 = 4 × <tspan fill="#2e7d32">CD</tspan>&nbsp;&nbsp;&nbsp;→&nbsp;&nbsp;&nbsp;<tspan fill="#2e7d32" font-weight="bold">CD = √3</tspan>');
                    break;
            }
            updateButtons();
        }

        function leaveStep(step) {
            switch (step) {
                case 0:
                    break;
                case 1:
                    hide(elLineOc);
                    elLineOc.classList.remove('scene1-oc-animate');
                    hide(elRightAngleOc);
                    hide(elAnnoOc);
                    break;
                case 2:
                    hide(elTriangleOcp);
                    hide(elAnnoOp);
                    hide(elAnnoPc);
                    break;
                case 3:
                    elLineCd.setAttribute('stroke', '#000000');
                    elLineCd.setAttribute('stroke-dasharray', 'none');
                    hide(elFormula);
                    break;
                case 4:
                    hide(elAnnoCd);
                    hide(elFormula);
                    break;
            }
        }

        function goToStep(newStep) {
            if (newStep < 0 || newStep > MAX_STEP) return;
            leaveStep(currentStep);
            currentStep = newStep;
            enterStep(currentStep);
        }

        btnPrev.addEventListener('click', function() {
            goToStep(currentStep - 1);
        });

        btnNext.addEventListener('click', function() {
            goToStep(currentStep + 1);
        });

        // 初始化
        enterStep(0);
    })();
    </script>
