    <script>
    (function() {
        // Scene 1 实现
        const currentStep = { value: 0 };
        const MAX_STEP = 6;

        // 元素引用
        const els = {
            edgeAC: document.getElementById('scene1-edge-AC'),
            edgeBD: document.getElementById('scene1-edge-BD'),
            edgeMD: document.getElementById('scene1-edge-MD'),
            edgeBM: document.getElementById('scene1-edge-BM'),
            edgePA: document.getElementById('scene1-edge-PA'),
            edgePC: document.getElementById('scene1-edge-PC'),
            vertexO: document.getElementById('scene1-vertex-O'),
            vertexM: document.getElementById('scene1-vertex-M'),
            planePAC: document.getElementById('scene1-plane-PAC'),
            planePCD: document.getElementById('scene1-plane-PCD'),
            planeMBD: document.getElementById('scene1-plane-MBD'),
            rightangleO: document.getElementById('scene1-rightangle-O'),
            rightangleM: document.getElementById('scene1-rightangle-M'),
            labelO: document.getElementById('scene1-label-O'),
            labelM: document.getElementById('scene1-label-M'),
            labelACBD: document.getElementById('scene1-label-AC-BD'),
            labelPlanePAC: document.getElementById('scene1-label-plane-PAC'),
            labelBDPlanePAC: document.getElementById('scene1-label-BD-planePAC'),
            labelBDPC: document.getElementById('scene1-label-BD-PC'),
            labelPlaneMBD: document.getElementById('scene1-label-plane-MBD'),
            labelPlanePCD: document.getElementById('scene1-label-plane-PCD'),
            labelPCPlaneMBD: document.getElementById('scene1-label-PC-planeMBD'),
            labelConclusion: document.getElementById('scene1-label-conclusion'),
            labelConclusionNote: document.getElementById('scene1-label-conclusion-note'),
            stepDesc: document.getElementById('scene1-step-desc'),
            btnPrev: document.getElementById('scene1-btn-prev'),
            btnNext: document.getElementById('scene1-btn-next'),
        };

        // M 点坐标
        const M_DEFAULT = { x: 500, y: 233 };
        const M_FINAL = { x: 581, y: 272 };

        // 标签坐标
        const M_LABEL_DEFAULT = { x: 570, y: 190 };
        const M_LABEL_FINAL = { x: 651, y: 229 };

        // 步骤说明文字
        const stepDescriptions = [
            "已知：PA⊥底面ABCD，底面为菱形，M是PC上的动点",
            "菱形性质：对角线 AC⊥BD，交于点 O",
            "PA⊥BD 且 AC⊥BD → BD⊥平面PAC",
            "由 BD⊥平面PAC，且 PC⊂平面PAC，得 BD⊥PC",
            "分析平面MBD 与平面PCD，交线为 MD，探究面面垂直条件",
            "补充条件 BM⊥PC，结合 BD⊥PC，得 PC⊥平面MBD",
            "PC⊥平面MBD 且 PC⊂平面PCD → 平面MBD⊥平面PCD"
        ];

        // 工具函数
        function setOpacity(el, opacity) {
            if (el) el.setAttribute('opacity', opacity);
        }
        function setStrokeWidth(el, width) {
            if (el) el.setAttribute('stroke-width', width);
        }
        function setStroke(el, color) {
            if (el) el.setAttribute('stroke', color);
        }
        function setFillOpacity(el, opacity) {
            if (el) el.setAttribute('fill-opacity', opacity);
        }
        function setStrokeOpacity(el, opacity) {
            if (el) el.setAttribute('stroke-opacity', opacity);
        }

        // 动画：M 点移动
        let animFrame = null;
        function animateMTo(targetX, targetY, duration, callback) {
            const startX = parseFloat(els.vertexM.getAttribute('cx'));
            const startY = parseFloat(els.vertexM.getAttribute('cy'));
            const labelStartX = parseFloat(els.labelM.getAttribute('x'));
            const labelStartY = parseFloat(els.labelM.getAttribute('y'));
            const startTime = performance.now();

            function easeInOut(t) {
                return t < 0.5 ? 2 * t * t : -1 + (4 - 2 * t) * t;
            }

            function frame(now) {
                const elapsed = now - startTime;
                const t = Math.min(elapsed / duration, 1);
                const et = easeInOut(t);
                const x = startX + (targetX - startX) * et;
                const y = startY + (targetY - startY) * et;

                // 更新 M 点
                els.vertexM.setAttribute('cx', x);
                els.vertexM.setAttribute('cy', y);

                // 更新 BM 线段终点
                els.edgeBM.setAttribute('x2', x);
                els.edgeBM.setAttribute('y2', y);

                // 更新 MD 线段起点
                els.edgeMD.setAttribute('x1', x);
                els.edgeMD.setAttribute('y1', y);

                // 更新平面MBD多边形
                els.planeMBD.setAttribute('points', `${x},${y} 520,400 340,300`);

                // 更新 M 标签位置（同步插值）
                const labelTargetX = targetX === M_FINAL.x ? M_LABEL_FINAL.x : M_LABEL_DEFAULT.x;
                const labelTargetY = targetY === M_FINAL.y ? M_LABEL_FINAL.y : M_LABEL_DEFAULT.y;
                const lx = labelStartX + (labelTargetX - labelStartX) * et;
                const ly = labelStartY + (labelTargetY - labelStartY) * et;
                els.labelM.setAttribute('x', lx);
                els.labelM.setAttribute('y', ly);

                if (t < 1) {
                    animFrame = requestAnimationFrame(frame);
                } else {
                    animFrame = null;
                    if (callback) callback();
                }
            }

            if (animFrame) cancelAnimationFrame(animFrame);
            animFrame = requestAnimationFrame(frame);
        }

        // 渲染当前步骤
        function render() {
            const step = currentStep.value;

            // 更新按钮状态
            els.btnPrev.disabled = step === 0;
            els.btnNext.disabled = step === MAX_STEP;

            // 更新步骤说明
            els.stepDesc.textContent = stepDescriptions[step];

            // ===== 步骤0：基础条件 =====
            if (step === 0) {
                setOpacity(els.edgeAC, 0);
                setOpacity(els.vertexO, 0);
                setOpacity(els.rightangleO, 0);
                setOpacity(els.labelO, 0);
                setOpacity(els.labelACBD, 0);
                setOpacity(els.labelPlanePAC, 0);
                setOpacity(els.labelBDPlanePAC, 0);
                setOpacity(els.labelBDPC, 0);
                setOpacity(els.labelPlaneMBD, 0);
                setOpacity(els.labelPlanePCD, 0);
                setOpacity(els.labelPCPlaneMBD, 0);
                setOpacity(els.labelConclusion, 0);
                setOpacity(els.labelConclusionNote, 0);

                // 隐藏平面多边形（通过opacity=0）
                setOpacity(els.planePAC, 0);
                setOpacity(els.planePCD, 0);
                setOpacity(els.planeMBD, 0);

                // 恢复棱线样式
                setStroke(els.edgeBD, '#888');
                setStrokeWidth(els.edgeBD, 1.5);
                setStroke(els.edgeAC, '#888');
                setStrokeWidth(els.edgeAC, 1.5);
                setStroke(els.edgePA, '#000');
                setStrokeWidth(els.edgePA, 2);
                setStroke(els.edgePC, '#000');
                setStrokeWidth(els.edgePC, 2);
                setStroke(els.edgeBM, '#000');
                setStrokeWidth(els.edgeBM, 2);
                setStroke(els.edgeMD, '#888');
                setStrokeWidth(els.edgeMD, 1.5);

                // M 回到默认位置
                els.vertexM.setAttribute('cx', M_DEFAULT.x);
                els.vertexM.setAttribute('cy', M_DEFAULT.y);
                els.edgeBM.setAttribute('x2', M_DEFAULT.x);
                els.edgeBM.setAttribute('y2', M_DEFAULT.y);
                els.edgeMD.setAttribute('x1', M_DEFAULT.x);
                els.edgeMD.setAttribute('y1', M_DEFAULT.y);
                els.planeMBD.setAttribute('points', `${M_DEFAULT.x},${M_DEFAULT.y} 520,400 340,300`);
                els.labelM.setAttribute('x', M_LABEL_DEFAULT.x);
                els.labelM.setAttribute('y', M_LABEL_DEFAULT.y);
                setOpacity(els.rightangleM, 0);
            }

            // ===== 步骤1：菱形对角线性质 =====
            if (step >= 1) {
                setOpacity(els.edgeAC, 1);
                setOpacity(els.vertexO, 1);
                setOpacity(els.rightangleO, 1);
                setOpacity(els.labelO, 1);
                setOpacity(els.labelACBD, 1);
                // 高亮 AC 和 BD
                setStroke(els.edgeAC, '#1565c0');
                setStrokeWidth(els.edgeAC, 2.5);
                setStroke(els.edgeBD, '#1565c0');
                setStrokeWidth(els.edgeBD, 2.5);
            } else {
                setStroke(els.edgeBD, '#888');
                setStrokeWidth(els.edgeBD, 1.5);
            }

            // ===== 步骤2：线面垂直 =====
            if (step >= 2) {
                setOpacity(els.planePAC, 1);
                setOpacity(els.labelPlanePAC, 1);
                setOpacity(els.labelBDPlanePAC, 1);
                // 高亮 PA
                setStroke(els.edgePA, '#1565c0');
                setStrokeWidth(els.edgePA, 2.5);
            } else if (step < 2) {
                setStroke(els.edgePA, '#000');
                setStrokeWidth(els.edgePA, 2);
            }

            // ===== 步骤3：线线垂直 =====
            if (step >= 3) {
                setOpacity(els.labelBDPC, 1);
                // 高亮 BD 和 PC（红色）
                setStroke(els.edgeBD, '#c62828');
                setStrokeWidth(els.edgeBD, 2.5);
                setStroke(els.edgePC, '#c62828');
                setStrokeWidth(els.edgePC, 2.5);
            } else if (step < 3 && step >= 1) {
                setStroke(els.edgeBD, '#1565c0');
                setStrokeWidth(els.edgeBD, 2.5);
                setStroke(els.edgePC, '#000');
                setStrokeWidth(els.edgePC, 2);
            }

            // ===== 步骤4：目标平面分析 =====
            if (step >= 4) {
                setOpacity(els.planeMBD, 1);
                setOpacity(els.planePCD, 1);
                setOpacity(els.labelPlaneMBD, 1);
                setOpacity(els.labelPlanePCD, 1);
                // 高亮 MD（红色）
                setStroke(els.edgeMD, '#c62828');
                setStrokeWidth(els.edgeMD, 2.5);
            } else if (step < 4) {
                setStroke(els.edgeMD, '#888');
                setStrokeWidth(els.edgeMD, 1.5);
            }

            // ===== 步骤5：补充垂直条件 =====
            if (step >= 5) {
                setOpacity(els.labelPCPlaneMBD, 1);
                setOpacity(els.rightangleM, 1);
                // 高亮 BM 和 PC
                setStroke(els.edgeBM, '#c62828');
                setStrokeWidth(els.edgeBM, 2.5);
                setStroke(els.edgePC, '#c62828');
                setStrokeWidth(els.edgePC, 2.5);
            } else if (step < 5) {
                setOpacity(els.labelPCPlaneMBD, 0);
                setOpacity(els.rightangleM, 0);
                setStroke(els.edgeBM, '#000');
                setStrokeWidth(els.edgeBM, 2);
            }

            // ===== 步骤6：结论 =====
            if (step >= 6) {
                setOpacity(els.labelConclusion, 1);
                setOpacity(els.labelConclusionNote, 1);
                setFillOpacity(els.planeMBD, 0.30);
                setStrokeWidth(els.planeMBD, 2);
                setFillOpacity(els.planePCD, 0.30);
                setStrokeWidth(els.planePCD, 2);
            } else if (step < 6) {
                setOpacity(els.labelConclusion, 0);
                setOpacity(els.labelConclusionNote, 0);
                if (step >= 4) {
                    setFillOpacity(els.planeMBD, 0.20);
                    setStrokeWidth(els.planeMBD, 1.5);
                    setFillOpacity(els.planePCD, 0.15);
                    setStrokeWidth(els.planePCD, 1.5);
                }
            }
        }

        // 按钮事件
        els.btnPrev.addEventListener('click', function() {
            if (currentStep.value > 0) {
                currentStep.value--;
                // 如果回退到步骤5之前，M 需要回到默认位置
                if (currentStep.value < 5) {
                    els.vertexM.setAttribute('cx', M_DEFAULT.x);
                    els.vertexM.setAttribute('cy', M_DEFAULT.y);
                    els.edgeBM.setAttribute('x2', M_DEFAULT.x);
                    els.edgeBM.setAttribute('y2', M_DEFAULT.y);
                    els.edgeMD.setAttribute('x1', M_DEFAULT.x);
                    els.edgeMD.setAttribute('y1', M_DEFAULT.y);
                    els.planeMBD.setAttribute('points', `${M_DEFAULT.x},${M_DEFAULT.y} 520,400 340,300`);
                    els.labelM.setAttribute('x', M_LABEL_DEFAULT.x);
                    els.labelM.setAttribute('y', M_LABEL_DEFAULT.y);
                }
                render();
            }
        });

        els.btnNext.addEventListener('click', function() {
            if (currentStep.value < MAX_STEP) {
                currentStep.value++;
                render();
                // 步骤5时启动 M 点动画
                if (currentStep.value === 5) {
                    animateMTo(M_FINAL.x, M_FINAL.y, 500);
                }
            }
        });

        // 初始渲染
        render();
    })();
    </script>
</body>
