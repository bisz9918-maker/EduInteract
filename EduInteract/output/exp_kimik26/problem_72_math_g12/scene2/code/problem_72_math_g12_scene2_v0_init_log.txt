    <script>
    (function() {
        var currentStep = 0;
        var maxStep = 6;

        var el = {
            title: document.getElementById('scene2-title'),
            hint: document.getElementById('scene2-hint'),
            formula: document.getElementById('scene2-formula'),
            summary: document.getElementById('scene2-summary'),
            faceBase: document.getElementById('scene2-face-base'),
            facePab: document.getElementById('scene2-face-pab'),
            facePbc: document.getElementById('scene2-face-pbc'),
            facePcd: document.getElementById('scene2-face-pcd'),
            facePda: document.getElementById('scene2-face-pda'),
            linePm: document.getElementById('scene2-line-pm'),
            lineOm: document.getElementById('scene2-line-om'),
            ptM: document.getElementById('scene2-pt-m'),
            labelM: document.getElementById('scene2-label-m'),
            labelPa: document.getElementById('scene2-label-pa'),
            labelPb: document.getElementById('scene2-label-pb'),
            labelPc: document.getElementById('scene2-label-pc'),
            labelPm: document.getElementById('scene2-label-pm'),
            raA: document.getElementById('scene2-ra-a'),
            raB: document.getElementById('scene2-ra-b'),
            raM: document.getElementById('scene2-ra-m'),
            rightBase: document.getElementById('scene2-right-base'),
            rightPab: document.getElementById('scene2-right-pab'),
            rightPad: document.getElementById('scene2-right-pad'),
            rightPbc: document.getElementById('scene2-right-pbc'),
            rightPcd: document.getElementById('scene2-right-pcd'),
            btnPrev: document.getElementById('scene2-btn-prev'),
            btnReset: document.getElementById('scene2-btn-reset'),
            btnNext: document.getElementById('scene2-btn-next')
        };

        function show(elem) { if(elem) { elem.style.display = ''; elem.style.opacity = '1'; } }
        function hide(elem) { if(elem) { elem.style.display = 'none'; } }
        function setText(elem, text) { if(elem) elem.textContent = text; }

        function setFaceFill(elem, color, opacity) {
            if(elem) {
                elem.setAttribute('fill', color);
                elem.setAttribute('fill-opacity', opacity);
            }
        }

        function renderStep(step) {
            hide(el.hint);
            hide(el.rightBase); hide(el.rightPab); hide(el.rightPad); hide(el.rightPbc); hide(el.rightPcd);
            hide(el.formula); hide(el.summary);
            hide(el.linePm); hide(el.lineOm); hide(el.ptM); hide(el.labelM);
            hide(el.labelPa); hide(el.labelPb); hide(el.labelPc); hide(el.labelPm);
            hide(el.raA); hide(el.raB); hide(el.raM);

            setFaceFill(el.faceBase, '#eeeeee', '0.30');
            setFaceFill(el.facePab, '#888888', '0.10');
            setFaceFill(el.facePbc, '#888888', '0.10');
            setFaceFill(el.facePcd, '#888888', '0.10');
            setFaceFill(el.facePda, '#888888', '0.10');

            el.btnPrev.disabled = (step <= 0);
            el.btnNext.disabled = (step >= maxStep);

            switch(step) {
                case 0:
                    el.title.textContent = '四棱锥表面积计算';
                    show(el.hint);
                    break;
                case 1:
                    el.title.textContent = '步骤1：计算底面积';
                    setFaceFill(el.faceBase, '#2e7d32', '0.20');
                    show(el.rightBase);
                    show(el.formula);
                    setText(el.formula, 'S​底 = 6 × 2 = 12');
                    break;
                case 2:
                    el.title.textContent = '步骤2：计算侧面△PAB面积';
                    setFaceFill(el.faceBase, '#2e7d32', '0.15');
                    setFaceFill(el.facePab, '#1565c0', '0.20');
                    show(el.rightPab);
                    show(el.formula);
                    setText(el.formula, 'S​△PAB = ½ × 6 × 2 = 6');
                    break;
                case 3:
                    el.title.textContent = '步骤3：计算侧面△PAD面积';
                    setFaceFill(el.faceBase, '#2e7d32', '0.15');
                    setFaceFill(el.facePab, '#2e7d32', '0.15');
                    setFaceFill(el.facePda, '#1565c0', '0.20');
                    show(el.rightPad);
                    show(el.labelPa);
                    show(el.raA);
                    show(el.formula);
                    setText(el.formula, 'PA = √(2²+3²) = √13    S​△PAD = ½ × 2 × √13 = √13');
                    break;
                case 4:
                    el.title.textContent = '步骤4：计算侧面△PBC面积';
                    setFaceFill(el.faceBase, '#2e7d32', '0.15');
                    setFaceFill(el.facePab, '#2e7d32', '0.15');
                    setFaceFill(el.facePda, '#2e7d32', '0.15');
                    setFaceFill(el.facePbc, '#1565c0', '0.20');
                    show(el.rightPbc);
                    show(el.labelPa);
                    show(el.raA);
                    show(el.labelPb);
                    show(el.raB);
                    show(el.formula);
                    setText(el.formula, 'S​△PBC = ½ × 2 × √13 = √13');
                    break;
                case 5:
                    el.title.textContent = '步骤5：计算侧面△PCD面积';
                    setFaceFill(el.faceBase, '#2e7d32', '0.15');
                    setFaceFill(el.facePab, '#2e7d32', '0.15');
                    setFaceFill(el.facePda, '#2e7d32', '0.15');
                    setFaceFill(el.facePbc, '#2e7d32', '0.15');
                    setFaceFill(el.facePcd, '#1565c0', '0.20');
                    show(el.linePm); show(el.lineOm); show(el.ptM); show(el.labelM);
                    show(el.labelPa); show(el.raA);
                    show(el.labelPb); show(el.raB);
                    show(el.labelPc); show(el.labelPm); show(el.raM);
                    show(el.rightPcd);
                    show(el.formula);
                    setText(el.formula, 'PM = √(2²+2²) = 2√2    S​△PCD = ½ × 6 × 2√2 = 6√2');
                    break;
                case 6:
                    el.title.textContent = '步骤6：表面积汇总';
                    setFaceFill(el.faceBase, '#2e7d32', '0.20');
                    setFaceFill(el.facePab, '#1565c0', '0.20');
                    setFaceFill(el.facePda, '#c62828', '0.15');
                    setFaceFill(el.facePbc, '#ff8f00', '0.15');
                    setFaceFill(el.facePcd, '#6a1b9a', '0.15');
                    show(el.linePm); show(el.lineOm); show(el.ptM); show(el.labelM);
                    show(el.labelPa); show(el.raA);
                    show(el.labelPb); show(el.raB);
                    show(el.labelPc); show(el.labelPm); show(el.raM);
                    show(el.summary);
                    break;
            }
        }

        el.btnPrev.addEventListener('click', function() {
            if(currentStep > 0) { currentStep--; renderStep(currentStep); }
        });
        el.btnNext.addEventListener('click', function() {
            if(currentStep < maxStep) { currentStep++; renderStep(currentStep); }
        });
        el.btnReset.addEventListener('click', function() {
            currentStep = 0; renderStep(currentStep);
        });

        renderStep(0);
    })();
    </script>
