    <script>
    (function() {
        // Scene 3 实现 - 导体棒下滑能量与电量

        let currentStep = 0;
        const totalSteps = 6;

        const els = {
            energyTitle: document.getElementById('scene3-energy-title'),
            arrowEp: document.getElementById('scene3-arrow-ep'),
            labelDep: document.getElementById('scene3-label-dep'),
            labelMgsin: document.getElementById('scene3-label-mgsin'),
            labelEq: document.getElementById('scene3-label-eq'),
            arrowEk: document.getElementById('scene3-arrow-ek'),
            labelDek: document.getElementById('scene3-label-dek'),
            labelMv2: document.getElementById('scene3-label-mv2'),
            labelPlus: document.getElementById('scene3-label-plus'),
            arrowQ: document.getElementById('scene3-arrow-q'),
            labelQelec: document.getElementById('scene3-label-qelec'),
            label15j: document.getElementById('scene3-label-15j'),
            fluxShade: document.getElementById('scene3-flux-shade'),
            labelDeltaPhi: document.getElementById('scene3-label-delta-phi'),
            formulaPanel: document.getElementById('scene3-formula-panel'),
            formulaEnergy: document.getElementById('scene3-formula-energy'),
            formulaCharge: document.getElementById('scene3-formula-charge'),
            formulaResult: document.getElementById('scene3-formula-result'),
            resultQ: document.getElementById('scene3-result-q'),
            btnPrev: document.getElementById('scene3-btn-prev'),
            btnNext: document.getElementById('scene3-btn-next'),
            stepIndicator: document.getElementById('scene3-step-indicator')
        };

        function show(el) {
            if (!el) return;
            el.classList.remove('scene3-hidden');
            el.classList.add('scene3-visible');
        }
        function hide(el) {
            if (!el) return;
            el.classList.remove('scene3-visible');
            el.classList.add('scene3-hidden');
        }
        function showPanel() {
            if (els.formulaPanel) els.formulaPanel.style.display = 'flex';
        }
        function hidePanel() {
            if (els.formulaPanel) els.formulaPanel.style.display = 'none';
        }
        function showFormula(contentEl) {
            [els.formulaEnergy, els.formulaCharge, els.formulaResult].forEach(function(el) {
                if (el) el.classList.remove('active');
            });
            if (contentEl) contentEl.classList.add('active');
        }

        function updateScene(step) {
            currentStep = step;
            els.stepIndicator.textContent = '步骤 ' + (step + 1) + '/' + totalSteps;
            els.btnPrev.disabled = (step === 0);
            els.btnNext.disabled = (step === totalSteps - 1);

            // 隐藏所有动态元素
            hide(els.energyTitle);
            hide(els.arrowEp);
            hide(els.labelDep);
            hide(els.labelMgsin);
            hide(els.labelEq);
            hide(els.arrowEk);
            hide(els.labelDek);
            hide(els.labelMv2);
            hide(els.labelPlus);
            hide(els.arrowQ);
            hide(els.labelQelec);
            hide(els.label15j);
            hide(els.fluxShade);
            hide(els.labelDeltaPhi);
            hidePanel();

            if (els.resultQ) {
                els.resultQ.style.transform = 'scale(1)';
                els.resultQ.style.opacity = '1';
            }

            if (step >= 1) {
                show(els.energyTitle);
                show(els.arrowEp);
                show(els.labelDep);
                show(els.labelMgsin);
                show(els.labelEq);
                show(els.arrowEk);
                show(els.labelDek);
                show(els.labelMv2);
                show(els.labelPlus);
                show(els.arrowQ);
                show(els.labelQelec);
                show(els.label15j);
            }

            if (step >= 2) {
                showPanel();
                showFormula(els.formulaEnergy);
            }

            if (step >= 3) {
                show(els.fluxShade);
                show(els.labelDeltaPhi);
            }

            if (step >= 4) {
                showPanel();
                showFormula(els.formulaCharge);
            }

            if (step >= 5) {
                showPanel();
                showFormula(els.formulaResult);

                if (els.resultQ) {
                    els.resultQ.style.transform = 'scale(0.8)';
                    els.resultQ.style.opacity = '0.5';
                    requestAnimationFrame(function() {
                        setTimeout(function() {
                            if (els.resultQ) {
                                els.resultQ.style.transform = 'scale(1)';
                                els.resultQ.style.opacity = '1';
                            }
                        }, 50);
                    });
                }
            }
        }

        els.btnPrev.addEventListener('click', function() {
            if (currentStep > 0) updateScene(currentStep - 1);
        });

        els.btnNext.addEventListener('click', function() {
            if (currentStep < totalSteps - 1) updateScene(currentStep + 1);
        });

        updateScene(0);
    })();
    </script>
