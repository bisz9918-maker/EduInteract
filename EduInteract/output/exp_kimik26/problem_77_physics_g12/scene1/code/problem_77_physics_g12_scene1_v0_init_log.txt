    <script>
    (function() {
        // Scene 1 实现
        var currentStep = 0;
        var totalSteps = 5;

        var btnPrev = document.getElementById('scene1-btn-prev');
        var btnNext = document.getElementById('scene1-btn-next');
        var stepIndicator = document.getElementById('scene1-step-indicator');

        var groupS1 = document.getElementById('scene1-group-s1');
        var groupS2 = document.getElementById('scene1-group-s2');
        var groupS3 = document.getElementById('scene1-group-s3');
        var groupS4 = document.getElementById('scene1-group-s4');

        function show(el) {
            if (!el) return;
            el.style.display = 'inline';
            // 强制重排后设置opacity实现淡入
            requestAnimationFrame(function() {
                el.style.opacity = '1';
            });
        }

        function hide(el) {
            if (!el) return;
            el.style.opacity = '0';
            // 等待过渡完成后隐藏
            setTimeout(function() {
                if (el.style.opacity === '0') {
                    el.style.display = 'none';
                }
            }, 200);
        }

        function renderStep() {
            // 更新步骤指示器
            stepIndicator.textContent = '步骤 ' + (currentStep + 1) + '/' + totalSteps;

            // 更新按钮状态
            btnPrev.disabled = (currentStep === 0);
            btnNext.disabled = (currentStep === totalSteps - 1);

            // 重置所有步骤专属元素
            hide(groupS1);
            hide(groupS2);
            hide(groupS3);
            hide(groupS4);

            // 根据当前步骤显示对应元素
            if (currentStep >= 1) {
                show(groupS1);
            }
            if (currentStep === 2) {
                show(groupS2);
            }
            if (currentStep === 3) {
                show(groupS3);
            }
            if (currentStep === 4) {
                show(groupS1); // 步骤4保留波形
                show(groupS4);
            }
        }

        btnPrev.addEventListener('click', function() {
            if (currentStep > 0) {
                currentStep--;
                renderStep();
            }
        });

        btnNext.addEventListener('click', function() {
            if (currentStep < totalSteps - 1) {
                currentStep++;
                renderStep();
            }
        });

        // 初始渲染
        renderStep();
    })();
    </script>
