    <script>
    (function() {
        // Scene 1 实现
        let currentStep = 0;
        const totalSteps = 4;

        const btnPrev = document.getElementById('scene1-btn-prev');
        const btnNext = document.getElementById('scene1-btn-next');
        const stepTitle = document.getElementById('scene1-step-title');
        const arcAb = document.getElementById('scene1-arc-ab');

        const stepTitles = [
            '步骤0：初始状态',
            '步骤1：摆动轨迹',
            '步骤2：能量守恒',
            '步骤3：受力分析',
            '步骤4：计算结果'
        ];

        function showStep(n) {
            currentStep = n;
            stepTitle.textContent = stepTitles[n];

            // 按钮状态
            btnPrev.disabled = (n === 0);
            btnNext.disabled = (n === totalSteps);

            // 获取各步骤组
            const g0 = document.getElementById('scene1-step0-group');
            const g1 = document.getElementById('scene1-step1-group');
            const g2 = document.getElementById('scene1-step2-group');
            const g3 = document.getElementById('scene1-step3-group');
            const g4 = document.getElementById('scene1-step4-group');

            // 步骤0：显示A点，隐藏B点及后续
            if (n === 0) {
                g0.style.display = 'block';
                g1.style.display = 'none';
                g2.style.display = 'none';
                g3.style.display = 'none';
                g4.style.display = 'none';
                // 重置圆弧动画
                arcAb.classList.remove('scene1-arc-draw');
            }
            // 步骤1：隐藏A点，显示B点、圆弧
            else if (n === 1) {
                g0.style.display = 'none';
                g1.style.display = 'block';
                g2.style.display = 'none';
                g3.style.display = 'none';
                g4.style.display = 'none';
                // 触发圆弧描边动画
                arcAb.classList.remove('scene1-arc-draw');
                void arcAb.offsetWidth; // 强制重绘以重启动画
                arcAb.classList.add('scene1-arc-draw');
            }
            // 步骤2：在步骤1基础上增加能量守恒
            else if (n === 2) {
                g0.style.display = 'none';
                g1.style.display = 'block';
                g2.style.display = 'block';
                g3.style.display = 'none';
                g4.style.display = 'none';
            }
            // 步骤3：在步骤2基础上增加受力分析
            else if (n === 3) {
                g0.style.display = 'none';
                g1.style.display = 'block';
                g2.style.display = 'block';
                g3.style.display = 'block';
                g4.style.display = 'none';
            }
            // 步骤4：在步骤3基础上增加计算结果
            else if (n === 4) {
                g0.style.display = 'none';
                g1.style.display = 'block';
                g2.style.display = 'block';
                g3.style.display = 'block';
                g4.style.display = 'block';
            }
        }

        btnPrev.addEventListener('click', function() {
            if (currentStep > 0) {
                showStep(currentStep - 1);
            }
        });

        btnNext.addEventListener('click', function() {
            if (currentStep < totalSteps) {
                showStep(currentStep + 1);
            }
        });

        // 初始化
        showStep(0);
    })();
    </script>
