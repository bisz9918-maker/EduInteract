    <script>
    (function() {
        // Scene 1 实现
        const STEP_COUNT = 5;
        let currentStep = 0;

        const btnPrev = document.getElementById('scene1-btn-prev');
        const btnNext = document.getElementById('scene1-btn-next');
        const stepIndicator = document.getElementById('scene1-step-indicator');

        const blockA = document.getElementById('scene1-block-a');
        const blockB = document.getElementById('scene1-block-b');
        const forcesA = document.getElementById('scene1-forces-a');
        const forcesB = document.getElementById('scene1-forces-b');
        const equations = document.getElementById('scene1-equations');
        const conclusion = document.getElementById('scene1-conclusion');

        function setGroupState(el, visible, opacityVal) {
            if (visible) {
                el.setAttribute('display', 'block');
                el.setAttribute('opacity', String(opacityVal !== undefined ? opacityVal : 1));
            } else {
                el.setAttribute('display', 'none');
                el.setAttribute('opacity', '0');
            }
        }

        function updateUI() {
            stepIndicator.textContent = '步骤 ' + (currentStep + 1) + '/' + STEP_COUNT;
            btnPrev.disabled = currentStep === 0;
            btnNext.disabled = currentStep === STEP_COUNT - 1;

            // 重置方块样式
            blockA.setAttribute('fill', 'rgba(21,101,192,0.2)');
            blockA.setAttribute('stroke-width', '2');
            blockB.setAttribute('fill', 'rgba(198,40,40,0.2)');
            blockB.setAttribute('stroke-width', '2');

            // 默认隐藏所有动态元素
            setGroupState(forcesA, false);
            setGroupState(forcesB, false);
            setGroupState(equations, false);
            setGroupState(conclusion, false);

            if (currentStep >= 1) {
                setGroupState(forcesA, true, 1);
            }
            if (currentStep >= 2) {
                setGroupState(forcesB, true, 1);
            }
            if (currentStep === 2) {
                setGroupState(forcesA, true, 0.4);
            }
            if (currentStep >= 3) {
                setGroupState(forcesA, true, 0.3);
                setGroupState(forcesB, true, 0.3);
                setGroupState(equations, true, 1);
            }
            if (currentStep === 4) {
                setGroupState(equations, true, 0.5);
                setGroupState(conclusion, true, 1);
            }

            // 高亮处理
            if (currentStep === 1) {
                blockA.setAttribute('fill', 'rgba(21,101,192,0.5)');
                blockA.setAttribute('stroke-width', '3');
            }
            if (currentStep === 2) {
                blockB.setAttribute('fill', 'rgba(198,40,40,0.5)');
                blockB.setAttribute('stroke-width', '3');
            }
        }

        btnPrev.addEventListener('click', function() {
            if (currentStep > 0) {
                currentStep--;
                updateUI();
            }
        });

        btnNext.addEventListener('click', function() {
            if (currentStep < STEP_COUNT - 1) {
                currentStep++;
                updateUI();
            }
        });

        // 初始化
        updateUI();
    })();
    </script>
