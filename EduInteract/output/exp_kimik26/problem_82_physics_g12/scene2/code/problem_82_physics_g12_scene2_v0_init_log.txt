      if (s.cx !== undefined) el.setAttribute('cx', s.cx);
      if (s.cy !== undefined) el.setAttribute('cy', s.cy);

      if (s.y2 !== undefined) el.setAttribute('y2', s.y2);

        if (s.dash === 'none') el.removeAttribute('stroke-dasharray');
        else el.setAttribute('stroke-dasharray', s.dash);
      }
    }

    document.getElementById('scene2-step-label').textContent = 'Step ' + step + ' / ' + totalSteps;
    document.getElementById('scene2-btn-prev').disabled = (step === 0);
    document.getElementById('scene2-btn-next').disabled = (step === totalSteps);
  }

    if (currentStep > 0) {
      currentStep--;
      render(currentStep);
    }
  });

  document.getElementById('scene2-btn-next').addEventListener('click', function() {
    if (currentStep < totalSteps) {
      currentStep++;
      render(currentStep);
    }
  });

  // 初始渲染 Step 0
  render(0);
})();
