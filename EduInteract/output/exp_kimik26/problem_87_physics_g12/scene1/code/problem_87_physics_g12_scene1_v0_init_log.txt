<script>
(function() {
    var currentStep = 0;
    var totalSteps = 4;
    var isAnimating = false;
    
    var ballA = document.getElementById('scene1-ball-a');
    var ballB = document.getElementById('scene1-ball-b');
    var formulaPanel = document.getElementById('scene1-formula-panel');
    var prevBtn = document.getElementById('scene1-prev-btn');
    var nextBtn = document.getElementById('scene1-next-btn');
    var dots = document.querySelectorAll('.scene1-dot');
    
    var step1Group = document.getElementById('scene1-step1-forces');
    var step2Group = document.getElementById('scene1-step2-decomp');
    var step3Group = document.getElementById('scene1-step3-forces');
    var step4Group = document.getElementById('scene1-step4-highlight');
    
    var formulas = [
        '<p style="margin:0 0 8px 0;"><b>已知条件</b></p>' +
        '<p style="margin:4px 0;">光滑绝缘斜面，θ = 37°</p>' +
        '<p style="margin:4px 0;">匀强电场 <i>E</i> 水平向右</p>' +
        '<p style="margin:4px 0;">A、B 质量均为 <i>m</i>，电量大小均为 <i>q</i></p>' +
        '<p style="margin:4px 0;">A 带负电，B 固定</p>' +
        '<p style="margin:4px 0;">间距 <i>L</i></p>' +
        '<p style="margin:4px 0;">sin37° = 0.6，cos37° = 0.8</p>',
        
        '<p style="margin:0 0 8px 0;"><b>A 球受力与电性判断</b></p>' +
        '<p style="margin:4px 0;">重力 <i>mg</i>（竖直向下）</p>' +
        '<p style="margin:4px 0;">支持力 <i>N<sub>A</sub></i>（垂直斜面向上）</p>' +
        '<p style="margin:4px 0;">电场力 <i>qE</i>（水平向左）</p>' +
        '<p style="margin:4px 0; color:#888;">假设 B 带正电：F′ 沿斜面向下</p>' +
        '<p style="margin:4px 0; color:#888;">→ 三力同向，无法平衡</p>' +
        '<p style="margin:4px 0; color:#7b1fa2;">∴ B 带负电，<i>F<sub>c</sub></i> 沿斜面向上</p>',
        
        '<p style="margin:0 0 8px 0;"><b>A 球平衡方程</b></p>' +
        '<p style="margin:4px 0;">沿斜面方向：</p>' +
        '<p style="margin:4px 0; color:#2e7d32; font-weight:bold;"><i>kq²/L² = mg·sin37° + qE·cos37°</i></p>' +
        '<p style="margin:4px 0;"><i>kq²/L² = 0.6mg + 0.8qE</i></p>' +
        '<p style="margin:4px 0;">解得：</p>' +
        '<p style="margin:4px 0; color:#2e7d32; font-weight:bold;"><i>E = 5kq/(4L²) − 3mg/(4q)</i></p>',
        
        '<p style="margin:0 0 8px 0;"><b>B 球受力分析</b></p>' +
        '<p style="margin:4px 0;">重力 <i>mg</i>（竖直向下）</p>' +
        '<p style="margin:4px 0;">支持力 <i>N<sub>B</sub></i>（垂直斜面向上）</p>' +
        '<p style="margin:4px 0;">电场力 <i>qE</i>（水平向左）</p>' +
        '<p style="margin:4px 0; color:#7b1fa2;">库仑力 <i>F<sub>c</sub> = kq²/L²</i>（沿斜面向下）</p>' +
        '<p style="margin:4px 0;">A 球受力淡化作参照</p>',
        
        '<p style="margin:0 0 8px 0;"><b>B 球加速度</b></p>' +
        '<p style="margin:4px 0;"><i>F<sub>合</sub> = mg·sin37° + qE·cos37° + kq²/L²</i></p>' +
        '<p style="margin:4px 0;">由(1)知：</p>' +
        '<p style="margin:4px 0;"><i>mg·sin37° + qE·cos37° = kq²/L²</i></p>' +
        '<p style="margin:4px 0;">∴ <i>F<sub>合</sub> = 2kq²/L²</i></p>' +
        '<p style="margin:4px 0; color:#2e7d32; font-weight:bold;"><i>a = 2kq²/(mL²)</i></p>' +
        '<p style="margin:4px 0; color:#c62828;">方向沿斜面向下</p>'
    ];
    
    function updateDots() {
        dots.forEach(function(dot, idx) {
            dot.classList.remove('active', 'completed');
            if (idx === currentStep) {
                dot.classList.add('active');
            } else if (idx < currentStep) {
                dot.classList.add('completed');
            }
        });
    }
    
    function updateButtons() {
        prevBtn.disabled = (currentStep === 0);
        nextBtn.disabled = (currentStep === totalSteps);
    }
    
    function updateBalls() {
        if (currentStep >= 1 && currentStep <= 2) {
            ballA.setAttribute('fill', '#1565c0');
            ballA.setAttribute('fill-opacity', '0.2');
        } else {
            ballA.setAttribute('fill', '#ffffff');
            ballA.setAttribute('fill-opacity', '1');
        }
        
        if (currentStep >= 3) {
            ballB.setAttribute('fill', '#c62828');
            ballB.setAttribute('fill-opacity', '0.2');
        } else {
            ballB.setAttribute('fill', '#ffffff');
            ballB.setAttribute('fill-opacity', '1');
        }
    }
    
    function showGroup(el, show, opacity) {
        if (!el) return;
        if (show) {
            el.style.display = 'block';
            requestAnimationFrame(function() {
                el.style.opacity = String(opacity);
            });
        } else {
            el.style.opacity = '0';
            setTimeout(function() {
                el.style.display = 'none';
            }, 300);
        }
    }
    
    function renderStep(step) {
        formulaPanel.innerHTML = formulas[step];
        
        var s1Opacity = (step >= 3) ? 0.3 : 1;
        showGroup(step1Group, step >= 1, s1Opacity);
        
        var s2Opacity = (step >= 3) ? 0.3 : 1;
        showGroup(step2Group, step >= 2, s2Opacity);
        
        showGroup(step3Group, step >= 3, 1);
        showGroup(step4Group, step >= 4, 1);
        
        updateBalls();
        updateDots();
        updateButtons();
    }
    
    function goToStep(newStep) {
        if (newStep < 0 || newStep > totalSteps) return;
        if (isAnimating) return;
        isAnimating = true;
        
        currentStep = newStep;
        renderStep(currentStep);
        
        setTimeout(function() {
            isAnimating = false;
        }, 350);
    }
    
    prevBtn.addEventListener('click', function() {
        goToStep(currentStep - 1);
    });
    
    nextBtn.addEventListener('click', function() {
        goToStep(currentStep + 1);
    });
    
    dots.forEach(function(dot) {
        dot.addEventListener('click', function() {
            var s = parseInt(dot.getAttribute('data-step'));
            goToStep(s);
        });
    });
    
    renderStep(0);
})();
</script>
