    <script>
    (function() {
        var currentStep = 0;
        var totalSteps = 4;
        
        var btnPrev = document.getElementById('scene2-btn-prev');
        var btnNext = document.getElementById('scene2-btn-next');
        var indicator = document.getElementById('scene2-step-indicator');
        
        var circuitGroup = document.getElementById('scene2-circuit-group');
        var step0Title = document.getElementById('scene2-step0-title');
        var formulaStep1 = document.getElementById('scene2-formula-step1');
        var graphLabels = document.getElementById('scene2-graph-labels');
        var cosineCurve = document.getElementById('scene2-cosine-curve');
        var integralFill = document.getElementById('scene2-integral-fill');
        var hintStep0 = document.getElementById('scene2-hint-step0');
        var hintStep1 = document.getElementById('scene2-hint-step1');
        var deriveStep2 = document.getElementById('scene2-derive-step2');
        var deriveStep3 = document.getElementById('scene2-derive-step3');
        
        function show(el) {
            if (el) el.classList.remove('scene2-hide');
        }
        function hide(el) {
            if (el) el.classList.add('scene2-hide');
        }
        
        function updateStep(step) {
            currentStep = step;
            indicator.textContent = '步骤 ' + (step + 1) + '/' + totalSteps;
            
            btnPrev.disabled = (step === 0);
            if (step === totalSteps - 1) {
                btnNext.textContent = '完成';
                btnNext.disabled = true;
            } else {
                btnNext.textContent = '下一步 ▶';
                btnNext.disabled = false;
            }
            
            // 步骤0标题：仅步骤0显示
            if (step === 0) {
                show(step0Title);
                step0Title.classList.remove('scene2-dimmed');
            } else {
                hide(step0Title);
                step0Title.classList.remove('scene2-dimmed');
            }
            
            // 电路图：步骤3淡化
            if (step === 3) {
                circuitGroup.classList.add('scene2-dimmed');
            } else {
                circuitGroup.classList.remove('scene2-dimmed');
            }
            
            // 步骤1公式：步骤1-2显示，步骤3淡化
            if (step >= 1) {
                show(formulaStep1);
                if (step === 3) {
                    formulaStep1.classList.add('scene2-dimmed');
                } else {
                    formulaStep1.classList.remove('scene2-dimmed');
                }
            } else {
                hide(formulaStep1);
                formulaStep1.classList.remove('scene2-dimmed');
            }
            
            // 坐标轴标签：步骤1+显示
            if (step >= 1) {
                show(graphLabels);
            } else {
                hide(graphLabels);
            }
            
            // 余弦曲线：步骤2+显示
            if (step >= 2) {
                show(cosineCurve);
            } else {
                hide(cosineCurve);
            }
            
            // 积分填充：步骤3显示
            if (step >= 3) {
                show(integralFill);
            } else {
                hide(integralFill);
            }
            
            // 右下提示/推导
            if (step === 0) {
                show(hintStep0);
                hide(hintStep1);
                hide(deriveStep2);
                hide(deriveStep3);
            } else if (step === 1) {
                hide(hintStep0);
                show(hintStep1);
                hide(deriveStep2);
                hide(deriveStep3);
            } else if (step === 2) {
                hide(hintStep0);
                hide(hintStep1);
                show(deriveStep2);
                hide(deriveStep3);
            } else if (step === 3) {
                hide(hintStep0);
                hide(hintStep1);
                hide(deriveStep2);
                show(deriveStep3);
            }
        }
        
        btnPrev.addEventListener('click', function() {
            if (currentStep > 0) {
                updateStep(currentStep - 1);
            }
        });
        
        btnNext.addEventListener('click', function() {
            if (currentStep < totalSteps - 1) {
                updateStep(currentStep + 1);
            }
        });
        
        // 初始化
        updateStep(0);
    })();
    </script>
