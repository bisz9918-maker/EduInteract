    <script>
    (function() {
        // Generate parabolic trajectory points: x=220+240t, y=140+280t^2, t in [0,1]
        const trajectory = document.getElementById('scene2-trajectory');
        const points = [];
        for (let i = 0; i <= 20; i++) {
            const t = i / 20;
            const x = 220 + 240 * t;
            const y = 140 + 280 * t * t;
            points.push(x + ',' + y);
        }
        trajectory.setAttribute('points', points.join(' '));
        
        // Step configuration
        let currentStep = 0;
        const stepNames = ['初态', '位移分解', '速度分解', '联立求解'];
        
        // Elements visible from step 0 (full opacity, except F0 handled separately)
        const step0Elements = [
            'scene2-axis-x', 'scene2-axis-y', 'scene2-origin',
            'scene2-label-o', 'scene2-label-x', 'scene2-label-y',
            'scene2-arrow-v1', 'scene2-label-v1',
            'scene2-arrow-a2', 'scene2-label-a2'
        ];
        
        // Elements added in step 1
        const step1Elements = [
            'scene2-trajectory', 'scene2-point-q', 'scene2-label-q',
            'scene2-rect-displacement',
            'scene2-arrow-d', 'scene2-label-d',
            'scene2-arrow-h', 'scene2-label-h',
            'scene2-formula-f1a', 'scene2-formula-f1b'
        ];
        
        // Elements added in step 2
        const step2Elements = [
            'scene2-arrow-vx', 'scene2-label-vx',
            'scene2-arrow-vy', 'scene2-label-vy',
            'scene2-arrow-vq', 'scene2-label-vq',
            'scene2-triangle-velocity',
            'scene2-formula-f2a', 'scene2-formula-f2b'
        ];
        
        // Elements added in step 3
        const step3Elements = [
            'scene2-formula-f3a', 'scene2-formula-f3b', 'scene2-formula-f3c',
            'scene2-formula-f3d', 'scene2-formula-f3e',
            'scene2-sub-arrow1', 'scene2-sub-arrow2', 'scene2-sub-arrow3'
        ];
        
        // Elements to dim in step 3
        const step3DimElements = [
            'scene2-formula-f0', 'scene2-formula-f1a', 'scene2-formula-f1b',
            'scene2-formula-f2a', 'scene2-formula-f2b'
        ];
        
        function setOpacity(id, value) {
            const el = document.getElementById(id);
            if (el) el.style.opacity = value;
        }
        
        function updateStep() {
            // Update controls
            document.getElementById('scene2-btn-prev').disabled = (currentStep === 0);
            document.getElementById('scene2-btn-next').disabled = (currentStep === 3);
            document.getElementById('scene2-step-info').textContent = stepNames[currentStep];
            
            // Step 0 base elements always at full opacity
            step0Elements.forEach(id => setOpacity(id, 1));
            
            // Step 1 elements
            step1Elements.forEach(id => setOpacity(id, currentStep >= 1 ? 1 : 0));
            
            // Step 2 elements
            step2Elements.forEach(id => setOpacity(id, currentStep >= 2 ? 1 : 0));
            
            // Step 3 elements and dimming
            if (currentStep === 3) {
                step3Elements.forEach(id => setOpacity(id, 1));
                step3DimElements.forEach(id => setOpacity(id, 0.3));
            } else {
                step3Elements.forEach(id => setOpacity(id, 0));
                // Restore dimmed elements to full opacity if they should be visible
                setOpacity('scene2-formula-f0', 1);
                setOpacity('scene2-formula-f1a', currentStep >= 1 ? 1 : 0);
                setOpacity('scene2-formula-f1b', currentStep >= 1 ? 1 : 0);
                setOpacity('scene2-formula-f2a', currentStep >= 2 ? 1 : 0);
                setOpacity('scene2-formula-f2b', currentStep >= 2 ? 1 : 0);
            }
        }
        
        document.getElementById('scene2-btn-prev').addEventListener('click', function() {
            if (currentStep > 0) {
                currentStep--;
                updateStep();
            }
        });
        
        document.getElementById('scene2-btn-next').addEventListener('click', function() {
            if (currentStep < 3) {
                currentStep++;
                updateStep();
            }
        });
        
        // Initialize
        updateStep();
    })();
    </script>
