<script>
(function() {
  const sqrt3 = Math.sqrt(3);
  const Py = 420 - 100 * sqrt3; // y坐标恒定 ≈246.79

  const state = {
    x: 2,
    showTrack: false,
    trailPoints: []
  };

  const els = {
    slider: document.getElementById('scene2-slider'),
    btnShow: document.getElementById('scene2-btn-show'),
    btnClear: document.getElementById('scene2-btn-clear'),
    triAcd: document.getElementById('scene2-tri-acd'),
    triBce: document.getElementById('scene2-tri-bce'),
    lineDe: document.getElementById('scene2-line-de'),
    pointC: document.getElementById('scene2-point-c'),
    pointD: document.getElementById('scene2-point-d'),
    pointE: document.getElementById('scene2-point-e'),
    pointP: document.getElementById('scene2-point-p'),
    labelC: document.getElementById('scene2-label-c'),
    labelD: document.getElementById('scene2-label-d'),
    labelE: document.getElementById('scene2-label-e'),
    labelP: document.getElementById('scene2-label-p'),
    labelCoord: document.getElementById('scene2-label-coord'),
    trackLine: document.getElementById('scene2-track-line'),
    trackTrail: document.getElementById('scene2-track-trail'),
    pointP1: document.getElementById('scene2-point-p1'),
    pointP2: document.getElementById('scene2-point-p2'),
    labelP1: document.getElementById('scene2-label-p1'),
    labelP2: document.getElementById('scene2-label-p2'),
    labelPath: document.getElementById('scene2-label-path'),
    refLine: document.getElementById('scene2-ref-line')
  };

  // 设置参考虚线和轨迹线的精确y坐标
  els.refLine.setAttribute('y1', Py);
  els.refLine.setAttribute('y2', Py);
  els.trackLine.setAttribute('y1', Py);
  els.trackLine.setAttribute('y2', Py);
  els.pointP1.setAttribute('cy', Py);
  els.pointP2.setAttribute('cy', Py);
  els.pointP.setAttribute('cy', Py);

  function render() {
    const x = state.x;
    const Cx = 200 + 50 * x;
    const Dx = 200 + 25 * x;
    const Dy = 420 - 25 * x * sqrt3;
    const Ex = 400 + 25 * x;
    const Ey = 420 - (200 - 25 * x) * sqrt3;
    const Px = 300 + 25 * x;

    // 更新三角形
    els.triAcd.setAttribute('points', `200,420 ${Cx},420 ${Dx},${Dy}`);
    els.triBce.setAttribute('points', `600,420 ${Cx},420 ${Ex},${Ey}`);

    // 更新DE线段
    els.lineDe.setAttribute('x1', Dx);
    els.lineDe.setAttribute('y1', Dy);
    els.lineDe.setAttribute('x2', Ex);
    els.lineDe.setAttribute('y2', Ey);

    // 更新点坐标
    els.pointC.setAttribute('cx', Cx);
    els.pointD.setAttribute('cx', Dx);
    els.pointD.setAttribute('cy', Dy);
    els.pointE.setAttribute('cx', Ex);
    els.pointE.setAttribute('cy', Ey);
    els.pointP.setAttribute('cx', Px);

    // 更新标签位置
    els.labelC.setAttribute('x', Cx - 70);
    els.labelD.setAttribute('x', Dx - 15);
    els.labelD.setAttribute('y', Dy - 10);
    els.labelE.setAttribute('x', Ex + 10);
    els.labelE.setAttribute('y', Ey - 10);
    els.labelP.setAttribute('x', Px - 8);
    els.labelCoord.setAttribute('x', Px);

    // 坐标值标注
    const pxVal = (x / 2 + 2).toFixed(1);
    els.labelCoord.textContent = `P(${pxVal}, 2√3)`;

    // 轨迹拖尾记录
    if (state.showTrack) {
      state.trailPoints.push(`${Px},${Py}`);
      if (state.trailPoints.length > 800) state.trailPoints.shift();
      els.trackTrail.setAttribute('points', state.trailPoints.join(' '));
    }
  }

  // 滑块事件
  els.slider.addEventListener('input', function() {
    state.x = parseFloat(this.value);
    render();
  });

  // 显示/隐藏轨迹按钮
  els.btnShow.addEventListener('click', function() {
    state.showTrack = !state.showTrack;
    this.textContent = state.showTrack ? '隐藏轨迹' : '显示轨迹';
    this.classList.toggle('active', state.showTrack);

    const opacity = state.showTrack ? '1' : '0';
    els.trackLine.setAttribute('opacity', opacity);
    els.pointP1.setAttribute('opacity', opacity);
    els.pointP2.setAttribute('opacity', opacity);
    els.labelP1.setAttribute('opacity', opacity);
    els.labelP2.setAttribute('opacity', opacity);
    els.labelPath.setAttribute('opacity', opacity);

    if (state.showTrack) {
      state.trailPoints = [];
      els.trackTrail.setAttribute('points', '');
      // 立即记录当前点
      const Px = 300 + 25 * state.x;
      state.trailPoints.push(`${Px},${Py}`);
      els.trackTrail.setAttribute('points', state.trailPoints.join(' '));
    }
  });

  // 清除轨迹按钮
  els.btnClear.addEventListener('click', function() {
    state.trailPoints = [];
    els.trackTrail.setAttribute('points', '');
  });

  // 初始渲染
  render();
})();
</script>
