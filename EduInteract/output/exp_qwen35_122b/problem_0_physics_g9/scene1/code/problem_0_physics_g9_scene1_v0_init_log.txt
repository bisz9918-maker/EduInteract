    <script>
    (function() {
        // Scene 1 实现 - 杠杆平衡条件动态演示
        
        // 配置参数
        const CONFIG = {
            adultX: 210,           // 大人初始 x 坐标
            childX: 605,           // 小孩初始 x 坐标
            fulcrumX: 480,         // 支点 x 坐标
            moveDistance: 40,      // 移动距离
            adultForce: 600,       // 大人重力 (N) - 假设值
            childForce: 480,       // 小孩重力 (N) - 假设值
            adultArm: 270,         // 大人初始力臂 (mm)
            childArm: 125,         // 小孩初始力臂 (mm)
            tiltAngle: 5           // 倾斜角度 (度)
        };
        
        // 状态变量
        let currentStep = 0;
        let adultX = CONFIG.adultX;
        let childX = CONFIG.childX;
        let isAnimating = false;
        
        // DOM 元素引用
        const svg = document.getElementById('scene1-svg');
        const adultGroup = document.getElementById('scene1-adult');
        const childGroup = document.getElementById('scene1-child');
        const gravityAdultLine = document.getElementById('scene1-gravity-adult');
        const gravityAdultArrow = document.getElementById('scene1-gravity-adult-arrow');
        const gravityChildLine = document.getElementById('scene1-gravity-child');
        const gravityChildArrow = document.getElementById('scene1-gravity-child-arrow');
        const armAdult = document.getElementById('scene1-arm-adult');
        const armChild = document.getElementById('scene1-arm-child');
        const arrowAdult = document.getElementById('scene1-arrow-adult');
        const arrowAdultHead = document.getElementById('scene1-arrow-adult-head');
        const arrowChild = document.getElementById('scene1-arrow-child');
        const arrowChildHead = document.getElementById('scene1-arrow-child-head');
        const deltaL = document.getElementById('scene1-deltaL');
        const torqueTextAdult = document.getElementById('scene1-torque-text-adult');
        const torqueTextChild = document.getElementById('scene1-torque-text-child');
        const highlightBox = document.getElementById('scene1-highlight-box');
        const compare = document.getElementById('scene1-compare');
        const tiltedLever = document.getElementById('scene1-tilted-lever');
        const sinkArrow = document.getElementById('scene1-sink-arrow');
        const sinkArrowHead = document.getElementById('scene1-sink-arrow-head');
        const hint = document.getElementById('scene1-hint');
        const conclusion = document.getElementById('scene1-conclusion');
        const answer = document.getElementById('scene1-answer');
        const btnPrev = document.getElementById('scene1-btn-prev');
        const btnNext = document.getElementById('scene1-btn-next');
        const stepIndicator = document.getElementById('scene1-step-indicator');
        
        // 工具函数：显示元素
        function show(el) {
            el.style.opacity = '1';
            el.style.display = 'block';
        }
        
        // 工具函数：隐藏元素
        function hide(el) {
            el.style.opacity = '0';
            if (el.tagName === 'text' || el.tagName === 'line' || el.tagName === 'polygon' || el.tagName === 'rect') {
                el.style.display = 'block'; // SVG 元素保持 display
            }
        }
        
        // 更新场景显示
        function updateScene() {
            // 更新步骤指示器
            stepIndicator.textContent = `步骤 ${currentStep + 1}/5`;
            
            // 更新按钮状态
            btnPrev.disabled = (currentStep === 0);
            btnNext.disabled = (currentStep === 4);
            
            // 根据步骤更新显示
            switch(currentStep) {
                case 0: // 初始平衡状态
                    // 隐藏移动相关元素
                    hide(arrowAdult); hide(arrowAdultHead);
                    hide(arrowChild); hide(arrowChildHead);
                    hide(deltaL);
                    hide(hint);
                    hide(highlightBox);
                    hide(compare);
                    hide(tiltedLever);
                    hide(sinkArrow); hide(sinkArrowHead);
                    hide(conclusion);
                    hide(answer);
                    
                    // 重置位置
                    adultX = CONFIG.adultX;
                    childX = CONFIG.childX;
                    updatePersonPositions();
                    
                    // 力矩显示初始值
                    torqueTextAdult.textContent = `M<sub>大人</sub> = G<sub>大人</sub> × L<sub>大人</sub> = 600 N·m`;
                    torqueTextChild.textContent = `M<sub>小孩</sub> = G<sub>小孩</sub> × L<sub>小孩</sub> = 600 N·m`;
                    
                    // 杠杆保持水平
                    tiltedLever.setAttribute('opacity', '0');
                    document.getElementById('scene1-lever').setAttribute('opacity', '1');
                    break;
                    
                case 1: // 开始移动
                    // 显示移动箭头和ΔL
                    show(arrowAdult); show(arrowAdultHead);
                    show(arrowChild); show(arrowChildHead);
                    show(deltaL);
                    show(hint);
                    hint.textContent = '两人同时以相等速度向支点移动 ΔL';
                    
                    hide(highlightBox);
                    hide(compare);
                    hide(tiltedLever);
                    hide(sinkArrow); hide(sinkArrowHead);
                    hide(conclusion);
                    hide(answer);
                    
                    // 力矩仍显示初始值
                    torqueTextAdult.textContent = `M<sub>大人</sub> = G<sub>大人</sub> × L<sub>大人</sub> = 600 N·m`;
                    torqueTextChild.textContent = `M<sub>小孩</sub> = G<sub>小孩</sub> × L<sub>小孩</sub> = 600 N·m`;
                    
                    document.getElementById('scene1-lever').setAttribute('opacity', '1');
                    break;
                    
                case 2: // 移动后状态
                    // 移动人物
                    adultX = CONFIG.adultX + CONFIG.moveDistance;
                    childX = CONFIG.childX - CONFIG.moveDistance;
                    updatePersonPositions();
                    
                    // 更新力臂
                    const newAdultArm = CONFIG.adultArm - CONFIG.moveDistance;
                    const newChildArm = CONFIG.childArm - CONFIG.moveDistance;
                    armAdult.setAttribute('x2', adultX);
                    armChild.setAttribute('x1', CONFIG.fulcrumX);
                    armChild.setAttribute('x2', childX);
                    
                    // 更新力矩标签位置
                    document.getElementById('scene1-label-l-adult').setAttribute('x', (adultX + CONFIG.fulcrumX) / 2);
                    document.getElementById('scene1-label-l-child').setAttribute('x', (CONFIG.fulcrumX + childX) / 2);
                    
                    // 计算新力矩
                    const newAdultTorque = CONFIG.adultForce * (newAdultArm / CONFIG.adultArm);
                    const newChildTorque = CONFIG.childForce * (newChildArm / CONFIG.childArm);
                    
                    torqueTextAdult.textContent = `M'<sub>大人</sub> = G<sub>大人</sub> × (L<sub>大人</sub> - ΔL) = ${Math.round(newAdultTorque)} N·m`;
                    torqueTextChild.textContent = `M'<sub>小孩</sub> = G<sub>小孩</sub> × (L<sub>小孩</sub> - ΔL) = ${Math.round(newChildTorque)} N·m`;
                    
                    hide(highlightBox);
                    hide(compare);
                    hide(tiltedLever);
                    hide(sinkArrow); hide(sinkArrowHead);
                    hide(conclusion);
                    hide(answer);
                    hide(hint);
                    
                    document.getElementById('scene1-lever').setAttribute('opacity', '1');
                    break;
                    
                case 3: // 力矩对比
                    // 保持移动后位置
                    updatePersonPositions();
                    
                    // 高亮小孩端力矩框
                    show(highlightBox);
                    highlightBox.setAttribute('x', '450');
                    
                    // 显示比较符号
                    show(compare);
                    
                    hint.textContent = '小孩端力矩更大';
                    show(hint);
                    
                    hide(tiltedLever);
                    hide(sinkArrow); hide(sinkArrowHead);
                    hide(conclusion);
                    hide(answer);
                    
                    document.getElementById('scene1-lever').setAttribute('opacity', '1');
                    break;
                    
                case 4: // 结论
                    // 显示倾斜杠杆
                    document.getElementById('scene1-lever').setAttribute('opacity', '0');
                    show(tiltedLever);
                    tiltedLever.setAttribute('opacity', '1');
                    tiltedLever.setAttribute('transform', `rotate(${CONFIG.tiltAngle}, 480, 250)`);
                    
                    // 显示下沉箭头
                    show(sinkArrow); show(sinkArrowHead);
                    
                    // 显示结论
                    show(conclusion);
                    show(answer);
                    
                    hide(highlightBox);
                    hide(compare);
                    hide(hint);
                    
                    // 更新力矩显示为最终值
                    const finalAdultTorque = CONFIG.adultForce * ((CONFIG.adultArm - CONFIG.moveDistance) / CONFIG.adultArm);
                    const finalChildTorque = CONFIG.childForce * ((CONFIG.childArm - CONFIG.moveDistance) / CONFIG.childArm);
                    torqueTextAdult.textContent = `M'<sub>大人</sub> = ${Math.round(finalAdultTorque)} N·m`;
                    torqueTextChild.textContent = `M'<sub>小孩</sub> = ${Math.round(finalChildTorque)} N·m`;
                    break;
            }
        }
        
        // 更新人物位置
        function updatePersonPositions() {
            // 更新大人位置
            adultGroup.setAttribute('transform', `translate(${adultX - 210}, 0)`);
            gravityAdultLine.setAttribute('x1', adultX);
            gravityAdultLine.setAttribute('x2', adultX);
            gravityAdultArrow.setAttribute('points', `${adultX},${250} ${adultX - 5},${240} ${adultX + 5},${240}`);
            document.getElementById('scene1-label-g-adult').setAttribute('x', adultX - 15);
            
            // 更新小孩位置
            childGroup.setAttribute('transform', `translate(${childX - 605}, 0)`);
            gravityChildLine.setAttribute('x1', childX);
            gravityChildLine.setAttribute('x2', childX);
            gravityChildArrow.setAttribute('points', `${childX},${250} ${childX - 5},${240} ${childX + 5},${240}`);
            document.getElementById('scene1-label-g-child').setAttribute('x', childX - 10);
        }
        
        // 事件绑定
        btnPrev.addEventListener('click', () => {
            if (currentStep > 0 && !isAnimating) {
                currentStep--;
                updateScene();
            }
        });
        
        btnNext.addEventListener('click', () => {
            if (currentStep < 4 && !isAnimating) {
                currentStep++;
                updateScene();
            }
        });
        
        // 初始化
        updateScene();
    })();
    </script>
</body>
