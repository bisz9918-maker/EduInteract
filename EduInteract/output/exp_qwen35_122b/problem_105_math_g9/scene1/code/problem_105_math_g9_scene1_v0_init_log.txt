    <script>
    (function() {
        const scene1 = {
            currentStep: 0,
            totalSteps: 5,
            
            // 元素分组映射
            stepGroups: {
                1: ['scene1-step1-group', 'scene1-fill-abc'],
                2: ['scene1-step2-group', 'scene1-fill-adf'],
                3: ['scene1-step3-group'],
                4: ['scene1-step4-group']
            },
            
            // 高亮元素映射
            highlightElements: {
                1: ['scene1-seg-ab', 'scene1-seg-ac'],
                2: ['scene1-seg-ad', 'scene1-seg-af'],
                3: ['scene1-angle-arc-dae']
            },
            
            init: function() {
                this.updateUI();
                this.bindEvents();
            },
            
            bindEvents: function() {
                document.getElementById('scene1-prev-btn').addEventListener('click', () => {
                    if (this.currentStep > 1) {
                        this.currentStep--;
                        this.updateUI();
                    }
                });
                
                document.getElementById('scene1-next-btn').addEventListener('click', () => {
                    if (this.currentStep < this.totalSteps) {
                        this.currentStep++;
                        this.updateUI();
                    }
                });
            },
            
            updateUI: function() {
                // 更新步骤指示器
                const indicator = document.getElementById('scene1-step-indicator');
                indicator.textContent = '步骤 ' + this.currentStep + '/' + this.totalSteps;
                
                // 更新按钮状态
                const prevBtn = document.getElementById('scene1-prev-btn');
                const nextBtn = document.getElementById('scene1-next-btn');
                prevBtn.disabled = this.currentStep <= 1;
                nextBtn.disabled = this.currentStep >= this.totalSteps;
                
                // 重置所有步骤组
                for (let i = 1; i <= 4; i++) {
                    const groups = this.stepGroups[i];
                    groups.forEach(groupId => {
                        const el = document.getElementById(groupId);
                        if (el) {
                            if (i <= this.currentStep) {
                                el.classList.remove('scene1-hidden');
                                el.classList.add('scene1-visible');
                            } else {
                                el.classList.add('scene1-hidden');
                                el.classList.remove('scene1-visible');
                            }
                        }
                    });
                }
                
                // 重置所有高亮
                for (let i = 1; i <= 3; i++) {
                    const elements = this.highlightElements[i];
                    elements.forEach(elemId => {
                        const el = document.getElementById(elemId);
                        if (el) {
                            if (i <= this.currentStep) {
                                if (i === 1) el.classList.add('scene1-highlight-green');
                                else if (i === 2) el.classList.add('scene1-highlight-green');
                                else if (i === 3) el.classList.add('scene1-highlight-red');
                            } else {
                                el.classList.remove('scene1-highlight-green');
                                el.classList.remove('scene1-highlight-red');
                            }
                        }
                    });
                }
                
                // 步骤 4 特殊处理：显示 DF 虚线
                const dfSeg = document.getElementById('scene1-seg-df');
                if (dfSeg) {
                    if (this.currentStep >= 4) {
                        dfSeg.classList.remove('scene1-hidden');
                        dfSeg.classList.add('scene1-visible');
                    } else {
                        dfSeg.classList.add('scene1-hidden');
                        dfSeg.classList.remove('scene1-visible');
                    }
                }
            }
        };
        
        scene1.init();
    })();
    </script>
