    <script>
    (function() {
        // Scene 2 实现
        // 布局坐标表（800×500 画布）：
        // | 元素 | 类型 | 位置/范围 | 大小 | 标签偏移方向 | 备注 |
        // |------|------|-----------|------|-------------|------|
        // | 金属板外框 | 矩形 | (220,140)-(580,360) | 360×220 | - | 实线黑色 |
        // | 列分隔线 1 | 虚线 | x=340, y=140~360 | 2×220 | - | 蓝色 |
        // | 列分隔线 2 | 虚线 | x=460, y=140~360 | 2×220 | - | 蓝色 |
        // | 行分隔线 1 | 虚线 | y=213, x=220~580 | 360×2 | - | 灰色 |
        // | 行分隔线 2 | 虚线 | y=287, x=220~580 | 360×2 | - | 灰色 |
        // | 接线柱 A 竖条 | 矩形 | (180,180)-(200,320) | 20×140 | - | 黑色填充 |
        // | 接线柱 B 竖条 | 矩形 | (600,180)-(620,320) | 20×140 | - | 黑色填充 |
        // | 接线柱 A 引线 | 线段 | (200,250)-(120,250) | 80×2 | - | 黑色 |
        // | 接线柱 B 引线 | 线段 | (600,250)-(680,250) | 80×2 | - | 黑色 |
        // | 接线柱 A 圆点 | 圆 | (110,250) | r=6 | - | 黑色 |
        // | 接线柱 B 圆点 | 圆 | (690,250) | r=6 | - | 黑色 |
        // | 挖去位置叉号 | 叉号 | (400,250) | 40×40 | - | 红色，第 2 列第 2 行中心 |
        // | 第 1 列标注 | text | (280,110) | ~40×16 | 上方 | 第 1 列上方 |
        // | 第 2 列标注 | text | (400,110) | ~40×16 | 上方 | 第 2 列上方 |
        // | 第 3 列标注 | text | (520,110) | ~40×16 | 上方 | 第 3 列上方 |
        // | A 标签 | text | (80,245) | ~16×16 | 左侧 | 接线柱 A 左侧 |
        // | B 标签 | text | (705,245) | ~16×16 | 右侧 | 接线柱 B 右侧 |
        // | r 标签 | text | (280,335) | ~12×16 | 下方 | 第 1 列第 3 行小块内（调整位置） |
        // | R'标签 | text | (400,412) | ~24×16 | - | 公式框内（调整位置） |
        // | 公式框 | 矩形 | (280,380)-(520,430) | 240×50 | - | 浅灰色背景 |
        // | 并联标注弧 1 | 弧线 | (340,250) 半径 30 | - | 左侧 | 第 2 列上下 2 块并联 |
        // | 并联标注弧 2 | 弧线 | (460,250) 半径 30 | - | 右侧 | 第 2 列上下 2 块并联 |

        const scene2 = {
            currentStep: 0,
            totalSteps: 3,
            
            elements: {
                // 步骤 1 显示的元素
                holeMark: document.getElementById('scene2-hole-mark'),
                labelR: document.getElementById('scene2-label-r'),
                
                // 步骤 2 显示的元素
                parallelArcs: document.getElementById('scene2-parallel-arcs'),
                
                // 步骤 3 显示的元素
                formulaBox: document.getElementById('scene2-formula-box'),
                calcText: document.getElementById('scene2-calc-text')
            },
            
            buttons: {
                prev: document.getElementById('scene2-btn-prev'),
                next: document.getElementById('scene2-btn-next')
            },
            
            stepDisplay: document.getElementById('scene2-step-display'),
            
            init: function() {
                this.updateDisplay();
                this.bindEvents();
            },
            
            bindEvents: function() {
                this.buttons.prev.addEventListener('click', () => {
                    if (this.currentStep > 0) {
                        this.currentStep--;
                        this.updateDisplay();
                    }
                });
                
                this.buttons.next.addEventListener('click', () => {
                    if (this.currentStep < this.totalSteps) {
                        this.currentStep++;
                        this.updateDisplay();
                    }
                });
            },
            
            updateDisplay: function() {
                // 更新步骤显示
                this.stepDisplay.textContent = '步骤 ' + (this.currentStep + 1) + '/' + this.totalSteps;
                
                // 更新按钮状态
                this.buttons.prev.disabled = (this.currentStep === 0);
                this.buttons.next.disabled = (this.currentStep === this.totalSteps);
                
                // 根据步骤显示/隐藏元素
                this.updateElementVisibility();
            },
            
            updateElementVisibility: function() {
                const el = this.elements;
                
                // 步骤 1：显示挖去标记和 r 标签
                if (this.currentStep >= 1) {
                    this.show(el.holeMark);
                    this.show(el.labelR);
                } else {
                    this.hide(el.holeMark);
                    this.hide(el.labelR);
                }
                
                // 步骤 2：显示并联标注弧
                if (this.currentStep >= 2) {
                    this.show(el.parallelArcs);
                } else {
                    this.hide(el.parallelArcs);
                }
                
                // 步骤 3：显示公式框和计算说明
                if (this.currentStep >= 3) {
                    this.show(el.formulaBox);
                    this.show(el.calcText);
                } else {
                    this.hide(el.formulaBox);
                    this.hide(el.calcText);
                }
            },
            
            show: function(el) {
                if (el) {
                    el.style.opacity = '1';
                    el.style.display = 'block';
                    el.classList.add('scene2-fade-in');
                }
            },
            
            hide: function(el) {
                if (el) {
                    el.style.opacity = '0';
                    el.style.display = 'none';
                    el.classList.remove('scene2-fade-in');
                }
            }
        };
        
        // 初始化
        scene2.init();
    })();
    </script>
