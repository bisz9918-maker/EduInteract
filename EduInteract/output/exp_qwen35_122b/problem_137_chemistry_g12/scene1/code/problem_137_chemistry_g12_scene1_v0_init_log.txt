    <script>
    (function() {
        // Scene 1 实现 - 有机化合物 A 的分子结构分析
        // 布局坐标表（参考）：
        // | 元素 | 类型 | 位置/范围 | 大小 | 标签偏移方向 | 备注 |
        // |------|------|-----------|------|-------------|------|
        // | 分子骨架主链 | 折线 | (150,250)-(670,250) | 520×4 | - | 横向链状结构 |
        // | 酯基 1 区域 | 多边形 | (90,200)-(150,250) | 60×50 | 左方 | 左侧乙酰氧基 |
        // | 酯基 2 区域 | 多边形 | (210,180)-(270,250) | 60×70 | 上方 | 中上部乙酰氧基 |
        // | 酯基 3 区域 | 多边形 | (310,320)-(370,350) | 60×30 | 下方 | 中下部乙酰氧基 |
        // | 双键 1-5 | 双线 | 见各双键坐标 | 36×4 | 上方 | 碳碳双键 |
        // | 手性碳 1-2 | 圆点 | (270,250)、(310,280) | 6×6 | 左上方/下方 | 手性中心 |
        
        const scene1 = {
            currentStep: 0,
            totalSteps: 4,
            
            // 步骤状态配置
            stepConfig: [
                // 步骤 0: 初始状态 - 只显示分子骨架
                {
                    esters: false,
                    doubles: false,
                    chiral: false,
                    labels: false
                },
                // 步骤 1: 高亮酯基
                {
                    esters: true,
                    doubles: false,
                    chiral: false,
                    labels: { ester: true }
                },
                // 步骤 2: 高亮双键
                {
                    esters: true,
                    doubles: true,
                    chiral: false,
                    labels: { ester: true, double: true }
                },
                // 步骤 3: 标记手性碳
                {
                    esters: true,
                    doubles: true,
                    chiral: true,
                    labels: { ester: true, double: true, chiral: true }
                },
                // 步骤 4: 完整标注
                {
                    esters: true,
                    doubles: true,
                    chiral: true,
                    labels: { ester: true, double: true, chiral: true, summary: true }
                }
            ],
            
            // 初始化
            init: function() {
                this.updateDisplay();
                this.bindEvents();
            },
            
            // 绑定事件
            bindEvents: function() {
                const prevBtn = document.getElementById('scene1-btn-prev');
                const nextBtn = document.getElementById('scene1-btn-next');
                
                prevBtn.addEventListener('click', () => {
                    if (this.currentStep > 0) {
                        this.currentStep--;
                        this.updateDisplay();
                    }
                });
                
                nextBtn.addEventListener('click', () => {
                    if (this.currentStep < this.totalSteps) {
                        this.currentStep++;
                        this.updateDisplay();
                    }
                });
            },
            
            // 更新显示
            updateDisplay: function() {
                const config = this.stepConfig[this.currentStep];
                
                // 更新酯基高亮
                const esterGroup = document.getElementById('scene1-ester-highlights-group');
                esterGroup.style.opacity = config.esters ? '1' : '0';
                
                // 更新双键高亮
                const doubleBonds = document.querySelectorAll('#scene1-double-bonds-group line');
                doubleBonds.forEach(line => {
                    line.style.opacity = config.doubles ? '1' : '0.3';
                });
                
                // 更新手性碳标记
                const chiralGroup = document.getElementById('scene1-chiral-group');
                chiralGroup.style.opacity = config.chiral ? '1' : '0';
                
                // 更新标签
                if (config.labels.ester) {
                    document.getElementById('scene1-label-ester-1').style.opacity = '1';
                    document.getElementById('scene1-label-ester-2').style.opacity = '1';
                    document.getElementById('scene1-label-ester-3').style.opacity = '1';
                } else {
                    document.getElementById('scene1-label-ester-1').style.opacity = '0';
                    document.getElementById('scene1-label-ester-2').style.opacity = '0';
                    document.getElementById('scene1-label-ester-3').style.opacity = '0';
                }
                
                if (config.labels.double) {
                    for (let i = 1; i <= 5; i++) {
                        document.getElementById('scene1-label-double-' + i).style.opacity = '1';
                    }
                } else {
                    for (let i = 1; i <= 5; i++) {
                        document.getElementById('scene1-label-double-' + i).style.opacity = '0';
                    }
                }
                
                if (config.labels.chiral) {
                    document.getElementById('scene1-label-chiral-1').style.opacity = '1';
                    document.getElementById('scene1-label-chiral-2').style.opacity = '1';
                } else {
                    document.getElementById('scene1-label-chiral-1').style.opacity = '0';
                    document.getElementById('scene1-label-chiral-2').style.opacity = '0';
                }
                
                if (config.labels.summary) {
                    document.getElementById('scene1-label-ester-summary').style.opacity = '1';
                    document.getElementById('scene1-label-double-summary').style.opacity = '1';
                    document.getElementById('scene1-label-chiral').style.opacity = '1';
                    document.getElementById('scene1-annotation-lines-group').style.opacity = '1';
                } else {
                    document.getElementById('scene1-label-ester-summary').style.opacity = '0';
                    document.getElementById('scene1-label-double-summary').style.opacity = '0';
                    document.getElementById('scene1-label-chiral').style.opacity = '0';
                    document.getElementById('scene1-annotation-lines-group').style.opacity = '0';
                }
                
                // 更新按钮状态
                const prevBtn = document.getElementById('scene1-btn-prev');
                const nextBtn = document.getElementById('scene1-btn-next');
                
                prevBtn.disabled = (this.currentStep === 0);
                
                if (this.currentStep === this.totalSteps) {
                    nextBtn.textContent = '完成';
                    nextBtn.disabled = true;
                } else {
                    nextBtn.textContent = '下一步 ▶';
                    nextBtn.disabled = false;
                }
            }
        };
        
        // 初始化场景
        scene1.init();
    })();
    </script>
