    <script>
    (function() {
        // Scene 2 实现：杂交后代遗传分析
        const currentStepMax = 5;
        let currentStep = 0;

        // 元素 ID 列表（按步骤分组）
        const stepElements = {
            1: [
                'scene2-f1-box', 'scene2-f1-title', 'scene2-f1-genotype', 'scene2-f1-phenotype'
            ],
            2: [
                'scene2-punnett-box',
                'scene2-punnett-header-row-1', 'scene2-punnett-header-row-2', 'scene2-punnett-header-row-3', 'scene2-punnett-header-row-4',
                'scene2-punnett-header-col-1', 'scene2-punnett-header-col-2', 'scene2-punnett-header-col-3', 'scene2-punnett-header-col-4',
                'scene2-punnett-cell-1', 'scene2-punnett-genotype-1',
                'scene2-punnett-cell-2', 'scene2-punnett-genotype-2',
                'scene2-punnett-cell-3', 'scene2-punnett-genotype-3',
                'scene2-punnett-cell-4', 'scene2-punnett-genotype-4',
                'scene2-punnett-cell-5', 'scene2-punnett-genotype-5',
                'scene2-punnett-cell-6', 'scene2-punnett-genotype-6',
                'scene2-punnett-cell-7', 'scene2-punnett-genotype-7',
                'scene2-punnett-cell-8', 'scene2-punnett-genotype-8',
                'scene2-punnett-cell-9', 'scene2-punnett-genotype-9',
                'scene2-punnett-cell-10', 'scene2-punnett-genotype-10',
                'scene2-punnett-cell-11', 'scene2-punnett-genotype-11',
                'scene2-punnett-cell-12', 'scene2-punnett-genotype-12',
                'scene2-punnett-cell-13', 'scene2-punnett-genotype-13',
                'scene2-punnett-cell-14', 'scene2-punnett-genotype-14',
                'scene2-punnett-cell-15', 'scene2-punnett-genotype-15',
                'scene2-punnett-cell-16', 'scene2-punnett-genotype-16'
            ],
            3: [
                'scene2-pod-label-box', 'scene2-pod-label-title', 'scene2-pod-label-text',
                'scene2-cotyledon-label-box', 'scene2-cotyledon-label-title', 'scene2-cotyledon-label-text'
            ],
            4: [
                'scene2-f2-pod-ratio'
            ],
            5: [
                'scene2-f2-embryo-ratio'
            ]
        };

        // 高亮元素（步骤 4 和 5）
        const highlightElements = {
            4: ['scene2-punnett-cell-11', 'scene2-punnett-cell-12', 'scene2-punnett-cell-15', 'scene2-punnett-cell-16'], // aa 基因型格子
            5: ['scene2-punnett-cell-2', 'scene2-punnett-cell-5'] // AABb 基因型格子
        };

        // 工具函数：显示元素
        function show(elId) {
            const el = document.getElementById(elId);
            if (el) {
                el.classList.remove('scene2-hidden');
                el.classList.add('scene2-fade-in');
            }
        }

        // 工具函数：隐藏元素
        function hide(elId) {
            const el = document.getElementById(elId);
            if (el) {
                el.classList.add('scene2-hidden');
                el.classList.remove('scene2-fade-in');
                el.classList.remove('scene2-highlight-red', 'scene2-highlight-green');
            }
        }

        // 更新步骤显示
        function updateStep() {
            // 更新步骤指示器
            document.getElementById('scene2-step-indicator').textContent = `步骤 ${currentStep}/6`;

            // 更新按钮状态
            document.getElementById('scene2-btn-prev').disabled = (currentStep === 0);
            document.getElementById('scene2-btn-next').disabled = (currentStep === currentStepMax);

            // 清除所有高亮
            for (let step = 1; step <= currentStepMax; step++) {
                if (highlightElements[step]) {
                    highlightElements[step].forEach(id => hide(id));
                }
            }

            // 显示当前步骤及之前步骤的元素
            for (let step = 1; step <= currentStep; step++) {
                if (stepElements[step]) {
                    stepElements[step].forEach(id => show(id));
                }
            }

            // 步骤 4：高亮 aa 基因型格子（红色）
            if (currentStep >= 4) {
                highlightElements[4].forEach(id => {
                    const cell = document.getElementById(id);
                    if (cell) {
                        cell.classList.remove('scene2-hidden');
                        cell.classList.add('scene2-highlight-red');
                    }
                });
            }

            // 步骤 5：高亮 AABb 基因型格子（绿色）
            if (currentStep >= 5) {
                highlightElements[5].forEach(id => {
                    const cell = document.getElementById(id);
                    if (cell) {
                        cell.classList.remove('scene2-hidden');
                        cell.classList.add('scene2-highlight-green');
                    }
                });
            }
        }

        // 按钮事件绑定
        document.getElementById('scene2-btn-prev').addEventListener('click', function() {
            if (currentStep > 0) {
                currentStep--;
                updateStep();
            }
        });

        document.getElementById('scene2-btn-next').addEventListener('click', function() {
            if (currentStep < currentStepMax) {
                currentStep++;
                updateStep();
            }
        });

        // 初始化
        updateStep();
    })();
    </script>
