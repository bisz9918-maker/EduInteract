    <script>
    (function() {
        // Scene 2: 杂交实验与遗传比例分析
        // 坐标表对照：
        // 标题 (400,25), 亲本左 (80,40)-(180,90), 亲本右 (520,40)-(620,90)
        // F₁(320,130)-(480,180), F₂棋盘格 (220,235)-(420,355)
        // 株系框 (100,380)-(350,430), 测交框 (450,380)-(700,430)
        
        const scene2 = {
            currentStep: 0,
            totalSteps: 7,
            
            // 元素 ID 列表
            elements: {
                parentBoxes: ['scene2-parent-left-box', 'scene2-parent-left-genotype', 'scene2-parent-left-phenotype',
                              'scene2-parent-right-box', 'scene2-parent-right-genotype', 'scene2-parent-right-phenotype'],
                crossArrow: ['scene2-cross-arrow', 'scene2-arrow-head'],
                f1Box: ['scene2-f1-box', 'scene2-f1-genotype', 'scene2-f1-phenotype'],
                f2Title: ['scene2-f2-title'],
                rowLabels: ['scene2-row-label-1', 'scene2-row-label-2', 'scene2-row-label-3', 'scene2-row-label-4'],
                colLabels: ['scene2-col-label-1', 'scene2-col-label-2', 'scene2-col-label-3', 'scene2-col-label-4'],
                punnettCells: [],
                punnettTexts: [],
                studyBox: ['scene2-study-box', 'scene2-study-title', 'scene2-study-conclusion'],
                testcrossBox: ['scene2-testcross-box', 'scene2-testcross-title', 'scene2-testcross-conclusion']
            },
            
            // 白花格子 ID（步骤 3 高亮）
            whiteCells: ['scene2-punnett-11', 'scene2-punnett-12', 'scene2-punnett-13', 'scene2-punnett-14',
                         'scene2-punnett-21', 'scene2-punnett-23', 'scene2-punnett-31', 'scene2-punnett-34',
                         'scene2-punnett-41', 'scene2-punnett-44'],
            
            init: function() {
                // 生成棋盘格 ID 列表
                for (let i = 1; i <= 4; i++) {
                    for (let j = 1; j <= 4; j++) {
                        this.elements.punnettCells.push('scene2-punnett-' + i + j);
                        this.elements.punnettTexts.push('scene2-punnett-text-' + i + j);
                    }
                }
                
                // 绑定按钮事件
                document.getElementById('scene2-prev-btn').addEventListener('click', () => this.prevStep());
                document.getElementById('scene2-next-btn').addEventListener('click', () => this.nextStep());
                
                // 初始化显示
                this.updateDisplay();
            },
            
            show: function(id) {
                const el = document.getElementById(id);
                if (el) {
                    el.style.display = 'block';
                    el.style.opacity = '1';
                }
            },
            
            hide: function(id) {
                const el = document.getElementById(id);
                if (el) {
                    el.style.display = 'none';
                    el.style.opacity = '0';
                }
            },
            
            showGroup: function(ids) {
                ids.forEach(id => this.show(id));
            },
            
            hideGroup: function(ids) {
                ids.forEach(id => this.hide(id));
            },
            
            highlightWhiteCells: function() {
                this.whiteCells.forEach(id => {
                    const el = document.getElementById(id);
                    if (el) {
                        el.setAttribute('stroke', '#1565c0');
                        el.setAttribute('stroke-width', '3');
                    }
                });
            },
            
            updateDisplay: function() {
                // 重置所有元素
                this.hideGroup(this.elements.f1Box);
                this.hideGroup(this.elements.f2Title);
                this.hideGroup(this.elements.rowLabels);
                this.hideGroup(this.elements.colLabels);
                this.hideGroup(this.elements.punnettCells);
                this.hideGroup(this.elements.punnettTexts);
                this.hideGroup(this.elements.studyBox);
                this.hideGroup(this.elements.testcrossBox);
                
                // 重置高亮
                this.whiteCells.forEach(id => {
                    const el = document.getElementById(id);
                    if (el) {
                        el.setAttribute('stroke', '#888880');
                        el.setAttribute('stroke-width', '1');
                    }
                });
                
                // 重置结论样式
                const studyConc = document.getElementById('scene2-study-conclusion');
                const testConc = document.getElementById('scene2-testcross-conclusion');
                if (studyConc) {
                    studyConc.setAttribute('fill', '#000000');
                    studyConc.setAttribute('font-weight', 'normal');
                }
                if (testConc) {
                    testConc.setAttribute('fill', '#000000');
                    testConc.setAttribute('font-weight', 'normal');
                }
                
                // 根据步骤显示元素
                switch (this.currentStep) {
                    case 0: // 初始状态：显示亲本信息
                        this.showGroup(this.elements.parentBoxes);
                        this.showGroup(this.elements.crossArrow);
                        break;
                        
                    case 1: // 展示 F₁代
                        this.showGroup(this.elements.parentBoxes);
                        this.showGroup(this.elements.crossArrow);
                        this.showGroup(this.elements.f1Box);
                        break;
                        
                    case 2: // 展示 F₂棋盘格
                        this.showGroup(this.elements.parentBoxes);
                        this.showGroup(this.elements.crossArrow);
                        this.showGroup(this.elements.f1Box);
                        this.showGroup(this.elements.f2Title);
                        this.showGroup(this.elements.rowLabels);
                        this.showGroup(this.elements.colLabels);
                        this.showGroup(this.elements.punnettCells);
                        this.showGroup(this.elements.punnettTexts);
                        break;
                        
                    case 3: // 高亮 F₂白花个体
                        this.showGroup(this.elements.parentBoxes);
                        this.showGroup(this.elements.crossArrow);
                        this.showGroup(this.elements.f1Box);
                        this.showGroup(this.elements.f2Title);
                        this.showGroup(this.elements.rowLabels);
                        this.showGroup(this.elements.colLabels);
                        this.showGroup(this.elements.punnettCells);
                        this.showGroup(this.elements.punnettTexts);
                        this.highlightWhiteCells();
                        break;
                        
                    case 4: // 展示白花基因型组成及比例
                        this.showGroup(this.elements.parentBoxes);
                        this.showGroup(this.elements.crossArrow);
                        this.showGroup(this.elements.f1Box);
                        this.showGroup(this.elements.f2Title);
                        this.showGroup(this.elements.rowLabels);
                        this.showGroup(this.elements.colLabels);
                        this.showGroup(this.elements.punnettCells);
                        this.showGroup(this.elements.punnettTexts);
                        this.highlightWhiteCells();
                        this.showGroup(this.elements.studyBox);
                        break;
                        
                    case 5: // 分析株系自交结果
                        this.showGroup(this.elements.parentBoxes);
                        this.showGroup(this.elements.crossArrow);
                        this.showGroup(this.elements.f1Box);
                        this.showGroup(this.elements.f2Title);
                        this.showGroup(this.elements.rowLabels);
                        this.showGroup(this.elements.colLabels);
                        this.showGroup(this.elements.punnettCells);
                        this.showGroup(this.elements.punnettTexts);
                        this.highlightWhiteCells();
                        this.showGroup(this.elements.studyBox);
                        // 高亮株系结论
                        const conclusion = document.getElementById('scene2-study-conclusion');
                        if (conclusion) {
                            conclusion.setAttribute('fill', '#1565c0');
                            conclusion.setAttribute('font-weight', 'bold');
                        }
                        break;
                        
                    case 6: // 展示紫花植株测交结果
                        this.showGroup(this.elements.parentBoxes);
                        this.showGroup(this.elements.crossArrow);
                        this.showGroup(this.elements.f1Box);
                        this.showGroup(this.elements.f2Title);
                        this.showGroup(this.elements.rowLabels);
                        this.showGroup(this.elements.colLabels);
                        this.showGroup(this.elements.punnettCells);
                        this.showGroup(this.elements.punnettTexts);
                        this.highlightWhiteCells();
                        this.showGroup(this.elements.studyBox);
                        this.showGroup(this.elements.testcrossBox);
                        break;
                        
                    case 7: // 结论
                        this.showGroup(this.elements.parentBoxes);
                        this.showGroup(this.elements.crossArrow);
                        this.showGroup(this.elements.f1Box);
                        this.showGroup(this.elements.f2Title);
                        this.showGroup(this.elements.rowLabels);
                        this.showGroup(this.elements.colLabels);
                        this.showGroup(this.elements.punnettCells);
                        this.showGroup(this.elements.punnettTexts);
                        this.highlightWhiteCells();
                        this.showGroup(this.elements.studyBox);
                        this.showGroup(this.elements.testcrossBox);
                        // 高亮所有结论
                        const studyConc2 = document.getElementById('scene2-study-conclusion');
                        const testConc2 = document.getElementById('scene2-testcross-conclusion');
                        if (studyConc2) {
                            studyConc2.setAttribute('fill', '#1565c0');
                            studyConc2.setAttribute('font-weight', 'bold');
                        }
                        if (testConc2) {
                            testConc2.setAttribute('fill', '#1565c0');
                            testConc2.setAttribute('font-weight', 'bold');
                        }
                        break;
                }
                
                // 更新步骤指示
                document.getElementById('scene2-step-indicator').textContent = '步骤 ' + (this.currentStep + 1) + '/7';
                
                // 更新按钮状态
                document.getElementById('scene2-prev-btn').disabled = (this.currentStep === 0);
                document.getElementById('scene2-next-btn').disabled = (this.currentStep === this.totalSteps);
            },
            
            nextStep: function() {
                if (this.currentStep < this.totalSteps) {
                    this.currentStep++;
                    this.updateDisplay();
                }
            },
            
            prevStep: function() {
                if (this.currentStep > 0) {
                    this.currentStep--;
                    this.updateDisplay();
                }
            }
        };
        
        // 初始化场景
        scene2.init();
    })();
    </script>
