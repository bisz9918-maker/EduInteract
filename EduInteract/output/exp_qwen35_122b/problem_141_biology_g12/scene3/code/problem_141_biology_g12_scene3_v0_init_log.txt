    <script>
    (function() {
        const TOTAL_STEPS = 6;
        let currentStep = 0;
        
        const btnPrev = document.getElementById('scene3-btn-prev');
        const btnNext = document.getElementById('scene3-btn-next');
        const criteriaBox = document.getElementById('scene3-criteria-box');
        const criteriaText = document.getElementById('scene3-criteria-text');
        const annotation = document.getElementById('scene3-annotation');
        const checkmark = document.getElementById('scene3-checkmark');
        
        const tableRows = {
            a: document.getElementById('scene3-table-row-a'),
            b: document.getElementById('scene3-table-row-b'),
            c: document.getElementById('scene3-table-row-c'),
            d: document.getElementById('scene3-table-row-d'),
            e: document.getElementById('scene3-table-row-e'),
            f: document.getElementById('scene3-table-row-f')
        };
        
        const selfCrossCells = {
            b: document.getElementById('scene3-table-cell-b-4'),
            e: document.getElementById('scene3-table-cell-e-4')
        };
        
        function show(el) { if (el) { el.style.display = 'block'; el.style.opacity = '1'; } }
        function hide(el) { if (el) { el.style.display = 'none'; el.style.opacity = '0'; } }
        function addHighlight(el, className) { if (el) el.classList.add(className); }
        function removeHighlight(el, className) { if (el) el.classList.remove(className); }
        function highlightRow(type, className) { if (tableRows[type]) tableRows[type].classList.add(className); }
        function unhighlightRow(type, className) { if (tableRows[type]) tableRows[type].classList.remove(className); }
        
        function clearAllHighlights() {
            ['a', 'b', 'c', 'd', 'e', 'f'].forEach(type => {
                unhighlightRow(type, 'scene3-row-highlight');
                unhighlightRow(type, 'scene3-row-crowded');
            });
            removeHighlight(tableRows.b, 'scene3-highlight-red');
            removeHighlight(tableRows.e, 'scene3-highlight-red');
            removeHighlight(tableRows.d, 'scene3-highlight-green');
            Object.values(selfCrossCells).forEach(cell => {
                if (cell) { cell.style.fontWeight = 'normal'; cell.style.fill = '#000000'; }
            });
        }
        
        function updateScene() {
            btnPrev.disabled = (currentStep === 0);
            btnNext.disabled = (currentStep === TOTAL_STEPS);
            clearAllHighlights();
            hide(annotation);
            hide(criteriaBox);
            hide(criteriaText);
            hide(checkmark);
            
            switch(currentStep) {
                case 0: break;
                case 1: break;
                case 2:
                    addHighlight(tableRows.b, 'scene3-highlight-red');
                    addHighlight(tableRows.e, 'scene3-highlight-red');
                    selfCrossCells.b.style.fontWeight = 'bold';
                    selfCrossCells.b.style.fill = '#c62828';
                    selfCrossCells.e.style.fontWeight = 'bold';
                    selfCrossCells.e.style.fill = '#c62828';
                    annotation.textContent = '自交目的 a：通过自交系一代的抗性，将 e 与 b/c 区分';
                    annotation.style.fill = '#c62828';
                    show(annotation);
                    break;
                case 3:
                    addHighlight(tableRows.d, 'scene3-highlight-green');
                    tableRows.d.style.strokeWidth = '3';
                    annotation.textContent = '自交目的 b：筛选纯合体（d 类型）';
                    annotation.style.fill = '#2e7d32';
                    show(annotation);
                    break;
                case 4:
                    highlightRow('a', 'scene3-row-highlight');
                    highlightRow('b', 'scene3-row-highlight');
                    highlightRow('d', 'scene3-row-highlight');
                    highlightRow('e', 'scene3-row-highlight');
                    highlightRow('c', 'scene3-row-crowded');
                    highlightRow('f', 'scene3-row-crowded');
                    annotation.textContent = '杂交系一代结果类型：2 种（全抗性/全无抗性）';
                    annotation.style.fill = '#2e7d32';
                    show(annotation);
                    break;
                case 5:
                    show(criteriaBox);
                    show(criteriaText);
                    highlightRow('d', 'scene3-row-highlight');
                    addHighlight(tableRows.d, 'scene3-highlight-green');
                    annotation.textContent = '符合要求的种子：d 类型的自交后代';
                    annotation.style.fill = '#2e7d32';
                    show(annotation);
                    break;
                case 6:
                    show(criteriaBox);
                    show(criteriaText);
                    highlightRow('d', 'scene3-row-highlight');
                    addHighlight(tableRows.d, 'scene3-highlight-green');
                    show(checkmark);
                    annotation.textContent = 'd 类型：染色体纯合 S+，线粒体 S-，抗性稳定遗传';
                    annotation.style.fill = '#000000';
                    show(annotation);
                    break;
            }
        }
        
        btnPrev.addEventListener('click', function() { if (currentStep > 0) { currentStep--; updateScene(); } });
        btnNext.addEventListener('click', function() { if (currentStep < TOTAL_STEPS) { currentStep++; updateScene(); } });
        updateScene();
    })();
    </script>
