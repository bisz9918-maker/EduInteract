    <script>
    (function() {
        // Scene 1 实现：基因互作与毛发颜色性状的代谢途径
        // 分步序列：
        // 步骤 0：基础代谢途径（白色→棕色→黑色）
        // 步骤 1：显示 B 基因标签
        // 步骤 2：显示 C 基因标签
        // 步骤 3：显示 A 基因和抑制关系
        // 步骤 4：显示基因型 - 表现型对照表

        const TOTAL_STEPS = 4;
        let currentStep = 0;

        // 元素引用缓存
        const elements = {
            // 基因标签
            labelA: document.getElementById('scene1-label-A'),
            labelB: document.getElementById('scene1-label-B'),
            labelC: document.getElementById('scene1-label-C'),
            // 抑制关系
            inhibitLine: document.getElementById('scene1-inhibit-line'),
            inhibitHead: document.getElementById('scene1-inhibit-head'),
            labelInhibit: document.getElementById('scene1-label-inhibit'),
            // 对照表
            tableBox: document.getElementById('scene1-table-box'),
            tableTitle: document.getElementById('scene1-table-title'),
            tableRow1: document.getElementById('scene1-table-row1'),
            tableRow2: document.getElementById('scene1-table-row2'),
            tableRow3: document.getElementById('scene1-table-row3'),
            // 按钮
            prevBtn: document.getElementById('scene1-prev-btn'),
            nextBtn: document.getElementById('scene1-next-btn')
        };

        // 工具函数：显示元素
        function show(el) {
            if (el) {
                el.classList.remove('scene1-hidden');
                el.style.display = 'block';
                el.style.opacity = '1';
            }
        }

        // 工具函数：隐藏元素
        function hide(el) {
            if (el) {
                el.classList.add('scene1-hidden');
                el.style.display = 'none';
                el.style.opacity = '0';
            }
        }

        // 更新步骤显示状态
        function updateStep() {
            // 步骤 0：只显示基础代谢途径（已默认显示，无需额外操作）
            // 步骤 1：显示 B 基因
            if (currentStep >= 1) {
                show(elements.labelB);
            } else {
                hide(elements.labelB);
            }

            // 步骤 2：显示 C 基因
            if (currentStep >= 2) {
                show(elements.labelC);
            } else {
                hide(elements.labelC);
            }

            // 步骤 3：显示 A 基因和抑制关系
            if (currentStep >= 3) {
                show(elements.labelA);
                show(elements.inhibitLine);
                show(elements.inhibitHead);
                show(elements.labelInhibit);
            } else {
                hide(elements.labelA);
                hide(elements.inhibitLine);
                hide(elements.inhibitHead);
                hide(elements.labelInhibit);
            }

            // 步骤 4：显示对照表
            if (currentStep >= 4) {
                show(elements.tableBox);
                show(elements.tableTitle);
                show(elements.tableRow1);
                show(elements.tableRow2);
                show(elements.tableRow3);
            } else {
                hide(elements.tableBox);
                hide(elements.tableTitle);
                hide(elements.tableRow1);
                hide(elements.tableRow2);
                hide(elements.tableRow3);
            }

            // 更新按钮状态
            elements.prevBtn.disabled = (currentStep === 0);
            elements.nextBtn.disabled = (currentStep === TOTAL_STEPS);
        }

        // 事件绑定
        elements.prevBtn.addEventListener('click', function() {
            if (currentStep > 0) {
                currentStep--;
                updateStep();
            }
        });

        elements.nextBtn.addEventListener('click', function() {
            if (currentStep < TOTAL_STEPS) {
                currentStep++;
                updateStep();
            }
        });

        // 初始化：确保步骤 0 状态正确
        updateStep();
    })();
    </script>
