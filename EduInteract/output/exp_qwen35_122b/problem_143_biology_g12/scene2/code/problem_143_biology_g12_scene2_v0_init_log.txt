            const stepData = state.stepData[state.currentStep];

            // 更新文本区域

            elements.valueText.textContent = stepData.value;
            elements.conclusionText.textContent = stepData.conclusion;

                if (elements.energyLabels[key]) {
                    elements.energyLabels[key].textContent = stepData.energyDisplay[key];
                }
            }

            // 更新箭头高亮
            updateArrowHighlighting(stepData.highlightArrows);

            // 更新按钮状态

            elements.nextBtn.disabled = state.currentStep === state.totalSteps;

            // 重置所有箭头

                const arrow = elements.arrows[key];
                arrow.setAttribute("stroke", "#000000");
                arrow.setAttribute("stroke-width", "2");
            }

            // 高亮指定箭头
            for (const arrowKey of highlightArrows) {
                if (elements.arrows[arrowKey]) {
                    const arrow = elements.arrows[arrowKey];
                    arrow.setAttribute("stroke", "#ff0000");
                    arrow.setAttribute("stroke-width", "3");
                }
            }
        }

        // 事件绑定
        elements.prevBtn.addEventListener("click", function() {
            if (state.currentStep > 0) {
                state.currentStep--;
                updateStep();
            }
        });

        elements.nextBtn.addEventListener("click", function() {
            if (state.currentStep < state.totalSteps) {
                state.currentStep++;
                updateStep();
