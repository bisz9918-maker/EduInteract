    <script>
    (function() {
        // Scene 1 实现 - 物质转化关系推断图
        // 布局坐标表（基于原图结构）：
        // | 元素 | 位置 | 备注 |
        // |------|------|------|
        // | 物质框 A | (60,40)-(125,105) | 左侧顶部 |
        // | 物质框 B | (60,160)-(125,225) | 左侧中部 |
        // | 物质框 C | (60,280)-(125,345) | 左侧底部 |
        // | 物质框 E | (420,40)-(485,105) | 右侧顶部 |
        // | 物质框 D | (420,160)-(485,225) | 右侧中上部 |
        // | 物质框 F | (420,280)-(485,345) | 右侧中下部 |
        // | 物质框 G | (580,280)-(645,345) | 右侧底部 |
        // | 反应① | (270,72.5) | A+B→E+D |
        // | 反应② | (270,312.5) | B+C→D+F |
        // | 反应③ | (515,192.5) | F+E→E+G |

        const scene1 = {
            currentStep: 0,
            totalSteps: 4,

            // 状态配置
            stepConfig: {
                0: {
                    // 步骤 0：初始状态 - 显示已知条件
                    highlights: { red: ['B','D','G'], blue: [], green: [] },
                    notes: { show: [] },
                    answers: false,
                    indicator: '步骤 1/4'
                },
                1: {
                    // 步骤 1：分析反应③
                    highlights: { red: [], blue: ['reaction3'], green: [] },
                    notes: { show: ['G', 'F', 'E'] },
                    answers: false,
                    indicator: '步骤 2/4'
                },
                2: {
                    // 步骤 2：分析反应①②
                    highlights: { red: [], blue: ['reaction12'], green: [] },
                    notes: { show: ['D', 'B'] },
                    answers: false,
                    indicator: '步骤 3/4'
                },
                3: {
                    // 步骤 3：最终结论
                    highlights: { red: [], blue: [], green: ['A', 'C'] },
                    notes: { show: ['A', 'C'] },
                    answers: true,
                    indicator: '步骤 4/4'
                }
            },

            // 初始化
            init: function() {
                this.renderStep(0);
                this.bindEvents();
            },

            // 渲染步骤
            renderStep: function(step) {
                const config = this.stepConfig[step];

                // 清除所有高亮
                this.hideAllHighlights();

                // 应用红色高亮（已知条件）
                if (config.highlights.red.length > 0) {
                    config.highlights.red.forEach(id => {
                        this.showHighlight('highlight-' + id);
                    });
                }

                // 应用蓝色高亮（当前分析的反应）
                if (config.highlights.blue.length > 0) {
                    config.highlights.blue.forEach(id => {
                        this.showHighlight('highlight-' + id);
                    });
                }

                // 应用绿色高亮（最终结论）
                if (config.highlights.green.length > 0) {
                    config.highlights.green.forEach(id => {
                        this.showHighlight('highlight-' + id);
                    });
                }

                // 显示/隐藏推理标注
                this.hideAllNotes();
                if (config.notes.show.length > 0) {
                    config.notes.show.forEach(id => {
                        this.showNote('note-' + id);
                    });
                }

                // 显示/隐藏最终答案
                if (config.answers) {
                    this.showAnswer();
                } else {
                    this.hideAnswer();
                }

                // 更新步骤指示器
                document.getElementById('scene1-step-indicator').textContent = config.indicator;

                // 更新按钮状态
                document.getElementById('scene1-prev-btn').disabled = (step === 0);
                document.getElementById('scene1-next-btn').disabled = (step === this.totalSteps - 1);
            },

            // 显示高亮框
            showHighlight: function(id) {
                const el = document.getElementById('scene1-' + id);
                if (el) {
                    el.style.display = 'block';
                    el.style.opacity = '0';
                    setTimeout(() => { el.style.opacity = '1'; }, 10);
                }
            },

            // 隐藏所有高亮
            hideAllHighlights: function() {
                const highlights = document.querySelectorAll('[id^="scene1-highlight-"]');
                highlights.forEach(el => {
                    el.style.display = 'none';
                    el.style.opacity = '1';
                });
            },

            // 显示推理标注
            showNote: function(id) {
                const el = document.getElementById('scene1-' + id);
                if (el) {
                    el.style.display = 'block';
                    el.style.opacity = '0';
                    setTimeout(() => { el.style.opacity = '1'; }, 10);
                }
            },

            // 隐藏所有推理标注
            hideAllNotes: function() {
                const notes = document.querySelectorAll('[id^="scene1-note-"]');
                notes.forEach(el => {
                    el.style.display = 'none';
                    el.style.opacity = '1';
                });
            },

            // 显示最终答案
            showAnswer: function() {
                document.getElementById('scene1-answer-box').style.display = 'block';
                document.getElementById('scene1-answer-title').style.display = 'block';
                document.getElementById('scene1-answer-1').style.display = 'block';
                document.getElementById('scene1-answer-2').style.display = 'block';
                document.getElementById('scene1-answer-3').style.display = 'block';
            },

            // 隐藏最终答案
            hideAnswer: function() {
                document.getElementById('scene1-answer-box').style.display = 'none';
                document.getElementById('scene1-answer-title').style.display = 'none';
                document.getElementById('scene1-answer-1').style.display = 'none';
                document.getElementById('scene1-answer-2').style.display = 'none';
                document.getElementById('scene1-answer-3').style.display = 'none';
            },

            // 绑定事件
            bindEvents: function() {
                const self = this;
                document.getElementById('scene1-prev-btn').addEventListener('click', function() {
                    if (self.currentStep > 0) {
                        self.currentStep--;
                        self.renderStep(self.currentStep);
                    }
                });

                document.getElementById('scene1-next-btn').addEventListener('click', function() {
                    if (self.currentStep < self.totalSteps - 1) {
                        self.currentStep++;
                        self.renderStep(self.currentStep);
                    }
                });
            }
        };

        // 初始化场景
        scene1.init();
    })();
    </script>
