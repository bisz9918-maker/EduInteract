    <script>
    (function() {
        // Scene 4 实现
        // 布局坐标表（修复后 - 坐标系缩小，标签移到外部，说明区左移）：
        // | 元素 | 类型 | 位置/范围 | 大小 | 标签偏移方向 | 备注 |
        // |------|------|-----------|------|-------------|------|
        // | 坐标系边框 | 矩形 | (40,40)-(410,390) | 370×350 | - | 左侧绘图区 |
        // | 横轴 | 线段 | (50,380)-(400,380) | 350×2 | - | 温度轴 |
        // | 纵轴 | 线段 | (50,50)-(50,380) | 2×330 | - | 溶解度轴 |
        // | 横轴标签"温度" | text | (415,405) | ~40×16 | 下方 | 横轴右侧外部 |
        // | 纵轴标签"溶解度/L" | text | (15,200) | ~80×16 | 左方 | 纵轴左侧旋转 90° |
        // | 原点"0" | text | (45,395) | ~16×16 | 左下 | 坐标原点外部 |
        // | 原图曲线 | 路径 | (80,330)-(390,100) | ~310×230 | - | 蓝色上升曲线（错误） |
        // | 正确曲线 | 路径 | (80,100)-(390,330) | ~310×230 | - | 绿色下降曲线（正确） |
        // | 原图标签"原图" | text | (415,150) | ~40×16 | 右方 | 蓝色曲线末端右侧外部 |
        // | 正确标签"正确" | text | (415,340) | ~40×16 | 右方 | 绿色曲线末端右侧外部 |
        // | 错误标记"×" | text | (415,150) | ~20×20 | 右方 | 状态 3 显示，红色 |
        // | 右侧说明区背景 | 矩形 | (450,50)-(780,380) | 330×330 | - | 浅灰背景 |
        // | 规律标题 | text | (465,75) | ~120×18 | - | "气体溶解度规律" |
        // | 规律内容 | text | (465,110) | ~280×20 | - | "气体溶解度随温度升高而降低" |
        // | 错误原因标题 | text | (465,210) | ~100×18 | - | "错误原因分析" |
        // | 错误原因内容 | text | (465,245) | ~280×40 | - | 解释原图错误点（行间距 20px） |
        
        // 状态变量
        let currentStep = 0;
        
        // 工具函数：显示元素（同时设置 display 和 opacity）
        function show(el) {
            if (el) {
                el.style.display = 'block';
                el.style.opacity = '1';
            }
        }
        
        // 工具函数：隐藏元素（同时设置 display 和 opacity）
        function hide(el) {
            if (el) {
                el.style.display = 'none';
                el.style.opacity = '0';
            }
        }
        
        // 更新按钮状态
        function updateButtons(step) {
            for (let i = 0; i <= 3; i++) {
                const btn = document.getElementById('scene4-btn-' + i);
                if (btn) {
                    if (i === step) {
                        btn.classList.add('active');
                    } else {
                        btn.classList.remove('active');
                    }
                }
            }
        }
        
        // 状态 0：原图曲线
        function showStep0() {
            const curveOriginal = document.getElementById('scene4-curve-original');
            const labelOriginal = document.getElementById('scene4-label-original');
            const guideLine = document.getElementById('scene4-guide-line');
            
            const curveCorrect = document.getElementById('scene4-curve-correct');
            const labelCorrect = document.getElementById('scene4-label-correct');
            const markError = document.getElementById('scene4-mark-error');
            const textLegend = document.getElementById('scene4-text-legend');
            const titleError = document.getElementById('scene4-title-error');
            const textError = document.getElementById('scene4-text-error');
            const textError2 = document.getElementById('scene4-text-error-2');
            const textError3 = document.getElementById('scene4-text-error-3');
            
            show(curveOriginal);
            show(labelOriginal);
            hide(curveCorrect);
            hide(labelCorrect);
            hide(markError);
            hide(guideLine);
            hide(textLegend);
            hide(titleError);
            hide(textError);
            hide(textError2);
            hide(textError3);
        }
        
        // 状态 1：溶解度规律
        function showStep1() {
            const curveOriginal = document.getElementById('scene4-curve-original');
            const labelOriginal = document.getElementById('scene4-label-original');
            const guideLine = document.getElementById('scene4-guide-line');
            const textLegend = document.getElementById('scene4-text-legend');
            
            const curveCorrect = document.getElementById('scene4-curve-correct');
            const labelCorrect = document.getElementById('scene4-label-correct');
            const markError = document.getElementById('scene4-mark-error');
            const titleError = document.getElementById('scene4-title-error');
            const textError = document.getElementById('scene4-text-error');
            const textError2 = document.getElementById('scene4-text-error-2');
            const textError3 = document.getElementById('scene4-text-error-3');
            
            show(curveOriginal);
            show(labelOriginal);
            show(guideLine);
            show(textLegend);
            hide(curveCorrect);
            hide(labelCorrect);
            hide(markError);
            hide(titleError);
            hide(textError);
            hide(textError2);
            hide(textError3);
        }
        
        // 状态 2：正确曲线
        function showStep2() {
            const curveOriginal = document.getElementById('scene4-curve-original');
            const labelOriginal = document.getElementById('scene4-label-original');
            const curveCorrect = document.getElementById('scene4-curve-correct');
            const labelCorrect = document.getElementById('scene4-label-correct');
            
            const guideLine = document.getElementById('scene4-guide-line');
            const textLegend = document.getElementById('scene4-text-legend');
            const markError = document.getElementById('scene4-mark-error');
            const titleError = document.getElementById('scene4-title-error');
            const textError = document.getElementById('scene4-text-error');
            const textError2 = document.getElementById('scene4-text-error-2');
            const textError3 = document.getElementById('scene4-text-error-3');
            
            hide(curveOriginal);
            hide(labelOriginal);
            show(curveCorrect);
            show(labelCorrect);
            hide(guideLine);
            hide(textLegend);
            hide(markError);
            hide(titleError);
            hide(textError);
            hide(textError2);
            hide(textError3);
        }
        
        // 状态 3：错误原因
        function showStep3() {
            const curveOriginal = document.getElementById('scene4-curve-original');
            const markError = document.getElementById('scene4-mark-error');
            const titleError = document.getElementById('scene4-title-error');
            const textError = document.getElementById('scene4-text-error');
            const textError2 = document.getElementById('scene4-text-error-2');
            const textError3 = document.getElementById('scene4-text-error-3');
            
            const labelOriginal = document.getElementById('scene4-label-original');
            const curveCorrect = document.getElementById('scene4-curve-correct');
            const labelCorrect = document.getElementById('scene4-label-correct');
            const guideLine = document.getElementById('scene4-guide-line');
            const textLegend = document.getElementById('scene4-text-legend');
            
            show(curveOriginal);
            show(markError);
            show(titleError);
            show(textError);
            show(textError2);
            show(textError3);
            hide(labelOriginal);
            hide(curveCorrect);
            hide(labelCorrect);
            hide(guideLine);
            hide(textLegend);
        }
        
        // 切换状态
        function switchStep(step) {
            currentStep = step;
            updateButtons(step);
            
            switch(step) {
                case 0: showStep0(); break;
                case 1: showStep1(); break;
                case 2: showStep2(); break;
                case 3: showStep3(); break;
            }
        }
        
        // 初始化事件绑定
        function init() {
            for (let i = 0; i <= 3; i++) {
                const btn = document.getElementById('scene4-btn-' + i);
                if (btn) {
                    btn.addEventListener('click', function() {
                        switchStep(i);
                    });
                }
            }
            
            switchStep(0);
        }
        
        if (document.readyState === 'loading') {
            document.addEventListener('DOMContentLoaded', init);
        } else {
            init();
        }
    })();
    </script>
