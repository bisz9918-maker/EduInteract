    <script>
    (function() {
        // Scene 1: 楼房平面图周长计算 - 平移法演示
        // 布局坐标表对照：
        // | 元素 | 位置/范围 | 备注 |
        // | 线段 a | (250,200)-(250,350) | 垂直，左侧边 |
        // | 线段 b | (150,350)-(650,350) | 水平，底部边 |
        // | 线段 c | (650,350)-(650,200) | 垂直，右侧边 |
        // | 线段 d | (600,200)-(650,200) | 水平，右上顶边 |
        // | 线段 e | (600,200)-(600,250) | 垂直，中间凸起垂直边 |
        // | 线段 f | (400,250)-(600,250) | 水平，中间水平边 |
        // | 线段 g | (400,250)-(400,200) | 垂直，中间凹陷垂直边 |
        // | 线段 h | (250,200)-(400,200) | 水平，左上顶边 |

        const currentStepKey = 'scene1-currentStep';
        let currentStep = 0;
        const totalSteps = 4;

        // 获取 DOM 元素
        const segments = {
            a: document.getElementById('scene1-seg-a'),
            b: document.getElementById('scene1-seg-b'),
            c: document.getElementById('scene1-seg-c'),
            d: document.getElementById('scene1-seg-d'),
            e: document.getElementById('scene1-seg-e'),
            f: document.getElementById('scene1-seg-f'),
            g: document.getElementById('scene1-seg-g'),
            h: document.getElementById('scene1-seg-h')
        };

        const labels = {
            a: document.getElementById('scene1-label-a'),
            b: document.getElementById('scene1-label-b'),
            c: document.getElementById('scene1-label-c'),
            d: document.getElementById('scene1-label-d'),
            e: document.getElementById('scene1-label-e'),
            f: document.getElementById('scene1-label-f'),
            g: document.getElementById('scene1-label-g'),
            h: document.getElementById('scene1-label-h')
        };

        const values = {
            b: document.getElementById('scene1-value-b'),
            c: document.getElementById('scene1-value-c'),
            g: document.getElementById('scene1-value-g')
        };

        const arrows = {
            h: { line: document.getElementById('scene1-arrow-h'), head: document.getElementById('scene1-arrow-h-head') },
            f: { line: document.getElementById('scene1-arrow-f'), head: document.getElementById('scene1-arrow-f-head') },
            d: { line: document.getElementById('scene1-arrow-d'), head: document.getElementById('scene1-arrow-d-head') },
            a: { line: document.getElementById('scene1-arrow-a'), head: document.getElementById('scene1-arrow-a-head') },
            e: { line: document.getElementById('scene1-arrow-e'), head: document.getElementById('scene1-arrow-e-head') }
        };

        const textRelations = {
            hfd: document.getElementById('scene1-text-hfd'),
            ac: document.getElementById('scene1-text-ac')
        };

        const formula = {
            box: document.getElementById('scene1-formula-box'),
            line1: document.getElementById('scene1-formula'),
            line2: document.getElementById('scene1-formula2'),
            line3: document.getElementById('scene1-formula3')
        };

        const optionC = {
            box: document.getElementById('scene1-option-c-box'),
            text: document.getElementById('scene1-option-c')
        };

        const btnPrev = document.getElementById('scene1-btn-prev');
        const btnNext = document.getElementById('scene1-btn-next');

        // 工具函数：显示元素
        function show(el) {
            if (el) {
                el.style.display = 'block';
                el.style.opacity = '1';
            }
        }

        // 工具函数：隐藏元素
        function hide(el) {
            if (el) {
                el.style.display = 'none';
                el.style.opacity = '0';
            }
        }

        // 工具函数：设置线段高亮样式
        function highlightSegment(seg, color) {
            if (seg) {
                seg.style.stroke = color;
                seg.style.strokeWidth = '3';
            }
        }

        // 工具函数：恢复线段默认样式
        function resetSegment(seg) {
            if (seg) {
                seg.style.stroke = '#000000';
                seg.style.strokeWidth = '2';
            }
        }

        // 重置所有线段为默认样式
        function resetAllSegments() {
            Object.values(segments).forEach(seg => resetSegment(seg));
        }

        // 更新按钮状态
        function updateButtons() {
            btnPrev.disabled = (currentStep === 0);
            btnNext.disabled = (currentStep === totalSteps - 1);
        }

        // 渲染当前步骤
        function renderStep() {
            // 重置所有元素
            resetAllSegments();
            Object.values(arrows).forEach(arrow => {
                hide(arrow.line);
                hide(arrow.head);
            });
            hide(textRelations.hfd);
            hide(textRelations.ac);
            hide(formula.box);
            hide(formula.line1);
            hide(formula.line2);
            hide(formula.line3);
            hide(optionC.box);
            hide(optionC.text);

            // 根据步骤设置显示内容
            if (currentStep === 1) {
                // 步骤 1：水平边平移分析
                highlightSegment(segments.b, '#1565c0');
                highlightSegment(segments.h, '#1565c0');
                highlightSegment(segments.f, '#1565c0');
                highlightSegment(segments.d, '#1565c0');
                
                show(arrows.h.line);
                show(arrows.h.head);
                show(arrows.f.line);
                show(arrows.f.head);
                show(arrows.d.line);
                show(arrows.d.head);
                
                show(textRelations.hfd);
            } else if (currentStep === 2) {
                // 步骤 2：垂直边对齐分析
                highlightSegment(segments.a, '#c62828');
                highlightSegment(segments.c, '#c62828');
                highlightSegment(segments.g, '#c62828');
                highlightSegment(segments.e, '#c62828');
                
                show(arrows.a.line);
                show(arrows.a.head);
                show(arrows.e.line);
                show(arrows.e.head);
                
                show(textRelations.ac);
            } else if (currentStep === 3) {
                // 步骤 3：周长计算
                show(formula.box);
                show(formula.line1);
                show(formula.line2);
                show(formula.line3);
                show(optionC.box);
                show(optionC.text);
            }

            updateButtons();
        }

        // 下一步
        function nextStep() {
            if (currentStep < totalSteps - 1) {
                currentStep++;
                renderStep();
            }
        }

        // 上一步
        function prevStep() {
            if (currentStep > 0) {
                currentStep--;
                renderStep();
            }
        }

        // 事件监听
        btnNext.addEventListener('click', nextStep);
        btnPrev.addEventListener('click', prevStep);

        // 初始化
        renderStep();
    })();
    </script>
