    <script>
    (function() {
        // Scene 2 实现 - 平移法详解（图形 B 阶梯形周长计算）
        // 坐标表对照：
        // 原始阶梯图形底边：(350,350)-(550,350) 长度 10
        // 原始阶梯图形左侧边：(350,180)-(350,350) 长度 6
        // 目标长方形：(350,180)-(550,350) 200×170
        // 阶梯边：5 级台阶，每级水平=40px，垂直=34px
        // 滑块：左侧 (60,220)-(200,240)
        // 标签："10" 在 (450,370)，"6" 在 (330,265)

        const svg = document.getElementById('scene2-svg');
        const slider = document.getElementById('scene2-slider');
        const progressText = document.getElementById('scene2-progress-text');
        const perimeterDisplay = document.getElementById('scene2-perimeter-display');
        const explanation = document.getElementById('scene2-explanation');

        // 阶梯边元素 ID 列表
        const hStairs = ['scene2-h1', 'scene2-h2', 'scene2-h3', 'scene2-h4', 'scene2-h5'];
        const vStairs = ['scene2-v1', 'scene2-v2', 'scene2-v3', 'scene2-v4', 'scene2-v5'];
        const hTrajectories = ['scene2-ht1', 'scene2-ht2', 'scene2-ht3', 'scene2-ht4', 'scene2-ht5'];
        const vTrajectories = ['scene2-vt1', 'scene2-vt2', 'scene2-vt3', 'scene2-vt4', 'scene2-vt5'];
        const hArrowHeads = ['scene2-ha1', 'scene2-ha2', 'scene2-ha3', 'scene2-ha4', 'scene2-ha5'];
        const vArrowHeads = ['scene2-va1', 'scene2-va2', 'scene2-va3', 'scene2-va4', 'scene2-va5'];
        const translatedH = ['scene2-th1', 'scene2-th2', 'scene2-th3', 'scene2-th4', 'scene2-th5'];
        const translatedV = ['scene2-tv1', 'scene2-tv2', 'scene2-tv3', 'scene2-tv4', 'scene2-tv5'];

        // 更新函数：根据滑块值更新所有元素
        function updateScene(progress) {
            const progressRatio = progress / 100;
            const numStairs = 5;
            const completedStairs = Math.floor(progressRatio * numStairs);
            const partialProgress = (progressRatio * numStairs) - completedStairs;

            // 更新进度文本
            progressText.textContent = progress + '%';

            // 更新说明文字
            if (progress === 0) {
                explanation.textContent = '拖动滑块，观察阶梯边如何平移成长方形';
            } else if (progress < 25) {
                explanation.textContent = '水平边向上平移，垂直边向右平移';
            } else if (progress < 50) {
                explanation.textContent = '平移过程中，总长度保持不变';
            } else if (progress < 75) {
                explanation.textContent = '所有边平移后形成完整长方形';
            } else if (progress < 100) {
                explanation.textContent = '接近完成，观察阶梯形如何变成规则长方形';
            } else {
                explanation.textContent = '阶梯形周长 = 长方形周长 = 2×(10+6) = 32';
            }

            // 更新水平阶梯边和轨迹
            for (let i = 0; i < numStairs; i++) {
                const hStair = document.getElementById(hStairs[i]);
                const hTraj = document.getElementById(hTrajectories[i]);
                const hArrow = document.getElementById(hArrowHeads[i]);
                const transH = document.getElementById(translatedH[i]);

                if (i < completedStairs) {
                    // 已完成平移：隐藏原边，显示轨迹，显示平移后的边
                    hStair.style.opacity = '0';
                    hTraj.style.opacity = '0.3';
                    hArrow.style.opacity = '0.3';
                    transH.style.opacity = '1';
                } else if (i === completedStairs && partialProgress > 0) {
                    // 正在平移：原边逐渐淡出，轨迹显示，平移后的边逐渐显示
                    hStair.style.opacity = (1 - partialProgress).toString();
                    hTraj.style.opacity = '0.6';
                    hArrow.style.opacity = '0.6';
                    transH.style.opacity = partialProgress.toString();
                } else {
                    // 未开始平移：显示原边，隐藏轨迹和平移后的边
                    hStair.style.opacity = '1';
                    hTraj.style.opacity = '0';
                    hArrow.style.opacity = '0';
                    transH.style.opacity = '0';
                }
            }

            // 更新垂直阶梯边和轨迹
            for (let i = 0; i < numStairs; i++) {
                const vStair = document.getElementById(vStairs[i]);
                const vTraj = document.getElementById(vTrajectories[i]);
                const vArrow = document.getElementById(vArrowHeads[i]);
                const transV = document.getElementById(translatedV[i]);

                if (i < completedStairs) {
                    // 已完成平移
                    vStair.style.opacity = '0';
                    vTraj.style.opacity = '0.3';
                    vArrow.style.opacity = '0.3';
                    transV.style.opacity = '1';
                } else if (i === completedStairs && partialProgress > 0) {
                    // 正在平移
                    vStair.style.opacity = (1 - partialProgress).toString();
                    vTraj.style.opacity = '0.6';
                    vArrow.style.opacity = '0.6';
                    transV.style.opacity = partialProgress.toString();
                } else {
                    // 未开始平移
                    vStair.style.opacity = '1';
                    vTraj.style.opacity = '0';
                    vArrow.style.opacity = '0';
                    transV.style.opacity = '0';
                }
            }

            // 更新周长显示
            perimeterDisplay.textContent = '周长：32';
        }

        // 滑块事件监听
        slider.addEventListener('input', function() {
            updateScene(parseInt(this.value));
        });

        // 初始化
        updateScene(0);
    })();
    </script>
