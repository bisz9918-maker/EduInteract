    <script>
    (function() {
        // Scene 1 实现
        // 布局坐标表对照：
        // | 元素 | 类型 | 位置/范围 | 大小 | 备注 |
        // | 顶层框 | 矩形 | (300,30)-(500,65) | 200×35 | 物质跨膜运输方式 |
        // | 被动运输框 | 矩形 | (120,120)-(260,155) | 140×35 | 蓝高亮背景 |
        // | ①框 | 矩形 | (540,120)-(680,155) | 140×35 | 红高亮背景 |
        // | 自由扩散框 | 矩形 | (60,210)-(180,245) | 120×35 | - |
        // | ②框 | 矩形 | (280,210)-(400,245) | 120×35 | 绿高亮背景 |
        // | ④框 | 矩形 | (540,210)-(680,245) | 140×35 | - |
        // | 氧气框 | 矩形 | (40,300)-(200,335) | 160×35 | - |
        // | ③框 | 矩形 | (280,300)-(400,335) | 120×35 | - |
        
        const MAX_STEP = 3;
        let currentStep = 0;
        
        // 获取 DOM 元素
        const btnPrev = document.getElementById('scene1-btn-prev');
        const btnNext = document.getElementById('scene1-btn-next');
        const stepIndicator = document.getElementById('scene1-step-indicator');
        const optionsGroup = document.getElementById('scene1-options-group');
        
        // 框元素
        const box1 = document.getElementById('scene1-box-1');
        const text1 = document.getElementById('scene1-text-1');
        const box2 = document.getElementById('scene1-box-2');
        const text2 = document.getElementById('scene1-text-2');
        
        // 更新步骤显示
        function updateScene() {
            // 更新步骤指示器
            stepIndicator.textContent = `步骤 ${currentStep + 1}/${MAX_STEP}`;
            
            // 更新按钮状态
            btnPrev.disabled = (currentStep === 0);
            btnNext.disabled = (currentStep === MAX_STEP);
            btnNext.textContent = (currentStep === MAX_STEP) ? '完成' : '下一步 ▶';
            
            // 根据步骤更新显示内容
            switch(currentStep) {
                case 0:
                    // 初始状态：显示分类框架，①②为序号
                    text1.textContent = '①';
                    text2.textContent = '②';
                    box1.setAttribute('stroke-width', '2');
                    box2.setAttribute('stroke-width', '2');
                    text1.setAttribute('font-weight', 'bold');
                    text2.setAttribute('font-weight', 'bold');
                    optionsGroup.style.display = 'none';
                    break;
                    
                case 1:
                    // 步骤 1：①变为"主动运输"
                    text1.textContent = '主动运输';
                    text2.textContent = '②';
                    box1.setAttribute('stroke-width', '3');
                    box2.setAttribute('stroke-width', '2');
                    text1.setAttribute('font-weight', 'bold');
                    text2.setAttribute('font-weight', 'bold');
                    optionsGroup.style.display = 'none';
                    break;
                    
                case 2:
                    // 步骤 2：②变为"协助扩散"
                    text1.textContent = '主动运输';
                    text2.textContent = '协助扩散';
                    box1.setAttribute('stroke-width', '3');
                    box2.setAttribute('stroke-width', '3');
                    text1.setAttribute('font-weight', 'bold');
                    text2.setAttribute('font-weight', 'bold');
                    optionsGroup.style.display = 'none';
                    break;
                    
                case 3:
                    // 步骤 3：显示选项，选项 B 高亮
                    text1.textContent = '主动运输';
                    text2.textContent = '协助扩散';
                    box1.setAttribute('stroke-width', '3');
                    box2.setAttribute('stroke-width', '3');
                    text1.setAttribute('font-weight', 'bold');
                    text2.setAttribute('font-weight', 'bold');
                    optionsGroup.style.display = 'block';
                    break;
            }
        }
        
        // 按钮事件绑定
        btnPrev.addEventListener('click', function() {
            if (currentStep > 0) {
                currentStep--;
                updateScene();
            }
        });
        
        btnNext.addEventListener('click', function() {
            if (currentStep < MAX_STEP) {
                currentStep++;
                updateScene();
            }
        });
        
        // 初始化
        updateScene();
    })();
    </script>
