    <script>
    (function() {
        // Scene 1: 碳循环生态系统成分识别
        // 布局坐标表：
        // | 元素 | 类型 | 位置/范围 | 大小 | 备注 |
        // |------|------|-----------|------|------|
        // | 大气 CO₂节点 | 椭圆 | (280,30)-(520,110) | 240×80 | 顶部中央 |
        // | 甲节点 | 椭圆 | (60,190)-(220,290) | 160×100 | 左侧中部 |
        // | 乙节点 | 椭圆 | (300,335)-(500,405) | 200×70 | 底部中央 |
        // | 丙节点 | 椭圆 | (580,190)-(740,290) | 160×100 | 右侧中部 |

        const TOTAL_STEPS = 6;
        let currentStep = 0;

        // 元素引用
        const arrows = {
            atmosToJia: document.getElementById('scene1-arrow-atmos-to-jia'),
            jiaToAtmos: document.getElementById('scene1-arrow-jia-to-atmos'),
            jiaToYi: document.getElementById('scene1-arrow-jia-to-yi'),
            jiaToBing: document.getElementById('scene1-arrow-jia-to-bing'),
            bingToYi: document.getElementById('scene1-arrow-bing-to-yi'),
            yiToAtmos: document.getElementById('scene1-arrow-yi-to-atmos')
        };

        const labels = {
            photosynthesis: document.getElementById('scene1-label-photosynthesis'),
            respiration: document.getElementById('scene1-label-respiration'),
            feeding: document.getElementById('scene1-label-feeding'),
            deadJia: document.getElementById('scene1-label-dead-jia'),
            deadBing: document.getElementById('scene1-label-dead-bing'),
            decomposition: document.getElementById('scene1-label-decomposition'),
            producer: document.getElementById('scene1-label-producer'),
            consumer: document.getElementById('scene1-label-consumer'),
            decomposer: document.getElementById('scene1-label-decomposer')
        };

        const btnPrev = document.getElementById('scene1-btn-prev');
        const btnNext = document.getElementById('scene1-btn-next');
        const stepIndicator = document.getElementById('scene1-step-indicator');

        // 工具函数：显示元素
        function show(el) {
            if (el) {
                el.style.display = 'block';
                el.style.opacity = '1';
            }
        }

        // 工具函数：隐藏元素
        function hide(el) {
            if (el) {
                el.style.display = 'none';
                el.style.opacity = '0';
            }
        }

        // 工具函数：设置箭头为激活状态
        function activateArrow(arrow) {
            if (arrow) {
                arrow.classList.remove('arrow-default');
                arrow.classList.add('arrow-active');
                arrow.setAttribute('marker-end', 'url(#scene1-arrowhead-active)');
            }
        }

        // 工具函数：设置箭头为默认状态
        function deactivateArrow(arrow) {
            if (arrow) {
                arrow.classList.remove('arrow-active');
                arrow.classList.add('arrow-default');
                arrow.setAttribute('marker-end', 'url(#scene1-arrowhead)');
            }
        }

        // 更新步骤显示
        function updateStep() {
            // 重置所有箭头为默认状态
            Object.values(arrows).forEach(deactivateArrow);

            // 重置所有过程标注
            hide(labels.photosynthesis);
            hide(labels.respiration);
            hide(labels.feeding);
            hide(labels.deadJia);
            hide(labels.deadBing);
            hide(labels.decomposition);

            // 重置所有结论标签
            hide(labels.producer);
            hide(labels.consumer);
            hide(labels.decomposer);

            // 根据步骤设置显示内容
            switch(currentStep) {
                case 0:
                    // 初始状态：只显示节点和灰色虚线箭头
                    stepIndicator.textContent = '步骤 0/6';
                    break;

                case 1:
                    // 高亮甲↔大气 CO₂双向箭头
                    activateArrow(arrows.atmosToJia);
                    activateArrow(arrows.jiaToAtmos);
                    show(labels.photosynthesis);
                    show(labels.respiration);
                    stepIndicator.textContent = '步骤 1/6';
                    break;

                case 2:
                    // 高亮甲→丙弧线箭头
                    activateArrow(arrows.jiaToBing);
                    show(labels.feeding);
                    stepIndicator.textContent = '步骤 2/6';
                    break;

                case 3:
                    // 高亮甲→乙、丙→乙两条箭头
                    activateArrow(arrows.jiaToYi);
                    activateArrow(arrows.bingToYi);
                    show(labels.deadJia);
                    show(labels.deadBing);
                    stepIndicator.textContent = '步骤 3/6';
                    break;

                case 4:
                    // 高亮乙→大气 CO₂箭头
                    activateArrow(arrows.yiToAtmos);
                    show(labels.decomposition);
                    stepIndicator.textContent = '步骤 4/6';
                    break;

                case 5:
                    // 结论：显示各成分角色
                    show(labels.producer);
                    show(labels.consumer);
                    show(labels.decomposer);
                    stepIndicator.textContent = '结论/6';
                    break;
            }

            // 更新按钮状态
            btnPrev.disabled = (currentStep === 0);
            btnNext.disabled = (currentStep === TOTAL_STEPS - 1);
        }

        // 事件绑定
        btnPrev.addEventListener('click', function() {
            if (currentStep > 0) {
                currentStep--;
                updateStep();
            }
        });

        btnNext.addEventListener('click', function() {
            if (currentStep < TOTAL_STEPS - 1) {
                currentStep++;
                updateStep();
            }
        });

        // 初始化
        updateStep();
    })();
    </script>
