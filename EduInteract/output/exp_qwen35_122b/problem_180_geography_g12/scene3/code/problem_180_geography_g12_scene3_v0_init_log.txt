    <script>
    (function() {
        // Scene 3: 地方时计算与北京时间推导
        // 坐标表对照：
        // 地球轮廓：圆心 (240,250)，半径 110
        // 北极 N：(240,140)，南极 S：(240,360)
        // A 点：(290,250)，B 点：(182,178)，C 点：(350,250)
        
        const TOTAL_STEPS = 6;
        let currentStep = 0;
        
        // 工具函数：显示元素
        function show(el) {
            if (el) {
                el.style.display = 'block';
                el.style.opacity = '1';
            }
        }
        
        // 工具函数：隐藏元素
        function hide(el) {
            if (el) {
                el.style.opacity = '0';
                setTimeout(() => {
                    if (el) el.style.display = 'none';
                }, 300);
            }
        }
        
        // 渲染当前步骤
        function renderStep() {
            const prevBtn = document.getElementById('scene3-prev-btn');
            const nextBtn = document.getElementById('scene3-next-btn');
            
            // 更新按钮状态
            prevBtn.disabled = (currentStep === 0);
            if (currentStep === TOTAL_STEPS - 1) {
                nextBtn.textContent = '完成 ✓';
                nextBtn.disabled = true;
            } else {
                nextBtn.textContent = '下一步 ▶';
                nextBtn.disabled = false;
            }
            
            // 根据步骤控制元素显示
            switch(currentStep) {
                case 0:
                    // 初始状态：只显示基础元素
                    break;
                    
                case 1:
                    // 显示 C 点经线高亮和 C 点地方时
                    show(document.getElementById('scene3-line-c-meridian'));
                    show(document.getElementById('scene3-label-c-time'));
                    break;
                    
                case 2:
                    // 显示西半球范围
                    show(document.getElementById('scene3-path-west-hemisphere'));
                    show(document.getElementById('scene3-label-west-hemisphere'));
                    break;
                    
                case 3:
                    // 显示时间计算框和步骤 1 内容
                    show(document.getElementById('scene3-rect-calc-box'));
                    show(document.getElementById('scene3-calc-content'));
                    show(document.getElementById('scene3-calc-step1'));
                    show(document.getElementById('scene3-calc-step1-1'));
                    show(document.getElementById('scene3-calc-step1-2'));
                    show(document.getElementById('scene3-calc-step1-3'));
                    break;
                    
                case 4:
                    // 显示 120°E 经线、经度差弧线和步骤 2 内容
                    show(document.getElementById('scene3-line-120e'));
                    show(document.getElementById('scene3-label-120e'));
                    show(document.getElementById('scene3-path-longitude-diff'));
                    show(document.getElementById('scene3-label-longitude-diff'));
                    show(document.getElementById('scene3-calc-step2'));
                    show(document.getElementById('scene3-calc-step2-1'));
                    show(document.getElementById('scene3-calc-step2-2'));
                    break;
                    
                case 5:
                    // 显示步骤 3、4 内容和结论
                    show(document.getElementById('scene3-calc-step3'));
                    show(document.getElementById('scene3-calc-step3-1'));
                    show(document.getElementById('scene3-calc-step4'));
                    show(document.getElementById('scene3-calc-step4-1'));
                    show(document.getElementById('scene3-calc-step4-2'));
                    show(document.getElementById('scene3-calc-conclusion'));
                    break;
            }
        }
        
        // 事件绑定
        document.getElementById('scene3-prev-btn').addEventListener('click', function() {
            if (currentStep > 0) {
                currentStep--;
                renderStep();
            }
        });
        
        document.getElementById('scene3-next-btn').addEventListener('click', function() {
            if (currentStep < TOTAL_STEPS - 1) {
                currentStep++;
                renderStep();
            }
        });
        
        // 初始化
        renderStep();
    })();
    </script>
