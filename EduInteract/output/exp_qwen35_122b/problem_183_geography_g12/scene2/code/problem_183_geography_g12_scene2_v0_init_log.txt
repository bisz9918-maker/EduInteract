    <script>
    (function() {
        // Scene 2 实现：地球公转与太阳直射点移动
        
        // 获取 DOM 元素
        const svg = document.getElementById('scene2-svg');
        const earth = document.getElementById('scene2-earth');
        const axis = document.getElementById('scene2-axis');
        const latPoint = document.getElementById('scene2-lat-point');
        const heightPoint = document.getElementById('scene2-height-point');
        const dayBar = document.getElementById('scene2-day-bar');
        const nightBar = document.getElementById('scene2-night-bar');
        const slider = document.getElementById('scene2-slider');
        const valueDisplay = document.getElementById('scene2-value-display');
        
        // 状态变量
        let earthPosition = 0; // 0-100，0=A(夏至), 25=B(秋分), 50=C(冬至), 75=D(春分), 100=A
        
        // 地球位置计算函数
        function getEarthPosition(p) {
            // p: 0-100, 0=A(夏至，左侧), 25=B(秋分，下方), 50=C(冬至，右侧), 75=D(春分，上方)
            const angle = 2 * Math.PI * (p / 100 - 0.25);
            return {
                x: 160 + 140 * Math.cos(angle),
                y: 250 + 100 * Math.sin(angle)
            };
        }
        
        // 太阳直射点纬度计算（正为北，负为南）
        function getDeclination(p) {
            return 23.5 * Math.sin(2 * Math.PI * p / 100);
        }
        
        // 北半球昼长计算（小时）
        function getDayLength(p) {
            return 12 + 3 * Math.sin(2 * Math.PI * p / 100);
        }
        
        // 南回归线以南正午太阳高度（度）
        function getNoonHeight(p) {
            const declination = getDeclination(p);
            return 90 - Math.abs(declination + 23.5);
        }
        
        // 获取节气名称
        function getSeasonName(p) {
            if (p < 12.5) return '夏至 (A)';
            if (p < 37.5) return '秋分 (B)';
            if (p < 62.5) return '冬至 (C)';
            if (p < 87.5) return '春分 (D)';
            return '夏至 (A)';
        }
        
        // 更新所有视觉元素
        function updateDisplay() {
            const pos = getEarthPosition(earthPosition);
            const declination = getDeclination(earthPosition);
            const dayLength = getDayLength(earthPosition);
            const nightLength = 24 - dayLength;
            const height = getNoonHeight(earthPosition);
            
            // 更新地球位置
            earth.setAttribute('cx', pos.x);
            earth.setAttribute('cy', pos.y);
            
            // 更新地轴（保持倾斜方向）
            const angle = 2 * Math.PI * (earthPosition / 100 - 0.25);
            const axisAngle = angle + Math.PI / 2; // 地轴垂直于轨道切线
            const axisLength = 20;
            axis.setAttribute('x1', pos.x - axisLength * Math.cos(axisAngle));
            axis.setAttribute('y1', pos.y - axisLength * Math.sin(axisAngle));
            axis.setAttribute('x2', pos.x + axisLength * Math.cos(axisAngle));
            axis.setAttribute('y2', pos.y + axisLength * Math.sin(axisAngle));
            
            // 更新太阳直射点曲线上的点
            const latY = 80 + (declination / 23.5) * 27.5; // 80 是中心，27.5 是半高
            latPoint.setAttribute('cx', 340 + (earthPosition / 100) * 440);
            latPoint.setAttribute('cy', latY);
            
            // 更新正午太阳高度曲线上的点
            const heightY = 490 - (height / 90) * 110; // 490 是底部，110 是高度范围
            heightPoint.setAttribute('cx', 340 + (earthPosition / 100) * 440);
            heightPoint.setAttribute('cy', heightY);
            
            // 更新昼长条形（最大 15h 对应 50px 高度）
            const dayHeight = (dayLength / 15) * 50;
            dayBar.setAttribute('y', 250 - dayHeight);
            dayBar.setAttribute('height', dayHeight);
            
            // 更新夜长条形
            const nightHeight = (nightLength / 15) * 50;
            nightBar.setAttribute('y', 250 - nightHeight);
            nightBar.setAttribute('height', nightHeight);
            
            // 更新数值显示
            const seasonName = getSeasonName(earthPosition);
            valueDisplay.textContent = `${seasonName} | 直射点：${declination.toFixed(1)}° | 昼长：${dayLength.toFixed(1)}h | 高度：${height.toFixed(1)}°`;
        }
        
        // 滑块事件监听
        slider.addEventListener('input', function(e) {
            earthPosition = parseFloat(e.target.value);
            updateDisplay();
        });
        
        // 初始化显示
        updateDisplay();
    })();
    </script>
