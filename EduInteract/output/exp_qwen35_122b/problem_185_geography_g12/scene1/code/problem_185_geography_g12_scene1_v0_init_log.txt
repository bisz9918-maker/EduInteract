    <script>
    (function() {
        // Scene 1 实现
        // 布局坐标表（调整后）：
        // | 元素 | 类型 | 位置/范围 | 大小 | 标签偏移方向 | 备注 |
        // |------|------|-----------|------|-------------|------|
        // | 纬线圈 | 圆形 | 圆心 (400,250) | 半径 180 | - | 黑色粗线 |
        // | 中心点（北极） | 实心圆 | (400,250) | 6×6 | - | 北极点投影 |
        // | 45°E 经线 | 线段 | (400,250)-(400,70) | 4×180 | - | 垂直虚线 |
        // | 45°W 经线 | 线段 | (400,250)-(580,250) | 4×180 | - | 水平虚线 |
        // | 晨线 | 线段 | (400,250)-(235,207) | 4×180 | - | 蓝色实线，与水平夹角 15° |
        // | 昏线 | 线段 | (400,250)-(565,207) | 4×180 | - | 红色实线，与水平夹角 15° |
        // | A 点 | 实心圆 | (235,207) | 8×8 | 左上方 | 晨线与纬线圈交点 |
        // | B 点 | 实心圆 | (565,207) | 8×8 | 右上方 | 昏线与纬线圈交点 |
        // | 昼弧 | 圆弧 | A 到 B 上半部分 | 半径 180 | - | 蓝色填充 20% 透明度 |
        // | 夜弧 | 圆弧 | A 到 B 下半部分 | 半径 180 | - | 黑色填充 10% 透明度 |
        // | 晨线 15°角弧 | 小圆弧 | 圆心处晨线与水平线之间 | 半径 30 | - | 灰色细线 |
        // | 昏线 15°角弧 | 小圆弧 | 圆心处昏线与水平线之间 | 半径 30 | - | 灰色细线 |
        
        const currentStepMax = 4;
        let currentStep = 0;
        
        // 元素显示数组 - 每个步骤需要显示的元素 ID
        const stepElements = {
            0: ['scene1-circle', 'scene1-center', 'scene1-line-45E', 'scene1-line-45W', 'scene1-label-30N', 'scene1-label-45E', 'scene1-label-45W'],
            1: ['scene1-morning-line', 'scene1-evening-line', 'scene1-A-point', 'scene1-B-point', 'scene1-label-A', 'scene1-label-B', 'scene1-label-morning', 'scene1-label-evening'],
            2: ['scene1-day-arc', 'scene1-night-arc', 'scene1-label-day', 'scene1-label-night'],
            3: ['scene1-angle-morning', 'scene1-angle-evening', 'scene1-label-angle-left', 'scene1-label-angle-right'],
            4: ['scene1-A-highlight']
        };
        
        // 工具函数：显示元素
        function show(elId) {
            const el = document.getElementById(elId);
            if (el) {
                el.style.display = 'block';
                el.style.opacity = '1';
            }
        }
        
        // 工具函数：隐藏元素
        function hide(elId) {
            const el = document.getElementById(elId);
            if (el) {
                el.style.opacity = '0';
                setTimeout(() => {
                    if (el.style.opacity === '0') {
                        el.style.display = 'none';
                    }
                }, 300);
            }
        }
        
        // 更新显示状态
        function updateDisplay() {
            // 隐藏所有步骤元素
            for (let step = 0; step <= currentStepMax; step++) {
                stepElements[step].forEach(elId => hide(elId));
            }
            
            // 显示当前步骤及之前步骤的所有元素
            for (let step = 0; step <= currentStep; step++) {
                stepElements[step].forEach(elId => show(elId));
            }
            
            // 更新按钮状态
            document.getElementById('scene1-prev-btn').disabled = (currentStep === 0);
            document.getElementById('scene1-next-btn').disabled = (currentStep === currentStepMax);
            
            // 更新提示文字
            const hint = document.getElementById('scene1-hint');
            if (currentStep === 4) {
                hint.style.opacity = '1';
            } else {
                hint.style.opacity = '0';
            }
        }
        
        // 按钮事件绑定
        document.getElementById('scene1-prev-btn').addEventListener('click', function() {
            if (currentStep > 0) {
                currentStep--;
                updateDisplay();
            }
        });
        
        document.getElementById('scene1-next-btn').addEventListener('click', function() {
            if (currentStep < currentStepMax) {
                currentStep++;
                updateDisplay();
            }
        });
        
        // 初始化显示
        updateDisplay();
    })();
    </script>
