    <script>
    (function() {
        // Scene 2: 太阳直射点坐标推导图
        // 布局坐标表对照：
        // 地球椭圆：中心 (250,200)，半径 100
        // 赤道线：(150,200)-(350,200)
        // 地轴：(250,80)-(250,320)
        // 70°N 纬线：(180,130)-(320,130)，标签 (345,130) - 移到右侧外部
        // 66.5°N 纬线：(165,115)-(335,115)，标签 (345,115) - 移到右侧外部
        // 晨昏线：(120,130)-(380,270)，标签 (390,200)
        // 切点：(250,130)
        // 直射点：(250,160)，标签 (265,145) - 移到上方
        // 直射点经线：(250,160)-(250,200)，标签 (250,225) - 移到下方
        // 20°N 标签：(345,160) - 移到右侧外部
        // 纬度角标注：(220,180) - 移到左侧
        // 推导框：step1(450,50), step2(450,140), step3(450,230)
        // 控件栏：prev(250,455), next(450,455)
        
        const scene2 = {
            currentStep: 0,
            totalSteps: 3,
            
            elements: {
                prevBtn: document.getElementById('scene2-prev-btn'),
                nextBtn: document.getElementById('scene2-next-btn'),
                step1Box: document.getElementById('scene2-step1-box'),
                step2Box: document.getElementById('scene2-step2-box'),
                step3Box: document.getElementById('scene2-step3-box'),
                sunRays: document.getElementById('scene2-sun-rays'),
                group20n: document.getElementById('scene2-20n-group'),
                subSolarPoint: document.getElementById('scene2-sub-solar-point')
            },
            
            init: function() {
                this.elements.prevBtn.addEventListener('click', () => this.prevStep());
                this.elements.nextBtn.addEventListener('click', () => this.nextStep());
                this.updateDisplay();
            },
            
            prevStep: function() {
                if (this.currentStep > 0) {
                    this.currentStep--;
                    this.updateDisplay();
                }
            },
            
            nextStep: function() {
                if (this.currentStep < this.totalSteps) {
                    this.currentStep++;
                    this.updateDisplay();
                }
            },
            
            updateDisplay: function() {
                const el = this.elements;
                
                // 更新按钮状态
                el.prevBtn.disabled = (this.currentStep === 0);
                el.nextBtn.disabled = (this.currentStep === this.totalSteps);
                
                // 更新推导框显示
                if (this.currentStep >= 1) {
                    this.showElement(el.step1Box);
                } else {
                    this.hideElement(el.step1Box);
                }
                
                if (this.currentStep >= 2) {
                    this.showElement(el.step2Box);
                } else {
                    this.hideElement(el.step2Box);
                }
                
                if (this.currentStep >= 3) {
                    this.showElement(el.step3Box);
                } else {
                    this.hideElement(el.step3Box);
                }
                
                // 更新太阳光线 (步骤 1+)
                if (this.currentStep >= 1) {
                    this.showElement(el.sunRays);
                } else {
                    this.hideElement(el.sunRays);
                }
                
                // 更新 20°N 相关元素 (步骤 2+)
                if (this.currentStep >= 2) {
                    this.showElement(el.group20n);
                } else {
                    this.hideElement(el.group20n);
                }
                
                // 步骤 3 高亮直射点
                if (this.currentStep >= 3) {
                    el.subSolarPoint.setAttribute('r', '6');
                    el.subSolarPoint.setAttribute('fill', '#d32f2f');
                } else {
                    el.subSolarPoint.setAttribute('r', '4');
                    el.subSolarPoint.setAttribute('fill', '#c62828');
                }
            },
            
            showElement: function(el) {
                el.style.display = 'block';
                el.style.opacity = '1';
            },
            
            hideElement: function(el) {
                el.style.display = 'none';
                el.style.opacity = '0';
            }
        };
        
        scene2.init();
    })();
    </script>
