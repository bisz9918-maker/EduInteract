    <script>
    (function() {
        // Scene 1 实现
        // 布局坐标表对照：
        // 热水器 A: 地面 (60,180)-(160,180), 集热板 (60,180)-(110,155), 倾角 6°34′
        // 热水器 B: 地面 (230,180)-(330,180), 集热板 (230,180)-(285,140), 倾角 26°34′
        // 热水器 C: 地面 (400,180)-(500,180), 集热板 (400,180)-(458,135), 倾角 28°34′
        // 热水器 D: 地面 (570,180)-(670,180), 集热板 (570,180)-(620,150), 倾角 13°34′
        
        const scene1 = {
            currentStep: 0,
            totalSteps: 5,
            
            // 计算结果数据
            calcData: {
                a: { angle: '6°34′', height: '83°26′' },
                b: { angle: '26°34′', height: '63°26′' },
                c: { angle: '28°34′', height: '61°26′' },
                d: { angle: '13°34′', height: '76°26′' }
            },
            
            // 工具函数
            show: function(el) {
                if (el) {
                    el.classList.remove('scene1-hidden');
                    el.classList.add('scene1-fade-in');
                }
            },
            
            hide: function(el) {
                if (el) {
                    el.classList.add('scene1-hidden');
                    el.classList.remove('scene1-fade-in');
                }
            },
            
            // 更新显示状态
            updateDisplay: function() {
                const s = this.currentStep;
                
                // 步骤 0：初始状态 - 只显示热水器和倾角
                if (s === 0) {
                    this.show(document.getElementById('scene1-angle-a'));
                    this.show(document.getElementById('scene1-angle-arc-a'));
                    this.show(document.getElementById('scene1-angle-b'));
                    this.show(document.getElementById('scene1-angle-arc-b'));
                    this.show(document.getElementById('scene1-angle-c'));
                    this.show(document.getElementById('scene1-angle-arc-c'));
                    this.show(document.getElementById('scene1-angle-d'));
                    this.show(document.getElementById('scene1-angle-arc-d'));
                    
                    this.hide(document.getElementById('scene1-sun-ray-a'));
                    this.hide(document.getElementById('scene1-sun-ray-b'));
                    this.hide(document.getElementById('scene1-sun-ray-c'));
                    this.hide(document.getElementById('scene1-sun-ray-d'));
                    this.hide(document.getElementById('scene1-calc-box'));
                    this.hide(document.getElementById('scene1-bar-chart'));
                    this.hide(document.getElementById('scene1-sort-label'));
                    
                    // 移除所有高亮
                    document.getElementById('scene1-heater-a-panel').classList.remove('scene1-highlight');
                    document.getElementById('scene1-heater-b-panel').classList.remove('scene1-highlight');
                    document.getElementById('scene1-heater-c-panel').classList.remove('scene1-highlight');
                    document.getElementById('scene1-heater-d-panel').classList.remove('scene1-highlight');
                }
                
                // 步骤 1：高亮 A 地计算
                if (s >= 1) {
                    this.show(document.getElementById('scene1-sun-ray-a'));
                    this.show(document.getElementById('scene1-calc-box'));
                    document.getElementById('scene1-calc-result').textContent = 'θ=6°34′ → H=90°-6°34′=83°26′';
                    document.getElementById('scene1-heater-a-panel').classList.add('scene1-highlight');
                }
                if (s < 1) {
                    this.hide(document.getElementById('scene1-sun-ray-a'));
                    this.hide(document.getElementById('scene1-calc-box'));
                    document.getElementById('scene1-heater-a-panel').classList.remove('scene1-highlight');
                }
                
                // 步骤 2：依次高亮 B、C、D
                if (s >= 2) {
                    this.show(document.getElementById('scene1-sun-ray-b'));
                    document.getElementById('scene1-calc-result').textContent = 'θ=26°34′ → H=90°-26°34′=63°26′';
                    document.getElementById('scene1-heater-a-panel').classList.remove('scene1-highlight');
                    document.getElementById('scene1-heater-b-panel').classList.add('scene1-highlight');
                }
                if (s < 2) {
                    this.hide(document.getElementById('scene1-sun-ray-b'));
                    document.getElementById('scene1-heater-b-panel').classList.remove('scene1-highlight');
                }
                
                if (s >= 3) {
                    this.show(document.getElementById('scene1-sun-ray-c'));
                    this.show(document.getElementById('scene1-sun-ray-d'));
                    document.getElementById('scene1-calc-result').textContent = 'θ=28°34′ → H=90°-28°34′=61°26′';
                    document.getElementById('scene1-heater-b-panel').classList.remove('scene1-highlight');
                    document.getElementById('scene1-heater-c-panel').classList.add('scene1-highlight');
                }
                if (s < 3) {
                    this.hide(document.getElementById('scene1-sun-ray-c'));
                    this.hide(document.getElementById('scene1-sun-ray-d'));
                    document.getElementById('scene1-heater-c-panel').classList.remove('scene1-highlight');
                }
                
                // 步骤 3：显示条形图
                if (s >= 4) {
                    this.show(document.getElementById('scene1-bar-chart'));
                    document.getElementById('scene1-calc-result').textContent = 'θ=13°34′ → H=90°-13°34′=76°26′';
                    document.getElementById('scene1-heater-c-panel').classList.remove('scene1-highlight');
                    document.getElementById('scene1-heater-d-panel').classList.add('scene1-highlight');
                }
                if (s < 4) {
                    this.hide(document.getElementById('scene1-bar-chart'));
                    document.getElementById('scene1-heater-d-panel').classList.remove('scene1-highlight');
                }
                
                // 步骤 4：显示排序结论
                if (s >= 5) {
                    this.show(document.getElementById('scene1-sort-label'));
                    document.getElementById('scene1-heater-d-panel').classList.remove('scene1-highlight');
                    // 条形图闪烁高亮效果
                    const bars = ['scene1-bar-a', 'scene1-bar-d', 'scene1-bar-b', 'scene1-bar-c'];
                    let idx = 0;
                    const flashInterval = setInterval(() => {
                        bars.forEach((id, i) => {
                            const bar = document.getElementById(id);
                            if (i === idx) {
                                bar.style.opacity = '1';
                                bar.style.filter = 'brightness(1.3)';
                            } else {
                                bar.style.opacity = '0.7';
                                bar.style.filter = 'none';
                            }
                        });
                        idx = (idx + 1) % bars.length;
                    }, 500);
                    // 5 秒后停止闪烁
                    setTimeout(() => clearInterval(flashInterval), 5000);
                }
                if (s < 5) {
                    this.hide(document.getElementById('scene1-sort-label'));
                    ['scene1-bar-a', 'scene1-bar-d', 'scene1-bar-b', 'scene1-bar-c'].forEach(id => {
                        const bar = document.getElementById(id);
                        bar.style.opacity = '1';
                        bar.style.filter = 'none';
                    });
                }
                
                // 更新按钮状态
                document.getElementById('scene1-btn-prev').disabled = (s === 0);
                const nextBtn = document.getElementById('scene1-btn-next');
                if (s === 4) {
                    nextBtn.textContent = '完成';
                } else {
                    nextBtn.textContent = '下一步 ▶';
                }
                nextBtn.disabled = (s >= 4);
            },
            
            // 初始化
            init: function() {
                document.getElementById('scene1-btn-prev').addEventListener('click', () => {
                    if (this.currentStep > 0) {
                        this.currentStep--;
                        this.updateDisplay();
                    }
                });
                
                document.getElementById('scene1-btn-next').addEventListener('click', () => {
                    if (this.currentStep < 4) {
                        this.currentStep++;
                        this.updateDisplay();
                    }
                });
                
                // 初始显示
                this.updateDisplay();
            }
        };
        
        // 启动
        scene1.init();
    })();
    </script>
