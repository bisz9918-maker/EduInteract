    <script>
    (function() {
        // Scene 1: 三视图与几何体组成关系
        // 布局坐标表（实现对照依据）：
        // 正视图：(60,40)-(200,140), 三角形顶点 (130,40), 底边 (60,140)-(200,140)
        // 侧视图：(240,40)-(380,140), 三角形顶点 (310,40), 底边 (240,140)-(380,140)
        // 俯视图：(60,260)-(300,420), 半圆圆心 (180,340) r=60, 三角形 (60,340)-(180,340)-(180,400)
        // 立体图：顶点 (585,80), 底面中心 (585,320), 半圆 rx=80 ry=40, 三角形 (505,320)-(665,320)-(585,360)

        const scene1 = {
            currentStep: 0,
            totalSteps: 5,

            // 元素显示/隐藏控制
            elements: {
                // 步骤 0 基础元素（始终显示）
                baseElements: [
                    'scene1-front-view', 'scene1-side-view', 'scene1-top-view',
                    'scene1-projection-lines',
                    'scene1-front-label', 'scene1-side-label', 'scene1-top-label',
                    'scene1-front-base-left', 'scene1-front-base-right', 'scene1-front-height',
                    'scene1-side-base-left', 'scene1-side-base-right', 'scene1-side-height',
                    'scene1-top-base-left', 'scene1-top-base-right', 'scene1-top-height'
                ],
                // 步骤 1 新增
                step1Elements: ['scene1-semicircle', 'scene1-semicone-label'],
                // 步骤 2 新增
                step2Elements: ['scene1-triangle-base', 'scene1-prism-label'],
                // 步骤 3 新增
                step3Elements: ['scene1-3d-view', 'scene1-3d-vertex'],
                // 步骤 4 新增
                step4Elements: ['scene1-3d-height']
            },

            init: function() {
                this.updateScene();
                document.getElementById('scene1-prev').addEventListener('click', () => this.prevStep());
                document.getElementById('scene1-next').addEventListener('click', () => this.nextStep());
            },

            updateScene: function() {
                // 显示基础元素
                this.elements.baseElements.forEach(id => this.show(id));

                // 根据步骤显示对应元素
                if (this.currentStep >= 1) {
                    this.elements.step1Elements.forEach(id => this.show(id));
                }
                if (this.currentStep >= 2) {
                    this.elements.step2Elements.forEach(id => this.show(id));
                }
                if (this.currentStep >= 3) {
                    this.elements.step3Elements.forEach(id => this.show(id));
                    // 显示高亮区域
                    this.show('scene1-3d-semicone-highlight');
                    this.show('scene1-3d-prism-highlight');
                }
                if (this.currentStep >= 4) {
                    this.elements.step4Elements.forEach(id => this.show(id));
                }

                // 更新按钮状态
                document.getElementById('scene1-prev').disabled = (this.currentStep === 0);
                document.getElementById('scene1-next').disabled = (this.currentStep === this.totalSteps - 1);

                // 更新步骤指示器
                document.getElementById('scene1-step-indicator').textContent = '步骤 ' + (this.currentStep + 1) + '/5';
            },

            show: function(id) {
                const el = document.getElementById(id);
                if (el) {
                    el.style.display = 'block';
                    el.style.opacity = '1';
                }
            },

            hide: function(id) {
                const el = document.getElementById(id);
                if (el) {
                    el.style.display = 'none';
                    el.style.opacity = '0';
                }
            },

            prevStep: function() {
                if (this.currentStep > 0) {
                    this.currentStep--;
                    this.updateScene();
                }
            },

            nextStep: function() {
                if (this.currentStep < this.totalSteps - 1) {
                    this.currentStep++;
                    this.updateScene();
                }
            }
        };

        scene1.init();
    })();
    </script>
