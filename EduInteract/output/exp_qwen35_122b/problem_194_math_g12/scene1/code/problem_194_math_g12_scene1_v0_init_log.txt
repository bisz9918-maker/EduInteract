    <script>
    (function() {
        // Scene 1: 四面体 ABCD 的三维结构与已知条件标注
        // 布局坐标表（修复后）:
        // | 元素 | 类型 | 位置/范围 | 大小 | 标签偏移方向 | 备注 |
        // | 顶点 A | 圆点 | (200,380) | 6×6 | 左下 | 左下角 |
        // | 顶点 B | 圆点 | (420,360) | 6×6 | 右下 | 右下前方 |
        // | 顶点 C | 圆点 | (580,220) | 6×6 | 右上 | 右上后方 |
        // | 顶点 D | 圆点 | (200,120) | 6×6 | 左上 | 左上角 |
        // | 标注 BC=2 | text | (550,250) | 60×18 | - | 棱 BC 外侧 |
        // | 标注 AD=2c | text | (125,250) | 60×18 | - | 棱 AD 左侧 |
        // | 标注 AB+BD=2a | text | (360,360) | 80×18 | - | 椭圆条件 |
        // | 标注 AC+CD=2a | text | (450,270) | 80×18 | - | 椭圆条件 |
        // | 垂直符号 | 直角 | (342,250) | 15×15 | - | AD 与 BC 公垂处 |
        // | 体积公式 | text+rect | (530,350) | 250×100 | - | 右下角提示框 |

        const TOTAL_STEPS = 5;  // 步骤 0-4，共 5 步
        let currentStep = 0;

        // 工具函数：显示元素（同时处理 display 和 opacity）
        function show(el) {
            if (el) {
                el.style.display = '';
                el.style.opacity = '1';
            }
        }

        // 工具函数：隐藏元素（同时处理 display 和 opacity）
        function hide(el) {
            if (el) {
                el.style.opacity = '0';
                setTimeout(() => {
                    if (el.style.opacity === '0') {
                        el.style.display = 'none';
                    }
                }, 300);
            }
        }

        // 更新当前步骤的显示状态
        function updateStep() {
            // 隐藏所有步骤组（除了 step0 始终显示）
            for (let i = 1; i < TOTAL_STEPS; i++) {
                const stepGroup = document.getElementById('scene1-step' + i);
                if (stepGroup) {
                    if (i <= currentStep) {
                        show(stepGroup);
                    } else {
                        hide(stepGroup);
                    }
                }
            }

            // 更新按钮状态
            const prevBtn = document.getElementById('scene1-btn-prev');
            const nextBtn = document.getElementById('scene1-btn-next');
            const indicator = document.getElementById('scene1-step-indicator');

            prevBtn.disabled = (currentStep === 0);
            nextBtn.disabled = (currentStep === TOTAL_STEPS - 1);
            indicator.textContent = '步骤 ' + (currentStep + 1) + '/' + TOTAL_STEPS;
        }

        // 初始化
        function init() {
            // 确保 step0 显示，其他步骤隐藏
            const step0 = document.getElementById('scene1-step0');
            show(step0);

            for (let i = 1; i < TOTAL_STEPS; i++) {
                const stepGroup = document.getElementById('scene1-step' + i);
                hide(stepGroup);
            }

            // 绑定按钮事件
            const prevBtn = document.getElementById('scene1-btn-prev');
            const nextBtn = document.getElementById('scene1-btn-next');

            prevBtn.addEventListener('click', function() {
                if (currentStep > 0) {
                    currentStep--;
                    updateStep();
                }
            });

            nextBtn.addEventListener('click', function() {
                if (currentStep < TOTAL_STEPS - 1) {
                    currentStep++;
                    updateStep();
                }
            });

            // 初始更新
            updateStep();
        }

        // 页面加载完成后初始化
        if (document.readyState === 'loading') {
            document.addEventListener('DOMContentLoaded', init);
        } else {
            init();
        }
    })();
    </script>
