    <script>
    (function() {
        // Scene 2 实现：四面体体积最大值的截面分析
        // 坐标表对照：
        // 椭圆：中心 (400,250)，rx=120, ry=89
        // 焦点 A：(320,250)，焦点 D：(480,250)
        // 长半轴 a=120，焦距 c=80，短半轴 b=√(120²-80²)=√8000≈89.4
        
        // 椭圆参数
        const centerX = 400;
        const centerY = 250;
        const rx = 120;  // 长半轴 a
        const ry = 89;   // 短半轴 b ≈ √(a²-c²) = √(120²-80²)
        const focusDist = 80;  // 焦距 c
        
        // BC 长度
        const BC_LENGTH = 2;
        
        // 获取 DOM 元素
        const pointB = document.getElementById('scene2-pointB');
        const pointC = document.getElementById('scene2-pointC');
        const labelB = document.getElementById('scene2-labelB');
        const labelC = document.getElementById('scene2-labelC');
        const segmentBC = document.getElementById('scene2-segmentBC');
        const labelBC = document.getElementById('scene2-labelBC');
        const distanceD = document.getElementById('scene2-distanceD');
        const labelD_val = document.getElementById('scene2-labelD_val');
        const slider = document.getElementById('scene2-slider');
        const valueDisplay = document.getElementById('scene2-value');
        const maxHighlight = document.getElementById('scene2-maxHighlight');
        
        // 计算 B 点在椭圆上的位置（参数方程）
        function getBPoint(angle) {
            const rad = angle * Math.PI / 180;
            return {
                x: centerX + rx * Math.cos(rad),
                y: centerY + ry * Math.sin(rad)
            };
        }
        
        // 计算 C 点位置：保持 BC=2 且 BC⊥AD（即 BC 垂直于 x 轴）
        function getCPoint(B) {
            const xOffset = B.x - centerX;
            const ratio = xOffset * xOffset / (rx * rx);
            
            if (ratio > 1) {
                return { x: B.x, y: centerY - (B.y - centerY) };
            }
            
            const yOffset = ry * Math.sqrt(1 - ratio);
            const C_y_below = centerY + yOffset;
            const C_y_above = centerY - yOffset;
            
            if (Math.abs(C_y_below - B.y) < Math.abs(C_y_above - B.y)) {
                return { x: B.x, y: C_y_below };
            } else {
                return { x: B.x, y: C_y_above };
            }
        }
        
        // 计算距离 d（BC 中点到中心的距离）
        function calculateDistanceD(B, C) {
            const midY = (B.y + C.y) / 2;
            return Math.abs(midY - centerY);
        }
        
        // 更新所有元素位置
        function updateScene() {
            const angle = parseInt(slider.value);
            const B = getBPoint(angle);
            const C = getCPoint(B);
            const d = calculateDistanceD(B, C);
            
            // 更新 B 点位置
            pointB.setAttribute('cx', B.x);
            pointB.setAttribute('cy', B.y);
            // 根据 B 点位置动态调整标签位置，确保在椭圆外且远离其他标签
            if (B.x > centerX + rx - 10) {
                labelB.setAttribute('x', B.x + 20);
            } else if (B.x < centerX - rx + 10) {
                labelB.setAttribute('x', B.x - 35);
            } else {
                labelB.setAttribute('x', B.x + 25);
            }
            if (B.y > centerY + ry - 10) {
                labelB.setAttribute('y', B.y + 15);
            } else if (B.y < centerY - ry + 10) {
                labelB.setAttribute('y', B.y - 5);
            } else {
                labelB.setAttribute('y', B.y - 5);
            }
            
            // 更新 C 点位置
            pointC.setAttribute('cx', C.x);
            pointC.setAttribute('cy', C.y);
            // 根据 C 点位置动态调整标签位置
            if (C.x > centerX + rx - 10) {
                labelC.setAttribute('x', C.x + 20);
            } else if (C.x < centerX - rx + 10) {
                labelC.setAttribute('x', C.x - 35);
            } else {
                labelC.setAttribute('x', C.x + 25);
            }
            if (C.y > centerY + ry - 10) {
                labelC.setAttribute('y', C.y + 15);
            } else if (C.y < centerY - ry + 10) {
                labelC.setAttribute('y', C.y - 5);
            } else {
                labelC.setAttribute('y', C.y + 15);
            }
            
            // 更新线段 BC
            segmentBC.setAttribute('x1', B.x);
            segmentBC.setAttribute('y1', B.y);
            segmentBC.setAttribute('x2', C.x);
            segmentBC.setAttribute('y2', C.y);
            
            // 更新 BC=2 标签位置（BC 右侧，远离 B、C 标签）
            const midX = (B.x + C.x) / 2;
            const midY = (B.y + C.y) / 2;
            labelBC.setAttribute('x', Math.max(B.x, C.x) + 40);
            labelBC.setAttribute('y', midY + 5);
            
            // 更新距离 d（BC 中点到中心的连线）
            distanceD.setAttribute('x1', midX);
            distanceD.setAttribute('y1', midY);
            distanceD.setAttribute('x2', centerX);
            distanceD.setAttribute('y2', centerY);
            
            // 更新距离 d 标签位置（左侧，远离椭圆）
            labelD_val.setAttribute('x', centerX - 50);
            labelD_val.setAttribute('y', midY - 5);
            
            // 更新数值显示
            valueDisplay.textContent = `d = ${d.toFixed(2)}`;
            
            // 检查是否接近最大值（B 点在短轴端点附近）
            const maxB = ry;
            const isMax = Math.abs(d - maxB) < 5;
            
            if (isMax) {
                maxHighlight.style.display = 'block';
                maxHighlight.style.opacity = '0.2';
                labelD_val.textContent = 'd_max';
            } else {
                maxHighlight.style.display = 'none';
                maxHighlight.style.opacity = '0';
                labelD_val.textContent = 'd';
            }
        }
        
        // 事件绑定
        slider.addEventListener('input', updateScene);
        
        // 初始化
        updateScene();
    })();
    </script>
