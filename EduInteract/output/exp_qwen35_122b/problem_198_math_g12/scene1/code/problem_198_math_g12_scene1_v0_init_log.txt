    <script>
    (function() {
        // Scene 1: 圆的切线性质与四边形 PADB 面积计算
        // 分步序列：
        // 步骤 0：基础图形（圆 O、点 P、O、C、D、线段 PO、标签、标注 PC=1、CD=1）
        // 步骤 1：添加切线 PA、PB，切点 A、B，标签 A、B，直角符号
        // 步骤 2：添加半径 OA、OB
        // 步骤 3：添加弦 AB、线段 AD、BD，四边形 PADB 填充
        // 步骤 4：添加标注 PO=1.5、OA=0.5、PA=√2

        const currentStepMax = 4;
        let currentStep = 0;

        // 元素按步骤分组
        const stepElements = {
            0: [
                'scene1-circle-o',
                'scene1-line-po',
                'scene1-point-p', 'scene1-point-o', 'scene1-point-c', 'scene1-point-d',
                'scene1-label-p', 'scene1-label-o', 'scene1-label-c', 'scene1-label-d',
                'scene1-annot-pc', 'scene1-annot-cd'
            ],
            1: [
                'scene1-line-pa', 'scene1-line-pb',
                'scene1-point-a', 'scene1-point-b',
                'scene1-label-a', 'scene1-label-b',
                'scene1-rightangle-a', 'scene1-rightangle-b'
            ],
            2: [
                'scene1-line-oa', 'scene1-line-ob'
            ],
            3: [
                'scene1-line-ab', 'scene1-line-ad', 'scene1-line-bd',
                'scene1-quad-padb'
            ],
            4: [
                'scene1-annot-po', 'scene1-annot-oa', 'scene1-annot-pa'
            ]
        };

        function showElement(id) {
            const el = document.getElementById(id);
            if (el) {
                el.style.display = 'block';
                el.style.opacity = '0';
                setTimeout(() => {
                    el.style.opacity = '1';
                }, 10);
            }
        }

        function hideElement(id) {
            const el = document.getElementById(id);
            if (el) {
                el.style.display = 'none';
                el.style.opacity = '1';
            }
        }

        function updateScene() {
            // 隐藏所有步骤 1-4 的元素
            for (let step = 1; step <= currentStepMax; step++) {
                stepElements[step].forEach(id => hideElement(id));
            }

            // 显示当前步骤及之前的所有元素
            for (let step = 0; step <= currentStep; step++) {
                stepElements[step].forEach(id => showElement(id));
            }

            // 更新按钮状态
            document.getElementById('scene1-btn-prev').disabled = (currentStep === 0);
            document.getElementById('scene1-btn-next').disabled = (currentStep === currentStepMax);
        }

        document.getElementById('scene1-btn-prev').addEventListener('click', function() {
            if (currentStep > 0) {
                currentStep--;
                updateScene();
            }
        });

        document.getElementById('scene1-btn-next').addEventListener('click', function() {
            if (currentStep < currentStepMax) {
                currentStep++;
                updateScene();
            }
        });

        // 初始化
        updateScene();
    })();
    </script>
