    <script>
    (function() {
        // 场景 1：带电粒子在复合场中的运动分解
        // 布局坐标表对照：
        // | 元素 | 位置 | 备注 |
        // | A 点 | (150,340) | 黑色圆点 |
        // | B 点 | (550,180) | 黑色圆点 |
        // | 电场线 y | 100/180/260/340/420 | 5 条水平线 |
        // | v₀矢量 | (150,340)→(150,260) | 蓝色竖直向上 |
        // | v_B矢量 | (550,180)→(630,180) | 红色水平向右 |
        // | mg 矢量 | (150,340)→(150,390) | 红色虚线向下 |
        // | qE 矢量 | (150,340)→(220,340) | 绿色虚线向右 |

        const TOTAL_STEPS = 5;
        let currentStep = 0;

        // 元素 ID 映射
        const elements = {
            // 步骤 1 新增
            pointB: 'scene1-point-b',
            labelB: 'scene1-label-b',
            vecMg: 'scene1-vec-mg',
            labelMg: 'scene1-label-mg',
            vecQe: 'scene1-vec-qe',
            labelQe: 'scene1-label-qe',
            // 步骤 2 新增
            trajectory: 'scene1-trajectory',
            trajectoryFill: 'scene1-trajectory-fill',
            dispHorizontal: 'scene1-disp-horizontal',
            dispVertical: 'scene1-disp-vertical',
            labelX: 'scene1-label-x',
            labelH: 'scene1-label-h',
            rightAngle1: 'scene1-right-angle-1',
            rightAngle2: 'scene1-right-angle-2',
            // 步骤 3 新增
            vecVb: 'scene1-vec-vb',
            labelVb: 'scene1-label-vb',
            // 步骤 4 新增
            velDecompA: 'scene1-vel-decomp-a',
            velDecompB: 'scene1-vel-decomp-b'
        };

        const controls = {
            prevBtn: document.getElementById('scene1-prev-btn'),
            nextBtn: document.getElementById('scene1-next-btn'),
            indicator: document.getElementById('scene1-step-indicator')
        };

        // 显示元素
        function show(id) {
            const el = document.getElementById(id);
            if (el) {
                el.style.display = 'block';
                el.style.opacity = '1';
            }
        }

        // 隐藏元素
        function hide(id) {
            const el = document.getElementById(id);
            if (el) {
                el.style.display = 'none';
                el.style.opacity = '0';
            }
        }

        // 更新按钮状态
        function updateButtons() {
            controls.prevBtn.disabled = (currentStep === 0);
            controls.nextBtn.disabled = (currentStep === TOTAL_STEPS - 1);
            controls.indicator.textContent = `步骤 ${currentStep + 1}/${TOTAL_STEPS}`;
        }

        // 更新显示状态
        function updateDisplay() {
            // 步骤 0：初始状态 - 只显示电场线、A 点、v₀
            // 步骤 1：新增 mg、qE、B 点
            if (currentStep >= 1) {
                show(elements.pointB);
                show(elements.labelB);
                show(elements.vecMg);
                show(elements.labelMg);
                show(elements.vecQe);
                show(elements.labelQe);
            } else {
                hide(elements.pointB);
                hide(elements.labelB);
                hide(elements.vecMg);
                hide(elements.labelMg);
                hide(elements.vecQe);
                hide(elements.labelQe);
            }

            // 步骤 2：新增轨迹、位移标注
            if (currentStep >= 2) {
                show(elements.trajectory);
                show(elements.trajectoryFill);
                show(elements.dispHorizontal);
                show(elements.dispVertical);
                show(elements.labelX);
                show(elements.labelH);
                show(elements.rightAngle1);
                show(elements.rightAngle2);
            } else {
                hide(elements.trajectory);
                hide(elements.trajectoryFill);
                hide(elements.dispHorizontal);
                hide(elements.dispVertical);
                hide(elements.labelX);
                hide(elements.labelH);
                hide(elements.rightAngle1);
                hide(elements.rightAngle2);
            }

            // 步骤 3：新增 v_B矢量
            if (currentStep >= 3) {
                show(elements.vecVb);
                show(elements.labelVb);
            } else {
                hide(elements.vecVb);
                hide(elements.labelVb);
            }

            // 步骤 4：新增速度分解标注
            if (currentStep >= 4) {
                show(elements.velDecompA);
                show(elements.velDecompB);
            } else {
                hide(elements.velDecompA);
                hide(elements.velDecompB);
            }

            updateButtons();
        }

        // 下一步
        function nextStep() {
            if (currentStep < TOTAL_STEPS - 1) {
                currentStep++;
                updateDisplay();
            }
        }

        // 上一步
        function prevStep() {
            if (currentStep > 0) {
                currentStep--;
                updateDisplay();
            }
        }

        // 绑定事件
        controls.nextBtn.addEventListener('click', nextStep);
        controls.prevBtn.addEventListener('click', prevStep);

        // 初始化
        updateDisplay();
    })();
    </script>
