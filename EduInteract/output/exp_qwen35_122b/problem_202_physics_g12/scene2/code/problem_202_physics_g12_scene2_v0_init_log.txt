    <script>
    (function() {
        // Scene 2: 电场中类平抛运动的分解分析
        // 坐标映射：物理坐标 (x,y) → 画布坐标 (X,Y)
        // x 范围 [-2L, 0] 映射到 [100, 500]，比例：200px/2L = 100px/L
        // y 范围 [-L, L] 映射到 [400, 100]，比例：300px/2L = 150px/L
        // 原点 O(0,0) → (500, 250)
        
        const scene2 = {
            currentStep: 0,
            totalSteps: 5,
            
            // 步骤配置：每个步骤显示/隐藏哪些元素
            stepConfig: [
                // 步骤 0：初始状态 - 显示电场环境、A 点、初速度
                {
                    show: ['scene2-efield-upper', 'scene2-efield-lower', 'scene2-point-A', 'scene2-point-C', 'scene2-v0-init'],
                    hide: ['scene2-trajectory', 'scene2-point-P', 'scene2-point-Ap', 'scene2-v0-final', 'scene2-velocity-decomp', 'scene2-horiz-decomp', 'scene2-vert-decomp', 'scene2-formula']
                },
                // 步骤 1：显示完整轨迹、P 点、A'点
                {
                    show: ['scene2-efield-upper', 'scene2-efield-lower', 'scene2-point-A', 'scene2-point-C', 'scene2-v0-init', 'scene2-trajectory', 'scene2-point-P', 'scene2-point-Ap', 'scene2-v0-final'],
                    hide: ['scene2-velocity-decomp', 'scene2-horiz-decomp', 'scene2-vert-decomp', 'scene2-formula']
                },
                // 步骤 2：显示速度分解
                {
                    show: ['scene2-efield-upper', 'scene2-efield-lower', 'scene2-point-A', 'scene2-point-C', 'scene2-v0-init', 'scene2-trajectory', 'scene2-point-P', 'scene2-point-Ap', 'scene2-v0-final', 'scene2-velocity-decomp'],
                    hide: ['scene2-horiz-decomp', 'scene2-vert-decomp', 'scene2-formula']
                },
                // 步骤 3：显示水平方向分解
                {
                    show: ['scene2-efield-upper', 'scene2-efield-lower', 'scene2-point-A', 'scene2-point-C', 'scene2-v0-init', 'scene2-trajectory', 'scene2-point-P', 'scene2-point-Ap', 'scene2-v0-final', 'scene2-velocity-decomp', 'scene2-horiz-decomp'],
                    hide: ['scene2-vert-decomp', 'scene2-formula']
                },
                // 步骤 4：显示竖直方向分解
                {
                    show: ['scene2-efield-upper', 'scene2-efield-lower', 'scene2-point-A', 'scene2-point-C', 'scene2-v0-init', 'scene2-trajectory', 'scene2-point-P', 'scene2-point-Ap', 'scene2-v0-final', 'scene2-velocity-decomp', 'scene2-horiz-decomp', 'scene2-vert-decomp'],
                    hide: ['scene2-formula']
                },
                // 步骤 5：显示公式推导
                {
                    show: ['scene2-efield-upper', 'scene2-efield-lower', 'scene2-point-A', 'scene2-point-C', 'scene2-v0-init', 'scene2-trajectory', 'scene2-point-P', 'scene2-point-Ap', 'scene2-v0-final', 'scene2-velocity-decomp', 'scene2-horiz-decomp', 'scene2-vert-decomp', 'scene2-formula'],
                    hide: []
                }
            ],
            
            init: function() {
                this.bindEvents();
                this.updateStep();
            },
            
            bindEvents: function() {
                const btnPrev = document.getElementById('scene2-btn-prev');
                const btnNext = document.getElementById('scene2-btn-next');
                
                btnPrev.addEventListener('click', () => {
                    if (this.currentStep > 0) {
                        this.currentStep--;
                        this.updateStep();
                    }
                });
                
                btnNext.addEventListener('click', () => {
                    if (this.currentStep < this.totalSteps) {
                        this.currentStep++;
                        this.updateStep();
                    }
                });
            },
            
            updateStep: function() {
                const config = this.stepConfig[this.currentStep];
                
                // 显示元素
                config.show.forEach(id => {
                    const el = document.getElementById(id);
                    if (el && el.classList.contains('scene2-hidden')) {
                        el.classList.remove('scene2-hidden');
                        el.classList.add('scene2-fade-in');
                    }
                });
                
                // 隐藏元素
                config.hide.forEach(id => {
                    const el = document.getElementById(id);
                    if (el) {
                        el.classList.add('scene2-hidden');
                        el.classList.remove('scene2-fade-in');
                    }
                });
                
                // 更新按钮状态
                document.getElementById('scene2-btn-prev').disabled = (this.currentStep === 0);
                document.getElementById('scene2-btn-next').disabled = (this.currentStep === this.totalSteps);
                
                // 更新步骤指示器
                document.getElementById('scene2-step-indicator').textContent = '步骤 ' + this.currentStep + '/' + this.totalSteps;
            }
        };
        
        scene2.init();
    })();
    </script>
