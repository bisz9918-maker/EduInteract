    <script>
    (function() {
        // 布局坐标表（实现对照依据）
        // | 元素 | 类型 | 位置/范围 | 大小 | 标签偏移方向 | 备注 |
        // | 画布背景 | rect | (0,0)-(800,500) | 800×500 | - | 白色背景 |
        // | 1 节链条轮廓 | path | (280,150)-(380,220) | 100×70 | - | 椭圆链节，两端半圆 |
        // | 2 节链条轮廓 | path | (200,150)-(460,220) | 260×70 | - | 步骤 1 显示 |
        // | 3 节链条轮廓 | path | (150,150)-(510,220) | 360×70 | - | 步骤 2 显示 |
        // | n 节链条轮廓 | path | (100,150)-(560,220) | 460×70 | - | 步骤 3 显示，含省略号 |
        // | 100 节链条轮廓 | path | (80,150)-(580,220) | 500×70 | - | 步骤 4 显示 |
        // | 连接圆 | circle | 根据节数动态计算 | r=16 | - | 重叠区域，直径 0.8cm 对应 |
        // | 尺寸标注线 2.5cm | line | (280,80)-(380,80) | 100×0 | 上方 | 辅助虚线（向上移动） |
        // | 尺寸标注线 0.8cm | line | (344,80)-(360,80) | 16×0 | 上方 | 重叠区域标注（向上移动） |
        // | 标签"2.5cm" | text | (330,70) | 60×20 | 上方 | 单节长度（向上移动） |
        // | 标签"0.8cm" | text | (352,70) | 50×20 | 上方 | 重叠直径（向上移动） |
        // | 标签"1 节/2 节/..." | text | (330,260) | 40×20 | 下方 | 步骤标签（向下移动） |
        // | 公式文本区 | text | (100,320)-(700,420) | 600×100 | - | 公式展示区（向下移动） |
        // | 按钮"◀ 上一步" | rect+text | (150,440)-(320,480) | 170×40 | - | 左侧按钮 |
        // | 按钮"下一步 ▶" | rect+text | (480,440)-(650,480) | 170×40 | - | 右侧按钮 |

        const scene1 = {
            currentStep: 0,
            totalSteps: 5,
            
            // 公式文本数组
            formulas: [
                { text: "1 节链条长度 = 2.5cm", className: "scene1-formula" },
                { text: "2 节链条长度 = 2.5 + 2.5 - 0.8 = 4.2cm", className: "scene1-formula" },
                { text: "3 节链条长度 = 2.5×3 - 0.8×2 = 5.9cm", className: "scene1-formula" },
                { text: "n 节链条长度 = 2.5n - 0.8(n-1) = 1.7n + 0.8", className: "scene1-formula-green" },
                { text: "当 n=100 时，总长度 = 1.7×100 + 0.8 = 170.8cm\n故选 C", className: "scene1-formula-red" }
            ],
            
            // 节数标签数组
            countLabels: ["1 节", "2 节", "3 节", "n 节", "100 节"],
            
            // 初始化
            init: function() {
                this.updateScene();
                this.bindEvents();
            },
            
            // 更新场景显示
            updateScene: function() {
                // 隐藏所有链条
                document.getElementById('scene1-chain-1').style.display = 'none';
                document.getElementById('scene1-chain-2').style.display = 'none';
                document.getElementById('scene1-chain-3').style.display = 'none';
                document.getElementById('scene1-chain-n').style.display = 'none';
                document.getElementById('scene1-chain-100').style.display = 'none';
                
                // 隐藏 0.8cm 标注
                document.getElementById('scene1-dim-0.8-line').style.display = 'none';
                document.getElementById('scene1-dim-0.8-arrow-left').style.display = 'none';
                document.getElementById('scene1-dim-0.8-arrow-right').style.display = 'none';
                document.getElementById('scene1-dim-0.8-label').style.display = 'none';
                
                // 显示当前步骤的链条
                const chainIds = ['scene1-chain-1', 'scene1-chain-2', 'scene1-chain-3', 'scene1-chain-n', 'scene1-chain-100'];
                document.getElementById(chainIds[this.currentStep]).style.display = 'block';
                
                // 更新节数标签
                document.getElementById('scene1-count-label').textContent = this.countLabels[this.currentStep];
                
                // 更新公式文本
                const formulaEl = document.getElementById('scene1-formula');
                const formulaData = this.formulas[this.currentStep];
                formulaEl.setAttribute('class', formulaData.className);
                
                // 处理多行文本
                if (formulaData.text.includes('\n')) {
                    const lines = formulaData.text.split('\n');
                    let tspanHTML = '';
                    lines.forEach((line, i) => {
                        tspanHTML += `<tspan x="100" dy="${i === 0 ? '0' : '24'}">${line}</tspan>`;
                    });
                    formulaEl.innerHTML = tspanHTML;
                } else {
                    formulaEl.innerHTML = `<tspan x="100" dy="0">${formulaData.text}</tspan>`;
                }
                
                // 显示 0.8cm 标注（步骤 1 及以后）
                if (this.currentStep >= 1) {
                    document.getElementById('scene1-dim-0.8-line').style.display = 'block';
                    document.getElementById('scene1-dim-0.8-arrow-left').style.display = 'block';
                    document.getElementById('scene1-dim-0.8-arrow-right').style.display = 'block';
                    document.getElementById('scene1-dim-0.8-label').style.display = 'block';
                }
                
                // 更新按钮状态
                document.getElementById('scene1-btn-prev').disabled = (this.currentStep === 0);
                document.getElementById('scene1-btn-next').disabled = (this.currentStep === this.totalSteps - 1);
            },
            
            // 绑定事件
            bindEvents: function() {
                document.getElementById('scene1-btn-prev').addEventListener('click', () => {
                    if (this.currentStep > 0) {
                        this.currentStep--;
                        this.updateScene();
                    }
                });
                
                document.getElementById('scene1-btn-next').addEventListener('click', () => {
                    if (this.currentStep < this.totalSteps - 1) {
                        this.currentStep++;
                        this.updateScene();
                    }
                });
            }
        };
        
        // 初始化场景
        scene1.init();
    })();
    </script>
