    <script>
    (function() {
        // Scene 1 实现 - 直角三角板摆放角度计算

        const sceneId = 'scene1';
        const totalSteps = 5;
        let currentStep = 0;

        // 步骤配置：每个步骤显示的元素 ID 列表
        const stepConfig = {
            0: {
                // 初始状态：只显示三角板轮廓和顶点标签
                triangles: ['scene1-tri-abc', 'scene1-tri-def'],
                vertices: ['scene1-vertex-a', 'scene1-vertex-b', 'scene1-vertex-c', 'scene1-vertex-d', 'scene1-vertex-e', 'scene1-vertex-f'],
                labels: ['scene1-label-a', 'scene1-label-b', 'scene1-label-c', 'scene1-label-d', 'scene1-label-e', 'scene1-label-f'],
                angleMarks: [],
                valueLabels: [],
                highlightTriangles: { 'scene1-tri-abc': '#000', 'scene1-tri-def': '#000' },
                highlightVertices: {},
                highlightLines: {}
            },
            1: {
                // 步骤 1：标注△ABC 条件
                triangles: ['scene1-tri-abc', 'scene1-tri-def'],
                vertices: ['scene1-vertex-a', 'scene1-vertex-b', 'scene1-vertex-c', 'scene1-vertex-d', 'scene1-vertex-e', 'scene1-vertex-f'],
                labels: ['scene1-label-a', 'scene1-label-b', 'scene1-label-c', 'scene1-label-d', 'scene1-label-e', 'scene1-label-f'],
                angleMarks: ['scene1-rect-a', 'scene1-equal-ab', 'scene1-equal-ac'],
                valueLabels: ['scene1-val-90a'],
                highlightTriangles: { 'scene1-tri-abc': '#1565c0', 'scene1-tri-def': '#000' },
                highlightVertices: {},
                highlightLines: {}
            },
            2: {
                // 步骤 2：标注△DEF 条件
                triangles: ['scene1-tri-abc', 'scene1-tri-def'],
                vertices: ['scene1-vertex-a', 'scene1-vertex-b', 'scene1-vertex-c', 'scene1-vertex-d', 'scene1-vertex-e', 'scene1-vertex-f'],
                labels: ['scene1-label-a', 'scene1-label-b', 'scene1-label-c', 'scene1-label-d', 'scene1-label-e', 'scene1-label-f'],
                angleMarks: ['scene1-rect-a', 'scene1-equal-ab', 'scene1-equal-ac', 'scene1-rect-d', 'scene1-arc-e'],
                valueLabels: ['scene1-val-90a', 'scene1-val-90d', 'scene1-val-30'],
                highlightTriangles: { 'scene1-tri-abc': '#1565c0', 'scene1-tri-def': '#c62828' },
                highlightVertices: {},
                highlightLines: {}
            },
            3: {
                // 步骤 3：标注位置关系
                triangles: ['scene1-tri-abc', 'scene1-tri-def'],
                vertices: ['scene1-vertex-a', 'scene1-vertex-b', 'scene1-vertex-c', 'scene1-vertex-d', 'scene1-vertex-e', 'scene1-vertex-f'],
                labels: ['scene1-label-a', 'scene1-label-b', 'scene1-label-c', 'scene1-label-d', 'scene1-label-e', 'scene1-label-f'],
                angleMarks: ['scene1-rect-a', 'scene1-equal-ab', 'scene1-equal-ac', 'scene1-rect-d', 'scene1-arc-e', 'scene1-arc-bce'],
                valueLabels: ['scene1-val-90a', 'scene1-val-90d', 'scene1-val-30', 'scene1-val-40'],
                highlightTriangles: { 'scene1-tri-abc': '#1565c0', 'scene1-tri-def': '#c62828' },
                highlightVertices: { 'scene1-vertex-c': '#2e7d32' },
                highlightLines: {}
            },
            4: {
                // 步骤 4：高亮待求角
                triangles: ['scene1-tri-abc', 'scene1-tri-def'],
                vertices: ['scene1-vertex-a', 'scene1-vertex-b', 'scene1-vertex-c', 'scene1-vertex-d', 'scene1-vertex-e', 'scene1-vertex-f'],
                labels: ['scene1-label-a', 'scene1-label-b', 'scene1-label-c', 'scene1-label-d', 'scene1-label-e', 'scene1-label-f'],
                angleMarks: ['scene1-rect-a', 'scene1-equal-ab', 'scene1-equal-ac', 'scene1-rect-d', 'scene1-arc-e', 'scene1-arc-bce', 'scene1-arc-cdf'],
                valueLabels: ['scene1-val-90a', 'scene1-val-90d', 'scene1-val-30', 'scene1-val-40', 'scene1-val-q'],
                highlightTriangles: { 'scene1-tri-abc': '#1565c0', 'scene1-tri-def': '#c62828' },
                highlightVertices: { 'scene1-vertex-c': '#2e7d32' },
                highlightLines: {}
            }
        };

        // 工具函数：显示元素
        function show(elId) {
            const el = document.getElementById(elId);
            if (el) {
                el.style.display = 'block';
                el.style.opacity = '1';
            }
        }

        // 工具函数：隐藏元素
        function hide(elId) {
            const el = document.getElementById(elId);
            if (el) {
                el.style.opacity = '0';
            }
        }

        // 渲染当前步骤
        function renderStep() {
            const config = stepConfig[currentStep];

            // 显示所有三角形
            config.triangles.forEach(id => show(id));

            // 显示所有顶点
            config.vertices.forEach(id => show(id));

            // 显示所有标签
            config.labels.forEach(id => show(id));

            // 显示角度标记
            config.angleMarks.forEach(id => show(id));

            // 显示数值标签
            config.valueLabels.forEach(id => show(id));

            // 设置三角形高亮颜色
            Object.keys(config.highlightTriangles).forEach(id => {
                const el = document.getElementById(id);
                if (el) el.setAttribute('stroke', config.highlightTriangles[id]);
            });

            // 设置顶点高亮颜色
            Object.keys(config.highlightVertices).forEach(id => {
                const el = document.getElementById(id);
                if (el) el.setAttribute('fill', config.highlightVertices[id]);
            });

            // 更新按钮状态
            document.getElementById('scene1-btn-prev').disabled = (currentStep === 0);
            document.getElementById('scene1-btn-next').disabled = (currentStep === totalSteps - 1);

            // 更新步骤指示器
            document.getElementById('scene1-step-indicator').textContent = `步骤 ${currentStep + 1}/${totalSteps}`;
        }

        // 初始化
        function init() {
            // 隐藏所有角度标记和数值标签（初始状态）
            const allAngleMarks = ['scene1-rect-a', 'scene1-equal-ab', 'scene1-equal-ac', 'scene1-rect-d', 'scene1-arc-e', 'scene1-arc-bce', 'scene1-arc-cdf'];
            const allValueLabels = ['scene1-val-90a', 'scene1-val-90d', 'scene1-val-30', 'scene1-val-40', 'scene1-val-q'];
            allAngleMarks.forEach(id => hide(id));
            allValueLabels.forEach(id => hide(id));

            // 绑定按钮事件
            document.getElementById('scene1-btn-prev').addEventListener('click', function() {
                if (currentStep > 0) {
                    currentStep--;
                    renderStep();
                }
            });

            document.getElementById('scene1-btn-next').addEventListener('click', function() {
                if (currentStep < totalSteps - 1) {
                    currentStep++;
                    renderStep();
                }
            });

            // 渲染初始步骤
            renderStep();
        }

        // 页面加载完成后初始化
        if (document.readyState === 'loading') {
            document.addEventListener('DOMContentLoaded', init);
        } else {
            init();
        }
    })();
    </script>
