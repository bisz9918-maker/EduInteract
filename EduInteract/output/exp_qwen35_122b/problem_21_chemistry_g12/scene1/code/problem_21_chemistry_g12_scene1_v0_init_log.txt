    (function() {
        // Scene 1 实现 - 原电池 - 电解池串联装置
        // 布局坐标表（参考）：
        // | 元素 | 类型 | 位置/范围 | 大小 | 标签偏移方向 | 备注 |
        // |------|------|-----------|------|-------------|------|
        // | 左侧烧杯轮廓 | path | (80,80)-(280,380) | 200×300 | - | 梯形烧杯 |
        // | 稀硫酸液面 | rect | (90,180)-(270,370) | 180×190 | - | 半透明蓝色填充 |
        // | 锌棒 | rect | (110,120)-(150,340) | 40×220 | - | 蓝色填充 #1565c0 |
        // | 铜棒 | rect | (190,120)-(230,340) | 40×220 | - | 橙色填充 #ff6f00 |
        // | 锌棒标签 | text | (125,105) | 30×16 | 上方 | 文字"锌" |
        // | 铜棒标签 | text | (205,105) | 30×16 | 上方 | 文字"铜" |
        // | 稀硫酸标签 | text | (180,395) | 60×16 | 下方 | 文字"稀硫酸" |
        // | 右侧滤纸轮廓 | ellipse | (550,230) | 200×120 | - | 灰色虚线椭圆 |
        // | 饱和食盐水填充 | ellipse | (550,230) | 200×120 | - | 半透明浅蓝色 |
        // | 石墨棒 a | rect | (510,150)-(540,280) | 30×130 | - | 灰色填充 #616161 |
        // | 石墨棒 b | rect | (610,150)-(640,280) | 30×130 | - | 灰色填充 #616161 |
        // | 石墨棒 a 标签 | text | (520,135) | 20×16 | 上方 | 文字"a" |
        // | 石墨棒 b 标签 | text | (625,135) | 20×16 | 上方 | 文字"b" |
        // | 滤纸标签 | text | (550,370) | 120×16 | 下方 | 文字"饱和食盐水滤纸" |
        // | 导线 1（锌→b） | path | (130,120)-(130,60)-(625,60)-(625,150) | - | - | 黑色实线 2px |
        // | 导线 2（铜→a） | path | (210,120)-(210,40)-(525,40)-(525,150) | - | - | 黑色实线 2px |
        // | 电子流向箭头 1 | path | (130,80)-(625,80) | - | - | 红色箭头，锌→b |
        // | 电子流向箭头 2 | path | (525,20)-(210,20) | - | - | 红色箭头，a→铜 |
        // | 负极标签 | text | (130,400) | 40×16 | 下方 | 文字"负极" |
        // | 正极标签 | text | (210,400) | 40×16 | 下方 | 文字"正极" |
        // | 阳极标签 | text | (525,400) | 40×16 | 下方 | 文字"阳极" |
        // | 阴极标签 | text | (625,400) | 40×16 | 下方 | 文字"阴极" |
        // | 酚酞变色区域 | ellipse | (625,230) | 40×30 | - | 红色半透明，b 极附近 |
        // | 反应式框 1 | rect | (50,420)-(250,470) | 200×50 | - | 步骤 1 显示 |
        // | 反应式框 2 | rect | (300,420)-(500,470) | 200×50 | - | 步骤 2 显示 |
        // | 反应式框 3 | rect | (550,420)-(750,470) | 200×50 | - | 步骤 3 显示 |

        // 状态变量
        var currentStep = 0;
        var totalSteps = 6;

        // 工具函数：显示元素
        function show(el) {
            if (el) {
                el.setAttribute('opacity', '1');
                el.style.display = 'block';
            }
        }

        // 工具函数：隐藏元素
        function hide(el) {
            if (el) {
                el.setAttribute('opacity', '0');
            }
        }

        // 工具函数：淡入动画
        function fadeIn(el, duration) {
            if (!el) return;
            var start = null;
            var opacity = 0;
            function animate(timestamp) {
                if (!start) start = timestamp;
                var progress = timestamp - start;
                opacity = Math.min(progress / duration, 1);
                el.setAttribute('opacity', opacity);
                if (progress < duration) {
                    requestAnimationFrame(animate);
                }
            }
            requestAnimationFrame(animate);
        }

        // 更新显示状态
        function updateDisplay() {
            // 更新步骤指示器
            document.getElementById('scene1-step-indicator').textContent = '步骤 ' + (currentStep + 1) + '/' + totalSteps;
            
            // 更新按钮状态
            document.getElementById('scene1-btn-prev').disabled = (currentStep === 0);
            document.getElementById('scene1-btn-next').disabled = (currentStep === totalSteps - 1);

            // 步骤 0：初始状态 - 只显示基础装置
            // 步骤 1：原电池电极标注
            // 步骤 2：电解池电极标注
            // 步骤 3：电子流向高亮
            // 步骤 4：电极反应式展示
            // 步骤 5：酚酞变色验证

            // 步骤 1 及以上显示的内容
            var showStep1 = currentStep >= 1;
            var showStep2 = currentStep >= 2;
            var showStep3 = currentStep >= 3;
            var showStep4 = currentStep >= 4;
            var showStep5 = currentStep >= 5;

            // 负极/正极标签（步骤 1+）
            showStep1 ? show(document.getElementById('scene1-label-negative')) : hide(document.getElementById('scene1-label-negative'));
            showStep1 ? show(document.getElementById('scene1-label-positive')) : hide(document.getElementById('scene1-label-positive'));

            // 阳极/阴极标签（步骤 2+）
            showStep2 ? show(document.getElementById('scene1-label-anode')) : hide(document.getElementById('scene1-label-anode'));
            showStep2 ? show(document.getElementById('scene1-label-cathode')) : hide(document.getElementById('scene1-label-cathode'));

            // 锌/铜电极高亮（步骤 1+）
            showStep1 ? fadeIn(document.getElementById('scene1-zinc-highlight'), 300) : hide(document.getElementById('scene1-zinc-highlight'));
            showStep1 ? fadeIn(document.getElementById('scene1-copper-highlight'), 300) : hide(document.getElementById('scene1-copper-highlight'));

            // 反应式框 1（步骤 1+）
            showStep1 ? fadeIn(document.getElementById('scene1-reaction-box1'), 300) : hide(document.getElementById('scene1-reaction-box1'));
            showStep1 ? fadeIn(document.getElementById('scene1-reaction-text1'), 300) : hide(document.getElementById('scene1-reaction-text1'));
            showStep1 ? fadeIn(document.getElementById('scene1-reaction-text1b'), 300) : hide(document.getElementById('scene1-reaction-text1b'));

            // 反应式框 2（步骤 2+）
            showStep2 ? fadeIn(document.getElementById('scene1-reaction-box2'), 300) : hide(document.getElementById('scene1-reaction-box2'));
            showStep2 ? fadeIn(document.getElementById('scene1-reaction-text2'), 300) : hide(document.getElementById('scene1-reaction-text2'));
            showStep2 ? fadeIn(document.getElementById('scene1-reaction-text2b'), 300) : hide(document.getElementById('scene1-reaction-text2b'));

            // 反应式框 3（步骤 2+）
            showStep2 ? fadeIn(document.getElementById('scene1-reaction-box3'), 300) : hide(document.getElementById('scene1-reaction-box3'));
            showStep2 ? fadeIn(document.getElementById('scene1-reaction-text3'), 300) : hide(document.getElementById('scene1-reaction-text3'));
            showStep2 ? fadeIn(document.getElementById('scene1-reaction-text3b'), 300) : hide(document.getElementById('scene1-reaction-text3b'));

            // 电子流向箭头（步骤 3+）
            showStep3 ? fadeIn(document.getElementById('scene1-electron-flow1'), 300) : hide(document.getElementById('scene1-electron-flow1'));
            showStep3 ? fadeIn(document.getElementById('scene1-e1-label'), 300) : hide(document.getElementById('scene1-e1-label'));
            showStep3 ? fadeIn(document.getElementById('scene1-electron-flow2'), 300) : hide(document.getElementById('scene1-electron-flow2'));
            showStep3 ? fadeIn(document.getElementById('scene1-e2-label'), 300) : hide(document.getElementById('scene1-e2-label'));

            // 酚酞变色区域（步骤 5+）
            showStep5 ? fadeIn(document.getElementById('scene1-phenolphthalein'), 300) : hide(document.getElementById('scene1-phenolphthalein'));
            showStep5 ? fadeIn(document.getElementById('scene1-oh-label'), 300) : hide(document.getElementById('scene1-oh-label'));
        }

        // 初始化
        function init() {
            updateDisplay();

            // 绑定按钮事件
            document.getElementById('scene1-btn-prev').addEventListener('click', function() {
                if (currentStep > 0) {
                    currentStep--;
                    updateDisplay();
                }
            });

            document.getElementById('scene1-btn-next').addEventListener('click', function() {
                if (currentStep < totalSteps - 1) {
                    currentStep++;
                    updateDisplay();
                }
            });
        }

        // 页面加载完成后初始化
        if (document.readyState === 'loading') {
            document.addEventListener('DOMContentLoaded', init);
        } else {
            init();
        }
    })();
