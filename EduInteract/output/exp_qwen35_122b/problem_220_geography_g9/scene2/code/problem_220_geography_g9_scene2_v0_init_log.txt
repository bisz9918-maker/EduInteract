    <script>
    (function() {
        // Scene 2 实现 - 中纬环流模式分析
        // 布局坐标表对照（已调整避免重叠）：
        // | 元素 | 位置 | 备注 |
        // | a 点 | (200,410) | 30°N 位置，向下移动避开地面 |
        // | b 点 | (600,410) | 60°N 位置，向下移动避开地面 |
        // | c 点 | (600,250) | 高空，向下移动 |
        // | d 点 | (200,250) | 高空，向下移动 |
        // | 箭头① | (600,250)→(200,250) | 高空水平气流 c→d |
        // | 箭头② | (600,410)→(600,250) | 垂直上升气流 b→c |
        // | 箭头③ | (200,410)→(600,410) | 地面水平气流 a→b |
        // | 箭头④ | (200,250)→(200,410) | 垂直下沉气流 d→a |
        // | 地面平面 | (150,390)-(650,450) | 向下移动避免与标签重叠 |
        
        const scene2 = {
            currentStep: 0,
            totalSteps: 6,
            
            // 工具函数
            show: function(el) {
                if (el) {
                    el.style.display = 'block';
                    el.style.opacity = '1';
                }
            },
            hide: function(el) {
                if (el) {
                    el.style.display = 'none';
                    el.style.opacity = '0';
                }
            },
            
            // 更新显示状态
            updateDisplay: function() {
                const stepIndicator = document.getElementById('scene2-step-indicator');
                const prevBtn = document.getElementById('scene2-prev-btn');
                const nextBtn = document.getElementById('scene2-next-btn');
                const conclusion = document.getElementById('scene2-conclusion');
                
                // 更新步骤指示器
                stepIndicator.textContent = '步骤 ' + (scene2.currentStep + 1) + '/6';
                
                // 更新按钮状态
                prevBtn.disabled = scene2.currentStep === 0;
                nextBtn.disabled = scene2.currentStep === scene2.totalSteps - 1;
                
                // 隐藏所有步骤相关元素
                document.getElementById('scene2-step1').style.opacity = '0';
                document.getElementById('scene2-step2').style.opacity = '0';
                document.getElementById('scene2-step3').style.opacity = '0';
                document.getElementById('scene2-step4').style.opacity = '0';
                document.getElementById('scene2-step5').style.opacity = '0';
                conclusion.style.display = 'none';
                
                // 重置点颜色
                document.getElementById('scene2-point-a').setAttribute('fill', '#000000');
                document.getElementById('scene2-point-b').setAttribute('fill', '#000000');
                document.getElementById('scene2-arrow-4').querySelector('line').setAttribute('stroke', '#000000');
                document.getElementById('scene2-arrow-4').querySelector('polygon').setAttribute('fill', '#000000');
                document.getElementById('scene2-arrow-2').querySelector('line').setAttribute('stroke', '#000000');
                document.getElementById('scene2-arrow-2').querySelector('polygon').setAttribute('fill', '#000000');
                
                // 根据当前步骤显示对应元素
                if (scene2.currentStep >= 1) {
                    document.getElementById('scene2-step1').style.opacity = '1';
                    document.getElementById('scene2-point-a').setAttribute('fill', '#c62828');
                    document.getElementById('scene2-arrow-4').querySelector('line').setAttribute('stroke', '#2e7d32');
                    document.getElementById('scene2-arrow-4').querySelector('polygon').setAttribute('fill', '#2e7d32');
                }
                if (scene2.currentStep >= 2) {
                    document.getElementById('scene2-step2').style.opacity = '1';
                    document.getElementById('scene2-point-b').setAttribute('fill', '#1565c0');
                }
                if (scene2.currentStep >= 3) {
                    document.getElementById('scene2-step3').style.opacity = '1';
                    document.getElementById('scene2-arrow-2').querySelector('line').setAttribute('stroke', '#2e7d32');
                    document.getElementById('scene2-arrow-2').querySelector('polygon').setAttribute('fill', '#2e7d32');
                }
                if (scene2.currentStep >= 4) {
                    document.getElementById('scene2-step4').style.opacity = '1';
                }
                if (scene2.currentStep >= 5) {
                    document.getElementById('scene2-step5').style.opacity = '1';
                    conclusion.style.display = 'block';
                }
            },
            
            // 下一步
            nextStep: function() {
                if (scene2.currentStep < scene2.totalSteps - 1) {
                    scene2.currentStep++;
                    scene2.updateDisplay();
                }
            },
            
            // 上一步
            prevStep: function() {
                if (scene2.currentStep > 0) {
                    scene2.currentStep--;
                    scene2.updateDisplay();
                }
            },
            
            // 初始化
            init: function() {
                document.getElementById('scene2-next-btn').addEventListener('click', scene2.nextStep);
                document.getElementById('scene2-prev-btn').addEventListener('click', scene2.prevStep);
                scene2.updateDisplay();
            }
        };
        
        // 启动
        scene2.init();
    })();
    </script>
