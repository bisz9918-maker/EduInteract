    <script>
    (function() {
        // Scene 2 实现
        const TOTAL_STEPS = 5;
        let currentStep = 0;
        
        // 获取所有 DOM 元素
        const titleEl = document.getElementById('scene2-title');
        const prevBtn = document.getElementById('scene2-prev-btn');
        const nextBtn = document.getElementById('scene2-next-btn');
        const stepIndicator = document.getElementById('scene2-step-indicator');
        
        // 按步骤分组的元素
        const stepElements = {
            0: [titleEl],
            1: [
                document.getElementById('scene2-moon-title'),
                document.getElementById('scene2-moon-info1'),
                document.getElementById('scene2-moon-info2'),
                document.getElementById('scene2-moon-time')
            ],
            2: [
                document.getElementById('scene2-panel-title'),
                document.getElementById('scene2-ground'),
                document.getElementById('scene2-wall'),
                document.getElementById('scene2-wall-label'),
                document.getElementById('scene2-sun-ray-right'),
                document.getElementById('scene2-sun-arrow-right'),
                document.getElementById('scene2-solar-panel'),
                document.getElementById('scene2-panel-label'),
                document.getElementById('scene2-normal'),
                document.getElementById('scene2-angle-h'),
                document.getElementById('scene2-h-label'),
                document.getElementById('scene2-h-formula'),
                document.getElementById('scene2-h-result')
            ],
            3: [
                document.getElementById('scene2-angle-alpha'),
                document.getElementById('scene2-alpha-label'),
                document.getElementById('scene2-alpha-formula'),
                document.getElementById('scene2-alpha-result')
            ],
            4: [
                document.getElementById('scene2-conclusion-title'),
                document.getElementById('scene2-ans1-title'),
                document.getElementById('scene2-ans1-a'),
                document.getElementById('scene2-ans1-b'),
                document.getElementById('scene2-ans2-title'),
                document.getElementById('scene2-ans2-a'),
                document.getElementById('scene2-ans2-b'),
                document.getElementById('scene2-ans3-title'),
                document.getElementById('scene2-ans3-a'),
                document.getElementById('scene2-ans4-title'),
                document.getElementById('scene2-ans4-a'),
                document.getElementById('scene2-ans4-b'),
                document.getElementById('scene2-ans5-title'),
                document.getElementById('scene2-ans5-a')
            ]
        };
        
        // 步骤标题
        const stepTitles = [
            '基础图示',
            '农历十五月出时间',
            '太阳高度角计算',
            '太阳能板角度确定',
            '综合答案'
        ];
        
        // 更新视图
        function updateView() {
            // 更新标题
            titleEl.textContent = stepTitles[currentStep];
            
            // 更新步骤指示器
            stepIndicator.textContent = `步骤 ${currentStep + 1}/${TOTAL_STEPS}`;
            
            // 更新按钮状态
            prevBtn.disabled = (currentStep === 0);
            if (currentStep === TOTAL_STEPS - 1) {
                nextBtn.textContent = '完成';
                nextBtn.disabled = true;
            } else {
                nextBtn.textContent = '下一步 ▶';
                nextBtn.disabled = false;
            }
            
            // 控制各步骤元素的显示
            for (let step = 0; step < TOTAL_STEPS; step++) {
                const elements = stepElements[step];
                if (!elements) continue;
                
                elements.forEach(el => {
                    if (!el) return;
                    if (step <= currentStep) {
                        el.style.opacity = '1';
                        el.style.display = 'block';
                    } else {
                        el.style.opacity = '0';
                        el.style.display = 'none';
                    }
                });
            }
            
            // 步骤 4 时隐藏右侧示意图，显示综合答案
            if (currentStep === 4) {
                const hideElements = [
                    'scene2-moon-title', 'scene2-moon-info1', 'scene2-moon-info2', 'scene2-moon-time',
                    'scene2-panel-title', 'scene2-ground', 'scene2-wall', 'scene2-wall-label',
                    'scene2-sun-ray-right', 'scene2-sun-arrow-right', 'scene2-solar-panel',
                    'scene2-panel-label', 'scene2-normal', 'scene2-angle-h', 'scene2-h-label',
                    'scene2-h-formula', 'scene2-h-result', 'scene2-angle-alpha', 'scene2-alpha-label',
                    'scene2-alpha-formula', 'scene2-alpha-result'
                ];
                hideElements.forEach(id => {
                    const el = document.getElementById(id);
                    if (el) {
                        el.style.opacity = '0';
                        el.style.display = 'none';
                    }
                });
            }
        }
        
        // 事件绑定
        prevBtn.addEventListener('click', () => {
            if (currentStep > 0) {
                currentStep--;
                updateView();
            }
        });
        
        nextBtn.addEventListener('click', () => {
            if (currentStep < TOTAL_STEPS - 1) {
                currentStep++;
                updateView();
            }
        });
        
        // 初始化
        updateView();
    })();
    </script>
