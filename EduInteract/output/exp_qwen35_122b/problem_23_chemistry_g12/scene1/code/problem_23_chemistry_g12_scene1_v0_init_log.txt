    <script>
    (function() {
        // Scene 1: H₂O₂分子结构与极性分析
        // 布局坐标表（调整后 v10）：
        // | 元素 | 位置 | 备注 |
        // |------|------|------|
        // | 标题 | (400,45) | 顶部居中，远离边缘 |
        // | 左侧平面 | (80,140)-(150,140)-(140,230)-(70,230) | 缩小并左移 |
        // | 右侧平面 | (350,140)-(420,140)-(410,230)-(340,230) | 缩小并右移 |
        // | O1 原子 | (200,250) | 左侧氧原子 |
        // | O2 原子 | (250,250) | 右侧氧原子 |
        // | H1 原子 | (170,200) | 左侧氢原子 |
        // | H2 原子 | (270,310) | 右侧氢原子（调整位置） |
        // | O1-O2 键 | (200,250)-(250,250) | 水平键 |
        // | O1-H1 键 | (200,250)-(170,200) | 斜键 |
        // | O2-H2 键 | (250,250)-(270,310) | 斜键（调整角度向左） |
        // | O1 标签 | (200,275) | O1 正下方 |
        // | O2 标签 | (275,285) | O2 右下方（右移远离线段） |
        // | H1 标签 | (170,180) | H1 正上方 |
        // | H2 标签 | (270,340) | H2 正下方（下移远离 O-O 键标签） |
        // | 控件栏 | 底部居中 | 按钮区 |

        // 状态变量
        var currentStep = 0;
        var totalSteps = 5;

        // 获取 DOM 元素
        var prevBtn = document.getElementById('scene1-prev-btn');
        var nextBtn = document.getElementById('scene1-next-btn');
        var stepIndicator = document.getElementById('scene1-step-indicator');

        // 键元素
        var oOBond = document.getElementById('scene1-o-o-bond');
        var o1H1Bond = document.getElementById('scene1-o1-h1-bond');
        var o2H2Bond = document.getElementById('scene1-o2-h2-bond');

        // 标签元素
        var oOLabel = document.getElementById('scene1-o-o-label');
        var oHLabel = document.getElementById('scene1-o-h-label');
        var dihedralArc = document.getElementById('scene1-dihedral-arc');
        var dihedralLabel = document.getElementById('scene1-dihedral-label');

        // 偶极矩元素
        var dipoleGroup = document.getElementById('scene1-dipole-group');
        var dipoleLabel = document.getElementById('scene1-dipole-label');

        // 结论元素
        var conclusionGroup = document.getElementById('scene1-conclusion-group');

        // 工具函数：显示元素
        function show(el) {
            el.style.display = 'block';
            el.style.opacity = '0';
            setTimeout(function() {
                el.style.opacity = '1';
            }, 10);
        }

        // 工具函数：隐藏元素
        function hide(el) {
            el.style.opacity = '0';
            setTimeout(function() {
                el.style.display = 'none';
            }, 200);
        }

        // 更新步骤显示
        function updateStep() {
            // 更新按钮状态
            prevBtn.disabled = (currentStep === 0);

            if (currentStep === totalSteps - 1) {
                nextBtn.textContent = '完成 ✓';
                nextBtn.disabled = true;
            } else {
                nextBtn.textContent = '下一步 ▶';
                nextBtn.disabled = false;
            }

            // 更新步骤指示器
            stepIndicator.textContent = '步骤 ' + (currentStep + 1) + '/' + totalSteps;

            // 根据步骤显示/隐藏元素
            switch(currentStep) {
                case 0:
                    // 初始状态：只显示基本骨架
                    oOBond.style.stroke = '#000000';
                    o1H1Bond.style.stroke = '#000000';
                    o2H2Bond.style.stroke = '#000000';
                    hide(oOLabel);
                    hide(oHLabel);
                    hide(dihedralArc);
                    hide(dihedralLabel);
                    hide(dipoleGroup);
                    hide(dipoleLabel);
                    hide(conclusionGroup);
                    break;

                case 1:
                    // 标注 O-O 非极性键
                    oOBond.style.stroke = '#2e7d32';
                    o1H1Bond.style.stroke = '#000000';
                    o2H2Bond.style.stroke = '#000000';
                    show(oOLabel);
                    show(dihedralArc);
                    hide(oHLabel);
                    hide(dihedralLabel);
                    hide(dipoleGroup);
                    hide(dipoleLabel);
                    hide(conclusionGroup);
                    break;

                case 2:
                    // 标注 O-H 极性键和二面角
                    oOBond.style.stroke = '#2e7d32';
                    o1H1Bond.style.stroke = '#1565c0';
                    o2H2Bond.style.stroke = '#1565c0';
                    show(oOLabel);
                    show(oHLabel);
                    show(dihedralArc);
                    show(dihedralLabel);
                    hide(dipoleGroup);
                    hide(dipoleLabel);
                    hide(conclusionGroup);
                    break;

                case 3:
                    // 显示偶极矩
                    oOBond.style.stroke = '#2e7d32';
                    o1H1Bond.style.stroke = '#1565c0';
                    o2H2Bond.style.stroke = '#1565c0';
                    show(oOLabel);
                    show(oHLabel);
                    show(dihedralArc);
                    show(dihedralLabel);
                    show(dipoleGroup);
                    show(dipoleLabel);
                    hide(conclusionGroup);
                    break;

                case 4:
                    // 结论
                    oOBond.style.stroke = '#2e7d32';
                    o1H1Bond.style.stroke = '#1565c0';
                    o2H2Bond.style.stroke = '#1565c0';
                    show(oOLabel);
                    show(oHLabel);
                    show(dihedralArc);
                    show(dihedralLabel);
                    show(dipoleGroup);
                    show(dipoleLabel);
                    show(conclusionGroup);
                    break;
            }
        }

        // 事件监听
        prevBtn.addEventListener('click', function() {
            if (currentStep > 0) {
                currentStep--;
                updateStep();
            }
        });

        nextBtn.addEventListener('click', function() {
            if (currentStep < totalSteps - 1) {
                currentStep++;
                updateStep();
            }
        });

        // 初始化
        updateStep();
    })();
    </script>
