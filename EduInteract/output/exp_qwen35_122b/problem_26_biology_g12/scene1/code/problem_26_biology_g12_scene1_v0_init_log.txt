    <script>
    (function() {
        const state = { currentStep: 0, totalSteps: 4 };
        
        // 元素显示配置：每个步骤需要显示的元素
        const stepElements = {
            0: ['scene1-p-container', 'scene1-arrow-p-f1'],
            1: ['scene1-p-container', 'scene1-arrow-p-f1', 'scene1-f1-container'],
            2: ['scene1-p-container', 'scene1-arrow-p-f1', 'scene1-f1-container', 'scene1-arrow-f1-f2', 'scene1-f2-container', 'scene1-f2-long-container'],
            3: ['scene1-p-container', 'scene1-arrow-p-f1', 'scene1-f1-container', 'scene1-arrow-f1-f2', 'scene1-f2-container', 'scene1-f2-long-container', 'scene1-f2-mini-container', 'scene1-f2-vestigial-container'],
            4: ['scene1-p-container', 'scene1-arrow-p-f1', 'scene1-f1-container', 'scene1-arrow-f1-f2', 'scene1-f2-container', 'scene1-f2-long-container', 'scene1-f2-mini-container', 'scene1-f2-vestigial-container', 'scene1-stat-container', 'scene1-observation-container']
        };
        
        // 箭头样式配置
        const arrowStyles = {
            0: { p_f1: 'dash', f1_f2: 'dash' },
            1: { p_f1: 'solid', f1_f2: 'dash' },
            2: { p_f1: 'solid', f1_f2: 'solid' },
            3: { p_f1: 'solid', f1_f2: 'solid' },
            4: { p_f1: 'solid', f1_f2: 'solid' }
        };
        
        function showElement(id) {
            const el = document.getElementById(id);
            if (el) {
                el.classList.remove('scene1-hidden');
                el.classList.add('scene1-fade-in');
            }
        }
        
        function hideElement(id) {
            const el = document.getElementById(id);
            if (el) {
                el.classList.remove('scene1-fade-in');
                el.classList.add('scene1-hidden');
            }
        }
        
        function setArrowStyle(arrowId, style) {
            const line = document.getElementById(arrowId + '-line');
            const head = document.getElementById(arrowId + '-head');
            if (line && head) {
                if (style === 'solid') {
                    line.setAttribute('stroke-dasharray', '0');
                    head.setAttribute('fill', '#000000');
                } else {
                    line.setAttribute('stroke-dasharray', '8,4');
                    head.setAttribute('fill', '#999999');
                }
            }
        }
        
        function updateDisplay() {
            // 隐藏所有元素
            const allElements = [
                'scene1-p-container', 'scene1-arrow-p-f1', 'scene1-f1-container',
                'scene1-arrow-f1-f2', 'scene1-f2-container', 'scene1-f2-long-container',
                'scene1-f2-mini-container', 'scene1-f2-vestigial-container',
                'scene1-stat-container', 'scene1-observation-container'
            ];
            allElements.forEach(hideElement);
            
            // 显示当前步骤的元素
            const elementsToShow = stepElements[state.currentStep] || [];
            elementsToShow.forEach(showElement);
            
            // 设置箭头样式
            const arrows = arrowStyles[state.currentStep];
            if (arrows) {
                setArrowStyle('scene1-arrow-p-f1', arrows.p_f1);
                setArrowStyle('scene1-arrow-f1-f2', arrows.f1_f2);
            }
            
            // 更新步骤指示器
            const indicator = document.getElementById('scene1-step-indicator');
            if (indicator) {
                indicator.textContent = '步骤 ' + (state.currentStep + 1) + '/5';
            }
            
            // 更新按钮状态
            const prevBtn = document.getElementById('scene1-btn-prev');
            const nextBtn = document.getElementById('scene1-btn-next');
            if (prevBtn) prevBtn.disabled = (state.currentStep === 0);
            if (nextBtn) nextBtn.disabled = (state.currentStep === state.totalSteps);
        }
        
        function prevStep() {
            if (state.currentStep > 0) {
                state.currentStep--;
                updateDisplay();
            }
        }
        
        function nextStep() {
            if (state.currentStep < state.totalSteps) {
                state.currentStep++;
                updateDisplay();
            }
        }
        
        // 绑定事件
        document.getElementById('scene1-btn-prev').addEventListener('click', prevStep);
        document.getElementById('scene1-btn-next').addEventListener('click', nextStep);
        
        // 初始化
        updateDisplay();
    })();
    </script>
