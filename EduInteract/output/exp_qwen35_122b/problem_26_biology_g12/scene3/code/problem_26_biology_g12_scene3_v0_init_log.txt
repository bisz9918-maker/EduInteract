    <script>
    (function() {
        // Scene 3 实现 - 反交实验设计与预测
        // 布局坐标表对照：
        // | 元素 | 位置/范围 | 备注 |
        // | 标题 | (400,20) | 反交实验设计与预测 |
        // | 小翅♀亲本框 | (80,80)-(220,180) | 左侧亲本 |
        // | 残翅♂亲本框 | (80,240)-(220,340) | 下方亲本 |
        // | F₁代区域 | (360,60)-(580,220) | F₁展示区 |
        // | F₂区域 | (100,300)-(700,430) | F₂柱状图区 |
        // | 对比框 | (620,80)-(780,200) | 与正交对比 |

        const TOTAL_STEPS = 5;
        let currentStep = 0;

        // 获取 DOM 元素
        const btnPrev = document.getElementById('scene3-btn-prev');
        const btnNext = document.getElementById('scene3-btn-next');

        // 更新步骤显示
        function updateStep() {
            // 隐藏所有步骤
            for (let i = 0; i < TOTAL_STEPS; i++) {
                const stepGroup = document.getElementById('scene3-step' + i);
                if (stepGroup) {
                    stepGroup.classList.remove('active');
                }
            }

            // 显示当前步骤及之前的所有步骤（累积显示）
            for (let i = 0; i <= currentStep; i++) {
                const stepGroup = document.getElementById('scene3-step' + i);
                if (stepGroup) {
                    stepGroup.classList.add('active');
                }
            }

            // 更新按钮状态
            btnPrev.disabled = (currentStep === 0);
            if (currentStep === TOTAL_STEPS - 1) {
                btnNext.textContent = '完成';
                btnNext.disabled = true;
            } else {
                btnNext.textContent = '下一步 ▶';
                btnNext.disabled = false;
            }
        }

        // 按钮事件绑定
        btnPrev.addEventListener('click', function() {
            if (currentStep > 0) {
                currentStep--;
                updateStep();
            }
        });

        btnNext.addEventListener('click', function() {
            if (currentStep < TOTAL_STEPS - 1) {
                currentStep++;
                updateStep();
            }
        });

        // 初始化
        updateStep();

        // 等待 MathJax 渲染完成
        if (window.MathJax) {
            MathJax.typesetPromise();
        }
    })();
    </script>
