                document.getElementById(this.elements.prevBtn).addEventListener('click', () => {
                    if (this.currentStep > 0) {
                        this.currentStep--;
                        this.updateStep();
                    }
                });
                document.getElementById(this.elements.nextBtn).addEventListener('click', () => {
                    if (this.currentStep < this.totalSteps - 1) {
                        this.currentStep++;
                        this.updateStep();
                    }
                });
            },

            updateStep: function() {
                // 更新步骤指示器
                document.getElementById(this.elements.stepIndicator).textContent = '步骤 ' + (this.currentStep + 1) + '/6';

                // 更新按钮状态
                document.getElementById(this.elements.prevBtn).disabled = (this.currentStep === 0);
                document.getElementById(this.elements.nextBtn).disabled = (this.currentStep === this.totalSteps - 1);

                // 更新提示文本
                document.getElementById(this.elements.promptText).textContent = this.prompts[this.currentStep];

                // 清除所有高亮
                this.clearHighlights();

                // 根据步骤更新视图
                switch(this.currentStep) {
                    case 0:
                        // 初始状态：无高亮
                        break;
                    case 1:
                        // 高亮甲病患者：Ⅱ3、Ⅲ2、Ⅲ3、Ⅲ4
                        this.highlightElements(['i3_1', 'iii2', 'iii3', 'iii4']);
                        this.showKeyInfo();
                        break;
                    case 2:
                        // 高亮乙病患者：Ⅰ3、Ⅰ4、Ⅱ7、Ⅱ9、Ⅱ10
                        this.highlightElements(['i3', 'i4', 'i7', 'i9', 'i10']);
                        break;
                    case 3:
                        // 显示甲病结论，高亮Ⅱ5、Ⅱ6、Ⅲ2

                        break;
                    case 4:
                        // 显示乙病结论，高亮Ⅰ3、Ⅰ4、Ⅱ8
                        this.showConclusion('乙病', '常染色体显性遗传', 'Ⅰ3、Ⅰ4 患病，Ⅱ8 正常→显性；男女均有患者→常染色体');
                        this.highlightElements(['i3', 'i4', 'i8']);
                        break;
                    case 5:
                        // 显示基因型标注
                        this.showGenotypes();
                        this.highlightElements(['i5', 'i6', 'iii2', 'iii3']);
                        break;
                }
            },

            clearHighlights: function() {
                const highlightGroup = document.getElementById(this.elements.highlightGroup);
                highlightGroup.innerHTML = '';
                highlightGroup.style.display = 'none';

                genotypeGroup.innerHTML = '';
                genotypeGroup.style.display = 'none';

                const conclusionGroup = document.getElementById(this.elements.conclusionGroup);
                conclusionGroup.innerHTML = '';
                conclusionGroup.style.display = 'none';

                keyinfoGroup.innerHTML = '';
                keyinfoGroup.style.display = 'none';
            },
