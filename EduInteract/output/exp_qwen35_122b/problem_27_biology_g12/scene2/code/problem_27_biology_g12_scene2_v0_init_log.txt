            // 显示元素
            showElement: function(id) {
                const el = document.getElementById(id);
                if (el) {
                    el.style.display = 'block';
                    el.style.opacity = '0';
                    setTimeout(() => { el.style.opacity = '1'; }, 50);
                }
            },

            // 隐藏元素
            hideElement: function(id) {
                const el = document.getElementById(id);
                if (el) {
                    el.style.opacity = '0';
                    setTimeout(() => { el.style.display = 'none'; }, 200);
                }
            },

            // 高亮元素
            highlightElement: function(id) {
                const el = document.getElementById(id);
                if (el) {
                    el.classList.add('scene2-arrow-highlight');
                }
            },

            // 移除高亮
            removeHighlight: function(id) {
                const el = document.getElementById(id);
                if (el) {
                    el.classList.remove('scene2-arrow-highlight');
                }
            },

            // 更新场景
            updateScene: function() {
                const step = this.steps[this.currentStep];

                // 先隐藏所有可能需要的元素
                const allElements = [];
                for (let s = 0; s <= this.totalSteps; s++) {
                    if (this.steps[s]) {
                        this.steps[s].show.forEach(id => {
                            if (!allElements.includes(id)) allElements.push(id);
                        });
                        this.steps[s].hide.forEach(id => {
                            if (!allElements.includes(id)) allElements.push(id);
                        });
                    }
                }

                // 先全部隐藏
                allElements.forEach(id => {
                    const el = document.getElementById(id);
                    if (el) {
                        el.style.display = 'none';
                        el.style.opacity = '1';
                        el.classList.remove('scene2-arrow-highlight');
                    }
                });

                // 再显示当前步骤需要的元素
                if (step) {
                    step.show.forEach(id => this.showElement(id));
                    if (step.highlight) {
                        step.highlight.forEach(id => this.highlightElement(id));
                    }
                }

                // 更新按钮状态
                document.getElementById('scene2-prev-btn').disabled = (this.currentStep === 0);
                document.getElementById('scene2-next-btn').disabled = (this.currentStep === this.totalSteps);
            },

            // 下一步
            nextStep: function() {
                if (this.currentStep < this.totalSteps) {
                    this.currentStep++;
                    this.updateScene();
                }
            },

            // 上一步
            prevStep: function() {
                if (this.currentStep > 0) {
                    this.currentStep--;
                    this.updateScene();
                }
            },

            // 初始化
            init: function() {
                document.getElementById('scene2-next-btn').addEventListener('click', () => this.nextStep());
                document.getElementById('scene2-prev-btn').addEventListener('click', () => this.prevStep());
                this.updateScene();
            }
        };

        // 启动场景
        scene2.init();
    })();
