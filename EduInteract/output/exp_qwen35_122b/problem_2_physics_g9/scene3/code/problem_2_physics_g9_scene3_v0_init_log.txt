    <script>
    (function() {
        // Scene 3 实现
        // 布局坐标表（供参考）：
        // 斜面：(80,150)-(200,350)
        // 水平木板：(200,350)-(700,360)
        // 小车（初始）：(90,170)-(130,210)
        // 小车（撞击后）：(180,320)-(220,360)
        // 木块：(400,310)-(460,350)
        // 速度矢量 v：(430,300)-(530,300)
        // 摩擦力 f：(430,360)-(490,360)
        // 动能指示条 E_k：(400,270)-(500,290)
        
        const TOTAL_STEPS = 4;
        let currentStep = 0;
        
        // DOM 元素引用
        const cartInitial = document.getElementById('scene3-cart-initial');
        const cartAfter = document.getElementById('scene3-cart-after');
        const velocity = document.getElementById('scene3-velocity');
        const friction = document.getElementById('scene3-friction');
        const kineticEnergy = document.getElementById('scene3-kinetic-energy');
        const labelCartAfter = document.getElementById('scene3-label-cart-after');
        const labelV = document.getElementById('scene3-label-v');
        const labelF = document.getElementById('scene3-label-f');
        const labelEk = document.getElementById('scene3-label-ek');
        const surfaceStatus = document.getElementById('scene3-surface-status');
        const stepDisplay = document.getElementById('scene3-step-display');
        const btnPrev = document.getElementById('scene3-btn-prev');
        const btnNext = document.getElementById('scene3-btn-next');
        const conclusion = document.getElementById('scene3-conclusion');
        const ekRect = kineticEnergy.querySelector('rect');
        
        // 工具函数：显示元素
        function show(el) {
            el.style.display = 'block';
            el.style.opacity = '1';
        }
        
        // 工具函数：隐藏元素
        function hide(el) {
            el.style.display = 'none';
            el.style.opacity = '0';
        }
        
        // 渲染当前步骤
        function renderStep() {
            // 重置所有动态元素
            hide(cartAfter);
            hide(labelCartAfter);
            hide(velocity);
            hide(friction);
            hide(kineticEnergy);
            hide(labelV);
            hide(labelF);
            hide(labelEk);
            hide(surfaceStatus);
            hide(conclusion);
            
            // 根据步骤设置元素状态
            switch(currentStep) {
                case 0: // 初始状态
                    show(cartInitial);
                    stepDisplay.textContent = '步骤 1/4';
                    btnPrev.disabled = true;
                    btnNext.disabled = false;
                    break;
                    
                case 1: // 撞击瞬间
                    hide(cartInitial);
                    show(cartAfter);
                    show(labelCartAfter);
                    stepDisplay.textContent = '步骤 2/4';
                    btnPrev.disabled = false;
                    btnNext.disabled = false;
                    break;
                    
                case 2: // 普通木板
                    show(cartAfter);
                    show(labelCartAfter);
                    show(velocity);
                    show(friction);
                    show(kineticEnergy);
                    show(labelV);
                    show(labelF);
                    show(labelEk);
                    show(surfaceStatus);
                    surfaceStatus.textContent = '普通';
                    // 设置动能条长度（普通木板 - 较短）
                    ekRect.setAttribute('width', '60');
                    stepDisplay.textContent = '步骤 3/4';
                    btnPrev.disabled = false;
                    btnNext.disabled = false;
                    break;
                    
                case 3: // 光滑木板
                    show(cartAfter);
                    show(labelCartAfter);
                    show(velocity);
                    hide(friction);
                    show(kineticEnergy);
                    show(labelV);
                    hide(labelF);
                    show(labelEk);
                    show(surfaceStatus);
                    surfaceStatus.textContent = '光滑';
                    // 设置动能条长度（光滑木板 - 较长）
                    ekRect.setAttribute('width', '100');
                    stepDisplay.textContent = '步骤 4/4';
                    btnPrev.disabled = false;
                    btnNext.disabled = true;
                    show(conclusion);
                    break;
            }
        }
        
        // 事件绑定
        btnPrev.addEventListener('click', function() {
            if (currentStep > 0) {
                currentStep--;
                renderStep();
            }
        });
        
        btnNext.addEventListener('click', function() {
            if (currentStep < TOTAL_STEPS - 1) {
                currentStep++;
                renderStep();
            }
        });
        
        // 初始化渲染
        renderStep();
    })();
    </script>
