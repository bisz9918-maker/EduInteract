    <script>
    (function() {
        // Scene 2: 物质反应特性对比表
        // 坐标表对照：
        // 表格：x=100~700, y=80~400
        // 列中心 x: 154, 262, 370, 478, 586
        // 行中心 y: 140, 194, 248, 302, 356
        // 列宽=108px, 行高=54px
        
        const TOTAL_STEPS = 5;
        let currentStep = 0;
        
        // 工具函数：设置 SVG 元素可见
        function svgShow(el) {
            if (el) {
                el.style.opacity = '1';
            }
        }
        
        // 工具函数：设置 SVG 元素隐藏
        function svgHide(el) {
            if (el) {
                el.style.opacity = '0';
            }
        }
        
        // 更新步骤显示
        function updateStep() {
            const stepIndicator = document.getElementById('scene2-step-indicator');
            const prevBtn = document.getElementById('scene2-btn-prev');
            const nextBtn = document.getElementById('scene2-btn-next');
            
            // 更新步骤指示器
            stepIndicator.textContent = '步骤 ' + (currentStep + 1) + '/' + TOTAL_STEPS;
            
            // 更新按钮状态
            prevBtn.disabled = (currentStep === 0);
            nextBtn.disabled = (currentStep === TOTAL_STEPS - 1);
            
            // 根据步骤显示/隐藏元素
            switch(currentStep) {
                case 0:
                    // 步骤 0: 空白表格框架
                    for (let i = 1; i <= 5; i++) {
                        svgHide(document.getElementById('scene2-header-' + i));
                        svgHide(document.getElementById('scene2-row-title-' + i));
                        svgHide(document.getElementById('scene2-header-count'));
                        for (let j = 1; j <= 5; j++) {
                            svgHide(document.getElementById('scene2-cell-' + i + '-' + j));
                        }
                        svgHide(document.getElementById('scene2-count-' + i));
                    }
                    svgHide(document.getElementById('scene2-header-bg'));
                    svgHide(document.getElementById('scene2-row-title-bg'));
                    svgHide(document.getElementById('scene2-highlight-row'));
                    svgHide(document.getElementById('scene2-highlight-border-top'));
                    svgHide(document.getElementById('scene2-highlight-border-bottom'));
                    svgHide(document.getElementById('scene2-conclusion'));
                    break;
                    
                case 1:
                    // 步骤 1: 显示行列标题
                    svgShow(document.getElementById('scene2-header-bg'));
                    svgShow(document.getElementById('scene2-row-title-bg'));
                    for (let i = 1; i <= 5; i++) {
                        svgShow(document.getElementById('scene2-header-' + i));
                        svgShow(document.getElementById('scene2-row-title-' + i));
                    }
                    svgShow(document.getElementById('scene2-header-count'));
                    for (let i = 1; i <= 5; i++) {
                        for (let j = 1; j <= 5; j++) {
                            svgHide(document.getElementById('scene2-cell-' + i + '-' + j));
                        }
                        svgHide(document.getElementById('scene2-count-' + i));
                    }
                    svgHide(document.getElementById('scene2-highlight-row'));
                    svgHide(document.getElementById('scene2-highlight-border-top'));
                    svgHide(document.getElementById('scene2-highlight-border-bottom'));
                    svgHide(document.getElementById('scene2-conclusion'));
                    break;
                    
                case 2:
                    // 步骤 2: 逐行显示反应矩阵
                    svgShow(document.getElementById('scene2-header-bg'));
                    svgShow(document.getElementById('scene2-row-title-bg'));
                    for (let i = 1; i <= 5; i++) {
                        svgShow(document.getElementById('scene2-header-' + i));
                        svgShow(document.getElementById('scene2-row-title-' + i));
                    }
                    svgShow(document.getElementById('scene2-header-count'));
                    
                    // 逐行显示单元格和反应数
                    for (let i = 1; i <= 5; i++) {
                        setTimeout(function(row) {
                            for (let j = 1; j <= 5; j++) {
                                svgShow(document.getElementById('scene2-cell-' + row + '-' + j));
                            }
                            svgShow(document.getElementById('scene2-count-' + row));
                        }, (i - 1) * 300, i);
                    }
                    
                    svgHide(document.getElementById('scene2-highlight-row'));
                    svgHide(document.getElementById('scene2-highlight-border-top'));
                    svgHide(document.getElementById('scene2-highlight-border-bottom'));
                    svgHide(document.getElementById('scene2-conclusion'));
                    break;
                    
                case 3:
                    // 步骤 3: 高亮稀盐酸行（第 3 行）
                    svgShow(document.getElementById('scene2-header-bg'));
                    svgShow(document.getElementById('scene2-row-title-bg'));
                    for (let i = 1; i <= 5; i++) {
                        svgShow(document.getElementById('scene2-header-' + i));
                        svgShow(document.getElementById('scene2-row-title-' + i));
                    }
                    svgShow(document.getElementById('scene2-header-count'));
                    
                    for (let i = 1; i <= 5; i++) {
                        for (let j = 1; j <= 5; j++) {
                            svgShow(document.getElementById('scene2-cell-' + i + '-' + j));
                        }
                        svgShow(document.getElementById('scene2-count-' + i));
                    }
                    
                    svgShow(document.getElementById('scene2-highlight-row'));
                    svgShow(document.getElementById('scene2-highlight-border-top'));
                    svgShow(document.getElementById('scene2-highlight-border-bottom'));
                    
                    const count3 = document.getElementById('scene2-count-3');
                    count3.style.fontSize = '20px';
                    count3.style.fill = '#c62828';
                    
                    svgHide(document.getElementById('scene2-conclusion'));
                    break;
                    
                case 4:
                    // 步骤 4: 显示结论
                    svgShow(document.getElementById('scene2-header-bg'));
                    svgShow(document.getElementById('scene2-row-title-bg'));
                    for (let i = 1; i <= 5; i++) {
                        svgShow(document.getElementById('scene2-header-' + i));
                        svgShow(document.getElementById('scene2-row-title-' + i));
                    }
                    svgShow(document.getElementById('scene2-header-count'));
                    
                    for (let i = 1; i <= 5; i++) {
                        for (let j = 1; j <= 5; j++) {
                            svgShow(document.getElementById('scene2-cell-' + i + '-' + j));
                        }
                        svgShow(document.getElementById('scene2-count-' + i));
                    }
                    
                    svgShow(document.getElementById('scene2-highlight-row'));
                    svgShow(document.getElementById('scene2-highlight-border-top'));
                    svgShow(document.getElementById('scene2-highlight-border-bottom'));
                    
                    const count3_4 = document.getElementById('scene2-count-3');
                    count3_4.style.fontSize = '20px';
                    count3_4.style.fill = '#c62828';
                    
                    svgShow(document.getElementById('scene2-conclusion'));
                    break;
            }
        }
        
        // 事件绑定
        document.getElementById('scene2-btn-prev').addEventListener('click', function() {
            if (currentStep > 0) {
                currentStep--;
                updateStep();
            }
        });
        
        document.getElementById('scene2-btn-next').addEventListener('click', function() {
            if (currentStep < TOTAL_STEPS - 1) {
                currentStep++;
                updateStep();
            }
        });
        
        // 初始化
        updateStep();
    })();
    </script>
