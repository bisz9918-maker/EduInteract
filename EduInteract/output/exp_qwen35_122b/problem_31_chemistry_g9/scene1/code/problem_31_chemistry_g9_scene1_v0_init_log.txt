    <script>
    (function() {
        // Scene 1: 体心立方晶胞配位数分析
        // 布局坐标表（调整后）：
        // | 元素 | 类型 | 位置/范围 | 大小 | 标签偏移方向 | 备注 |
        // |------|------|-----------|------|-------------|------|
        // | 立方体前上左顶点 | 点 | (300, 180) | 8px 球 | 左上 | 顶点原子 |
        // | 立方体前上右顶点 | 点 | (500, 180) | 8px 球 | 右上 | 顶点原子 |
        // | 立方体前下左顶点 | 点 | (300, 320) | 8px 球 | 左下 | 顶点原子 |
        // | 立方体前下右顶点 | 点 | (500, 320) | 8px 球 | 右下 | 顶点原子 |
        // | 立方体后上左顶点 | 点 | (360, 140) | 8px 球 | 左上 | 顶点原子 |
        // | 立方体后上右顶点 | 点 | (560, 140) | 8px 球 | 右上 | 顶点原子 |
        // | 立方体后下左顶点 | 点 | (360, 280) | 8px 球 | 左下 | 顶点原子 |
        // | 立方体后下右顶点 | 点 | (560, 280) | 8px 球 | 右下 | 顶点原子 |
        // | 体心原子 | 点 | (400, 230) | 8px 球 | 下方 | 中心位置 |

        const MAX_STEP = 3;
        let currentStep = 0;

        // 获取 DOM 元素
        const btnPrev = document.getElementById('scene1-btn-prev');
        const btnNext = document.getElementById('scene1-btn-next');
        const atomCenter = document.getElementById('scene1-atom-center');
        const vertexAtoms = [
            document.getElementById('scene1-atom-vertex-1'),
            document.getElementById('scene1-atom-vertex-2'),
            document.getElementById('scene1-atom-vertex-3'),
            document.getElementById('scene1-atom-vertex-4'),
            document.getElementById('scene1-atom-vertex-5'),
            document.getElementById('scene1-atom-vertex-6'),
            document.getElementById('scene1-atom-vertex-7'),
            document.getElementById('scene1-atom-vertex-8')
        ];
        const connectLines = [
            document.getElementById('scene1-line-connect-1'),
            document.getElementById('scene1-line-connect-2'),
            document.getElementById('scene1-line-connect-3'),
            document.getElementById('scene1-line-connect-4'),
            document.getElementById('scene1-line-connect-5'),
            document.getElementById('scene1-line-connect-6'),
            document.getElementById('scene1-line-connect-7'),
            document.getElementById('scene1-line-connect-8')
        ];
        const labelCoordination = document.getElementById('scene1-label-coordination');

        // 工具函数：显示元素
        function show(el) {
            el.style.display = 'block';
            el.style.opacity = '1';
        }

        // 工具函数：隐藏元素
        function hide(el) {
            el.style.display = 'none';
            el.style.opacity = '0';
        }

        // 更新场景状态
        function updateScene() {
            // 步骤 0: 初始状态 - 无高亮
            // 步骤 1: 体心原子高亮蓝色
            // 步骤 2: 显示连接线，顶点原子依次高亮红色
            // 步骤 3: 显示配位数结论

            // 重置体心原子颜色
            atomCenter.setAttribute('fill', '#000000');

            // 重置顶点原子颜色
            vertexAtoms.forEach(atom => atom.setAttribute('fill', '#000000'));

            // 隐藏连接线
            connectLines.forEach(line => {
                line.style.opacity = '0';
            });

            // 隐藏配位数标签
            labelCoordination.style.opacity = '0';

            // 根据当前步骤应用状态
            if (currentStep >= 1) {
                // 步骤 1+: 体心原子高亮蓝色
                atomCenter.setAttribute('fill', '#1565c0');
            }

            if (currentStep >= 2) {
                // 步骤 2+: 显示连接线
                connectLines.forEach(line => {
                    line.style.opacity = '1';
                });
                // 顶点原子高亮红色（依次动画）
                animateVertexHighlight();
            }

            if (currentStep >= 3) {
                // 步骤 3+: 显示配位数结论
                labelCoordination.style.opacity = '1';
            }

            // 更新按钮状态
            btnPrev.disabled = (currentStep === 0);
            btnNext.disabled = (currentStep === MAX_STEP);
        }

        // 顶点原子依次高亮动画
        function animateVertexHighlight() {
            let index = 0;
            function highlightNext() {
                if (index < vertexAtoms.length) {
                    vertexAtoms[index].setAttribute('fill', '#c62828');
                    index++;
                    setTimeout(highlightNext, 300);
                }
            }
            // 重置所有顶点原子为黑色，然后依次高亮
            vertexAtoms.forEach(atom => atom.setAttribute('fill', '#000000'));
            highlightNext();
        }

        // 按钮事件
        btnPrev.addEventListener('click', () => {
            if (currentStep > 0) {
                currentStep--;
                updateScene();
            }
        });

        btnNext.addEventListener('click', () => {
            if (currentStep < MAX_STEP) {
                currentStep++;
                updateScene();
            }
        });

        // 初始化场景
        updateScene();
    })();
    </script>
