    <script>
    (function() {
        // 坐标表对照（网格单元 40px）
        // 甲图形：(100,100)-(340,260) 240×160 凹字形
        // 乙图形：(460,100)-(700,260) 240×160 凸字形
        // 甲平移虚线：水平 (180,140)-(220,140)，竖直 (180,140)-(180,100) 和 (220,140)-(220,100)
        // 乙平移虚线：水平 (540,220)-(580,220)，竖直 (540,220)-(540,260) 和 (580,220)-(580,260)
        // 标签位置调整：
        // - "甲"标签：从 (200,180) 移到 (140,200) 左下角空白处
        // - "乙"标签：从 (560,180) 移到 (660,200) 右下角空白处
        // - "平移后"标签：移到图形上方 (200,80) 和 (560,80)
        // - "外轮廓相同"标签：移到图形上方 (220,70) 和 (580,70)
        // - "+2 竖边"标签：移到图形外侧 (70,120) 和 (730,240)
        
        const currentStepMax = 4;
        let currentStep = 0;
        
        // 工具函数：显示元素
        function show(el) {
            if (el) {
                el.style.display = 'block';
                el.style.opacity = '0';
                setTimeout(() => {
                    el.style.transition = 'opacity 300ms';
                    el.style.opacity = '1';
                }, 10);
            }
        }
        
        // 工具函数：隐藏元素
        function hide(el) {
            if (el) {
                el.style.opacity = '0';
                setTimeout(() => {
                    el.style.display = 'none';
                }, 300);
            }
        }
        
        // 更新步骤显示
        function updateStep() {
            const stepIndicator = document.getElementById('scene1-step-indicator');
            const prevBtn = document.getElementById('scene1-prev-btn');
            const nextBtn = document.getElementById('scene1-next-btn');
            const conclusion = document.getElementById('scene1-conclusion');
            
            // 更新步骤指示器
            stepIndicator.textContent = `步骤 ${currentStep + 1}/${currentStepMax + 1}`;
            
            // 更新按钮状态
            prevBtn.disabled = (currentStep === 0);
            nextBtn.disabled = (currentStep === currentStepMax);
            
            // 隐藏所有步骤组
            document.getElementById('scene1-group-step1').style.display = 'none';
            document.getElementById('scene1-group-step2').style.display = 'none';
            document.getElementById('scene1-group-step3').style.display = 'none';
            
            // 根据当前步骤显示对应元素
            if (currentStep >= 1) {
                const group1 = document.getElementById('scene1-group-step1');
                group1.style.display = 'block';
                show(group1);
            }
            
            if (currentStep >= 2) {
                const group2 = document.getElementById('scene1-group-step2');
                group2.style.display = 'block';
                show(group2);
            }
            
            if (currentStep >= 3) {
                const group3 = document.getElementById('scene1-group-step3');
                group3.style.display = 'block';
                show(group3);
            }
            
            // 步骤 4 显示结论
            if (currentStep === 4) {
                conclusion.style.opacity = '0';
                setTimeout(() => {
                    conclusion.style.transition = 'opacity 300ms';
                    conclusion.style.opacity = '1';
                }, 10);
            } else {
                conclusion.style.opacity = '0';
            }
        }
        
        // 绑定按钮事件
        document.getElementById('scene1-prev-btn').addEventListener('click', function() {
            if (currentStep > 0) {
                currentStep--;
                updateStep();
            }
        });
        
        document.getElementById('scene1-next-btn').addEventListener('click', function() {
            if (currentStep < currentStepMax) {
                currentStep++;
                updateStep();
            }
        });
        
        // 初始化
        updateStep();
    })();
    </script>
