    <script>
    (function() {
        // Scene 1 实现
        // 布局坐标表（调整后）：
        // | 元素 | 类型 | 位置/范围 | 大小 | 标签偏移方向 | 备注 |
        // |------|------|-----------|------|-------------|------|
        // | 圆 A（左侧） | 圆 | 圆心 (200,200) | 半径 60 | - | 左圆 |
        // | 圆 B（中间） | 圆 | 圆心 (350,200) | 半径 60 | - | 中圆 |
        // | 圆 C（右侧） | 圆 | 圆心 (500,320) | 半径 60 | - | 右圆，圆心在直线 a 下方 |
        // | 直线 a | 线段 | (50,200)-(750,200) | 700×2 | - | 水平穿过圆 A、圆 B 圆心 |
        // | 圆心点 A | 实心圆 | (200,200) | 4×4 | - | 红色 |
        // | 圆心点 B | 实心圆 | (350,200) | 4×4 | - | 红色 |
        // | 圆心点 C | 实心圆 | (500,320) | 4×4 | - | 红色 |
        // | 重叠区域 1 | 阴影填充 | 圆 A 与圆 B 交集 | - | - | 蓝色 20% 透明度 |
        // | 重叠区域 2 | 阴影填充 | 圆 B 与圆 C 交集 | - | - | 蓝色 20% 透明度 |
        // | 标签"A" | text | (130,205) | 16×16 | 左侧 | 圆 A 左侧 |
        // | 标签"B" | text | (345,110) | 16×16 | 上方 | 圆 B 上方（更靠上） |
        // | 标签"C" | text | (570,325) | 16×16 | 右侧 | 圆 C 右侧 |
        // | 标签"a" | text | (760,195) | 16×16 | 右侧 | 直线 a 右端 |
        // | 标签"S" | text | (195,130) | 16×16 | 上方 | 圆 A 上方 |
        // | 标签"S" | text | (345,130) | 16×16 | 上方 | 圆 B 上方 |
        // | 标签"S" | text | (495,250) | 16×16 | 上方 | 圆 C 上方 |
        // | 标签"x" | text | (275,150) | 16×16 | 上方 | 重叠区域 1 上方 |
        // | 标签"x" | text | (465,240) | 16×16 | 上方 | 重叠区域 2 上方 |
        // | 标签"2S+2" | text | (320,420) | 40×16 | - | 底部居中，总覆盖面积 |
        // | 标签"9" | text | (340,460) | 16×16 | - | 底部，直线 a 下方覆盖面积 |

        const scene1 = {
            currentStep: 0,
            totalSteps: 4,
            
            // 步骤配置
            steps: [
                // 步骤 0：初始状态 - 显示三个圆、圆心、标签 A/B/C、直线 a
                {
                    overlap1: 0, overlap2: 0,
                    x1: 0, x2: 0,
                    sA: 0, sB: 0, sC: 0,
                    total: 0, below: 0
                },
                // 步骤 1：高亮重叠区域，显示标签 x
                {
                    overlap1: 0.2, overlap2: 0.2,
                    x1: 1, x2: 1,
                    sA: 0, sB: 0, sC: 0,
                    total: 0, below: 0
                },
                // 步骤 2：显示标签 S 和总覆盖面积
                {
                    overlap1: 0.2, overlap2: 0.2,
                    x1: 1, x2: 1,
                    sA: 1, sB: 1, sC: 1,
                    total: 1, below: 0
                },
                // 步骤 3：显示标签 9，完成所有标注
                {
                    overlap1: 0.2, overlap2: 0.2,
                    x1: 1, x2: 1,
                    sA: 1, sB: 1, sC: 1,
                    total: 1, below: 1
                }
            ],
            
            // 获取元素
            elements: {
                overlap1: document.getElementById('scene1-overlap1'),
                overlap2: document.getElementById('scene1-overlap2'),
                x1: document.getElementById('scene1-label-x1'),
                x2: document.getElementById('scene1-label-x2'),
                sA: document.getElementById('scene1-label-s-a'),
                sB: document.getElementById('scene1-label-s-b'),
                sC: document.getElementById('scene1-label-s-c'),
                total: document.getElementById('scene1-label-total'),
                below: document.getElementById('scene1-label-below'),
                prevBtn: document.getElementById('scene1-btn-prev'),
                nextBtn: document.getElementById('scene1-btn-next'),
                stepIndicator: document.getElementById('scene1-step-indicator')
            },
            
            // 更新步骤显示
            updateStep: function() {
                const config = this.steps[this.currentStep];
                
                // 使用 requestAnimationFrame 驱动过渡动画
                requestAnimationFrame(() => {
                    // 更新重叠区域透明度
                    this.elements.overlap1.style.opacity = config.overlap1;
                    this.elements.overlap2.style.opacity = config.overlap2;
                    
                    // 更新标签 x 的显示
                    this.elements.x1.style.opacity = config.x1;
                    this.elements.x2.style.opacity = config.x2;
                    
                    // 更新标签 S 的显示
                    this.elements.sA.style.opacity = config.sA;
                    this.elements.sB.style.opacity = config.sB;
                    this.elements.sC.style.opacity = config.sC;
                    
                    // 更新底部标签
                    this.elements.total.style.opacity = config.total;
                    this.elements.below.style.opacity = config.below;
                    
                    // 更新按钮状态
                    this.elements.prevBtn.disabled = (this.currentStep === 0);
                    this.elements.nextBtn.disabled = (this.currentStep === this.totalSteps - 1);
                    
                    // 更新步骤指示器
                    this.elements.stepIndicator.textContent = `步骤 ${this.currentStep + 1}/${this.totalSteps}`;
                });
            },
            
            // 初始化
            init: function() {
                // 绑定按钮事件
                this.elements.prevBtn.addEventListener('click', () => {
                    if (this.currentStep > 0) {
                        this.currentStep--;
                        this.updateStep();
                    }
                });
                
                this.elements.nextBtn.addEventListener('click', () => {
                    if (this.currentStep < this.totalSteps - 1) {
                        this.currentStep++;
                        this.updateStep();
                    }
                });
                
                // 初始化显示
                this.updateStep();
            }
        };
        
        // 启动
        scene1.init();
    })();
    </script>
