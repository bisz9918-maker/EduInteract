    <script>
    (function() {
        // Scene 1: 小旗绕点 O 顺时针旋转 90° 动态演示
        // 坐标换算：网格坐标 (gx, gy) → 画布坐标 (cx, cy)
        // cx = 100 + gx × 50
        // cy = 400 - (gy - 1) × 50
        // 例如：(5,7) → (350, 100), (5,1) → (350, 400), (9,3) → (550, 300)
        
        const scene1 = {
            currentStep: 0,
            totalSteps: 4,
            elements: {
                svg: document.getElementById('scene1-svg'),
                prevBtn: document.getElementById('scene1-btn-prev'),
                nextBtn: document.getElementById('scene1-btn-next'),
                stepIndicator: document.getElementById('scene1-step-indicator'),
                // 步骤 0：基础元素（始终显示）
                gridBg: document.getElementById('scene1-grid-bg'),
                gridLines: document.getElementById('scene1-grid-lines'),
                xAxis: document.getElementById('scene1-x-axis'),
                xArrow: document.getElementById('scene1-x-arrow'),
                labelX: document.getElementById('scene1-label-x'),
                yAxis: document.getElementById('scene1-y-axis'),
                yArrow: document.getElementById('scene1-y-arrow'),
                labelY: document.getElementById('scene1-label-y'),
                pointO: document.getElementById('scene1-point-o'),
                labelO: document.getElementById('scene1-label-o'),
                flagPole: document.getElementById('scene1-flag-pole'),
                flagFace: document.getElementById('scene1-flag-face'),
                pointA: document.getElementById('scene1-point-a'),
                labelA: document.getElementById('scene1-label-a'),
                // 步骤 1：旋转要素
                radiusOA: document.getElementById('scene1-radius-oa'),
                rotateArrow: document.getElementById('scene1-rotate-arrow'),
                rotateArrowHead: document.getElementById('scene1-rotate-arrow-head'),
                angleLabel: document.getElementById('scene1-angle-label'),
                // 步骤 2：旋转后旗子
                flagPoleRotated: document.getElementById('scene1-flag-pole-rotated'),
                flagFaceRotated: document.getElementById('scene1-flag-face-rotated'),
                // 步骤 3：顶点 A'
                pointAPrime: document.getElementById('scene1-point-a-prime'),
                labelAPrime: document.getElementById('scene1-label-a-prime')
            },
            
            // 显示元素
            show: function(el) {
                if (el) {
                    el.style.display = 'block';
                    el.style.opacity = '1';
                }
            },
            
            // 隐藏元素
            hide: function(el) {
                if (el) {
                    el.style.display = 'none';
                    el.style.opacity = '0';
                }
            },
            
            // 更新步骤显示
            updateStep: function() {
                const el = this.elements;
                
                // 步骤 0：基础元素 + 原旗子
                this.show(el.gridBg);
                this.show(el.gridLines);
                this.show(el.xAxis);
                this.show(el.xArrow);
                this.show(el.labelX);
                this.show(el.yAxis);
                this.show(el.yArrow);
                this.show(el.labelY);
                this.show(el.pointO);
                this.show(el.labelO);
                this.show(el.flagPole);
                this.show(el.flagFace);
                this.show(el.pointA);
                this.show(el.labelA);
                
                // 步骤 1 及以上：显示旋转要素
                if (this.currentStep >= 1) {
                    this.show(el.radiusOA);
                    this.show(el.rotateArrow);
                    this.show(el.rotateArrowHead);
                    this.show(el.angleLabel);
                } else {
                    this.hide(el.radiusOA);
                    this.hide(el.rotateArrow);
                    this.hide(el.rotateArrowHead);
                    this.hide(el.angleLabel);
                }
                
                // 步骤 2 及以上：显示旋转后旗子
                if (this.currentStep >= 2) {
                    this.show(el.flagPoleRotated);
                    this.show(el.flagFaceRotated);
                } else {
                    this.hide(el.flagPoleRotated);
                    this.hide(el.flagFaceRotated);
                }
                
                // 步骤 3：显示顶点 A'
                if (this.currentStep >= 3) {
                    this.show(el.pointAPrime);
                    this.show(el.labelAPrime);
                } else {
                    this.hide(el.pointAPrime);
                    this.hide(el.labelAPrime);
                }
                
                // 更新按钮状态
                el.prevBtn.disabled = (this.currentStep === 0);
                el.nextBtn.disabled = (this.currentStep === this.totalSteps - 1);
                
                // 更新步骤指示器
                el.stepIndicator.textContent = '步骤 ' + (this.currentStep + 1) + '/' + this.totalSteps;
            },
            
            // 下一步
            nextStep: function() {
                if (this.currentStep < this.totalSteps - 1) {
                    this.currentStep++;
                    this.updateStep();
                    // 步骤 1 时触发旋转箭头闪烁动画
                    if (this.currentStep === 1) {
                        this.animateRotateArrow();
                    }
                }
            },
            
            // 上一步
            prevStep: function() {
                if (this.currentStep > 0) {
                    this.currentStep--;
                    this.updateStep();
                }
            },
            
            // 旋转箭头闪烁动画
            animateRotateArrow: function() {
                const el = this.elements;
                let count = 0;
                const maxCount = 2;
                const duration = 300;
                
                const animate = () => {
                    let opacity = 0;
                    let rising = true;
                    let startTime = null;
                    
                    const flash = (timestamp) => {
                        if (!startTime) startTime = timestamp;
                        const elapsed = timestamp - startTime;
                        
                        if (elapsed < duration) {
                            if (rising) {
                                opacity += 0.05;
                                if (opacity >= 1) {
                                    opacity = 1;
                                    rising = false;
                                }
                            } else {
                                opacity -= 0.05;
                                if (opacity <= 0) {
                                    opacity = 0;
                                    rising = true;
                                }
                            }
                            el.rotateArrow.style.opacity = opacity;
                            el.rotateArrowHead.style.opacity = opacity;
                            el.angleLabel.style.opacity = opacity;
                            requestAnimationFrame(flash);
                        } else {
                            count++;
                            if (count < maxCount) {
                                requestAnimationFrame(animate);
                            } else {
                                // 动画结束，保持显示
                                el.rotateArrow.style.opacity = '1';
                                el.rotateArrowHead.style.opacity = '1';
                                el.angleLabel.style.opacity = '1';
                            }
                        }
                    };
                    
                    requestAnimationFrame(flash);
                };
                
                animate();
            },
            
            // 初始化
            init: function() {
                this.elements.prevBtn.addEventListener('click', () => this.prevStep());
                this.elements.nextBtn.addEventListener('click', () => this.nextStep());
                this.updateStep();
            }
        };
        
        scene1.init();
    })();
    </script>
