    <script>
    (function() {
        // Scene 1 实现
        // 坐标表对照：
        // 正方体正面：(120,200)-(200,200)-(180,320)-(100,320)
        // 正方体顶面：(120,200)-(240,140)-(260,260)-(140,320)
        // 正方体右侧面：(200,200)-(280,260)-(260,380)-(180,320)
        // 爱心图案：(140,240) 中心
        // 太阳图案：(180,190) 中心
        // 笑脸图案：(220,270) 中心
        // 爱心标签：(80,265)
        // 太阳标签：(180,115)
        // 笑脸标签：(280,295)
        // 右侧标题：(500,80)
        // 说明框 1-3：(480,140)-(760,190/260/330)
        // 说明文字：(500,168/238/308) - 在框内居中
        
        const scene1 = {
            currentStep: 0,
            totalSteps: 4,
            
            elements: {
                btnPrev: document.getElementById('scene1-btn-prev'),
                btnNext: document.getElementById('scene1-btn-next'),
                highlight1: document.getElementById('scene1-highlight-1'),
                highlight2: document.getElementById('scene1-highlight-2'),
                highlight3: document.getElementById('scene1-highlight-3'),
                iconHeart: document.getElementById('scene1-icon-heart'),
                iconSun: document.getElementById('scene1-icon-sun'),
                iconSmile: document.getElementById('scene1-icon-smile'),
                labelHeart: document.getElementById('scene1-label-heart'),
                labelSun: document.getElementById('scene1-label-sun'),
                labelSmile: document.getElementById('scene1-label-smile'),
                text1: document.getElementById('scene1-text-1'),
                text2: document.getElementById('scene1-text-2'),
                text3: document.getElementById('scene1-text-3')
            },
            
            init: function() {
                this.elements.btnPrev.addEventListener('click', () => this.prevStep());
                this.elements.btnNext.addEventListener('click', () => this.nextStep());
                this.renderStep();
            },
            
            prevStep: function() {
                if (this.currentStep > 0) {
                    this.currentStep--;
                    this.renderStep();
                }
            },
            
            nextStep: function() {
                if (this.currentStep < this.totalSteps) {
                    this.currentStep++;
                    this.renderStep();
                }
            },
            
            renderStep: function() {
                const el = this.elements;
                
                // 更新按钮状态
                el.btnPrev.disabled = (this.currentStep === 0);
                el.btnNext.disabled = (this.currentStep === this.totalSteps);
                
                // 重置所有高亮
                el.highlight1.style.opacity = '0';
                el.highlight2.style.opacity = '0';
                el.highlight3.style.opacity = '0';
                el.iconHeart.classList.remove('scene1-icon-highlight');
                el.iconSun.classList.remove('scene1-icon-highlight');
                el.iconSmile.classList.remove('scene1-icon-highlight');
                el.labelHeart.style.fontWeight = 'normal';
                el.labelSun.style.fontWeight = 'normal';
                el.labelSmile.style.fontWeight = 'normal';
                
                // 根据步骤设置高亮
                switch(this.currentStep) {
                    case 0:
                        // 初始状态：无高亮
                        el.text1.textContent = '爱心 ↔ 爱心';
                        el.text2.textContent = '太阳 ↔ 太阳';
                        el.text3.textContent = '笑脸 ↔ 笑脸';
                        break;
                        
                    case 1:
                        // 高亮爱心组
                        el.highlight1.style.opacity = '1';
                        el.iconHeart.classList.add('scene1-icon-highlight');
                        el.labelHeart.style.fontWeight = 'bold';
                        el.text1.textContent = '爱心 ↔ 爱心（正面与背面）';
                        el.text2.textContent = '太阳 ↔ 太阳';
                        el.text3.textContent = '笑脸 ↔ 笑脸';
                        break;
                        
                    case 2:
                        // 高亮太阳组
                        el.highlight2.style.opacity = '1';
                        el.iconSun.classList.add('scene1-icon-highlight');
                        el.labelSun.style.fontWeight = 'bold';
                        el.text1.textContent = '爱心 ↔ 爱心';
                        el.text2.textContent = '太阳 ↔ 太阳（顶面与底面）';
                        el.text3.textContent = '笑脸 ↔ 笑脸';
                        break;
                        
                    case 3:
                        // 高亮笑脸组
                        el.highlight3.style.opacity = '1';
                        el.iconSmile.classList.add('scene1-icon-highlight');
                        el.labelSmile.style.fontWeight = 'bold';
                        el.text1.textContent = '爱心 ↔ 爱心';
                        el.text2.textContent = '太阳 ↔ 太阳';
                        el.text3.textContent = '笑脸 ↔ 笑脸（右侧面与左侧面）';
                        break;
                        
                    case 4:
                        // 最终状态：全部高亮
                        el.highlight1.style.opacity = '1';
                        el.highlight2.style.opacity = '1';
                        el.highlight3.style.opacity = '1';
                        el.iconHeart.classList.add('scene1-icon-highlight');
                        el.iconSun.classList.add('scene1-icon-highlight');
                        el.iconSmile.classList.add('scene1-icon-highlight');
                        el.labelHeart.style.fontWeight = 'bold';
                        el.labelSun.style.fontWeight = 'bold';
                        el.labelSmile.style.fontWeight = 'bold';
                        el.text1.textContent = '三组对面图案分别相同';
                        el.text2.textContent = '';
                        el.text3.textContent = '';
                        break;
                }
            }
        };
        
        scene1.init();
    })();
    </script>
