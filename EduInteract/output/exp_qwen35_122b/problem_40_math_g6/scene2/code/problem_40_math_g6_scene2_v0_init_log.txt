                border.setAttribute('stroke', '#000000');
                border.setAttribute('stroke-width', '2');
            });
            
            // 隐藏所有标记
            document.getElementById('scene2-opt-a-check').setAttribute('opacity', '0');
            document.getElementById('scene2-opt-b-cross').setAttribute('opacity', '0');
            document.getElementById('scene2-opt-c-cross').setAttribute('opacity', '0');
            document.getElementById('scene2-opt-d-cross').setAttribute('opacity', '0');
            
            // 隐藏所有对面连线
            ['a', 'b', 'd'].forEach(opt => {
                document.getElementById(`scene2-opt-${opt}-line1`).setAttribute('opacity', '0');
                document.getElementById(`scene2-opt-${opt}-line2`).setAttribute('opacity', '0');
                if (document.getElementById(`scene2-opt-${opt}-line3`)) {
                    document.getElementById(`scene2-opt-${opt}-line3`).setAttribute('opacity', '0');
                }
            });
            
            // 隐藏参考图对面连线
            document.getElementById('scene2-ref-line1').setAttribute('opacity', '0');
            document.getElementById('scene2-ref-line2').setAttribute('opacity', '0');
            document.getElementById('scene2-ref-line3').setAttribute('opacity', '0');
            
            // 隐藏立体验证图
            cubeVerify.setAttribute('opacity', '0');
            
            // 根据步骤更新显示
            switch(step) {
                case 0:
                    resultText.textContent = '请分析各选项的对面关系';
                    // 显示参考图对面连线
                    document.getElementById('scene2-ref-line1').setAttribute('opacity', '1');
                    document.getElementById('scene2-ref-line2').setAttribute('opacity', '1');
                    document.getElementById('scene2-ref-line3').setAttribute('opacity', '1');
                    break;
                    
                case 1:
                    // 高亮选项 A
                    optBorders.a.setAttribute('stroke', highlightColor);
                    optBorders.a.setAttribute('stroke-width', '3');
                    // 显示选项 A 对面连线
                    document.getElementById('scene2-opt-a-line1').setAttribute('opacity', '1');
                    document.getElementById('scene2-opt-a-line2').setAttribute('opacity', '1');
                    document.getElementById('scene2-opt-a-line3').setAttribute('opacity', '1');
                    // 显示正确标记
                    document.getElementById('scene2-opt-a-check').setAttribute('opacity', '1');
                    resultText.textContent = '✓ 选项 A：三组对面图案相同';
                    break;
                    
                case 2:
                    // 高亮选项 B
                    optBorders.b.setAttribute('stroke', highlightColor);
                    optBorders.b.setAttribute('stroke-width', '3');
                    // 显示选项 B 对面连线
                    document.getElementById('scene2-opt-b-line1').setAttribute('opacity', '1');
                    document.getElementById('scene2-opt-b-line2').setAttribute('opacity', '1');
                    // 显示错误标记
                    document.getElementById('scene2-opt-b-cross').setAttribute('opacity', '1');
                    resultText.textContent = '✗ 选项 B：顶面图案不符（应为太阳）';
                    break;
                    
                case 3:
                    // 高亮选项 C
                    optBorders.c.setAttribute('stroke', highlightColor);
                    optBorders.c.setAttribute('stroke-width', '3');
                    // 显示错误标记
                    document.getElementById('scene2-opt-c-cross').setAttribute('opacity', '1');
                    resultText.textContent = '✗ 选项 C：图案数量不对（太阳 3 个，笑脸 1 个）';
                    break;
                    
                case 4:
                    // 高亮选项 D
                    optBorders.d.setAttribute('stroke', highlightColor);
                    optBorders.d.setAttribute('stroke-width', '3');
                    // 显示选项 D 对面连线
                    document.getElementById('scene2-opt-d-line1').setAttribute('opacity', '1');
                    document.getElementById('scene2-opt-d-line2').setAttribute('opacity', '1');
                    // 显示错误标记
                    document.getElementById('scene2-opt-d-cross').setAttribute('opacity', '1');
                    resultText.textContent = '✗ 选项 D：相邻面关系不符';
                    break;
                    
                case 5:
                    // 保持选项 A 高亮
                    optBorders.a.setAttribute('stroke', highlightColor);
                    optBorders.a.setAttribute('stroke-width', '3');
                    // 显示选项 A 对面连线
                    document.getElementById('scene2-opt-a-line1').setAttribute('opacity', '1');
                    document.getElementById('scene2-opt-a-line2').setAttribute('opacity', '1');
                    document.getElementById('scene2-opt-a-line3').setAttribute('opacity', '1');
                    // 显示立体验证图
                    cubeVerify.setAttribute('opacity', '1');
                    resultText.textContent = '✓ 选项 A 可折叠成题目立体图';
                    break;
                    
                case 6:
                    // 选项 A 绿色高亮
                    optBorders.a.setAttribute('stroke', correctColor);
                    optBorders.a.setAttribute('stroke-width', '4');
                    // 显示正确标记
                    document.getElementById('scene2-opt-a-check').setAttribute('opacity', '1');
                    // 保持立体验证图
                    cubeVerify.setAttribute('opacity', '1');
                    resultText.textContent = '✓ 正确答案：A';
                    break;
            }
            
            // 更新按钮状态
            btnPrev.disabled = (step === 0);
            btnNext.disabled = (step === totalSteps);
        }
        
        // 按钮事件
        btnPrev.addEventListener('click', () => {
            if (currentStep.value > 0) {
                currentStep.value--;
                updateStep(currentStep.value);
            }
        });
        
        btnNext.addEventListener('click', () => {
            if (currentStep.value < totalSteps) {
                currentStep.value++;
                updateStep(currentStep.value);
            }
        });
        
        // 初始化
        updateStep(0);
    })();
