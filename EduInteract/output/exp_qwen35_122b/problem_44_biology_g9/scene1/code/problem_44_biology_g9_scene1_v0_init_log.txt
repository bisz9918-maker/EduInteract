        // Scene 1: 血液循环路径与不同器官的血液成分变化对比

        // | 元素 | 类型 | 位置/范围 | 大小 | 标签偏移方向 | 备注 |

        // | 血管丙（出血管） | 箭头线段 | (520,250)-(720,250) | 200×6 | - | 右侧出血管 |
        // | 丙标签 | text | (620,220) | ~20×16 | 上方 | 在血管丙上方 |
        // | 步骤 0-5 说明区 | text | (50,350)-(750,420) | 700×70 | - | 底部说明文字区 |

        // 状态变量

        // 颜色定义

            arterial: '#1565c0',  // 动脉血（蓝色）

                stepText: '步骤 5/5',

            }
        ];

        // 获取 DOM 元素
        const elements = {
            arrowA: document.getElementById('scene1-arrow-a'),
            arrowheadA: document.getElementById('scene1-arrowhead-a'),
            rectB: document.getElementById('scene1-rect-b'),
            arrowC: document.getElementById('scene1-arrow-c'),
            arrowheadC: document.getElementById('scene1-arrowhead-c'),
            labelA: document.getElementById('scene1-label-a'),
            labelB: document.getElementById('scene1-label-b'),
            labelC: document.getElementById('scene1-label-c'),
            labelArterial: document.getElementById('scene1-label-arterial'),
            labelVenous: document.getElementById('scene1-label-venous'),
            labelNutrient: document.getElementById('scene1-label-nutrient'),
            labelInsulin: document.getElementById('scene1-label-insulin'),
            highlightA: document.getElementById('scene1-highlight-a'),
            highlightD: document.getElementById('scene1-highlight-d'),
            textDesc: document.getElementById('scene1-text-desc'),
            textStep: document.getElementById('scene1-text-step'),
            btnPrevBg: document.getElementById('scene1-btn-prev-bg'),
            btnPrevText: document.getElementById('scene1-btn-prev-text'),
            btnNextBg: document.getElementById('scene1-btn-next-bg'),
            btnNextText: document.getElementById('scene1-btn-next-text')
        };

        // 显示/隐藏工具函数
        function show(el) {
            el.style.display = 'block';
            el.style.opacity = '1';
        }

        function hide(el) {
            el.style.display = 'none';
            el.style.opacity = '0';
        }

        // 更新场景
        function updateScene(step) {
            const data = stepData[step];

            // 更新乙结构标签
            elements.labelB.textContent = data.labelB;

            // 更新血液颜色
            elements.arrowA.style.stroke = COLORS[data.bloodA];
            elements.arrowheadA.style.fill = COLORS[data.bloodA];
            elements.arrowC.style.stroke = COLORS[data.bloodC];
            elements.arrowheadC.style.fill = COLORS[data.bloodC];
            elements.rectB.style.fill = data.bloodA === 'arterial' ? '#bbdefb' : (data.bloodC === 'venous' ? '#ffcdd2' : '#e0e0e0');

            // 更新动态标签显示
            data.showArterial ? show(elements.labelArterial) : hide(elements.labelArterial);
            data.showVenous ? show(elements.labelVenous) : hide(elements.labelVenous);
            data.showNutrient ? show(elements.labelNutrient) : hide(elements.labelNutrient);
            data.showInsulin ? show(elements.labelInsulin) : hide(elements.labelInsulin);

            // 更新高亮框
            if (data.highlightA) {
                show(elements.highlightA);
                elements.highlightA.style.stroke = COLORS.highlight;
                elements.highlightA.setAttribute('fill', 'rgba(46, 125, 50, 0.1)');
            } else {
                hide(elements.highlightA);
            }

            if (data.highlightD) {
                show(elements.highlightD);
                elements.highlightD.style.stroke = COLORS.highlight;
                elements.highlightD.setAttribute('fill', 'rgba(46, 125, 50, 0.1)');
            } else {
                hide(elements.highlightD);
            }

            // 更新说明文字
            elements.textDesc.textContent = data.desc;
            elements.textStep.textContent = data.stepText;

            // 更新按钮状态
            elements.btnPrevBg.style.opacity = step === 0 ? '0.5' : '1';
            elements.btnPrevText.style.opacity = step === 0 ? '0.5' : '1';
            elements.btnNextBg.style.opacity = step === totalSteps ? '0.5' : '1';
            elements.btnNextText.style.opacity = step === totalSteps ? '0.5' : '1';
        }

        // 上一步按钮点击
        function prevStep() {
            if (currentStep > 0) {
                currentStep--;
                updateScene(currentStep);
            }
        }

        // 下一步按钮点击
        function nextStep() {
            if (currentStep < totalSteps) {
                currentStep++;
                updateScene(currentStep);
            }
        }

        // 绑定事件
        elements.btnPrevBg.addEventListener('click', prevStep);
        elements.btnNextBg.addEventListener('click', nextStep);

        // 初始化
        updateScene(currentStep);
    })();
