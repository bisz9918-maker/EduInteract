    <script>
    (function() {
        // Scene 2: 胰岛调节血糖机制
        // 状态变量
        let glucoseLevel = 3.0;  // 血糖浓度 (mmol/L), 范围 3.0-12.0
        let isDragging = false;

        // DOM 元素引用
        const svg = document.getElementById('scene2-svg');
        const sliderHandle = document.getElementById('scene2-slider-handle');
        const glucoseValueText = document.getElementById('scene2-glucose-value');
        const statusText = document.getElementById('scene2-status-text');
        const insulinBar = document.getElementById('scene2-bar-insulin');
        const moleculesGroup = document.getElementById('scene2-molecules-group');

        // 坐标参数
        const SLIDER_X_MIN = 80;
        const SLIDER_X_MAX = 180;
        const GLUCOSE_MIN = 3.0;
        const GLUCOSE_MAX = 12.0;

        // 初始化
        function init() {
            updateDisplay(glucoseLevel);
            bindEvents();
        }

        // 根据血糖值计算滑块位置
        function sliderXFromGlucose(glucose) {
            const ratio = (glucose - GLUCOSE_MIN) / (GLUCOSE_MAX - GLUCOSE_MIN);
            return SLIDER_X_MIN + ratio * (SLIDER_X_MAX - SLIDER_X_MIN);
        }

        // 根据滑块位置计算血糖值
        function glucoseFromSliderX(x) {
            const ratio = (x - SLIDER_X_MIN) / (SLIDER_X_MAX - SLIDER_X_MIN);
            return GLUCOSE_MIN + ratio * (GLUCOSE_MAX - GLUCOSE_MIN);
        }

        // 计算胰岛素柱状图高度 (10-100px)
        function insulinHeightFromGlucose(glucose) {
            return (glucose - 3) / 9 * 90 + 10;
        }

        // 计算胰岛素分子数量 (2-15 个)
        function moleculeCountFromGlucose(glucose) {
            return Math.floor((glucose - 3) / 9 * 13) + 2;
        }

        // 获取血糖状态文本和颜色
        function getStatusInfo(glucose) {
            if (glucose < 4.0) {
                return { text: '低血糖', color: '#1565c0' };
            } else if (glucose <= 6.0) {
                return { text: '正常血糖', color: '#2e7d32' };
            } else if (glucose <= 8.0) {
                return { text: '血糖偏高', color: '#f57c00' };
            } else {
                return { text: '高血糖', color: '#c62828' };
            }
        }

        // 更新显示
        function updateDisplay(glucose) {
            glucoseLevel = glucose;

            // 更新滑块手柄位置
            const sliderX = sliderXFromGlucose(glucose);
            sliderHandle.setAttribute('x', sliderX);

            // 更新血糖数值显示
            glucoseValueText.textContent = glucose.toFixed(1) + ' mmol/L';

            // 更新胰岛素柱状图高度
            const height = insulinHeightFromGlucose(glucose);
            insulinBar.setAttribute('y', 350 - height);
            insulinBar.setAttribute('height', height);

            // 更新胰岛素分子
            const count = moleculeCountFromGlucose(glucose);
            updateMolecules(count);

            // 更新状态指示
            const statusInfo = getStatusInfo(glucose);
            statusText.textContent = statusInfo.text;
            statusText.setAttribute('fill', statusInfo.color);
        }

        // 更新胰岛素分子显示
        function updateMolecules(count) {
            moleculesGroup.innerHTML = '';
            const colors = ['#2e7d32', '#43a047', '#66bb6a'];
            
            for (let i = 0; i < count; i++) {
                const circle = document.createElementNS('http://www.w3.org/2000/svg', 'circle');
                // 在 (520-620, 180-220) 区域内随机分布
                const cx = 520 + (i * 67) % 100;
                const cy = 185 + (i * 23) % 35;
                circle.setAttribute('cx', cx);
                circle.setAttribute('cy', cy);
                circle.setAttribute('r', 6);
                circle.setAttribute('fill', colors[i % colors.length]);
                circle.setAttribute('opacity', '0.7');
                moleculesGroup.appendChild(circle);
            }
        }

        // 绑定事件
        function bindEvents() {
            // 鼠标事件
            sliderHandle.addEventListener('mousedown', onMouseDown);
            document.addEventListener('mousemove', onMouseMove);
            document.addEventListener('mouseup', onMouseUp);

            // 触摸事件
            sliderHandle.addEventListener('touchstart', onTouchStart, { passive: false });
            document.addEventListener('touchmove', onTouchMove, { passive: false });
            document.addEventListener('touchend', onTouchEnd);
        }

        // 鼠标按下
        function onMouseDown(e) {
            isDragging = true;
            e.preventDefault();
        }

        // 鼠标移动
        function onMouseMove(e) {
            if (!isDragging) return;
            const rect = svg.getBoundingClientRect();
            const x = e.clientX - rect.left;
            updateGlucoseFromPosition(x);
        }

        // 鼠标释放
        function onMouseUp(e) {
            isDragging = false;
        }

        // 触摸开始
        function onTouchStart(e) {
            isDragging = true;
            e.preventDefault();
        }

        // 触摸移动
        function onTouchMove(e) {
            if (!isDragging) return;
            e.preventDefault();
            const rect = svg.getBoundingClientRect();
            const touch = e.touches[0];
            const x = touch.clientX - rect.left;
            updateGlucoseFromPosition(x);
        }

        // 触摸结束
        function onTouchEnd(e) {
            isDragging = false;
        }

        // 根据位置更新血糖值
        function updateGlucoseFromPosition(x) {
            let glucose = glucoseFromSliderX(x);
            // 限制范围
            glucose = Math.max(GLUCOSE_MIN, Math.min(GLUCOSE_MAX, glucose));
            updateDisplay(glucose);
        }

        // 启动
        init();
    })();
    </script>
