    <script>
    (function() {
        // Scene 1: 地球经纬网示意图
        // 布局坐标表对照：
        // 经线 80°E: x=200, y=100~400
        // 经线 120°E: x=600, y=100~400
        // 经线 60°E: x=0, y=100~400 (辅助虚线)
        // 经线 100°E: x=400, y=100~400 (辅助虚线)
        // 纬线 60°N: x=0~800, y=150
        // 纬线 40°N: x=0~800, y=300
        // 纬线 30°N: x=0~800, y=400 (辅助虚线)
        // M 点：(200, 150)
        // N 点：(600, 150)
        // 标签 80°: (200, 90) - 调整确保距顶部边缘≥20px
        // 标签 120°: (600, 90) - 调整确保距顶部边缘≥20px
        // 标签 60°: (720, 155)
        // 标签 40°: (720, 305)
        // 标签 M: (210, 170)
        // 标签 N: (610, 170)
        // 提示框：(200, 430) 400×40 - 调整确保距底部边缘≥20px
        // 提示文字：(400, 455) - 垂直居中于提示框

        const currentStepMax = 3;
        let currentStep = 0;

        // 元素引用
        const elements = {
            // 经度标签
            lon80: document.getElementById('scene1-label-lon-80'),
            lon120: document.getElementById('scene1-label-lon-120'),
            // 纬度标签
            lat60: document.getElementById('scene1-label-lat-60'),
            lat40: document.getElementById('scene1-label-lat-40'),
            // M 点相关
            pointM: document.getElementById('scene1-point-m'),
            labelM: document.getElementById('scene1-label-m'),
            // N 点相关
            pointN: document.getElementById('scene1-point-n'),
            labelN: document.getElementById('scene1-label-n'),
            // 提示框
            hintBox: document.getElementById('scene1-hint-box'),
            hintText: document.getElementById('scene1-hint-text'),
            // 控件
            btnPrev: document.getElementById('scene1-btn-prev'),
            btnNext: document.getElementById('scene1-btn-next'),
            indicator: document.getElementById('scene1-step-indicator')
        };

        // 工具函数：显示元素（同时设置 display 和 opacity）
        function show(el) {
            if (el) {
                el.style.display = 'block';
                el.style.opacity = '1';
                el.classList.add('fade-in');
            }
        }

        // 工具函数：隐藏元素
        function hide(el) {
            if (el) {
                el.style.opacity = '0';
                el.style.display = 'none';
                el.classList.remove('fade-in');
            }
        }

        // 更新显示状态
        function updateDisplay() {
            // 步骤 0: 只显示经纬网，不显示点、标签、提示
            // 步骤 1: 显示 M 点、标签 M、经度 80°、纬度 60°
            // 步骤 2: 显示 N 点、标签 N、经度 120°、纬度 40°
            // 步骤 3: 显示提示框和文字

            // 步骤 0 基础状态：所有点、标签、提示都隐藏
            hide(elements.lon80);
            hide(elements.lon120);
            hide(elements.lat60);
            hide(elements.lat40);
            hide(elements.pointM);
            hide(elements.labelM);
            hide(elements.pointN);
            hide(elements.labelN);
            hide(elements.hintBox);
            hide(elements.hintText);

            // 根据当前步骤显示对应元素
            if (currentStep >= 1) {
                show(elements.lon80);
                show(elements.lat60);
                show(elements.pointM);
                show(elements.labelM);
            }

            if (currentStep >= 2) {
                show(elements.lon120);
                show(elements.lat40);
                show(elements.pointN);
                show(elements.labelN);
            }

            if (currentStep >= 3) {
                show(elements.hintBox);
                show(elements.hintText);
            }

            // 更新按钮状态
            elements.btnPrev.disabled = (currentStep === 0);
            elements.btnNext.disabled = (currentStep === currentStepMax);

            // 更新指示器文字
            elements.indicator.textContent = `步骤 ${currentStep + 1}/${currentStepMax}`;
        }

        // 按钮事件处理
        elements.btnPrev.addEventListener('click', function() {
            if (currentStep > 0) {
                currentStep--;
                updateDisplay();
            }
        });

        elements.btnNext.addEventListener('click', function() {
            if (currentStep < currentStepMax) {
                currentStep++;
                updateDisplay();
            }
        });

        // 初始化显示
        updateDisplay();
    })();
    </script>
