    <script>
    (function() {
        // ===== 布局坐标表（实现对照依据）=====
        // | 元素 | 类型 | 位置/范围 | 大小 | 标签偏移方向 | 备注 |
        // | 主视图背景 | 矩形 | (60,130)-(460,350) | 400×220 | - | 经纬网主图区（不覆盖标签） |
        // | 经线 80° | 竖线 | (100,80)-(100,340) | 2×260 | - | 主线 |
        // | 经线 100° | 竖线 | (200,80)-(200,340) | 2×260 | - | 中线 |
        // | 经线 120° | 竖线 | (300,80)-(300,340) | 2×260 | - | 主线 |
        // | 经线 140° | 竖线 | (400,80)-(400,340) | 2×260 | - | 辅助线 |
        // | 纬线 40° | 横线 | (60,280)-(460,280) | 400×2 | - | 辅助线 |
        // | 纬线 60° | 横线 | (60,140)-(460,140) | 400×2 | - | 主线 |
        // | 经度标签 80° | text | (90,60) | 30×16 | 上方 | 经线上方 |
        // | 经度标签 100° | text | (185,60) | 40×16 | 上方 | 经线上方 |
        // | 经度标签 120° | text | (285,60) | 40×16 | 上方 | 经线上方 |
        // | 经度标签 140° | text | (385,60) | 40×16 | 上方 | 经线上方 |
        // | 纬度标签 40° | text | (470,285) | 30×16 | 右侧 | 纬线右侧 |
        // | 纬度标签 60° | text | (470,145) | 30×16 | 右侧 | 纬线右侧 |
        // | M 点 | 圆点 | (100,140) | 8px 直径 | - | 80°E,60°N |
        // | N 点 | 圆点 | (300,140) | 8px 直径 | - | 120°E,60°N |
        // | M 标签 | text | (100,120) | 20×18 | 上方 | M 点上方 |
        // | N 标签 | text | (300,120) | 20×18 | 上方 | N 点上方 |
        // | 纬线航线 | 直线 | (100,140)-(300,140) | 200×2 | - | 灰色虚线 |
        // | 大圆航线 | 弧线 | (100,140) 至 (300,140)，控制点 (150,100)/(250,100) | 曲线 | - | 绿色实线 |
        // | 方向箭头 1 | 箭头 | (140,125) | 15×15 | - | 指向东北 |
        // | 方向箭头 2 | 箭头 | (200,115) | 15×15 | - | 指向正东 |
        // | 方向箭头 3 | 箭头 | (260,135) | 15×15 | - | 指向东南 |
        // | 箭头标签 1 | text | (470,100) | 30×14 | 右侧 | "东北"（主视图右侧，垂直排列） |
        // | 箭头标签 2 | text | (470,120) | 30×14 | 右侧 | "正东"（主视图右侧，垂直排列） |
        // | 箭头标签 3 | text | (470,140) | 30×14 | 右侧 | "东南"（主视图右侧，垂直排列） |
        // | 副视图背景 | 矩形 | (560,180)-(780,380) | 220×200 | - | 地球示意图 |
        // | 地球轮廓 | 圆 | (670,280) | 100px 直径 | - | 副视图 |
        // | M' 点 | 圆点 | (590,220) | 8px 直径 | - | 100°W,60°S |
        // | M' 标签 | text | (500,215) | 25×18 | 左上方 | M' 点左上方（副视图外） |
        // | 地心连线 | 虚线 | (100,140)-(590,220) | 直线 | - | 穿过地心示意 |
        // | 坐标标注 | text | (500,185) | 80×16 | - | "(60°S, 100°W)"（副视图外） |
        
        // ===== 状态管理 =====
        let currentStep = 0;
        const maxStep = 4;
        
        // ===== DOM 元素引用 =====
        const elements = {
            // 纬线航线
            latitudeRoute: document.getElementById('scene2-latitude-route'),
            // 大圆航线
            greatCircle: document.getElementById('scene2-great-circle'),
            // 方向箭头组
            arrows: document.getElementById('scene2-arrows'),
            // 地心连线
            centerLine: document.getElementById('scene2-center-line'),
            // 副视图背景
            subviewBg: document.getElementById('scene2-subview-bg'),
            // 地球轮廓
            earth: document.getElementById('scene2-earth'),
            // M' 点
            mPrimePoint: document.getElementById('scene2-m-prime-point'),
            // M' 标签
            mPrimeLabel: document.getElementById('scene2-m-prime-label'),
            // 坐标标注
            coordinateLabel: document.getElementById('scene2-coordinate-label'),
            // 按钮
            prevBtn: document.getElementById('scene2-prev'),
            nextBtn: document.getElementById('scene2-next')
        };
        
        // ===== 工具函数：显示/隐藏元素 =====
        function show(el) {
            if (el) {
                el.classList.remove('hidden');
                el.style.display = '';
                el.style.opacity = '1';
            }
        }
        
        function hide(el) {
            if (el) {
                el.classList.add('hidden');
                el.style.display = 'none';
                el.style.opacity = '0';
            }
        }
        
        // ===== 场景更新函数 =====
        function updateScene() {
            // 步骤 0（初始状态）：显示经纬网、M/N 点、纬线航线
            // 步骤 1：新增大圆航线
            // 步骤 2：新增方向箭头
            // 步骤 3：新增副视图、M' 点、地心连线
            // 步骤 4：新增 M' 坐标标注
            
            // 纬线航线：步骤 0-4 都显示
            if (currentStep >= 0) {
                show(elements.latitudeRoute);
            } else {
                hide(elements.latitudeRoute);
            }
            
            // 大圆航线：步骤 1-4 显示
            if (currentStep >= 1) {
                show(elements.greatCircle);
            } else {
                hide(elements.greatCircle);
            }
            
            // 方向箭头：步骤 2-4 显示
            if (currentStep >= 2) {
                show(elements.arrows);
            } else {
                hide(elements.arrows);
            }
            
            // 地心连线：步骤 3-4 显示
            if (currentStep >= 3) {
                show(elements.centerLine);
            } else {
                hide(elements.centerLine);
            }
            
            // 副视图背景：步骤 3-4 显示
            if (currentStep >= 3) {
                show(elements.subviewBg);
            } else {
                hide(elements.subviewBg);
            }
            
            // 地球轮廓：步骤 3-4 显示
            if (currentStep >= 3) {
                show(elements.earth);
            } else {
                hide(elements.earth);
            }
            
            // M' 点：步骤 3-4 显示
            if (currentStep >= 3) {
                show(elements.mPrimePoint);
            } else {
                hide(elements.mPrimePoint);
            }
            
            // M' 标签：步骤 3-4 显示
            if (currentStep >= 3) {
                show(elements.mPrimeLabel);
            } else {
                hide(elements.mPrimeLabel);
            }
            
            // 坐标标注：步骤 4 显示
            if (currentStep >= 4) {
                show(elements.coordinateLabel);
            } else {
                hide(elements.coordinateLabel);
            }
            
            // 更新按钮状态
            elements.prevBtn.disabled = (currentStep === 0);
            elements.nextBtn.disabled = (currentStep === maxStep);
        }
        
        // ===== 事件绑定 =====
        elements.prevBtn.addEventListener('click', function() {
            if (currentStep > 0) {
                currentStep--;
                updateScene();
            }
        });
        
        elements.nextBtn.addEventListener('click', function() {
            if (currentStep < maxStep) {
                currentStep++;
                updateScene();
            }
        });
        
        // ===== 初始化 =====
        updateScene();
    })();
    </script>
