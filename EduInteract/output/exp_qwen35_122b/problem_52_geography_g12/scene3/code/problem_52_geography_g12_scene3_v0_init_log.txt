    <script>
    (function() {
        // Scene 3 实现 - 时区计算、昼长计算与太阳高度
        // 布局坐标表对照：
        // 左半球：圆心 (200,250)，半径 150
        // 右半球：圆心 (600,250)，半径 150
        // 甲点：(125,250) - 晨昏线与赤道交点
        // 乙点：(650,140) - 北极圈附近
        // C 点：(600,250) - 右半球赤道中央
        // 太阳直射点：(600,343) - 23.5°S 位置
        // 北京点：(680,180) - 39°N, 116°E
        
        const TOTAL_STEPS = 7;
        let currentStep = 0;
        
        // DOM 元素引用
        const btnPrev = document.getElementById('scene3-btn-prev');
        const btnNext = document.getElementById('scene3-btn-next');
        const stepIndicator = document.getElementById('scene3-step-indicator');
        const resultContent = document.getElementById('scene3-result-content');
        const resultBox = document.getElementById('scene3-result-box');
        const resultTitle = document.getElementById('scene3-result-title');
        
        // 元素显示/隐藏辅助函数
        function show(el) {
            if (el) {
                el.style.opacity = '1';
                el.style.display = 'block';
            }
        }
        
        function hide(el) {
            if (el) {
                el.style.opacity = '0';
                el.style.display = 'none';
            }
        }
        
        function showSVG(el) {
            if (el) {
                el.style.opacity = '1';
            }
        }
        
        function hideSVG(el) {
            if (el) {
                el.style.opacity = '0';
            }
        }
        
        // 高亮效果
        function highlight(el) {
            if (el) {
                el.classList.remove('scene3-highlight');
                void el.offsetWidth; // 触发重绘
                el.classList.add('scene3-highlight');
            }
        }
        
        // 更新步骤显示
        function renderStep() {
            stepIndicator.textContent = `步骤 ${currentStep + 1}/${TOTAL_STEPS}`;
            btnPrev.disabled = (currentStep === 0);
            btnNext.disabled = (currentStep === TOTAL_STEPS - 1);
            
            // 重置所有动态元素
            hideSVG(document.getElementById('scene3-terminator'));
            hideSVG(document.getElementById('scene3-shadow-left'));
            hideSVG(document.getElementById('scene3-day-arc'));
            hideSVG(document.getElementById('scene3-longitude-diff'));
            hideSVG(document.getElementById('scene3-orbit'));
            hideSVG(document.getElementById('scene3-sun'));
            hideSVG(document.getElementById('scene3-perihelion'));
            hideSVG(document.getElementById('scene3-aphelion'));
            hideSVG(document.getElementById('scene3-label-perihelion'));
            hideSVG(document.getElementById('scene3-label-aphelion'));
            hideSVG(document.getElementById('scene3-label-orbit'));
            hideSVG(document.getElementById('scene3-solar-zenith'));
            hideSVG(document.getElementById('scene3-label-solar-zenith'));
            hideSVG(document.getElementById('scene3-circle-beijing'));
            hideSVG(document.getElementById('scene3-label-beijing'));
            hide(resultBox);
            hide(resultTitle);
            hide(resultContent);
            
            // 移除所有高亮
            document.getElementById('scene3-circle-jia').classList.remove('scene3-highlight');
            document.getElementById('scene3-circle-yi').classList.remove('scene3-highlight');
            document.getElementById('scene3-circle-c').classList.remove('scene3-highlight');
            
            // 根据步骤显示对应元素
            switch(currentStep) {
                case 0: // 初始状态
                    showSVG(document.getElementById('scene3-label-20w'));
                    showSVG(document.getElementById('scene3-label-n'));
                    showSVG(document.getElementById('scene3-label-s'));
                    showSVG(document.getElementById('scene3-label-a'));
                    break;
                    
                case 1: // 时区计算
                    showSVG(document.getElementById('scene3-terminator'));
                    showSVG(document.getElementById('scene3-shadow-left'));
                    showSVG(document.getElementById('scene3-longitude-diff'));
                    highlight(document.getElementById('scene3-circle-jia'));
                    highlight(document.getElementById('scene3-circle-yi'));
                    show(resultBox);
                    show(resultTitle);
                    resultTitle.textContent = '计算结果';
                    resultContent.textContent = '甲地 22 日 16 时 → 乙地时间计算\n甲乙经度差 = 90° → 时间差 = 6 小时\n乙地在甲地东侧 → 乙地 = 22 日 22 时';
                    showSVG(resultContent);
                    break;
                    
                case 2: // C 点昼长计算
                    showSVG(document.getElementById('scene3-terminator'));
                    showSVG(document.getElementById('scene3-shadow-left'));
                    showSVG(document.getElementById('scene3-day-arc'));
                    highlight(document.getElementById('scene3-circle-c'));
                    show(resultBox);
                    show(resultTitle);
                    resultTitle.textContent = '计算结果';
                    resultContent.textContent = 'C 点昼长计算\n昼弧所跨经度 = 120°\n昼长 = 120° ÷ 15°/小时 = 8 小时';
                    showSVG(resultContent);
                    break;
                    
                case 3: // 太阳直射纬度
                    showSVG(document.getElementById('scene3-terminator'));
                    showSVG(document.getElementById('scene3-shadow-left'));
                    showSVG(document.getElementById('scene3-solar-zenith'));
                    showSVG(document.getElementById('scene3-label-solar-zenith'));
                    show(resultBox);
                    show(resultTitle);
                    resultTitle.textContent = '计算结果';
                    resultContent.textContent = '太阳直射纬度判断\n晨昏线与南极圈相切 → 太阳直射南回归线\n太阳直射纬度 = 23.5°S';
                    showSVG(resultContent);
                    break;
                    
                case 4: // 北京正午太阳高度
                    showSVG(document.getElementById('scene3-terminator'));
                    showSVG(document.getElementById('scene3-shadow-left'));
                    showSVG(document.getElementById('scene3-solar-zenith'));
                    showSVG(document.getElementById('scene3-label-solar-zenith'));
                    showSVG(document.getElementById('scene3-circle-beijing'));
                    showSVG(document.getElementById('scene3-label-beijing'));
                    show(resultBox);
                    show(resultTitle);
                    resultTitle.textContent = '计算结果';
                    resultContent.textContent = '北京正午太阳高度计算\nH = 90° - |φ - δ|\nH = 90° - |39° - (-23.5°)| = 90° - 62.5° = 27.5°';
                    showSVG(resultContent);
                    break;
                    
                case 5: // 地球公转位置
                    showSVG(document.getElementById('scene3-terminator'));
                    showSVG(document.getElementById('scene3-shadow-left'));
                    showSVG(document.getElementById('scene3-orbit'));
                    showSVG(document.getElementById('scene3-sun'));
                    showSVG(document.getElementById('scene3-perihelion'));
                    showSVG(document.getElementById('scene3-aphelion'));
                    showSVG(document.getElementById('scene3-label-perihelion'));
                    showSVG(document.getElementById('scene3-label-aphelion'));
                    showSVG(document.getElementById('scene3-label-orbit'));
                    highlight(document.getElementById('scene3-perihelion'));
                    show(resultBox);
                    show(resultTitle);
                    resultTitle.textContent = '计算结果';
                    resultContent.textContent = '地球公转位置\n日期为 12 月 22 日（冬至）\n接近 1 月初 → 地球位于近日点附近\n地球距太阳较近';
                    showSVG(resultContent);
                    break;
                    
                case 6: // 结论汇总
                    showSVG(document.getElementById('scene3-terminator'));
                    showSVG(document.getElementById('scene3-shadow-left'));
                    showSVG(document.getElementById('scene3-day-arc'));
                    showSVG(document.getElementById('scene3-solar-zenith'));
                    showSVG(document.getElementById('scene3-label-solar-zenith'));
                    showSVG(document.getElementById('scene3-circle-beijing'));
                    showSVG(document.getElementById('scene3-label-beijing'));
                    showSVG(document.getElementById('scene3-orbit'));
                    showSVG(document.getElementById('scene3-sun'));
                    showSVG(document.getElementById('scene3-perihelion'));
                    showSVG(document.getElementById('scene3-aphelion'));
                    showSVG(document.getElementById('scene3-label-perihelion'));
                    showSVG(document.getElementById('scene3-label-aphelion'));
                    showSVG(document.getElementById('scene3-label-orbit'));
                    highlight(document.getElementById('scene3-circle-jia'));
                    highlight(document.getElementById('scene3-circle-yi'));
                    highlight(document.getElementById('scene3-circle-c'));
                    highlight(document.getElementById('scene3-solar-zenith'));
                    highlight(document.getElementById('scene3-circle-beijing'));
                    show(resultBox);
                    show(resultTitle);
                    resultTitle.textContent = '计算结果汇总';
                    resultContent.textContent = '(4) 乙地是 12 月 22 日 22 时，C 点昼长 8 小时\n(5) 太阳直射纬度 23.5°S，北京正午太阳高度 27.5°\n(6) 地球距太阳较近（近日点附近）';
                    showSVG(resultContent);
                    break;
            }
        }
        
        // 事件绑定
        btnPrev.addEventListener('click', function() {
            if (currentStep > 0) {
                currentStep--;
                renderStep();
            }
        });
        
        btnNext.addEventListener('click', function() {
            if (currentStep < TOTAL_STEPS - 1) {
                currentStep++;
                renderStep();
            }
        });
        
        // 初始化
        renderStep();
    })();
    </script>
