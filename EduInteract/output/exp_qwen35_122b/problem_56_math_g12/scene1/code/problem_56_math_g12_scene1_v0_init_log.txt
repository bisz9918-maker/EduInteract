    (function() {
        // Scene 1: 圆与正六边形嵌套构造
        // 布局坐标表（参考）：

        // | 正六边形 H1 | 内接于 C1 | 黑色实线 |
        // | 正六边形 H2 | 内接于 C2 | 黑色实线 |
        // | 标签 R₁=1m | (570, 145) | 移到半径线上方 |

        // 状态变量
        let currentStep = 0;
        const maxStep = 5;
        
        // 元素定义：每个步骤显示/隐藏的元素
        const stepElements = {

                       'scene1-radius-2', 'scene1-radius-3', 'scene1-label-R2', 'scene1-label-R3',
                       'scene1-vertex-A1', 'scene1-vertex-A2', 'scene1-vertex-A3', 'scene1-vertex-A4',
                       'scene1-vertex-A5', 'scene1-vertex-A6', 'scene1-vertex-B1', 'scene1-vertex-B2',
                       'scene1-vertex-B3', 'scene1-vertex-B4', 'scene1-vertex-B5', 'scene1-vertex-B6',
                       'scene1-infinite-hint']
            },
            // 步骤 1：新增 H1 和顶点标签 A₁~A₆
            1: {
                show: ['scene1-center', 'scene1-circle-1', 'scene1-radius-1', 'scene1-label-O', 'scene1-label-R1',
                       'scene1-hex-1', 'scene1-vertex-A1', 'scene1-vertex-A2', 'scene1-vertex-A3',
                       'scene1-vertex-A4', 'scene1-vertex-A5', 'scene1-vertex-A6'],
                hide: ['scene1-circle-2', 'scene1-circle-3', 'scene1-hex-2',
                       'scene1-radius-2', 'scene1-radius-3', 'scene1-label-R2', 'scene1-label-R3',
                       'scene1-vertex-B1', 'scene1-vertex-B2', 'scene1-vertex-B3', 'scene1-vertex-B4',
                       'scene1-vertex-B5', 'scene1-vertex-B6', 'scene1-infinite-hint']
            },
            // 步骤 2：新增 C2、R2、标签 R₂
            2: {
                show: ['scene1-center', 'scene1-circle-1', 'scene1-radius-1', 'scene1-label-O', 'scene1-label-R1',
                       'scene1-hex-1', 'scene1-vertex-A1', 'scene1-vertex-A2', 'scene1-vertex-A3',
                       'scene1-vertex-A4', 'scene1-vertex-A5', 'scene1-vertex-A6',
                       'scene1-circle-2', 'scene1-radius-2', 'scene1-label-R2'],
                hide: ['scene1-circle-3', 'scene1-hex-2',
                       'scene1-radius-3', 'scene1-label-R3',
                       'scene1-vertex-B1', 'scene1-vertex-B2', 'scene1-vertex-B3', 'scene1-vertex-B4',
                       'scene1-vertex-B5', 'scene1-vertex-B6', 'scene1-infinite-hint']
            },
            // 步骤 3：新增 H2 和顶点标签 B₁~B₆

            4: {
                show: ['scene1-center', 'scene1-circle-1', 'scene1-radius-1', 'scene1-label-O', 'scene1-label-R1',
                       'scene1-hex-1', 'scene1-vertex-A1', 'scene1-vertex-A2', 'scene1-vertex-A3',

                       'scene1-hex-2', 'scene1-vertex-B1', 'scene1-vertex-B2', 'scene1-vertex-B3',

            },
            // 步骤 5：结论状态 - 高亮所有圆，显示无限提示
            5: {
                show: ['scene1-center', 'scene1-circle-1', 'scene1-radius-1', 'scene1-label-O', 'scene1-label-R1',
                       'scene1-hex-1', 'scene1-vertex-A1', 'scene1-vertex-A2', 'scene1-vertex-A3',
                       'scene1-vertex-A4', 'scene1-vertex-A5', 'scene1-vertex-A6',
                       'scene1-circle-2', 'scene1-radius-2', 'scene1-label-R2',
                       'scene1-hex-2', 'scene1-vertex-B1', 'scene1-vertex-B2', 'scene1-vertex-B3',
                       'scene1-vertex-B4', 'scene1-vertex-B5', 'scene1-vertex-B6',
                       'scene1-circle-3', 'scene1-radius-3', 'scene1-label-R3',
                       'scene1-infinite-hint'],
                hide: []
            }
        };
        
        // 高亮元素：每个步骤需要高亮的元素
        const highlightElements = {
            0: [],
            1: ['scene1-hex-1'],
            2: ['scene1-circle-2'],
            3: ['scene1-hex-2'],
            4: ['scene1-circle-3'],
            5: ['scene1-circle-1', 'scene1-circle-2', 'scene1-circle-3']
        };
        
        // 工具函数：显示元素
        function showElement(id) {
            const el = document.getElementById(id);
            if (el) {
                el.style.display = 'block';
                el.style.opacity = '1';
            }
        }
        
        // 工具函数：隐藏元素
        function hideElement(id) {
            const el = document.getElementById(id);
            if (el) {
                el.style.display = 'none';
                el.style.opacity = '0';
            }
        }
        
        // 工具函数：添加高亮
        function addHighlight(id) {
            const el = document.getElementById(id);
            if (el) {
                el.classList.add('highlight');
            }
        }
        
        // 工具函数：移除高亮
        function removeHighlight(id) {
            const el = document.getElementById(id);
            if (el) {
                el.classList.remove('highlight');
            }
        }
        
        // 更新场景显示
        function updateScene() {
            const elements = stepElements[currentStep];
            
            // 先移除所有高亮
            for (let i = 0; i <= maxStep; i++) {
                highlightElements[i].forEach(id => removeHighlight(id));
            }
            
            // 显示当前步骤的元素
            elements.show.forEach(id => showElement(id));
            
            // 隐藏其他元素
            elements.hide.forEach(id => hideElement(id));
            
            // 添加当前步骤的高亮
            highlightElements[currentStep].forEach(id => addHighlight(id));
            
            // 更新步骤指示器
            document.getElementById('scene1-step-indicator').textContent = `步骤 ${currentStep}/${maxStep}`;
            
            // 更新按钮状态
            document.getElementById('scene1-prev-btn').disabled = (currentStep === 0);
            document.getElementById('scene1-next-btn').disabled = (currentStep === maxStep);
        }
        
        // 绑定按钮事件
        document.getElementById('scene1-prev-btn').addEventListener('click', function() {
            if (currentStep > 0) {
                currentStep--;
                updateScene();
            }
        });
        
        document.getElementById('scene1-next-btn').addEventListener('click', function() {
            if (currentStep < maxStep) {
                currentStep++;
                updateScene();
            }
        });
        
        // 初始化场景
        updateScene();
    })();
