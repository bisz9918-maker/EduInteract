    <script>
    (function() {
        'use strict';
        
        // 布局坐标表（实现对照依据）
        // 标签位置已调整以避免重叠（圆半径 120，圆心 400,220，圆边界 x:280-520, y:100-340）：
        // O(400,360) - 圆心正下方圆外
        // A(460,80) - A 点正上方圆外
        // B(540,170) - B 点右侧圆外（y=170）
        // M(540,210) - M 点右侧圆外（y=210，与 B 错开 40px）
        // R₁(250,200) - 圆外左侧
        // R₂(420,360) - 圆外下方
        // ∠AOM=30°(560,260) - 右侧圆外（与 M 错开 50px，满足 12px 最小间距）
        
        // 状态变量
        let currentStep = 0;
        const maxStep = 4;
        
        // 获取 DOM 元素
        const btnPrev = document.getElementById('scene2-btn-prev');
        const btnNext = document.getElementById('scene2-btn-next');
        
        // 元素 ID 列表
        const elements = {
            // 步骤 1 显示的元素
            step1: [
                'scene2-line-oa',
                'scene2-line-om',
                'scene2-circle-m',
                'scene2-rect-symbol',
                'scene2-label-m',
                'scene2-label-r1',
                'scene2-label-r2'
            ],
            // 步骤 2 显示的元素
            step2: [
                'scene2-arc-angle',
                'scene2-label-angle'
            ],
            // 步骤 3 显示的元素
            step3: [
                'scene2-text-formula'
            ],
            // 步骤 4 显示的元素
            step4: [
                'scene2-text-conclusion'
            ]
        };
        
        // 工具函数：显示元素
        function show(id) {
            const el = document.getElementById(id);
            if (el) {
                el.style.display = '';
                el.style.opacity = '0';
                requestAnimationFrame(() => {
                    el.style.opacity = '1';
                });
            }
        }
        
        // 工具函数：隐藏元素
        function hide(id) {
            const el = document.getElementById(id);
            if (el) {
                el.style.display = 'none';
                el.style.opacity = '1';
            }
        }
        
        // 工具函数：添加高亮动画
        function highlight(id) {
            const el = document.getElementById(id);
            if (el) {
                el.classList.remove('highlight');
                void el.offsetWidth; // 触发重绘
                el.classList.add('highlight');
            }
        }
        
        // 更新场景状态
        function updateScene() {
            // 重置所有步骤特定元素
            elements.step1.forEach(id => hide(id));
            elements.step2.forEach(id => hide(id));
            elements.step3.forEach(id => hide(id));
            elements.step4.forEach(id => hide(id));
            
            // 根据当前步骤显示元素
            if (currentStep >= 1) {
                elements.step1.forEach(id => show(id));
            }
            if (currentStep >= 2) {
                elements.step2.forEach(id => show(id));
            }
            if (currentStep >= 3) {
                elements.step3.forEach(id => show(id));
                // 高亮 OA 和 OM
                highlight('scene2-line-oa');
                highlight('scene2-line-om');
            }
            if (currentStep >= 4) {
                elements.step4.forEach(id => show(id));
                // 高亮整个三角形
                highlight('scene2-line-oa');
                highlight('scene2-line-om');
                highlight('scene2-line-ab');
            }
            
            // 更新按钮状态
            btnPrev.disabled = (currentStep === 0);
            
            if (currentStep === maxStep) {
                btnNext.textContent = '完成';
                btnNext.disabled = true;
            } else {
                btnNext.textContent = '下一步 ▶';
                btnNext.disabled = false;
            }
        }
        
        // 事件监听
        btnPrev.addEventListener('click', () => {
            if (currentStep > 0) {
                currentStep--;
                updateScene();
            }
        });
        
        btnNext.addEventListener('click', () => {
            if (currentStep < maxStep) {
                currentStep++;
                updateScene();
            }
        });
        
        // 初始化
        updateScene();
    })();
    </script>
