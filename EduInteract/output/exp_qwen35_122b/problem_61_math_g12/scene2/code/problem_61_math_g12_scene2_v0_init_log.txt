    <script>
    (function() {
        // Scene 2 实现 - 球心角与球面距离计算
        // 布局坐标表（修复后）：
        // | 元素 | 位置/范围 | 备注 |
        // | 球体轮廓 | (100,100)-(500,500) | 400×400 |
        // | 截面圆 | (180,280)-(420,340) | 240×60 虚线 |
        // | 球心 O | (300,250) | 球心位置 |
        // | 点 B | (220,320) | 球面右下偏左 |
        // | 点 C | (380,320) | 球面右下偏右 |
        // | 线段 OB | (300,250)-(220,320) | 半径，蓝色 |
        // | 线段 OC | (300,250)-(380,320) | 半径，蓝色 |
        // | 线段 BC | (220,320)-(380,320) | 弦，黑色 160px |
        // | 标签 O | (305,40) | 移到球体上方外侧 |
        // | 标签 B | (80,340) | 移到球体左方外侧 |
        // | 标签 C | (520,340) | 移到球体右方外侧 |
        // | 数值标注 | (550,180+) | 移到右侧空白区域 |
        
        const scene2 = {
            currentStep: 0,
            totalSteps: 5,
            
            // 元素 ID 映射
            elements: {
                sphere: 'scene2-sphere',
                section: 'scene2-section',
                triangleFill: 'scene2-triangle-fill',
                lineOB: 'scene2-line-OB',
                lineOC: 'scene2-line-OC',
                lineBC: 'scene2-line-BC',
                angleArc: 'scene2-angle-arc',
                surfaceArc: 'scene2-surface-arc',
                markOB: 'scene2-mark-OB',
                markOC: 'scene2-mark-OC',
                markBC: 'scene2-mark-BC',
                pointO: 'scene2-point-O',
                pointB: 'scene2-point-B',
                pointC: 'scene2-point-C',
                labelO: 'scene2-label-O',
                labelB: 'scene2-label-B',
                labelC: 'scene2-label-C',
                labelRadius: 'scene2-label-radius',
                labelChord: 'scene2-label-chord',
                labelEquilateral: 'scene2-label-equilateral',
                labelAngle: 'scene2-label-angle',
                labelFormula: 'scene2-label-formula'
            },
            
            // 步骤定义
            steps: [
                // 步骤 0：初始状态 - 显示基本结构
                {
                    show: ['sphere', 'section', 'triangleFill', 'lineOB', 'lineOC', 'lineBC', 'pointO', 'pointB', 'pointC', 'labelO', 'labelB', 'labelC'],
                    hide: ['angleArc', 'surfaceArc', 'markOB', 'markOC', 'markBC', 'labelRadius', 'labelChord', 'labelEquilateral', 'labelAngle', 'labelFormula'],
                    lineOBWidth: 2,
                    lineOCWidth: 2,
                    lineBCWidth: 1.5,
                    triangleOpacity: 0.1
                },
                // 步骤 1：标注半径和弦长
                {
                    show: ['sphere', 'section', 'triangleFill', 'lineOB', 'lineOC', 'lineBC', 'pointO', 'pointB', 'pointC', 'labelO', 'labelB', 'labelC', 'labelRadius', 'labelChord'],
                    hide: ['angleArc', 'surfaceArc', 'markOB', 'markOC', 'markBC', 'labelEquilateral', 'labelAngle', 'labelFormula'],
                    lineOBWidth: 3,
                    lineOCWidth: 3,
                    lineBCWidth: 2,
                    triangleOpacity: 0.1
                },
                // 步骤 2：等边三角形判定
                {
                    show: ['sphere', 'section', 'triangleFill', 'lineOB', 'lineOC', 'lineBC', 'pointO', 'pointB', 'pointC', 'labelO', 'labelB', 'labelC', 'labelRadius', 'labelChord', 'markOB', 'markOC', 'markBC', 'labelEquilateral'],
                    hide: ['angleArc', 'surfaceArc', 'labelAngle', 'labelFormula'],
                    lineOBWidth: 3,
                    lineOCWidth: 3,
                    lineBCWidth: 2,
                    triangleOpacity: 0.2
                },
                // 步骤 3：球心角计算
                {
                    show: ['sphere', 'section', 'triangleFill', 'lineOB', 'lineOC', 'lineBC', 'pointO', 'pointB', 'pointC', 'labelO', 'labelB', 'labelC', 'labelRadius', 'labelChord', 'markOB', 'markOC', 'markBC', 'labelEquilateral', 'angleArc', 'labelAngle'],
                    hide: ['surfaceArc', 'labelFormula'],
                    lineOBWidth: 3,
                    lineOCWidth: 3,
                    lineBCWidth: 2,
                    triangleOpacity: 0.2
                },
                // 步骤 4：球面距离公式（最终）
                {
                    show: ['sphere', 'section', 'triangleFill', 'lineOB', 'lineOC', 'lineBC', 'pointO', 'pointB', 'pointC', 'labelO', 'labelB', 'labelC', 'labelRadius', 'labelChord', 'markOB', 'markOC', 'markBC', 'labelEquilateral', 'angleArc', 'surfaceArc', 'labelAngle', 'labelFormula'],
                    hide: [],
                    lineOBWidth: 3,
                    lineOCWidth: 3,
                    lineBCWidth: 2,
                    triangleOpacity: 0.2
                }
            ],
            
            // 获取元素引用
            getEl: function(key) {
                return document.getElementById(this.elements[key]);
            },
            
            // 显示元素
            showEl: function(key) {
                const el = this.getEl(key);
                if (el) {
                    el.style.display = 'block';
                    el.style.opacity = '1';
                }
            },
            
            // 隐藏元素
            hideEl: function(key) {
                const el = this.getEl(key);
                if (el) {
                    el.style.display = 'none';
                    el.style.opacity = '0';
                }
            },
            
            // 更新步骤显示
            updateScene: function() {
                const step = this.steps[this.currentStep];
                
                // 显示元素
                step.show.forEach(key => this.showEl(key));
                
                // 隐藏元素
                step.hide.forEach(key => this.hideEl(key));
                
                // 更新线段粗细
                const lineOB = this.getEl('lineOB');
                const lineOC = this.getEl('lineOC');
                const lineBC = this.getEl('lineBC');
                const triangleFill = this.getEl('triangleFill');
                
                if (lineOB) lineOB.setAttribute('stroke-width', step.lineOBWidth);
                if (lineOC) lineOC.setAttribute('stroke-width', step.lineOCWidth);
                if (lineBC) lineBC.setAttribute('stroke-width', step.lineBCWidth);
                if (triangleFill) triangleFill.setAttribute('opacity', step.triangleOpacity);
                
                // 更新步骤指示器
                const indicator = document.getElementById('scene2-step-indicator');
                if (indicator) {
                    indicator.textContent = '步骤 ' + (this.currentStep + 1) + '/' + this.totalSteps;
                }
                
                // 更新按钮状态
                const btnPrev = document.getElementById('scene2-btn-prev');
                const btnNext = document.getElementById('scene2-btn-next');
                
                if (btnPrev) btnPrev.disabled = (this.currentStep === 0);
                if (btnNext) btnNext.disabled = (this.currentStep === this.totalSteps - 1);
            },
            
            // 下一步
            nextStep: function() {
                if (this.currentStep < this.totalSteps - 1) {
                    this.currentStep++;
                    this.updateScene();
                }
            },
            
            // 上一步
            prevStep: function() {
                if (this.currentStep > 0) {
                    this.currentStep--;
                    this.updateScene();
                }
            },
            
            // 初始化
            init: function() {
                // 绑定按钮事件
                document.getElementById('scene2-btn-prev').addEventListener('click', () => this.prevStep());
                document.getElementById('scene2-btn-next').addEventListener('click', () => this.nextStep());
                
                // 初始化显示
                this.updateScene();
            }
        };
        
        // 启动
        scene2.init();
    })();
    </script>
