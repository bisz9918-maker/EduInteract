    <script>
    (function() {
        // Scene 1 实现 - 四面体平面旋转演示
        // 布局坐标表：
        // A(400,80), B(150,350), C(400,420), D(650,350)
        // E(275,215) = AB 中点, F(525,385) = CD 中点
        // G = B + t*(C-B) = (150+250t, 350+70t)
        // H = A + t*(D-A) = (400+250t, 80+270t)
        
        const svg = document.getElementById('scene1-svg');
        const slider = document.getElementById('scene1-slider');
        const valueDisplay = document.getElementById('scene1-value');
        
        // 固定点坐标
        const A = { x: 400, y: 80 };
        const B = { x: 150, y: 350 };
        const C = { x: 400, y: 420 };
        const D = { x: 650, y: 350 };
        const E = { x: 275, y: 215 };
        const F = { x: 525, y: 385 };
        
        // 获取动态元素引用
        const polygonAlpha = document.getElementById('scene1-polygon-alpha');
        const lineEG = document.getElementById('scene1-line-EG');
        const lineGF = document.getElementById('scene1-line-GF');
        const lineFH = document.getElementById('scene1-line-FH');
        const lineHE = document.getElementById('scene1-line-HE');
        const pointG = document.getElementById('scene1-point-G');
        const pointH = document.getElementById('scene1-point-H');
        const labelG = document.getElementById('scene1-label-G');
        const labelH = document.getElementById('scene1-label-H');
        
        // 计算 G、H 坐标（参数化）
        function calculatePoints(t) {
            // G = B + t*(C-B)
            const G = {
                x: B.x + t * (C.x - B.x),
                y: B.y + t * (C.y - B.y)
            };
            // H = A + t*(D-A)
            const H = {
                x: A.x + t * (D.x - A.x),
                y: A.y + t * (D.y - A.y)
            };
            return { G, H };
        }
        
        // 更新画面
        function updateScene(t) {
            const { G, H } = calculatePoints(t);
            
            // 更新 G、H 点位置
            pointG.setAttribute('cx', G.x);
            pointG.setAttribute('cy', G.y);
            pointH.setAttribute('cx', H.x);
            pointH.setAttribute('cy', H.y);
            
            // 更新平面α四边形顶点
            polygonAlpha.setAttribute('points', `${E.x},${E.y} ${G.x},${G.y} ${F.x},${F.y} ${H.x},${H.y}`);
            
            // 更新平面边界线
            lineEG.setAttribute('x1', E.x);
            lineEG.setAttribute('y1', E.y);
            lineEG.setAttribute('x2', G.x);
            lineEG.setAttribute('y2', G.y);
            
            lineGF.setAttribute('x1', G.x);
            lineGF.setAttribute('y1', G.y);
            lineGF.setAttribute('x2', F.x);
            lineGF.setAttribute('y2', F.y);
            
            lineFH.setAttribute('x1', F.x);
            lineFH.setAttribute('y1', F.y);
            lineFH.setAttribute('x2', H.x);
            lineFH.setAttribute('y2', H.y);
            
            lineHE.setAttribute('x1', H.x);
            lineHE.setAttribute('y1', H.y);
            lineHE.setAttribute('x2', E.x);
            lineHE.setAttribute('y2', E.y);
            
            // 更新 G、H 标签位置（远离线条，避免重叠）
            // G 点标签：放在 G 点下方偏左，远离 BC 线和平面边界
            labelG.setAttribute('x', G.x - 15);
            labelG.setAttribute('y', G.y + 25);
            // H 点标签：放在 H 点右上方，远离 AD 线和平面边界
            labelH.setAttribute('x', H.x + 20);
            labelH.setAttribute('y', H.y - 10);
            
            // 更新数值显示
            valueDisplay.textContent = Math.round(t * 100);
        }
        
        // 滑块事件处理
        slider.addEventListener('input', function() {
            const t = this.value / 100;
            updateScene(t);
        });
        
        // 初始化（t=0）
        updateScene(0);
    })();
    </script>
