    <script>
    (function() {
        // Scene 1 实现
        // 布局坐标表（实现对照依据）：
        // | 元素 | 类型 | 位置/范围 | 大小 | 标签偏移方向 | 备注 |
        // |------|------|-----------|------|-------------|------|
        // | 水平参考线 | 虚线 | (100,380)-(700,380) | 600×1 | - | 底部水平参考 |
        // | U 形框架左轨 | 线段 | (150,350)-(650,150) | - | - | 倾斜 30° |
        // | U 形框架右轨 | 线段 | (150,370)-(650,170) | - | - | 与左轨平行，间距 20px 表示宽度 |
        // | 导体棒 MN | 矩形 | (320,265)-(480,235) | 160×30 | - | 横跨导轨，浅灰填充 |
        // | 磁场 B 箭头 | 箭头 | (400,250)-(400,120) | - | 左上方 | 垂直框架平面向上 |
        // | 速度 v 箭头 | 箭头 | (400,250)-(480,280) | - | 右下方 | 沿斜面向下 |
        // | 安培力 F_安箭头 | 箭头 | (400,250)-(320,220) | - | 左上方 | 沿斜面向上 |
        // | 重力分力箭头 | 箭头 | (400,250)-(480,280) | - | 右下方 | 沿斜面向下，与 v 同向 |
        // | 角度θ弧线 | 圆弧 | (180,360) 半径 40 | - | 下方 | 框架与水平线夹角 |
        // | 标签 L=1m | text | (660,250) | ~50×16 | 右侧 | 框架宽度标注 |
        // | 标签 θ=30° | text | (200,390) | ~50×16 | 下方 | 角度标注 |
        // | 标签 B=0.2 | text | (380,100) | ~70×16 | 上方 | 磁场强度 |
        // | 标签 M | text | (215,230) | ~16×16 | 左侧 | 导体棒左端（调整：远离导轨） |
        // | 标签 N | text | (560,250) | ~16×16 | 右侧 | 导体棒右端（调整：远离导轨） |
        // | 标签 v | text | (570,245) | ~16×16 | 右侧 | 速度（调整：远离箭头和 N） |
        // | 标签 I | text | (400,205) | ~16×16 | 上方 | 电流方向（调整：远离箭头） |
        // | 标签 F_安 | text | (200,175) | ~50×16 | 左上方 | 安培力（调整：远离箭头） |
        // | 标签 mg sinθ | text | (570,280) | ~70×16 | 右下方 | 重力分力（调整：远离箭头） |
        // | 标签 m=0.2kg | text | (660,280) | ~70×16 | 右侧 | 质量 |
        // | 标签 R=0.2Ω | text | (660,310) | ~70×16 | 右侧 | 电阻 |
        // | 电流方向箭头 | 箭头 | (360,250)-(440,250) | - | 向右 | 导体棒上 N→M |
        
        const TOTAL_STEPS = 6;
        let currentStep = 0;
        
        // 工具函数：显示元素
        function show(el) {
            if (el) {
                el.style.display = 'block';
                el.style.opacity = '1';
            }
        }
        
        // 工具函数：隐藏元素
        function hide(el) {
            if (el) {
                el.style.display = 'none';
                el.style.opacity = '0';
            }
        }
        
        // 更新当前步骤显示
        function updateStep() {
            // 隐藏所有步骤元素
            for (let i = 1; i <= TOTAL_STEPS; i++) {
                const stepGroup = document.getElementById('scene1-step' + i);
                if (stepGroup) {
                    hide(stepGroup);
                }
            }
            
            // 移除高亮
            const highlightElements = document.querySelectorAll('.highlight');
            highlightElements.forEach(el => el.classList.remove('highlight'));
            
            // 显示当前步骤及之前的所有元素
            for (let i = 1; i <= currentStep; i++) {
                const stepGroup = document.getElementById('scene1-step' + i);
                if (stepGroup) {
                    show(stepGroup);
                }
            }
            
            // 步骤 6 特殊处理：添加高亮
            if (currentStep === TOTAL_STEPS) {
                const arrowFAmp = document.getElementById('scene1-arrow-F-amp');
                const arrowMgSin = document.getElementById('scene1-arrow-mg-sin');
                if (arrowFAmp) arrowFAmp.classList.add('highlight');
                if (arrowMgSin) arrowMgSin.classList.add('highlight');
            }
            
            // 更新步骤指示器
            document.getElementById('scene1-step-indicator').textContent = '步骤 ' + currentStep + '/' + TOTAL_STEPS;
            
            // 更新按钮状态
            document.getElementById('scene1-btn-prev').disabled = (currentStep === 0);
            document.getElementById('scene1-btn-next').disabled = (currentStep === TOTAL_STEPS);
        }
        
        // 初始化
        function init() {
            // 绑定按钮事件
            document.getElementById('scene1-btn-prev').addEventListener('click', function() {
                if (currentStep > 0) {
                    currentStep--;
                    updateStep();
                }
            });
            
            document.getElementById('scene1-btn-next').addEventListener('click', function() {
                if (currentStep < TOTAL_STEPS) {
                    currentStep++;
                    updateStep();
                }
            });
            
            // 初始显示步骤 0
            updateStep();
        }
        
        // 页面加载完成后初始化
        if (document.readyState === 'loading') {
            document.addEventListener('DOMContentLoaded', init);
        } else {
            init();
        }
    })();
    </script>
