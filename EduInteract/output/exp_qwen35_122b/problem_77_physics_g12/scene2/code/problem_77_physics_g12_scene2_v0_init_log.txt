    <script>
    (function() {
        // ===== Scene 2 实现 =====
        // 坐标映射：时间 t∈[0,2T] 映射到 x∈[90,690]
        // 比例因子：px_per_T = 300 (因为 T 对应 300px)
        // 电压跳变点：2T/3≈290, T=390, 5T/3≈590
        
        const px_per_T = 300;
        const x_origin = 90;
        const voltage_y_positive = 80;
        const voltage_y_negative = 120;
        const voltage_y_zero = 170;
        const accel_y_positive = 320;
        const accel_y_negative = 360;
        const accel_y_zero = 390;
        
        // 状态变量
        let t0 = 0;  // 入射时刻，范围 [0, T)
        let isDragging = false;
        
        // DOM 元素
        const sliderTrack = document.getElementById('scene2-slider-track');
        const sliderHandle = document.getElementById('scene2-slider-handle');
        const t0Display = document.getElementById('scene2-t0-display');
        const highlightRect = document.getElementById('scene2-highlight-rect');
        const t0Line = document.getElementById('scene2-t0-line');
        const t0Label = document.getElementById('scene2-t0-label');
        const accelGraph = document.getElementById('scene2-accel-graph');
        
        // 时间转像素坐标
        function timeToX(t) {
            return x_origin + t * px_per_T;
        }
        
        // 像素坐标转时间
        function xToTime(x) {
            return (x - x_origin) / px_per_T;
        }
        
        // 更新 UI
        function updateUI() {
            const x_t0 = timeToX(t0);
            const x_t0_plus_T = timeToX(t0 + 1);
            
            // 更新高亮区间矩形
            highlightRect.setAttribute('x', x_t0);
            highlightRect.setAttribute('y', 70);
            highlightRect.setAttribute('width', x_t0_plus_T - x_t0);
            highlightRect.setAttribute('height', 100);
            
            // 更新入射时刻竖线
            t0Line.setAttribute('x1', x_t0);
            t0Line.setAttribute('y1', 70);
            t0Line.setAttribute('x2', x_t0);
            t0Line.setAttribute('y2', 170);
            
            // 更新 t₀标签位置（避免与 U 标签重叠）
            if (x_t0 < 110) {
                t0Label.setAttribute('x', x_t0 - 25);
                t0Label.setAttribute('y', 75);
            } else {
                t0Label.setAttribute('x', x_t0 + 5);
                t0Label.setAttribute('y', 75);
            }
            
            // 更新滑块手柄位置
            const sliderLeft = (t0 / 1) * 300;
            sliderHandle.style.left = sliderLeft + 'px';
            
            // 更新 t₀显示
            if (t0 < 0.01) {
                t0Display.textContent = 't₀ = 0';
            } else if (t0 >= 0.99) {
                t0Display.textContent = 't₀ = T';
            } else {
                t0Display.textContent = 't₀ = ' + t0.toFixed(2) + 'T';
            }
            
            // 更新加速度图
            updateAccelerationGraph();
        }
        
        // 更新加速度图
        function updateAccelerationGraph() {
            const points = [];
            const t_start = t0;
            const t_end = t0 + 1;
            
            // 计算加速度分段点
            // 电压波形周期：0~2T/3: +U₀, 2T/3~T: -U₀, T~5T/3: +U₀, 5T/3~2T: -U₀
            // 加速度与电压成正比，方向相反（正电荷）
            
            let current_t = t_start;
            let current_a = getAcceleration(t_start);
            
            points.push({ t: t_start, a: current_a });
            
            // 找到所有跳变点
            const jumpPoints = [];
            for (let k = 0; k < 4; k++) {
                const t_jump1 = k * 1 + 2/3;  // 2T/3, T+2T/3, ...
                const t_jump2 = k * 1 + 1;    // T, 2T, ...
                if (t_jump1 > t_start && t_jump1 < t_end) jumpPoints.push(t_jump1);
                if (t_jump2 > t_start && t_jump2 < t_end) jumpPoints.push(t_jump2);
            }
            jumpPoints.sort((a, b) => a - b);
            jumpPoints.push(t_end);
            
            // 构建加速度折线
            for (const t_jump of jumpPoints) {
                const a_before = getAcceleration(t_jump - 0.001);
                const a_after = getAcceleration(t_jump + 0.001);
                
                const x_before = timeToX(t_jump);
                const x_after = timeToX(t_jump);
                
                // 添加跳变前的点
                points.push({ t: t_jump, a: a_before, x: x_before });
                // 添加跳变后的点
                points.push({ t: t_jump, a: a_after, x: x_after });
                
                current_t = t_jump;
                current_a = a_after;
            }
            
            // 添加终点
            points.push({ t: t_end, a: getAcceleration(t_end - 0.001) });
            
            // 构建 SVG polyline
            let svgPoints = '';
            for (let i = 0; i < points.length; i++) {
                const p = points[i];
                const x = timeToX(p.t);
                const y = p.a > 0 ? accel_y_positive : accel_y_negative;
                
                if (i === 0) {
                    svgPoints += x + ',' + y;
                } else {
                    // 如果是跳变点，先画垂直线
                    const prevP = points[i - 1];
                    const prevY = prevP.a > 0 ? accel_y_positive : accel_y_negative;
                    if (Math.abs(p.t - prevP.t) < 0.01 && prevY !== y) {
                        svgPoints += ',' + x + ',' + prevY + ' ' + x + ',' + y;
                    } else {
                        svgPoints += ',' + x + ',' + y;
                    }
                }
            }
            
            accelGraph.setAttribute('points', svgPoints);
            
            // 设置颜色：向下加速度（正值）为红色，向上加速度（负值）为蓝色
            accelGraph.setAttribute('stroke', '#888880');
        }
        
        // 获取时刻 t 的加速度方向（简化模型）
        function getAcceleration(t) {
            // 归一化到 [0, 2) 周期
            const t_norm = t % 2;
            
            if (t_norm < 2/3) {
                return 1;  // 向下加速度（正值）
            } else if (t_norm < 1) {
                return -1; // 向上加速度（负值）
            } else if (t_norm < 5/3) {
                return 1;  // 向下加速度（正值）
            } else {
                return -1; // 向上加速度（负值）
            }
        }
        
        // 滑块交互
        function handleSliderMove(clientX) {
            const rect = sliderTrack.getBoundingClientRect();
            let x = clientX - rect.left;
            x = Math.max(0, Math.min(300, x));
            t0 = x / 300;  // 映射到 [0, 1]
            t0 = Math.max(0, Math.min(0.999, t0));
            updateUI();
        }
        
        sliderHandle.addEventListener('mousedown', function(e) {
            isDragging = true;
            e.preventDefault();
        });
        
        sliderTrack.addEventListener('mousedown', function(e) {
            handleSliderMove(e.clientX);
            isDragging = true;
            e.preventDefault();
        });
        
        document.addEventListener('mousemove', function(e) {
            if (isDragging) {
                handleSliderMove(e.clientX);
            }
        });
        
        document.addEventListener('mouseup', function() {
            isDragging = false;
        });
        
        // 触摸支持
        sliderHandle.addEventListener('touchstart', function(e) {
            isDragging = true;
            e.preventDefault();
        });
        
        sliderTrack.addEventListener('touchstart', function(e) {
            handleSliderMove(e.touches[0].clientX);
            isDragging = true;
            e.preventDefault();
        });
        
        document.addEventListener('touchmove', function(e) {
            if (isDragging) {
                handleSliderMove(e.touches[0].clientX);
            }
        });
        
        document.addEventListener('touchend', function() {
            isDragging = false;
        });
        
        // 初始化
        updateUI();
    })();
    </script>
