    <script>
    (function() {
        // Scene 3 实现 - 运动过程动态演示（F=10N 情况）
        
        // 物理参数
        const SCALE = 250; // 1m = 250px
        const L = 1.6;     // 木板长度 (m)
        const t0 = 0.8;    // 临界作用时间 (s)
        const a_m = 1;     // 物块加速度 (m/s²)
        const a_M = 3;     // 木板 F 作用时加速度 (m/s²)
        const a_M_prime = -1/3; // 木板 F 撤去后加速度 (m/s²)
        
        // 初始位置（px）
        const 木板初始 X = 100;
        const 物块初始 X = 460;
        
        // DOM 元素
        const svg = document.getElementById('scene3-svg');
        const 滑块 = document.getElementById('scene3-slider');
        const 滑块手柄 = document.getElementById('scene3-slider-handle');
        const 滑块值显示 = document.getElementById('scene3-slider-value');
        const 状态文本 = document.getElementById('scene3-status-text');
        const t 值显示 = document.getElementById('scene3-t-value');
        const xM 值显示 = document.getElementById('scene3-xM-value');
        const xm 值显示 = document.getElementById('scene3-xm-value');
        const delta 值显示 = document.getElementById('scene3-delta-value');
        const 木板当前 = document.getElementById('scene3-木板当前');
        const 物块当前 = document.getElementById('scene3-物块当前');
        const F 箭头线 = document.getElementById('scene3-F-arrow-line');
        const F 箭头头 = document.getElementById('scene3-F-arrow-head');
        const F 标签 = document.getElementById('scene3-F-label');
        const xM 线 = document.getElementById('scene3-xM-line');
        const xm 线 = document.getElementById('scene3-xm-line');
        const delta 线 = document.getElementById('scene3-delta-line');
        const 临界文本 = document.getElementById('scene3-critical-text');
        
        // 状态变量
        let currentTime = 0; // 秒
        let isDragging = false;
        
        // 计算位移（米）
        function calculateDisplacement(t) {
            let x_M, x_m;
            
            if (t <= t0) {
                // F 作用阶段
                x_M = 0.5 * a_M * t * t;
                x_m = 0.5 * a_m * t * t;
            } else {
                // F 撤去后阶段
                const t1 = t - t0;
                const x_M_t0 = 0.5 * a_M * t0 * t0;
                const v_M_t0 = a_M * t0;
                const x_m_t0 = 0.5 * a_m * t0 * t0;
                const v_m_t0 = a_m * t0;
                
                x_M = x_M_t0 + v_M_t0 * t1 + 0.5 * a_M_prime * t1 * t1;
                x_m = x_m_t0 + v_m_t0 * t1 + 0.5 * a_m * t1 * t1;
            }
            
            return { x_M, x_m, delta: x_M - x_m };
        }
        
        // 更新画面
        function updateScene() {
            const disp = calculateDisplacement(currentTime);
            const x_M_px = disp.x_M * SCALE;
            const x_m_px = disp.x_m * SCALE;
            const delta_px = disp.delta * SCALE;
            
            // 更新滑块手柄位置
            const sliderMin = 0;
            const sliderMax = 300;
            const sliderPercent = (currentTime * 100) / 300; // 0-3s 映射到 0-100%
            const handleX = 200 + (sliderPercent / 100) * 400;
            滑块手柄.setAttribute('cx', handleX);
            
            // 更新滑块值显示
            滑块值显示.textContent = currentTime.toFixed(2) + 's';
            
            // 更新状态指示
            if (currentTime <= t0) {
                状态文本.textContent = 'F 作用中 (F=10N)';
                F 箭头线.style.display = 'block';
                F 箭头头.style.display = 'block';
                F 标签.style.display = 'block';
            } else {
                状态文本.textContent = 'F 已撤去';
                F 箭头线.style.display = 'none';
                F 箭头头.style.display = 'none';
                F 标签.style.display = 'none';
            }
            
            // 更新木板当前位置
            木板当前.setAttribute('x', 木板初始 X + x_M_px);
            
            // 更新物块当前位置
            物块当前.setAttribute('x', 物块初始 X + x_m_px);
            
            // 更新位移线
            xM 线.setAttribute('x2', 木板初始 X + x_M_px);
            xm 线.setAttribute('x2', 物块初始 X + x_m_px);
            delta 线.setAttribute('x2', 木板初始 X + delta_px);
            
            // 更新数值显示
            t 值显示.textContent = 't=' + currentTime.toFixed(2) + 's';
            xM 值显示.textContent = 'x_M=' + disp.x_M.toFixed(2) + 'm';
            xm 值显示.textContent = 'x_m=' + disp.x_m.toFixed(2) + 'm';
            delta 值显示.textContent = 'Δx=' + disp.delta.toFixed(2) + 'm';
            
            // 临界状态提示
            if (Math.abs(disp.delta - L) < 0.05) {
                临界文本.style.display = 'block';
                delta 线.setAttribute('stroke-width', '4');
            } else {
                临界文本.style.display = 'none';
                delta 线.setAttribute('stroke-width', '2');
            }
        }
        
        // 滑块事件处理
        function handleSliderInput(e) {
            const value = parseInt(e.target.value);
            currentTime = value / 100; // 0-300 映射到 0-3s
            updateScene();
        }
        
        滑块.addEventListener('input', handleSliderInput);
        
        // 初始化
        updateScene();
    })();
    </script>
