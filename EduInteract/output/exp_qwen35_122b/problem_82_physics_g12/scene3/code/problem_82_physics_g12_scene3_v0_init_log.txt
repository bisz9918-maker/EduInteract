    <script>
    (function() {
        const TOTAL_STEPS = 4;
        let currentStep = 0;

        function showSvg(el) {
            if (el) {
                el.style.display = 'inline';
                el.style.opacity = '1';
            }
        }

        function hideSvg(el) {
            if (el) {
                el.style.display = 'none';
                el.style.opacity = '0';
            }
        }

        function updateScene() {
            const stepIndicator = document.getElementById('scene3-step-indicator');
            const btnPrev = document.getElementById('scene3-btn-prev');
            const btnNext = document.getElementById('scene3-btn-next');

            stepIndicator.textContent = `步骤 ${currentStep + 1}/${TOTAL_STEPS}`;
            btnPrev.disabled = (currentStep === 0);
            btnNext.disabled = (currentStep === TOTAL_STEPS - 1);
            if (currentStep === TOTAL_STEPS - 1) {
                btnNext.textContent = '完成';
            } else {
                btnNext.textContent = '下一步 ▶';
            }

            // 斜面组始终显示
            showSvg(document.getElementById('scene3-slope-group'));

            // 受力分析 B 球始终显示
            showSvg(document.getElementById('scene3-ball-b-force'));
            showSvg(document.getElementById('scene3-label-b'));
            
            if (currentStep >= 0) {
                showSvg(document.getElementById('scene3-force-t1'));
                showSvg(document.getElementById('scene3-force-t1-head'));
                showSvg(document.getElementById('scene3-label-t1'));
                showSvg(document.getElementById('scene3-force-t2'));
                showSvg(document.getElementById('scene3-force-t2-head'));
                showSvg(document.getElementById('scene3-label-t2'));
                showSvg(document.getElementById('scene3-force-gb'));
                showSvg(document.getElementById('scene3-force-gb-head'));
                showSvg(document.getElementById('scene3-label-gb'));
                showSvg(document.getElementById('scene3-force-n'));
                showSvg(document.getElementById('scene3-force-n-head'));
                showSvg(document.getElementById('scene3-label-n'));
            }

            if (currentStep >= 0) {
                showSvg(document.getElementById('scene3-formula-box'));
                showSvg(document.getElementById('scene3-formula-title'));
            }

            if (currentStep >= 1) {
                document.getElementById('scene3-force-t1').setAttribute('stroke-width', '4');
                document.getElementById('scene3-force-t2').setAttribute('stroke-width', '4');
                showSvg(document.getElementById('scene3-disp-s1'));
                showSvg(document.getElementById('scene3-disp-s1-head'));
                showSvg(document.getElementById('scene3-label-s1'));
                showSvg(document.getElementById('scene3-wt1'));
                showSvg(document.getElementById('scene3-wt2'));
            }

            if (currentStep >= 2) {
                showSvg(document.getElementById('scene3-disp-s2'));
                showSvg(document.getElementById('scene3-disp-s2-head'));
                showSvg(document.getElementById('scene3-label-s2'));
                showSvg(document.getElementById('scene3-wtotal'));
                showSvg(document.getElementById('scene3-ke-box'));
                showSvg(document.getElementById('scene3-ke-title'));
                showSvg(document.getElementById('scene3-ke-formula'));
            }

            if (currentStep >= 3) {
                showSvg(document.getElementById('scene3-calc-t1'));
                showSvg(document.getElementById('scene3-result-box'));
                showSvg(document.getElementById('scene3-result-label'));
                showSvg(document.getElementById('scene3-result-value'));
            }
        }

        function nextStep() {
            if (currentStep < TOTAL_STEPS - 1) {
                currentStep++;
                updateScene();
            }
        }

        function prevStep() {
            if (currentStep > 0) {
                currentStep--;
                updateScene();
            }
        }

        function init() {
            document.getElementById('scene3-btn-next').addEventListener('click', nextStep);
            document.getElementById('scene3-btn-prev').addEventListener('click', prevStep);
            updateScene();
        }

        init();
    })();
    </script>
