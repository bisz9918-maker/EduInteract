                       'scene1-vec-mg', 'scene1-label-mg', 'scene1-trajectory', 'scene1-point-O',

        function showElement(id) {
            const el = document.getElementById(id);
            if (el) {
                el.style.display = '';
                el.style.opacity = '1';
            }
        }

        function hideElement(id) {
            const el = document.getElementById(id);
            if (el) {
                el.style.display = 'none';
                el.style.opacity = '0';
            }
        }

        function updateDisplay() {
            const visibility = stepVisibility[currentStep];

            // 先隐藏所有可能隐藏的元素
            visibility.hide.forEach(id => hideElement(id));

            // 再显示所有需要显示的元素
            visibility.show.forEach(id => showElement(id));

            // 更新步骤指示器
            document.getElementById('scene1-step-indicator').textContent = '步骤 ' + (currentStep + 1) + '/5';

            // 更新按钮状态
            document.getElementById('scene1-btn-prev').disabled = (currentStep === 0);
            document.getElementById('scene1-btn-next').disabled = (currentStep === currentStepMax);
        }

        // 按钮事件绑定
        document.getElementById('scene1-btn-prev').addEventListener('click', function() {
            if (currentStep > 0) {
                currentStep--;
                updateDisplay();
            }
        });

        document.getElementById('scene1-btn-next').addEventListener('click', function() {
            if (currentStep < currentStepMax) {
                currentStep++;
                updateDisplay();
            }
        });

        // 初始化显示
        updateDisplay();
    })();
