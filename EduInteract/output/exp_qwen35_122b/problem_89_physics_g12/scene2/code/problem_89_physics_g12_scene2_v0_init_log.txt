    <script>
    (function() {
        // Scene 2 实现 - 速度矢量分解与合成分析
        // 布局坐标表对照：
        // | 元素 | 类型 | 位置/范围 | 大小 | 标签偏移方向 | 备注 |
        // | 坐标轴 x | 线段 | (80,400)-(750,400) | 670×2 | - | 水平轴，箭头向右 |
        // | 坐标轴 y | 线段 | (80,400)-(80,50) | 2×350 | - | 竖直轴，箭头向上 |
        // | 原点 O | 点 | (80,400) | - | 左下 (65,415) | 坐标原点 |
        // | 虚线 ab | 线段 | (280,50)-(280,400) | 2×350 | - | x=L 分界线 |
        // | E₁区域 | 矩形边框 | (80,50)-(280,400) | 200×350 | - | 左区电场范围 |
        // | E₂区域 | 矩形边框 | (280,50)-(750,400) | 470×350 | - | 右区电场范围 |
        // | P 点 | 实心圆 | (80,200) | 5px 半径 | - | (0,h) 位置 |
        // | Q 点 | 实心圆 | (600,400) | 5px 半径 | - | x 轴上出射点 |
        // | v₁矢量 | 箭头 | (280,200)-(400,200) | 120×3 | - | 步骤 1 显示，水平向右 |
        // | v₂矢量 | 箭头 | (600,400)-(700,280) | - | - | 步骤 2 显示，斜向左上 |
        // | v_x 矢量 | 箭头 | (600,400)-(700,400) | 100×3 | - | 水平分量，蓝色 |
        // | v_y 矢量 | 箭头 | (700,400)-(700,280) | 3×120 | - | 竖直分量，红色 |
        
        const TOTAL_STEPS = 4;
        let currentStep = 0;
        
        // DOM 元素引用
        const els = {
            stepIndicator: document.getElementById('scene2-step-indicator'),
            btnPrev: document.getElementById('scene2-btn-prev'),
            btnNext: document.getElementById('scene2-btn-next'),
            // 步骤 0 元素
            labelV0: document.getElementById('scene2-label-v0'),
            // 步骤 1 元素
            vecV1Group: document.getElementById('scene2-vec-v1-group'),
            // 步骤 2 元素
            vecDecompGroup: document.getElementById('scene2-vec-decomp-group'),
            // 步骤 3 元素
            labelFormula: document.getElementById('scene2-label-formula')
        };
        
        // 工具函数：显示元素
        function show(el) {
            if (!el) return;
            el.classList.remove('scene2-hidden');
            el.style.display = 'block';
            el.style.opacity = '1';
        }
        
        // 工具函数：隐藏元素
        function hide(el) {
            if (!el) return;
            el.classList.add('scene2-hidden');
            el.style.display = 'none';
            el.style.opacity = '0';
        }
        
        // 更新场景显示
        function updateScene() {
            // 更新步骤指示器
            els.stepIndicator.textContent = `步骤 ${currentStep + 1}/${TOTAL_STEPS}`;
            
            // 更新按钮状态
            els.btnPrev.disabled = (currentStep === 0);
            els.btnNext.disabled = (currentStep === TOTAL_STEPS - 1);
            if (currentStep === TOTAL_STEPS - 1) {
                els.btnNext.textContent = '完成';
            } else {
                els.btnNext.textContent = '下一步 ▶';
            }
            
            // 根据当前步骤显示/隐藏元素
            switch(currentStep) {
                case 0:
                    // 步骤 0: 显示 v₀=0 标签，隐藏所有速度矢量
                    show(els.labelV0);
                    hide(els.vecV1Group);
                    hide(els.vecDecompGroup);
                    hide(els.labelFormula);
                    break;
                    
                case 1:
                    // 步骤 1: 显示 v₁矢量，隐藏 v₀=0
                    hide(els.labelV0);
                    show(els.vecV1Group);
                    hide(els.vecDecompGroup);
                    hide(els.labelFormula);
                    break;
                    
                case 2:
                    // 步骤 2: 显示速度分解矢量
                    hide(els.labelV0);
                    hide(els.vecV1Group);
                    show(els.vecDecompGroup);
                    hide(els.labelFormula);
                    break;
                    
                case 3:
                    // 步骤 3: 显示公式
                    hide(els.labelV0);
                    hide(els.vecV1Group);
                    show(els.vecDecompGroup);
                    show(els.labelFormula);
                    break;
            }
        }
        
        // 下一步
        function nextStep() {
            if (currentStep < TOTAL_STEPS - 1) {
                currentStep++;
                updateScene();
            }
        }
        
        // 上一步
        function prevStep() {
            if (currentStep > 0) {
                currentStep--;
                updateScene();
            }
        }
        
        // 事件绑定
        els.btnNext.addEventListener('click', nextStep);
        els.btnPrev.addEventListener('click', prevStep);
        
        // 初始化场景
        updateScene();
    })();
    </script>
