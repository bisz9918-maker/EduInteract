    <script>
    (function() {
        // Scene 1 实现
        // 状态机：currentStep (0-4)
        let currentStep = 0;
        
        // DOM 元素引用
        const elements = {
            battery: document.getElementById('scene1-battery'),
            switch: document.getElementById('scene1-switch'),
            lamp1: document.getElementById('scene1-lamp1'),
            lamp2: document.getElementById('scene1-lamp2'),
            voltmeter: document.getElementById('scene1-voltmeter'),
            wires: document.getElementById('scene1-wires'),
            currentArrow: document.getElementById('scene1-current-arrow'),
            measureBox: document.getElementById('scene1-measure-box'),
            labels: {
                batteryPlus: document.getElementById('scene1-label-battery-plus'),
                batteryMinus: document.getElementById('scene1-label-battery-minus'),
                switch: document.getElementById('scene1-label-switch'),
                lamp1: document.getElementById('scene1-label-lamp1'),
                lamp2: document.getElementById('scene1-label-lamp2'),
                voltmeter: document.getElementById('scene1-label-voltmeter')
            },
            btnPrev: document.getElementById('scene1-btn-prev'),
            btnNext: document.getElementById('scene1-btn-next')
        };
        
        // 步骤效果配置
        const stepEffects = {
            0: { 
                highlight: [], 
                showLabels: [],
                showArrow: false,
                showMeasureBox: false
            },
            1: { 
                highlight: ['battery', 'switch'], 
                showLabels: ['batteryPlus', 'batteryMinus', 'switch'],
                showArrow: false,
                showMeasureBox: false
            },
            2: { 
                highlight: ['lamp1', 'lamp2'], 
                showLabels: ['lamp1', 'lamp2'],
                showArrow: false,
                showMeasureBox: false
            },
            3: { 
                highlight: [], 
                showLabels: [],
                showArrow: true,
                showMeasureBox: false
            },
            4: { 
                highlight: ['voltmeter'], 
                showLabels: ['voltmeter'],
                showArrow: true,
                showMeasureBox: true
            }
        };
        
        // 工具函数：显示元素
        function show(el) {
            if (el) {
                el.style.display = 'block';
                el.style.opacity = '1';
            }
        }
        
        // 工具函数：隐藏元素
        function hide(el) {
            if (el) {
                el.style.display = 'none';
                el.style.opacity = '0';
            }
        }
        
        // 工具函数：移除高亮
        function removeHighlight(el) {
            if (el) {
                el.classList.remove('highlight-blue');
                // 恢复默认样式
                const children = el.querySelectorAll('*');
                children.forEach(child => {
                    child.classList.remove('highlight-blue');
                });
            }
        }
        
        // 工具函数：添加蓝色高亮
        function addHighlight(el) {
            if (el) {
                el.classList.add('highlight-blue');
            }
        }
        
        // 更新步骤
        function updateStep() {
            const effect = stepEffects[currentStep];
            
            // 清除所有高亮
            ['battery', 'switch', 'lamp1', 'lamp2', 'voltmeter'].forEach(key => {
                removeHighlight(elements[key]);
            });
            
            // 隐藏所有标签
            Object.values(elements.labels).forEach(label => {
                hide(label);
            });
            
            // 隐藏电流箭头和测量框
            hide(elements.currentArrow);
            hide(elements.measureBox);
            
            // 应用当前步骤效果
            effect.highlight.forEach(key => {
                addHighlight(elements[key]);
            });
            
            effect.showLabels.forEach(key => {
                show(elements.labels[key]);
            });
            
            if (effect.showArrow) {
                show(elements.currentArrow);
            }
            
            if (effect.showMeasureBox) {
                show(elements.measureBox);
            }
            
            // 更新按钮状态
            elements.btnPrev.disabled = (currentStep === 0);
            elements.btnNext.disabled = (currentStep === 4);
        }
        
        // 事件绑定
        elements.btnPrev.addEventListener('click', () => {
            if (currentStep > 0) {
                currentStep--;
                updateStep();
            }
        });
        
        elements.btnNext.addEventListener('click', () => {
            if (currentStep < 4) {
                currentStep++;
                updateStep();
            }
        });
        
        // 初始化
        updateStep();
    })();
    </script>
