    <script>
    (function() {
        // Scene 2 实现 - 带电小球系统受力分析
        
        const MAX_STEP = 6;
        let currentStep = 0;
        
        // DOM 元素引用
        const prevBtn = document.getElementById('scene2-prev-btn');
        const nextBtn = document.getElementById('scene2-next-btn');
        const leftGroup = document.getElementById('scene2-left-group');
        const rightGroup = document.getElementById('scene2-right-group');
        const equationBox = document.getElementById('scene2-equation-box');
        const equationText = document.getElementById('scene2-equation-text');
        const resultBox = document.getElementById('scene2-result-box');
        const resultText = document.getElementById('scene2-result-text');
        
        // 步骤对应的显示配置
        const stepConfig = {
            0: {
                showLeft: true,
                showRight: false,
                showEquation: false,
                showResult: false,
                equationText: '',
                resultText: '',
                arrows: { fac: true, fbc: true, fheqc: false, fExternal: false }
            },
            1: {
                showLeft: true,
                showRight: false,
                showEquation: false,
                showResult: false,
                equationText: '',
                resultText: '',
                arrows: { fac: true, fbc: true, fheqc: true, fExternal: false }
            },
            2: {
                showLeft: true,
                showRight: false,
                showEquation: true,
                showResult: false,
                equationText: 'C 球牛顿第二定律：F - F_合 C = ma',
                resultText: '',
                arrows: { fac: true, fbc: true, fheqc: true, fExternal: true }
            },
            3: {
                showLeft: false,
                showRight: true,
                showEquation: false,
                showResult: false,
                equationText: '',
                resultText: '',
                arrows: { fba: true, fca: true, fcaX: false, fcaY: false }
            },
            4: {
                showLeft: false,
                showRight: true,
                showEquation: false,
                showResult: false,
                equationText: '',
                resultText: '',
                arrows: { fba: true, fca: true, fcaX: true, fcaY: true }
            },
            5: {
                showLeft: false,
                showRight: true,
                showEquation: true,
                showResult: false,
                equationText: 'A 球水平方向：F_BA + F_CA·cos30° = ma',
                resultText: '',
                arrows: { fba: true, fca: true, fcaX: true, fcaY: true }
            },
            6: {
                showLeft: false,
                showRight: true,
                showEquation: false,
                showResult: true,
                equationText: '',
                resultText: '解得：Q = 2q（C 带负电），F = 3kq²/r²',
                arrows: { fba: true, fca: true, fcaX: true, fcaY: true }
            }
        };
        
        function updateScene() {
            const config = stepConfig[currentStep];
            
            // 控制左右组显示
            if (config.showLeft) {
                leftGroup.style.opacity = '1';
                rightGroup.style.opacity = '0';
            } else {
                leftGroup.style.opacity = '0';
                rightGroup.style.opacity = '1';
            }
            
            // 控制箭头显示
            if (config.showLeft) {
                document.getElementById('scene2-arrow-fac').style.opacity = config.arrows.fac ? '1' : '0';
                document.getElementById('scene2-arrow-fbc').style.opacity = config.arrows.fbc ? '1' : '0';
                document.getElementById('scene2-arrow-fheqc').style.opacity = config.arrows.fheqc ? '1' : '0';
                document.getElementById('scene2-arrow-f-external').style.opacity = config.arrows.fExternal ? '1' : '0';
            } else {
                document.getElementById('scene2-arrow-fba').style.opacity = config.arrows.fba ? '1' : '0';
                document.getElementById('scene2-arrow-fca').style.opacity = config.arrows.fca ? '1' : '0';
                document.getElementById('scene2-arrow-fca-x').style.opacity = config.arrows.fcaX ? '1' : '0';
                document.getElementById('scene2-arrow-fca-y').style.opacity = config.arrows.fcaY ? '1' : '0';
            }
            
            // 控制方程/结果显示
            if (config.showEquation) {
                equationBox.style.opacity = '1';
                resultBox.style.opacity = '0';
                equationText.style.opacity = '1';
                resultText.style.opacity = '0';
                equationText.textContent = config.equationText;
            } else if (config.showResult) {
                equationBox.style.opacity = '0';
                resultBox.style.opacity = '1';
                equationText.style.opacity = '0';
                resultText.style.opacity = '1';
                resultText.textContent = config.resultText;
            } else {
                equationBox.style.opacity = '0';
                resultBox.style.opacity = '0';
                equationText.style.opacity = '0';
                resultText.style.opacity = '0';
            }
            
            // 更新按钮状态
            prevBtn.disabled = (currentStep === 0);
            if (currentStep === MAX_STEP) {
                nextBtn.textContent = '完成 ✓';
                nextBtn.disabled = true;
            } else {
                nextBtn.textContent = '下一步 ▶';
                nextBtn.disabled = false;
            }
            
            // 触发 MathJax 重新渲染（如果有公式）
            if (window.MathJax) {
                MathJax.typesetPromise();
            }
        }
        
        // 按钮事件绑定
        prevBtn.addEventListener('click', function() {
            if (currentStep > 0) {
                currentStep--;
                updateScene();
            }
        });
        
        nextBtn.addEventListener('click', function() {
            if (currentStep < MAX_STEP) {
                currentStep++;
                updateScene();
            }
        });
        
        // 初始化场景
        updateScene();
    })();
    </script>
