    <script>
    (function() {
        // Scene 1: 等边三角形折叠问题
        // 布局坐标表（参考）：
        // | 元素 | 位置 | 备注 |
        // | A(220,380), B(580,380), C(400,68) | 三角形顶点 |
        // | D(340,380) | AB 上的点，AD=1, BD=2 |
        // | E(316,214) | AC 上的点 |
        // | F(505,250) | BC 上的点，BF=1.25, CF=1.75 |
        // | 折痕 EF(316,214)-(505,250) | 虚线 |
        
        const scene1 = {
            currentStep: 0,
            totalSteps: 4,
            elements: {},
            animationFrame: null
        };
        
        // 初始化元素引用
        function initElements() {
            scene1.elements = {
                // 图形元素
                triCefDashed: document.getElementById('scene1-tri-cef-dashed'),
                lineEf: document.getElementById('scene1-line-ef'),
                lineDe: document.getElementById('scene1-line-de'),
                lineDf: document.getElementById('scene1-line-df'),
                pointC: document.getElementById('scene1-point-c'),
                pointE: document.getElementById('scene1-point-e'),
                pointF: document.getElementById('scene1-point-f'),
                pointCMoving: document.getElementById('scene1-point-c-moving'),
                
                // 标签元素
                labelC: document.getElementById('scene1-label-c'),
                labelE: document.getElementById('scene1-label-e'),
                labelF: document.getElementById('scene1-label-f'),
                labelCf: document.getElementById('scene1-label-cf'),
                labelCe: document.getElementById('scene1-label-ce'),
                labelCeDe: document.getElementById('scene1-label-ce-de'),
                labelCfDf: document.getElementById('scene1-label-cf-df'),
                answerBox: document.getElementById('scene1-answer-box'),
                answerText: document.getElementById('scene1-answer-text'),
                answerOption: document.getElementById('scene1-answer-option'),
                
                // 控件
                btnPrev: document.getElementById('scene1-btn-prev'),
                btnNext: document.getElementById('scene1-btn-next'),
                stepIndicator: document.getElementById('scene1-step-indicator')
            };
        }
        
        // 显示/隐藏工具函数
        function show(el) {
            if (el) {
                el.style.display = 'block';
                el.style.opacity = '0';
                setTimeout(() => { el.style.opacity = '1'; }, 10);
            }
        }
        
        function hide(el) {
            if (el) {
                el.style.opacity = '0';
                setTimeout(() => { el.style.display = 'none'; }, 300);
            }
        }
        
        // 更新场景显示
        function updateScene() {
            const el = scene1.elements;
            
            // 重置所有步骤相关元素
            const stepElements = [
                el.triCefDashed, el.lineEf, el.pointC, el.pointE, el.pointF,
                el.labelC, el.labelE, el.labelF, el.labelCf,
                el.lineDe, el.lineDf, el.labelCeDe, el.labelCfDf,
                el.labelCe, el.answerBox, el.answerText, el.answerOption
            ];
            stepElements.forEach(e => { if (e) { e.style.display = 'none'; e.style.opacity = '0'; } });
            
            // 步骤 0：初始状态 - 显示三角形 ABC、点 D、标签 A/B/C/D、数值 AD=1, BD=2, BF=1.25
            // 步骤 1：显示折痕 EF、虚线三角形 CEF、点 E/F/C、标签 E/F/C、数值 CF=1.75
            // 步骤 2：显示线段 DE/DF、等长标记 CE=DE/CF=DF、折叠动画
            // 步骤 3：显示 CE=?、答案区域
            
            if (scene1.currentStep >= 0) {
                // 基础元素始终显示
            }
            
            if (scene1.currentStep >= 1) {
                show(el.triCefDashed);
                show(el.lineEf);
                show(el.pointC);
                show(el.pointE);
                show(el.pointF);
                show(el.labelC);
                show(el.labelE);
                show(el.labelF);
                show(el.labelCf);
            }
            
            if (scene1.currentStep >= 2) {
                show(el.lineDe);
                show(el.lineDf);
                show(el.labelCeDe);
                show(el.labelCfDf);
                // 折叠动画：C 点移动到 D 点
                animateFold();
            }
            
            if (scene1.currentStep >= 3) {
                show(el.labelCe);
                show(el.answerBox);
                show(el.answerText);
                show(el.answerOption);
            }
            
            // 更新按钮状态
            el.btnPrev.disabled = scene1.currentStep === 0;
            el.btnNext.disabled = scene1.currentStep === scene1.totalSteps - 1;
            
            // 更新步骤指示器
            el.stepIndicator.textContent = `步骤 ${scene1.currentStep + 1}/${scene1.totalSteps}`;
        }
        
        // 折叠动画：C 点从 (400,68) 移动到 (340,380)
        function animateFold() {
            if (scene1.animationFrame) {
                cancelAnimationFrame(scene1.animationFrame);
            }
            
            const el = scene1.elements;
            const startX = 400, startY = 68;
            const endX = 340, endY = 380;
            const duration = 500;
            const startTime = performance.now();
            
            // 隐藏静态 C 点，显示移动 C 点
            hide(el.pointC);
            show(el.pointCMoving);
            
            function animate(currentTime) {
                const elapsed = currentTime - startTime;
                const progress = Math.min(elapsed / duration, 1);
                
                // 线性插值
                const currentX = startX + (endX - startX) * progress;
                const currentY = startY + (endY - startY) * progress;
                
                el.pointCMoving.setAttribute('cx', currentX);
                el.pointCMoving.setAttribute('cy', currentY);
                
                if (progress < 1) {
                    scene1.animationFrame = requestAnimationFrame(animate);
                } else {
                    // 动画结束，显示静态 D 点（已存在）
                    hide(el.pointCMoving);
                }
            }
            
            scene1.animationFrame = requestAnimationFrame(animate);
        }
        
        // 事件绑定
        function bindEvents() {
            scene1.elements.btnPrev.addEventListener('click', () => {
                if (scene1.currentStep > 0) {
                    scene1.currentStep--;
                    updateScene();
                }
            });
            
            scene1.elements.btnNext.addEventListener('click', () => {
                if (scene1.currentStep < scene1.totalSteps - 1) {
                    scene1.currentStep++;
                    updateScene();
                }
            });
        }
        
        // 初始化
        function init() {
            initElements();
            bindEvents();
            updateScene();
        }
        
        // 页面加载完成后初始化
        if (document.readyState === 'loading') {
            document.addEventListener('DOMContentLoaded', init);
        } else {
            init();
        }
    })();
    </script>
