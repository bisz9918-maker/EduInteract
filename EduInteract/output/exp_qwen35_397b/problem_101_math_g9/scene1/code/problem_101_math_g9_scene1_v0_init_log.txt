    <script>
    (function() {
        // Scene 1: 正方形折叠变换过程
        // 布局坐标表参考：
        // | 元素 | 类型 | 位置/范围 | 大小 | 备注 |
        // |------|------|-----------|------|------|
        // | 正方形 | rect | (150,150)-(350,350) | 200×200 | 左侧主图 |
        // | 对角线折痕 | line | (150,150)-(350,350) | 283px | 虚线 |
        // | 折叠箭头 | path | (260,200) 弧形指向 (260,300) | 60×80 | 蓝色 |
        // | 过程箭头 | line | (380,240)-(420,240) | 40×12 | 黑色 |
        // | 三角形 | polygon | (480,150)-(480,350)-(680,350) | 200×200 | 右侧 |
        // | 圆洞 1/2/3 | circle | (500,180), (500,330), (650,330) | r=8 | 白色填充 |
        // | 直角标记 | path | (480,335)-(495,335)-(495,350) | 15×15 | 红色 |
        // | 标签"对角线" | text | (265,260) | 14px #888888 |
        // | 标签"直角顶点" | text | (500,375) | 14px #000000 |
        // | 标签"斜边" | text | (590,225) | 14px #000000 |
        // | 步骤标题 | text | (400,40) | 18px 粗体 #1565c0 |
        // | 控件栏 | div | (200,450)-(600,490) | 400×40 | 底部居中 |

        const TOTAL_STEPS = 4;
        let currentStep = 0;

        // 获取 DOM 元素
        const stepTitle = document.getElementById('scene1-step-title');
        const stepIndicator = document.getElementById('scene1-step-indicator');
        const btnPrev = document.getElementById('scene1-btn-prev');
        const btnNext = document.getElementById('scene1-btn-next');

        // 各步骤元素
        const diagonal = document.getElementById('scene1-diagonal');
        const labelDiagonal = document.getElementById('scene1-label-diagonal');
        const foldArrow = document.getElementById('scene1-fold-arrow');
        const triangle = document.getElementById('scene1-triangle');
        const hole1 = document.getElementById('scene1-hole1');
        const hole2 = document.getElementById('scene1-hole2');
        const hole3 = document.getElementById('scene1-hole3');
        const rightAngleMark = document.getElementById('scene1-right-angle-mark');
        const labelRightAngle = document.getElementById('scene1-label-right-angle');
        const labelHypotenuse = document.getElementById('scene1-label-hypotenuse');

        // 工具函数：显示元素
        function show(el) {
            el.classList.remove('scene1-hidden');
            el.classList.add('scene1-visible');
            el.style.display = 'block';
            el.style.opacity = '1';
        }

        // 工具函数：隐藏元素
        function hide(el) {
            el.classList.remove('scene1-visible');
            el.classList.add('scene1-hidden');
            el.style.display = 'none';
            el.style.opacity = '0';
        }

        // 更新视图
        function updateView() {
            // 更新标题和指示器
            const stepTitles = [
                '步骤 1/4：初始正方形',
                '步骤 2/4：显示对角线折痕',
                '步骤 3/4：沿对角线折叠',
                '步骤 4/4：形成等腰直角三角形'
            ];
            stepTitle.textContent = stepTitles[currentStep];
            stepIndicator.textContent = `步骤 ${currentStep + 1}/${TOTAL_STEPS}`;

            // 更新按钮状态
            btnPrev.disabled = (currentStep === 0);
            if (currentStep === TOTAL_STEPS - 1) {
                btnNext.textContent = '完成';
            } else {
                btnNext.textContent = '下一步 ▶';
            }

            // 步骤 0：初始状态
            if (currentStep === 0) {
                hide(diagonal);
                hide(labelDiagonal);
                hide(foldArrow);
                triangle.setAttribute('fill', '#e0e0e0');
                hide(hole1);
                hide(hole2);
                hide(hole3);
                hide(rightAngleMark);
                hide(labelRightAngle);
                hide(labelHypotenuse);
            }
            // 步骤 1：显示对角线
            else if (currentStep === 1) {
                show(diagonal);
                show(labelDiagonal);
                hide(foldArrow);
                triangle.setAttribute('fill', '#e0e0e0');
                hide(hole1);
                hide(hole2);
                hide(hole3);
                hide(rightAngleMark);
                hide(labelRightAngle);
                hide(labelHypotenuse);
            }
            // 步骤 2：显示折叠箭头
            else if (currentStep === 2) {
                show(diagonal);
                show(labelDiagonal);
                show(foldArrow);
                triangle.setAttribute('fill', '#e0e0e0');
                hide(hole1);
                hide(hole2);
                hide(hole3);
                hide(rightAngleMark);
                hide(labelRightAngle);
                hide(labelHypotenuse);
            }
            // 步骤 3：显示三角形和圆洞
            else if (currentStep === 3) {
                show(diagonal);
                show(labelDiagonal);
                show(foldArrow);
                triangle.setAttribute('fill', '#e3f2fd');
                show(hole1);
                show(hole2);
                show(hole3);
                show(rightAngleMark);
                show(labelRightAngle);
                show(labelHypotenuse);
            }
        }

        // 按钮事件处理
        btnPrev.addEventListener('click', function() {
            if (currentStep > 0) {
                currentStep--;
                updateView();
            }
        });

        btnNext.addEventListener('click', function() {
            if (currentStep < TOTAL_STEPS - 1) {
                currentStep++;
                updateView();
            }
        });

        // 初始化视图
        updateView();
    })();
    </script>
