    <script>
    (function() {
        // Scene 2 实现：半圆旋转阴影面积割补法演示
        
        const state = { step: 0 };
        const totalSteps = 5;
        
        // 步骤配置
        const steps = [
            {
                formula: "分析阴影部分面积...",
                sectorOpacity: 0,
                rotatedOpacity: 0.25,
                originalFill: "#ffffff",
                shadowFill: "#888888",
                shadowOpacity: 0.4,
                extractedOpacity: 0
            },
            {
                formula: "S_阴影 = S_旋转后半圆 + S_扇形 ABA' - S_原始半圆",
                sectorOpacity: 0.15,
                rotatedOpacity: 0.3,
                originalFill: "#ffffff",
                shadowFill: "#888888",
                shadowOpacity: 0.4,
                extractedOpacity: 0
            },
            {
                formula: "S_阴影 = S_旋转后半圆 + S_扇形 ABA' - S_原始半圆",
                sectorOpacity: 0.3,
                rotatedOpacity: 0.3,
                originalFill: "#ffffff",
                shadowFill: "#c62828",
                shadowOpacity: 0.3,
                extractedOpacity: 0
            },
            {
                formula: "S_旋转后半圆 = S_原始半圆  ∴ S_阴影 = S_扇形 ABA'",
                sectorOpacity: 0.4,
                rotatedOpacity: 0.3,
                originalFill: "#ffffff",
                shadowFill: "#c62828",
                shadowOpacity: 0.3,
                extractedOpacity: 0.5
            },
            {
                formula: "S_扇形 = (45°/360°) × π × 4² = 2π",
                sectorOpacity: 0.4,
                rotatedOpacity: 0.3,
                originalFill: "#ffffff",
                shadowFill: "#c62828",
                shadowOpacity: 0.3,
                extractedOpacity: 1
            }
        ];
        
        // 获取 DOM 元素
        const sectorAba = document.getElementById('scene2-sector-aba');
        const semicircleRotated = document.getElementById('scene2-semicircle-rotated');
        const semicircleOriginal = document.getElementById('scene2-semicircle-original');
        const shadowRegion = document.getElementById('scene2-shadow-region');
        const extractedGroup = document.getElementById('scene2-extracted-sector-group');
        const formulaText = document.getElementById('scene2-formula-text');
        const stepIndicator = document.getElementById('scene2-step-indicator');
        const btnPrev = document.getElementById('scene2-btn-prev');
        const btnNext = document.getElementById('scene2-btn-next');
        const btnPrevText = btnPrev.querySelector('text');
        const btnPrevRect = btnPrev.querySelector('rect');
        const btnNextText = btnNext.querySelector('text');
        const btnNextRect = btnNext.querySelector('rect');
        
        // 更新步骤显示
        function renderStep(stepIndex) {
            const step = steps[stepIndex];
            
            sectorAba.setAttribute('opacity', step.sectorOpacity);
            semicircleRotated.setAttribute('opacity', step.rotatedOpacity);
            semicircleOriginal.setAttribute('fill', step.originalFill);
            shadowRegion.setAttribute('fill', step.shadowFill);
            shadowRegion.setAttribute('opacity', step.shadowOpacity);
            extractedGroup.setAttribute('opacity', step.extractedOpacity);
            formulaText.textContent = step.formula;
            stepIndicator.textContent = `步骤 ${stepIndex + 1}/${totalSteps}`;
            
            // 更新上一步按钮
            if (stepIndex === 0) {
                btnPrev.classList.add('scene2-btn-disabled');
                btnPrevRect.setAttribute('fill', '#e0e0e0');
                btnPrevText.setAttribute('fill', '#999');
            } else {
                btnPrev.classList.remove('scene2-btn-disabled');
                btnPrevRect.setAttribute('fill', '#e0e0e0');
                btnPrevText.setAttribute('fill', '#333');
            }
            
            // 更新下一步按钮
            if (stepIndex === totalSteps - 1) {
                btnNext.classList.add('scene2-btn-disabled');
                btnNextRect.setAttribute('fill', '#e0e0e0');
                btnNextText.setAttribute('fill', '#999');
            } else {
                btnNext.classList.remove('scene2-btn-disabled');
                btnNextRect.setAttribute('fill', '#2196f3');
                btnNextText.setAttribute('fill', '#ffffff');
            }
        }
        
        // 按钮点击事件
        btnPrev.addEventListener('click', function() {
            if (state.step > 0) {
                state.step--;
                renderStep(state.step);
            }
        });
        
        btnNext.addEventListener('click', function() {
            if (state.step < totalSteps - 1) {
                state.step++;
                renderStep(state.step);
            }
        });
        
        // 初始化
        renderStep(0);
    })();
    </script>
