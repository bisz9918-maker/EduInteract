    <script>
    (function() {
        // Scene 1 实现 - 经纬网坐标读取与点位识别

        // 状态变量
        let currentStep = 0;
        const totalSteps = 6;

        // 各点坐标信息
        const pointData = {
            jia: { label: '甲', coord: '(0°, 66.5°N)' },
            yi: { label: '乙', coord: '(20°W, 23.5°N)' },
            bing: { label: '丙', coord: '(0°, 0°)' },
            ding: { label: '丁', coord: '(20°E, 23.5°S)' },
            E: { label: 'E', coord: '(20°E, 0°)' },
            F: { label: 'F', coord: '(40°E, 66.5°N)' }
        };

        // DOM 元素
        const coordDisplay = document.getElementById('scene1-coord-display');
        const stepIndicator = document.getElementById('scene1-step-indicator');
        const btnPrev = document.getElementById('scene1-btn-prev');
        const btnNext = document.getElementById('scene1-btn-next');
        const arrowsGroup = document.getElementById('scene1-arrows');

        // 高亮元素
        const highlights = {
            jia: document.getElementById('scene1-highlight-jia'),
            yi: document.getElementById('scene1-highlight-yi'),
            bing: document.getElementById('scene1-highlight-bing'),
            ding: document.getElementById('scene1-highlight-ding'),
            E: document.getElementById('scene1-highlight-E'),
            F: document.getElementById('scene1-highlight-F')
        };

        // 隐藏所有高亮
        function hideAllHighlights() {
            Object.values(highlights).forEach(el => {
                el.style.display = 'none';
            });
        }

        // 显示指定高亮
        function showHighlight(keys) {
            hideAllHighlights();
            if (Array.isArray(keys)) {
                keys.forEach(key => {
                    if (highlights[key]) {
                        highlights[key].style.display = 'block';
                    }
                });
            } else if (keys && highlights[keys]) {
                highlights[keys].style.display = 'block';
            }
        }

        // 更新坐标显示
        function updateCoordDisplay(text) {
            coordDisplay.textContent = text;
        }

        // 更新步骤指示器
        function updateStepIndicator() {
            stepIndicator.textContent = `步骤 ${currentStep}/${totalSteps}`;
            btnPrev.disabled = currentStep === 0;
            btnNext.disabled = currentStep === totalSteps;
        }

        // 执行步骤
        function goToStep(step) {
            currentStep = step;
            
            switch(step) {
                case 0:
                    hideAllHighlights();
                    arrowsGroup.style.display = 'none';
                    updateCoordDisplay('请选择下一步查看点位坐标');
                    break;
                case 1:
                    showHighlight('jia');
                    arrowsGroup.style.display = 'none';
                    updateCoordDisplay('甲：(0°, 66.5°N)');
                    break;
                case 2:
                    showHighlight('yi');
                    arrowsGroup.style.display = 'none';
                    updateCoordDisplay('乙：(20°W, 23.5°N)');
                    break;
                case 3:
                    showHighlight('bing');
                    arrowsGroup.style.display = 'none';
                    updateCoordDisplay('丙：(0°, 0°)');
                    break;
                case 4:
                    showHighlight('ding');
                    arrowsGroup.style.display = 'none';
                    updateCoordDisplay('丁：(20°E, 23.5°S)');
                    break;
                case 5:
                    showHighlight(['E', 'F']);
                    arrowsGroup.style.display = 'none';
                    updateCoordDisplay('E：(20°E, 0°)  F：(40°E, 66.5°N)');
                    break;
                case 6:
                    showHighlight(['E', 'F']);
                    arrowsGroup.style.display = 'block';
                    updateCoordDisplay('方向指示：上北下南，左西右东');
                    break;
            }
            
            updateStepIndicator();
        }

        // 按钮事件
        btnPrev.addEventListener('click', function() {
            if (currentStep > 0) {
                goToStep(currentStep - 1);
            }
        });

        btnNext.addEventListener('click', function() {
            if (currentStep < totalSteps) {
                goToStep(currentStep + 1);
            }
        });

        // 初始化
        updateStepIndicator();
    })();
    </script>
