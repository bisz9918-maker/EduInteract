    <script>
    (function() {
        // Scene 1 实现 - 斜经纬网方向判断

        // 状态变量
        let currentStep = 0;
        const totalSteps = 5;

        // DOM 元素引用
        const btnPrev = document.getElementById('scene1-btn-prev');
        const btnNext = document.getElementById('scene1-btn-next');
        const stepIndicator = document.getElementById('scene1-step-indicator');

        // 各步骤需要显示的元素
        const stepElements = {
            0: [],  // 初始状态：只显示基础网格和点
            1: ['scene1-label-a-coord'],  // 显示 A 点坐标
            2: ['scene1-label-b-coord'],  // 显示 B 点坐标
            3: ['scene1-ref-lines'],  // 显示参考线
            4: ['scene1-direction-arrow'],  // 显示方向箭头
            5: ['scene1-conclusion']  // 显示结论
        };

        // 显示元素函数
        function showElement(id) {
            const el = document.getElementById(id);
            if (el) {
                el.classList.remove('scene1-hidden');
                el.classList.add('scene1-visible');
            }
        }

        // 隐藏元素函数
        function hideElement(id) {
            const el = document.getElementById(id);
            if (el) {
                el.classList.remove('scene1-visible');
                el.classList.add('scene1-hidden');
            }
        }

        // 高亮 A 点
        function highlightPointA() {
            const pointA = document.getElementById('scene1-point-a');
            if (pointA) {
                pointA.setAttribute('r', '8');
                pointA.style.filter = 'drop-shadow(0 0 3px #c62828)';
            }
        }

        // 恢复 A 点
        function restorePointA() {
            const pointA = document.getElementById('scene1-point-a');
            if (pointA) {
                pointA.setAttribute('r', '6');
                pointA.style.filter = 'none';
            }
        }

        // 高亮 B 点
        function highlightPointB() {
            const pointB = document.getElementById('scene1-point-b');
            if (pointB) {
                pointB.setAttribute('r', '8');
                pointB.style.filter = 'drop-shadow(0 0 3px #c62828)';
            }
        }

        // 恢复 B 点
        function restorePointB() {
            const pointB = document.getElementById('scene1-point-b');
            if (pointB) {
                pointB.setAttribute('r', '6');
                pointB.style.filter = 'none';
            }
        }

        // 更新步骤显示
        function updateStep() {
            // 重置所有点状态
            restorePointA();
            restorePointB();

            // 根据当前步骤显示/隐藏元素
            for (let step = 1; step <= totalSteps; step++) {
                const elements = stepElements[step];
                if (step <= currentStep) {
                    elements.forEach(id => showElement(id));
                } else {
                    elements.forEach(id => hideElement(id));
                }
            }

            // 特殊步骤处理
            if (currentStep >= 1) {
                highlightPointA();
            }
            if (currentStep >= 2) {
                restorePointA();
                highlightPointB();
            }

            // 更新按钮状态
            btnPrev.disabled = currentStep === 0;
            btnNext.disabled = currentStep === totalSteps;

            // 更新步骤指示器
            stepIndicator.textContent = `步骤 ${currentStep}/${totalSteps}`;
        }

        // 事件监听
        btnPrev.addEventListener('click', function() {
            if (currentStep > 0) {
                currentStep--;
                updateStep();
            }
        });

        btnNext.addEventListener('click', function() {
            if (currentStep < totalSteps) {
                currentStep++;
                updateStep();
            }
        });

        // 初始化
        updateStep();
    })();
    </script>
