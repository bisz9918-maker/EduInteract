    <script>
    (function() {
        // Scene 1 实现 - 电路结构及电表测量对象识别
        
        // 布局坐标表参考：
        // | 元素 | 位置 | 备注 |
        // |------|------|------|
        // | 电源 | (370-390, 385-415) | 底部中央 |
        // | 开关 S | (520-580, 400) | 底部右侧 |
        // | R₁ | (100-120, 250-350) | 左侧竖直 |
        // | R₂ | (250-550, 100-120) | 顶部水平 |
        // | 滑片 P | (400, 60-110) | R₂上方 |
        // | 电流表 A | (620, 100) | 右上角 |
        // | V₁ | (400, 40) | 顶部中央 |
        // | V₂ | (400, 460) | 底部中央 |

        let currentStep = 0;
        const totalSteps = 6;

        const btnPrev = document.getElementById('scene1-btn-prev');
        const btnNext = document.getElementById('scene1-btn-next');
        const stepIndicator = document.getElementById('scene1-step-indicator');

        // 各步骤需要显示的元素
        const stepConfig = {
            0: { // 初始状态
                series: false,
                current: false,
                v1Box: false,
                v2Box: false,
                arrow: false,
                labelSeries: false,
                labelTotalCurrent: false,
                labelMeasureR2: false,
                labelMeasurePower: false,
                labelMoveLeft: false,
                labelI: false,
                labelU1: false,
                labelU2: false
            },
            1: { // 高亮串联
                series: true,
                current: false,
                v1Box: false,
                v2Box: false,
                arrow: false,
                labelSeries: true,
                labelTotalCurrent: false,
                labelMeasureR2: false,
                labelMeasurePower: false,
                labelMoveLeft: false,
                labelI: false,
                labelU1: false,
                labelU2: false
            },
            2: { // 高亮电流路径
                series: true,
                current: true,
                v1Box: false,
                v2Box: false,
                arrow: false,
                labelSeries: true,
                labelTotalCurrent: true,
                labelMeasureR2: false,
                labelMeasurePower: false,
                labelMoveLeft: false,
                labelI: true,
                labelU1: false,
                labelU2: false
            },
            3: { // 高亮 V₁测量
                series: true,
                current: true,
                v1Box: true,
                v2Box: false,
                arrow: false,
                labelSeries: true,
                labelTotalCurrent: true,
                labelMeasureR2: true,
                labelMeasurePower: false,
                labelMoveLeft: false,
                labelI: true,
                labelU1: true,
                labelU2: false
            },
            4: { // 高亮 V₂测量
                series: true,
                current: true,
                v1Box: true,
                v2Box: true,
                arrow: false,
                labelSeries: true,
                labelTotalCurrent: true,
                labelMeasureR2: true,
                labelMeasurePower: true,
                labelMoveLeft: false,
                labelI: true,
                labelU1: true,
                labelU2: true
            },
            5: { // 显示滑片移动
                series: true,
                current: true,
                v1Box: true,
                v2Box: true,
                arrow: true,
                labelSeries: true,
                labelTotalCurrent: true,
                labelMeasureR2: true,
                labelMeasurePower: true,
                labelMoveLeft: true,
                labelI: true,
                labelU1: true,
                labelU2: true
            }
        };

        function updateDisplay() {
            const config = stepConfig[currentStep];

            // 更新高亮元素
            document.getElementById('scene1-highlight-series').style.opacity = config.series ? '1' : '0';
            document.getElementById('scene1-highlight-current').style.opacity = config.current ? '1' : '0';
            
            const v1Box = document.getElementById('scene1-highlight-v1-box');
            if (config.v1Box) {
                v1Box.classList.add('scene1-visible');
            } else {
                v1Box.classList.remove('scene1-visible');
            }

            const v2Box = document.getElementById('scene1-highlight-v2-box');
            if (config.v2Box) {
                v2Box.classList.add('scene1-visible');
            } else {
                v2Box.classList.remove('scene1-visible');
            }

            const arrow = document.getElementById('scene1-arrow-left');
            if (config.arrow) {
                arrow.classList.add('scene1-visible');
            } else {
                arrow.classList.remove('scene1-visible');
            }

            // 更新额外标签
            const extraLabels = [
                { id: 'scene1-label-series', show: config.labelSeries },
                { id: 'scene1-label-total-current', show: config.labelTotalCurrent },
                { id: 'scene1-label-measure-r2', show: config.labelMeasureR2 },
                { id: 'scene1-label-measure-power', show: config.labelMeasurePower },
                { id: 'scene1-label-move-left', show: config.labelMoveLeft },
                { id: 'scene1-label-i', show: config.labelI },
                { id: 'scene1-label-u1', show: config.labelU1 },
                { id: 'scene1-label-u2', show: config.labelU2 }
            ];

            extraLabels.forEach(item => {
                const el = document.getElementById(item.id);
                if (item.show) {
                    el.classList.add('scene1-visible');
                } else {
                    el.classList.remove('scene1-visible');
                }
            });

            // 更新步骤指示器
            stepIndicator.textContent = `步骤 ${currentStep + 1}/${totalSteps}`;

            // 更新按钮状态
            btnPrev.disabled = currentStep === 0;
            btnNext.disabled = currentStep === totalSteps - 1;
        }

        function goToStep(step) {
            if (step < 0 || step >= totalSteps) return;
            currentStep = step;
            updateDisplay();
        }

        btnPrev.addEventListener('click', () => {
            if (currentStep > 0) {
                goToStep(currentStep - 1);
            }
        });

        btnNext.addEventListener('click', () => {
            if (currentStep < totalSteps - 1) {
                goToStep(currentStep + 1);
            }
        });

        // 初始化显示
        updateDisplay();
    })();
    </script>
