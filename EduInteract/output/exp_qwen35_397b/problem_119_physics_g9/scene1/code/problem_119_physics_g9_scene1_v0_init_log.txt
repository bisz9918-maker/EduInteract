    <script>
    (function() {
        // Scene 1: 并联电路分析
        
        const currentState = { step: 0 };
        const totalSteps = 3;
        
        // 获取 DOM 元素
        const btnPrev = document.getElementById('scene1-btn-prev-group');
        const btnNext = document.getElementById('scene1-btn-next-group');
        const stepIndicator = document.getElementById('scene1-step-indicator');
        const explanation = document.getElementById('scene1-explanation');
        
        // 高亮区域
        const highlightL1 = document.getElementById('scene1-highlight-L1');
        const highlightL2 = document.getElementById('scene1-highlight-L2');
        
        // 电流箭头
        const currentI1 = document.getElementById('scene1-current-I1');
        const currentI2 = document.getElementById('scene1-current-I2');
        
        // 滑片移动箭头
        const slideArrow = document.getElementById('scene1-slide-arrow');
        
        // L₂ 发光效果
        const bulbL2Glow = document.getElementById('scene1-bulb-L2-glow');
        
        // 电压标注
        const labelU = document.getElementById('scene1-label-U');
        
        // 解释文字
        const explanations = [
            "点击「下一步」开始分析",
            "L₁ 支路电压 = 电源电压 U（不变）",
            "滑片 P 右移 → L₂ 支路电阻减小 → 电流 I₂ 增大",
            "L₁ 亮度不变，L₂ 变亮"
        ];
        
        // 更新显示状态
        function updateDisplay() {
            // 重置所有元素
            highlightL1.setAttribute('opacity', '0');
            highlightL2.setAttribute('opacity', '0');
            currentI1.setAttribute('opacity', '0');
            currentI2.setAttribute('opacity', '0');
            slideArrow.setAttribute('opacity', '0');
            bulbL2Glow.setAttribute('opacity', '0');
            labelU.setAttribute('opacity', '0');
            
            // 根据步骤显示相应元素
            if (currentState.step >= 1) {
                // 步骤 1: L₁ 支路高亮
                highlightL1.setAttribute('opacity', '0.2');
                currentI1.setAttribute('opacity', '1');
                labelU.setAttribute('opacity', '1');
            }
            
            if (currentState.step >= 2) {
                // 步骤 2: L₂ 支路高亮 + 滑片移动箭头
                highlightL2.setAttribute('opacity', '0.2');
                currentI2.setAttribute('opacity', '1');
                slideArrow.setAttribute('opacity', '1');
            }
            
            if (currentState.step >= 3) {
                // 步骤 3: L₂ 发光动画
                bulbL2Glow.setAttribute('opacity', '0.6');
                // 启动发光动画
                animateGlow();
            }
            
            // 更新解释文字
            explanation.textContent = explanations[currentState.step];
            explanation.setAttribute('opacity', '1');
            
            // 更新步骤指示器
            stepIndicator.textContent = `步骤 ${currentState.step}/${totalSteps}`;
            
            // 更新按钮状态
            updateButtons();
        }
        
        // 更新按钮状态
        function updateButtons() {
            const prevRect = btnPrev.querySelector('rect');
            const prevText = btnPrev.querySelector('text');
            const nextRect = btnNext.querySelector('rect');
            const nextText = btnNext.querySelector('text');
            
            if (currentState.step === 0) {
                prevRect.setAttribute('fill', '#e0e0e0');
                prevText.setAttribute('fill', '#999');
            } else {
                prevRect.setAttribute('fill', '#e0e0e0');
                prevText.setAttribute('fill', '#333');
            }
            
            if (currentState.step >= totalSteps) {
                nextRect.setAttribute('fill', '#e0e0e0');
                nextText.setAttribute('fill', '#999');
            } else {
                nextRect.setAttribute('fill', '#4CAF50');
                nextText.setAttribute('fill', '#fff');
            }
        }
        
        // L₂ 发光动画
        let glowAnimationId = null;
        function animateGlow() {
            if (glowAnimationId) {
                cancelAnimationFrame(glowAnimationId);
            }
            
            let opacity = 0.6;
            let increasing = true;
            
            function animate() {
                if (currentState.step < 3) return;
                
                if (increasing) {
                    opacity += 0.02;
                    if (opacity >= 1.0) {
                        opacity = 1.0;
                        increasing = false;
                    }
                } else {
                    opacity -= 0.02;
                    if (opacity <= 0.6) {
                        opacity = 0.6;
                        increasing = true;
                    }
                }
                
                bulbL2Glow.setAttribute('opacity', opacity.toString());
                glowAnimationId = requestAnimationFrame(animate);
            }
            
            animate();
        }
        
        // 按钮点击事件
        btnPrev.addEventListener('click', function() {
            if (currentState.step > 0) {
                currentState.step--;
                if (currentState.step < 3 && glowAnimationId) {
                    cancelAnimationFrame(glowAnimationId);
                }
                updateDisplay();
            }
        });
        
        btnNext.addEventListener('click', function() {
            if (currentState.step < totalSteps) {
                currentState.step++;
                updateDisplay();
            }
        });
        
        // 初始化显示
        updateDisplay();
        
    })();
    </script>
