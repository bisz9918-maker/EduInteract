    <script>
    (function() {
        // Scene 1 实现：整体法与隔离法受力分析
        
        // 状态变量
        let currentStep = 0;
        const totalSteps = 4;
        
        // 元素分组：每步需要显示的元素 ID 列表
        const stepElements = {
            0: [],  // 初始状态：只显示基础元素（物块、水平面、外力 F）
            1: ['scene1-frictions', 'scene1-label-f1', 'scene1-highlight', 'scene1-label-ftotal'],
            2: ['scene1-pressure-group'],
            3: ['scene1-conclusion']
        };
        
        // 获取 DOM 元素
        const btnPrev = document.getElementById('scene1-btn-prev');
        const btnNext = document.getElementById('scene1-btn-next');
        const stepIndicator = document.getElementById('scene1-step-indicator');
        
        // 工具函数：显示元素
        function showElement(elementId) {
            const el = document.getElementById(elementId);
            if (el) {
                el.classList.remove('scene1-hidden');
                el.classList.add('scene1-visible');
            }
        }
        
        // 工具函数：隐藏元素
        function hideElement(elementId) {
            const el = document.getElementById(elementId);
            if (el) {
                el.classList.remove('scene1-visible');
                el.classList.add('scene1-hidden');
            }
        }
        
        // 更新步骤显示
        function updateStep() {
            // 隐藏所有步骤特有元素
            for (let step = 1; step < totalSteps; step++) {
                stepElements[step].forEach(id => hideElement(id));
            }
            
            // 显示当前步骤及之前步骤的元素
            for (let step = 1; step <= currentStep; step++) {
                stepElements[step].forEach(id => showElement(id));
            }
            
            // 更新步骤指示器
            stepIndicator.textContent = `步骤 ${currentStep + 1}/${totalSteps}`;
            
            // 更新按钮状态
            btnPrev.disabled = (currentStep === 0);
            btnNext.disabled = (currentStep === totalSteps - 1);
        }
        
        // 事件处理：上一步
        btnPrev.addEventListener('click', function() {
            if (currentStep > 0) {
                currentStep--;
                updateStep();
            }
        });
        
        // 事件处理：下一步
        btnNext.addEventListener('click', function() {
            if (currentStep < totalSteps - 1) {
                currentStep++;
                updateStep();
            }
        });
        
        // 初始化
        updateStep();
    })();
    </script>
