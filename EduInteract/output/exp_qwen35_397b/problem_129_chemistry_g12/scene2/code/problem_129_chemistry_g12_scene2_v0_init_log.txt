    <script>
    (function() {
        // Scene 2 实现：电极反应式推导与验证
        
        let currentStep = 0;
        const totalSteps = 8;
        
        // 获取所有步骤元素
        const stepElements = {
            0: ['scene2-anode-frame', 'scene2-cathode-frame'],
            1: ['scene2-anode-frame', 'scene2-cathode-frame', 'scene2-step1'],
            2: ['scene2-anode-frame', 'scene2-cathode-frame', 'scene2-step1', 'scene2-step2'],
            3: ['scene2-anode-frame', 'scene2-cathode-frame', 'scene2-step1', 'scene2-step2', 'scene2-step3'],
            4: ['scene2-anode-frame', 'scene2-cathode-frame', 'scene2-step1', 'scene2-step2', 'scene2-step3', 'scene2-step4'],
            5: ['scene2-anode-frame', 'scene2-cathode-frame', 'scene2-step1', 'scene2-step2', 'scene2-step3', 'scene2-step4', 'scene2-step5'],
            6: ['scene2-anode-frame', 'scene2-cathode-frame', 'scene2-step6'],
            7: ['scene2-anode-frame', 'scene2-cathode-frame', 'scene2-step6', 'scene2-step7'],
            8: ['scene2-anode-frame', 'scene2-cathode-frame', 'scene2-step6', 'scene2-step7', 'scene2-step8']
        };
        
        // 显示/隐藏元素工具函数
        function showElement(id) {
            const el = document.getElementById(id);
            if (el) {
                el.classList.remove('scene2-hidden');
                el.classList.add('scene2-visible');
            }
        }
        
        function hideElement(id) {
            const el = document.getElementById(id);
            if (el) {
                el.classList.remove('scene2-visible');
                el.classList.add('scene2-hidden');
            }
        }
        
        // 更新步骤显示
        function updateStep() {
            // 隐藏所有步骤元素
            for (let i = 1; i <= 8; i++) {
                const el = document.getElementById('scene2-step' + i);
                if (el) {
                    el.classList.remove('scene2-visible');
                    el.classList.add('scene2-hidden');
                }
            }
            
            // 显示当前步骤及之前的所有元素
            const visibleElements = stepElements[currentStep];
            if (visibleElements) {
                visibleElements.forEach(id => showElement(id));
            }
            
            // 更新步骤指示器
            const indicator = document.getElementById('scene2-step-indicator');
            if (indicator) {
                indicator.textContent = '步骤 ' + currentStep + '/' + totalSteps;
            }
            
            // 更新按钮状态
            const prevBtnBg = document.getElementById('scene2-prev-btn-bg');
            const nextBtnBg = document.getElementById('scene2-next-btn-bg');
            const prevText = document.getElementById('scene2-prev-text');
            const nextText = document.getElementById('scene2-next-text');
            
            if (prevBtnBg && prevText) {
                if (currentStep === 0) {
                    prevBtnBg.setAttribute('fill', '#ccc');
                    prevText.setAttribute('fill', '#999');
                } else {
                    prevBtnBg.setAttribute('fill', '#1565c0');
                    prevText.setAttribute('fill', '#fff');
                }
            }
            
            if (nextBtnBg && nextText) {
                if (currentStep === totalSteps) {
                    nextBtnBg.setAttribute('fill', '#ccc');
                    nextText.setAttribute('fill', '#999');
                } else {
                    nextBtnBg.setAttribute('fill', '#1565c0');
                    nextText.setAttribute('fill', '#fff');
                }
            }
        }
        
        // 点击事件处理
        function handlePrevClick() {
            if (currentStep > 0) {
                currentStep--;
                updateStep();
            }
        }
        
        function handleNextClick() {
            if (currentStep < totalSteps) {
                currentStep++;
                updateStep();
            }
        }
        
        // 绑定按钮点击事件
        const prevBtnBg = document.getElementById('scene2-prev-btn-bg');
        const nextBtnBg = document.getElementById('scene2-next-btn-bg');
        
        if (prevBtnBg) {
            prevBtnBg.style.cursor = 'pointer';
            prevBtnBg.addEventListener('click', handlePrevClick);
        }
        
        if (nextBtnBg) {
            nextBtnBg.style.cursor = 'pointer';
            nextBtnBg.addEventListener('click', handleNextClick);
        }
        
        // 初始化
        updateStep();
    })();
    </script>
