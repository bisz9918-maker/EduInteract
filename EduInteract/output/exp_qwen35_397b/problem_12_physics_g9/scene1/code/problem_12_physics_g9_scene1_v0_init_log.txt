    <script>
    (function() {
        // Scene 1: 弹簧测力计受力分析
        const totalSteps = 5;
        let currentStep = 0;

        // 获取 DOM 元素
        const btnPrev = document.getElementById('scene1-btnPrev');
        const btnNext = document.getElementById('scene1-btnNext');
        const stepIndicator = document.getElementById('scene1-stepIndicator');

        // 元素引用
        const elements = {
            springA: document.getElementById('scene1-springA'),
            arrowLeft: document.getElementById('scene1-arrowLeftGroup'),
            arrowRight: document.getElementById('scene1-arrowRightGroup'),
            labelLeftForce: document.getElementById('scene1-labelLeftForce'),
            labelRightForce: document.getElementById('scene1-labelRightForce'),
            readoutA: document.getElementById('scene1-readoutAGroup'),
            interactionForce: document.getElementById('scene1-interactionForceGroup'),
            readoutB: document.getElementById('scene1-readoutBGroup'),
            answerBox: document.getElementById('scene1-answerBoxGroup')
        };

        // 显示/隐藏工具函数
        function show(el) {
            if (el) {
                el.classList.remove('scene1-hidden');
                el.classList.add('scene1-visible');
                el.style.display = 'block';
                el.style.opacity = '1';
            }
        }

        function hide(el) {
            if (el) {
                el.classList.remove('scene1-visible');
                el.classList.add('scene1-hidden');
                el.style.display = 'none';
                el.style.opacity = '0';
            }
        }

        // 更新场景状态
        function updateScene() {
            // 重置所有元素
            hide(elements.arrowLeft);
            hide(elements.arrowRight);
            hide(elements.labelLeftForce);
            hide(elements.labelRightForce);
            hide(elements.readoutA);
            hide(elements.interactionForce);
            hide(elements.readoutB);
            hide(elements.answerBox);

            // 根据当前步骤显示相应元素
            if (currentStep >= 1) {
                show(elements.readoutA);
            }
            if (currentStep >= 2) {
                show(elements.interactionForce);
            }
            if (currentStep >= 3) {
                show(elements.readoutB);
            }
            if (currentStep >= 4) {
                show(elements.arrowLeft);
                show(elements.arrowRight);
                show(elements.labelLeftForce);
                show(elements.labelRightForce);
            }
            if (currentStep >= 5) {
                show(elements.answerBox);
            }

            // 更新步骤指示器
            stepIndicator.textContent = `步骤 ${currentStep}/${totalSteps}`;

            // 更新按钮状态
            btnPrev.disabled = currentStep === 0;
            btnNext.disabled = currentStep === totalSteps;
        }

        // 按钮事件处理
        btnPrev.addEventListener('click', function() {
            if (currentStep > 0) {
                currentStep--;
                updateScene();
            }
        });

        btnNext.addEventListener('click', function() {
            if (currentStep < totalSteps) {
                currentStep++;
                updateScene();
            }
        });

        // 初始化
        updateScene();
    })();
    </script>
