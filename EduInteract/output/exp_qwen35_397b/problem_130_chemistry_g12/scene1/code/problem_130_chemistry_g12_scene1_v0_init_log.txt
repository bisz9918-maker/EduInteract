    <script>
    (function() {
        // Scene 1 实现：基团组成及价键规则
        
        // 状态变量
        let currentStep = 0;
        const totalSteps = 5;
        
        // 元素引用
        const stepGroups = {
            1: document.getElementById('scene1-step1-group'),
            2: document.getElementById('scene1-step2-group'),
            3: document.getElementById('scene1-step3-group'),
            4: document.getElementById('scene1-step4-group'),
            5: document.getElementById('scene1-step5-group')
        };
        
        const btnPrev = document.getElementById('scene1-btn-prev-bg');
        const btnPrevText = document.getElementById('scene1-btn-prev-text');
        const btnNext = document.getElementById('scene1-btn-next-bg');
        const btnNextText = document.getElementById('scene1-btn-next-text');
        const stepIndicator = document.getElementById('scene1-step-indicator');
        
        // 工具函数：显示元素
        function showElement(el) {
            if (el) {
                el.classList.add('scene1-visible');
                el.style.opacity = '1';
            }
        }
        
        // 工具函数：隐藏元素
        function hideElement(el) {
            if (el) {
                el.classList.remove('scene1-visible');
                el.style.opacity = '0';
            }
        }
        
        // 更新按钮状态
        function updateButtons() {
            if (currentStep === 0) {
                btnPrev.style.fill = '#bdbdbd';
                btnPrev.style.stroke = '#9e9e9e';
                btnPrevText.classList.add('disabled');
            } else {
                btnPrev.style.fill = '#1976d2';
                btnPrev.style.stroke = '#1565c0';
                btnPrevText.classList.remove('disabled');
            }
            
            if (currentStep === totalSteps) {
                btnNext.style.fill = '#bdbdbd';
                btnNext.style.stroke = '#9e9e9e';
                btnNextText.classList.add('disabled');
            } else {
                btnNext.style.fill = '#1976d2';
                btnNext.style.stroke = '#1565c0';
                btnNextText.classList.remove('disabled');
            }
            
            stepIndicator.textContent = '步骤 ' + currentStep + '/' + totalSteps;
        }
        
        // 更新显示状态
        function updateDisplay() {
            // 根据当前步骤显示/隐藏各步骤组
            for (let step = 1; step <= totalSteps; step++) {
                if (step <= currentStep) {
                    showElement(stepGroups[step]);
                } else {
                    hideElement(stepGroups[step]);
                }
            }
            updateButtons();
        }
        
        // 上一步
        function prevStep() {
            if (currentStep > 0) {
                currentStep--;
                updateDisplay();
            }
        }
        
        // 下一步
        function nextStep() {
            if (currentStep < totalSteps) {
                currentStep++;
                updateDisplay();
            }
        }
        
        // 事件绑定
        btnPrev.addEventListener('click', prevStep);
        btnNext.addEventListener('click', nextStep);
        
        // 初始化
        updateDisplay();
        
    })();
    </script>
