    <script>
    (function() {
        // Scene 3: C₅H₁₁Cl 卤代烃结构异构体展示
        
        let currentStep = 0;
        const totalSteps = 5;
        
        const btnPrev = document.getElementById('scene3-btn-prev');
        const btnNext = document.getElementById('scene3-btn-next');
        const stepIndicator = document.getElementById('scene3-step-indicator');
        
        const struct1 = document.getElementById('scene3-struct1');
        const struct2 = document.getElementById('scene3-struct2');
        const struct3 = document.getElementById('scene3-struct3');
        const struct4 = document.getElementById('scene3-struct4');
        const answer = document.getElementById('scene3-answer');
        
        function updateDisplay() {
            // 更新步骤指示器
            stepIndicator.textContent = '步骤 ' + currentStep + '/' + totalSteps;
            
            // 更新按钮状态
            btnPrev.disabled = (currentStep === 0);
            btnNext.disabled = (currentStep === totalSteps);
            if (currentStep === totalSteps) {
                btnNext.textContent = '完成';
            } else {
                btnNext.textContent = '下一步 ▶';
            }
            
            // 控制各结构可见性
            if (currentStep >= 1) {
                struct1.classList.add('struct-visible');
            } else {
                struct1.classList.remove('struct-visible');
            }
            
            if (currentStep >= 2) {
                struct2.classList.add('struct-visible');
            } else {
                struct2.classList.remove('struct-visible');
            }
            
            if (currentStep >= 3) {
                struct3.classList.add('struct-visible');
            } else {
                struct3.classList.remove('struct-visible');
            }
            
            if (currentStep >= 4) {
                struct4.classList.add('struct-visible');
            } else {
                struct4.classList.remove('struct-visible');
            }
            
            if (currentStep >= 5) {
                answer.classList.add('struct-visible');
            } else {
                answer.classList.remove('struct-visible');
            }
        }
        
        btnPrev.addEventListener('click', function() {
            if (currentStep > 0) {
                currentStep--;
                updateDisplay();
            }
        });
        
        btnNext.addEventListener('click', function() {
            if (currentStep < totalSteps) {
                currentStep++;
                updateDisplay();
            }
        });
        
        // 初始化显示
        updateDisplay();
    })();
    </script>
