    <script>
    (function() {
        // Scene 2 实现：H₂与 O₂反应——两种过量情况分析

        // 状态变量：0 = H₂过量，1 = O₂过量
        let currentState = 0;

        // 获取 DOM 元素
        const state1Group = document.getElementById('scene2-state1-group');
        const state2Group = document.getElementById('scene2-state2-group');
        const btnH2Excess = document.getElementById('scene2-btn-h2-excess');
        const btnO2Excess = document.getElementById('scene2-btn-o2-excess');

        // 切换状态函数
        function switchState(newState) {
            if (newState === currentState) return;

            currentState = newState;

            if (currentState === 0) {
                // 显示状态 1（H₂过量），隐藏状态 2
                state1Group.classList.remove('scene2-hidden');
                state2Group.classList.add('scene2-hidden');

                // 按钮样式切换
                btnH2Excess.classList.remove('scene2-btn-normal');
                btnH2Excess.classList.add('scene2-btn-active');
                btnO2Excess.classList.remove('scene2-btn-active');
                btnO2Excess.classList.add('scene2-btn-normal');
            } else {
                // 显示状态 2（O₂过量），隐藏状态 1
                state1Group.classList.add('scene2-hidden');
                state2Group.classList.remove('scene2-hidden');

                // 按钮样式切换
                btnH2Excess.classList.remove('scene2-btn-active');
                btnH2Excess.classList.add('scene2-btn-normal');
                btnO2Excess.classList.remove('scene2-btn-normal');
                btnO2Excess.classList.add('scene2-btn-active');
            }
        }

        // 绑定按钮点击事件
        btnH2Excess.addEventListener('click', function() {
            switchState(0);
        });

        btnO2Excess.addEventListener('click', function() {
            switchState(1);
        });

        // 初始状态：显示 H₂过量情况
        switchState(0);
    })();
    </script>
