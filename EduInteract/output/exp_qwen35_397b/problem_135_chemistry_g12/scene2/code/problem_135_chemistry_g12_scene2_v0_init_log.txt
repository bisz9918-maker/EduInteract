    <script>
    (function() {
        // 离子颜色配置
        const ION_COLORS = {
            'H⁺': '#c62828',
            'OH⁻': '#1565c0',
            'CH₃COO⁻': '#2e7d32',
            'NH₄⁺': '#7b1fa2',
            'Ag⁺': '#ff9800',
            'NO₃⁻': '#00bcd4',
            'Cl⁻': '#607d8b',
            'Ba²⁺': '#e91e63',
            'SO₄²⁻': '#3f51b5',
            'Na⁺': '#ffa726'
        };
        
        // 四个选项的数据
        const optionsData = [
            {
                // 选项 A: 醋酸 + 氨水
                equation: 'CH₃COOH + NH₃·H₂O → CH₃COONH₄ + H₂O',
                before: [
                    { ion: 'CH₃COO⁻', concentration: 15 },
                    { ion: 'H⁺', concentration: 15 }
                ],
                after: [
                    { ion: 'CH₃COO⁻', concentration: 70 },
                    { ion: 'NH₄⁺', concentration: 70 }
                ],
                conductivityBefore: { level: '弱', color: '#bdbdbd', glow: false },
                conductivityAfter: { level: '强', color: '#ffd54f', glow: true },
                precipitate: null
            },
            {
                // 选项 B: AgNO₃ + HCl
                equation: 'AgNO₃ + HCl → AgCl↓ + HNO₃',
                before: [
                    { ion: 'Ag⁺', concentration: 50 },
                    { ion: 'NO₃⁻', concentration: 50 }
                ],
                after: [
                    { ion: 'H⁺', concentration: 50 },
                    { ion: 'NO₃⁻', concentration: 50 }
                ],
                conductivityBefore: { level: '中', color: '#ffd54f', glow: true },
                conductivityAfter: { level: '中', color: '#ffd54f', glow: true },
                precipitate: { label: 'AgCl↓', color: '#888888' }
            },
            {
                // 选项 C: HCl + NaCl
                equation: 'HCl + NaCl → H⁺ + Cl⁻ + Na⁺ + Cl⁻',
                before: [
                    { ion: 'H⁺', concentration: 50 },
                    { ion: 'Cl⁻', concentration: 50 }
                ],
                after: [
                    { ion: 'H⁺', concentration: 40 },
                    { ion: 'Cl⁻', concentration: 60 },
                    { ion: 'Na⁺', concentration: 40 }
                ],
                conductivityBefore: { level: '中', color: '#ffd54f', glow: true },
                conductivityAfter: { level: '中强', color: '#ffeb3b', glow: true },
                precipitate: null
            },
            {
                // 选项 D: H₂SO₄ + BaCl₂
                equation: 'H₂SO₄ + BaCl₂ → BaSO₄↓ + 2HCl',
                before: [
                    { ion: 'H⁺', concentration: 70 },
                    { ion: 'SO₄²⁻', concentration: 35 },
                    { ion: 'Ba²⁺', concentration: 35 },
                    { ion: 'Cl⁻', concentration: 70 }
                ],
                after: [
                    { ion: 'H⁺', concentration: 70 },
                    { ion: 'Cl⁻', concentration: 70 }
                ],
                conductivityBefore: { level: '强', color: '#ffd54f', glow: true },
                conductivityAfter: { level: '中', color: '#ffeb3b', glow: true },
                precipitate: { label: 'BaSO₄↓', color: '#888888' }
            }
        ];
        
        // 当前状态
        let currentState = 0;
        
        // 获取 DOM 元素
        const equationEl = document.getElementById('scene2-equation');
        const barBeforeEl = document.getElementById('scene2-bar-before');
        const barAfterEl = document.getElementById('scene2-bar-after');
        const precipitateEl = document.getElementById('scene2-precipitate');
        const legendEl = document.getElementById('scene2-legend');
        const bulbBeforeEl = document.getElementById('scene2-bulb-before');
        const bulbAfterEl = document.getElementById('scene2-bulb-after');
        const condTextBeforeEl = document.getElementById('scene2-cond-text-before');
        const condTextAfterEl = document.getElementById('scene2-cond-text-after');
        
        // 渲染柱状图
        function renderBarChart(container, ions, isBefore) {
            container.innerHTML = '';
            const baseX = isBefore ? 150 : 550;
            const barWidth = 100;
            const maxHeight = 160;
            const maxConcentration = 100;
            const labelX = isBefore ? 260 : 660;
            
            let currentY = 310; // 从底部开始
            
            ions.forEach((item, index) => {
                const height = (item.concentration / maxConcentration) * maxHeight;
                const y = currentY - height;
                const color = ION_COLORS[item.ion] || '#999999';
                
                // 绘制矩形
                const rect = document.createElementNS('http://www.w3.org/2000/svg', 'rect');
                rect.setAttribute('x', baseX);
                rect.setAttribute('y', y);
                rect.setAttribute('width', barWidth);
                rect.setAttribute('height', height);
                rect.setAttribute('fill', color);
                rect.setAttribute('opacity', '0.85');
                rect.setAttribute('class', 'scene2-bar-rect');
                rect.setAttribute('rx', '2');
                container.appendChild(rect);
                
                // 绘制离子标签
                const text = document.createElementNS('http://www.w3.org/2000/svg', 'text');
                text.setAttribute('x', labelX);
                text.setAttribute('y', y + height / 2 + 5);
                text.setAttribute('font-size', '12px');
                text.setAttribute('fill', color);
                text.setAttribute('font-weight', 'bold');
                text.textContent = item.ion;
                container.appendChild(text);
                
                currentY = y;
            });
        }
        
        // 渲染沉淀符号
        function renderPrecipitate(precipitate) {
            precipitateEl.innerHTML = '';
            if (precipitate) {
                // 沉淀符号在反应后柱状图顶部
                const circle = document.createElementNS('http://www.w3.org/2000/svg', 'circle');
                circle.setAttribute('cx', '600');
                circle.setAttribute('cy', '125');
                circle.setAttribute('r', '10');
                circle.setAttribute('fill', precipitate.color);
                precipitateEl.appendChild(circle);
                
                const text = document.createElementNS('http://www.w3.org/2000/svg', 'text');
                text.setAttribute('x', '600');
                text.setAttribute('y', '110');
                text.setAttribute('text-anchor', 'middle');
                text.setAttribute('font-size', '11px');
                text.setAttribute('fill', precipitate.color);
                text.setAttribute('font-weight', 'bold');
                text.textContent = '↓';
                precipitateEl.appendChild(text);
                
                const label = document.createElementNS('http://www.w3.org/2000/svg', 'text');
                label.setAttribute('x', '600');
                label.setAttribute('y', '155');
                label.setAttribute('text-anchor', 'middle');
                label.setAttribute('font-size', '11px');
                label.setAttribute('fill', '#666666');
                label.textContent = precipitate.label;
                precipitateEl.appendChild(label);
            }
        }
        
        // 渲染离子图例
        function renderLegend(ions) {
            legendEl.innerHTML = '';
            const allIons = [];
            ions.forEach(item => {
                if (!allIons.includes(item.ion)) {
                    allIons.push(item.ion);
                }
            });
            
            let startX = 20;
            let startY = 370;
            const itemsPerRow = 5;
            
            allIons.forEach((ion, index) => {
                const row = Math.floor(index / itemsPerRow);
                const col = index % itemsPerRow;
                const x = startX + col * 65;
                const y = startY + row * 25;
                const color = ION_COLORS[ion] || '#999999';
                
                // 色块
                const rect = document.createElementNS('http://www.w3.org/2000/svg', 'rect');
                rect.setAttribute('x', x);
                rect.setAttribute('y', y - 10);
                rect.setAttribute('width', '14');
                rect.setAttribute('height', '14');
                rect.setAttribute('fill', color);
                rect.setAttribute('rx', '2');
                legendEl.appendChild(rect);
                
                // 标签
                const text = document.createElementNS('http://www.w3.org/2000/svg', 'text');
                text.setAttribute('x', x + 18);
                text.setAttribute('y', y);
                text.setAttribute('font-size', '11px');
                text.setAttribute('fill', '#333333');
                text.textContent = ion;
                legendEl.appendChild(text);
            });
        }
        
        // 更新导电性指示
        function updateConductivity(before, after) {
            bulbBeforeEl.setAttribute('fill', before.color);
            bulbAfterEl.setAttribute('fill', after.color);
            
            if (before.glow) {
                bulbBeforeEl.classList.add('scene2-bulb-glow');
            } else {
                bulbBeforeEl.classList.remove('scene2-bulb-glow');
            }
            
            if (after.glow) {
                bulbAfterEl.classList.add('scene2-bulb-glow');
            } else {
                bulbAfterEl.classList.remove('scene2-bulb-glow');
            }
            
            condTextBeforeEl.textContent = before.level;
            condTextAfterEl.textContent = after.level;
        }
        
        // 更新按钮状态
        function updateButtons(activeIndex) {
            ['a', 'b', 'c', 'd'].forEach((letter, index) => {
                const btn = document.getElementById(`scene2-btn-${letter}`);
                const rect = btn.querySelector('rect');
                const text = btn.querySelector('text');
                
                if (index === activeIndex) {
                    btn.classList.add('scene2-btn-active');
                    rect.setAttribute('fill', '#e3f2fd');
                    rect.setAttribute('stroke', '#1976d2');
                    rect.setAttribute('stroke-width', '3');
                    text.setAttribute('fill', '#1976d2');
                } else {
                    btn.classList.remove('scene2-btn-active');
                    rect.setAttribute('fill', '#f5f5f5');
                    rect.setAttribute('stroke', '#999999');
                    rect.setAttribute('stroke-width', '1');
                    text.setAttribute('fill', '#666666');
                }
            });
        }
        
        // 更新场景
        function updateScene(optionIndex) {
            const data = optionsData[optionIndex];
            
            // 更新方程式
            equationEl.textContent = data.equation;
            
            // 更新柱状图
            renderBarChart(barBeforeEl, data.before, true);
            renderBarChart(barAfterEl, data.after, false);
            
            // 更新沉淀
            renderPrecipitate(data.precipitate);
            
            // 更新导电性
            updateConductivity(data.conductivityBefore, data.conductivityAfter);
            
            // 更新按钮
            updateButtons(optionIndex);
            
            // 更新图例
            const allIons = [...data.before, ...data.after];
            renderLegend(allIons);
            
            currentState = optionIndex;
        }
        
        // 绑定按钮事件
        document.getElementById('scene2-btn-a').addEventListener('click', () => updateScene(0));
        document.getElementById('scene2-btn-b').addEventListener('click', () => updateScene(1));
        document.getElementById('scene2-btn-c').addEventListener('click', () => updateScene(2));
        document.getElementById('scene2-btn-d').addEventListener('click', () => updateScene(3));
        
        // 初始化场景
        updateScene(0);
    })();
    </script>
</body>
