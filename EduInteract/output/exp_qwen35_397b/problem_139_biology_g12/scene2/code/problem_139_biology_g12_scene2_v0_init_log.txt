    <script>
    (function() {
        // Scene 2: 基因型与表现型对应关系
        // 布局坐标表（实现对照）：
        // 卡片 1 (A_B_): (40,40), 340×180 | 卡片 2 (A_bb): (420,40), 340×180
        // 卡片 3 (aaB_): (40,260), 340×180 | 卡片 4 (aabb): (420,260), 340×180
        // 按钮区: y=450, 4 个按钮各 150×35

        let currentState = 0; // 0-3 对应 4 个卡片

        // 获取 DOM 元素
        const cards = [
            document.getElementById('scene2-card-1'),
            document.getElementById('scene2-card-2'),
            document.getElementById('scene2-card-3'),
            document.getElementById('scene2-card-4')
        ];

        const buttons = [
            document.getElementById('scene2-btn-1'),
            document.getElementById('scene2-btn-2'),
            document.getElementById('scene2-btn-3'),
            document.getElementById('scene2-btn-4')
        ];

        // 切换状态函数
        function switchState(cardIndex) {
            currentState = cardIndex;

            // 更新卡片状态
            cards.forEach((card, i) => {
                if (i === cardIndex) {
                    card.classList.remove('card-inactive');
                    card.classList.add('card-active');
                } else {
                    card.classList.remove('card-active');
                    card.classList.add('card-inactive');
                }
            });

            // 更新按钮状态
            buttons.forEach((btn, i) => {
                if (i === cardIndex) {
                    btn.classList.remove('btn-inactive');
                    btn.classList.add('btn-active');
                } else {
                    btn.classList.remove('btn-active');
                    btn.classList.add('btn-inactive');
                }
            });
        }

        // 绑定按钮点击事件
        buttons.forEach((btn, index) => {
            btn.addEventListener('click', () => {
                switchState(index);
            });
        });

        // 初始状态：卡片 1 高亮
        switchState(0);
    })();
    </script>
