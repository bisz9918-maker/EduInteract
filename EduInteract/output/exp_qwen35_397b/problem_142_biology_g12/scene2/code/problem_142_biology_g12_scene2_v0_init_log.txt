    <script>
    (function() {
        // Scene 2 实现：三种黑青色突变体染色体结构对比
        // 布局坐标表（根据题目原图修正）：
        // 突变体甲：x=50~270, y=70~290，染色体对 1(90,110)/(130,110)，染色体对 2(190,155)/(215,155)
        // 突变体乙：x=290~510, y=70~290，染色体对 1(330,110)/(370,110)，染色体对 2(420,110)/(460,110)
        // 突变体丙：x=530~750, y=70~290，染色体对 1(570,110)/(610,110)，染色体对 2(670,155)/(695,155)
        
        let currentState = 'a'; // 当前状态：'a', 'b', 'c'
        
        const mutantA = document.getElementById('scene2-mutant-a');
        const mutantB = document.getElementById('scene2-mutant-b');
        const mutantC = document.getElementById('scene2-mutant-c');
        
        const btnA = document.getElementById('scene2-btn-a');
        const btnB = document.getElementById('scene2-btn-b');
        const btnC = document.getElementById('scene2-btn-c');
        
        // 显示指定突变体
        function showMutant(mutant) {
            // 隐藏所有突变体
            mutantA.style.display = 'none';
            mutantA.style.opacity = '0';
            mutantB.style.display = 'none';
            mutantB.style.opacity = '0';
            mutantC.style.display = 'none';
            mutantC.style.opacity = '0';
            
            // 移除所有按钮高亮
            btnA.classList.remove('scene2-btn-active');
            btnB.classList.remove('scene2-btn-active');
            btnC.classList.remove('scene2-btn-active');
            
            // 显示目标突变体
            if (mutant === 'a') {
                mutantA.style.display = 'block';
                setTimeout(() => { mutantA.style.opacity = '1'; }, 10);
                btnA.classList.add('scene2-btn-active');
            } else if (mutant === 'b') {
                mutantB.style.display = 'block';
                setTimeout(() => { mutantB.style.opacity = '1'; }, 10);
                btnB.classList.add('scene2-btn-active');
            } else if (mutant === 'c') {
                mutantC.style.display = 'block';
                setTimeout(() => { mutantC.style.opacity = '1'; }, 10);
                btnC.classList.add('scene2-btn-active');
            }
            
            currentState = mutant;
        }
        
        // 绑定按钮事件
        btnA.addEventListener('click', function() {
            showMutant('a');
        });
        
        btnB.addEventListener('click', function() {
            showMutant('b');
        });
        
        btnC.addEventListener('click', function() {
            showMutant('c');
        });
        
        // 初始状态：显示突变体甲
        showMutant('a');
    })();
    </script>
