    <script>
    (function() {
        // Scene 3: 人类干预前后能量对比
        let currentState = 0; // 0=干预前, 1=干预后, 2=对比总结
        
        const pieBefore = document.getElementById('scene3-pie-before');
        const pieAfter = document.getElementById('scene3-pie-after');
        const energyBefore = document.getElementById('scene3-energy-before');
        const energyAfter = document.getElementById('scene3-energy-after');
        const arrowGroup = document.getElementById('scene3-arrow-group');
        
        const btnBefore = document.getElementById('scene3-btn-before');
        const btnAfter = document.getElementById('scene3-btn-after');
        const btnCompare = document.getElementById('scene3-btn-compare');
        
        function hideElement(el) {
            el.classList.add('scene3-hidden');
            el.classList.remove('scene3-semi');
        }
        
        function showElement(el, semi = false) {
            el.classList.remove('scene3-hidden');
            if (semi) {
                el.classList.add('scene3-semi');
            } else {
                el.classList.remove('scene3-semi');
            }
        }
        
        function updateButtons(activeIdx) {
            btnBefore.classList.remove('scene3-btn-active');
            btnAfter.classList.remove('scene3-btn-active');
            btnCompare.classList.remove('scene3-btn-active');
            
            if (activeIdx === 0) btnBefore.classList.add('scene3-btn-active');
            else if (activeIdx === 1) btnAfter.classList.add('scene3-btn-active');
            else if (activeIdx === 2) btnCompare.classList.add('scene3-btn-active');
        }
        
        function updateState(state) {
            currentState = state;
            
            if (state === 0) {
                // 干预前状态
                showElement(pieBefore, false);
                hideElement(pieAfter);
                showElement(energyBefore, false);
                hideElement(energyAfter);
                hideElement(arrowGroup);
                updateButtons(0);
            } else if (state === 1) {
                // 干预后状态
                hideElement(pieBefore);
                showElement(pieAfter, false);
                hideElement(energyBefore);
                showElement(energyAfter, false);
                hideElement(arrowGroup);
                updateButtons(1);
            } else if (state === 2) {
                // 对比总结状态
                showElement(pieBefore, true);
                showElement(pieAfter, true);
                showElement(energyBefore, true);
                showElement(energyAfter, true);
                showElement(arrowGroup, false);
                updateButtons(2);
            }
        }
        
        // 按钮事件绑定
        btnBefore.addEventListener('click', function() {
            updateState(0);
        });
        
        btnAfter.addEventListener('click', function() {
            updateState(1);
        });
        
        btnCompare.addEventListener('click', function() {
            updateState(2);
        });
        
        // 初始化
        updateState(0);
    })();
    </script>
