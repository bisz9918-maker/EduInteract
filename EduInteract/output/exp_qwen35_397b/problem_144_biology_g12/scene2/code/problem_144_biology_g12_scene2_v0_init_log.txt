    <script>
    (function() {
        // Scene 2 - 三种遗传假设的 F2 预期比例对比
        
        // 状态数据
        const stateData = {
            0: { // 常染色体显性
                pText: "AA(棒眼♀) × aa(圆眼♂)",
                f1Text: "Aa(全部棒眼)",
                f2Text: "AA : Aa : aa = 1 : 2 : 1",
                f2Phenotype: "表型：棒眼 : 圆眼 = 3 : 1",
                ratioText: "3:1（棒眼:圆眼）",
                ratioColor: "#1565c0",
                isMatch: false,
                btnClass: "scene2-btn-active",
                btnColor: "#1565c0"
            },
            1: { // 常染色体隐性
                pText: "aa(棒眼♀) × AA(圆眼♂)",
                f1Text: "Aa(全部圆眼) — 与 F1 实际结果矛盾！",
                f1Color: "#c62828",
                f2Text: "AA : Aa : aa = 1 : 2 : 1",
                f2Phenotype: "表型：圆眼 : 棒眼 = 3 : 1",
                ratioText: "此假设与 F1 结果矛盾",
                ratioColor: "#c62828",
                isMatch: false,
                btnClass: "scene2-btn-active",
                btnColor: "#c62828"
            },
            2: { // X 染色体显性
                pText: "X^A X^A(棒眼♀) × X^a Y(圆眼♂)",
                f1Text: "X^A X^a(棒眼♀), X^A Y(棒眼♂)",
                f2Text: "X^A X^a : X^a X^a : X^A Y : X^a Y = 1 : 1 : 1 : 1",
                f2Phenotype: "♀棒眼 : ♀圆眼 : ♂棒眼 : ♂圆眼 = 1 : 1 : 1 : 1",
                ratioText: "1:1:1:1（与实验结果匹配）",
                ratioColor: "#2e7d32",
                isMatch: true,
                btnClass: "scene2-btn-match",
                btnColor: "#2e7d32"
            }
        };
        
        let currentState = 0;
        
        // 获取元素
        const btnConstDom = document.getElementById('scene2-btn-const-dom');
        const btnConstRec = document.getElementById('scene2-btn-const-rec');
        const btnXDom = document.getElementById('scene2-btn-x-dom');
        const buttons = [btnConstDom, btnConstRec, btnXDom];
        
        const pText = document.getElementById('scene2-p-text');
        const f1Text = document.getElementById('scene2-f1-text');
        const f2Text = document.getElementById('scene2-f2-text');
        const f2Phenotype = document.getElementById('scene2-f2-phenotype');
        const ratioText = document.getElementById('scene2-ratio-text');
        const matchIndicator = document.getElementById('scene2-match-indicator');
        
        const state1Content = document.getElementById('scene2-state1-content');
        const state2Content = document.getElementById('scene2-state2-content');
        
        // 更新状态函数
        function updateState(newState) {
            if (newState === currentState) return;
            
            const data = stateData[newState];
            
            // 更新按钮样式
            buttons.forEach((btn, idx) => {
                const rect = btn.querySelector('rect');
                const text = btn.querySelector('text');
                
                btn.classList.remove('scene2-btn-active', 'scene2-btn-match');
                
                if (idx === newState) {
                    btn.classList.add(data.btnClass);
                    rect.setAttribute('fill', data.btnColor);
                    text.setAttribute('fill', '#ffffff');
                } else if (idx === 2 && newState !== 2) {
                    // X 显按钮在未选中时，如果是匹配状态显示绿色边框提示
                    rect.setAttribute('fill', '#e0e0e0');
                    text.setAttribute('fill', '#333333');
                } else {
                    rect.setAttribute('fill', '#e0e0e0');
                    text.setAttribute('fill', '#333333');
                }
            });
            
            // 更新主内容
            pText.textContent = data.pText;
            f1Text.textContent = data.f1Text;
            if (data.f1Color) {
                f1Text.setAttribute('fill', data.f1Color);
            } else {
                f1Text.setAttribute('fill', '#333333');
            }
            f2Text.textContent = data.f2Text;
            f2Phenotype.textContent = data.f2Phenotype;
            ratioText.textContent = data.ratioText;
            ratioText.setAttribute('fill', data.ratioColor);
            
            // 更新匹配指示器
            if (data.isMatch) {
                matchIndicator.textContent = '✓';
                matchIndicator.setAttribute('fill', '#2e7d32');
                matchIndicator.nextElementSibling.textContent = '与实验结果';
                matchIndicator.nextElementSibling.nextElementSibling.textContent = '完全匹配';
            } else {
                matchIndicator.textContent = '✗';
                matchIndicator.setAttribute('fill', '#c62828');
                matchIndicator.nextElementSibling.textContent = '与实验结果';
                matchIndicator.nextElementSibling.nextElementSibling.textContent = '不匹配';
            }
            
            // 显示/隐藏状态特定内容
            if (newState === 1) {
                state1Content.classList.remove('scene2-hidden');
                state1Content.classList.add('scene2-visible');
                state2Content.classList.remove('scene2-visible');
                state2Content.classList.add('scene2-hidden');
                // 隐藏默认 F2 内容
                f2Text.classList.add('scene2-hidden');
                f2Phenotype.classList.add('scene2-hidden');
            } else if (newState === 2) {
                state1Content.classList.remove('scene2-visible');
                state1Content.classList.add('scene2-hidden');
                state2Content.classList.remove('scene2-hidden');
                state2Content.classList.add('scene2-visible');
                // 隐藏默认 F2 内容
                f2Text.classList.add('scene2-hidden');
                f2Phenotype.classList.add('scene2-hidden');
            } else {
                state1Content.classList.remove('scene2-visible');
                state1Content.classList.add('scene2-hidden');
                state2Content.classList.remove('scene2-visible');
                state2Content.classList.add('scene2-hidden');
                // 显示默认 F2 内容
                f2Text.classList.remove('scene2-hidden');
                f2Text.classList.add('scene2-visible');
                f2Phenotype.classList.remove('scene2-hidden');
                f2Phenotype.classList.add('scene2-visible');
            }
            
            currentState = newState;
        }
        
        // 绑定按钮点击事件
        btnConstDom.addEventListener('click', () => updateState(0));
        btnConstRec.addEventListener('click', () => updateState(1));
        btnXDom.addEventListener('click', () => updateState(2));
        
        // 初始化 MathJax 渲染（如果有 LaTeX 公式）
        if (window.MathJax) {
            MathJax.typesetPromise();
        }
    })();
    </script>
