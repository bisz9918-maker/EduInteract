    <script>
    (function() {
        // Scene 1: 甲烷与水反应微观过程
        // 布局坐标表参考：
        // | 元素 | 位置/范围 | 备注 |
        // |------|-----------|------|
        // | 反应前框 | (50,80)-(350,280) | 300×200 |
        // | 反应后框 | (450,80)-(750,280) | 300×200 |
        // | 甲分子-碳原子 | (150,180) r=25 | 斜线填充 |
        // | 甲分子-氢 | (120,150),(180,150),(120,210),(180,210) r=15 |
        // | 乙分子-氧原子 | (280,180) r=25 | 黑色实心 |
        // | 乙分子-氢 | (255,210),(305,210) r=15 |
        // | 丙分子 | (520,180),(560,180) | CO |
        // | 丁分子 | (650,160),(670,160) 等 3组 | 3个H₂ |

        let currentStep = 0;
        const totalSteps = 5;

        const btnPrev = document.getElementById('scene1-btn-prev');
        const btnNext = document.getElementById('scene1-btn-next');
        const stepIndicator = document.getElementById('scene1-step-indicator');

        // 元素引用
        const reactantBox = document.getElementById('scene1-reactant-box');
        const productBox = document.getElementById('scene1-product-box');
        const arrowGroup = document.getElementById('scene1-arrow-group');
        const highTempText = document.getElementById('scene1-high-temp');
        const moleculeA = document.getElementById('scene1-molecule-a');
        const moleculeB = document.getElementById('scene1-molecule-b');
        const moleculeC = document.getElementById('scene1-molecule-c');
        const moleculeD = document.getElementById('scene1-molecule-d');
        const labelA = document.getElementById('scene1-label-a');
        const labelB = document.getElementById('scene1-label-b');
        const labelC = document.getElementById('scene1-label-c');
        const labelD = document.getElementById('scene1-label-d');
        const titleBefore = document.getElementById('scene1-title-before');
        const titleAfter = document.getElementById('scene1-title-after');
        const equation = document.getElementById('scene1-equation');

        // 原子元素
        const cAtom = document.getElementById('scene1-c-atom');
        const hA1 = document.getElementById('scene1-h-a1');
        const hA2 = document.getElementById('scene1-h-a2');
        const hA3 = document.getElementById('scene1-h-a3');
        const hA4 = document.getElementById('scene1-h-a4');
        const oAtom = document.getElementById('scene1-o-atom');
        const hB1 = document.getElementById('scene1-h-b1');
        const hB2 = document.getElementById('scene1-h-b2');

        // 化学键
        const bondsA = [
            document.getElementById('scene1-bond-a1'),
            document.getElementById('scene1-bond-a2'),
            document.getElementById('scene1-bond-a3'),
            document.getElementById('scene1-bond-a4')
        ];
        const bondsB = [
            document.getElementById('scene1-bond-b1'),
            document.getElementById('scene1-bond-b2')
        ];

        // 更新步骤显示
        function updateStepDisplay() {
            stepIndicator.textContent = `步骤 ${currentStep + 1}/${totalSteps}`;
            btnPrev.disabled = currentStep === 0;
            btnNext.disabled = currentStep === totalSteps - 1;
        }

        // 设置步骤状态
        function setStep(step) {
            currentStep = step;

            // 重置所有元素状态
            reactantBox.classList.remove('scene1-box-highlight');
            productBox.classList.remove('scene1-box-highlight');
            arrowGroup.style.opacity = '0';
            highTempText.style.opacity = '0';
            moleculeA.style.opacity = '1';
            moleculeB.style.opacity = '1';
            moleculeC.style.opacity = '0';
            moleculeD.style.opacity = '0';
            labelA.style.opacity = '1';
            labelB.style.opacity = '1';
            labelC.style.opacity = '0';
            labelD.style.opacity = '0';
            titleBefore.style.opacity = '1';
            titleAfter.style.opacity = '0';
            equation.style.opacity = '0';

            // 重置化学键
            bondsA.forEach(b => b.style.opacity = '1');
            bondsB.forEach(b => b.style.opacity = '1');

            // 重置原子位置
            cAtom.setAttribute('cx', '150');
            cAtom.setAttribute('cy', '180');
            hA1.setAttribute('cx', '120');
            hA1.setAttribute('cy', '150');
            hA2.setAttribute('cx', '180');
            hA2.setAttribute('cy', '150');
            hA3.setAttribute('cx', '120');
            hA3.setAttribute('cy', '210');
            hA4.setAttribute('cx', '180');
            hA4.setAttribute('cy', '210');
            oAtom.setAttribute('cx', '280');
            oAtom.setAttribute('cy', '180');
            hB1.setAttribute('cx', '255');
            hB1.setAttribute('cy', '210');
            hB2.setAttribute('cx', '305');
            hB2.setAttribute('cy', '210');

            // 根据步骤设置状态
            switch(step) {
                case 0: // 初始状态 - 反应前
                    reactantBox.classList.add('scene1-box-highlight');
                    break;

                case 1: // 化学键断裂
                    reactantBox.classList.add('scene1-box-highlight');
                    arrowGroup.style.opacity = '1';
                    highTempText.style.opacity = '1';
                    // 化学键淡出
                    bondsA.forEach(b => b.style.opacity = '0');
                    bondsB.forEach(b => b.style.opacity = '0');
                    // 原子分散
                    hA1.setAttribute('cx', '105');
                    hA1.setAttribute('cy', '135');
                    hA2.setAttribute('cx', '195');
                    hA2.setAttribute('cy', '135');
                    hA3.setAttribute('cx', '105');
                    hA3.setAttribute('cy', '225');
                    hA4.setAttribute('cx', '195');
                    hA4.setAttribute('cy', '225');
                    hB1.setAttribute('cx', '240');
                    hB1.setAttribute('cy', '230');
                    hB2.setAttribute('cx', '320');
                    hB2.setAttribute('cy', '230');
                    break;

                case 2: // 原子重组中
                    arrowGroup.style.opacity = '1';
                    highTempText.style.opacity = '1';
                    reactantBox.classList.add('scene1-box-highlight');
                    // 化学键消失
                    bondsA.forEach(b => b.style.opacity = '0');
                    bondsB.forEach(b => b.style.opacity = '0');
                    // 原子开始移动向生成物位置
                    cAtom.setAttribute('cx', '400');
                    cAtom.setAttribute('cy', '180');
                    oAtom.setAttribute('cx', '450');
                    oAtom.setAttribute('cy', '180');
                    hA1.setAttribute('cx', '500');
                    hA1.setAttribute('cy', '160');
                    hA2.setAttribute('cx', '500');
                    hA2.setAttribute('cy', '200');
                    hA3.setAttribute('cx', '500');
                    hA3.setAttribute('cy', '240');
                    hA4.setAttribute('cx', '520');
                    hA4.setAttribute('cy', '160');
                    hB1.setAttribute('cx', '520');
                    hB1.setAttribute('cy', '200');
                    hB2.setAttribute('cx', '520');
                    hB2.setAttribute('cy', '240');
                    break;

                case 3: // 生成物形成
                    productBox.classList.add('scene1-box-highlight');
                    arrowGroup.style.opacity = '1';
                    highTempText.style.opacity = '1';
                    moleculeC.style.opacity = '1';
                    moleculeD.style.opacity = '1';
                    labelC.style.opacity = '1';
                    labelD.style.opacity = '1';
                    titleAfter.style.opacity = '1';
                    moleculeA.style.opacity = '0';
                    moleculeB.style.opacity = '0';
                    labelA.style.opacity = '0';
                    labelB.style.opacity = '0';
                    titleBefore.style.opacity = '0';
                    break;

                case 4: // 结论 - 化学方程式
                    productBox.classList.add('scene1-box-highlight');
                    arrowGroup.style.opacity = '1';
                    highTempText.style.opacity = '1';
                    moleculeC.style.opacity = '1';
                    moleculeD.style.opacity = '1';
                    labelC.style.opacity = '1';
                    labelD.style.opacity = '1';
                    titleAfter.style.opacity = '1';
                    moleculeA.style.opacity = '0.3';
                    moleculeB.style.opacity = '0.3';
                    labelA.style.opacity = '0.5';
                    labelB.style.opacity = '0.5';
                    titleBefore.style.opacity = '0.5';
                    equation.style.opacity = '1';
                    break;
            }

            updateStepDisplay();
        }

        // 按钮事件
        btnPrev.addEventListener('click', function() {
            if (currentStep > 0) {
                setStep(currentStep - 1);
            }
        });

        btnNext.addEventListener('click', function() {
            if (currentStep < totalSteps - 1) {
                setStep(currentStep + 1);
            }
        });

        // 初始化
        setStep(0);
    })();
    </script>
</body>
