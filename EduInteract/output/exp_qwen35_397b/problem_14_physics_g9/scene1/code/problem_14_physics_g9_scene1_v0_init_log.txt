    <script>
    (function() {
        // Scene 1 实现 - 滑轮组受力分析

        // 布局坐标表（对照依据）：
        // | 元素 | 类型 | 位置/范围 | 大小 | 备注 |
        // |------|------|-----------|------|------|
        // | 墙面 | 竖直线 | (20,120)-(20,280) | 4×160 | 左侧固定端 |
        // | 地面 | 水平线 | (20,280)-(480,280) | 460×4 | 带斜线阴影 |
        // | 定滑轮 | 圆形 | 中心(100,200) | r=35 | 固定在墙面 |
        // | 动滑轮 | 圆形 | 中心(260,200) | r=35 | 连接物体A |
        // | 物体A | 矩形 | (340,150)-(440,280) | 100×130 | 填充#e0e0e0 |
        // | 绳子段1 | 线段 | (100,165)-(260,165) | 160×3 | 上段水平绳 |
        // | 绳子段2 | 线段 | (100,235)-(260,235) | 160×3 | 下段水平绳 |
        // | 绳子段3 | 线段 | (260,200)-(340,200) | 80×3 | 连接物体A |
        // | 拉力F箭头 | 箭头 | (280,140)-(220,140) | 60×8 | 蓝色，向左 |
        // | 摩擦力f箭头 | 箭头 | (390,260)-(340,260) | 50×8 | 红色，向左 |
        // | F=5N标签 | text | (250,125) | ~50×16 | 拉力数值 |
        // | n=3标签 | text | (180,180) | ~30×16 | 绳子段数，蓝色 |
        // | 物体A标签 | text | (390,200) | ~40×16 | 物体标识 |
        // | v=1m/s标签 | text | (390,290) | ~60×16 | 下方 |
        // | 受力分析图框 | 矩形 | (520,150)-(760,320) | 240×170 | 右侧放大图 |

        // 状态管理
        let currentStep = 0;
        const totalSteps = 3;

        // DOM元素引用
        const btnPrev = document.getElementById('scene1-btn-prev');
        const btnNext = document.getElementById('scene1-btn-next');
        const stepIndicator = document.getElementById('scene1-step-indicator');

        // 承重绳段元素
        const ropeSeg1 = document.getElementById('scene1-rope-seg1');
        const ropeSeg2 = document.getElementById('scene1-rope-seg2');
        const ropeSeg3 = document.getElementById('scene1-rope-seg3');

        // 步骤相关元素
        const labelN3 = document.getElementById('scene1-label-n3');
        const groupF = document.getElementById('scene1-group-f');
        const analysisGroup = document.getElementById('scene1-analysis-group');
        const formulaText = document.getElementById('scene1-formula');
        const netForceText = document.getElementById('scene1-net-force');

        // 工具函数：显示元素
        function show(el) {
            if (el) {
                el.classList.remove('scene1-hidden');
                el.classList.add('scene1-visible');
                el.style.display = 'block';
                el.style.opacity = '1';
            }
        }

        // 工具函数：隐藏元素
        function hide(el) {
            if (el) {
                el.classList.remove('scene1-visible');
                el.classList.add('scene1-hidden');
                el.style.display = 'none';
                el.style.opacity = '0';
            }
        }

        // 更新步骤状态
        function updateStep() {
            // 更新步骤指示器
            stepIndicator.textContent = '步骤 ' + currentStep + '/' + totalSteps;

            // 更新按钮状态
            btnPrev.disabled = (currentStep === 0);
            btnNext.disabled = (currentStep === totalSteps);
            btnPrev.style.background = (currentStep === 0) ? '#bdbdbd' : '#9e9e9e';
            btnNext.style.background = (currentStep === totalSteps) ? '#90caf9' : '#1565c0';

            // 根据步骤设置元素可见性
            switch (currentStep) {
                case 0:
                    // 初始状态：只显示基础结构
                    hide(labelN3);
                    hide(groupF);
                    hide(analysisGroup);
                    // 绳子恢复黑色细线
                    ropeSeg1.setAttribute('stroke', '#000000');
                    ropeSeg1.setAttribute('stroke-width', '2');
                    ropeSeg2.setAttribute('stroke', '#000000');
                    ropeSeg2.setAttribute('stroke-width', '2');
                    ropeSeg3.setAttribute('stroke', '#000000');
                    ropeSeg3.setAttribute('stroke-width', '2');
                    break;

                case 1:
                    // 步骤1：高亮3段承重绳子，显示n=3
                    show(labelN3);
                    hide(groupF);
                    hide(analysisGroup);
                    // 绳子高亮为蓝色加粗
                    ropeSeg1.setAttribute('stroke', '#1565c0');
                    ropeSeg1.setAttribute('stroke-width', '5');
                    ropeSeg2.setAttribute('stroke', '#1565c0');
                    ropeSeg2.setAttribute('stroke-width', '5');
                    ropeSeg3.setAttribute('stroke', '#1565c0');
                    ropeSeg3.setAttribute('stroke-width', '5');
                    break;

                case 2:
                    // 步骤2：显示受力分析图和关系式
                    show(labelN3);
                    show(groupF);
                    show(analysisGroup);
                    hide(netForceText);
                    // 绳子保持高亮
                    ropeSeg1.setAttribute('stroke', '#1565c0');
                    ropeSeg1.setAttribute('stroke-width', '5');
                    ropeSeg2.setAttribute('stroke', '#1565c0');
                    ropeSeg2.setAttribute('stroke-width', '5');
                    ropeSeg3.setAttribute('stroke', '#1565c0');
                    ropeSeg3.setAttribute('stroke-width', '5');
                    break;

                case 3:
                    // 步骤3：展示匀速运动合力为0
                    show(labelN3);
                    show(groupF);
                    show(analysisGroup);
                    show(netForceText);
                    // 绳子保持高亮
                    ropeSeg1.setAttribute('stroke', '#1565c0');
                    ropeSeg1.setAttribute('stroke-width', '5');
                    ropeSeg2.setAttribute('stroke', '#1565c0');
                    ropeSeg2.setAttribute('stroke-width', '5');
                    ropeSeg3.setAttribute('stroke', '#1565c0');
                    ropeSeg3.setAttribute('stroke-width', '5');
                    break;
            }
        }

        // 按钮事件绑定
        btnPrev.addEventListener('click', function() {
            if (currentStep > 0) {
                currentStep--;
                updateStep();
            }
        });

        btnNext.addEventListener('click', function() {
            if (currentStep < totalSteps) {
                currentStep++;
                updateStep();
            }
        });

        // 初始化
        updateStep();

    })();
    </script>
