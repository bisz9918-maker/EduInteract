    <script>
    (function() {
        // Scene 2 实现 - 甲烷燃烧反应原子守恒验证

        // 状态变量
        let currentState = 0;
        const totalSteps = 4;

        // DOM 元素引用
        const btnPrev = document.getElementById('scene2-btn-prev');
        const btnNext = document.getElementById('scene2-btn-next');
        const stepIndicator = document.getElementById('scene2-step-indicator');
        const conclusion = document.getElementById('scene2-conclusion');

        // 表格行
        const rowCarbon = document.getElementById('scene2-table-row-carbon');
        const rowHydrogen = document.getElementById('scene2-table-row-hydrogen');
        const rowOxygen = document.getElementById('scene2-table-row-oxygen');

        // 原子元素
        const carbonAtoms = document.querySelectorAll('.scene2-carbon-atom');
        const hydrogenAtoms = document.querySelectorAll('.scene2-hydrogen-atom');
        const oxygenAtoms = document.querySelectorAll('.scene2-oxygen-atom');

        // 高亮光环组
        let highlightRings = [];

        // 创建高亮光环
        function createHighlightRing(cx, cy) {
            const svg = document.getElementById('scene2-svg');
            const ring = document.createElementNS('http://www.w3.org/2000/svg', 'circle');
            ring.setAttribute('cx', cx);
            ring.setAttribute('cy', cy);
            ring.setAttribute('r', '18');
            ring.setAttribute('fill', 'none');
            ring.setAttribute('stroke', '#1565c0');
            ring.setAttribute('stroke-width', '3');
            ring.classList.add('scene2-highlight');
            svg.appendChild(ring);
            highlightRings.push(ring);
            return ring;
        }

        // 清除所有高亮光环
        function clearHighlightRings() {
            highlightRings.forEach(ring => ring.remove());
            highlightRings = [];
        }

        // 高亮指定原子
        function highlightAtoms(atoms) {
            atoms.forEach(atom => {
                const cx = atom.getAttribute('cx');
                const cy = atom.getAttribute('cy');
                createHighlightRing(parseFloat(cx), parseFloat(cy));
            });
        }

        // 更新步骤显示
        function updateStepDisplay() {
            stepIndicator.textContent = `步骤 ${currentState}/${totalSteps}`;
            
            // 更新按钮状态
            btnPrev.disabled = currentState === 0;
            if (currentState === totalSteps) {
                btnNext.textContent = '完成';
                btnNext.disabled = true;
            } else {
                btnNext.textContent = '下一步 ▶';
                btnNext.disabled = false;
            }
        }

        // 执行步骤
        function goToStep(step) {
            // 清除之前的高亮和表格显示
            clearHighlightRings();
            rowCarbon.classList.remove('visible');
            rowHydrogen.classList.remove('visible');
            rowOxygen.classList.remove('visible');
            conclusion.classList.remove('visible');

            currentState = step;

            // 根据步骤执行相应操作
            switch(step) {
                case 0:
                    // 初始状态 - 无高亮
                    break;
                case 1:
                    // 高亮碳原子
                    highlightAtoms(carbonAtoms);
                    rowCarbon.classList.add('visible');
                    break;
                case 2:
                    // 高亮碳原子 + 氢原子
                    highlightAtoms(carbonAtoms);
                    highlightAtoms(hydrogenAtoms);
                    rowCarbon.classList.add('visible');
                    rowHydrogen.classList.add('visible');
                    break;
                case 3:
                    // 高亮碳原子 + 氢原子 + 氧原子
                    highlightAtoms(carbonAtoms);
                    highlightAtoms(hydrogenAtoms);
                    highlightAtoms(oxygenAtoms);
                    rowCarbon.classList.add('visible');
                    rowHydrogen.classList.add('visible');
                    rowOxygen.classList.add('visible');
                    break;
                case 4:
                    // 全部高亮 + 显示结论
                    highlightAtoms(carbonAtoms);
                    highlightAtoms(hydrogenAtoms);
                    highlightAtoms(oxygenAtoms);
                    rowCarbon.classList.add('visible');
                    rowHydrogen.classList.add('visible');
                    rowOxygen.classList.add('visible');
                    conclusion.classList.add('visible');
                    break;
            }

            updateStepDisplay();
        }

        // 事件监听
        btnPrev.addEventListener('click', function() {
            if (currentState > 0) {
                goToStep(currentState - 1);
            }
        });

        btnNext.addEventListener('click', function() {
            if (currentState < totalSteps) {
                goToStep(currentState + 1);
            }
        });

        // 初始化
        updateStepDisplay();

    })();
    </script>
