    <script>
    (function() {
        // Scene 2 实现：各装置反应及气体成分变化

        // 布局坐标表（实现对照依据）：
        // | 元素 | 类型 | 位置/范围 | 大小 | 备注 |
        // |------|------|-----------|------|------|
        // | 标题 | text | (400,25) | ~200×20 | 居中 |
        // | 装置 A 方程式 | text | (60,60-80) | ~280×40 | 分两行 |
        // | 装置 B 方程式 | text | (60,115-155) | ~320×40 | 分三行 |
        // | 装置 C 方程式 | text | (60,185-205) | ~280×40 | 分两行 |
        // | 装置 D 方程式 | text | (60,245-265) | ~240×20 | 分两行 |
        // | 表格背景 | rect | (40,285)-(760,425) | 720×140 | 表格区域 |
        // | 表格列标题 | text | y=310 | 6 列气体 | H₂,N₂,HCl,H₂O,CO,CO₂ |
        // | 表格行 | text | y=340/370/400/430/460/490 | 6 行 | 初始/A/B/C/D/最终 |
        // | 上一步按钮 | rect+text | (260,465) | 100×35 | 控件栏 |
        // | 下一步按钮 | rect+text | (440,465) | 100×35 | 控件栏 |
        // | 步骤指示 | text | (400,490) | ~80×16 | "步骤 X/5" |

        let currentState = 0;
        const totalSteps = 5;

        // 工具函数：显示元素
        function show(el) {
            if (el) {
                el.style.display = 'block';
                el.style.opacity = '1';
            }
        }

        // 工具函数：隐藏元素
        function hide(el) {
            if (el) {
                el.style.display = 'none';
                el.style.opacity = '0';
            }
        }

        // 更新按钮状态
        function updateButtons() {
            const prevBg = document.getElementById('scene2-btn-prev-bg');
            const prevText = document.getElementById('scene2-btn-prev');
            const nextBg = document.getElementById('scene2-btn-next-bg');
            const nextText = document.getElementById('scene2-btn-next');
            const stepIndicator = document.getElementById('scene2-step-indicator');

            stepIndicator.textContent = '步骤 ' + currentState + '/' + totalSteps;

            if (currentState === 0) {
                prevBg.setAttribute('fill', '#bdbdbd');
                prevText.classList.add('scene2-btn-disabled');
            } else {
                prevBg.setAttribute('fill', '#1976d2');
                prevText.classList.remove('scene2-btn-disabled');
            }

            if (currentState >= totalSteps) {
                nextBg.setAttribute('fill', '#bdbdbd');
                nextText.classList.add('scene2-btn-disabled');
            } else {
                nextBg.setAttribute('fill', '#1976d2');
                nextText.classList.remove('scene2-btn-disabled');
            }
        }

        // 渲染当前状态
        function render() {
            const eqA = document.getElementById('scene2-eq-a');
            const eqB = document.getElementById('scene2-eq-b');
            const eqC = document.getElementById('scene2-eq-c');
            const eqD = document.getElementById('scene2-eq-d');
            const tableBg = document.getElementById('scene2-table-bg');
            const tableHeaders = document.getElementById('scene2-table-headers');
            const rowInit = document.getElementById('scene2-table-row-init');
            const rowA = document.getElementById('scene2-table-row-a');
            const rowB = document.getElementById('scene2-table-row-b');
            const rowC = document.getElementById('scene2-table-row-c');
            const rowD = document.getElementById('scene2-table-row-d');
            const rowFinal = document.getElementById('scene2-table-row-final');

            // 始终显示标题和装置 A
            show(eqA);

            // 根据状态显示其他方程式
            if (currentState >= 1) {
                show(eqB);
            } else {
                hide(eqB);
            }

            if (currentState >= 2) {
                show(eqC);
            } else {
                hide(eqC);
            }

            if (currentState >= 3) {
                show(eqD);
            } else {
                hide(eqD);
            }

            // 表格显示逻辑
            if (currentState >= 0) {
                show(tableBg);
                show(tableHeaders);
                show(rowInit);
            }

            if (currentState >= 1) {
                show(rowA);
            } else {
                hide(rowA);
            }

            if (currentState >= 2) {
                show(rowB);
            } else {
                hide(rowB);
            }

            if (currentState >= 3) {
                show(rowC);
            } else {
                hide(rowC);
            }

            if (currentState >= 4) {
                show(rowD);
            } else {
                hide(rowD);
            }

            if (currentState >= 5) {
                show(rowFinal);
            } else {
                hide(rowFinal);
            }

            updateButtons();
        }

        // 上一步
        function prevStep() {
            if (currentState > 0) {
                currentState--;
                render();
            }
        }

        // 下一步
        function nextStep() {
            if (currentState < totalSteps) {
                currentState++;
                render();
            }
        }

        // 绑定事件
        document.getElementById('scene2-btn-prev').addEventListener('click', prevStep);
        document.getElementById('scene2-btn-prev-bg').addEventListener('click', prevStep);
        document.getElementById('scene2-btn-next').addEventListener('click', nextStep);
        document.getElementById('scene2-btn-next-bg').addEventListener('click', nextStep);

        // 初始渲染
        render();
    })();
    </script>
