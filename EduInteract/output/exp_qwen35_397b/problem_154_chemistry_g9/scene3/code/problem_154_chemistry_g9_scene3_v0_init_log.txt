    <script>
    (function() {
        // Scene 3: 复分解反应微观本质
        // 布局坐标表参考:
        // 烧杯1 (CuSO₄): (80,150)-(180,320), 烧杯2 (NaOH): (200,150)-(300,320)
        // 烧杯3 (混合后): (480,150)-(630,350)
        // 滑块轨道: (200,460)-(600,485), 滑块手柄初始: (200,472)

        const sliderTrack = document.getElementById('scene3-slider-track');
        const sliderHandle = document.getElementById('scene3-slider-handle');
        const beaker1Ions = document.getElementById('scene3-beaker1-ions');
        const beaker2Ions = document.getElementById('scene3-beaker2-ions');
        const beaker3Content = document.getElementById('scene3-beaker3-content');
        const equationGroup = document.getElementById('scene3-equation-group');
        const reactionType = document.getElementById('scene3-reaction-type');
        const labelSolution = document.getElementById('scene3-label-solution');

        // 离子初始位置和目标位置
        const ionPositions = {
            cu2_1: { start: { x: 110, y: 200 }, target: { x: 530, y: 310 }, type: 'precipitate' },
            cu2_2: { start: { x: 140, y: 240 }, target: { x: 545, y: 315 }, type: 'precipitate' },
            so4_1: { start: { x: 120, y: 270 }, target: { x: 540, y: 260 }, type: 'solution' },
            so4_2: { start: { x: 150, y: 290 }, target: { x: 600, y: 270 }, type: 'solution' },
            na_1: { start: { x: 230, y: 200 }, target: { x: 510, y: 200 }, type: 'solution' },
            na_2: { start: { x: 260, y: 230 }, target: { x: 580, y: 210 }, type: 'solution' },
            na_3: { start: { x: 240, y: 260 }, target: { x: 530, y: 230 }, type: 'solution' },
            na_4: { start: { x: 270, y: 280 }, target: { x: 590, y: 240 }, type: 'solution' },
            oh_1: { start: { x: 220, y: 250 }, target: { x: 520, y: 325 }, type: 'precipitate' },
            oh_2: { start: { x: 250, y: 290 }, target: { x: 540, y: 330 }, type: 'precipitate' },
            oh_3: { start: { x: 280, y: 260 }, target: { x: 555, y: 320 }, type: 'precipitate' },
            oh_4: { start: { x: 235, y: 300 }, target: { x: 540, y: 310 }, type: 'precipitate' }
        };

        let sliderPosition = 0; // 0-100
        let isDragging = false;

        // 更新离子位置
        function updateIonPositions(progress) {
            // progress: 0-1, 0=初始, 1=完成
            const ions = [
                { id: 'scene3-ion-cu2-1', labelId: 'scene3-ion-cu2-1-label', pos: ionPositions.cu2_1 },
                { id: 'scene3-ion-cu2-2', labelId: 'scene3-ion-cu2-2-label', pos: ionPositions.cu2_2 },
                { id: 'scene3-ion-so4-1', labelId: 'scene3-ion-so4-1-label', pos: ionPositions.so4_1 },
                { id: 'scene3-ion-so4-2', labelId: 'scene3-ion-so4-2-label', pos: ionPositions.so4_2 },
                { id: 'scene3-ion-na-1', labelId: 'scene3-ion-na-1-label', pos: ionPositions.na_1 },
                { id: 'scene3-ion-na-2', labelId: 'scene3-ion-na-2-label', pos: ionPositions.na_2 },
                { id: 'scene3-ion-na-3', labelId: 'scene3-ion-na-3-label', pos: ionPositions.na_3 },
                { id: 'scene3-ion-na-4', labelId: 'scene3-ion-na-4-label', pos: ionPositions.na_4 },
                { id: 'scene3-ion-oh-1', labelId: 'scene3-ion-oh-1-label', pos: ionPositions.oh_1 },
                { id: 'scene3-ion-oh-2', labelId: 'scene3-ion-oh-2-label', pos: ionPositions.oh_2 },
                { id: 'scene3-ion-oh-3', labelId: 'scene3-ion-oh-3-label', pos: ionPositions.oh_3 },
                { id: 'scene3-ion-oh-4', labelId: 'scene3-ion-oh-4-label', pos: ionPositions.oh_4 }
            ];

            ions.forEach(ion => {
                const circle = document.getElementById(ion.id);
                const label = document.getElementById(ion.labelId);
                const currentX = ion.pos.start.x + (ion.pos.target.x - ion.pos.start.x) * progress;
                const currentY = ion.pos.start.y + (ion.pos.target.y - ion.pos.start.y) * progress;

                circle.setAttribute('cx', currentX);
                circle.setAttribute('cy', currentY);
                label.setAttribute('x', currentX);
                label.setAttribute('y', ion.pos.start.y < 250 ? currentY - 12 : currentY + 18);

                // 根据进度调整透明度
                if (ion.pos.type === 'precipitate') {
                    // 沉淀离子在 progress > 0.7 时逐渐消失
                    if (progress < 0.7) {
                        circle.setAttribute('opacity', 1);
                        label.setAttribute('opacity', 1);
                    } else {
                        const fadeOut = 1 - (progress - 0.7) / 0.3;
                        circle.setAttribute('opacity', fadeOut);
                        label.setAttribute('opacity', fadeOut);
                    }
                } else {
                    // 溶液离子逐渐移动到目标位置
                    if (progress < 0.3) {
                        circle.setAttribute('opacity', 1 - progress / 0.3);
                        label.setAttribute('opacity', 1 - progress / 0.3);
                    } else {
                        circle.setAttribute('opacity', 1);
                        label.setAttribute('opacity', 1);
                    }
                }
            });
        }

        // 更新最终离子显示
        function updateFinalIons(progress) {
            const finalIons = [
                'scene3-ion-na-final-1', 'scene3-ion-na-final-1-label',
                'scene3-ion-na-final-2', 'scene3-ion-na-final-2-label',
                'scene3-ion-na-final-3', 'scene3-ion-na-final-3-label',
                'scene3-ion-na-final-4', 'scene3-ion-na-final-4-label',
                'scene3-ion-so4-final-1', 'scene3-ion-so4-final-1-label',
                'scene3-ion-so4-final-2', 'scene3-ion-so4-final-2-label'
            ];

            finalIons.forEach(id => {
                const el = document.getElementById(id);
                if (progress > 0.7) {
                    el.setAttribute('opacity', (progress - 0.7) / 0.3);
                } else {
                    el.setAttribute('opacity', 0);
                }
            });

            // 沉淀和溶液标签
            if (progress > 0.7) {
                const precipOpacity = (progress - 0.7) / 0.3;
                document.getElementById('scene3-label-precipitate').setAttribute('opacity', precipOpacity);
                labelSolution.setAttribute('opacity', precipOpacity);
            } else {
                document.getElementById('scene3-label-precipitate').setAttribute('opacity', 0);
                labelSolution.setAttribute('opacity', 0);
            }
        }

        // 主更新函数
        function updateScene() {
            const progress = sliderPosition / 100;

            // 更新滑块手柄位置
            const handleX = 200 + progress * 400;
            sliderHandle.setAttribute('cx', handleX);

            // 更新离子位置
            updateIonPositions(progress);

            // 更新最终离子显示
            updateFinalIons(progress);

            // 更新烧杯3 内容透明度
            beaker3Content.setAttribute('opacity', progress > 0.3 ? Math.min(1, (progress - 0.3) / 0.4) : 0);

            // 更新方程式框透明度
            equationGroup.setAttribute('opacity', 0.5 + progress * 0.5);

            // 更新反应类型标签
            if (progress > 0.8) {
                reactionType.setAttribute('opacity', (progress - 0.8) / 0.2);
            } else {
                reactionType.setAttribute('opacity', 0);
            }

            // 更新初始离子组透明度
            beaker1Ions.setAttribute('opacity', progress < 0.3 ? 1 - progress / 0.3 : 0);
            beaker2Ions.setAttribute('opacity', progress < 0.3 ? 1 - progress / 0.3 : 0);
        }

        // 滑块拖拽事件处理
        function handleMouseDown(e) {
            isDragging = true;
            handleMouseMove(e);
        }

        function handleMouseMove(e) {
            if (!isDragging) return;

            const rect = sliderTrack.getBoundingClientRect();
            const svgRect = document.getElementById('scene3-svg').getBoundingClientRect();
            let clientX = e.clientX || (e.touches && e.touches[0].clientX);

            if (clientX === undefined) return;

            let newX = clientX - svgRect.left;
            newX = Math.max(200, Math.min(600, newX));
            sliderPosition = ((newX - 200) / 400) * 100;

            updateScene();
        }

        function handleMouseUp() {
            isDragging = false;
        }

        // 绑定事件
        sliderHandle.addEventListener('mousedown', handleMouseDown);
        sliderHandle.addEventListener('touchstart', handleMouseDown);

        document.addEventListener('mousemove', handleMouseMove);
        document.addEventListener('touchmove', handleMouseMove);

        document.addEventListener('mouseup', handleMouseUp);
        document.addEventListener('touchend', handleMouseUp);

        // 初始化
        updateScene();
    })();
    </script>
