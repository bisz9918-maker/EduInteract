    <script>
    (function() {
        // Scene 2: 温度变化对A点饱和溶液的影响
        
        // 坐标映射参数
        const ORIGIN_X = 80;
        const ORIGIN_Y = 420;
        const X_SCALE = 100; // 每单位温度对应x像素
        const Y_SCALE = 0.6; // 溶解度到y的缩放
        
        // 温度刻度对应的x坐标
        const T1_X = 200;
        const T2_X = 400;
        const T3_X = 600;
        
        // 溶解度刻度对应的y坐标
        const S1_Y = 320;
        const S2_Y = 200;
        
        // 计算溶解度曲线上的点（根据x坐标返回y坐标）
        // 曲线方程近似：从(80,380)到(520,80)的曲线
        function getCurveY(x) {
            // 使用二次曲线拟合
            const normalizedX = (x - 80) / 440; // 0到1
            // 曲线从y=380降到y=80
            const baseY = 380 - 300 * normalizedX;
            // 添加曲率（温度越高溶解度增长越快）
            const curvature = 50 * normalizedX * normalizedX;
            return baseY - curvature + 50;
        }
        
        // 根据温度x坐标计算溶解度值（用于显示）
        function getSolubility(x) {
            const y = getCurveY(x);
            // y从380(低温)到80(高温)，溶解度从低到高
            // S1对应y=320, S2对应y=200
            const s1 = 320, s2 = 200;
            // 线性插值估算溶解度标签
            if (x <= T1_X) {
                return "S₁";
            } else if (x <= T2_X) {
                // 在S1和S2之间
                const ratio = (x - T1_X) / (T2_X - T1_X);
                // 不显示具体值，用温度表示
                return null;
            } else {
                return "S₂+";
            }
        }
        
        // 计算质量分数
        function getMassFraction(x) {
            const y = getCurveY(x);
            // 假设S2=50g, S1=30g（相对值）
            const s2 = 50, s1 = 30;
            const ratio = (y - S1_Y) / (S2_Y - S1_Y); // 0到1之间
            const s = s1 + (s2 - s1) * (1 - ratio);
            return (s / (100 + s) * 100).toFixed(1);
        }
        
        // 获取温度标签
        function getTempLabel(x) {
            if (Math.abs(x - T1_X) < 15) return "t₁";
            if (Math.abs(x - T2_X) < 15) return "t₂";
            if (Math.abs(x - T3_X) < 15) return "t₃";
            // 计算相对温度
            const ratio = (x - T1_X) / (T2_X - T1_X);
            if (x < T1_X) {
                return "<t₁";
            } else if (x < T2_X) {
                return "t₁~t₂";
            } else {
                return ">t₂";
            }
        }
        
        // DOM元素
        const pointP = document.getElementById('scene2-point-p');
        const trail = document.getElementById('scene2-trail');
        const dashV = document.getElementById('scene2-dash-a-v');
        const dashH = document.getElementById('scene2-dash-a-h');
        const crystalGroup = document.getElementById('scene2-crystal-group');
        const crystalPile = document.getElementById('scene2-crystal-pile');
        const sliderHandle = document.getElementById('scene2-slider-handle');
        const tempValue = document.getElementById('scene2-temp-value');
        const solValue = document.getElementById('scene2-sol-value');
        const massValue = document.getElementById('scene2-mass-value');
        
        // 状态变量
        let currentX = T2_X; // 当前滑块x位置，初始在t₂
        let isDragging = false;
        let trailPoints = []; // 轨迹点集合
        
        // 更新场景
        function updateScene(x) {
            // 限制x范围
            x = Math.max(100, Math.min(600, x));
            currentX = x;
            
            const y = getCurveY(x);
            
            // 更新动点P位置
            pointP.setAttribute('cx', x);
            pointP.setAttribute('cy', y);
            
            // 更新滑块手柄位置
            sliderHandle.setAttribute('cx', x);
            
            // 更新投影虚线
            dashV.setAttribute('x1', x);
            dashV.setAttribute('x2', x);
            dashV.setAttribute('y2', 420);
            dashH.setAttribute('x2', x);
            dashH.setAttribute('y1', y);
            dashH.setAttribute('y2', y);
            
            // 更新数值显示
            tempValue.textContent = getTempLabel(x);
            
            // 溶解度显示
            if (Math.abs(x - T1_X) < 10) {
                solValue.textContent = "S₁";
            } else if (Math.abs(x - T2_X) < 10) {
                solValue.textContent = "S₂";
            } else {
                solValue.textContent = "S";
            }
            
            // 质量分数
            massValue.textContent = getMassFraction(x) + "%";
            
            // 更新轨迹
            if (trailPoints.length === 0 || 
                Math.abs(x - trailPoints[trailPoints.length-1].x) > 5) {
                trailPoints.push({x: x, y: y});
                // 限制轨迹点数量
                if (trailPoints.length > 100) {
                    trailPoints.shift();
                }
            }
            
            // 绘制轨迹线
            if (trailPoints.length > 1) {
                let d = `M ${trailPoints[0].x},${trailPoints[0].y}`;
                for (let i = 1; i < trailPoints.length; i++) {
                    d += ` L ${trailPoints[i].x},${trailPoints[i].y}`;
                }
                trail.setAttribute('d', d);
                trail.setAttribute('opacity', '0.6');
            }
            
            // 晶体析出效果（当温度低于t₂时）
            if (x < T2_X - 5) {
                crystalGroup.setAttribute('opacity', '1');
                // 晶体高度随温度降低而增加
                const crystalHeight = Math.min(40, (T2_X - x) * 0.15);
                const pileY = 410 - crystalHeight;
                crystalPile.setAttribute('points', 
                    `360,410 440,410 440,${pileY} 400,${pileY - 5} 360,${pileY}`);
            } else {
                crystalGroup.setAttribute('opacity', '0');
            }
        }
        
        // 滑块拖拽事件处理
        function handleStart(e) {
            isDragging = true;
            e.preventDefault();
        }
        
        function handleMove(e) {
            if (!isDragging) return;
            
            const svg = document.getElementById('scene2-svg');
            const rect = svg.getBoundingClientRect();
            let clientX;
            
            if (e.touches) {
                clientX = e.touches[0].clientX;
            } else {
                clientX = e.clientX;
            }
            
            const x = clientX - rect.left;
            updateScene(x);
        }
        
        function handleEnd() {
            isDragging = false;
        }
        
        // 绑定事件
        sliderHandle.addEventListener('mousedown', handleStart);
        sliderHandle.addEventListener('touchstart', handleStart);
        
        document.addEventListener('mousemove', handleMove);
        document.addEventListener('touchmove', handleMove);
        
        document.addEventListener('mouseup', handleEnd);
        document.addEventListener('touchend', handleEnd);
        
        // 初始化
        updateScene(T2_X);
        
    })();
    </script>
