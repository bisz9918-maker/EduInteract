    <script>
    (function() {
        // Scene 1 实现 - 化学实验图像分析
        
        // 状态变量
        let currentStep = 0;
        const totalSteps = 5;
        
        // 步骤标题
        const stepTitles = [
            "请分析四个实验操作图像",
            "选项 A：向饱和石灰水中加入氧化钙——错误",
            "选项 B：加热高锰酸钾产生氧气——错误",
            "选项 C：高温煅烧碳酸钙——正确",
            "选项 D：氧气在水中的溶解性——错误",
            "正确答案：C"
        ];
        
        // 步骤指示器文本
        const stepIndicators = [
            "步骤 1/5",
            "步骤 2/5",
            "步骤 3/5",
            "步骤 4/5",
            "步骤 5/5",
            "步骤 6/5"
        ];
        
        // 获取 DOM 元素
        const titleEl = document.getElementById('scene1-title');
        const stepIndicatorEl = document.getElementById('scene1-step-indicator');
        const summaryEl = document.getElementById('scene1-summary');
        const btnPrevBg = document.getElementById('scene1-btn-prev-bg');
        const btnPrevText = document.getElementById('scene1-btn-prev-text');
        const btnNextBg = document.getElementById('scene1-btn-next-bg');
        const btnNextText = document.getElementById('scene1-btn-next-text');
        
        // 高亮框元素
        const highlights = {
            a: document.getElementById('scene1-highlight-a'),
            b: document.getElementById('scene1-highlight-b'),
            c: document.getElementById('scene1-highlight-c'),
            d: document.getElementById('scene1-highlight-d')
        };
        
        // 正确曲线元素
        const correctCurves = {
            a: document.getElementById('scene1-curve-a-correct'),
            b: document.getElementById('scene1-curve-b-correct'),
            d: document.getElementById('scene1-curve-d-correct')
        };
        
        // 对错标记元素
        const marks = {
            a: document.getElementById('scene1-mark-a'),
            b: document.getElementById('scene1-mark-b'),
            c: document.getElementById('scene1-mark-c'),
            d: document.getElementById('scene1-mark-d')
        };
        
        // 显示/隐藏工具函数
        function show(el) {
            if (el) {
                el.classList.remove('scene1-hidden');
                el.style.display = '';
                el.style.opacity = '1';
            }
        }
        
        function hide(el) {
            if (el) {
                el.classList.add('scene1-hidden');
                el.style.display = 'none';
                el.style.opacity = '0';
            }
        }
        
        // 隐藏所有高亮和标记
        function hideAllHighlights() {
            Object.values(highlights).forEach(hide);
            Object.values(marks).forEach(hide);
            Object.values(correctCurves).forEach(hide);
        }
        
        // 更新视图
        function updateView() {
            // 更新标题和步骤指示器
            titleEl.textContent = stepTitles[currentStep];
            stepIndicatorEl.textContent = stepIndicators[currentStep];
            
            // 隐藏所有高亮和标记
            hideAllHighlights();
            hide(summaryEl);
            
            // 根据当前步骤显示对应元素
            switch (currentStep) {
                case 0:
                    // 初始状态：只显示原图
                    break;
                    
                case 1:
                    // 分析选项 A
                    show(highlights.a);
                    show(correctCurves.a);
                    show(marks.a);
                    break;
                    
                case 2:
                    // 分析选项 B
                    show(highlights.b);
                    show(correctCurves.b);
                    show(marks.b);
                    break;
                    
                case 3:
                    // 分析选项 C
                    show(highlights.c);
                    show(marks.c);
                    break;
                    
                case 4:
                    // 分析选项 D
                    show(highlights.d);
                    show(correctCurves.d);
                    show(marks.d);
                    break;
                    
                case 5:
                    // 总结：显示所有对错标记
                    show(marks.a);
                    show(marks.b);
                    show(marks.c);
                    show(marks.d);
                    show(summaryEl);
                    break;
            }
            
            // 更新按钮状态
            updateButtons();
        }
        
        // 更新按钮状态
        function updateButtons() {
            if (currentStep === 0) {
                btnPrevBg.setAttribute('fill', '#e0e0e0');
                btnPrevText.setAttribute('fill', '#999');
            } else {
                btnPrevBg.setAttribute('fill', '#f5f5f5');
                btnPrevText.setAttribute('fill', '#333');
            }
            
            if (currentStep >= totalSteps) {
                btnNextBg.setAttribute('fill', '#e0e0e0');
                btnNextText.setAttribute('fill', '#999');
            } else {
                btnNextBg.setAttribute('fill', '#f5f5f5');
                btnNextText.setAttribute('fill', '#333');
            }
        }
        
        // 按钮点击处理
        btnPrevBg.addEventListener('click', function() {
            if (currentStep > 0) {
                currentStep--;
                updateView();
            }
        });
        
        btnNextBg.addEventListener('click', function() {
            if (currentStep < totalSteps) {
                currentStep++;
                updateView();
            }
        });
        
        // 鼠标悬停效果
        btnPrevBg.addEventListener('mouseenter', function() {
            if (currentStep > 0) {
                btnPrevBg.setAttribute('fill', '#e0e0e0');
            }
        });
        btnPrevBg.addEventListener('mouseleave', function() {
            if (currentStep > 0) {
                btnPrevBg.setAttribute('fill', '#f5f5f5');
            }
        });
        
        btnNextBg.addEventListener('mouseenter', function() {
            if (currentStep < totalSteps) {
                btnNextBg.setAttribute('fill', '#e0e0e0');
            }
        });
        btnNextBg.addEventListener('mouseleave', function() {
            if (currentStep < totalSteps) {
                btnNextBg.setAttribute('fill', '#f5f5f5');
            }
        });
        
        // 初始化视图
        updateView();
        
    })();
    </script>
