    <script>
    (function() {
        // Scene 3 实现：图形 C 的周长计算与凹槽分析
        
        // 状态变量
        let currentState = 0;
        const totalSteps = 4;
        
        // 获取 DOM 元素
        const btnPrev = document.getElementById('scene3-btn-prev');
        const btnNext = document.getElementById('scene3-btn-next');
        const stepIndicator = document.getElementById('scene3-step-indicator');
        
        // 图形元素
        const equivRect = document.getElementById('scene3-equiv-rect');
        const topSegment = document.getElementById('scene3-top-segment');
        const grooveLeft = document.getElementById('scene3-groove-left');
        const grooveRight = document.getElementById('scene3-groove-right');
        
        // 公式元素
        const formulaInitial = document.getElementById('scene3-formula-initial');
        const formulaStep1 = document.getElementById('scene3-formula-step1');
        const formulaStep1Result = document.getElementById('scene3-formula-step1-result');
        const formulaStep2 = document.getElementById('scene3-formula-step2');
        const formulaStep3 = document.getElementById('scene3-formula-step3');
        const conclusion = document.getElementById('scene3-conclusion');
        
        // 工具函数：显示元素
        function show(el) {
            el.classList.remove('scene3-hidden');
            el.classList.add('scene3-visible');
            if (el.tagName === 'text' || el.tagName === 'rect' || el.tagName === 'line' || el.tagName === 'path') {
                el.style.opacity = '1';
            } else {
                el.style.display = 'block';
                el.style.opacity = '1';
            }
        }
        
        // 工具函数：隐藏元素
        function hide(el) {
            el.classList.remove('scene3-visible');
            el.classList.add('scene3-hidden');
            if (el.tagName === 'text' || el.tagName === 'rect' || el.tagName === 'line' || el.tagName === 'path') {
                el.style.opacity = '0';
            } else {
                el.style.display = 'none';
                el.style.opacity = '0';
            }
        }
        
        // 更新状态显示
        function updateState() {
            // 重置所有高亮
            topSegment.classList.remove('highlight');
            grooveLeft.classList.remove('highlight');
            grooveRight.classList.remove('highlight');
            
            // 隐藏所有公式
            hide(formulaInitial);
            hide(formulaStep1);
            hide(formulaStep1Result);
            hide(formulaStep2);
            hide(formulaStep3);
            hide(conclusion);
            
            // 根据状态显示对应元素
            switch (currentState) {
                case 0:
                    // 初始状态：只显示原始图形
                    hide(equivRect);
                    show(formulaInitial);
                    break;
                    
                case 1:
                    // 步骤 1：显示等效长方形，高亮顶部线段
                    show(equivRect);
                    topSegment.classList.add('highlight');
                    show(formulaStep1);
                    show(formulaStep1Result);
                    break;
                    
                case 2:
                    // 步骤 2：高亮凹槽竖边
                    show(equivRect);
                    topSegment.classList.add('highlight');
                    grooveLeft.classList.add('highlight');
                    grooveRight.classList.add('highlight');
                    show(formulaStep1);
                    show(formulaStep1Result);
                    show(formulaStep2);
                    break;
                    
                case 3:
                    // 步骤 3：显示完整公式和结论
                    show(equivRect);
                    topSegment.classList.add('highlight');
                    grooveLeft.classList.add('highlight');
                    grooveRight.classList.add('highlight');
                    show(formulaStep1);
                    show(formulaStep1Result);
                    show(formulaStep2);
                    show(formulaStep3);
                    show(conclusion);
                    break;
            }
            
            // 更新按钮状态
            btnPrev.disabled = (currentState === 0);
            btnNext.disabled = (currentState === totalSteps - 1);
            
            // 更新步骤指示器
            stepIndicator.textContent = `步骤 ${currentState + 1}/${totalSteps}`;
        }
        
        // 事件绑定
        btnPrev.addEventListener('click', function() {
            if (currentState > 0) {
                currentState--;
                updateState();
            }
        });
        
        btnNext.addEventListener('click', function() {
            if (currentState < totalSteps - 1) {
                currentState++;
                updateState();
            }
        });
        
        // 初始化
        updateState();
    })();
    </script>
