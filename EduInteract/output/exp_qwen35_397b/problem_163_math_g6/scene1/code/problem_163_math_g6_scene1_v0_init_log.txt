        let currentStep = 0;

        // 工具函数：显示元素
        function showElement(id) {
            const el = document.getElementById(id);
            if (el) {
                el.classList.remove('scene1-hidden');
                el.classList.add('scene1-visible');
                el.style.display = '';
                el.style.opacity = '1';
            }
        }
        
        // 工具函数：隐藏元素
        function hideElement(id) {
            const el = document.getElementById(id);
            if (el) {
                el.classList.remove('scene1-visible');
                el.classList.add('scene1-hidden');
                el.style.display = 'none';
                el.style.opacity = '0';
            }
        }
        
        // 更新步骤显示
        function updateStep(stepIndex) {
            const step = steps[stepIndex];
            
            // 清除所有特殊效果
            document.querySelectorAll('[id^="scene1-"]').forEach(el => {
                el.classList.remove('scene1-pulse', 'scene1-blink');
                if (el.id.startsWith('scene1-point-')) {
                    el.setAttribute('r', '5');
                }
                if (el.id === 'scene1-tri-abc-fill') {
                    el.setAttribute('fill', 'rgba(21,101,192,0.2)');
                }
                if (el.id === 'scene1-line-ab') {
                    el.setAttribute('stroke-width', '1.5');
                }
            });
            
            // 应用隐藏列表
            step.hidden.forEach(id => hideElement(id));
            
            // 应用显示列表
            step.visible.forEach(id => showElement(id));
            
            // 应用高亮效果（步骤1）
            if (step.highlight) {
                step.highlight.forEach(id => {
                    const el = document.getElementById(id);
                    if (el) {
                        if (el.tagName === 'circle') {
                            el.setAttribute('r', '7');
                        }
                        if (el.id === 'scene1-tri-abc-fill') {
                            el.setAttribute('fill', 'rgba(21,101,192,0.3)');
                        }
                    }
                });
            }
            
            // 应用加粗效果（步骤2）
            if (step.bold) {
                step.bold.forEach(id => {
                    const el = document.getElementById(id);
                    if (el) {
                        el.setAttribute('stroke-width', '3');
                    }
                });
            }
            
            // 应用闪烁效果（步骤3）
            if (step.blink) {
                step.blink.forEach(id => {
                    const el = document.getElementById(id);
                    if (el) {
                        el.classList.add('scene1-blink');
                    }
                });
            }
            
            // 应用脉冲效果（步骤5）
            if (step.pulse) {
                step.pulse.forEach(id => {
                    const el = document.getElementById(id);
                    if (el) {
                        el.classList.add('scene1-pulse');
                    }
                });
            }
            
            // 更新步骤指示器
            document.getElementById('scene1-step-indicator').textContent = `步骤 ${stepIndex + 1}/${totalSteps}`;
            
            // 更新按钮状态
            document.getElementById('scene1-btn-prev').disabled = (stepIndex === 0);
            document.getElementById('scene1-btn-next').disabled = (stepIndex === totalSteps - 1);
        }
        
        // 下一步
        function nextStep() {
            if (currentStep < totalSteps - 1) {
                currentStep++;
                updateStep(currentStep);
            }
        }
        
        // 上一步
        function prevStep() {
            if (currentStep > 0) {
                currentStep--;
                updateStep(currentStep);
            }
        }
        
        // 绑定事件
        document.getElementById('scene1-btn-next').addEventListener('click', nextStep);
        document.getElementById('scene1-btn-prev').addEventListener('click', prevStep);
        
        // 初始化
        updateStep(0);
