    <script>
    (function() {
        // Scene 2 实现：三个物体的俯视图对比

        // 状态变量：0=初始，1=物体①，2=物体②，3=物体③
        let currentState = 0;

        // 获取 DOM 元素
        const obj1 = document.getElementById('scene2-obj1');
        const obj2 = document.getElementById('scene2-obj2');
        const obj3 = document.getElementById('scene2-obj3');
        const topviewShape = document.getElementById('scene2-topview-shape');
        const judgment = document.getElementById('scene2-judgment');
        const btn1Bg = document.getElementById('scene2-btn1-bg');
        const btn2Bg = document.getElementById('scene2-btn2-bg');
        const btn3Bg = document.getElementById('scene2-btn3-bg');
        const btn1Text = document.getElementById('scene2-btn1-text');
        const btn2Text = document.getElementById('scene2-btn2-text');
        const btn3Text = document.getElementById('scene2-btn3-text');

        // 俯视图形状定义
        const topviewShapes = {
            1: '150,320 310,320 310,360 150,360', // 矩形（长方体俯视图）
            2: '230,320 280,330 270,360 190,360 180,330', // 五边形（五棱柱俯视图）
            3: '160,320 300,320 290,360 170,360' // 梯形（梯形四棱柱俯视图）
        };

        // 设置物体样式
        function setObjectStyle(obj, isHighlighted) {
            const elements = obj.querySelectorAll('polygon, line');
            elements.forEach(el => {
                if (isHighlighted) {
                    el.setAttribute('stroke', '#1565c0');
                    el.setAttribute('stroke-width', '3');
                } else {
                    el.setAttribute('stroke', '#888888');
                    el.setAttribute('stroke-width', '2');
                }
            });
        }

        // 设置按钮样式
        function setButtonStyle(btnBg, btnText, isActive) {
            if (isActive) {
                btnBg.setAttribute('fill', '#1565c0');
                btnBg.setAttribute('stroke', '#1565c0');
                btnText.setAttribute('fill', '#ffffff');
            } else {
                btnBg.setAttribute('fill', '#f5f5f5');
                btnBg.setAttribute('stroke', '#888888');
                btnText.setAttribute('fill', '#000000');
            }
        }

        // 设置状态
        function setState(state) {
            currentState = state;

            // 重置所有物体样式
            setObjectStyle(obj1, false);
            setObjectStyle(obj2, false);
            setObjectStyle(obj3, false);

            // 重置所有按钮样式
            setButtonStyle(btn1Bg, btn1Text, false);
            setButtonStyle(btn2Bg, btn2Text, false);
            setButtonStyle(btn3Bg, btn3Text, false);

            if (state === 0) {
                // 初始状态：隐藏俯视图和判断标记
                topviewShape.setAttribute('opacity', '0');
                judgment.setAttribute('opacity', '0');
            } else {
                // 高亮对应物体
                if (state === 1) setObjectStyle(obj1, true);
                if (state === 2) setObjectStyle(obj2, true);
                if (state === 3) setObjectStyle(obj3, true);

                // 高亮对应按钮
                if (state === 1) setButtonStyle(btn1Bg, btn1Text, true);
                if (state === 2) setButtonStyle(btn2Bg, btn2Text, true);
                if (state === 3) setButtonStyle(btn3Bg, btn3Text, true);

                // 显示俯视图
                topviewShape.setAttribute('points', topviewShapes[state]);
                topviewShape.setAttribute('opacity', '1');

                // 显示判断标记
                if (state === 3) {
                    judgment.textContent = '✓';
                    judgment.setAttribute('fill', '#2e7d32');
                } else {
                    judgment.textContent = '✗';
                    judgment.setAttribute('fill', '#c62828');
                }
                judgment.setAttribute('opacity', '1');
            }
        }

        // 绑定按钮事件
        document.getElementById('scene2-btn1-bg').addEventListener('click', () => setState(1));
        document.getElementById('scene2-btn2-bg').addEventListener('click', () => setState(2));
        document.getElementById('scene2-btn3-bg').addEventListener('click', () => setState(3));

        // 初始状态
        setState(0);
    })();
    </script>
