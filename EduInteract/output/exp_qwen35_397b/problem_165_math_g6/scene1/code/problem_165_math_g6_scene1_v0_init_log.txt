    <script>
    (function() {
        // Scene 1 实现：正方形地砖面积对比
        
        // 状态变量
        let currentState = 0;
        const totalSteps = 4;
        
        // 每步需要显示的元素 ID 列表
        const steps = {
            0: [],  // 初始状态：仅显示正方形轮廓
            1: ['scene1-line-left', 'scene1-line-right', 'scene1-label-2dm', 'scene1-label-3dm'],
            2: ['scene1-area-left', 'scene1-area-right', 'scene1-formula-left', 'scene1-formula-right'],
            3: ['scene1-ratio', 'scene1-highlight-left', 'scene1-highlight-right']
        };
        
        // 每步正方形的填充透明度
        const squareOpacity = {
            0: 0,
            1: 0.2,
            2: 0.2,
            3: 0.2
        };
        
        // 每步正方形的描边宽度
        const squareStrokeWidth = {
            0: 2,
            1: 2,
            2: 2,
            3: 4
        };
        
        // 获取元素
        const squareLeft = document.getElementById('scene1-square-left');
        const squareRight = document.getElementById('scene1-square-right');
        const stepIndicator = document.getElementById('scene1-step-indicator');
        const btnPrevGroup = document.getElementById('scene1-btn-prev-group');
        const btnNextGroup = document.getElementById('scene1-btn-next-group');
        
        // 显示元素函数
        function showElement(id) {
            const el = document.getElementById(id);
            if (el) {
                el.classList.remove('hidden');
                el.classList.add('visible', 'fade-in');
            }
        }
        
        // 隐藏元素函数
        function hideElement(id) {
            const el = document.getElementById(id);
            if (el) {
                el.classList.remove('visible', 'fade-in');
                el.classList.add('hidden');
            }
        }
        
        // 更新步骤显示
        function updateStep() {
            // 更新正方形样式
            squareLeft.setAttribute('fill-opacity', squareOpacity[currentState]);
            squareRight.setAttribute('fill-opacity', squareOpacity[currentState]);
            squareLeft.setAttribute('stroke-width', squareStrokeWidth[currentState]);
            squareRight.setAttribute('stroke-width', squareStrokeWidth[currentState]);
            
            // 步骤 0：隐藏所有标注
            if (currentState === 0) {
                for (let step = 1; step <= 3; step++) {
                    steps[step].forEach(id => hideElement(id));
                }
            }
            // 步骤 1-3：显示对应步骤及之前步骤的元素
            else {
                for (let step = 1; step <= currentState; step++) {
                    steps[step].forEach(id => showElement(id));
                }
                // 隐藏后续步骤的元素
                for (let step = currentState + 1; step <= 3; step++) {
                    steps[step].forEach(id => hideElement(id));
                }
            }
            
            // 更新步骤指示器
            stepIndicator.textContent = '步骤 ' + (currentState + 1) + '/4';
            
            // 更新按钮状态
            updateButtons();
        }
        
        // 更新按钮状态
        function updateButtons() {
            // 上一步按钮
            if (currentState === 0) {
                btnPrevGroup.style.opacity = '0.5';
                btnPrevGroup.style.cursor = 'not-allowed';
            } else {
                btnPrevGroup.style.opacity = '1';
                btnPrevGroup.style.cursor = 'pointer';
            }
            
            // 下一步按钮
            if (currentState === totalSteps - 1) {
                btnNextGroup.style.opacity = '0.5';
                btnNextGroup.style.cursor = 'not-allowed';
            } else {
                btnNextGroup.style.opacity = '1';
                btnNextGroup.style.cursor = 'pointer';
            }
        }
        
        // 上一步
        function prevStep() {
            if (currentState > 0) {
                currentState--;
                updateStep();
            }
        }
        
        // 下一步
        function nextStep() {
            if (currentState < totalSteps - 1) {
                currentState++;
                updateStep();
            }
        }
        
        // 绑定按钮事件
        btnPrevGroup.addEventListener('click', prevStep);
        btnNextGroup.addEventListener('click', nextStep);
        
        // 初始化
        updateStep();
    })();
    </script>
