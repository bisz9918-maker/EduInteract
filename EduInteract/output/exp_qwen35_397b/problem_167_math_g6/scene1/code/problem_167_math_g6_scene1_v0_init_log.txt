    <script>
    (function() {
        // Scene 1 实现
        // 布局坐标表（实现对照依据）：
        // 画布尺寸：800×500px
        // 网格参数：每格 80×80px，6 列×2 行，网格区域 480×160px
        // 网格起始位置：x=160, y=150
        // 甲三角形：顶点 (160,150)-(400,150)-(400,310)
        // 乙三角形 1：顶点 (480,230)-(560,230)-(560,310)
        // 乙三角形 2：顶点 (560,230)-(640,230)-(640,310)

        let currentStep = 0;
        const totalSteps = 5;

        const btnPrev = document.getElementById('scene1-btn-prev');
        const btnNext = document.getElementById('scene1-btn-next');
        const stepText = document.getElementById('scene1-step-text');

        // 元素引用
        const triJia = document.getElementById('scene1-tri-jia');
        const dimJia = document.getElementById('scene1-dim-jia');
        const dimYi = document.getElementById('scene1-dim-yi');
        const formulaGroup = document.getElementById('scene1-formula-group');
        const formulaText = document.getElementById('scene1-formula-text');
        const conclusion = document.getElementById('scene1-conclusion');

        // 工具函数：显示元素
        function show(el) {
            el.style.display = 'block';
            // 对于 SVG 元素，使用 opacity 控制可见性
            if (el.tagName.toLowerCase() === 'g') {
                el.setAttribute('opacity', '1');
            } else {
                el.style.opacity = '1';
            }
            el.classList.add('scene1-fade-in');
        }

        // 工具函数：隐藏元素
        function hide(el) {
            el.style.display = 'none';
            if (el.tagName.toLowerCase() === 'g') {
                el.setAttribute('opacity', '0');
            } else {
                el.style.opacity = '0';
            }
            el.classList.remove('scene1-fade-in');
        }

        // 更新显示状态
        function updateDisplay() {
            // 重置所有元素状态
            triJia.setAttribute('stroke', '#1565c0');
            triJia.setAttribute('stroke-width', '1');

            hide(dimJia);
            hide(dimYi);
            hide(formulaGroup);
            hide(conclusion);

            // 根据步骤设置显示
            switch(currentStep) {
                case 0:
                    // 初始状态：只显示网格和三角形
                    break;

                case 1:
                    // 高亮甲并标注底高
                    triJia.setAttribute('stroke', '#000000');
                    triJia.setAttribute('stroke-width', '2');
                    show(dimJia);
                    break;

                case 2:
                    // 显示甲面积计算
                    triJia.setAttribute('stroke', '#000000');
                    triJia.setAttribute('stroke-width', '2');
                    show(dimJia);
                    formulaText.textContent = 'S_甲 = 底×高÷2 = 3×2÷2 = 3';
                    show(formulaGroup);
                    break;

                case 3:
                    // 高亮乙并标注底高
                    show(dimYi);
                    break;

                case 4:
                    // 显示乙面积计算
                    show(dimYi);
                    formulaText.textContent = 'S_乙 = (底×高÷2)×2 = (1×1÷2)×2 = 1';
                    show(formulaGroup);
                    break;

                case 5:
                    // 面积对比结论
                    show(dimYi);
                    formulaText.textContent = 'S_乙 = (底×高÷2)×2 = (1×1÷2)×2 = 1';
                    show(formulaGroup);
                    show(conclusion);
                    break;
            }

            // 更新按钮状态
            btnPrev.disabled = currentStep === 0;
            btnNext.disabled = currentStep === totalSteps;
            stepText.textContent = '步骤 ' + currentStep + '/' + totalSteps;
        }

        // 事件绑定
        btnPrev.addEventListener('click', function() {
            if (currentStep > 0) {
                currentStep--;
                updateDisplay();
            }
        });

        btnNext.addEventListener('click', function() {
            if (currentStep < totalSteps) {
                currentStep++;
                updateDisplay();
            }
        });

        // 初始化
        updateDisplay();
    })();
    </script>
