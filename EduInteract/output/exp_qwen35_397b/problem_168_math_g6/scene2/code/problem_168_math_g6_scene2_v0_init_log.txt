    <script>
    (function() {
        // Scene 2 实现 - 圆周长展开对比演示
        
        // 布局坐标表（实现对照依据）：
        // | 元素 | 类型 | 位置/范围 | 大小 | 备注 |
        // |------|------|-----------|------|------|
        // | 大圆 | 圆 | 中心(400,160) | r=75 | 黑色轮廓 2px |
        // | 小圆1(左) | 圆 | 中心(355,160) | r=30 | 黑色轮廓 2px，在大圆内 |
        // | 小圆2(右) | 圆 | 中心(445,160) | r=45 | 黑色轮廓 2px，在大圆内 |
        // | 直径线 | 线段 | (325,160)-(475,160) | 150×2 | 贯穿大圆 |
        // | 大圆直径标注 D | text | (400,60) | ~20×16 | 大圆顶部上方 25px |
        // | 小圆1直径标注 d₁ | text | (355,80) | ~18×16 | 小圆1顶部上方 50px |
        // | 小圆2直径标注 d₂ | text | (445,75) | ~18×16 | 小圆2顶部上方 40px |
        // | 大圆周长展开条 | 矩形 | (100,290)-(700,320) | 600×30 | 蓝色填充 20% |
        // | 大圆周长标签 πD | text | (400,280) | ~35×16 | 展开条中点上方 |
        // | 小圆1周长展开条 | 矩形 | (100,340)-(340,370) | 240×30 | 绿色填充 20% |
        // | 小圆2周长展开条 | 矩形 | (340,340)-(700,370) | 360×30 | 绿色填充 20% |
        // | 小圆1周长标签 πd₁ | text | (220,330) | ~30×16 | 展开条中点上方 |
        // | 小圆2周长标签 πd₂ | text | (520,330) | ~30×16 | 展开条中点上方 |
        // | 对比对齐线1 | 虚线 | (100,320)-(100,340) | 1×20 | 左端对齐 |
        // | 对比对齐线2 | 虚线 | (700,320)-(700,370) | 1×50 | 右端对齐 |
        // | 等号标识 | text | (720,345) | ~25×30 | 红色加粗 |
        // | 推导公式 | text | (400,395) | ~300×20 | 居中显示 |
        // | 结论文本 | text | (400,430) | ~80×18 | 红色加粗居中，与公式间距 35px |
        // | 上一步按钮 | 按钮 | (280,460) | 80×30 | 控件栏左侧 |
        // | 下一步按钮 | 按钮 | (440,460) | 80×30 | 控件栏右侧 |
        
        // 状态机：currentStep (0-4)
        let currentStep = 0;
        const totalSteps = 5;
        
        // 每步可见元素配置
        // 步骤 0：三个圆 + 直径线 + 直径标注
        // 步骤 1：+ 大圆周长展开条 + πD 标签，圆透明度降低
        // 步骤 2：+ 两个小圆周长展开条 + πd₁/πd₂ 标签
        // 步骤 3：+ 对比对齐虚线 + 等号
        // 步骤 4：+ 推导公式 + 结论
        
        const stepConfig = {
            0: {
                visible: ['scene2-circle-big', 'scene2-circle-small1', 'scene2-circle-small2',
                          'scene2-diameter-line', 'scene2-label-D', 'scene2-label-d1', 'scene2-label-d2'],
                fade: []
            },
            1: {
                visible: ['scene2-circle-big', 'scene2-circle-small1', 'scene2-circle-small2',
                          'scene2-diameter-line', 'scene2-label-D', 'scene2-label-d1', 'scene2-label-d2',
                          'scene2-expand-big', 'scene2-label-piD'],
                fade: ['scene2-circle-big', 'scene2-circle-small1', 'scene2-circle-small2',
                       'scene2-diameter-line', 'scene2-label-D', 'scene2-label-d1', 'scene2-label-d2']
            },
            2: {
                visible: ['scene2-circle-big', 'scene2-circle-small1', 'scene2-circle-small2',
                          'scene2-diameter-line', 'scene2-label-D', 'scene2-label-d1', 'scene2-label-d2',
                          'scene2-expand-big', 'scene2-label-piD',
                          'scene2-expand-small1', 'scene2-label-pid1',
                          'scene2-expand-small2', 'scene2-label-pid2'],
                fade: ['scene2-circle-big', 'scene2-circle-small1', 'scene2-circle-small2',
                       'scene2-diameter-line', 'scene2-label-D', 'scene2-label-d1', 'scene2-label-d2']
            },
            3: {
                visible: ['scene2-circle-big', 'scene2-circle-small1', 'scene2-circle-small2',
                          'scene2-diameter-line', 'scene2-label-D', 'scene2-label-d1', 'scene2-label-d2',
                          'scene2-expand-big', 'scene2-label-piD',
                          'scene2-expand-small1', 'scene2-label-pid1',
                          'scene2-expand-small2', 'scene2-label-pid2',
                          'scene2-align-left', 'scene2-align-right', 'scene2-equals-sign'],
                fade: ['scene2-circle-big', 'scene2-circle-small1', 'scene2-circle-small2',
                       'scene2-diameter-line', 'scene2-label-D', 'scene2-label-d1', 'scene2-label-d2']
            },
            4: {
                visible: ['scene2-circle-big', 'scene2-circle-small1', 'scene2-circle-small2',
                          'scene2-diameter-line', 'scene2-label-D', 'scene2-label-d1', 'scene2-label-d2',
                          'scene2-expand-big', 'scene2-label-piD',
                          'scene2-expand-small1', 'scene2-label-pid1',
                          'scene2-expand-small2', 'scene2-label-pid2',
                          'scene2-align-left', 'scene2-align-right', 'scene2-equals-sign',
                          'scene2-formula', 'scene2-conclusion'],
                fade: ['scene2-circle-big', 'scene2-circle-small1', 'scene2-circle-small2',
                       'scene2-diameter-line', 'scene2-label-D', 'scene2-label-d1', 'scene2-label-d2']
            }
        };
        
        // 工具函数：显示元素
        function showElement(id) {
            const el = document.getElementById(id);
            if (el) {
                el.classList.remove('scene2-hidden');
                el.classList.add('scene2-visible');
            }
        }
        
        // 工具函数：隐藏元素
        function hideElement(id) {
            const el = document.getElementById(id);
            if (el) {
                el.classList.remove('scene2-visible');
                el.classList.add('scene2-hidden');
            }
        }
        
        // 工具函数：设置淡入淡出
        function setFade(id, isFaded) {
            const el = document.getElementById(id);
            if (el) {
                if (isFaded) {
                    el.classList.add('scene2-fade');
                } else {
                    el.classList.remove('scene2-fade');
                }
            }
        }
        
        // 更新步骤显示
        function updateStep() {
            const config = stepConfig[currentStep];
            
            // 获取所有可控元素 ID
            const allIds = [];
            for (let s = 0; s < totalSteps; s++) {
                stepConfig[s].visible.forEach(id => {
                    if (!allIds.includes(id)) allIds.push(id);
                });
            }
            
            // 先隐藏所有元素
            allIds.forEach(id => hideElement(id));
            
            // 显示当前步骤的元素
            config.visible.forEach(id => showElement(id));
            
            // 设置淡出效果
            allIds.forEach(id => {
                const isFaded = config.fade.includes(id);
                setFade(id, isFaded);
            });
            
            // 更新步骤指示器
            document.getElementById('scene2-step-indicator').textContent = 
                `步骤 ${currentStep + 1}/${totalSteps}`;
            
            // 更新按钮状态
            document.getElementById('scene2-btn-prev').disabled = (currentStep === 0);
            document.getElementById('scene2-btn-next').disabled = (currentStep === totalSteps - 1);
        }
        
        // 上一步
        function prevStep() {
            if (currentStep > 0) {
                currentStep--;
                updateStep();
            }
        }
        
        // 下一步
        function nextStep() {
            if (currentStep < totalSteps - 1) {
                currentStep++;
                updateStep();
            }
        }
        
        // 绑定事件
        document.getElementById('scene2-btn-prev').addEventListener('click', prevStep);
        document.getElementById('scene2-btn-next').addEventListener('click', nextStep);
        
        // 初始化
        updateStep();
    })();
    </script>
