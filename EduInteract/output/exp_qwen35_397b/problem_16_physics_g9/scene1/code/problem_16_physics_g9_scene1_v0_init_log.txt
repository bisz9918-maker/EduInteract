    <script>
    (function() {
        // Scene 1 实现 - 电路结构识别交互图示

        // 状态变量
        let currentStep = 0;
        const totalSteps = 4;

        // DOM 元素引用
        const btnPrev = document.getElementById('scene1-btn-prev');
        const btnNext = document.getElementById('scene1-btn-next');
        const stepIndicator = document.getElementById('scene1-step-indicator');
        const currentPath = document.getElementById('scene1-current-path');
        const equivalentCircuit = document.getElementById('scene1-equivalent-circuit');
        const transitionArrow = document.getElementById('scene1-transition-arrow');
        const arrowPolygon = transitionArrow.querySelector('polygon');
        const v1Box = document.getElementById('scene1-v1-box');
        const v2Box = document.getElementById('scene1-v2-box');
        const realCircuit = document.getElementById('scene1-real-circuit');

        // 工具函数：显示元素
        function show(el) {
            el.classList.remove('scene1-hidden');
            el.classList.add('scene1-visible');
            el.style.display = 'block';
            el.style.opacity = '1';
        }

        // 工具函数：隐藏元素
        function hide(el) {
            el.classList.remove('scene1-visible');
            el.classList.add('scene1-hidden');
            el.style.display = 'none';
            el.style.opacity = '0';
        }

        // 更新步骤显示
        function updateStepDisplay() {
            stepIndicator.textContent = `步骤 ${currentStep + 1}/${totalSteps}`;
            
            // 更新按钮状态
            btnPrev.disabled = (currentStep === 0);
            btnNext.disabled = (currentStep === totalSteps - 1);

            // 根据步骤显示/隐藏元素
            switch(currentStep) {
                case 0:
                    // 初始状态：只显示实物图
                    hide(currentPath);
                    hide(equivalentCircuit);
                    arrowPolygon.classList.remove('scene1-arrow-green');
                    arrowPolygon.classList.add('scene1-arrow-gray');
                    realCircuit.classList.remove('scene1-semi-transparent');
                    hide(v1Box);
                    hide(v2Box);
                    break;
                case 1:
                    // 步骤1：高亮电流路径
                    show(currentPath);
                    hide(equivalentCircuit);
                    arrowPolygon.classList.remove('scene1-arrow-green');
                    arrowPolygon.classList.add('scene1-arrow-gray');
                    realCircuit.classList.remove('scene1-semi-transparent');
                    hide(v1Box);
                    hide(v2Box);
                    break;
                case 2:
                    // 步骤2：显示等效电路图
                    show(currentPath);
                    show(equivalentCircuit);
                    arrowPolygon.classList.remove('scene1-arrow-gray');
                    arrowPolygon.classList.add('scene1-arrow-green');
                    realCircuit.classList.add('scene1-semi-transparent');
                    hide(v1Box);
                    hide(v2Box);
                    break;
                case 3:
                    // 步骤3：标注电表测量对象
                    show(currentPath);
                    show(equivalentCircuit);
                    arrowPolygon.classList.remove('scene1-arrow-gray');
                    arrowPolygon.classList.add('scene1-arrow-green');
                    realCircuit.classList.add('scene1-semi-transparent');
                    show(v1Box);
                    show(v2Box);
                    break;
            }
        }

        // 上一步按钮点击事件
        btnPrev.addEventListener('click', function() {
            if (currentStep > 0) {
                currentStep--;
                updateStepDisplay();
            }
        });

        // 下一步按钮点击事件
        btnNext.addEventListener('click', function() {
            if (currentStep < totalSteps - 1) {
                currentStep++;
                updateStepDisplay();
            }
        });

        // 初始化显示
        updateStepDisplay();
    })();
    </script>
