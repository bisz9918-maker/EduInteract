    <script>
    (function() {
        // Scene 1 实现 - 物质跨膜运输方式分类

        // 状态变量
        let currentStep = 0;
        const totalSteps = 5;

        // 步骤定义：每步显示的元素
        const steps = {
            0: {
                // 初始状态：仅顶层
                boxes: ['scene1-box-top', 'scene1-text-top'],
                groups: [],
                lines: [],
                nums: [],
                labels: [],
                passiveHighlight: false,
                activeHighlight: false,
                meaningHighlight: false
            },
            1: {
                // 第二层：被动运输、①
                boxes: ['scene1-box-top', 'scene1-text-top'],
                groups: ['scene1-group-passive', 'scene1-group-active'],
                lines: ['scene1-line-1', 'scene1-line-2'],
                nums: ['scene1-num-1', 'scene1-num-1-text'],
                labels: [],
                passiveHighlight: false,
                activeHighlight: false,
                meaningHighlight: false
            },
            2: {
                // 第三层：自由扩散、②
                boxes: ['scene1-box-top', 'scene1-text-top'],
                groups: ['scene1-group-passive', 'scene1-group-active', 'scene1-group-free', 'scene1-group-facilitated'],
                lines: ['scene1-line-1', 'scene1-line-2', 'scene1-line-3', 'scene1-line-4'],
                nums: ['scene1-num-1', 'scene1-num-1-text', 'scene1-num-2', 'scene1-num-2-text'],
                labels: [],
                passiveHighlight: true,
                activeHighlight: false,
                meaningHighlight: false
            },
            3: {
                // 第四层：氧气、③、④
                boxes: ['scene1-box-top', 'scene1-text-top'],
                groups: ['scene1-group-passive', 'scene1-group-active', 'scene1-group-free', 'scene1-group-facilitated', 'scene1-group-oxygen', 'scene1-group-glucose', 'scene1-group-ion'],
                lines: ['scene1-line-1', 'scene1-line-2', 'scene1-line-3', 'scene1-line-4', 'scene1-line-5', 'scene1-line-6', 'scene1-line-7'],
                nums: ['scene1-num-1', 'scene1-num-1-text', 'scene1-num-2', 'scene1-num-2-text', 'scene1-num-3', 'scene1-num-3-text', 'scene1-num-4', 'scene1-num-4-text'],
                labels: [],
                passiveHighlight: false,
                activeHighlight: true,
                meaningHighlight: false
            },
            4: {
                // 含义标注①、②
                boxes: ['scene1-box-top', 'scene1-text-top'],
                groups: ['scene1-group-passive', 'scene1-group-active', 'scene1-group-free', 'scene1-group-facilitated', 'scene1-group-oxygen', 'scene1-group-glucose', 'scene1-group-ion'],
                lines: ['scene1-line-1', 'scene1-line-2', 'scene1-line-3', 'scene1-line-4', 'scene1-line-5', 'scene1-line-6', 'scene1-line-7'],
                nums: ['scene1-num-1', 'scene1-num-1-text', 'scene1-num-2', 'scene1-num-2-text', 'scene1-num-3', 'scene1-num-3-text', 'scene1-num-4', 'scene1-num-4-text'],
                labels: ['scene1-label-1', 'scene1-label-2'],
                passiveHighlight: false,
                activeHighlight: false,
                meaningHighlight: true
            },
            5: {
                // 含义标注③、④ - 完整状态
                boxes: ['scene1-box-top', 'scene1-text-top'],
                groups: ['scene1-group-passive', 'scene1-group-active', 'scene1-group-free', 'scene1-group-facilitated', 'scene1-group-oxygen', 'scene1-group-glucose', 'scene1-group-ion'],
                lines: ['scene1-line-1', 'scene1-line-2', 'scene1-line-3', 'scene1-line-4', 'scene1-line-5', 'scene1-line-6', 'scene1-line-7'],
                nums: ['scene1-num-1', 'scene1-num-1-text', 'scene1-num-2', 'scene1-num-2-text', 'scene1-num-3', 'scene1-num-3-text', 'scene1-num-4', 'scene1-num-4-text'],
                labels: ['scene1-label-1', 'scene1-label-2', 'scene1-label-3', 'scene1-label-4'],
                passiveHighlight: false,
                activeHighlight: false,
                meaningHighlight: true
            }
        };

        // 工具函数：显示元素
        function showElement(id) {
            const el = document.getElementById(id);
            if (el) {
                el.classList.remove('scene1-hidden');
                el.classList.add('scene1-visible');
            }
        }

        // 工具函数：隐藏元素
        function hideElement(id) {
            const el = document.getElementById(id);
            if (el) {
                el.classList.remove('scene1-visible');
                el.classList.add('scene1-hidden');
            }
        }

        // 更新步骤显示
        function updateStep(step) {
            const stepData = steps[step];

            // 隐藏所有可选元素
            const allGroups = ['scene1-group-passive', 'scene1-group-active', 'scene1-group-free', 'scene1-group-facilitated', 'scene1-group-oxygen', 'scene1-group-glucose', 'scene1-group-ion'];
            const allLines = ['scene1-line-1', 'scene1-line-2', 'scene1-line-3', 'scene1-line-4', 'scene1-line-5', 'scene1-line-6', 'scene1-line-7'];
            const allNums = ['scene1-num-1', 'scene1-num-1-text', 'scene1-num-2', 'scene1-num-2-text', 'scene1-num-3', 'scene1-num-3-text', 'scene1-num-4', 'scene1-num-4-text'];
            const allLabels = ['scene1-label-1', 'scene1-label-2', 'scene1-label-3', 'scene1-label-4'];

            allGroups.forEach(id => hideElement(id));
            allLines.forEach(id => hideElement(id));
            allNums.forEach(id => hideElement(id));
            allLabels.forEach(id => hideElement(id));

            // 移除高亮
            document.getElementById('scene1-box-passive').classList.remove('scene1-box-passive-highlight');
            document.getElementById('scene1-box-active').classList.remove('scene1-box-active-highlight');
            document.getElementById('scene1-box-active').classList.remove('scene1-box-meaning-highlight');
            document.getElementById('scene1-box-facilitated').classList.remove('scene1-box-meaning-highlight');

            // 显示当前步骤的元素
            stepData.boxes.forEach(id => showElement(id));
            stepData.groups.forEach(id => showElement(id));
            stepData.lines.forEach(id => showElement(id));
            stepData.nums.forEach(id => showElement(id));
            stepData.labels.forEach(id => showElement(id));

            // 应用高亮
            if (stepData.passiveHighlight) {
                document.getElementById('scene1-box-passive').classList.add('scene1-box-passive-highlight');
            }
            if (stepData.activeHighlight) {
                document.getElementById('scene1-box-active').classList.add('scene1-box-active-highlight');
            }
            if (stepData.meaningHighlight) {
                document.getElementById('scene1-box-active').classList.add('scene1-box-meaning-highlight');
                document.getElementById('scene1-box-facilitated').classList.add('scene1-box-meaning-highlight');
            }

            // 更新步骤指示器
            document.getElementById('scene1-step-indicator').textContent = '步骤 ' + step + '/' + totalSteps;

            // 更新按钮状态
            document.getElementById('scene1-btn-prev').disabled = (step === 0);
            document.getElementById('scene1-btn-next').disabled = (step === totalSteps);
        }

        // 事件绑定
        document.getElementById('scene1-btn-prev').addEventListener('click', function() {
            if (currentStep > 0) {
                currentStep--;
                updateStep(currentStep);
            }
        });

        document.getElementById('scene1-btn-next').addEventListener('click', function() {
            if (currentStep < totalSteps) {
                currentStep++;
                updateStep(currentStep);
            }
        });

        // 初始化
        updateStep(0);
    })();
    </script>
