    <script>
    (function() {
        // Scene 3 实现 - 物质跨膜运输方式选项分析

        // 状态变量
        let currentStep = 0;
        const totalSteps = 5;

        // 每步需要显示的元素 ID 映射
        const stepElements = {
            1: ['scene3-mark-a', 'scene3-reason-a', 'scene3-arrow-a'],
            2: ['scene3-mark-b', 'scene3-reason-b', 'scene3-arrow-b', 'scene3-highlight-b'],
            3: ['scene3-mark-c', 'scene3-reason-c', 'scene3-arrow-c'],
            4: ['scene3-mark-d', 'scene3-reason-d', 'scene3-arrow-d'],
            5: ['scene3-conclusion']
        };

        // 显示元素工具函数
        function showElement(id) {
            const el = document.getElementById(id);
            if (el) {
                el.style.display = 'block';
                el.style.opacity = '1';
                el.classList.add('scene3-fade-in');
            }
        }

        // 隐藏元素工具函数
        function hideElement(id) {
            const el = document.getElementById(id);
            if (el) {
                el.style.display = 'none';
                el.style.opacity = '0';
                el.classList.remove('scene3-fade-in');
            }
        }

        // 更新按钮状态
        function updateButtons() {
            const prevBg = document.getElementById('scene3-btn-prev-bg');
            const prevText = document.getElementById('scene3-btn-prev-text');
            const nextBg = document.getElementById('scene3-btn-next-bg');
            const nextText = document.getElementById('scene3-btn-next-text');

            if (currentStep === 0) {
                prevBg.classList.add('scene3-btn-disabled');
                prevText.style.fill = '#cccccc';
            } else {
                prevBg.classList.remove('scene3-btn-disabled');
                prevText.style.fill = '#ffffff';
            }

            if (currentStep === totalSteps) {
                nextBg.classList.add('scene3-btn-disabled');
                nextText.style.fill = '#cccccc';
            } else {
                nextBg.classList.remove('scene3-btn-disabled');
                nextText.style.fill = '#ffffff';
            }
        }

        // 更新步骤指示器
        function updateStepIndicator() {
            const indicator = document.getElementById('scene3-step-indicator');
            if (indicator) {
                indicator.textContent = '步骤 ' + currentStep + '/' + totalSteps;
            }
        }

        // 更新显示状态
        function updateDisplay() {
            // 隐藏所有步骤相关元素
            for (let step = 1; step <= totalSteps; step++) {
                if (stepElements[step]) {
                    stepElements[step].forEach(function(id) {
                        hideElement(id);
                    });
                }
            }

            // 显示当前步骤及之前步骤的元素
            for (let step = 1; step <= currentStep; step++) {
                if (stepElements[step]) {
                    stepElements[step].forEach(function(id) {
                        showElement(id);
                    });
                }
            }

            updateButtons();
            updateStepIndicator();
        }

        // 更新步骤
        function updateStep(delta) {
            const newStep = currentStep + delta;
            if (newStep >= 0 && newStep <= totalSteps) {
                currentStep = newStep;
                updateDisplay();
            }
        }

        // 事件绑定
        document.getElementById('scene3-btn-prev-bg').addEventListener('click', function() {
            updateStep(-1);
        });

        document.getElementById('scene3-btn-next-bg').addEventListener('click', function() {
            updateStep(1);
        });

        // 初始化
        updateDisplay();
    })();
    </script>
