    <script>
    (function() {
        // Scene 1 实现 - 突触结构详解
        
        // 布局坐标表（实现对照依据）：
        // | 元素 | 位置 | 备注 |
        // |------|------|------|
        // | 神经元 A | (80,280)-(180,340) | 绿色 #2e7d32 |
        // | 神经元 B | (200,180)-(320,360) | 蓝色 #1565c0 |
        // | 神经元 C | (380,200)-(520,320) | 红色 #c62828 |
        // | 电流计 | (400,60) 半径 25 | 顶部中央 |
        // | a 点 | (80,260) | 标签 (65,255) |
        // | d 点 | (550,260) | 标签 (560,255) |
        // | b 点 | (260,160) | 标签 (260,128) |
        // | c 点 | (420,180) | 标签 (420,148) |
        // | 突触 1 | (185,305) 椭圆 35×40 | 黄色高亮 |
        // | 突触 2 | (365,275) 椭圆 40×45 | 黄色高亮 |
        // | 放大图 | (620,150)-(780,330) | 步骤 3 显示 |
        // | 上一步按钮 | (280,455) 80×30 | 控件栏左侧 |
        // | 下一步按钮 | (440,455) 80×30 | 控件栏右侧 |

        let currentStep = 0;
        const totalSteps = 3;

        // 工具函数：显示元素
        function show(el) {
            if (el) {
                el.classList.remove('scene1-hidden');
                el.classList.add('scene1-visible');
            }
        }

        // 工具函数：隐藏元素
        function hide(el) {
            if (el) {
                el.classList.remove('scene1-visible');
                el.classList.add('scene1-hidden');
            }
        }

        // 更新步骤显示
        function updateStepDisplay() {
            document.getElementById('scene1-step-indicator').textContent = '步骤 ' + currentStep + '/' + totalSteps;

            // 步骤 0：初始状态
            if (currentStep === 0) {
                // 隐藏所有彩色高亮和标签
                hide(document.getElementById('scene1-label-neuron-a'));
                hide(document.getElementById('scene1-label-neuron-b'));
                hide(document.getElementById('scene1-label-neuron-c'));
                hide(document.getElementById('scene1-synapses'));
                hide(document.getElementById('scene1-synapse-detail'));
                
                // 神经元恢复灰度
                document.getElementById('scene1-neuron-a').setAttribute('stroke', '#999');
                document.getElementById('scene1-neuron-b').setAttribute('stroke', '#999');
                document.getElementById('scene1-neuron-c').setAttribute('stroke', '#999');
            }

            // 步骤 1：高亮 3 个神经元
            if (currentStep >= 1) {
                show(document.getElementById('scene1-label-neuron-a'));
                show(document.getElementById('scene1-label-neuron-b'));
                show(document.getElementById('scene1-label-neuron-c'));
                
                // 神经元彩色高亮
                document.getElementById('scene1-neuron-a').setAttribute('stroke', '#2e7d32');
                document.getElementById('scene1-neuron-b').setAttribute('stroke', '#1565c0');
                document.getElementById('scene1-neuron-c').setAttribute('stroke', '#c62828');
            }

            // 步骤 2：高亮 2 个突触
            if (currentStep >= 2) {
                show(document.getElementById('scene1-synapses'));
                show(document.getElementById('scene1-label-synapse-1'));
                show(document.getElementById('scene1-label-synapse-2'));
                show(document.getElementById('scene1-synapse-count'));
            }

            // 步骤 3：放大突触结构
            if (currentStep >= 3) {
                show(document.getElementById('scene1-synapse-detail'));
            }

            // 更新按钮状态
            const prevBtn = document.getElementById('scene1-btn-prev-text');
            const nextBtn = document.getElementById('scene1-btn-next-text');
            
            if (currentStep === 0) {
                prevBtn.parentElement.style.opacity = '0.5';
                prevBtn.parentElement.style.cursor = 'not-allowed';
            } else {
                prevBtn.parentElement.style.opacity = '1';
                prevBtn.parentElement.style.cursor = 'pointer';
            }
            
            if (currentStep === totalSteps) {
                nextBtn.parentElement.style.opacity = '0.5';
                nextBtn.parentElement.style.cursor = 'not-allowed';
            } else {
                nextBtn.parentElement.style.opacity = '1';
                nextBtn.parentElement.style.cursor = 'pointer';
            }
        }

        // 上一步
        function stepBackward() {
            if (currentStep > 0) {
                currentStep--;
                updateStepDisplay();
            }
        }

        // 下一步
        function stepForward() {
            if (currentStep < totalSteps) {
                currentStep++;
                updateStepDisplay();
            }
        }

        // 绑定按钮点击事件
        document.getElementById('scene1-btn-prev-text').parentElement.addEventListener('click', function() {
            if (currentStep > 0) {
                stepBackward();
            }
        });

        document.getElementById('scene1-btn-next-text').parentElement.addEventListener('click', function() {
            if (currentStep < totalSteps) {
                stepForward();
            }
        });

        // 初始化显示
        updateStepDisplay();
    })();
    </script>
