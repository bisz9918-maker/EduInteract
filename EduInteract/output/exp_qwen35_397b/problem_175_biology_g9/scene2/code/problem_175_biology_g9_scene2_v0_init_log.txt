    <script>
    (function() {
        // Scene 2 实现：植物类群分类对比表
        
        // 状态变量
        let currentStep = 0;
        const totalSteps = 5;
        
        // 获取所有需要控制的元素
        const rowElements = {
            1: [
                document.getElementById('scene2-row1-bg'),
                document.getElementById('scene2-row1-name'),
                document.getElementById('scene2-row1-num-circle'),
                document.getElementById('scene2-row1-num-text'),
                document.getElementById('scene2-row1-seed'),
                document.getElementById('scene2-row1-stem'),
                document.getElementById('scene2-row1-vascular'),
                document.getElementById('scene2-row1-root'),
                document.getElementById('scene2-row1-habitat')
            ],
            2: [
                document.getElementById('scene2-row2-bg'),
                document.getElementById('scene2-row2-name'),
                document.getElementById('scene2-row2-num-circle'),
                document.getElementById('scene2-row2-num-text'),
                document.getElementById('scene2-row2-seed'),
                document.getElementById('scene2-row2-stem'),
                document.getElementById('scene2-row2-vascular'),
                document.getElementById('scene2-row2-root'),
                document.getElementById('scene2-row2-habitat')
            ],
            3: [
                document.getElementById('scene2-row3-bg'),
                document.getElementById('scene2-row3-name'),
                document.getElementById('scene2-row3-num-circle'),
                document.getElementById('scene2-row3-num-text'),
                document.getElementById('scene2-row3-seed'),
                document.getElementById('scene2-row3-stem'),
                document.getElementById('scene2-row3-vascular'),
                document.getElementById('scene2-row3-root'),
                document.getElementById('scene2-row3-habitat')
            ],
            4: [
                document.getElementById('scene2-row4-bg'),
                document.getElementById('scene2-row4-name'),
                document.getElementById('scene2-row4-num-circle'),
                document.getElementById('scene2-row4-num-text'),
                document.getElementById('scene2-row4-seed'),
                document.getElementById('scene2-row4-stem'),
                document.getElementById('scene2-row4-vascular'),
                document.getElementById('scene2-row4-root'),
                document.getElementById('scene2-row4-habitat')
            ]
        };
        
        const step5Elements = [
            document.getElementById('scene2-tree-group'),
            document.getElementById('scene2-line2'),
            document.getElementById('scene2-line3'),
            document.getElementById('scene2-line4')
        ];
        
        const btnPrevBg = document.getElementById('scene2-btn-prev-bg');
        const btnPrevText = document.getElementById('scene2-btn-prev-text');
        const btnNextBg = document.getElementById('scene2-btn-next-bg');
        const btnNextText = document.getElementById('scene2-btn-next-text');
        
        // 辅助函数：显示元素
        function showElements(elements) {
            elements.forEach(el => {
                if (el) {
                    el.classList.add('visible');
                    el.style.display = '';
                    el.style.opacity = '1';
                }
            });
        }
        
        // 辅助函数：隐藏元素
        function hideElements(elements) {
            elements.forEach(el => {
                if (el) {
                    el.classList.remove('visible');
                    el.style.opacity = '0';
                }
            });
        }
        
        // 辅助函数：设置高亮行
        function setHighlight(rowNum) {
            // 移除所有行的高亮
            for (let i = 1; i <= 4; i++) {
                if (rowElements[i][0]) {
                    rowElements[i][0].classList.remove('scene2-highlight');
                }
            }
            // 添加当前行高亮
            if (rowNum >= 1 && rowNum <= 4 && rowElements[rowNum][0]) {
                rowElements[rowNum][0].classList.add('scene2-highlight');
            }
        }
        
        // 辅助函数：更新按钮状态
        function updateButtons() {
            // 上一步按钮
            if (currentStep === 0) {
                btnPrevBg.classList.add('scene2-btn-disabled');
                btnPrevText.style.fill = '#cccccc';
            } else {
                btnPrevBg.classList.remove('scene2-btn-disabled');
                btnPrevText.style.fill = '#ffffff';
            }
            
            // 下一步按钮
            if (currentStep === totalSteps) {
                btnNextBg.classList.add('scene2-btn-disabled');
                btnNextText.style.fill = '#cccccc';
            } else {
                btnNextBg.classList.remove('scene2-btn-disabled');
                btnNextText.style.fill = '#ffffff';
            }
        }
        
        // 更新显示状态
        function updateDisplay() {
            // 隐藏所有行
            for (let i = 1; i <= 4; i++) {
                hideElements(rowElements[i]);
            }
            
            // 隐藏步骤5元素
            hideElements(step5Elements);
            
            // 根据当前步骤显示相应内容
            if (currentStep >= 1) {
                showElements(rowElements[1]);
                setHighlight(1);
            }
            if (currentStep >= 2) {
                showElements(rowElements[2]);
                setHighlight(2);
            }
            if (currentStep >= 3) {
                showElements(rowElements[3]);
                setHighlight(3);
            }
            if (currentStep >= 4) {
                showElements(rowElements[4]);
                setHighlight(4);
            }
            if (currentStep >= 5) {
                showElements(step5Elements);
                setHighlight(-1); // 取消高亮
            }
            
            updateButtons();
        }
        
        // 上一步按钮点击处理
        btnPrevBg.addEventListener('click', function() {
            if (currentStep > 0) {
                currentStep--;
                updateDisplay();
            }
        });
        
        // 下一步按钮点击处理
        btnNextBg.addEventListener('click', function() {
            if (currentStep < totalSteps) {
                currentStep++;
                updateDisplay();
            }
        });
        
        // 初始化显示
        updateDisplay();
        
    })();
    </script>
