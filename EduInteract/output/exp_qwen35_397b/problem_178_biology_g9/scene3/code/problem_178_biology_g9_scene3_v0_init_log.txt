    <script>
    (function() {
        // Scene 3 实现：F₂代黄花与白花比例计算可视化

        // 状态管理
        let currentStep = 0;
        const totalSteps = 4;

        // 每步可见元素配置
        const stepConfig = {
            0: {
                // 初始状态：只显示 F₁ 构成
                visible: ['scene3-f1-title', 'scene3-f1-yellow-box', 'scene3-f1-yellow-label', 'scene3-f1-white-box', 'scene3-f1-white-label'],
                highlights: []
            },
            1: {
                // 步骤 1：显示黄花支系计算
                visible: ['scene3-f1-title', 'scene3-f1-yellow-box', 'scene3-f1-yellow-label', 'scene3-f1-white-box', 'scene3-f1-white-label',
                          'scene3-arrow1', 'scene3-yellow-branch', 'scene3-yellow-branch-title', 'scene3-yellow-calc', 'scene3-yellow-result', 'scene3-yellow-contrib-arrow'],
                highlights: ['scene3-yellow-calc', 'scene3-yellow-result']
            },
            2: {
                // 步骤 2：显示白花支系上半部分（黄花贡献）
                visible: ['scene3-f1-title', 'scene3-f1-yellow-box', 'scene3-f1-yellow-label', 'scene3-f1-white-box', 'scene3-f1-white-label',
                          'scene3-arrow1', 'scene3-arrow2', 'scene3-yellow-branch', 'scene3-yellow-branch-title', 'scene3-yellow-calc', 'scene3-yellow-result', 'scene3-yellow-contrib-arrow',
                          'scene3-white-branch', 'scene3-white-branch-title', 'scene3-white-calc1', 'scene3-white-result1', 'scene3-white-yellow-arrow'],
                highlights: ['scene3-white-calc1', 'scene3-white-result1']
            },
            3: {
                // 步骤 3：显示白花支系下半部分（白花贡献）
                visible: ['scene3-f1-title', 'scene3-f1-yellow-box', 'scene3-f1-yellow-label', 'scene3-f1-white-box', 'scene3-f1-white-label',
                          'scene3-arrow1', 'scene3-arrow2', 'scene3-yellow-branch', 'scene3-yellow-branch-title', 'scene3-yellow-calc', 'scene3-yellow-result', 'scene3-yellow-contrib-arrow',
                          'scene3-white-branch', 'scene3-white-branch-title', 'scene3-white-calc1', 'scene3-white-result1', 'scene3-white-calc2', 'scene3-white-result2',
                          'scene3-white-yellow-arrow', 'scene3-white-white-arrow'],
                highlights: ['scene3-white-calc2', 'scene3-white-result2']
            },
            4: {
                // 步骤 4：显示汇总结果
                visible: ['scene3-f1-title', 'scene3-f1-yellow-box', 'scene3-f1-yellow-label', 'scene3-f1-white-box', 'scene3-f1-white-label',
                          'scene3-arrow1', 'scene3-arrow2', 'scene3-yellow-branch', 'scene3-yellow-branch-title', 'scene3-yellow-calc', 'scene3-yellow-result', 'scene3-yellow-contrib-arrow',
                          'scene3-white-branch', 'scene3-white-branch-title', 'scene3-white-calc1', 'scene3-white-result1', 'scene3-white-calc2', 'scene3-white-result2',
                          'scene3-white-yellow-arrow', 'scene3-white-white-arrow',
                          'scene3-summary', 'scene3-summary-box', 'scene3-summary-title', 'scene3-summary-result'],
                highlights: ['scene3-summary-result']
            }
        };

        // 工具函数：显示元素
        function showElement(id) {
            const el = document.getElementById(id);
            if (el) {
                el.style.opacity = '1';
                if (el.tagName === 'g') {
                    const children = el.querySelectorAll('*');
                    children.forEach(child => {
                        child.style.opacity = '1';
                    });
                }
            }
        }

        // 工具函数：隐藏元素
        function hideElement(id) {
            const el = document.getElementById(id);
            if (el) {
                el.style.opacity = '0';
                if (el.tagName === 'g') {
                    const children = el.querySelectorAll('*');
                    children.forEach(child => {
                        child.style.opacity = '0';
                    });
                }
            }
        }

        // 更新步骤
        function updateStep() {
            const config = stepConfig[currentStep];

            // 隐藏所有元素
            const allElements = [];
            Object.values(stepConfig).forEach(step => {
                step.visible.forEach(id => {
                    if (!allElements.includes(id)) allElements.push(id);
                });
            });
            allElements.forEach(id => hideElement(id));

            // 显示当前步骤的元素
            config.visible.forEach(id => {
                const el = document.getElementById(id);
                if (el) {
                    el.style.opacity = '1';
                    el.classList.add('scene3-fade-in');
                    if (el.tagName === 'g') {
                        const children = el.querySelectorAll('*');
                        children.forEach(child => {
                            child.style.opacity = '1';
                        });
                    }
                }
            });

            // 清除所有高亮
            document.querySelectorAll('.scene3-highlight-yellow, .scene3-highlight-blue, .scene3-highlight-green').forEach(el => {
                el.classList.remove('scene3-highlight-yellow', 'scene3-highlight-blue', 'scene3-highlight-green');
            });

            // 应用高亮
            config.highlights.forEach(id => {
                const el = document.getElementById(id);
                if (el) {
                    if (id.includes('yellow')) {
                        el.classList.add('scene3-highlight-yellow');
                    } else if (id.includes('white') && !id.includes('result2')) {
                        el.classList.add('scene3-highlight-yellow');
                    } else if (id.includes('result2')) {
                        el.classList.add('scene3-highlight-blue');
                    } else if (id.includes('summary')) {
                        el.classList.add('scene3-highlight-green');
                    }
                }
            });

            // 更新步骤指示器
            const indicator = document.getElementById('scene3-step-indicator');
            if (indicator) {
                indicator.textContent = '步骤 ' + currentStep + '/' + totalSteps;
            }

            // 更新按钮状态
            const prevBtn = document.getElementById('scene3-prev-btn');
            const nextBtn = document.getElementById('scene3-next-btn');
            if (prevBtn) prevBtn.style.opacity = currentStep === 0 ? '0.5' : '1';
            if (nextBtn) nextBtn.style.opacity = currentStep === totalSteps ? '0.5' : '1';
        }

        // 按钮点击事件
        document.getElementById('scene3-prev-btn').addEventListener('click', function() {
            if (currentStep > 0) {
                currentStep--;
                updateStep();
            }
        });

        document.getElementById('scene3-next-btn').addEventListener('click', function() {
            if (currentStep < totalSteps) {
                currentStep++;
                updateStep();
            }
        });

        // 初始化
        updateStep();
    })();
    </script>
