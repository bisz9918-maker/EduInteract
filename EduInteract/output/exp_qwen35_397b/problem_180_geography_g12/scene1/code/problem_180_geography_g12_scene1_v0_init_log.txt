    <script>
    (function() {
        // Scene 1 实现：光照图基础结构——晨昏线、昼半球、夜半球的识别
        
        // 布局坐标表（实现对照依据）：
        // | 元素 | 类型 | 位置/范围 | 大小 | 标签偏移方向 | 备注 |
        // |------|------|-----------|------|-------------|------|
        // | 地球球体 | 圆形 | 中心 (400,250) | 半径 140px | - | 居中 |
        // | 夜半球 | 半圆填充 | 球体左半部分 | 半径 140px | - | 灰色阴影 |
        // | 晨昏线 | 弧线 | 球体内倾斜大圆 | 线宽 3px | - | 蓝色粗线 |
        // | 太阳光线 1 | 箭头 | (650,170)→(590,185) | 长 60px | - | 右上 |
        // | 太阳光线 2 | 箭头 | (670,240)→(610,255) | 长 60px | - | 右中 |
        // | 太阳光线 3 | 箭头 | (650,310)→(590,325) | 长 60px | - | 右下 |
        // | 赤道 | 椭圆弧 | 球体中部水平线 | 线宽 1px | - | 虚线 |
        // | N 极点标签 | text | (400,95) | ~20×16 | 上方 | 球体顶部 |
        // | S 极点标签 | text | (400,415) | ~20×16 | 下方 | 球体底部 |
        // | A 点 | 实心圆 | (330,250) | 半径 5px | 右上方 | 晨昏线与赤道交点 |
        // | B 点 | 实心圆 | (300,180) | 半径 5px | 右上方 | 北半球昼半球 |
        // | C 点 | 实心圆 | (540,250) | 半径 5px | 右上方 | 赤道最右侧 |
        // | 73°41' 标注 | text | (455,95) | ~50×16 | 上方 | 北极附近 |
        // | 上一步按钮 | 按钮 | 底部左侧 | 100×34 | - | 控件栏 |
        // | 下一步按钮 | 按钮 | 底部右侧 | 100×34 | - | 控件栏 |
        
        // 状态机：0-4 共 5 个状态
        let currentState = 0;
        const totalStates = 5;
        
        // 获取 DOM 元素
        const btnPrev = document.getElementById('scene1-btn-prev');
        const btnNext = document.getElementById('scene1-btn-next');
        
        // 各状态需要显示的元素
        const stateElements = {
            0: [], // 初始状态：只显示基础元素（已在 HTML 中默认显示）
            1: ['scene1-sunrays', 'scene1-label-sunray'], // 太阳光线
            2: ['scene1-morning-line', 'scene1-label-morning'], // 晨昏线
            3: ['scene1-day-hemisphere', 'scene1-label-day', 'scene1-night-hemisphere', 'scene1-label-night'], // 昼夜半球
            4: ['scene1-point-a', 'scene1-label-a', 'scene1-point-b', 'scene1-label-b', 'scene1-point-c', 'scene1-label-c'] // ABC 点高亮
        };
        
        // 已显示的元素集合（用于追踪）
        const shownElements = new Set();
        
        // 显示元素函数
        function showElement(elementId) {
            const el = document.getElementById(elementId);
            if (el) {
                el.classList.remove('scene1-hidden');
                el.classList.add('scene1-visible');
                shownElements.add(elementId);
            }
        }
        
        // 隐藏元素函数
        function hideElement(elementId) {
            const el = document.getElementById(elementId);
            if (el) {
                el.classList.remove('scene1-visible');
                el.classList.add('scene1-hidden');
                shownElements.delete(elementId);
            }
        }
        
        // 更新到指定状态
        function updateState(newState) {
            // 确定需要显示哪些新元素
            const targetElements = new Set();
            for (let i = 1; i <= newState; i++) {
                stateElements[i].forEach(id => targetElements.add(id));
            }
            
            // 显示目标元素
            targetElements.forEach(id => showElement(id));
            
            // 隐藏不再需要的元素（回退时）
            shownElements.forEach(id => {
                if (!targetElements.has(id)) {
                    hideElement(id);
                }
            });
            
            // 特殊处理：晨昏线高亮（状态 2）
            const morningLine = document.getElementById('scene1-morning-line');
            if (newState === 2) {
                morningLine.classList.add('scene1-morning-line-highlight');
            } else {
                morningLine.classList.remove('scene1-morning-line-highlight');
            }
            
            // 特殊处理：太阳光线高亮（状态 1）
            const sunrays = document.getElementById('scene1-sunrays');
            if (newState === 1) {
                sunrays.classList.add('scene1-sunray-highlight');
            } else {
                sunrays.classList.remove('scene1-sunray-highlight');
            }
            
            // 特殊处理：ABC 点高亮（状态 4）
            ['scene1-point-a', 'scene1-point-b', 'scene1-point-c'].forEach(id => {
                const point = document.getElementById(id);
                if (newState === 4) {
                    point.classList.add('scene1-point-highlight');
                } else {
                    point.classList.remove('scene1-point-highlight');
                }
            });
            
            // 更新按钮状态
            btnPrev.disabled = (newState === 0);
            btnNext.disabled = (newState === totalStates - 1);
            
            currentState = newState;
        }
        
        // 按钮事件绑定
        btnPrev.addEventListener('click', function() {
            if (currentState > 0) {
                updateState(currentState - 1);
            }
        });
        
        btnNext.addEventListener('click', function() {
            if (currentState < totalStates - 1) {
                updateState(currentState + 1);
            }
        });
        
        // 初始化：确保所有特殊元素初始隐藏
        function init() {
            // 隐藏所有需要逐步显示的元素
            const allHiddenElements = [];
            Object.values(stateElements).forEach(arr => {
                arr.forEach(id => allHiddenElements.push(id));
            });
            
            allHiddenElements.forEach(id => hideElement(id));
            
            // 确保 ABC 点初始可见（题目基础条件）
            ['scene1-point-a', 'scene1-label-a', 'scene1-point-b', 'scene1-label-b', 'scene1-point-c', 'scene1-label-c'].forEach(id => {
                const el = document.getElementById(id);
                if (el) {
                    el.classList.remove('scene1-hidden');
                    el.classList.add('scene1-visible');
                }
            });
            
            // 更新按钮状态
            btnPrev.disabled = true;
            btnNext.disabled = false;
        }
        
        // 页面加载完成后初始化
        init();
    })();
    </script>
