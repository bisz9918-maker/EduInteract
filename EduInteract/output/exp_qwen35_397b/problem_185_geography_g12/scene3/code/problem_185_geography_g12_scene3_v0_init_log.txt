    <script>
    (function() {
        // Scene 3: 昼夜弧长与日出时间计算
        
        let currentState = 0;
        const totalSteps = 4;
        
        // 获取元素
        const arcDay = document.getElementById('scene3-arc-day');
        const arcNight = document.getElementById('scene3-arc-night');
        const label150 = document.getElementById('scene3-label-150');
        const calcText1 = document.getElementById('scene3-calc-text1');
        const calcText2 = document.getElementById('scene3-calc-text2');
        const calcText3 = document.getElementById('scene3-calc-text3');
        const btnPrev = document.getElementById('scene3-btn-prev');
        const btnNext = document.getElementById('scene3-btn-next');
        const stepIndicator = document.getElementById('scene3-step-indicator');
        
        // 更新步骤显示
        function updateStepIndicator() {
            stepIndicator.textContent = `步骤 ${currentState}/${totalSteps}`;
            btnPrev.disabled = currentState === 0;
            btnNext.disabled = currentState === totalSteps;
            if (currentState === totalSteps) {
                btnNext.textContent = '完成';
            } else {
                btnNext.textContent = '下一步 ▶';
            }
        }
        
        // 渲染步骤
        function renderStep(step) {
            // 步骤 0: 初始状态 - 只显示基础元素
            // 步骤 1: 显示昼弧和夜弧，150°标注
            // 步骤 2: 显示计算第一行
            // 步骤 3: 显示计算第二行
            // 步骤 4: 显示计算第三行
            
            if (step >= 1) {
                arcDay.style.opacity = '1';
                arcNight.style.opacity = '1';
                label150.style.opacity = '1';
            } else {
                arcDay.style.opacity = '0';
                arcNight.style.opacity = '0';
                label150.style.opacity = '0';
            }
            
            if (step >= 2) {
                calcText1.style.opacity = '1';
            } else {
                calcText1.style.opacity = '0';
            }
            
            if (step >= 3) {
                calcText2.style.opacity = '1';
            } else {
                calcText2.style.opacity = '0';
            }
            
            if (step >= 4) {
                calcText3.style.opacity = '1';
            } else {
                calcText3.style.opacity = '0';
            }
            
            updateStepIndicator();
        }
        
        // 按钮事件
        btnPrev.addEventListener('click', function() {
            if (currentState > 0) {
                currentState--;
                renderStep(currentState);
            }
        });
        
        btnNext.addEventListener('click', function() {
            if (currentState < totalSteps) {
                currentState++;
                renderStep(currentState);
            }
        });
        
        // 初始化
        renderStep(0);
    })();
    </script>
