    <script>
    (function() {
        // Scene 3: 四季形成原理流程图
        // 布局坐标表（实现对照依据）：
        // | 元素 | 位置/范围 | 大小 | 备注 |
        // | 自转框 | (260,20) | 80×35 | 顶层左分支 |
        // | 公转框 | (460,20) | 80×35 | 顶层右分支 |
        // | 赤道平面框 | (210,70) | 140×35 | 第 2 层左 |
        // | 黄道平面框 | (450,70) | 140×35 | 第 2 层右 |
        // | 黄赤交角框 | (340,120) | 120×35 | 第 3 层居中 |
        // | 直射点回归框 | (320,170) | 160×35 | 第 4 层居中 |
        // | 正午太阳高度框 | (200,220) | 160×35 | 第 5 层左 |
        // | 昼夜长短框 | (440,220) | 160×35 | 第 5 层右 |
        // | 地面受热框 | (280,270) | 240×35 | 第 6 层居中 |
        // | 季节变化框 | (340,320) | 120×35 | 第 7 层居中 |
        // | 四季交替框 | (330,370) | 140×35 | 第 8 层居中 |

        let currentStep = 0;
        const maxStep = 5;

        const prevBtn = document.getElementById('scene3-prev-btn');
        const nextBtn = document.getElementById('scene3-next-btn');
        const stepIndicator = document.getElementById('scene3-step-indicator');

        // 获取所有占位符元素
        const placeholders = [
            document.getElementById('scene3-placeholder1'), // 赤道位置
            document.getElementById('scene3-placeholder2'), // 黄道位置
            document.getElementById('scene3-placeholder3'), // 黄赤位置
            document.getElementById('scene3-placeholder4'), // 正午太阳高度位置
            document.getElementById('scene3-placeholder5'), // 昼夜长短位置
            document.getElementById('scene3-placeholder6')  // 季节位置
        ];

        // 获取所有答案元素
        const answers = [
            document.getElementById('scene3-answer1'), // 赤道
            document.getElementById('scene3-answer2'), // 黄道
            document.getElementById('scene3-answer3'), // 黄赤
            document.getElementById('scene3-answer4'), // 正午太阳高度
            document.getElementById('scene3-answer5'), // 昼夜长短
            document.getElementById('scene3-answer6')  // 季节
        ];

        // 获取所有流程框
        const boxes = [
            document.getElementById('scene3-box1'),  // 自转
            document.getElementById('scene3-box2'),  // 公转
            document.getElementById('scene3-box3'),  // 赤道平面
            document.getElementById('scene3-box4'),  // 黄道平面
            document.getElementById('scene3-box5'),  // 黄赤交角
            document.getElementById('scene3-box6'),  // 直射点回归
            document.getElementById('scene3-box7'),  // 正午太阳高度
            document.getElementById('scene3-box8'),  // 昼夜长短
            document.getElementById('scene3-box9'),  // 地面受热
            document.getElementById('scene3-box10'), // 季节变化
            document.getElementById('scene3-box11')  // 四季交替
        ];

        // 步骤配置：[显示的答案索引数组，高亮的框索引数组]
        const stepConfig = [
            { answers: [], highlights: [] },                    // 步骤 0: 空白
            { answers: [0, 1], highlights: [2, 3] },            // 步骤 1: 赤道、黄道
            { answers: [0, 1, 2], highlights: [4] },            // 步骤 2: 黄赤
            { answers: [0, 1, 2, 3, 4], highlights: [6, 7] },   // 步骤 3: 正午太阳高度、昼夜长短
            { answers: [0, 1, 2, 3, 4, 5], highlights: [9] },   // 步骤 4: 季节
            { answers: [0, 1, 2, 3, 4, 5], highlights: [] }     // 步骤 5: 完成
        ];

        function show(el) {
            el.style.display = '';
        }

        function hide(el) {
            el.style.display = 'none';
        }

        function updateStep() {
            const config = stepConfig[currentStep];

            // 更新占位符和答案显示：显示答案时隐藏占位符
            placeholders.forEach((ph, idx) => {
                if (config.answers.includes(idx)) {
                    hide(ph);
                    show(answers[idx]);
                } else {
                    show(ph);
                    hide(answers[idx]);
                }
            });

            // 更新框高亮
            boxes.forEach((box, idx) => {
                box.classList.remove('scene3-box-highlight');
                if (config.highlights.includes(idx)) {
                    box.classList.add('scene3-box-highlight');
                }
            });

            // 更新步骤指示器
            stepIndicator.textContent = `步骤 ${currentStep}/${maxStep}`;

            // 更新按钮状态
            prevBtn.disabled = currentStep === 0;
            nextBtn.disabled = currentStep === maxStep;
        }

        prevBtn.addEventListener('click', function() {
            if (currentStep > 0) {
                currentStep--;
                updateStep();
            }
        });

        nextBtn.addEventListener('click', function() {
            if (currentStep < maxStep) {
                currentStep++;
                updateStep();
            }
        });

        // 初始化
        updateStep();
    })();
    </script>
