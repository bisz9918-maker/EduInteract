    <script>
    (function() {
        // Scene 3 实现：四地经度计算与世界时对照
        
        let currentStep = 0;
        const totalSteps = 6;
        
        // 每步需要显示的元素 ID 列表
        const stepElements = {
            0: ['scene3-point-a', 'scene3-point-b', 'scene3-point-c', 'scene3-point-d',
                'scene3-label-a', 'scene3-label-b', 'scene3-label-c', 'scene3-label-d'],
            1: ['scene3-table-bg', 'scene3-table-title', 'scene3-table-header'],
            2: ['scene3-row-a-bg', 'scene3-row-a', 'scene3-formula-bg', 'scene3-formula-text'],
            3: ['scene3-row-b-bg', 'scene3-row-b', 'scene3-lon-point-a', 'scene3-lon-label-a'],
            4: ['scene3-row-c-bg', 'scene3-row-c', 'scene3-lon-point-b', 'scene3-lon-label-b'],
            5: ['scene3-row-d-bg', 'scene3-row-d', 'scene3-lon-point-c', 'scene3-lon-label-c'],
            6: ['scene3-lon-point-d', 'scene3-lon-label-d']
        };
        
        // 每步需要高亮的元素
        const stepHighlights = {
            0: [],
            1: ['scene3-point-a'],
            2: ['scene3-row-a-bg'],
            3: ['scene3-row-b-bg'],
            4: ['scene3-row-c-bg'],
            5: ['scene3-row-d-bg'],
            6: []
        };
        
        // 显示元素函数
        function showElement(id) {
            const el = document.getElementById(id);
            if (el) {
                el.style.display = '';
                el.style.opacity = '1';
            }
        }
        
        // 隐藏元素函数
        function hideElement(id) {
            const el = document.getElementById(id);
            if (el) {
                el.style.display = 'none';
                el.style.opacity = '0';
            }
        }
        
        // 清除高亮
        function clearHighlights() {
            document.querySelectorAll('.scene3-highlight').forEach(el => {
                el.classList.remove('scene3-highlight');
            });
            document.querySelectorAll('.scene3-row-highlight').forEach(el => {
                el.classList.remove('scene3-row-highlight');
            });
        }
        
        // 应用高亮
        function applyHighlights(step) {
            clearHighlights();
            const highlights = stepHighlights[step] || [];
            highlights.forEach(id => {
                const el = document.getElementById(id);
                if (el) {
                    if (id.includes('row')) {
                        el.classList.add('scene3-row-highlight');
                    } else {
                        el.classList.add('scene3-highlight');
                    }
                }
            });
        }
        
        // 更新按钮状态
        function updateButtons() {
            const prevBtn = document.getElementById('scene3-btn-prev');
            const nextBtn = document.getElementById('scene3-btn-next');
            
            if (currentStep === 0) {
                prevBtn.style.opacity = '0.5';
                prevBtn.style.pointerEvents = 'none';
            } else {
                prevBtn.style.opacity = '1';
                prevBtn.style.pointerEvents = 'auto';
            }
            
            if (currentStep >= totalSteps) {
                nextBtn.style.opacity = '0.5';
                nextBtn.style.pointerEvents = 'none';
            } else {
                nextBtn.style.opacity = '1';
                nextBtn.style.pointerEvents = 'auto';
            }
        }
        
        // 更新显示状态
        function updateDisplay() {
            // 显示当前步骤及之前所有步骤的元素
            for (let s = 0; s <= currentStep; s++) {
                const elements = stepElements[s] || [];
                elements.forEach(id => showElement(id));
            }
            
            // 隐藏后续步骤的元素
            for (let s = currentStep + 1; s <= totalSteps; s++) {
                const elements = stepElements[s] || [];
                elements.forEach(id => hideElement(id));
            }
            
            // 应用高亮
            applyHighlights(currentStep);
            
            // 更新按钮状态
            updateButtons();
        }
        
        // 上一步
        function prevStep() {
            if (currentStep > 0) {
                currentStep--;
                updateDisplay();
            }
        }
        
        // 下一步
        function nextStep() {
            if (currentStep < totalSteps) {
                currentStep++;
                updateDisplay();
            }
        }
        
        // 绑定按钮事件
        document.getElementById('scene3-btn-prev').addEventListener('click', prevStep);
        document.getElementById('scene3-btn-next').addEventListener('click', nextStep);
        
        // 初始化显示
        updateDisplay();
    })();
    </script>
