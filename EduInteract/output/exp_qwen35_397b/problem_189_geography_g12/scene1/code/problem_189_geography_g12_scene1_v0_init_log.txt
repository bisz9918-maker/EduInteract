    <script>
    (function() {
        // Scene 1 实现：太阳直射点移动轨迹图

        // 状态变量
        let currentStep = 0;
        const totalSteps = 5;

        // DOM 元素引用
        const btnPrev = document.getElementById('scene1-btn-prev');
        const btnNext = document.getElementById('scene1-btn-next');
        const stepIndicator = document.getElementById('scene1-step-indicator');

        // 点元素
        const points = {
            a: document.getElementById('scene1-point-a'),
            b: document.getElementById('scene1-point-b'),
            c: document.getElementById('scene1-point-c'),
            d: document.getElementById('scene1-point-d'),
            e: document.getElementById('scene1-point-e'),
            f: document.getElementById('scene1-point-f')
        };

        // 标注框元素
        const labels = {
            a: document.getElementById('scene1-label-a'),
            b: document.getElementById('scene1-label-b'),
            c: document.getElementById('scene1-label-c'),
            d: document.getElementById('scene1-label-d'),
            e: document.getElementById('scene1-label-e'),
            f: document.getElementById('scene1-label-f')
        };

        // 箭头元素
        const arrows = [
            document.getElementById('scene1-arrow1'),
            document.getElementById('scene1-arrow2'),
            document.getElementById('scene1-arrow3'),
            document.getElementById('scene1-arrow4'),
            document.getElementById('scene1-arrow5')
        ];

        // 标题和总结
        const title = document.getElementById('scene1-title');
        const summary = document.getElementById('scene1-summary');
        const trajectory = document.getElementById('scene1-trajectory');

        // 重置所有点到默认状态
        function resetPoints() {
            Object.values(points).forEach(point => {
                point.setAttribute('r', '6');
                point.setAttribute('fill', '#9e9e9e');
            });
        }

        // 隐藏所有标注框
        function hideAllLabels() {
            Object.values(labels).forEach(label => {
                label.classList.remove('visible');
            });
        }

        // 隐藏所有箭头
        function hideAllArrows() {
            arrows.forEach(arrow => {
                arrow.classList.remove('visible');
            });
        }

        // 更新步骤显示
        function updateStep() {
            // 更新步骤指示器
            stepIndicator.textContent = `步骤 ${currentStep}/${totalSteps}`;

            // 更新按钮状态
            btnPrev.disabled = currentStep === 0;
            btnNext.disabled = currentStep === totalSteps;
            btnNext.textContent = currentStep === totalSteps ? '完成' : '下一步 ▶';

            // 重置所有元素
            resetPoints();
            hideAllLabels();
            hideAllArrows();
            title.classList.remove('visible');
            summary.classList.remove('visible');
            trajectory.setAttribute('opacity', '0.5');

            // 根据当前步骤设置显示
            switch (currentStep) {
                case 0:
                    // 初始状态：只显示基础图
                    break;

                case 1:
                    // 高亮 A 点（冬至）
                    points.a.setAttribute('r', '8');
                    points.a.setAttribute('fill', '#c62828');
                    labels.a.classList.add('visible');
                    trajectory.setAttribute('opacity', '0.7');
                    break;

                case 2:
                    // 高亮 B 点和 E 点（春分/秋分）
                    points.b.setAttribute('r', '8');
                    points.b.setAttribute('fill', '#2e7d32');
                    points.e.setAttribute('r', '8');
                    points.e.setAttribute('fill', '#2e7d32');
                    labels.b.classList.add('visible');
                    labels.e.classList.add('visible');
                    trajectory.setAttribute('opacity', '0.7');
                    break;

                case 3:
                    // 高亮 C 点和 D 点（夏至）
                    points.c.setAttribute('r', '8');
                    points.c.setAttribute('fill', '#1565c0');
                    points.d.setAttribute('r', '8');
                    points.d.setAttribute('fill', '#1565c0');
                    labels.c.classList.add('visible');
                    labels.d.classList.add('visible');
                    trajectory.setAttribute('opacity', '0.7');
                    break;

                case 4:
                    // 高亮 F 点，显示所有箭头
                    points.f.setAttribute('r', '8');
                    points.f.setAttribute('fill', '#c62828');
                    labels.f.classList.add('visible');
                    arrows.forEach(arrow => arrow.classList.add('visible'));
                    trajectory.setAttribute('opacity', '1');
                    break;

                case 5:
                    // 最终状态：显示所有标注和标题
                    points.a.setAttribute('r', '7');
                    points.a.setAttribute('fill', '#c62828');
                    points.b.setAttribute('r', '7');
                    points.b.setAttribute('fill', '#2e7d32');
                    points.c.setAttribute('r', '7');
                    points.c.setAttribute('fill', '#1565c0');
                    points.d.setAttribute('r', '7');
                    points.d.setAttribute('fill', '#1565c0');
                    points.e.setAttribute('r', '7');
                    points.e.setAttribute('fill', '#2e7d32');
                    points.f.setAttribute('r', '7');
                    points.f.setAttribute('fill', '#c62828');

                    Object.values(labels).forEach(label => {
                        label.classList.add('visible');
                    });
                    arrows.forEach(arrow => arrow.classList.add('visible'));
                    title.classList.add('visible');
                    summary.classList.add('visible');
                    trajectory.setAttribute('opacity', '1');
                    break;
            }
        }

        // 按钮事件绑定
        btnPrev.addEventListener('click', function() {
            if (currentStep > 0) {
                currentStep--;
                updateStep();
            }
        });

        btnNext.addEventListener('click', function() {
            if (currentStep < totalSteps) {
                currentStep++;
                updateStep();
            }
        });

        // 初始化
        updateStep();
    })();
    </script>
