    <script>
    (function() {
        // Scene 1 实现：牛顿调和三角形递推关系演示

        let currentStep = 0;
        const maxSteps = 5;

        // 状态配置
        const steps = {
            0: {
                highlights: [],
                arrows: [],
                equations: [],
                row6Style: 'dashed',
                row6TextVisible: false,
                answerVisible: false
            },
            1: {
                highlights: ['scene1-highlight-2-1', 'scene1-highlight-3-1', 'scene1-highlight-3-2'],
                arrows: ['scene1-arrow-1-1', 'scene1-arrow-1-2'],
                equations: ['scene1-equation-box-1', 'scene1-equation-1'],
                row6Style: 'dashed',
                row6TextVisible: false,
                answerVisible: false
            },
            2: {
                highlights: ['scene1-highlight-3-1-s2', 'scene1-highlight-4-1', 'scene1-highlight-4-2'],
                arrows: ['scene1-arrow-2-1', 'scene1-arrow-2-2'],
                equations: ['scene1-equation-box-2', 'scene1-equation-2'],
                row6Style: 'dashed',
                row6TextVisible: false,
                answerVisible: false
            },
            3: {
                highlights: ['scene1-highlight-4-1-s3', 'scene1-highlight-5-1', 'scene1-highlight-5-2'],
                arrows: ['scene1-arrow-3-1', 'scene1-arrow-3-2'],
                equations: ['scene1-equation-box-3', 'scene1-equation-3'],
                row6Style: 'dashed',
                row6TextVisible: false,
                answerVisible: false
            },
            4: {
                highlights: [],
                arrows: [],
                equations: [],
                row6Style: 'solid',
                row6TextVisible: true,
                answerVisible: false
            },
            5: {
                highlights: [],
                arrows: [],
                equations: [],
                row6Style: 'solid',
                row6TextVisible: true,
                answerVisible: true
            }
        };

        // 辅助函数：显示元素
        function show(el) {
            if (el) {
                el.style.display = 'block';
                el.style.opacity = '1';
            }
        }

        // 辅助函数：隐藏元素
        function hide(el) {
            if (el) {
                el.style.display = 'none';
                el.style.opacity = '0';
            }
        }

        // 更新状态
        function updateState() {
            const state = steps[currentStep];

            // 清除所有高亮
            document.querySelectorAll('.scene1-highlight').forEach(el => {
                el.classList.remove('scene1-highlight-visible');
            });
            document.getElementById('scene1-answer-highlight').classList.remove('scene1-answer-visible');

            // 清除所有箭头可见性
            document.querySelectorAll('.scene1-arrow').forEach(el => {
                el.classList.remove('scene1-arrow-visible');
            });

            // 清除所有等式可见性
            document.querySelectorAll('.scene1-equation-box').forEach(el => {
                el.classList.remove('scene1-equation-visible');
            });
            document.querySelectorAll('.scene1-equation-text').forEach(el => {
                el.classList.remove('scene1-equation-visible');
            });

            // 清除第 6 行实线样式
            document.querySelectorAll('.scene1-row6-box').forEach(el => {
                el.classList.remove('scene1-row6-solid');
            });
            document.querySelectorAll('.scene1-row6-text').forEach(el => {
                el.classList.remove('scene1-row6-text-visible');
            });

            // 隐藏答案标注
            document.getElementById('scene1-answer-label').classList.remove('scene1-answer-visible');

            // 应用当前状态
            state.highlights.forEach(id => {
                const el = document.getElementById(id);
                if (el) el.classList.add('scene1-highlight-visible');
            });

            if (currentStep === 5) {
                document.getElementById('scene1-answer-highlight').classList.add('scene1-answer-visible');
                document.getElementById('scene1-answer-label').classList.add('scene1-answer-visible');
            }

            state.arrows.forEach(id => {
                const el = document.getElementById(id);
                if (el) el.classList.add('scene1-arrow-visible');
            });

            state.equations.forEach(id => {
                const el = document.getElementById(id);
                if (el) el.classList.add('scene1-equation-visible');
            });

            // 第 6 行样式
            if (state.row6Style === 'solid') {
                document.querySelectorAll('.scene1-row6-box').forEach(el => {
                    el.classList.add('scene1-row6-solid');
                });
            }

            // 第 6 行文字可见性
            if (state.row6TextVisible) {
                document.querySelectorAll('.scene1-row6-text').forEach(el => {
                    el.classList.add('scene1-row6-text-visible');
                });
            }

            // 更新按钮状态
            const prevBtn = document.getElementById('scene1-btn-prev-bg');
            const nextBtn = document.getElementById('scene1-btn-next-bg');
            const prevText = document.getElementById('scene1-btn-prev-text');
            const nextText = document.getElementById('scene1-btn-next-text');

            if (currentStep === 0) {
                prevBtn.classList.add('scene1-btn-disabled');
                prevText.style.fill = '#999999';
            } else {
                prevBtn.classList.remove('scene1-btn-disabled');
                prevText.style.fill = '#333333';
            }

            if (currentStep >= maxSteps) {
                nextBtn.classList.add('scene1-btn-disabled');
                nextText.style.fill = '#999999';
            } else {
                nextBtn.classList.remove('scene1-btn-disabled');
                nextText.style.fill = '#333333';
            }

            // 更新步骤指示器
            const indicator = document.getElementById('scene1-step-indicator');
            if (currentStep === maxSteps) {
                indicator.textContent = '完成';
            } else {
                indicator.textContent = `步骤 ${currentStep}/5`;
            }

            // 触发 MathJax 重新渲染
            if (window.MathJax) {
                MathJax.typesetPromise();
            }
        }

        // 按钮事件绑定
        document.getElementById('scene1-btn-prev-bg').addEventListener('click', function() {
            if (currentStep > 0) {
                currentStep--;
                updateState();
            }
        });

        document.getElementById('scene1-btn-next-bg').addEventListener('click', function() {
            if (currentStep < maxSteps) {
                currentStep++;
                updateState();
            }
        });

        // 初始化
        updateState();
    })();
    </script>
