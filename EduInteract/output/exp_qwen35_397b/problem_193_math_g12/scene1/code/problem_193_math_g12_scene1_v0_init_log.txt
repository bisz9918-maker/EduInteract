    <script>
    (function() {
        // Scene 1 实现
        let currentStep = 0;
        const totalSteps = 4;
        
        // 获取元素
        const stepIndicator = document.getElementById('scene1-step-indicator');
        const btnPrevBg = document.getElementById('scene1-btn-prev-bg');
        const btnPrevText = document.getElementById('scene1-btn-prev-text');
        const btnNextBg = document.getElementById('scene1-btn-next-bg');
        const btnNextText = document.getElementById('scene1-btn-next-text');
        
        const projLines = document.getElementById('scene1-proj-lines');
        const projLine1 = document.getElementById('scene1-proj-line-1');
        const projLine2 = document.getElementById('scene1-proj-line-2');
        const projLine3 = document.getElementById('scene1-proj-line-3');
        
        const halfCone = document.getElementById('scene1-half-cone');
        const triPyramid = document.getElementById('scene1-tri-pyramid');
        const apex = document.getElementById('scene1-apex');
        const heightLine = document.getElementById('scene1-height-line');
        const heightTop = document.getElementById('scene1-height-top');
        const heightBottom = document.getElementById('scene1-height-bottom');
        const heightLabel = document.getElementById('scene1-height-label');
        const title3d = document.getElementById('scene1-3d-title');
        const volumeFormula = document.getElementById('scene1-volume-formula');
        
        // 显示/隐藏辅助函数
        function show(el) {
            el.classList.remove('scene1-hidden');
            el.classList.add('scene1-visible');
            if (el.tagName === 'text' || el.tagName === 'line' || el.tagName === 'circle' || el.tagName === 'path' || el.tagName === 'g') {
                el.style.opacity = '1';
                el.style.display = '';
            } else {
                el.style.display = '';
                el.style.opacity = '1';
            }
        }
        
        function hide(el) {
            el.classList.remove('scene1-visible');
            el.classList.add('scene1-hidden');
            if (el.tagName === 'text' || el.tagName === 'line' || el.tagName === 'circle' || el.tagName === 'path' || el.tagName === 'g') {
                el.style.opacity = '0';
            } else {
                el.style.opacity = '0';
            }
        }
        
        // 更新按钮状态
        function updateButtons() {
            if (currentStep === 0) {
                btnPrevBg.setAttribute('fill', '#cccccc');
                btnPrevText.setAttribute('fill', '#888');
            } else {
                btnPrevBg.setAttribute('fill', '#e0e0e0');
                btnPrevText.setAttribute('fill', '#333');
            }
            
            if (currentStep === totalSteps - 1) {
                btnNextBg.setAttribute('fill', '#cccccc');
                btnNextText.setAttribute('fill', '#888');
            } else {
                btnNextBg.setAttribute('fill', '#4CAF50');
                btnNextText.setAttribute('fill', '#fff');
            }
        }
        
        // 更新步骤显示
        function updateStep() {
            stepIndicator.textContent = '步骤 ' + (currentStep + 1) + '/' + totalSteps;
            
            // 步骤0: 仅显示三视图
            if (currentStep === 0) {
                hide(projLines);
                hide(halfCone);
                hide(triPyramid);
                hide(apex);
                hide(heightLine);
                hide(heightTop);
                hide(heightBottom);
                hide(heightLabel);
                hide(title3d);
                hide(volumeFormula);
            }
            // 步骤1: 显示半圆锥
            else if (currentStep === 1) {
                show(projLines);
                projLine1.style.opacity = '1';
                projLine2.style.opacity = '0';
                projLine3.style.opacity = '0';
                show(halfCone);
                hide(triPyramid);
                hide(apex);
                hide(heightLine);
                hide(heightTop);
                hide(heightBottom);
                hide(heightLabel);
                hide(title3d);
                hide(volumeFormula);
            }
            // 步骤2: 显示三棱锥
            else if (currentStep === 2) {
                show(projLines);
                projLine1.style.opacity = '1';
                projLine2.style.opacity = '1';
                projLine3.style.opacity = '1';
                show(halfCone);
                show(triPyramid);
                show(apex);
                show(heightLine);
                show(heightTop);
                show(heightBottom);
                show(heightLabel);
                show(title3d);
                hide(volumeFormula);
            }
            // 步骤3: 显示体积公式
            else if (currentStep === 3) {
                show(projLines);
                projLine1.style.opacity = '1';
                projLine2.style.opacity = '1';
                projLine3.style.opacity = '1';
                show(halfCone);
                show(triPyramid);
                show(apex);
                show(heightLine);
                show(heightTop);
                show(heightBottom);
                show(heightLabel);
                show(title3d);
                show(volumeFormula);
            }
            
            updateButtons();
        }
        
        // 点击事件处理
        const svg = document.getElementById('scene1-svg');
        
        svg.addEventListener('click', function(e) {
            const target = e.target;
            
            // 检查是否点击上一步按钮
            if (target === btnPrevBg || target === btnPrevText) {
                if (currentStep > 0) {
                    currentStep--;
                    updateStep();
                }
                return;
            }
            
            // 检查是否点击下一步按钮
            if (target === btnNextBg || target === btnNextText) {
                if (currentStep < totalSteps - 1) {
                    currentStep++;
                    updateStep();
                }
                return;
            }
        });
        
        // 初始化
        updateStep();
    })();
    </script>
