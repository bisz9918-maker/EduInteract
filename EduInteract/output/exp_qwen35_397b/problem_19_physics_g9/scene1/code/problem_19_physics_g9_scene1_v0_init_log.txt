    <script>
    (function() {
        // Scene 1 实现：并联电路结构分解与电流路径

        // 状态变量
        let currentStep = 0;
        const totalSteps = 4;

        // 元素引用
        const btnPrev = document.getElementById('scene1-btn-prev-bg');
        const btnPrevText = document.getElementById('scene1-btn-prev-text');
        const btnNext = document.getElementById('scene1-btn-next-bg');
        const btnNextText = document.getElementById('scene1-btn-next-text');
        const stepIndicator = document.getElementById('scene1-step-indicator');

        // 步骤元素分组
        const stepElements = {
            0: {
                show: ['scene1-original-frame', 'scene1-original-title', 'scene1-battery-original', 
                       'scene1-switch-original', 'scene1-ammeter-a1-original', 'scene1-ammeter-a2-original',
                       'scene1-pencil-ab', 'scene1-point-a', 'scene1-point-b', 'scene1-point-c', 
                       'scene1-point-d', 'scene1-slider-p', 'scene1-label-a', 'scene1-label-b',
                       'scene1-label-c', 'scene1-label-d', 'scene1-label-p', 'scene1-equivalent-frame',
                       'scene1-equivalent-title'],
                hide: ['scene1-current-arrows-original', 'scene1-highlight-nodes', 'scene1-equivalent-circuit']
            },
            1: {
                show: ['scene1-current-arrows-original', 'scene1-highlight-nodes'],
                hide: []
            },
            2: {
                show: ['scene1-equivalent-circuit'],
                hide: []
            },
            3: {
                show: ['scene1-resistor-labels', 'scene1-resistor-highlight'],
                hide: []
            }
        };

        // 显示/隐藏元素工具函数
        function showElement(id) {
            const el = document.getElementById(id);
            if (el) {
                el.classList.remove('hidden');
                el.style.display = '';
                el.style.opacity = '1';
            }
        }

        function hideElement(id) {
            const el = document.getElementById(id);
            if (el) {
                el.classList.add('hidden');
                el.style.display = 'none';
                el.style.opacity = '0';
            }
        }

        // 更新按钮状态
        function updateButtons() {
            // 上一步按钮
            if (currentStep === 0) {
                btnPrev.setAttribute('fill', '#bdbdbd');
                btnPrev.style.cursor = 'not-allowed';
                btnPrevText.setAttribute('fill', '#e0e0e0');
            } else {
                btnPrev.setAttribute('fill', '#9e9e9e');
                btnPrev.style.cursor = 'pointer';
                btnPrevText.setAttribute('fill', 'white');
            }

            // 下一步按钮
            if (currentStep === totalSteps - 1) {
                btnNext.setAttribute('fill', '#bdbdbd');
                btnNext.style.cursor = 'not-allowed';
                btnNextText.setAttribute('fill', '#e0e0e0');
            } else {
                btnNext.setAttribute('fill', '#1976d2');
                btnNext.style.cursor = 'pointer';
                btnNextText.setAttribute('fill', 'white');
            }

            // 更新步骤指示器
            stepIndicator.textContent = '步骤 ' + (currentStep + 1) + '/' + totalSteps;
        }

        // 应用步骤状态
        function applyStep(step) {
            // 步骤0：显示基础原图
            if (step === 0) {
                stepElements[0].show.forEach(showElement);
                stepElements[0].hide.forEach(hideElement);
                // 隐藏后续步骤的元素
                for (let i = 1; i < totalSteps; i++) {
                    stepElements[i].show.forEach(hideElement);
                }
            } else {
                // 步骤1-3：累积显示
                // 先显示步骤0的所有元素
                stepElements[0].show.forEach(showElement);
                stepElements[0].hide.forEach(hideElement);
                
                // 然后依次显示到当前步骤的元素
                for (let i = 1; i <= step; i++) {
                    stepElements[i].show.forEach(showElement);
                    stepElements[i].hide.forEach(hideElement);
                }
                // 隐藏后续步骤的元素
                for (let i = step + 1; i < totalSteps; i++) {
                    stepElements[i].show.forEach(hideElement);
                }
            }
            updateButtons();
        }

        // 按钮点击事件
        btnPrev.addEventListener('click', function() {
            if (currentStep > 0) {
                currentStep--;
                applyStep(currentStep);
            }
        });

        btnNext.addEventListener('click', function() {
            if (currentStep < totalSteps - 1) {
                currentStep++;
                applyStep(currentStep);
            }
        });

        // 设置按钮点击区域
        btnPrev.style.cursor = 'pointer';
        btnNext.style.cursor = 'pointer';

        // 初始化
        applyStep(0);

    })();
    </script>
