    <script>
    (function() {
        // Scene 1: 杠杆平衡条件与力臂比较分析
        // 布局坐标表（实现对照依据）：
        // | 元素 | 类型 | 位置/范围 | 大小 | 标签偏移方向 |
        // |------|------|-----------|------|-------------|
        // | 杠杆 | 矩形 | (100,190)-(700,210) | 600×20 | - |
        // | 支点 O 三角形 | 三角形 | (100,210)-(120,240) | 20×30 | - |
        // | 重物 G 矩形 | 矩形 | (335,240)-(365,280) | 30×40 | - |
        // | 挂线 | 线段 | (350,210)-(350,240) | 1×30 | - |
        // | F1 箭头 | 箭头 | B(700,200)→(750,250) | 线长~70 | 右下方 |
        // | F2 箭头 | 箭头 | B(700,200)→(750,150) | 线长~70 | 右上方 |
        // | F3 箭头 | 箭头 | B(700,200)→(700,130) | 线长~70 | 正上方 |
        // | F4 箭头 | 箭头 | B(700,200)→(650,150) | 线长~70 | 左上方 |
        // | l3 力臂 | 线段 | O(100,200)→B(700,200) | 600×2 | 上方中点 |
        // | l1 力臂 | 虚线 | O(100,200)→垂足(530,270) | ~450×2 | 下方 |
        // | l2 力臂 | 虚线 | O(100,200)→垂足(530,130) | ~450×2 | 上方 |
        // | l4 力臂 | 虚线 | O(100,200)→垂足(480,130) | ~430×2 | 上方偏左 |
        
        let currentStep = 0;
        const totalSteps = 4;
        
        // 元素分组：每个步骤需要显示的元素
        const stepElements = {
            0: ['scene1-lever', 'scene1-fulcrum', 'scene1-weight-string', 'scene1-weight-g',
                'scene1-label-o', 'scene1-label-a', 'scene1-label-b', 'scene1-label-g'],
            1: ['scene1-f1-group', 'scene1-f2-group', 'scene1-f3-group', 'scene1-f4-group'],
            2: ['scene1-l3-group'],
            3: ['scene1-l1-group', 'scene1-l2-group', 'scene1-l4-group'],
            4: ['scene1-conclusion-group']
        };
        
        // 显示元素函数
        function showElement(id) {
            const el = document.getElementById(id);
            if (el) {
                el.classList.remove('scene1-hidden');
                el.style.display = '';
                el.style.opacity = '1';
            }
        }
        
        // 隐藏元素函数
        function hideElement(id) {
            const el = document.getElementById(id);
            if (el) {
                el.classList.add('scene1-hidden');
                el.style.display = 'none';
                el.style.opacity = '0';
            }
        }
        
        // 更新显示状态
        function updateDisplay() {
            // 步骤 0 的元素始终显示
            stepElements[0].forEach(id => showElement(id));
            
            // 根据当前步骤显示对应元素
            for (let step = 1; step <= totalSteps; step++) {
                if (step <= currentStep) {
                    stepElements[step].forEach(id => showElement(id));
                } else {
                    stepElements[step].forEach(id => hideElement(id));
                }
            }
            
            // 步骤 2 时高亮 F3
            const f3Arrow = document.getElementById('scene1-f3-arrow');
            if (currentStep >= 2) {
                f3Arrow.setAttribute('stroke', '#1565c0');
                f3Arrow.setAttribute('stroke-width', '4');
            } else {
                f3Arrow.setAttribute('stroke', '#000000');
                f3Arrow.setAttribute('stroke-width', '2');
            }
            
            // 更新按钮状态
            const prevBtn = document.getElementById('scene1-btn-prev');
            const nextBtn = document.getElementById('scene1-btn-next');
            
            prevBtn.disabled = (currentStep === 0);
            nextBtn.disabled = (currentStep === totalSteps);
        }
        
        // 上一步
        function prevStep() {
            if (currentStep > 0) {
                currentStep--;
                updateDisplay();
            }
        }
        
        // 下一步
        function nextStep() {
            if (currentStep < totalSteps) {
                currentStep++;
                updateDisplay();
            }
        }
        
        // 绑定事件
        document.getElementById('scene1-btn-prev').addEventListener('click', prevStep);
        document.getElementById('scene1-btn-next').addEventListener('click', nextStep);
        
        // 初始化显示
        updateDisplay();
    })();
    </script>
