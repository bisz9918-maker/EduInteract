    <script>
    (function() {
        // Scene 1 实现 - 分步按钮导航

        // 状态管理
        let currentState = 0;
        const totalSteps = 5;

        // 元素分组 - 每步需要显示的元素
        const stepElements = {
            0: [], // 初始状态，只显示基础元素
            1: ['scene1-step1-group'],
            2: ['scene1-step1-group', 'scene1-step2-group'],
            3: ['scene1-step1-group', 'scene1-step2-group', 'scene1-step3-group'],
            4: ['scene1-step1-group', 'scene1-step2-group', 'scene1-step3-group', 'scene1-step4-group']
        };

        // 高亮元素映射
        const highlightElements = {
            1: 'scene1-piston-highlight',
            2: 'scene1-mercury-a-highlight',
            3: 'scene1-mercury-b-highlight',
            4: 'scene1-gas-highlight'
        };

        // 获取元素
        function getElement(id) {
            return document.getElementById(id);
        }

        // 显示元素
        function showElement(id) {
            const el = getElement(id);
            if (el) {
                el.classList.remove('scene1-hidden');
                el.classList.add('scene1-visible');
                el.style.display = 'block';
                el.style.opacity = '1';
            }
        }

        // 隐藏元素
        function hideElement(id) {
            const el = getElement(id);
            if (el) {
                el.classList.remove('scene1-visible');
                el.classList.add('scene1-hidden');
                el.style.display = 'none';
                el.style.opacity = '0';
            }
        }

        // 显示高亮
        function showHighlight(step) {
            if (highlightElements[step]) {
                const el = getElement(highlightElements[step]);
                if (el) {
                    el.style.opacity = '1';
                }
            }
        }

        // 隐藏所有高亮
        function hideAllHighlights() {
            for (let i = 1; i <= 4; i++) {
                if (highlightElements[i]) {
                    const el = getElement(highlightElements[i]);
                    if (el) {
                        el.style.opacity = '0';
                    }
                }
            }
        }

        // 更新步骤显示
        function updateStep(step) {
            // 隐藏所有步骤元素
            for (let i = 1; i <= 4; i++) {
                hideElement('scene1-step' + i + '-group');
            }
            hideAllHighlights();

            // 显示当前步骤及之前步骤的元素
            const elementsToShow = stepElements[step];
            elementsToShow.forEach(function(id) {
                showElement(id);
            });

            // 显示当前步骤的高亮
            if (step > 0) {
                showHighlight(step);
            }

            // 更新步骤指示器
            const indicator = getElement('scene1-step-indicator');
            if (indicator) {
                indicator.textContent = '步骤 ' + (step + 1) + '/' + totalSteps;
            }

            // 更新按钮状态
            const prevBtn = getElement('scene1-btn-prev');
            const nextBtn = getElement('scene1-btn-next');
            if (prevBtn) {
                prevBtn.disabled = (step === 0);
            }
            if (nextBtn) {
                nextBtn.disabled = (step === totalSteps - 1);
            }
        }

        // 上一步
        function prevStep() {
            if (currentState > 0) {
                currentState--;
                updateStep(currentState);
            }
        }

        // 下一步
        function nextStep() {
            if (currentState < totalSteps - 1) {
                currentState++;
                updateStep(currentState);
            }
        }

        // 绑定事件
        function init() {
            const prevBtn = getElement('scene1-btn-prev');
            const nextBtn = getElement('scene1-btn-next');

            if (prevBtn) {
                prevBtn.addEventListener('click', prevStep);
            }
            if (nextBtn) {
                nextBtn.addEventListener('click', nextStep);
            }

            // 初始化显示
            updateStep(currentState);
        }

        // 页面加载完成后初始化
        if (document.readyState === 'loading') {
            document.addEventListener('DOMContentLoaded', init);
        } else {
            init();
        }
    })();
    </script>
