    <script>
    (function() {
        // Scene 2: 飞出电场时的速度矢量分解
        // 布局坐标表（已调整标签位置避免重叠）：
        // | 元素 | 类型 | 位置/范围 | 大小 | 标签偏移方向 | 备注 |
        // |------|------|-----------|------|-------------|------|
        // | 中心线 RD | 虚线 | (150,250)-(650,250) | 500×2 | - | 水平居中 |
        // | MN 界面 | 虚线 | (400,180)-(400,320) | 2×140 | - | 电场右边界 |
        // | 粒子位置 P | 圆点 | (400,280) | 6×6 | - | MN 界面处偏离中心线 |
        // | vx 矢量 | 箭头 | (400,280)-(520,280) | 120×3 | 上方 | 蓝色，水平向右 |
        // | vy 矢量 | 箭头 | (400,280)-(400,360) | 3×80 | 左侧 | 绿色，竖直向下 |
        // | 合速度 v | 箭头 | (400,280)-(520,360) | 144×3 | 上方偏右 | 红色，沿合方向 |
        // | 辅助虚线 1 | 虚线 | (520,280)-(520,360) | 2×80 | - | 构成直角三角形 |
        // | 辅助虚线 2 | 虚线 | (400,360)-(520,360) | 120×2 | - | 构成直角三角形 |
        // | 偏转角θ弧 | 弧线 | (400,280) 半径 40 | 弧长~30 | 弧线外侧 | 从 vx 到 v 的夹角 |
        // | vx 标签 | text | (460,270) | ~50×16 | - | vx = v₀，放在 vx 箭头正上方 |
        // | vy 标签 | text | (370,325) | ~50×16 | 左侧 | vy = at₁，放在 vy 箭头左侧 |
        // | v 标签 | text | (470,265) | ~20×16 | - | v，放在合速度箭头上方 |
        // | θ标签 | text | (425,295) | ~15×16 | - | θ，放在弧线内部靠近圆心 |
        // | 公式标签 | text | (550,320) | ~120×30 | - | tanθ = vy/vx |
        // | N 标签 | text | (405,390) | ~15×16 | - | 向下调整避免与 vy 重叠 |
        // | 控件栏 | 按钮组 | (250,450)-(550,490) | 300×40 | - | 上一步/下一步按钮 |
        
        // 状态变量
        let currentStep = 0;
        const totalSteps = 4;
        
        // 元素引用
        const vxArrow = document.getElementById('scene2-vx-arrow');
        const vyArrow = document.getElementById('scene2-vy-arrow');
        const vArrow = document.getElementById('scene2-v-arrow');
        const vxLabel = document.getElementById('scene2-vx-label');
        const vyLabel = document.getElementById('scene2-vy-label');
        const vLabel = document.getElementById('scene2-v-label');
        const auxLine1 = document.getElementById('scene2-aux-line1');
        const auxLine2 = document.getElementById('scene2-aux-line2');
        const thetaArc = document.getElementById('scene2-theta-arc');
        const thetaLabel = document.getElementById('scene2-theta-label');
        const formula = document.getElementById('scene2-formula');
        const prevBtn = document.getElementById('scene2-prev-btn');
        const nextBtn = document.getElementById('scene2-next-btn');
        const stepIndicator = document.getElementById('scene2-step-indicator');

        // 显示/隐藏工具函数
        function show(el) {
            el.style.display = '';
            el.style.opacity = '1';
        }
        
        function hide(el) {
            el.style.opacity = '0';
            el.style.display = 'none';
        }

        // 更新按钮状态
        function updateButtons() {
            prevBtn.disabled = (currentStep === 0);
            nextBtn.disabled = (currentStep === totalSteps);
            stepIndicator.textContent = `步骤 ${currentStep}/${totalSteps}`;
        }

        // 渲染当前步骤
        function renderStep(step) {
            // 重置所有元素
            hide(vxArrow);
            hide(vyArrow);
            hide(vArrow);
            hide(vxLabel);
            hide(vyLabel);
            hide(vLabel);
            hide(auxLine1);
            hide(auxLine2);
            hide(thetaArc);
            hide(thetaLabel);
            hide(formula);

            // 根据步骤显示相应元素
            if (step >= 1) {
                // 步骤 1: 显示 vx
                vxArrow.style.opacity = '1';
                vxArrow.style.display = '';
                vxArrow.setAttribute('x2', '520');
                vxLabel.style.opacity = '1';
                vxLabel.style.display = '';
            }
            
            if (step >= 2) {
                // 步骤 2: 显示 vy
                vyArrow.style.opacity = '1';
                vyArrow.style.display = '';
                vyArrow.setAttribute('y2', '360');
                vyLabel.style.opacity = '1';
                vyLabel.style.display = '';
            }
            
            if (step >= 3) {
                // 步骤 3: 显示合速度 v 和辅助三角形
                vArrow.style.opacity = '1';
                vArrow.style.display = '';
                vArrow.setAttribute('x2', '520');
                vArrow.setAttribute('y2', '360');
                vLabel.style.opacity = '1';
                vLabel.style.display = '';
                auxLine1.style.opacity = '1';
                auxLine1.style.display = '';
                auxLine2.style.opacity = '1';
                auxLine2.style.display = '';
            }
            
            if (step >= 4) {
                // 步骤 4: 显示偏转角θ和公式
                thetaArc.style.opacity = '1';
                thetaArc.style.display = '';
                // 绘制弧线：从 vx 方向 (0°) 到 v 方向 (约 34°)，半径 40
                // 起点：(400+40, 280) = (440, 280)
                // 终点：(400+40*cos(34°), 280+40*sin(34°)) ≈ (433, 303)
                thetaArc.setAttribute('d', 'M 440 280 A 40 40 0 0 1 433 303');
                thetaLabel.style.opacity = '1';
                thetaLabel.style.display = '';
                formula.style.opacity = '1';
                formula.style.display = '';
            }

            updateButtons();
        }

        // 上一步按钮点击
        prevBtn.addEventListener('click', function() {
            if (currentStep > 0) {
                currentStep--;
                renderStep(currentStep);
            }
        });

        // 下一步按钮点击
        nextBtn.addEventListener('click', function() {
            if (currentStep < totalSteps) {
                currentStep++;
                renderStep(currentStep);
            }
        });

        // 初始化
        renderStep(0);
    })();
    </script>
