    <script>
    (function() {
        // Scene 1 实现
        // 布局坐标表（实现对照依据）：
        // | 元素 | 类型 | 位置/范围 | 大小 | 标签偏移方向 | 备注 |
        // |------|------|-----------|------|-------------|------|
        // | 上电场线 | 线段+箭头 | (50,100)-(750,100) | 700×3 | - | 带向右箭头 |
        // | 下电场线 | 线段+箭头 | (50,350)-(750,350) | 700×3 | - | 带向右箭头 |
        // | E 标签 | text | (760,95) | ~20×18 | 右侧 | 上电场线右端 |
        // | 轨迹虚线 | 虚线 | (100,225)-(700,225) | 600×2 | - | 两球运动轨迹 |
        // | A 球 | 实心圆 | (200,225) | r=12 | - | 左侧小球 |
        // | B 球 | 实心圆 | (500,225) | r=12 | - | 右侧小球 |
        // | A 球标签 | text | (200,255) | ~12×16 | 下方 | "A" 在球正下方 |
        // | B 球标签 | text | (500,255) | ~12×16 | 下方 | "B" 在球正下方 |
        // | A 球质量 | text | (175,195) | ~14×14 | 左上 | "m" 在 A 球左上方 |
        // | A 球电荷 | text | (225,180) | ~20×14 | 右上 | "+q" 在 A 球右上方（向上调整避免与 qE 拥挤） |
        // | B 球质量 | text | (485,205) | ~14×14 | 左上 | "m" 在 B 球左上方 |
        // | 距离 L 双向箭头 | 双向箭头 | (200,280)-(500,280) | 300×3 | - | 两球下方 |
        // | L 标签 | text | (350,295) | ~14×16 | 下方 | 双向箭头中点下方 |
        // | 电场力 qE 箭头 | 箭头 | (212,225)-(292,225) | 80×4 | - | 从 A 球右边缘向右 |
        // | qE 标签 | text | (252,210) | ~24×14 | 上方 | 电场力箭头上方 |
        // | 速度 v₀ 箭头 | 箭头 | (212,225)-(312,225) | 100×4 | - | 从 A 球右边缘向右 |
        // | v₀ 标签 | text | (262,210) | ~20×14 | 上方 | 速度箭头上方 |
        // | 碰撞后 B 速度箭头 | 箭头 | (512,225)-(612,225) | 100×4 | - | 从 B 球右边缘向右 |
        // | 碰撞后 A 速度标签 | text | (200,175) | ~12×14 | 上方 | "0" 表示 A 静止 |
        // | 控件栏背景 | 矩形 | (0,440)-(800,500) | 800×60 | - | SVG 内底部 |
        // | 上一步按钮 | 按钮 | (250,455) | 100×35 | - | 控件栏左侧 |
        // | 下一步按钮 | 按钮 | (450,455) | 100×35 | - | 控件栏右侧 |
        // | 步骤指示 | text | (400,470) | ~80×16 | - | 控件栏中央 |

        const totalSteps = 5; // 步骤 0-4
        let currentStep = 0;

        // 获取 DOM 元素
        const btnPrev = document.getElementById('scene1-btn-prev');
        const btnNext = document.getElementById('scene1-btn-next');
        const stepIndicator = document.getElementById('scene1-step-indicator');
        
        // 元素组
        const groupInitial = document.getElementById('scene1-group-distance-L');
        const groupForce = document.getElementById('scene1-group-force');
        const groupVelocity = document.getElementById('scene1-group-velocity');
        const groupCollision = document.getElementById('scene1-group-collision');
        const groupBallAMoved = document.getElementById('scene1-group-ball-a-moved');
        
        // 电场力箭头（用于脉冲动画）
        const forceArrow = document.getElementById('scene1-force-arrow');
        const forceLabel = document.getElementById('scene1-force-label');

        // 步骤说明文本
        const stepDescriptions = [
            '步骤 0/4：初始状态',
            '步骤 1/4：受力分析',
            '步骤 2/4：加速运动',
            '步骤 3/4：碰撞瞬间',
            '步骤 4/4：速度交换'
        ];

        // 显示/隐藏工具函数
        function showElement(el) {
            if (el) {
                el.classList.remove('scene1-hidden');
                el.style.display = '';
                el.style.opacity = '';
            }
        }

        function hideElement(el) {
            if (el) {
                el.classList.add('scene1-hidden');
                el.style.display = 'none';
                el.style.opacity = '0';
            }
        }

        // 更新显示状态
        function updateDisplay() {
            // 更新按钮状态
            btnPrev.disabled = (currentStep === 0);
            btnNext.disabled = (currentStep === totalSteps - 1);
            
            // 更新步骤指示
            stepIndicator.textContent = stepDescriptions[currentStep];
            
            // 根据步骤显示/隐藏元素
            switch (currentStep) {
                case 0: // 初始状态
                    showElement(groupInitial);
                    hideElement(groupForce);
                    hideElement(groupVelocity);
                    hideElement(groupCollision);
                    hideElement(groupBallAMoved);
                    forceArrow.classList.remove('scene1-pulsing');
                    forceLabel.classList.remove('scene1-pulsing');
                    break;
                    
                case 1: // 受力分析
                    showElement(groupInitial);
                    showElement(groupForce);
                    hideElement(groupVelocity);
                    hideElement(groupCollision);
                    hideElement(groupBallAMoved);
                    forceArrow.classList.remove('scene1-pulsing');
                    forceLabel.classList.remove('scene1-pulsing');
                    break;
                    
                case 2: // 加速运动（电场力脉冲动画）
                    showElement(groupInitial);
                    showElement(groupForce);
                    hideElement(groupVelocity);
                    hideElement(groupCollision);
                    hideElement(groupBallAMoved);
                    forceArrow.classList.add('scene1-pulsing');
                    forceLabel.classList.add('scene1-pulsing');
                    break;
                    
                case 3: // 碰撞瞬间
                    hideElement(groupInitial);
                    hideElement(groupForce);
                    showElement(groupVelocity);
                    hideElement(groupCollision);
                    showElement(groupBallAMoved);
                    forceArrow.classList.remove('scene1-pulsing');
                    forceLabel.classList.remove('scene1-pulsing');
                    break;
                    
                case 4: // 速度交换
                    hideElement(groupInitial);
                    hideElement(groupForce);
                    hideElement(groupVelocity);
                    showElement(groupCollision);
                    showElement(groupBallAMoved);
                    forceArrow.classList.remove('scene1-pulsing');
                    forceLabel.classList.remove('scene1-pulsing');
                    break;
            }
        }

        // 按钮事件绑定
        btnPrev.addEventListener('click', function() {
            if (currentStep > 0) {
                currentStep--;
                updateDisplay();
            }
        });

        btnNext.addEventListener('click', function() {
            if (currentStep < totalSteps - 1) {
                currentStep++;
                updateDisplay();
            }
        });

        // 初始化显示
        updateDisplay();
    })();
    </script>
