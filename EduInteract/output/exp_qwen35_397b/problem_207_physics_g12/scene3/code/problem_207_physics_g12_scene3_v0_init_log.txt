    <script>
    (function() {
        // Scene 3 实现 - v-t 图像分析
        
        const currentState = { step: 0 };
        const totalSteps = 5;
        
        // 每步可见元素配置
        const stepConfig = {
            0: { // 初始状态：仅坐标系和图例
                lines: [],
                areas: [],
                points: [],
                labels: [],
                equation: false
            },
            1: { // A 的 v-t 图线
                lines: ['scene3-line-a'],
                areas: ['scene3-area-a'],
                points: [],
                labels: ['scene3-label-v0', 'scene3-label-sa'],
                equation: false
            },
            2: { // B 的 v-t 图线
                lines: ['scene3-line-a', 'scene3-line-b'],
                areas: ['scene3-area-a', 'scene3-area-b'],
                points: [],
                labels: ['scene3-label-v0', 'scene3-label-sa', 'scene3-label-sb'],
                equation: false
            },
            3: { // 交点和相对位移
                lines: ['scene3-line-a', 'scene3-line-b'],
                areas: ['scene3-area-a', 'scene3-area-b', 'scene3-area-rel'],
                points: ['scene3-point-intersect'],
                labels: ['scene3-label-v0', 'scene3-label-sa', 'scene3-label-sb', 'scene3-label-t-common', 'scene3-label-v-common', 'scene3-label-rel'],
                equation: false
            },
            4: { // 完整方程
                lines: ['scene3-line-a', 'scene3-line-b'],
                areas: ['scene3-area-a', 'scene3-area-b', 'scene3-area-rel'],
                points: ['scene3-point-intersect'],
                labels: ['scene3-label-v0', 'scene3-label-sa', 'scene3-label-sb', 'scene3-label-t-common', 'scene3-label-v-common', 'scene3-label-rel'],
                equation: true
            }
        };
        
        // 显示/隐藏元素工具函数
        function showElement(id) {
            const el = document.getElementById(id);
            if (el) {
                el.style.opacity = '1';
                el.style.display = '';
            }
        }
        
        function hideElement(id) {
            const el = document.getElementById(id);
            if (el) {
                el.style.opacity = '0';
                el.style.display = '';
            }
        }
        
        // 更新步骤状态
        function updateStep(step) {
            currentState.step = step;
            
            const config = stepConfig[step];
            
            // 更新所有线条
            ['scene3-line-a', 'scene3-line-b'].forEach(id => {
                if (config.lines.includes(id)) {
                    showElement(id);
                } else {
                    hideElement(id);
                }
            });
            
            // 更新所有面积
            ['scene3-area-a', 'scene3-area-b', 'scene3-area-rel'].forEach(id => {
                if (config.areas.includes(id)) {
                    showElement(id);
                } else {
                    hideElement(id);
                }
            });
            
            // 更新所有点
            ['scene3-point-intersect'].forEach(id => {
                if (config.points.includes(id)) {
                    showElement(id);
                } else {
                    hideElement(id);
                }
            });
            
            // 更新所有标签
            ['scene3-label-v0', 'scene3-label-t-common', 'scene3-label-v-common', 'scene3-label-sa', 'scene3-label-sb', 'scene3-label-rel'].forEach(id => {
                if (config.labels.includes(id)) {
                    showElement(id);
                } else {
                    hideElement(id);
                }
            });
            
            // 更新方程
            if (config.equation) {
                showElement('scene3-equation');
            } else {
                hideElement('scene3-equation');
            }
            
            // 更新步骤指示器
            document.getElementById('scene3-step-indicator').textContent = `步骤 ${step + 1}/${totalSteps}`;
            
            // 更新按钮状态
            document.getElementById('scene3-btn-prev').disabled = (step === 0);
            document.getElementById('scene3-btn-next').disabled = (step === totalSteps - 1);
        }
        
        // 按钮事件绑定
        document.getElementById('scene3-btn-prev').addEventListener('click', function() {
            if (currentState.step > 0) {
                updateStep(currentState.step - 1);
            }
        });
        
        document.getElementById('scene3-btn-next').addEventListener('click', function() {
            if (currentState.step < totalSteps - 1) {
                updateStep(currentState.step + 1);
            }
        });
        
        // 初始化
        updateStep(0);
    })();
    </script>
