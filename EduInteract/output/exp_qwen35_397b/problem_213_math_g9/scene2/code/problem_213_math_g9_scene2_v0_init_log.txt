    <script>
    (function() {
        // Scene 2: 节数与总长度的函数关系验证
        let currentStep = 0;
        const totalSteps = 4;

        // 元素引用
        const btnPrev = document.getElementById('scene2-btn-prev');
        const btnNext = document.getElementById('scene2-btn-next');

        // 步骤 1 元素
        const point1 = document.getElementById('scene2-point-1');
        const point2 = document.getElementById('scene2-point-2');
        const point3 = document.getElementById('scene2-point-3');
        const label1 = document.getElementById('scene2-label-1');
        const label2 = document.getElementById('scene2-label-2');
        const label3 = document.getElementById('scene2-label-3');

        // 步骤 2 元素
        const lineFunc = document.getElementById('scene2-line-func');
        const labelFormula = document.getElementById('scene2-label-formula');

        // 步骤 3 元素
        const point100 = document.getElementById('scene2-point-100');
        const label100 = document.getElementById('scene2-label-100');
        const dashV = document.getElementById('scene2-dash-v');
        const dashH = document.getElementById('scene2-dash-h');

        // 步骤 4 元素
        const conclusionRect = document.getElementById('scene2-conclusion-rect');
        const conclusionText = document.getElementById('scene2-conclusion-text');

        // 显示元素函数
        function showElement(el) {
            el.style.opacity = '1';
            el.classList.add('scene2-fade-in');
        }

        // 隐藏元素函数
        function hideElement(el) {
            el.style.opacity = '0';
            el.classList.remove('scene2-fade-in');
        }

        // 更新按钮状态
        function updateButtons() {
            btnPrev.disabled = currentStep === 0;
            btnNext.disabled = currentStep === totalSteps;
        }

        // 更新场景状态
        function updateScene() {
            // 步骤 1: 显示 n=1,2,3 的数据点
            if (currentStep >= 1) {
                showElement(point1);
                showElement(point2);
                showElement(point3);
                showElement(label1);
                showElement(label2);
                showElement(label3);
            } else {
                hideElement(point1);
                hideElement(point2);
                hideElement(point3);
                hideElement(label1);
                hideElement(label2);
                hideElement(label3);
            }

            // 步骤 2: 显示函数直线
            if (currentStep >= 2) {
                showElement(lineFunc);
                showElement(labelFormula);
            } else {
                hideElement(lineFunc);
                hideElement(labelFormula);
            }

            // 步骤 3: 显示 n=100 的点和辅助线
            if (currentStep >= 3) {
                showElement(point100);
                showElement(label100);
                showElement(dashV);
                showElement(dashH);
            } else {
                hideElement(point100);
                hideElement(label100);
                hideElement(dashV);
                hideElement(dashH);
            }

            // 步骤 4: 显示结论框
            if (currentStep >= 4) {
                showElement(conclusionRect);
                showElement(conclusionText);
            } else {
                hideElement(conclusionRect);
                hideElement(conclusionText);
            }

            updateButtons();
        }

        // 按钮事件
        btnPrev.addEventListener('click', function() {
            if (currentStep > 0) {
                currentStep--;
                updateScene();
            }
        });

        btnNext.addEventListener('click', function() {
            if (currentStep < totalSteps) {
                currentStep++;
                updateScene();
            }
        });

        // 初始化
        updateButtons();
    })();
    </script>
