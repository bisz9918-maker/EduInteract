    <script>
    (function() {
        // Scene 1 实现
        let currentStep = 0;
        const totalSteps = 5;
        
        // 显示/隐藏工具函数
        function showElement(id) {
            const el = document.getElementById(id);
            if (el) {
                el.style.display = 'block';
                el.style.opacity = '1';
            }
        }
        
        function hideElement(id) {
            const el = document.getElementById(id);
            if (el) {
                el.style.display = 'none';
                el.style.opacity = '0';
            }
        }
        
        // 更新显示状态
        function updateDisplay() {
            // 隐藏所有步骤组
            for (let i = 1; i < totalSteps; i++) {
                hideElement('scene1-step' + i);
            }
            
            // 显示当前及之前步骤的元素
            for (let i = 1; i <= currentStep; i++) {
                showElement('scene1-step' + i);
            }
            
            // 更新按钮状态
            const prevBtn = document.getElementById('scene1-btn-prev-bg');
            const nextBtn = document.getElementById('scene1-btn-next-bg');
            const prevText = document.getElementById('scene1-btn-prev-text');
            const nextText = document.getElementById('scene1-btn-next-text');
            
            if (currentStep === 0) {
                prevBtn.setAttribute('fill', '#cccccc');
                prevText.setAttribute('fill', '#999999');
            } else {
                prevBtn.setAttribute('fill', '#f0f0f0');
                prevText.setAttribute('fill', '#000000');
            }
            
            if (currentStep === totalSteps - 1) {
                nextBtn.setAttribute('fill', '#cccccc');
                nextText.setAttribute('fill', '#999999');
            } else {
                nextBtn.setAttribute('fill', '#f0f0f0');
                nextText.setAttribute('fill', '#000000');
            }
            
            // 更新步骤指示器
            document.getElementById('scene1-step-indicator').textContent = '步骤 ' + (currentStep + 1) + '/' + totalSteps;
        }
        
        // 上一步
        function prevStep() {
            if (currentStep > 0) {
                currentStep--;
                updateDisplay();
            }
        }
        
        // 下一步
        function nextStep() {
            if (currentStep < totalSteps - 1) {
                currentStep++;
                updateDisplay();
            }
        }
        
        // 绑定按钮点击事件
        document.getElementById('scene1-btn-prev-bg').addEventListener('click', prevStep);
        document.getElementById('scene1-btn-next-bg').addEventListener('click', nextStep);
        
        // 初始化
        updateDisplay();
    })();
    </script>
