    <script>
    (function() {
        // Scene 3 实现：二次函数 y = x² - 2x - 3 的性质综合展示
        
        // 布局坐标表（实现对照依据）：
        // | 元素 | 类型 | 位置/范围 | 大小 | 标签偏移方向 | 备注 |
        // | x 轴 | 线段 | (80,280)-(720,280) | 640×3 | - | 水平居中 |
        // | y 轴 | 线段 | (360,50)-(360,450) | 3×400 | - | 垂直居中 |
        // | 抛物线 | 路径 | x∈[-2,4] 对应画布 | 曲线 | - | y=x²-2x-3 |
        // | 对称轴 x=1 | 虚线 | (440,50)-(440,450) | 2×400 | 上方 | 灰色虚线 |
        // | 顶点标记 | 圆点 | (440,440) | r=6 | 右下方 | 红色实心圆 |
        // | 区间 [0,2] 填充 | 矩形 | (360,280)-(520,420) | 160×140 | 下方 | 蓝色半透明 |
        // | x>1 区域填充 | 矩形 | (440,50)-(640,400) | 200×350 | - | 绿色半透明 12% |
        // | 递增箭头 | 箭头路径 | (500,350)→(640,120) | 曲线 | - | 沿抛物线右侧向上 |
        // | A 点 | 标记 | (280,280) | r=4 | 左上方 | x 轴交点 (-1,0) |
        // | B 点 | 标记 | (600,280) | r=4 | 右上方 | x 轴交点 (3,0) |
        // | C 点 | 标记 | (360,400) | r=4 | 左侧 | y 轴交点 (0,-3) |
        
        // 状态管理
        let currentState = 0;
        const totalSteps = 4;
        
        // 每步需要显示的元素
        const steps = {
            0: ['scene3-sym-axis', 'scene3-label-sym'],
            1: ['scene3-vertex', 'scene3-label-vertex'],
            2: ['scene3-interval-rect', 'scene3-interval-parabola', 'scene3-label-interval'],
            3: ['scene3-label-range'],
            4: ['scene3-increase-area', 'scene3-increase-arrow', 'scene3-label-increase-x', 'scene3-label-increase-y']
        };
        
        // 工具函数：显示元素
        function showElement(id) {
            const el = document.getElementById(id);
            if (el) {
                el.classList.remove('scene3-hidden');
                el.classList.add('scene3-visible', 'scene3-fade-in');
            }
        }
        
        // 工具函数：隐藏元素
        function hideElement(id) {
            const el = document.getElementById(id);
            if (el) {
                el.classList.remove('scene3-visible', 'scene3-fade-in');
                el.classList.add('scene3-hidden');
            }
        }
        
        // 更新显示状态
        function updateDisplay() {
            // 隐藏所有步骤专属元素
            for (let step = 0; step <= totalSteps; step++) {
                if (steps[step]) {
                    steps[step].forEach(id => hideElement(id));
                }
            }
            
            // 显示当前步骤及之前步骤的元素
            for (let step = 0; step <= currentState; step++) {
                if (steps[step]) {
                    steps[step].forEach(id => showElement(id));
                }
            }
            
            // 更新步骤指示器
            document.getElementById('scene3-step-indicator').textContent = `步骤 ${currentState} / ${totalSteps}`;
            
            // 更新按钮状态
            document.getElementById('scene3-btn-prev').disabled = (currentState === 0);
            document.getElementById('scene3-btn-next').disabled = (currentState === totalSteps);
        }
        
        // 事件处理：上一步
        document.getElementById('scene3-btn-prev').addEventListener('click', function() {
            if (currentState > 0) {
                currentState--;
                updateDisplay();
            }
        });
        
        // 事件处理：下一步
        document.getElementById('scene3-btn-next').addEventListener('click', function() {
            if (currentState < totalSteps) {
                currentState++;
                updateDisplay();
            }
        });
        
        // 初始化显示
        updateDisplay();
    })();
    </script>
