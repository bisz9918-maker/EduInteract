    <script>
    (function() {
        // Scene 2 实现：面积割补法——揭示 S₁ 与 S₂ 的等量关系
        
        // 状态机：当前步骤 (0-5)
        let currentStep = 0;
        const totalSteps = 5;
        
        // 元素可见性配置：每步显示哪些元素
        const stepConfig = {
            0: {
                // 初始状态：只显示基础图形
                triangles: [],
                labels: [],
                formulas: [],
                conclusions: []
            },
            1: {
                // 步骤 1：填充△OAD
                triangles: ['scene2-tri-oad'],
                labels: ['scene2-label-halfk-oad'],
                formulas: [],
                conclusions: []
            },
            2: {
                // 步骤 2：填充△OBC
                triangles: ['scene2-tri-oad', 'scene2-tri-obc'],
                labels: ['scene2-label-halfk-oad', 'scene2-label-halfk-obc'],
                formulas: [],
                conclusions: []
            },
            3: {
                // 步骤 3：显示面积相等关系
                triangles: ['scene2-tri-oad', 'scene2-tri-obc'],
                labels: ['scene2-label-halfk-oad', 'scene2-label-halfk-obc'],
                formulas: ['scene2-formula-relation'],
                conclusions: []
            },
            4: {
                // 步骤 4：填充△OAB 并显示面积割补公式
                triangles: ['scene2-tri-oad', 'scene2-tri-obc', 'scene2-tri-oab'],
                labels: ['scene2-label-halfk-oad', 'scene2-label-halfk-obc', 'scene2-label-s1', 'scene2-label-s2'],
                formulas: ['scene2-formula-relation', 'scene2-formula-cut'],
                conclusions: []
            },
            5: {
                // 步骤 5：得出结论
                triangles: ['scene2-tri-oad', 'scene2-tri-obc', 'scene2-tri-oab'],
                labels: ['scene2-label-halfk-oad', 'scene2-label-halfk-obc', 'scene2-label-s1', 'scene2-label-s2'],
                formulas: ['scene2-formula-relation', 'scene2-formula-cut'],
                conclusions: ['scene2-conclusion-eq', 'scene2-conclusion-ratio']
            }
        };
        
        // 工具函数：显示元素
        function showElement(id) {
            const el = document.getElementById('scene2-' + id.replace('scene2-', ''));
            if (el) {
                el.classList.remove('scene2-hidden');
                el.style.opacity = '1';
                el.style.display = '';
            }
        }
        
        // 工具函数：隐藏元素
        function hideElement(id) {
            const el = document.getElementById('scene2-' + id.replace('scene2-', ''));
            if (el) {
                el.classList.add('scene2-hidden');
                el.style.opacity = '0';
            }
        }
        
        // 更新步骤显示
        function updateStep() {
            const config = stepConfig[currentStep];
            
            // 所有三角形元素
            const allTriangles = ['scene2-tri-oad', 'scene2-tri-obc', 'scene2-tri-oab'];
            allTriangles.forEach(id => {
                if (config.triangles.includes(id)) {
                    showElement(id);
                } else {
                    hideElement(id);
                }
            });
            
            // 所有标签元素
            const allLabels = ['scene2-label-halfk-oad', 'scene2-label-halfk-obc', 'scene2-label-s1', 'scene2-label-s2'];
            allLabels.forEach(id => {
                if (config.labels.includes(id)) {
                    showElement(id);
                } else {
                    hideElement(id);
                }
            });
            
            // 所有公式元素
            const allFormulas = ['scene2-formula-relation', 'scene2-formula-cut'];
            allFormulas.forEach(id => {
                if (config.formulas.includes(id)) {
                    showElement(id);
                } else {
                    hideElement(id);
                }
            });
            
            // 所有结论元素
            const allConclusions = ['scene2-conclusion-eq', 'scene2-conclusion-ratio'];
            allConclusions.forEach(id => {
                if (config.conclusions.includes(id)) {
                    showElement(id);
                } else {
                    hideElement(id);
                }
            });
            
            // 更新步骤指示器
            document.getElementById('scene2-step-indicator').textContent = '步骤 ' + currentStep + '/' + totalSteps;
            
            // 更新按钮状态
            const prevBtn = document.getElementById('scene2-btn-prev');
            const nextBtn = document.getElementById('scene2-btn-next');
            prevBtn.disabled = (currentStep === 0);
            nextBtn.disabled = (currentStep === totalSteps);
            
            // 触发 MathJax 重新渲染（如果有公式变化）
            if (window.MathJax && window.MathJax.typesetPromise) {
                window.MathJax.typesetPromise();
            }
        }
        
        // 按钮事件处理
        function goToPrevStep() {
            if (currentStep > 0) {
                currentStep--;
                updateStep();
            }
        }
        
        function goToNextStep() {
            if (currentStep < totalSteps) {
                currentStep++;
                updateStep();
            }
        }
        
        // 初始化
        document.getElementById('scene2-btn-prev').addEventListener('click', goToPrevStep);
        document.getElementById('scene2-btn-next').addEventListener('click', goToNextStep);
        
        // 初始状态
        updateStep();
    })();
    </script>
