    <script>
    (function() {
        // 状态管理
        let currentState = 0;
        const totalSteps = 5;

        // DOM 元素引用
        const btnPrev = document.getElementById('scene2-btn-prev');
        const btnNext = document.getElementById('scene2-btn-next');
        const stepIndicator = document.getElementById('scene2-step-indicator');

        // 各步骤需要显示的元素 ID 列表
        const stepElements = {
            1: [
                'scene2-latitude-30N',
                'scene2-latitude-60N',
                'scene2-pressure-subtropical',
                'scene2-pressure-subpolar'
            ],
            2: [
                'scene2-warm-flow',
                'scene2-cold-flow',
                'scene2-warm-rise',
                'scene2-label-cold-warm-meet'
            ],
            3: [
                'scene2-box-dynamic-cause',
                'scene2-label-dynamic-cause'
            ],
            4: [
                'scene2-box-cause-compare',
                'scene2-label-cause-compare'
            ],
            5: [
                'scene2-box-conclusion',
                'scene2-label-conclusion'
            ]
        };

        // 显示元素函数
        function showElement(id) {
            const el = document.getElementById(id);
            if (el) {
                el.style.display = 'block';
                // 触发动画
                setTimeout(() => {
                    el.classList.add('scene2-visible');
                }, 10);
            }
        }

        // 隐藏元素函数
        function hideElement(id) {
            const el = document.getElementById(id);
            if (el) {
                el.style.display = 'none';
                el.classList.remove('scene2-visible');
            }
        }

        // 更新状态
        function updateState() {
            // 更新按钮状态
            btnPrev.disabled = currentState === 0;
            btnNext.disabled = currentState === totalSteps;
            btnNext.textContent = currentState === totalSteps ? '已完成' : '下一步 ▶';

            // 更新步骤指示器
            stepIndicator.textContent = `步骤 ${currentState}/${totalSteps}`;

            // 步骤 3 特殊处理：高亮④箭头为绿色
            const arrow4 = document.getElementById('scene2-line-da');
            if (currentState >= 3) {
                arrow4.style.stroke = '#2e7d32';
                arrow4.setAttribute('marker-end', 'url(#scene2-arrowhead-green)');
            } else {
                arrow4.style.stroke = '#000000';
                arrow4.setAttribute('marker-end', 'url(#scene2-arrowhead)');
            }
        }

        // 上一步
        function prevStep() {
            if (currentState > 0) {
                // 隐藏当前步骤的元素
                const elementsToHide = stepElements[currentState];
                if (elementsToHide) {
                    elementsToHide.forEach(id => hideElement(id));
                }
                currentState--;
                updateState();
            }
        }

        // 下一步
        function nextStep() {
            if (currentState < totalSteps) {
                currentState++;
                // 显示新步骤的元素
                const elementsToShow = stepElements[currentState];
                if (elementsToHide) {
                    elementsToShow.forEach(id => showElement(id));
                }
                updateState();
            }
        }

        // 绑定事件
        btnPrev.addEventListener('click', prevStep);
        btnNext.addEventListener('click', nextStep);

        // 初始化
        updateState();
    })();
    </script>
