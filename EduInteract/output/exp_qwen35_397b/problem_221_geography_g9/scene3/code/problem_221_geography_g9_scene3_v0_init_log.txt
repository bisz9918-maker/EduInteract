        // | 电池板 | (580,200)-(700,260) | 绿色斜线 |

        let currentStep = 0;
        const totalSteps = 4;

        // 每步显示的元素 ID 列表
        const stepElements = {
            0: [
                'scene3-title',

                'scene3-wall', 'scene3-right-angle',

                'scene3-formula-h'
            ],
            3: [
                'scene3-title',

            ]
        };

        // 工具函数：显示元素
        function showElement(id) {
            const el = document.getElementById(id);
            if (el) {
                el.classList.remove('scene3-hidden');
                el.classList.add('scene3-visible');
            }
        }

        // 工具函数：隐藏元素
        function hideElement(id) {
            const el = document.getElementById(id);
            if (el) {
                el.classList.remove('scene3-visible');
                el.classList.add('scene3-hidden');
            }
        }

        // 更新显示状态
        function updateDisplay() {
            // 隐藏所有元素
            for (let step = 0; step < totalSteps; step++) {
                stepElements[step].forEach(id => hideElement(id));
            }

            // 显示当前步骤及之前步骤的所有元素
            for (let step = 0; step <= currentStep; step++) {
                stepElements[step].forEach(id => showElement(id));
            }

            // 更新步骤指示器
            document.getElementById('scene3-step-indicator').textContent = `步骤 ${currentStep + 1}/${totalSteps}`;

            // 更新按钮状态
            document.getElementById('scene3-prev-btn').disabled = currentStep === 0;
            document.getElementById('scene3-next-btn').disabled = currentStep === totalSteps - 1;
        }

        // 事件处理
        document.getElementById('scene3-prev-btn').addEventListener('click', function() {
            if (currentStep > 0) {
                currentStep--;
                updateDisplay();
            }
        });

        document.getElementById('scene3-next-btn').addEventListener('click', function() {
            if (currentStep < totalSteps - 1) {
                currentStep++;
                updateDisplay();
            }
        });

        // 初始化
        updateDisplay();
    })();
