    <script>
    (function() {
        // Scene 2 实现：经纬网方向判断

        // 状态变量
        let currentStep = 0;
        const totalSteps = 4; // 0-3

        // 步骤说明文字
        const stepTexts = [
            '观察①点和③点在经纬网上的位置',
            '第一步：比较纬度——①点 (0°) 在③点 (35°N) 的<span class="scene2-highlight-text">南方</span>',
            '第二步：比较经度——①点 (120°W) 在③点 (120°E) 的<span class="scene2-highlight-text">西方</span>',
            '结论：①点在③点的<span class="scene2-highlight-text">东南方向</span>'
        ];

        // 获取 DOM 元素
        const nsLine = document.getElementById('scene2-ns-line');
        const ewLine = document.getElementById('scene2-ew-line');
        const directionArrow = document.getElementById('scene2-direction-arrow');
        const conclusion = document.getElementById('scene2-conclusion');
        const highlightNs = document.getElementById('scene2-highlight-ns');
        const highlightEw = document.getElementById('scene2-highlight-ew');
        const stepText = document.getElementById('scene2-step-text');
        const prevBtn = document.getElementById('scene2-prev-btn');
        const nextBtn = document.getElementById('scene2-next-btn');

        // 更新视图函数
        function updateView() {
            // 隐藏所有辅助元素
            nsLine.style.opacity = '0';
            ewLine.style.opacity = '0';
            directionArrow.style.opacity = '0';
            conclusion.style.opacity = '0';
            highlightNs.style.opacity = '0';
            highlightEw.style.opacity = '0';

            // 根据当前步骤显示相应元素
            switch (currentStep) {
                case 0:
                    // 初始状态：只显示基础元素
                    stepText.innerHTML = stepTexts[0];
                    break;
                case 1:
                    // 步骤 1：显示南北辅助线和高亮
                    nsLine.style.opacity = '1';
                    highlightNs.style.opacity = '0.6';
                    stepText.innerHTML = stepTexts[1];
                    break;
                case 2:
                    // 步骤 2：显示东西辅助线和高亮
                    ewLine.style.opacity = '1';
                    highlightEw.style.opacity = '0.6';
                    stepText.innerHTML = stepTexts[2];
                    break;
                case 3:
                    // 步骤 3：显示方向箭头和结论
                    directionArrow.style.opacity = '1';
                    conclusion.style.opacity = '1';
                    stepText.innerHTML = stepTexts[3];
                    break;
            }

            // 更新按钮状态
            prevBtn.disabled = (currentStep === 0);
            if (currentStep === totalSteps - 1) {
                nextBtn.textContent = '完成';
                nextBtn.disabled = true;
            } else {
                nextBtn.textContent = '下一步 ▶';
                nextBtn.disabled = false;
            }
        }

        // 事件处理
        prevBtn.addEventListener('click', function() {
            if (currentStep > 0) {
                currentStep--;
                updateView();
            }
        });

        nextBtn.addEventListener('click', function() {
            if (currentStep < totalSteps - 1) {
                currentStep++;
                updateView();
            }
        });

        // 初始化视图
        updateView();
    })();
    </script>
