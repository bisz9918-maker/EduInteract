    <script>
    (function() {
        // Scene 2 实现：球面最短路径与航向变化

        // 贝塞尔曲线参数
        const P0 = { x: 160, y: 350 };  // B 点
        const P1 = { x: 400, y: 220 };  // 控制点
        const P2 = { x: 640, y: 350 };  // D 点

        // 获取 DOM 元素
        const slider = document.getElementById('scene2-slider');
        const headingText = document.getElementById('scene2-heading-text');
        const currentPoint = document.getElementById('scene2-current-point');
        const headingArrow = document.getElementById('scene2-heading-arrow');
        const dynamicGroup = document.getElementById('scene2-dynamic-group');

        // 计算贝塞尔曲线上的点
        function getBezierPoint(t) {
            const oneMinusT = 1 - t;
            const x = oneMinusT * oneMinusT * P0.x + 2 * oneMinusT * t * P1.x + t * t * P2.x;
            const y = oneMinusT * oneMinusT * P0.y + 2 * oneMinusT * t * P1.y + t * t * P2.y;
            return { x, y };
        }

        // 计算贝塞尔曲线切线方向
        function getBezierTangent(t) {
            const oneMinusT = 1 - t;
            // 导数：B'(t) = 2(1-t)(P1-P0) + 2t(P2-P1)
            const dx = 2 * oneMinusT * (P1.x - P0.x) + 2 * t * (P2.x - P1.x);
            const dy = 2 * oneMinusT * (P1.y - P0.y) + 2 * t * (P2.y - P1.y);
            const angle = Math.atan2(dy, dx) * 180 / Math.PI;
            return angle;
        }

        // 根据滑块值更新航向文字
        function updateHeadingText(t) {
            let heading;
            if (t < 0.35) {
                heading = '东北';
            } else if (t < 0.65) {
                heading = '东';
            } else {
                heading = '东南';
            }
            headingText.textContent = heading;
        }

        // 更新动态元素位置
        function updateDynamicElements(t) {
            const point = getBezierPoint(t);
            const angle = getBezierTangent(t);

            // 更新当前位置点
            currentPoint.setAttribute('cx', point.x);
            currentPoint.setAttribute('cy', point.y);

            // 更新航向箭头位置和旋转
            headingArrow.setAttribute('transform', `translate(${point.x}, ${point.y}) rotate(${angle})`);

            // 更新航向文字
            updateHeadingText(t);
        }

        // 滑块事件监听
        slider.addEventListener('input', function() {
            const t = this.value / 100;
            updateDynamicElements(t);
        });

        // 初始化
        updateDynamicElements(0);

    })();
    </script>
