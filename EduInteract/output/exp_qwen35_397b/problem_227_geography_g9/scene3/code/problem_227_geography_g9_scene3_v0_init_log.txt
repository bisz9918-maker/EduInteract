    <script>
    (function() {
        // Scene 3 实现 - 经纬网图题目解答总结
        
        // 状态变量
        let currentState = 0; // 0=初始，1=小题 1, 2=小题 2, 3=完整答案
        
        // 获取 DOM 元素
        const btnBg1 = document.getElementById('scene3-btn-bg1');
        const btnBg2 = document.getElementById('scene3-btn-bg2');
        const btnBg3 = document.getElementById('scene3-btn-bg3');
        const btn1Text = document.getElementById('scene3-btn1-text');
        const btn2Text = document.getElementById('scene3-btn2-text');
        const btn3Text = document.getElementById('scene3-btn3-text');
        
        const ctrlBtn1Bg = document.getElementById('scene3-ctrl-btn1-bg');
        const ctrlBtn2Bg = document.getElementById('scene3-ctrl-btn2-bg');
        const ctrlBtn3Bg = document.getElementById('scene3-ctrl-btn3-bg');
        const ctrlBtn1 = document.getElementById('scene3-ctrl-btn1');
        const ctrlBtn2 = document.getElementById('scene3-ctrl-btn2');
        const ctrlBtn3 = document.getElementById('scene3-ctrl-btn3');
        
        const hemisphereGroup = document.getElementById('scene3-hemisphere');
        const pathGroup = document.getElementById('scene3-path-group');
        const pathBd = document.getElementById('scene3-path-bd');
        const markQ1 = document.getElementById('scene3-mark-q1');
        const markQ2 = document.getElementById('scene3-mark-q2');
        const explainText = document.getElementById('scene3-explain-text');
        const summaryText = document.getElementById('scene3-summary-text');
        
        // 更新按钮样式
        function updateButtons() {
            // 重置所有按钮
            btnBg1.setAttribute('fill', '#e0e0e0');
            btnBg2.setAttribute('fill', '#e0e0e0');
            btnBg3.setAttribute('fill', '#e0e0e0');
            btn1Text.setAttribute('fill', '#000000');
            btn2Text.setAttribute('fill', '#000000');
            btn3Text.setAttribute('fill', '#000000');
            
            ctrlBtn1Bg.setAttribute('fill', '#e0e0e0');
            ctrlBtn2Bg.setAttribute('fill', '#e0e0e0');
            ctrlBtn3Bg.setAttribute('fill', '#e0e0e0');
            ctrlBtn1.setAttribute('fill', '#000000');
            ctrlBtn2.setAttribute('fill', '#000000');
            ctrlBtn3.setAttribute('fill', '#000000');
            
            // 设置当前激活按钮
            if (currentState === 1) {
                btnBg1.setAttribute('fill', '#1565c0');
                btn1Text.setAttribute('fill', '#ffffff');
                ctrlBtn1Bg.setAttribute('fill', '#1565c0');
                ctrlBtn1.setAttribute('fill', '#ffffff');
            } else if (currentState === 2) {
                btnBg2.setAttribute('fill', '#1565c0');
                btn2Text.setAttribute('fill', '#ffffff');
                ctrlBtn2Bg.setAttribute('fill', '#1565c0');
                ctrlBtn2.setAttribute('fill', '#ffffff');
            } else if (currentState === 3) {
                btnBg3.setAttribute('fill', '#1565c0');
                btn3Text.setAttribute('fill', '#ffffff');
                ctrlBtn3Bg.setAttribute('fill', '#1565c0');
                ctrlBtn3.setAttribute('fill', '#ffffff');
            }
        }
        
        // 更新显示内容
        function updateDisplay() {
            // 隐藏所有状态特定元素
            hemisphereGroup.style.opacity = '0';
            pathGroup.style.opacity = '0';
            markQ1.style.opacity = '0';
            markQ2.style.opacity = '0';
            explainText.style.opacity = '0';
            summaryText.style.opacity = '0';
            pathBd.classList.remove('solid');
            
            if (currentState === 0) {
                // 初始状态 - 只显示基础经纬网和点
                updateButtons();
            } else if (currentState === 1) {
                // 小题 1 解析
                hemisphereGroup.style.opacity = '1';
                markQ1.style.opacity = '1';
                explainText.textContent = '东半球范围是 20°W~160°E，A、B、C、D 均位于西半球';
                explainText.style.opacity = '1';
                updateButtons();
            } else if (currentState === 2) {
                // 小题 2 解析
                pathGroup.style.opacity = '1';
                pathBd.classList.add('solid');
                markQ2.style.opacity = '1';
                explainText.textContent = '球面上两点间最短距离是大圆劣弧，从 B 到 D 先向东北再向东南';
                explainText.style.opacity = '1';
                updateButtons();
            } else if (currentState === 3) {
                // 完整答案
                markQ1.style.opacity = '1';
                markQ2.style.opacity = '1';
                markQ1.setAttribute('y', '300');
                markQ2.setAttribute('y', '350');
                explainText.textContent = '【答案】小题 1：错误；小题 2：正确';
                explainText.style.opacity = '1';
                summaryText.textContent = '【知识点】①东西半球分界：20°W 和 160°E；②球面最短路径为大圆劣弧';
                summaryText.style.opacity = '1';
                updateButtons();
            }
        }
        
        // 按钮点击事件
        function setState(state) {
            currentState = state;
            updateDisplay();
        }
        
        // 顶部按钮
        btnBg1.addEventListener('click', () => setState(1));
        btnBg2.addEventListener('click', () => setState(2));
        btnBg3.addEventListener('click', () => setState(3));
        btn1Text.addEventListener('click', () => setState(1));
        btn2Text.addEventListener('click', () => setState(2));
        btn3Text.addEventListener('click', () => setState(3));
        
        // 底部控件栏按钮
        ctrlBtn1Bg.addEventListener('click', () => setState(1));
        ctrlBtn2Bg.addEventListener('click', () => setState(2));
        ctrlBtn3Bg.addEventListener('click', () => setState(3));
        ctrlBtn1.addEventListener('click', () => setState(1));
        ctrlBtn2.addEventListener('click', () => setState(2));
        ctrlBtn3.addEventListener('click', () => setState(3));
        
        // 初始化
        updateDisplay();
    })();
    </script>
