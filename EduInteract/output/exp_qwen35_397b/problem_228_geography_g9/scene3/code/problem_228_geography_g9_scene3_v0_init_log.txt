    <script>
    (function() {
        // Scene 3 实现 - 晨昏线与地方时计算
        
        let currentStep = 0;
        const totalSteps = 5;
        
        const btnPrev = document.getElementById('scene3-btn-prev');
        const btnNext = document.getElementById('scene3-btn-next');
        const stepIndicator = document.getElementById('scene3-step-indicator');
        
        const stepElements = {
            1: ['scene3-step1-group', 'scene3-answer-q2'],
            2: ['scene3-step2-group', 'scene3-answer-q3'],
            3: ['scene3-step3-group', 'scene3-answer-q4'],
            4: ['scene3-step4-group', 'scene3-answer-q5']
        };
        
        function showElement(elementId) {
            const el = document.getElementById(elementId);
            if (el) {
                el.style.display = 'block';
                el.style.opacity = '0';
                setTimeout(() => {
                    el.style.opacity = '1';
                    el.classList.add('scene3-fade-in');
                }, 10);
            }
        }
        
        function hideElement(elementId) {
            const el = document.getElementById(elementId);
            if (el) {
                el.style.display = 'none';
                el.style.opacity = '0';
                el.classList.remove('scene3-fade-in');
            }
        }
        
        function updateButtons() {
            const prevRect = btnPrev.querySelector('rect');
            const prevText = btnPrev.querySelector('text');
            const nextRect = btnNext.querySelector('rect');
            const nextText = btnNext.querySelector('text');
            
            if (currentStep === 0) {
                prevRect.setAttribute('fill', '#e0e0e0');
                prevText.setAttribute('fill', '#888');
                btnPrev.classList.add('disabled');
            } else {
                prevRect.setAttribute('fill', '#f5f5f5');
                prevText.setAttribute('fill', '#000000');
                btnPrev.classList.remove('disabled');
            }
            
            if (currentStep === totalSteps - 1) {
                nextRect.setAttribute('fill', '#e0e0e0');
                nextText.setAttribute('fill', '#888');
                btnNext.classList.add('disabled');
            } else {
                nextRect.setAttribute('fill', '#f5f5f5');
                nextText.setAttribute('fill', '#000000');
                btnNext.classList.remove('disabled');
            }
        }
        
        function updateStepDisplay() {
            stepIndicator.textContent = `步骤 ${currentStep}/5`;
            updateButtons();
            
            for (let step = 1; step <= 4; step++) {
                const elements = stepElements[step];
                if (elements) {
                    elements.forEach(elId => {
                        if (step <= currentStep) {
                            showElement(elId);
                        } else {
                            hideElement(elId);
                        }
                    });
                }
            }
        }
        
        btnPrev.addEventListener('click', function() {
            if (currentStep > 0) {
                currentStep--;
                updateStepDisplay();
            }
        });
        
        btnNext.addEventListener('click', function() {
            if (currentStep < totalSteps - 1) {
                currentStep++;
                updateStepDisplay();
            }
        });
        
        updateStepDisplay();
        
    })();
    </script>
