    <script>
    (function() {
        // Scene 2: 原电池工作原理
        
        let currentStep = 0;
        const totalSteps = 4;
        
        // 元素引用
        const wire = document.getElementById('scene2-wire');
        const electronArrow = document.getElementById('scene2-electron-arrow');
        const labelE = document.getElementById('scene2-label-e');
        const labelPositive = document.getElementById('scene2-label-positive');
        const labelNegative = document.getElementById('scene2-label-negative');
        const ionCuArrow = document.getElementById('scene2-ion-cu-arrow');
        const labelCuIon = document.getElementById('scene2-label-cu-ion');
        const ionSo4Arrow = document.getElementById('scene2-ion-so4-arrow');
        const labelSo4Ion = document.getElementById('scene2-label-so4-ion');
        const ions = document.getElementById('scene2-ions');
        const reactionCu = document.getElementById('scene2-reaction-cu');
        const reactionFe = document.getElementById('scene2-reaction-fe');
        const solution = document.getElementById('scene2-solution');
        const electrodeCu = document.getElementById('scene2-electrode-cu');
        const electrodeFe = document.getElementById('scene2-electrode-fe');
        const stepIndicator = document.getElementById('scene2-step-indicator');
        const btnPrev = document.getElementById('scene2-btn-prev');
        const btnNext = document.getElementById('scene2-btn-next');
        
        // 工具函数：显示/隐藏元素
        function show(el) {
            el.classList.remove('scene2-hidden');
            el.classList.add('scene2-visible');
            el.style.display = 'block';
            el.style.opacity = '1';
        }
        
        function hide(el) {
            el.classList.remove('scene2-visible');
            el.classList.add('scene2-hidden');
            el.style.display = 'none';
            el.style.opacity = '0';
        }
        
        // 更新步骤显示
        function updateStep() {
            stepIndicator.textContent = `步骤 ${currentStep}/${totalSteps}`;
            
            // 按钮状态
            btnPrev.disabled = currentStep === 0;
            btnPrev.style.opacity = currentStep === 0 ? '0.5' : '1';
            btnNext.disabled = currentStep === totalSteps;
            btnNext.style.opacity = currentStep === totalSteps ? '0.5' : '1';
            
            // 根据步骤显示/隐藏元素
            resetElements();
            
            if (currentStep >= 0) {
                // 步骤 0: 显示导线
                show(wire);
            }
            if (currentStep >= 1) {
                // 步骤 1: 显示电子流向、正负极标识
                show(electronArrow);
                show(labelE);
                show(labelPositive);
                show(labelNegative);
            }
            if (currentStep >= 2) {
                // 步骤 2: 显示离子移动
                show(ionCuArrow);
                show(labelCuIon);
                show(ionSo4Arrow);
                show(labelSo4Ion);
                show(ions);
            }
            if (currentStep >= 3) {
                // 步骤 3: 显示电极反应式、开始溶解/析出效果
                show(reactionCu);
                show(reactionFe);
                // 铁电极溶解（高度减少）
                electrodeFe.setAttribute('height', '180');
                // 铜电极析出（宽度增加）
                electrodeCu.setAttribute('width', '45');
                electrodeCu.setAttribute('x', '277.5');
            }
            if (currentStep >= 4) {
                // 步骤 4: 溶液颜色变化
                solution.setAttribute('fill', '#a5d6a7');
            }
        }
        
        // 重置所有元素到初始状态
        function resetElements() {
            hide(electronArrow);
            hide(labelE);
            hide(labelPositive);
            hide(labelNegative);
            hide(ionCuArrow);
            hide(labelCuIon);
            hide(ionSo4Arrow);
            hide(labelSo4Ion);
            hide(ions);
            hide(reactionCu);
            hide(reactionFe);
            
            // 重置电极形状
            electrodeFe.setAttribute('height', '200');
            electrodeFe.setAttribute('y', '180');
            electrodeCu.setAttribute('width', '40');
            electrodeCu.setAttribute('x', '280');
            
            // 重置溶液颜色
            solution.setAttribute('fill', '#bbdefb');
        }
        
        // 事件监听
        btnPrev.addEventListener('click', function() {
            if (currentStep > 0) {
                currentStep--;
                updateStep();
            }
        });
        
        btnNext.addEventListener('click', function() {
            if (currentStep < totalSteps) {
                currentStep++;
                updateStep();
            }
        });
        
        // 初始化
        updateStep();
    })();
    </script>
