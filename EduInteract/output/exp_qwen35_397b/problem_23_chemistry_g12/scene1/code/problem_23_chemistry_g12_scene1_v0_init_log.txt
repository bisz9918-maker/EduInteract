    <script>
    (function() {
        // Scene 1: H₂O₂分子结构与极性分析

        // 状态管理
        let currentStep = 0;
        const totalSteps = 4;

        // 各步骤配置
        const steps = [
            {
                // 步骤 0：初始状态
                analysisTitle: "H₂O₂分子的空间结构",
                analysisLines: ["", "", "", "", ""],
                showAngleArc: false,
                showChargeMarks: false,
                showBondLabels: false,
                showMoleculeOutline: false,
                showConclusion: false,
                prevDisabled: true,
                nextDisabled: false
            },
            {
                // 步骤 1：显示键类型
                analysisTitle: "H₂O₂分子的空间结构",
                analysisLines: [
                    "• O-H 键：极性共价键",
                    "• O-O 键：非极性共价键",
                    "", "", ""
                ],
                showAngleArc: false,
                showChargeMarks: false,
                showBondLabels: true,
                showMoleculeOutline: false,
                showConclusion: false,
                prevDisabled: false,
                nextDisabled: false
            },
            {
                // 步骤 2：显示电荷中心
                analysisTitle: "H₂O₂分子的空间结构",
                analysisLines: [
                    "• O-H 键：极性共价键",
                    "• O-O 键：非极性共价键",
                    "• 分子呈二面角构型",
                    "• 正负电荷中心不重合",
                    ""
                ],
                showAngleArc: true,
                showChargeMarks: true,
                showBondLabels: true,
                showMoleculeOutline: false,
                showConclusion: false,
                prevDisabled: false,
                nextDisabled: false
            },
            {
                // 步骤 3：最终结论
                analysisTitle: "H₂O₂分子的空间结构",
                analysisLines: [
                    "• O-H 键：极性共价键",
                    "• O-O 键：非极性共价键",
                    "• 分子呈二面角构型",
                    "• 正负电荷中心不重合",
                    "→ 电荷分布不对称"
                ],
                showAngleArc: true,
                showChargeMarks: true,
                showBondLabels: true,
                showMoleculeOutline: true,
                showConclusion: true,
                prevDisabled: false,
                nextDisabled: true
            }
        ];

        // 获取 DOM 元素
        const elements = {
            // 化学键
            bondOO: document.getElementById('scene1-bond-oo'),
            bondOH1: document.getElementById('scene1-bond-oh1'),
            bondOH2: document.getElementById('scene1-bond-oh2'),
            // 二面角
            angleArc: document.getElementById('scene1-angle-arc'),
            labelAngle: document.getElementById('scene1-label-angle'),
            // 电荷中心
            chargePlus: document.getElementById('scene1-charge-plus'),
            labelChargePlus: document.getElementById('scene1-label-charge-plus'),
            chargeMinus: document.getElementById('scene1-charge-minus'),
            labelChargeMinus: document.getElementById('scene1-label-charge-minus'),
            // 键类型标签
            labelPolar: document.getElementById('scene1-label-polar'),
            labelNonpolar: document.getElementById('scene1-label-nonpolar'),
            // 分子轮廓
            moleculeOutline: document.getElementById('scene1-molecule-outline'),
            // 分析文字
            analysisTitle: document.getElementById('scene1-analysis-title'),
            analysisLine1: document.getElementById('scene1-analysis-line1'),
            analysisLine2: document.getElementById('scene1-analysis-line2'),
            analysisLine3: document.getElementById('scene1-analysis-line3'),
            analysisLine4: document.getElementById('scene1-analysis-line4'),
            analysisLine5: document.getElementById('scene1-analysis-line5'),
            // 结论
            conclusion: document.getElementById('scene1-conclusion'),
            // 按钮
            btnPrevBg: document.getElementById('scene1-btn-prev-bg'),
            btnPrevText: document.getElementById('scene1-btn-prev-text'),
            btnNextBg: document.getElementById('scene1-btn-next-bg'),
            btnNextText: document.getElementById('scene1-btn-next-text')
        };

        // 工具函数：设置元素可见性
        function show(el) {
            el.style.opacity = '1';
        }

        function hide(el) {
            el.style.opacity = '0';
        }

        // 更新步骤显示
        function updateStep(stepIndex) {
            const step = steps[stepIndex];

            // 更新分析文字
            elements.analysisTitle.textContent = step.analysisTitle;

            const analysisLines = [
                elements.analysisLine1,
                elements.analysisLine2,
                elements.analysisLine3,
                elements.analysisLine4,
                elements.analysisLine5
            ];

            for (let i = 0; i < 5; i++) {
                const line = analysisLines[i];
                const text = step.analysisLines[i];
                if (text) {
                    line.textContent = text;
                    show(line);
                } else {
                    hide(line);
                }
            }

            // 更新二面角弧线
            if (step.showAngleArc) {
                show(elements.angleArc);
                show(elements.labelAngle);
            } else {
                hide(elements.angleArc);
                hide(elements.labelAngle);
            }

            // 更新电荷中心标记
            if (step.showChargeMarks) {
                show(elements.chargePlus);
                show(elements.labelChargePlus);
                show(elements.chargeMinus);
                show(elements.labelChargeMinus);
            } else {
                hide(elements.chargePlus);
                hide(elements.labelChargePlus);
                hide(elements.chargeMinus);
                hide(elements.labelChargeMinus);
            }

            // 更新键类型标签和高亮
            if (step.showBondLabels) {
                show(elements.labelPolar);
                show(elements.labelNonpolar);
                // 高亮 O-H 键
                elements.bondOH1.classList.remove('scene1-bond-oh');
                elements.bondOH1.classList.add('scene1-bond-oh-highlight');
                elements.bondOH2.classList.remove('scene1-bond-oh');
                elements.bondOH2.classList.add('scene1-bond-oh-highlight');
            } else {
                hide(elements.labelPolar);
                hide(elements.labelNonpolar);
                // 恢复 O-H 键正常样式
                elements.bondOH1.classList.remove('scene1-bond-oh-highlight');
                elements.bondOH1.classList.add('scene1-bond-oh');
                elements.bondOH2.classList.remove('scene1-bond-oh-highlight');
                elements.bondOH2.classList.add('scene1-bond-oh');
            }

            // 更新分子轮廓
            if (step.showMoleculeOutline) {
                show(elements.moleculeOutline);
            } else {
                hide(elements.moleculeOutline);
            }

            // 更新结论
            if (step.showConclusion) {
                show(elements.conclusion);
            } else {
                hide(elements.conclusion);
            }

            // 更新按钮状态
            if (step.prevDisabled) {
                elements.btnPrevBg.classList.remove('scene1-btn');
                elements.btnPrevBg.classList.add('scene1-btn-disabled');
                elements.btnPrevText.classList.add('scene1-btn-text-disabled');
            } else {
                elements.btnPrevBg.classList.remove('scene1-btn-disabled');
                elements.btnPrevBg.classList.add('scene1-btn');
                elements.btnPrevText.classList.remove('scene1-btn-text-disabled');
            }

            if (step.nextDisabled) {
                elements.btnNextBg.classList.remove('scene1-btn');
                elements.btnNextBg.classList.add('scene1-btn-disabled');
                elements.btnNextText.classList.add('scene1-btn-text-disabled');
                elements.btnNextText.textContent = '重播 ⟲';
            } else {
                elements.btnNextBg.classList.remove('scene1-btn-disabled');
                elements.btnNextBg.classList.add('scene1-btn');
                elements.btnNextText.classList.remove('scene1-btn-text-disabled');
                elements.btnNextText.textContent = '下一步 ▶';
            }

            currentStep = stepIndex;
        }

        // 按钮点击事件
        document.getElementById('scene1-btn-prev-group').addEventListener('click', function() {
            if (currentStep > 0) {
                updateStep(currentStep - 1);
            }
        });

        document.getElementById('scene1-btn-next-group').addEventListener('click', function() {
            if (currentStep < totalSteps - 1) {
                updateStep(currentStep + 1);
            } else {
                // 重播
                updateStep(0);
            }
        });

        // 初始化
        updateStep(0);

    })();
    </script>
