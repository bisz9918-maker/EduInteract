    <script>
    (function() {
        // Scene 3: F₂表型比例可视化
        // 布局坐标表参考:
        // | 元素 | 类型 | 位置/范围 | 大小 | 备注 |
        // |------|------|-----------|------|------|
        // | 状态切换按钮组 | 按钮容器 | (250,20)-(550,60) | 300×40 | 顶部居中 |
        // | 状态 0-饼图 | 圆形图表 | (400,230) 圆心 | r=100 | 三扇区 |
        // | 状态 1-♀柱状图 | 分组柱状图 | (120,150)-(280,350) | 160×200 | 左侧 |
        // | 状态 1-♂柱状图 | 分组柱状图 | (520,150)-(680,350) | 160×200 | 右侧 |
        // | 状态 2-对比柱状图 | 分组柱状图 | (100,120)-(700,360) | 600×240 | 理论 vs 实际 |
        // | 图例区 | 容器 | (50,390)-(750,430) | 700×40 | 底部 |

        let currentState = 0;

        const btnState0 = document.getElementById('scene3-btn-state0');
        const btnState1 = document.getElementById('scene3-btn-state1');
        const btnState2 = document.getElementById('scene3-btn-state2');

        const state0Group = document.getElementById('scene3-state0');
        const state1Group = document.getElementById('scene3-state1');
        const state2Group = document.getElementById('scene3-state2');

        function switchState(state) {
            currentState = state;

            // 更新按钮状态
            btnState0.classList.remove('active');
            btnState1.classList.remove('active');
            btnState2.classList.remove('active');

            // 隐藏所有状态
            state0Group.style.display = 'none';
            state1Group.style.display = 'none';
            state2Group.style.display = 'none';

            if (state === 0) {
                btnState0.classList.add('active');
                state0Group.style.display = 'block';
            } else if (state === 1) {
                btnState1.classList.add('active');
                state1Group.style.display = 'block';
            } else if (state === 2) {
                btnState2.classList.add('active');
                state2Group.style.display = 'block';
            }
        }

        // 绑定按钮事件
        btnState0.addEventListener('click', function() {
            switchState(0);
        });

        btnState1.addEventListener('click', function() {
            switchState(1);
        });

        btnState2.addEventListener('click', function() {
            switchState(2);
        });

        // 初始化显示状态 0
        switchState(0);
    })();
    </script>
