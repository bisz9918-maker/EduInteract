    <script>
    (function() {
        // Scene 2 实现：引物与模板链互补配对详解
        
        // 布局坐标表对照（来自 spec.json）：
        // 标题：(400,35) 居中
        // 左侧区域：(50,80)-(380,260) 330×180
        // 右侧区域：(420,80)-(750,260) 330×180
        // a 链左：(90,130)-(340,130)
        // b 链左：(90,180)-(340,180)
        // 引物Ⅰ：(90,215)-(160,215)
        // b 链右：(460,130)-(710,130)
        // a 链右：(460,180)-(710,180)
        // 引物Ⅱ：(460,215)-(530,215)
        // 高亮区 1：(90,120)-(140,140)
        // 高亮区 2：(460,120)-(510,140)
        // 总结区：(150,290)-(650,360)
        // 按钮：(280,455) 和 (420,455)
        
        let currentStep = 0;
        const totalSteps = 5;
        
        // 状态定义：每个步骤显示/隐藏的元素
        const states = [
            // 步骤 0：初始状态
            {
                show: [
                    'scene2-title', 'scene2-left-border', 'scene2-right-border',
                    'scene2-a-chain-left-line', 'scene2-a-chain-left-5prime', 'scene2-a-chain-left-3prime',
                    'scene2-a-chain-left-label', 'scene2-a-chain-left-g1', 'scene2-a-chain-left-g2',
                    'scene2-a-chain-left-dots', 'scene2-a-chain-left-t', 'scene2-a-chain-left-c',
                    'scene2-b-chain-left-line', 'scene2-b-chain-left-3prime', 'scene2-b-chain-left-5prime',
                    'scene2-b-chain-left-label', 'scene2-b-chain-left-c1', 'scene2-b-chain-left-c2',
                    'scene2-b-chain-left-dots', 'scene2-b-chain-left-a', 'scene2-b-chain-left-g',
                    'scene2-b-chain-right-line', 'scene2-b-chain-right-3prime', 'scene2-b-chain-right-5prime',
                    'scene2-b-chain-right-label', 'scene2-b-chain-right-c1', 'scene2-b-chain-right-c2',
                    'scene2-b-chain-right-dots', 'scene2-b-chain-right-a', 'scene2-b-chain-right-g',
                    'scene2-a-chain-right-line', 'scene2-a-chain-right-5prime', 'scene2-a-chain-right-3prime',
                    'scene2-a-chain-right-label', 'scene2-a-chain-right-g1', 'scene2-a-chain-right-g2',
                    'scene2-a-chain-right-dots', 'scene2-a-chain-right-t', 'scene2-a-chain-right-c',
                    'scene2-arrow1', 'scene2-arrow2'
                ],
                hide: [
                    'scene2-primer1-line', 'scene2-primer1-5prime', 'scene2-primer1-oh',
                    'scene2-primer1-g1', 'scene2-primer1-g2', 'scene2-primer1-label',
                    'scene2-primer2-line', 'scene2-primer2-5prime', 'scene2-primer2-oh',
                    'scene2-primer2-g1', 'scene2-primer2-g2', 'scene2-primer2-label',
                    'scene2-highlight1', 'scene2-highlight2',
                    'scene2-dash1', 'scene2-dash2', 'scene2-dash3', 'scene2-dash4',
                    'scene2-summary-bg', 'scene2-summary-text1', 'scene2-summary-text2'
                ]
            },
            // 步骤 1：高亮 a 链 3'端
            {
                show: ['scene2-highlight1'],
                hide: []
            },
            // 步骤 2：显示引物Ⅰ
            {
                show: [
                    'scene2-primer1-line', 'scene2-primer1-5prime', 'scene2-primer1-oh',
                    'scene2-primer1-g1', 'scene2-primer1-g2', 'scene2-primer1-label',
                    'scene2-dash1', 'scene2-dash2'
                ],
                hide: []
            },
            // 步骤 3：高亮 b 链 3'端
            {
                show: ['scene2-highlight2'],
                hide: []
            },
            // 步骤 4：显示引物Ⅱ和总结区
            {
                show: [
                    'scene2-primer2-line', 'scene2-primer2-5prime', 'scene2-primer2-oh',
                    'scene2-primer2-g1', 'scene2-primer2-g2', 'scene2-primer2-label',
                    'scene2-dash3', 'scene2-dash4',
                    'scene2-summary-bg', 'scene2-summary-text1', 'scene2-summary-text2'
                ],
                hide: []
            },
            // 步骤 5：最终状态（同步骤 4）
            {
                show: [],
                hide: []
            }
        ];
        
        // 工具函数：显示元素
        function showElement(id) {
            const el = document.getElementById(id);
            if (el) {
                el.classList.remove('hidden');
                el.classList.add('visible');
            }
        }
        
        // 工具函数：隐藏元素
        function hideElement(id) {
            const el = document.getElementById(id);
            if (el) {
                el.classList.remove('visible');
                el.classList.add('hidden');
            }
        }
        
        // 更新状态
        function updateState(step) {
            // 累积显示所有步骤的元素
            const allShow = new Set(states[0].show);
            for (let i = 1; i <= step; i++) {
                states[i].show.forEach(id => allShow.add(id));
            }
            
            // 显示应该显示的元素
            allShow.forEach(id => showElement(id));
            
            // 隐藏不应该显示的元素
            for (let i = step + 1; i < states.length; i++) {
                states[i].show.forEach(id => hideElement(id));
            }
            
            // 更新按钮状态
            const prevBtn = document.getElementById('scene2-btn-prev');
            const nextBtn = document.getElementById('scene2-btn-next');
            
            if (prevBtn) {
                prevBtn.disabled = (step === 0);
            }
            if (nextBtn) {
                nextBtn.disabled = (step >= totalSteps);
            }
        }
        
        // 事件绑定
        document.getElementById('scene2-btn-prev').addEventListener('click', function() {
            if (currentStep > 0) {
                currentStep--;
                updateState(currentStep);
            }
        });
        
        document.getElementById('scene2-btn-next').addEventListener('click', function() {
            if (currentStep < totalSteps) {
                currentStep++;
                updateState(currentStep);
            }
        });
        
        // 初始化
        updateState(0);
    })();
    </script>
