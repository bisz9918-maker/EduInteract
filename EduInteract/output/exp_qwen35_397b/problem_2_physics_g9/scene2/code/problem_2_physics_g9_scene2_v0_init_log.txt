    <script>
    (function() {
        // Scene 2: 木块受力分析 - 平衡力与作用力反作用力

        // 状态变量
        let currentState = 0;
        const totalSteps = 4;

        // 获取 DOM 元素
        const btnPrev = document.getElementById('scene2-btn-prev');
        const btnNext = document.getElementById('scene2-btn-next');
        const stepIndicator = document.getElementById('scene2-step-indicator');

        // 力箭头元素
        const arrowG = document.getElementById('scene2-arrow-g');
        const arrowN = document.getElementById('scene2-arrow-n');
        const arrowF = document.getElementById('scene2-arrow-f');

        // 高亮框元素
        const highlightBalance = document.getElementById('scene2-highlight-balance');
        const highlightActionReaction = document.getElementById('scene2-highlight-action-reaction');

        // 说明文字和表格
        const tableRow1 = document.getElementById('scene2-table-row1');
        const tableRow2 = document.getElementById('scene2-table-row2');
        const comparisonTable = document.getElementById('scene2-table');

        // 辅助函数：显示元素
        function show(el) {
            el.style.display = 'block';
            el.style.opacity = '1';
            if (el.tagName === 'rect') {
                el.style.display = '';
            }
            if (el.tagName === 'TABLE') {
                el.classList.add('visible');
            }
        }

        // 辅助函数：隐藏元素
        function hide(el) {
            el.style.display = 'none';
            el.style.opacity = '0';
            if (el.tagName === 'rect') {
                el.style.display = 'none';
            }
            if (el.tagName === 'TABLE') {
                el.classList.remove('visible');
            }
        }

        // 设置箭头线宽
        function setArrowWidth(arrow, width) {
            arrow.setAttribute('stroke-width', width);
        }

        // 渲染当前步骤
        function renderStep() {
            // 重置所有高亮
            hide(highlightBalance);
            hide(highlightActionReaction);
            hide(tableRow1);
            hide(tableRow2);
            hide(comparisonTable);

            // 重置箭头线宽
            setArrowWidth(arrowG, 2);
            setArrowWidth(arrowN, 2);
            setArrowWidth(arrowF, 2);

            // 更新按钮状态
            btnPrev.disabled = (currentState === 0);
            if (currentState === totalSteps - 1) {
                btnNext.textContent = '完成';
            } else {
                btnNext.textContent = '下一步 ▶';
            }

            // 更新步骤指示器
            stepIndicator.textContent = `步骤 ${currentState + 1}/${totalSteps}`;

            // 根据当前状态显示相应元素
            switch (currentState) {
                case 0:
                    // 初始状态：只显示基础图形
                    break;

                case 1:
                    // 平衡力高亮
                    show(highlightBalance);
                    setArrowWidth(arrowG, 3);
                    setArrowWidth(arrowN, 3);
                    show(tableRow1);
                    break;

                case 2:
                    // 作用力反作用力高亮
                    show(highlightActionReaction);
                    setArrowWidth(arrowN, 3);
                    setArrowWidth(arrowF, 3);
                    show(tableRow2);
                    break;

                case 3:
                    // 显示对比表格
                    show(comparisonTable);
                    break;
            }
        }

        // 按钮事件绑定
        btnPrev.addEventListener('click', function() {
            if (currentState > 0) {
                currentState--;
                renderStep();
            }
        });

        btnNext.addEventListener('click', function() {
            if (currentState < totalSteps - 1) {
                currentState++;
                renderStep();
            }
        });

        // 初始化渲染
        renderStep();
    })();
    </script>
