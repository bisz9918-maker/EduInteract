    <script>
    (function() {
        // Scene 3 - 氨气溶解与喷泉效应
        
        let currentStep = 0;
        const totalSteps = 4;
        
        // 元素引用
        const elements = {
            stepIndicator: document.getElementById('scene3-step-indicator'),
            bottleB: document.getElementById('scene3-bottle-b'),
            waterDrop: document.getElementById('scene3-water-drop'),
            pressureArrow: document.getElementById('scene3-pressure-arrow'),
            flowArrow: document.getElementById('scene3-flow-arrow'),
            tubeFlow: document.getElementById('scene3-tube-flow'),
            fountain: document.getElementById('scene3-fountain'),
            labelSqueeze: document.getElementById('scene3-label-squeeze'),
            labelDissolve: document.getElementById('scene3-label-dissolve'),
            labelPressure: document.getElementById('scene3-label-pressure'),
            labelOpen: document.getElementById('scene3-label-open'),
            labelFountain: document.getElementById('scene3-label-fountain'),
            labelAlkaline: document.getElementById('scene3-label-alkaline'),
            pistonA: document.getElementById('scene3-piston-a'),
            waterLevel: document.getElementById('scene3-water-level'),
            btnPrevText: document.getElementById('scene3-btn-prev-text'),
            btnNextText: document.getElementById('scene3-btn-next-text')
        };
        
        // 显示/隐藏工具函数
        function show(el) {
            if (el) {
                el.classList.remove('scene3-hidden');
                el.style.display = '';
                el.style.opacity = '1';
            }
        }
        
        function hide(el) {
            if (el) {
                el.classList.add('scene3-hidden');
                el.style.display = 'none';
                el.style.opacity = '0';
            }
        }
        
        // 更新场景状态
        function updateScene() {
            // 更新步骤指示器
            elements.stepIndicator.textContent = `步骤 ${currentStep + 1}/${totalSteps}`;
            
            // 步骤 0：初始状态
            if (currentStep === 0) {
                hide(elements.waterDrop);
                hide(elements.pressureArrow);
                hide(elements.flowArrow);
                hide(elements.tubeFlow);
                hide(elements.fountain);
                hide(elements.labelSqueeze);
                hide(elements.labelDissolve);
                hide(elements.labelPressure);
                hide(elements.labelOpen);
                hide(elements.labelFountain);
                hide(elements.labelAlkaline);
                elements.bottleB.setAttribute('fill-opacity', '0.2');
                elements.bottleB.setAttribute('fill', '#2e7d32');
                elements.pistonA.setAttribute('fill', '#ff9800');
                elements.waterLevel.setAttribute('y', '450');
                elements.waterLevel.setAttribute('height', '40');
            }
            // 步骤 1：挤入水
            else if (currentStep === 1) {
                show(elements.waterDrop);
                hide(elements.pressureArrow);
                hide(elements.flowArrow);
                hide(elements.tubeFlow);
                hide(elements.fountain);
                show(elements.labelSqueeze);
                hide(elements.labelDissolve);
                hide(elements.labelPressure);
                hide(elements.labelOpen);
                hide(elements.labelFountain);
                hide(elements.labelAlkaline);
            }
            // 步骤 2：氨气溶解，压强减小
            else if (currentStep === 2) {
                hide(elements.waterDrop);
                show(elements.pressureArrow);
                hide(elements.flowArrow);
                hide(elements.tubeFlow);
                hide(elements.fountain);
                hide(elements.labelSqueeze);
                show(elements.labelDissolve);
                show(elements.labelPressure);
                hide(elements.labelOpen);
                hide(elements.labelFountain);
                hide(elements.labelAlkaline);
                elements.bottleB.setAttribute('fill-opacity', '0.1');
            }
            // 步骤 3：打开活塞，水被吸入
            else if (currentStep === 3) {
                hide(elements.waterDrop);
                hide(elements.pressureArrow);
                show(elements.flowArrow);
                show(elements.tubeFlow);
                hide(elements.fountain);
                hide(elements.labelSqueeze);
                show(elements.labelDissolve);
                show(elements.labelPressure);
                show(elements.labelOpen);
                hide(elements.labelFountain);
                hide(elements.labelAlkaline);
                elements.pistonA.setAttribute('fill', '#4caf50');
                elements.waterLevel.setAttribute('y', '460');
                elements.waterLevel.setAttribute('height', '30');
            }
            // 步骤 4：喷泉形成，酚酞变红
            else if (currentStep === 4) {
                hide(elements.waterDrop);
                hide(elements.pressureArrow);
                show(elements.flowArrow);
                show(elements.tubeFlow);
                show(elements.fountain);
                hide(elements.labelSqueeze);
                show(elements.labelDissolve);
                show(elements.labelPressure);
                show(elements.labelOpen);
                show(elements.labelFountain);
                show(elements.labelAlkaline);
                elements.pistonA.setAttribute('fill', '#4caf50');
                elements.bottleB.setAttribute('fill', '#c62828');
                elements.bottleB.setAttribute('fill-opacity', '0.4');
                elements.waterLevel.setAttribute('y', '475');
                elements.waterLevel.setAttribute('height', '15');
            }
            
            // 更新按钮状态
            elements.btnPrevText.textContent = currentStep > 0 ? '◀ 上一步' : '◀ 上一步';
            elements.btnPrevText.style.opacity = currentStep > 0 ? '1' : '0.5';
            elements.btnNextText.textContent = currentStep < totalSteps ? '下一步 ▶' : '完成';
            elements.btnNextText.style.opacity = currentStep < totalSteps ? '1' : '0.5';
        }
        
        // 点击区域检测
        function checkClick(x, y) {
            const svg = document.getElementById('scene3-svg');
            const rect = svg.getBoundingClientRect();
            const svgX = x - rect.left;
            const svgY = y - rect.top;
            
            // 上一步按钮区域 (80, 455) - (180, 490)
            if (svgX >= 80 && svgX <= 180 && svgY >= 455 && svgY <= 490) {
                if (currentStep > 0) {
                    currentStep--;
                    updateScene();
                }
                return;
            }
            
            // 下一步按钮区域 (620, 455) - (720, 490)
            if (svgX >= 620 && svgX <= 720 && svgY >= 455 && svgY <= 490) {
                if (currentStep < totalSteps) {
                    currentStep++;
                    updateScene();
                }
                return;
            }
        }
        
        // 事件监听
        document.getElementById('scene3-svg').addEventListener('click', function(e) {
            checkClick(e.clientX, e.clientY);
        });
        
        // 初始化
        updateScene();
    })();
    </script>
