    <script>
    (function() {
        // Scene 1 实现
        let currentStep = 0;
        const totalSteps = 4;

        const prevBtn = document.getElementById('scene1-prev-btn');
        const nextBtn = document.getElementById('scene1-next-btn');
        const stepIndicator = document.getElementById('scene1-step-indicator');

        // 元素引用
        const elements = {
            // 步骤1显示的元素
            step1: [
                document.getElementById('scene1-center-a'),
                document.getElementById('scene1-center-b'),
                document.getElementById('scene1-label-A'),
                document.getElementById('scene1-label-B'),
                document.getElementById('scene1-label-a')
            ],
            // 步骤2显示的元素
            step2: [
                document.getElementById('scene1-overlap1'),
                document.getElementById('scene1-overlap2')
            ],
            // 步骤3显示的元素
            step3: [
                document.getElementById('scene1-label-x1'),
                document.getElementById('scene1-label-x2')
            ]
        };

        // 显示/隐藏元素工具函数
        function showElement(el) {
            if (el) {
                el.style.opacity = '1';
                el.style.display = '';
            }
        }

        function hideElement(el) {
            if (el) {
                el.style.opacity = '0';
                el.style.display = 'none';
            }
        }

        // 更新视图
        function updateView() {
            // 隐藏所有步骤元素
            elements.step1.forEach(el => hideElement(el));
            elements.step2.forEach(el => hideElement(el));
            elements.step3.forEach(el => hideElement(el));

            // 根据当前步骤显示相应元素
            if (currentStep >= 1) {
                elements.step1.forEach(el => showElement(el));
            }
            if (currentStep >= 2) {
                elements.step2.forEach(el => showElement(el));
            }
            if (currentStep >= 3) {
                elements.step3.forEach(el => showElement(el));
            }

            // 更新步骤指示器
            stepIndicator.textContent = '步骤 ' + currentStep;

            // 更新按钮状态
            prevBtn.disabled = (currentStep === 0);
            nextBtn.disabled = (currentStep === totalSteps - 1);
        }

        // 按钮事件
        prevBtn.addEventListener('click', function() {
            if (currentStep > 0) {
                currentStep--;
                updateView();
            }
        });

        nextBtn.addEventListener('click', function() {
            if (currentStep < totalSteps - 1) {
                currentStep++;
                updateView();
            }
        });

        // 初始化视图
        updateView();
    })();
    </script>
