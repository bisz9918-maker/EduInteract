    <script>
    (function() {
        // Scene 2: 容斥原理可视化
        // 布局坐标表参考:
        // | 元素 | 类型 | 位置/范围 | 大小 | 备注 |
        // |------|------|-----------|------|------|
        // | 圆 A | 圆 | 圆心(200,260) | r=65 | 左侧圆 |
        // | 圆 B | 圆 | 圆心(340,260) | r=65 | 中间圆 |
        // | 圆 C | 圆 | 圆心(480,310) | r=65 | 右下圆 |
        // | 直线 a | 线段 | (80,260)-(600,260) | 520×2 | 水平穿过圆心 A、B |
        // | 重叠区 AB | 路径 | 圆 A 与圆 B 交集 | ~40×50 | 填充#c62828 30% |
        // | 重叠区 BC | 路径 | 圆 B 与圆 C 交集 | ~40×50 | 填充#c62828 30% |
        // | 标签 A | text | (200,245) | ~12×16 | 圆心正上方 |
        // | 标签 B | text | (340,245) | ~12×16 | 圆心正上方 |
        // | 标签 C | text | (480,295) | ~12×16 | 圆心正上方 |
        // | 标签 a | text | (615,265) | ~12×16 | 直线右端右侧 |
        // | 面积 S 标注 A | text+框 | (200,170) | ~30×20 | 圆 A 正上方 |
        // | 面积 S 标注 B | text+框 | (340,170) | ~30×20 | 圆 B 正上方 |
        // | 面积 S 标注 C | text+框 | (480,390) | ~30×20 | 圆 C 正下方 |
        // | 重叠 x 标注 AB | text | (270,235) | ~20×16 | 重叠区 AB 上方 |
        // | 重叠 x 标注 BC | text | (420,290) | ~20×16 | 重叠区 BC 右侧 |
        // | 表达式框 | 矩形 | (620,120)-(780,200) | 160×80 | 绿色边框 |

        // 状态定义
        const STATE_COUNT = 4;
        let currentState = 0;

        // 获取 DOM 元素
        const btnPrevBg = document.getElementById('scene2-btn-prev-bg');
        const btnPrevText = document.getElementById('scene2-btn-prev-text');
        const btnNextBg = document.getElementById('scene2-btn-next-bg');
        const btnNextText = document.getElementById('scene2-btn-next-text');
        const stepIndicator = document.getElementById('scene2-step-indicator');

        // 需要控制的元素
        const overlapAB = document.getElementById('scene2-overlap-ab');
        const overlapBC = document.getElementById('scene2-overlap-bc');
        const overlapXAB = document.getElementById('scene2-overlap-x-ab');
        const overlapXBC = document.getElementById('scene2-overlap-x-bc');
        const arrow1 = document.getElementById('scene2-arrow-1');
        const arrow2 = document.getElementById('scene2-arrow-2');
        const exprGroup = document.getElementById('scene2-expr-group');
        const exprText = document.getElementById('scene2-expr-text');
        const exprNote = document.getElementById('scene2-expr-note');

        // 工具函数：显示元素
        function show(el) {
            if (el) {
                el.classList.remove('scene2-hidden');
                el.classList.add('scene2-visible');
            }
        }

        // 工具函数：隐藏元素
        function hide(el) {
            if (el) {
                el.classList.remove('scene2-visible');
                el.classList.add('scene2-hidden');
            }
        }

        // 更新按钮状态
        function updateButtons() {
            if (currentState === 0) {
                btnPrevBg.style.fill = '#cccccc';
                btnPrevText.style.fill = '#999999';
            } else {
                btnPrevBg.style.fill = '#1565c0';
                btnPrevText.style.fill = '#ffffff';
            }

            if (currentState === STATE_COUNT - 1) {
                btnNextBg.style.fill = '#cccccc';
                btnNextText.style.fill = '#999999';
            } else {
                btnNextBg.style.fill = '#1565c0';
                btnNextText.style.fill = '#ffffff';
            }

            stepIndicator.textContent = '步骤 ' + (currentState + 1) + '/' + STATE_COUNT;
        }

        // 更新场景状态
        function updateScene() {
            // 重置所有动态元素
            hide(overlapAB);
            hide(overlapBC);
            hide(overlapXAB);
            hide(overlapXBC);
            hide(arrow1);
            hide(arrow2);
            hide(exprGroup);

            switch (currentState) {
                case 0:
                    // 初始状态：只显示三个圆、直线、标签、面积 S 标注
                    break;

                case 1:
                    // 步骤 1：显示重叠区域和 x 标注
                    show(overlapAB);
                    show(overlapBC);
                    show(overlapXAB);
                    show(overlapXBC);
                    break;

                case 2:
                    // 步骤 2：显示箭头和表达式 3S - 2x
                    show(overlapAB);
                    show(overlapBC);
                    show(overlapXAB);
                    show(overlapXBC);
                    show(arrow1);
                    show(arrow2);
                    show(exprGroup);
                    exprText.textContent = '3S - 2x';
                    exprNote.textContent = '容斥原理';
                    break;

                case 3:
                    // 步骤 3：显示最终表达式 2S + 2
                    show(overlapAB);
                    show(overlapBC);
                    show(overlapXAB);
                    show(overlapXBC);
                    show(arrow1);
                    show(arrow2);
                    show(exprGroup);
                    exprText.textContent = '2S + 2';
                    exprNote.textContent = '题目给定覆盖面积';
                    break;
            }

            updateButtons();
        }

        // 按钮点击处理
        btnPrevBg.addEventListener('click', function() {
            if (currentState > 0) {
                currentState--;
                updateScene();
            }
        });

        btnNextBg.addEventListener('click', function() {
            if (currentState < STATE_COUNT - 1) {
                currentState++;
                updateScene();
            }
        });

        // 添加按钮悬停效果
        btnPrevBg.addEventListener('mouseenter', function() {
            if (currentState > 0) {
                btnPrevBg.style.fill = '#0d47a1';
            }
        });
        btnPrevBg.addEventListener('mouseleave', function() {
            if (currentState > 0) {
                btnPrevBg.style.fill = '#1565c0';
            }
        });

        btnNextBg.addEventListener('mouseenter', function() {
            if (currentState < STATE_COUNT - 1) {
                btnNextBg.style.fill = '#0d47a1';
            }
        });
        btnNextBg.addEventListener('mouseleave', function() {
            if (currentState < STATE_COUNT - 1) {
                btnNextBg.style.fill = '#1565c0';
            }
        });

        // 初始化场景
        updateScene();

    })();
    </script>
