    <script>
    (function() {
        // Scene 2 实现：从前面观察形状对比
        
        // 状态变量
        let currentStep = 0;
        const totalSteps = 3;
        
        // 元素引用
        const leftView = document.getElementById('scene2-left-view');
        const rightView = document.getElementById('scene2-right-view');
        const conclusion = document.getElementById('scene2-conclusion');
        const btnPrevText = document.getElementById('scene2-btn-prev-text');
        const btnNextText = document.getElementById('scene2-btn-next-text');
        const controlsGroup = document.getElementById('scene2-controls');
        
        // 显示/隐藏工具函数
        function show(el) {
            el.style.display = '';
            el.style.opacity = '1';
        }
        
        function hide(el) {
            el.style.display = 'none';
            el.style.opacity = '0';
        }
        
        // 渲染函数
        function render() {
            // 步骤 0：只显示左图
            // 步骤 1：显示左图 + 右图
            // 步骤 2：显示左图 + 右图 + 结论
            
            if (currentStep >= 0) {
                show(leftView);
            }
            
            if (currentStep >= 1) {
                show(rightView);
            } else {
                hide(rightView);
            }
            
            if (currentStep >= 2) {
                show(conclusion);
            } else {
                hide(conclusion);
            }
            
            // 更新按钮状态
            if (currentStep === 0) {
                btnPrevText.textContent = '◀ 上一步';
                btnPrevText.parentElement.style.fill = '#f5f5f5';
                btnPrevText.parentElement.style.cursor = 'not-allowed';
            } else {
                btnPrevText.textContent = '◀ 上一步';
                btnPrevText.parentElement.style.fill = '#e0e0e0';
                btnPrevText.parentElement.style.cursor = 'pointer';
            }
            
            if (currentStep >= totalSteps - 1) {
                btnNextText.textContent = '下一步 ▶';
                btnNextText.parentElement.style.fill = '#f5f5f5';
                btnNextText.parentElement.style.cursor = 'not-allowed';
            } else {
                btnNextText.textContent = '下一步 ▶';
                btnNextText.parentElement.style.fill = '#e0e0e0';
                btnNextText.parentElement.style.cursor = 'pointer';
            }
        }
        
        // 点击事件处理
        controlsGroup.addEventListener('click', function(e) {
            const target = e.target;
            
            // 检查是否点击了上一步按钮区域
            if (target.tagName === 'rect' && target.getAttribute('x') === '280') {
                if (currentStep > 0) {
                    currentStep--;
                    render();
                }
            } else if (target.tagName === 'text' && target.id === 'scene2-btn-prev-text') {
                if (currentStep > 0) {
                    currentStep--;
                    render();
                }
            }
            
            // 检查是否点击了下一步按钮区域
            if (target.tagName === 'rect' && target.getAttribute('x') === '440') {
                if (currentStep < totalSteps - 1) {
                    currentStep++;
                    render();
                }
            } else if (target.tagName === 'text' && target.id === 'scene2-btn-next-text') {
                if (currentStep < totalSteps - 1) {
                    currentStep++;
                    render();
                }
            }
        });
        
        // 初始化
        render();
    })();
    </script>
