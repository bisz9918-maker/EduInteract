    <script>
    (function() {
        // Scene 1 实现
        let currentState = 0;
        const totalSteps = 4;
        
        // 获取 DOM 元素
        const highlightTop = document.getElementById('scene1-highlight-top');
        const highlightFront = document.getElementById('scene1-highlight-front');
        const highlightRight = document.getElementById('scene1-highlight-right');
        const arrow1 = document.getElementById('scene1-arrow1');
        const arrow2 = document.getElementById('scene1-arrow2');
        const arrow3 = document.getElementById('scene1-arrow3');
        const relationLabel = document.getElementById('scene1-relation-label');
        const prevBtn = document.getElementById('scene1-prev-btn');
        const nextBtn = document.getElementById('scene1-next-btn');
        const stepIndicator = document.getElementById('scene1-step-indicator');
        
        // 更新视图函数
        function updateView() {
            // 隐藏所有高亮框和箭头
            highlightTop.style.display = 'none';
            highlightFront.style.display = 'none';
            highlightRight.style.display = 'none';
            arrow1.style.display = 'none';
            arrow2.style.display = 'none';
            arrow3.style.display = 'none';
            relationLabel.style.display = 'none';
            
            // 根据当前状态显示对应元素
            switch (currentState) {
                case 0:
                    // 初始状态：只显示正方体和图案、标签
                    break;
                case 1:
                    // 步骤 1：高亮顶面
                    highlightTop.style.display = 'block';
                    break;
                case 2:
                    // 步骤 2：高亮正面
                    highlightFront.style.display = 'block';
                    break;
                case 3:
                    // 步骤 3：高亮右侧面
                    highlightRight.style.display = 'block';
                    break;
                case 4:
                    // 步骤 4：显示箭头和相邻关系标签
                    arrow1.style.display = 'block';
                    arrow2.style.display = 'block';
                    arrow3.style.display = 'block';
                    relationLabel.style.display = 'block';
                    break;
            }
            
            // 更新按钮状态
            prevBtn.disabled = (currentState === 0);
            nextBtn.disabled = (currentState === totalSteps);
            
            // 更新步骤指示
            stepIndicator.textContent = `步骤 ${currentState}/${totalSteps}`;
        }
        
        // 按钮事件监听
        prevBtn.addEventListener('click', function() {
            if (currentState > 0) {
                currentState--;
                updateView();
            }
        });
        
        nextBtn.addEventListener('click', function() {
            if (currentState < totalSteps) {
                currentState++;
                updateView();
            }
        });
        
        // 初始化视图
        updateView();
    })();
    </script>
