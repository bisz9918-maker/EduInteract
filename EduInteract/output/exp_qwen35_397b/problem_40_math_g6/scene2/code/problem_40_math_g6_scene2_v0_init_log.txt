    <script>
    (function() {
        // Scene 2 实现 - 正方体展开图与对面概念
        
        let currentState = 0;
        const totalSteps = 4;
        
        const btnPrev = document.getElementById('scene2-btn-prev');
        const btnNext = document.getElementById('scene2-btn-next');
        const stepIndicator = document.getElementById('scene2-step-indicator');
        
        // 元素引用
        const elements = {
            arrow: document.getElementById('scene2-arrow'),
            // 展开图各面
            sunTop: document.getElementById('scene2-net-sun-top'),
            sunBottom: document.getElementById('scene2-net-sun-bottom'),
            heartLeft: document.getElementById('scene2-net-heart-left'),
            heartRight: document.getElementById('scene2-net-heart-right'),
            smileCenter: document.getElementById('scene2-net-smile-center'),
            smileRight: document.getElementById('scene2-net-smile-right'),
            // 虚线
            dashSun: document.getElementById('scene2-dash-sun'),
            dashHeart: document.getElementById('scene2-dash-heart'),
            dashSmile: document.getElementById('scene2-dash-smile'),
            // 标签
            labelSun: document.getElementById('scene2-label-sun'),
            labelHeart: document.getElementById('scene2-label-heart'),
            labelSmile: document.getElementById('scene2-label-smile'),
            labelOpposite: document.getElementById('scene2-label-opposite'),
            // 路径
            pathTop: document.getElementById('scene2-path-top'),
            pathBottom: document.getElementById('scene2-path-bottom'),
            pathLeft: document.getElementById('scene2-path-left'),
            pathRight: document.getElementById('scene2-path-right'),
            pathSmileRight: document.getElementById('scene2-path-smile-right')
        };
        
        function updateStep() {
            // 更新步骤指示器
            stepIndicator.textContent = `步骤 ${currentState}/${totalSteps}`;
            
            // 更新按钮状态
            btnPrev.disabled = currentState === 0;
            btnNext.disabled = currentState === totalSteps;
            
            // 根据当前步骤显示/隐藏元素
            resetAll();
            
            if (currentState >= 1) {
                // 步骤 1: 顶面太阳和底面太阳展开
                showElement(elements.sunTop, 490, 100);
                showElement(elements.sunBottom, 490, 260);
                elements.dashSun.style.opacity = '1';
                elements.labelSun.style.opacity = '1';
                elements.pathTop.style.opacity = '1';
                elements.pathBottom.style.opacity = '1';
            }
            
            if (currentState >= 2) {
                // 步骤 2: 左右爱心面展开
                showElement(elements.heartLeft, 370, 180);
                showElement(elements.heartRight, 530, 180);
                elements.dashHeart.style.opacity = '1';
                elements.labelHeart.style.opacity = '1';
                elements.pathLeft.style.opacity = '1';
                elements.pathRight.style.opacity = '1';
            }
            
            if (currentState >= 3) {
                // 步骤 3: 前后笑脸面展开
                showElement(elements.smileCenter, 450, 180);
                showElement(elements.smileRight, 610, 180);
                elements.dashSmile.style.opacity = '1';
                elements.labelSmile.style.opacity = '1';
                elements.pathSmileRight.style.opacity = '1';
            }
            
            if (currentState >= 4) {
                // 步骤 4: 所有路径高亮，对面标签高亮
                elements.labelOpposite.style.opacity = '1';
            }
        }
        
        function resetAll() {
            // 隐藏所有展开图元素
            [elements.sunTop, elements.sunBottom, elements.heartLeft, elements.heartRight, 
             elements.smileCenter, elements.smileRight].forEach(el => {
                el.style.opacity = '0';
            });
            
            // 隐藏所有虚线和标签
            [elements.dashSun, elements.dashHeart, elements.dashSmile].forEach(el => {
                el.style.opacity = '0';
            });
            [elements.labelSun, elements.labelHeart, elements.labelSmile, elements.labelOpposite].forEach(el => {
                el.style.opacity = '0';
            });
            
            // 隐藏所有路径
            [elements.pathTop, elements.pathBottom, elements.pathLeft, elements.pathRight, elements.pathSmileRight].forEach(el => {
                el.style.opacity = '0';
            });
        }
        
        function showElement(element, x, y) {
            element.style.opacity = '1';
            element.setAttribute('transform', `translate(${x - 80}, ${y - 180})`);
        }
        
        // 事件监听
        btnPrev.addEventListener('click', () => {
            if (currentState > 0) {
                currentState--;
                updateStep();
            }
        });
        
        btnNext.addEventListener('click', () => {
            if (currentState < totalSteps) {
                currentState++;
                updateStep();
            }
        });
        
        // 初始化
        updateStep();
    })();
    </script>
