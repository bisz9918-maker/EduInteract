    <script>
    (function() {
        // Scene 2 实现：四种器官情况下血液性质变化对比

        // 状态数据
        const scene2States = {
            0: { // 心脏
                organName: '心脏',
                bloodA: 'venous',    // 静脉血
                bloodC: 'arterial',  // 动脉血
                showO2: false,
                showCO2: false,
                icon: 'heart'
            },
            1: { // 肺
                organName: '肺',
                bloodA: 'venous',    // 肺动脉，静脉血
                bloodC: 'arterial',  // 肺静脉，动脉血
                showO2: true,
                showCO2: true,
                icon: 'lung'
            },
            2: { // 小肠
                organName: '小肠',
                bloodA: 'arterial',  // 肠动脉，动脉血
                bloodC: 'venous',    // 肠静脉，静脉血
                showO2: true,
                showCO2: true,
                icon: 'intestine'
            },
            3: { // 组织细胞
                organName: '组织细胞',
                bloodA: 'arterial',  // 体动脉，动脉血
                bloodC: 'venous',    // 体静脉，静脉血
                showO2: true,
                showCO2: true,
                icon: 'tissue'
            }
        };

        let scene2CurrentState = 0;

        // 工具函数：显示/隐藏元素
        function scene2Show(el) {
            el.style.display = '';
            el.style.opacity = '1';
        }
        function scene2Hide(el) {
            el.style.display = 'none';
            el.style.opacity = '0';
        }

        // 更新状态
        function scene2UpdateState(newState) {
            scene2CurrentState = newState;
            const state = scene2States[newState];

            // 更新器官名称
            document.getElementById('scene2-label-organ').textContent = state.organName;

            // 更新血管 a 血液颜色和标签
            const bloodA = document.getElementById('scene2-blood-a');
            const labelA = document.getElementById('scene2-blood-label-a');
            if (state.bloodA === 'arterial') {
                bloodA.setAttribute('fill', '#c62828');
                labelA.textContent = '动脉血';
                labelA.classList.remove('scene2-blood-venous');
                labelA.classList.add('scene2-blood-arterial');
            } else {
                bloodA.setAttribute('fill', '#1565c0');
                labelA.textContent = '静脉血';
                labelA.classList.remove('scene2-blood-arterial');
                labelA.classList.add('scene2-blood-venous');
            }

            // 更新血管 c 血液颜色和标签
            const bloodC = document.getElementById('scene2-blood-c');
            const labelC = document.getElementById('scene2-blood-label-c');
            if (state.bloodC === 'arterial') {
                bloodC.setAttribute('fill', '#c62828');
                labelC.textContent = '动脉血';
                labelC.classList.remove('scene2-blood-venous');
                labelC.classList.add('scene2-blood-arterial');
            } else {
                bloodC.setAttribute('fill', '#1565c0');
                labelC.textContent = '静脉血';
                labelC.classList.remove('scene2-blood-arterial');
                labelC.classList.add('scene2-blood-venous');
            }

            // 更新气体交换箭头可见性
            const arrowO2 = document.getElementById('scene2-arrow-o2');
            const labelO2 = document.getElementById('scene2-label-o2');
            const arrowCO2 = document.getElementById('scene2-arrow-co2');
            const labelCO2 = document.getElementById('scene2-label-co2');

            if (state.showO2) {
                scene2Show(arrowO2);
                scene2Show(labelO2);
            } else {
                scene2Hide(arrowO2);
                scene2Hide(labelO2);
            }

            if (state.showCO2) {
                scene2Show(arrowCO2);
                scene2Show(labelCO2);
            } else {
                scene2Hide(arrowCO2);
                scene2Hide(labelCO2);
            }

            // 更新器官图标
            document.querySelectorAll('.scene2-organ-icon').forEach(icon => {
                icon.style.opacity = '0';
            });
            const activeIcon = document.getElementById('scene2-icon-' + state.icon);
            if (activeIcon) {
                activeIcon.style.opacity = '1';
            }

            // 更新按钮激活状态
            document.querySelectorAll('.scene2-btn').forEach(btn => {
                btn.classList.remove('active');
            });
            const activeBtn = document.getElementById('scene2-btn-' + state.icon);
            if (activeBtn) {
                activeBtn.classList.add('active');
            }
        }

        // 绑定按钮事件
        document.getElementById('scene2-btn-heart').addEventListener('click', function() {
            scene2UpdateState(0);
        });
        document.getElementById('scene2-btn-lung').addEventListener('click', function() {
            scene2UpdateState(1);
        });
        document.getElementById('scene2-btn-intestine').addEventListener('click', function() {
            scene2UpdateState(2);
        });
        document.getElementById('scene2-btn-tissue').addEventListener('click', function() {
            scene2UpdateState(3);
        });

        // 初始化
        scene2UpdateState(0);
    })();
    </script>
