    <script>
    (function() {
        // Scene 2: 晨昏线切点与太阳直射点纬度的互余关系
        // 布局坐标表（实现对照依据）：
        // | 元素 | 类型 | 位置/范围 | 大小 | 标签偏移方向 | 备注 |
        // |------|------|-----------|------|-------------|------|
        // | 俯视图外圆 | 圆弧 | 圆心 (180,320) | 半径 140px | - | 南极俯视图边界 |
        // | 70°纬线圈 | 圆 | 圆心 (180,320) | 半径 48px | - | 70°纬线 |
        // | 切点 M | 实心圆 | (180,272) | 直径 8px | 右上 | 晨昏线与 70°纬线切点 |
        // | 极点 O | 点 | (180,320) | 直径 4px | 下方 | 南极点 |
        // | 极夜区域 | 扇形填充 | 70°纬线以内 | 半径 48px | - | 透明度 20% |
        // | 辅助线 OM | 虚线 | (180,320)-(180,272) | 48px 长 | - | 地心到切点半径 |
        // | 晨昏线 MA | 线段 | (180,272)-(300,200) | - | - | 切线方向 |
        // | 角度弧 | 圆弧 | (180,320) 附近 | 半径 25px | - | 标注 20°角 |
        // | 角度值 20° | text | (195,305) | ~24×16 | - | 绿色加粗 |
        // | 90°-70°= | text | (150,290) | ~45×16 | - | 灰色小字 |
        // | 推导箭头 | 箭头 | (340,320)-(380,320) | 40px 长 | - | 从左到右 |
        // | 侧视图地球 | 圆 | 圆心 (520,320) | 半径 70px | - | 地球侧视 |
        // | 赤道线 | 线段 | (450,320)-(590,320) | 140px | - | 水平穿过地心 |
        // | 直射点 | 实心圆 | (520,265) | 直径 6px | 右侧 | 20°N 位置 |
        // | 太阳光线 | 箭头 | (620,265)-(540,265) | - | 上方 | 平行光线 |
        // | 20°N 标签 | text | (535,260) | ~35×16 | 右侧 | 绿色加粗 |
        // | 70°标签 | text | (180,265) | ~20×16 | 上方 | 70°纬线标注 |
        // | O 标签 | text | (180,335) | ~12×16 | 下方 | 极点标识 |
        // | M 标签 | text | (195,265) | ~12×16 | 右上 | 切点标识（修正后） |

        let currentStep = 0;
        const totalSteps = 6;

        // 每步显示的元素 ID 列表
        const steps = [
            // 步骤 0：基础图形
            ['scene2-outer-circle', 'scene2-lat70-circle', 'scene2-point-o', 'scene2-point-m', 'scene2-lat70-label', 'scene2-label-o', 'scene2-label-m'],
            // 步骤 1：极夜区域
            ['scene2-outer-circle', 'scene2-lat70-circle', 'scene2-point-o', 'scene2-point-m', 'scene2-lat70-label', 'scene2-label-o', 'scene2-label-m', 'scene2-polar-night'],
            // 步骤 2：辅助线 OM + 晨昏线 MA
            ['scene2-outer-circle', 'scene2-lat70-circle', 'scene2-point-o', 'scene2-point-m', 'scene2-lat70-label', 'scene2-label-o', 'scene2-label-m', 'scene2-polar-night', 'scene2-line-om', 'scene2-line-ma'],
            // 步骤 3：角度弧 + 计算文字
            ['scene2-outer-circle', 'scene2-lat70-circle', 'scene2-point-o', 'scene2-point-m', 'scene2-lat70-label', 'scene2-label-o', 'scene2-label-m', 'scene2-polar-night', 'scene2-line-om', 'scene2-line-ma', 'scene2-angle-arc', 'scene2-calc-text'],
            // 步骤 4：角度值 20° + 推导箭头
            ['scene2-outer-circle', 'scene2-lat70-circle', 'scene2-point-o', 'scene2-point-m', 'scene2-lat70-label', 'scene2-label-o', 'scene2-label-m', 'scene2-polar-night', 'scene2-line-om', 'scene2-line-ma', 'scene2-angle-arc', 'scene2-calc-text', 'scene2-angle-value', 'scene2-derive-arrow'],
            // 步骤 5：侧视图 + 直射点 + 太阳光线
            ['scene2-outer-circle', 'scene2-lat70-circle', 'scene2-point-o', 'scene2-point-m', 'scene2-lat70-label', 'scene2-label-o', 'scene2-label-m', 'scene2-polar-night', 'scene2-line-om', 'scene2-line-ma', 'scene2-angle-arc', 'scene2-calc-text', 'scene2-angle-value', 'scene2-derive-arrow', 'scene2-side-earth', 'scene2-equator', 'scene2-subsolar-point', 'scene2-sun-ray', 'scene2-subsolar-label']
        ];

        // 工具函数：显示元素
        function showElement(id) {
            const el = document.getElementById(id);
            if (el) {
                el.classList.remove('scene2-hidden');
                el.classList.add('scene2-visible');
            }
        }

        // 工具函数：隐藏元素
        function hideElement(id) {
            const el = document.getElementById(id);
            if (el) {
                el.classList.remove('scene2-visible');
                el.classList.add('scene2-hidden');
            }
        }

        // 渲染指定步骤
        function renderStep(stepIndex) {
            const currentIds = steps[stepIndex];
            const allIds = steps[totalSteps - 1]; // 所有可能的元素 ID

            // 隐藏所有元素
            allIds.forEach(id => hideElement(id));

            // 显示当前步骤的元素
            currentIds.forEach(id => showElement(id));

            // 更新步骤指示器
            document.getElementById('scene2-step-indicator').textContent = `步骤 ${stepIndex + 1}/${totalSteps}`;

            // 更新按钮状态
            const prevBtn = document.getElementById('scene2-btn-prev-bg');
            const nextBtn = document.getElementById('scene2-btn-next-bg');
            const prevText = document.getElementById('scene2-btn-prev-text');
            const nextText = document.getElementById('scene2-btn-next-text');

            if (stepIndex === 0) {
                prevBtn.style.opacity = '0.5';
                prevText.style.opacity = '0.5';
            } else {
                prevBtn.style.opacity = '1';
                prevText.style.opacity = '1';
            }

            if (stepIndex === totalSteps - 1) {
                nextBtn.style.opacity = '0.5';
                nextText.style.opacity = '0.5';
            } else {
                nextBtn.style.opacity = '1';
                nextText.style.opacity = '1';
            }
        }

        // 按钮点击事件
        document.getElementById('scene2-btn-prev-bg').addEventListener('click', function() {
            if (currentStep > 0) {
                currentStep--;
                renderStep(currentStep);
            }
        });

        document.getElementById('scene2-btn-next-bg').addEventListener('click', function() {
            if (currentStep < totalSteps - 1) {
                currentStep++;
                renderStep(currentStep);
            }
        });

        // 添加点击反馈效果
        document.getElementById('scene2-btn-prev-bg').style.cursor = 'pointer';
        document.getElementById('scene2-btn-next-bg').style.cursor = 'pointer';
        document.getElementById('scene2-btn-prev-text').style.cursor = 'pointer';
        document.getElementById('scene2-btn-next-text').style.cursor = 'pointer';

        // 初始化
        renderStep(currentStep);
    })();
    </script>
