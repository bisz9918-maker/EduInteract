    <script>
    (function() {
        // Scene 3: A、B、C 三点物理量对比——地转偏向力与自转线速度
        
        // 状态变量
        let currentStep = 0;
        const totalSteps = 4;
        
        // DOM 元素引用
        const btnPrev = document.getElementById('scene3-btn-prev');
        const btnNext = document.getElementById('scene3-btn-next');
        const stepIndicator = document.getElementById('scene3-step-indicator');
        
        // 各步骤需要显示的元素
        const stepElements = {
            0: ['scene3-lat-a', 'scene3-lat-b', 'scene3-lat-c', 'scene3-meridian', 
                'scene3-point-o', 'scene3-label-o', 'scene3-point-a', 'scene3-label-a-bg', 'scene3-label-a',
                'scene3-point-b', 'scene3-label-b-bg', 'scene3-label-b',
                'scene3-point-c', 'scene3-label-c-bg', 'scene3-label-c'],
            1: ['scene3-relation-box'],
            2: ['scene3-force-a', 'scene3-force-b', 'scene3-force-c', 'scene3-conclusion-1'],
            3: ['scene3-velocity-a', 'scene3-velocity-b', 'scene3-velocity-c', 'scene3-conclusion-2'],
            4: ['scene3-highlight-c']
        };
        
        // 显示元素函数
        function showElement(id) {
            const el = document.getElementById(id);
            if (el) {
                el.style.display = '';
                el.style.opacity = '0';
                el.classList.add('fade-in');
                // 强制重绘后设置 opacity
                requestAnimationFrame(() => {
                    el.style.opacity = '1';
                });
            }
        }
        
        // 隐藏元素函数
        function hideElement(id) {
            const el = document.getElementById(id);
            if (el) {
                el.style.display = 'none';
                el.style.opacity = '0';
                el.classList.remove('fade-in');
            }
        }
        
        // 更新显示状态
        function updateDisplay() {
            // 步骤 0 的元素始终显示
            stepElements[0].forEach(id => {
                const el = document.getElementById(id);
                if (el) {
                    el.style.display = '';
                    el.style.opacity = '1';
                }
            });
            
            // 根据当前步骤显示/隐藏其他元素
            for (let step = 1; step <= totalSteps; step++) {
                if (step <= currentStep) {
                    stepElements[step].forEach(id => showElement(id));
                } else {
                    stepElements[step].forEach(id => hideElement(id));
                }
            }
            
            // 更新按钮状态
            btnPrev.disabled = currentStep === 0;
            btnNext.disabled = currentStep === totalSteps;
            
            // 更新步骤指示器
            stepIndicator.textContent = `步骤 ${currentStep}/${totalSteps}`;
        }
        
        // 上一步
        function stepBackward() {
            if (currentStep > 0) {
                currentStep--;
                updateDisplay();
            }
        }
        
        // 下一步
        function stepForward() {
            if (currentStep < totalSteps) {
                currentStep++;
                updateDisplay();
            }
        }
        
        // 绑定事件
        btnPrev.addEventListener('click', stepBackward);
        btnNext.addEventListener('click', stepForward);
        
        // 初始化显示
        updateDisplay();
        
    })();
    </script>
