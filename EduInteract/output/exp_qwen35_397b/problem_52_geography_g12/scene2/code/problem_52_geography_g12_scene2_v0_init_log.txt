    <script>
    (function() {
        // Scene 2: 地方时计算与昼长计算示意图
        // 布局坐标表：
        // | 元素 | 类型 | 位置/范围 | 大小 | 备注 |
        // |------|------|-----------|------|------|
        // | 南半球俯视图 | 圆形 | 圆心(200,180) | 半径 120 | 左侧 |
        // | 北半球俯视图 | 圆形 | 圆心(400,180) | 半径 120 | 右侧 |
        // | 甲点 | 圆点 | (140,150) | r=6 | 南半球 |
        // | 乙点 | 圆点 | (400,80) | r=6 | 北半球 |
        // | C 点 | 圆点 | (400,140) | r=6 | 北半球赤道 |
        // | 计算框 | 矩形 | (100,340)-(700,450) | 600×110 | 下方 |
        
        let currentStep = 0;
        const totalSteps = 4;
        
        // 获取 DOM 元素
        const btnPrev = document.getElementById('scene2-btn-prev');
        const btnNext = document.getElementById('scene2-btn-next');
        const stepIndicatorSvg = document.getElementById('scene2-step-indicator');
        const stepIndicatorHtml = document.getElementById('scene2-step-indicator-html');
        const calcBox = document.getElementById('scene2-calc-box');
        const calcText1 = document.getElementById('scene2-calc-text1');
        const calcText2 = document.getElementById('scene2-calc-text2');
        const calcResult = document.getElementById('scene2-calc-result');
        const step1Group = document.getElementById('scene2-step1-group');
        const step3Group = document.getElementById('scene2-step3-group');
        
        // 更新步骤显示
        function updateStepDisplay() {
            stepIndicatorSvg.textContent = `步骤 ${currentStep}/${totalSteps}`;
            stepIndicatorHtml.textContent = `步骤 ${currentStep}/${totalSteps}`;
            
            // 更新按钮状态
            btnPrev.disabled = currentStep === 0;
            btnNext.disabled = currentStep === totalSteps;
            
            // 根据步骤显示/隐藏元素
            updateStepElements();
        }
        
        // 更新步骤元素可见性
        function updateStepElements() {
            // 步骤 1 元素
            if (currentStep >= 1) {
                step1Group.classList.add('scene2-visible');
            } else {
                step1Group.classList.remove('scene2-visible');
            }
            
            // 步骤 2：计算框显示时差计算
            if (currentStep >= 2) {
                calcBox.classList.add('scene2-visible');
                calcText1.textContent = '时差 = 经度差 ÷ 15°/小时 = 90° ÷ 15° = 6 小时';
                calcText2.textContent = '乙地时间 = 甲地时间 + 时差 = 16 时 + 6 时 = 22 时';
                calcResult.textContent = '乙地时间：22 日 22 时';
            } else {
                calcBox.classList.remove('scene2-visible');
            }
            
            // 步骤 3 元素（昼弧）
            if (currentStep >= 3) {
                step3Group.classList.add('scene2-visible');
            } else {
                step3Group.classList.remove('scene2-visible');
            }
            
            // 步骤 3：计算框显示昼长计算
            if (currentStep >= 3) {
                calcBox.classList.add('scene2-visible');
                calcText1.textContent = 'C 点位于赤道，全年昼夜平分';
                calcText2.textContent = '昼长 = 昼弧所跨经度 ÷ 15°/小时 = 180° ÷ 15° = 12 小时';
                calcResult.textContent = 'C 点昼长：12 小时';
            }
        }
        
        // 上一步
        function stepBackward() {
            if (currentStep > 0) {
                currentStep--;
                updateStepDisplay();
            }
        }
        
        // 下一步
        function stepForward() {
            if (currentStep < totalSteps) {
                currentStep++;
                updateStepDisplay();
            }
        }
        
        // 绑定事件
        btnPrev.addEventListener('click', stepBackward);
        btnNext.addEventListener('click', stepForward);
        
        // 初始化
        updateStepDisplay();
    })();
    </script>
