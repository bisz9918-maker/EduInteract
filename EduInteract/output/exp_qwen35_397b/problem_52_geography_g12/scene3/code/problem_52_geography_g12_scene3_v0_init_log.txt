    <script>
    (function() {
        // Scene 3: 太阳直射点与正午太阳高度计算
        // 布局坐标表（实现对照依据）：
        // | 元素 | 类型 | 位置/范围 | 大小 | 标签偏移方向 |
        // | 太阳 | 圆形 | (80,200) | 60×60 | 上方 |
        // | 地球 | 圆形 | (250,200) | 80×80 | - |
        // | 南回归线 | 弧线 | (250,235) 附近 | 60×8 | 右侧 |
        // | 直射点 | 圆点 | (250,235) | 8×8 | 右下方 |
        // | 北京位置 | 圆点 | (250,160) | 8×8 | 右上方 |
        // | 公转轨道 | 椭圆 | (500,250) 中心 | 240×180 | - |
        // | 公式区背景 | 矩形 | (50,415)-(750,485) | 700×70 | - |
        // | 控件栏 | 按钮组 | 底部 | 300×25 | - |

        let currentStep = 0;
        const totalSteps = 4;

        // 元素可见性控制函数
        function show(el) {
            if (el) {
                el.style.display = '';
                el.style.opacity = '1';
                el.classList.add('scene3-visible');
                el.classList.remove('scene3-hidden');
            }
        }

        function hide(el) {
            if (el) {
                el.style.display = 'none';
                el.style.opacity = '0';
                el.classList.remove('scene3-visible');
                el.classList.add('scene3-hidden');
            }
        }

        // 获取所有需要控制的元素
        const elements = {
            // 步骤1显示的元素（南回归线、直射点）
            step1: [
                document.getElementById('scene3-tropic'),
                document.getElementById('scene3-tropic-label'),
                document.getElementById('scene3-direct-point'),
                document.getElementById('scene3-direct-label')
            ],
            // 步骤2显示的元素（北京位置、光线、高度角）
            step2: [
                document.getElementById('scene3-beijing-point'),
                document.getElementById('scene3-beijing-label'),
                document.getElementById('scene3-beijing-ray'),
                document.getElementById('scene3-beijing-ray-arrow'),
                document.getElementById('scene3-horizon'),
                document.getElementById('scene3-angle-arc'),
                document.getElementById('scene3-angle-label')
            ],
            // 步骤3显示的元素（公转轨道图）
            step3: [
                document.getElementById('scene3-orbit'),
                document.getElementById('scene3-orbit-sun'),
                document.getElementById('scene3-orbit-sun-label'),
                document.getElementById('scene3-winter-point'),
                document.getElementById('scene3-winter-label'),
                document.getElementById('scene3-winter-date'),
                document.getElementById('scene3-perihelion'),
                document.getElementById('scene3-perihelion-label'),
                document.getElementById('scene3-perihelion-date'),
                document.getElementById('scene3-aphelion'),
                document.getElementById('scene3-aphelion-label'),
                document.getElementById('scene3-aphelion-date'),
                document.getElementById('scene3-summer-point'),
                document.getElementById('scene3-summer-label'),
                document.getElementById('scene3-spring-point'),
                document.getElementById('scene3-spring-label'),
                document.getElementById('scene3-autumn-point'),
                document.getElementById('scene3-autumn-label'),
                document.getElementById('scene3-orbit-arrow')
            ],
            // 公式文本
            formulaText: document.getElementById('scene3-formula-text'),
            formulaText2: document.getElementById('scene3-formula-text2'),
            stepIndicator: document.getElementById('scene3-step-indicator'),
            prevBtn: document.getElementById('scene3-prev-btn'),
            nextBtn: document.getElementById('scene3-next-btn')
        };

        // 更新步骤显示
        function updateStep() {
            // 更新步骤指示器
            elements.stepIndicator.textContent = `步骤 ${currentStep + 1}/${totalSteps}`;

            // 更新按钮状态
            elements.prevBtn.disabled = currentStep === 0;
            elements.prevBtn.style.opacity = currentStep === 0 ? '0.5' : '1';
            elements.prevBtn.style.cursor = currentStep === 0 ? 'not-allowed' : 'pointer';

            elements.nextBtn.disabled = currentStep === totalSteps - 1;
            elements.nextBtn.style.opacity = currentStep === totalSteps - 1 ? '0.5' : '1';
            elements.nextBtn.style.cursor = currentStep === totalSteps - 1 ? 'not-allowed' : 'pointer';

            // 根据步骤显示/隐藏元素
            // 步骤0：只显示基础框架（太阳、地球、地轴、赤道）
            // 步骤1：+ 南回归线、直射点
            // 步骤2：+ 北京位置、光线、高度角
            // 步骤3：+ 公转轨道图

            // 步骤1元素
            if (currentStep >= 1) {
                elements.step1.forEach(el => show(el));
            } else {
                elements.step1.forEach(el => hide(el));
            }

            // 步骤2元素
            if (currentStep >= 2) {
                elements.step2.forEach(el => show(el));
            } else {
                elements.step2.forEach(el => hide(el));
            }

            // 步骤3元素
            if (currentStep >= 3) {
                elements.step3.forEach(el => show(el));
            } else {
                elements.step3.forEach(el => hide(el));
            }

            // 更新公式文本
            if (currentStep === 0) {
                elements.formulaText.textContent = '冬至日（12 月 22 日前后）→ 太阳直射南回归线（23°26\'S）';
                elements.formulaText2.textContent = '';
            } else if (currentStep === 1) {
                elements.formulaText.textContent = '正午太阳高度公式：H = 90° - |φ - δ|';
                elements.formulaText2.textContent = '北京：H = 90° - |39°N - (-23°26\'|) = 90° - 62°26\' = 27°34\'';
            } else if (currentStep === 2) {
                elements.formulaText.textContent = '正午太阳高度公式：H = 90° - |φ - δ|';
                elements.formulaText2.textContent = '北京：H = 90° - |39°N - (-23°26\'|) = 90° - 62°26\' = 27°34\'';
            } else if (currentStep === 3) {
                elements.formulaText.textContent = '12 月 22 日接近近日点（1 月初），地球距太阳较近';
                elements.formulaText2.textContent = '';
            }
        }

        // 按钮事件绑定
        elements.prevBtn.addEventListener('click', function() {
            if (currentStep > 0) {
                currentStep--;
                updateStep();
            }
        });

        elements.nextBtn.addEventListener('click', function() {
            if (currentStep < totalSteps - 1) {
                currentStep++;
                updateStep();
            }
        });

        // 初始化：隐藏所有步骤相关元素，只显示基础框架
        function init() {
            // 初始隐藏所有步骤元素
            elements.step1.forEach(el => hide(el));
            elements.step2.forEach(el => hide(el));
            elements.step3.forEach(el => hide(el));

            // 初始显示基础元素
            show(document.getElementById('scene3-sun'));
            show(document.getElementById('scene3-sun-label'));
            show(document.getElementById('scene3-sunray-direct'));
            show(document.getElementById('scene3-sunray-direct-arrow'));
            show(document.getElementById('scene3-earth'));
            show(document.getElementById('scene3-axis'));
            show(document.getElementById('scene3-axis-n'));
            show(document.getElementById('scene3-axis-s'));
            show(document.getElementById('scene3-equator'));
            show(document.getElementById('scene3-equator-label'));
            show(document.getElementById('scene3-formula-bg'));
            show(document.getElementById('scene3-formula-text'));
            show(document.getElementById('scene3-formula-text2'));
            show(document.getElementById('scene3-controls-bg'));
            show(document.getElementById('scene3-step-indicator'));
            show(elements.prevBtn);
            show(elements.nextBtn);

            currentStep = 0;
            updateStep();
        }

        // 页面加载完成后初始化
        init();
    })();
    </script>
