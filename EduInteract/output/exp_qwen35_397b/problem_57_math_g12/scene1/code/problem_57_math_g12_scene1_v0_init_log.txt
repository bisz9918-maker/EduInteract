    <script>
    (function() {
        // Scene 1 实现：空间几何轨迹问题分步演示

        // 状态变量
        let currentStep = 0;
        const totalSteps = 5;

        // 每步需要显示的元素 ID 映射
        const stepElements = {
            0: ['scene1-plane-m', 'scene1-point-o'],
            1: ['scene1-line-po', 'scene1-point-p', 'scene1-perp-symbol'],
            2: ['scene1-line-pa', 'scene1-point-a', 'scene1-line-oa', 'scene1-angle-alpha', 'scene1-label-alpha'],
            3: ['scene1-line-pb1', 'scene1-point-b1', 'scene1-line-ob', 'scene1-line-ab', 'scene1-angle-beta', 'scene1-label-beta', 'scene1-label-b'],
            4: ['scene1-cone-outline', 'scene1-line-pb2', 'scene1-line-pb3'],
            5: ['scene1-trajectory']
        };

        // 工具函数：显示元素
        function showElement(elementId) {
            const el = document.getElementById(elementId);
            if (el) {
                el.style.display = '';
                // 触发动画
                requestAnimationFrame(() => {
                    el.classList.add('scene1-visible');
                });
            }
        }

        // 工具函数：隐藏元素
        function hideElement(elementId) {
            const el = document.getElementById(elementId);
            if (el) {
                el.classList.remove('scene1-visible');
                // 等待动画完成后隐藏
                setTimeout(() => {
                    el.style.display = 'none';
                }, 400);
            }
        }

        // 更新步骤显示
        function updateStep() {
            // 更新步骤指示器
            document.getElementById('scene1-step-indicator').textContent = `步骤 ${currentStep}/${totalSteps}`;

            // 更新按钮状态
            document.getElementById('scene1-btn-prev').disabled = (currentStep === 0);
            document.getElementById('scene1-btn-next').disabled = (currentStep === totalSteps);

            // 根据当前步骤显示/隐藏元素
            for (let step = 0; step <= totalSteps; step++) {
                const elements = stepElements[step];
                if (step <= currentStep) {
                    // 显示当前步骤及之前步骤的元素
                    elements.forEach(id => showElement(id));
                } else {
                    // 隐藏后续步骤的元素
                    elements.forEach(id => hideElement(id));
                }
            }
        }

        // 初始化：隐藏所有步骤元素，只显示步骤 0
        function init() {
            // 初始隐藏所有可控制的元素
            for (let step = 1; step <= totalSteps; step++) {
                stepElements[step].forEach(id => {
                    const el = document.getElementById(id);
                    if (el) {
                        el.style.display = 'none';
                        el.classList.remove('scene1-visible');
                    }
                });
            }
            updateStep();
        }

        // 事件绑定
        document.getElementById('scene1-btn-prev').addEventListener('click', () => {
            if (currentStep > 0) {
                currentStep--;
                updateStep();
            }
        });

        document.getElementById('scene1-btn-next').addEventListener('click', () => {
            if (currentStep < totalSteps) {
                currentStep++;
                updateStep();
            }
        });

        // 页面加载完成后初始化
        document.addEventListener('DOMContentLoaded', init);
        // 如果 DOM 已加载，直接执行
        if (document.readyState !== 'loading') {
            init();
        }
    })();
    </script>
