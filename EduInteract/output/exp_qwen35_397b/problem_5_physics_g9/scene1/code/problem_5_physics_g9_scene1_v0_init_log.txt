    <script>
    (function() {
        // Scene 1 实现：实物电路与等效电路图对照展示

        // 状态管理
        let currentState = 0;
        const totalSteps = 4;

        // DOM 元素引用
        const prevBtn = document.getElementById('scene1-prev-btn');
        const nextBtn = document.getElementById('scene1-next-btn');
        const stepIndicator = document.getElementById('scene1-step-indicator');

        // 可切换显示的元素
        const nodesGroup = document.getElementById('scene1-nodes');
        const schematicCircuit = document.getElementById('scene1-schematic-circuit');
        const arrowLine = document.getElementById('scene1-arrow-line');
        const arrowHead = document.getElementById('scene1-arrow-head');

        // 高亮支路元素
        const L1Physical = document.getElementById('scene1-L1-physical');
        const L2Physical = document.getElementById('scene1-L2-physical');
        const L1Schematic = document.getElementById('scene1-L1-schematic');
        const L2Schematic = document.getElementById('scene1-L2-schematic');

        // 辅助函数：显示元素
        function showElement(el) {
            if (el) {
                el.classList.remove('fade-hidden');
                el.classList.add('fade-visible');
            }
        }

        // 辅助函数：隐藏元素
        function hideElement(el) {
            if (el) {
                el.classList.remove('fade-visible');
                el.classList.add('fade-hidden');
            }
        }

        // 辅助函数：添加高亮类
        function addHighlight(el, colorClass) {
            if (el) {
                el.classList.add(colorClass);
            }
        }

        // 辅助函数：移除高亮类
        function removeHighlight(el, colorClass) {
            if (el) {
                el.classList.remove(colorClass);
            }
        }

        // 更新按钮状态
        function updateButtons() {
            prevBtn.disabled = (currentState === 0);
            nextBtn.disabled = (currentState === totalSteps - 1);
            stepIndicator.textContent = '步骤 ' + (currentState + 1) + '/' + totalSteps;
        }

        // 更新场景状态
        function updateScene() {
            // 重置所有高亮
            removeHighlight(L1Physical, 'highlight-blue');
            removeHighlight(L2Physical, 'highlight-red');
            removeHighlight(L1Schematic, 'highlight-blue');
            removeHighlight(L2Schematic, 'highlight-red');

            // 根据当前状态显示/隐藏元素
            switch (currentState) {
                case 0:
                    // 初始状态：只显示实物图
                    hideElement(nodesGroup);
                    hideElement(schematicCircuit);
                    // 箭头为虚线灰色
                    arrowLine.setAttribute('stroke', '#888');
                    arrowLine.setAttribute('stroke-dasharray', '5,5');
                    arrowHead.setAttribute('fill', '#888');
                    break;

                case 1:
                    // 步骤 1：高亮节点
                    showElement(nodesGroup);
                    hideElement(schematicCircuit);
                    // 箭头保持虚线灰色
                    arrowLine.setAttribute('stroke', '#888');
                    arrowLine.setAttribute('stroke-dasharray', '5,5');
                    arrowHead.setAttribute('fill', '#888');
                    break;

                case 2:
                    // 步骤 2：显示等效电路图，实物图变淡
                    showElement(nodesGroup);
                    showElement(schematicCircuit);
                    // 箭头变实线蓝色
                    arrowLine.setAttribute('stroke', '#1976d2');
                    arrowLine.setAttribute('stroke-dasharray', 'none');
                    arrowHead.setAttribute('fill', '#1976d2');
                    // 实物图变淡
                    document.getElementById('scene1-physical-circuit').style.opacity = '0.5';
                    break;

                case 3:
                    // 步骤 3：高亮并联支路
                    showElement(nodesGroup);
                    showElement(schematicCircuit);
                    // 箭头保持实线蓝色
                    arrowLine.setAttribute('stroke', '#1976d2');
                    arrowLine.setAttribute('stroke-dasharray', 'none');
                    arrowHead.setAttribute('fill', '#1976d2');
                    // 实物图保持变淡
                    document.getElementById('scene1-physical-circuit').style.opacity = '0.5';
                    // 高亮支路
                    addHighlight(L1Schematic, 'highlight-blue');
                    addHighlight(L2Schematic, 'highlight-red');
                    break;
            }

            updateButtons();
        }

        // 事件监听
        prevBtn.addEventListener('click', function() {
            if (currentState > 0) {
                currentState--;
                updateScene();
            }
        });

        nextBtn.addEventListener('click', function() {
            if (currentState < totalSteps - 1) {
                currentState++;
                updateScene();
            }
        });

        // 初始化
        updateScene();
    })();
    </script>
