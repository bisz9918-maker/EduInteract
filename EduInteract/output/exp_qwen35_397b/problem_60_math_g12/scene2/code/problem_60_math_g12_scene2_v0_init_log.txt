    <script>
    (function() {
        // Scene 2: 椭圆切线性质探究
        
        // 椭圆参数
        const CX = 400, CY = 220;  // 中心
        const RX = 280, RY = 140;  // 长半轴、短半轴
        const A = RX;              // 长半轴
        const B = RY;              // 短半轴
        const B2 = B * B;          // b² = 19600
        const C_FOCUS = Math.sqrt(A*A - B*B);  // 焦距 ≈ 242.5
        const F1_X = CX + C_FOCUS;  // 右焦点 x
        const F2_X = CX - C_FOCUS;  // 左焦点 x
        
        // 切线端点
        const C_X = 120, C_Y = 100;  // 点 C
        const D_X = 680, D_Y = 80;   // 点 D
        
        // 当前 P 点参数角（初始 120°）
        let theta = 2 * Math.PI / 3;
        
        // DOM 元素
        const pointP = document.getElementById('scene2-point-p');
        const labelP = document.getElementById('scene2-label-p');
        const labelPBg = document.getElementById('scene2-label-p-bg');
        const segPC = document.getElementById('scene2-seg-pc');
        const segPD = document.getElementById('scene2-seg-pd');
        const segPF1 = document.getElementById('scene2-seg-pf1');
        const segPF2 = document.getElementById('scene2-seg-pf2');
        const tangentCD = document.getElementById('scene2-tangent-cd');
        const pointC = document.getElementById('scene2-point-c');
        const pointD = document.getElementById('scene2-point-d');
        const labelC = document.getElementById('scene2-label-c');
        const labelCBg = document.getElementById('scene2-label-c-bg');
        const labelD = document.getElementById('scene2-label-d');
        const labelDBg = document.getElementById('scene2-label-d-bg');
        
        // 数值显示
        const valPC = document.getElementById('scene2-val-pc');
        const valPD = document.getElementById('scene2-val-pd');
        const valProduct = document.getElementById('scene2-val-product');
        const valPFProduct = document.getElementById('scene2-val-pf-product');
        const valB2 = document.getElementById('scene2-val-b2');
        
        // 初始化 b² 显示
        valB2.textContent = `b² = ${B2.toFixed(0)}`;
        
        // 根据参数角计算 P 点坐标
        function getPPoint(theta) {
            return {
                x: CX + RX * Math.cos(theta),
                y: CY + RY * Math.sin(theta)
            };
        }
        
        // 计算椭圆上点 P 的切线与 x=±a 的交点 C 和 D
        // 椭圆切线方程: (x0*x)/a² + (y0*y)/b² = 1
        // 与 x = -a 交点 C: y = b²(1 - x0*(-a)/a²) / y0 = b²(1 + x0/a) / y0
        // 与 x = a 交点 D: y = b²(1 - x0*a/a²) / y0 = b²(1 - x0/a) / y0
        function getTangentIntersections(px, py) {
            // 切线方程: (px*x)/A² + (py*y)/B² = 1
            // 当 x = C_X = 120 (即 -a 位置): y = B² * (1 - px*C_X/A²) / py
            // 当 x = D_X = 680 (即 +a 位置): y = B² * (1 - px*D_X/A²) / py
            
            const cY = (B2 * (1 - (px * C_X) / (A * A))) / py;
            const dY = (B2 * (1 - (px * D_X) / (A * A))) / py;
            
            return {
                c: { x: C_X, y: cY },
                d: { x: D_X, y: dY }
            };
        }
        
        // 两点距离
        function distance(p1, p2) {
            return Math.sqrt((p1.x - p2.x)**2 + (p1.y - p2.y)**2);
        }
        
        // 更新显示
        function updateDisplay() {
            const p = getPPoint(theta);
            
            // 获取切线交点 C, D
            const intersections = getTangentIntersections(p.x, p.y);
            const c = intersections.c;
            const d = intersections.d;
            
            // 更新 P 点位置
            pointP.setAttribute('cx', p.x);
            pointP.setAttribute('cy', p.y);
            
            // 更新 C, D 点位置
            pointC.setAttribute('cx', c.x);
            pointC.setAttribute('cy', c.y);
            pointD.setAttribute('cx', d.x);
            pointD.setAttribute('cy', d.y);
            
            // 更新切线 CD
            tangentCD.setAttribute('x1', c.x);
            tangentCD.setAttribute('y1', c.y);
            tangentCD.setAttribute('x2', d.x);
            tangentCD.setAttribute('y2', d.y);
            
            // 更新标签 C, D 位置
            labelC.setAttribute('x', c.x - 12);
            labelC.setAttribute('y', c.y - 12);
            labelCBg.setAttribute('x', c.x - 17);
            labelCBg.setAttribute('y', c.y - 22);
            
            labelD.setAttribute('x', d.x + 5);
            labelD.setAttribute('y', d.y - 12);
            labelDBg.setAttribute('x', d.x);
            labelDBg.setAttribute('y', d.y - 22);
            
            // 更新 P 标签位置（根据 P 的位置调整偏移）
            let pLabelOffsetX = 15, pLabelOffsetY = -15;
            if (p.x < CX) {
                pLabelOffsetX = 15;
            } else {
                pLabelOffsetX = -25;
            }
            if (p.y < CY) {
                pLabelOffsetY = -10;
            } else {
                pLabelOffsetY = 20;
            }
            labelP.setAttribute('x', p.x + pLabelOffsetX);
            labelP.setAttribute('y', p.y + pLabelOffsetY);
            labelPBg.setAttribute('x', p.x + pLabelOffsetX - 5);
            labelPBg.setAttribute('y', p.y + pLabelOffsetY - 22);
            
            // 更新线段 PC, PD, PF1, PF2
            segPC.setAttribute('x1', p.x);
            segPC.setAttribute('y1', p.y);
            segPC.setAttribute('x2', c.x);
            segPC.setAttribute('y2', c.y);
            
            segPD.setAttribute('x1', p.x);
            segPD.setAttribute('y1', p.y);
            segPD.setAttribute('x2', d.x);
            segPD.setAttribute('y2', d.y);
            
            segPF1.setAttribute('x1', p.x);
            segPF1.setAttribute('y1', p.y);
            segPF1.setAttribute('x2', F1_X);
            segPF1.setAttribute('y2', CY);
            
            segPF2.setAttribute('x1', p.x);
            segPF2.setAttribute('y1', p.y);
            segPF2.setAttribute('x2', F2_X);
            segPF2.setAttribute('y2', CY);
            
            // 计算长度
            const pc = distance(p, c);
            const pd = distance(p, d);
            const pf1 = distance(p, {x: F1_X, y: CY});
            const pf2 = distance(p, {x: F2_X, y: CY});
            
            // 更新数值显示
            valPC.textContent = `PC = ${pc.toFixed(2)}`;
            valPD.textContent = `PD = ${pd.toFixed(2)}`;
            valProduct.textContent = `PC·PD = ${(pc * pd).toFixed(2)}`;
            valPFProduct.textContent = `PF₁·PF₂ = ${(pf1 * pf2).toFixed(2)}`;
        }
        
        // 拖拽交互
        let isDragging = false;
        
        pointP.addEventListener('mousedown', function(e) {
            isDragging = true;
            e.preventDefault();
        });
        
        document.addEventListener('mousemove', function(e) {
            if (!isDragging) return;
            
            const svg = document.getElementById('scene2-svg');
            const rect = svg.getBoundingClientRect();
            const mouseX = e.clientX - rect.left;
            const mouseY = e.clientY - rect.top;
            
            // 将鼠标位置映射到椭圆参数角
            // 使用 atan2 计算相对于椭圆中心的参数角
            const dx = mouseX - CX;
            const dy = mouseY - CY;
            
            // 考虑椭圆缩放，计算参数角
            theta = Math.atan2(dy / RY, dx / RX);
            
            // 限制 theta 在 [0, 2π)
            if (theta < 0) theta += 2 * Math.PI;
            
            updateDisplay();
        });
        
        document.addEventListener('mouseup', function() {
            isDragging = false;
        });
        
        // 触摸支持
        pointP.addEventListener('touchstart', function(e) {
            isDragging = true;
            e.preventDefault();
        });
        
        document.addEventListener('touchmove', function(e) {
            if (!isDragging) return;
            
            const svg = document.getElementById('scene2-svg');
            const rect = svg.getBoundingClientRect();
            const touch = e.touches[0];
            const mouseX = touch.clientX - rect.left;
            const mouseY = touch.clientY - rect.top;
            
            const dx = mouseX - CX;
            const dy = mouseY - CY;
            theta = Math.atan2(dy / RY, dx / RX);
            
            if (theta < 0) theta += 2 * Math.PI;
            
            updateDisplay();
            e.preventDefault();
        });
        
        document.addEventListener('touchend', function() {
            isDragging = false;
        });
        
        // 初始化显示
        updateDisplay();
        
    })();
    </script>
