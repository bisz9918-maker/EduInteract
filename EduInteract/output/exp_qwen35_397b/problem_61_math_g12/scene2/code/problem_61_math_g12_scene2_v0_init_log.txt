    <script>
    (function() {
        // Scene 2: 球心角与球面距离计算
        
        let currentStep = 0;
        const totalSteps = 5;
        
        const btnPrev = document.getElementById('scene2-btn-prev');
        const btnNext = document.getElementById('scene2-btn-next');
        const stepIndicator = document.getElementById('scene2-step-indicator');
        
        // 元素引用
        const elements = {
            triAbc: ['scene2-tri-abc-ab', 'scene2-tri-abc-bc', 'scene2-tri-abc-ac'],
            triObc: ['scene2-tri-obc-ob', 'scene2-tri-obc-oc', 'scene2-tri-obc-bc'],
            labels: {
                bcLen: 'scene2-label-bc-len',
                obLen: 'scene2-label-ob-len',
                ocLen: 'scene2-label-oc-len',
                angleBoc: 'scene2-label-angle-boc'
            },
            arcBc: 'scene2-arc-bc',
            angleBoc: 'scene2-angle-boc',
            unfoldedTri: 'scene2-unfolded-tri',
            formulaBg: 'scene2-formula-bg',
            formulaText: 'scene2-formula-text',
            calcText: 'scene2-calc-text',
            deriveText: 'scene2-derive-text'
        };
        
        // 辅助函数：设置元素透明度
        function setOpacity(id, opacity, duration = 400) {
            const el = document.getElementById(id);
            if (el) {
                el.style.transition = `opacity ${duration}ms ease`;
                el.style.opacity = opacity;
            }
        }
        
        // 辅助函数：设置线段粗细
        function setStrokeWidth(ids, width, duration = 400) {
            ids.forEach(id => {
                const el = document.getElementById(id);
                if (el) {
                    el.style.transition = `stroke-width ${duration}ms ease, opacity ${duration}ms ease`;
                    el.style.strokeWidth = width;
                }
            });
        }
        
        // 更新步骤显示
        function updateStepDisplay() {
            stepIndicator.textContent = `步骤 ${currentStep}/${totalSteps}`;
            btnPrev.disabled = currentStep === 0;
            btnNext.disabled = currentStep === totalSteps;
        }
        
        // 更新场景状态
        function updateScene() {
            // 重置所有元素状态
            elements.triAbc.forEach(id => {
                const el = document.getElementById(id);
                if (el) {
                    el.style.strokeWidth = 2;
                    el.style.opacity = 0.5;
                }
            });
            elements.triObc.forEach(id => {
                const el = document.getElementById(id);
                if (el) {
                    el.style.strokeWidth = 2;
                    el.style.opacity = 0.5;
                }
            });
            setOpacity(elements.labels.bcLen, 0);
            setOpacity(elements.labels.obLen, 0);
            setOpacity(elements.labels.ocLen, 0);
            setOpacity(elements.labels.angleBoc, 0);
            setOpacity(elements.arcBc, 0);
            setOpacity(elements.angleBoc, 0);
            setOpacity(elements.unfoldedTri, 0);
            setOpacity(elements.formulaBg, 0);
            setOpacity(elements.formulaText, 0);
            setOpacity(elements.calcText, 0);
            setOpacity(elements.deriveText, 0);
            
            // 根据当前步骤设置状态
            switch(currentStep) {
                case 0:
                    // 初始状态：显示基础图形
                    break;
                    
                case 1:
                    // 高亮△ABC，显示 BC 长度
                    setStrokeWidth(elements.triAbc, 3);
                    setOpacity(elements.labels.bcLen, 1);
                    setOpacity(elements.deriveText, 1);
                    break;
                    
                case 2:
                    // 高亮△OBC，显示 OB、OC 长度，显示展开图
                    setStrokeWidth(elements.triAbc, 2);
                    elements.triAbc.forEach(id => {
                        const el = document.getElementById(id);
                        if (el) el.style.opacity = 0.3;
                    });
                    setStrokeWidth(elements.triObc, 3);
                    setOpacity(elements.triObc, 1);
                    setOpacity(elements.labels.obLen, 1);
                    setOpacity(elements.labels.ocLen, 1);
                    setOpacity(elements.labels.bcLen, 1);
                    setOpacity(elements.unfoldedTri, 1);
                    break;
                    
                case 3:
                    // 显示∠BOC 角标记
                    setStrokeWidth(elements.triAbc, 2);
                    elements.triAbc.forEach(id => {
                        const el = document.getElementById(id);
                        if (el) el.style.opacity = 0.3;
                    });
                    setStrokeWidth(elements.triObc, 3);
                    setOpacity(elements.triObc, 1);
                    setOpacity(elements.labels.obLen, 1);
                    setOpacity(elements.labels.ocLen, 1);
                    setOpacity(elements.labels.bcLen, 1);
                    setOpacity(elements.unfoldedTri, 1);
                    setOpacity(elements.angleBoc, 1);
                    setOpacity(elements.labels.angleBoc, 1);
                    break;
                    
                case 4:
                    // 显示公式
                    setStrokeWidth(elements.triAbc, 2);
                    elements.triAbc.forEach(id => {
                        const el = document.getElementById(id);
                        if (el) el.style.opacity = 0.3;
                    });
                    setStrokeWidth(elements.triObc, 3);
                    setOpacity(elements.triObc, 1);
                    setOpacity(elements.labels.obLen, 1);
                    setOpacity(elements.labels.ocLen, 1);
                    setOpacity(elements.labels.bcLen, 1);
                    setOpacity(elements.unfoldedTri, 1);
                    setOpacity(elements.angleBoc, 1);
                    setOpacity(elements.labels.angleBoc, 1);
                    setOpacity(elements.formulaBg, 1);
                    setOpacity(elements.formulaText, 1);
                    setOpacity(elements.calcText, 1);
                    break;
                    
                case 5:
                    // 显示大圆弧 BC，降低其他元素透明度
                    setStrokeWidth(elements.triAbc, 2);
                    elements.triAbc.forEach(id => {
                        const el = document.getElementById(id);
                        if (el) el.style.opacity = 0.3;
                    });
                    setStrokeWidth(elements.triObc, 2);
                    elements.triObc.forEach(id => {
                        const el = document.getElementById(id);
                        if (el) el.style.opacity = 0.3;
                    });
                    setOpacity(elements.labels.obLen, 0.5);
                    setOpacity(elements.labels.ocLen, 0.5);
                    setOpacity(elements.labels.bcLen, 0.5);
                    setOpacity(elements.unfoldedTri, 0.5);
                    setOpacity(elements.angleBoc, 1);
                    setOpacity(elements.labels.angleBoc, 1);
                    setOpacity(elements.formulaBg, 1);
                    setOpacity(elements.formulaText, 1);
                    setOpacity(elements.calcText, 1);
                    setOpacity(elements.arcBc, 1);
                    break;
            }
            
            updateStepDisplay();
        }
        
        // 按钮事件绑定
        btnPrev.addEventListener('click', function() {
            if (currentStep > 0) {
                currentStep--;
                updateScene();
            }
        });
        
        btnNext.addEventListener('click', function() {
            if (currentStep < totalSteps) {
                currentStep++;
                updateScene();
            }
        });
        
        // 初始化
        updateStepDisplay();
    })();
    </script>
