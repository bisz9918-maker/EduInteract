    <script>
    (function() {
        // Scene 2: 四面体中点截面命题验证
        // 布局坐标表参考：
        // A(400,80), B(200,350), C(400,420), D(600,350)
        // E(300,215), F(500,385), G(300,385), H(500,215)
        
        let currentStep = 0;
        const totalSteps = 5;
        
        const titles = [
            "基础结构",
            "命题①：GF、EH、BD 三线共点（✓真）",
            "命题②：存在平面α₀使 GF∥EH∥BD（✓真）",
            "命题③：H 可在 AD 延长线上（✓真）",
            "命题④：S△EFG = S△EFH（✓真）"
        ];
        
        // 工具函数：显示/隐藏元素
        function show(el) {
            if (el) {
                el.classList.remove('scene2-hidden');
                el.classList.add('scene2-visible');
            }
        }
        
        function hide(el) {
            if (el) {
                el.classList.remove('scene2-visible');
                el.classList.add('scene2-hidden');
            }
        }
        
        // 更新步骤指示器
        function updateStepIndicators() {
            for (let i = 0; i < totalSteps; i++) {
                const ind = document.getElementById('scene2-step' + i + '-ind');
                if (i === currentStep) {
                    ind.classList.remove('scene2-step-indicator');
                    ind.classList.add('scene2-step-active');
                } else {
                    ind.classList.remove('scene2-step-active');
                    ind.classList.add('scene2-step-indicator');
                }
            }
        }
        
        // 更新标题
        function updateTitle() {
            const title = document.getElementById('scene2-title');
            title.textContent = titles[currentStep];
        }
        
        // 更新按钮状态
        function updateButtons() {
            const prevBg = document.getElementById('scene2-btn-prev-bg');
            const prevText = document.getElementById('scene2-btn-prev-text');
            const nextBg = document.getElementById('scene2-btn-next-bg');
            const nextText = document.getElementById('scene2-btn-next-text');
            
            if (currentStep === 0) {
                prevBg.setAttribute('fill', '#e0e0e0');
                prevText.setAttribute('fill', '#999');
            } else {
                prevBg.setAttribute('fill', '#f5f5f5');
                prevText.setAttribute('fill', '#333');
            }
            
            if (currentStep === totalSteps - 1) {
                nextBg.setAttribute('fill', '#e0e0e0');
                nextText.setAttribute('fill', '#999');
            } else {
                nextBg.setAttribute('fill', '#f5f5f5');
                nextText.setAttribute('fill', '#333');
            }
        }
        
        // 渲染步骤
        function renderStep(step) {
            // 隐藏所有命题相关元素
            hide(document.getElementById('scene2-prop1'));
            hide(document.getElementById('scene2-prop2'));
            hide(document.getElementById('scene2-prop3'));
            hide(document.getElementById('scene2-prop4'));
            
            // 显示基础结构（始终可见）
            show(document.getElementById('scene2-tetrahedron'));
            show(document.getElementById('scene2-midpoints'));
            show(document.getElementById('scene2-section'));
            
            // 根据步骤显示特定元素
            switch (step) {
                case 0:
                    // 基础结构：只显示四面体、中点、截面点
                    break;
                case 1:
                    // 命题①：三线共点
                    show(document.getElementById('scene2-prop1'));
                    break;
                case 2:
                    // 命题②：平行关系
                    show(document.getElementById('scene2-prop2'));
                    break;
                case 3:
                    // 命题③：延长线位置
                    show(document.getElementById('scene2-prop3'));
                    break;
                case 4:
                    // 命题④：面积相等
                    show(document.getElementById('scene2-prop4'));
                    break;
            }
            
            updateStepIndicators();
            updateTitle();
            updateButtons();
        }
        
        // 按钮点击处理
        function handlePrev() {
            if (currentStep > 0) {
                currentStep--;
                renderStep(currentStep);
            }
        }
        
        function handleNext() {
            if (currentStep < totalSteps - 1) {
                currentStep++;
                renderStep(currentStep);
            }
        }
        
        // 绑定按钮事件
        const prevBtn = document.getElementById('scene2-btn-prev-bg');
        const nextBtn = document.getElementById('scene2-btn-next-bg');
        
        prevBtn.addEventListener('click', handlePrev);
        nextBtn.addEventListener('click', handleNext);
        
        // 初始化
        renderStep(0);
    })();
    </script>
