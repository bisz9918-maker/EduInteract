    <script>
    (function() {
        // 状态变量
        var currentStep = 0;
        var totalSteps = 4;

        // 每步需要显示的元素 ID 列表
        var steps = {
            0: [
                // 基础图形：三角形、圆、点、标签、直角标记 C
                'scene1-line-bc', 'scene1-line-ac', 'scene1-line-ab',
                'scene1-circle',
                'scene1-point-a', 'scene1-point-b', 'scene1-point-c', 'scene1-point-d', 'scene1-point-o',
                'scene1-label-a', 'scene1-label-b', 'scene1-label-c', 'scene1-label-d', 'scene1-label-o',
                'scene1-right-angle-c'
            ],
            1: [
                // 步骤 1：新增长度标注
                'scene1-line-bc', 'scene1-line-ac', 'scene1-line-ab',
                'scene1-circle',
                'scene1-point-a', 'scene1-point-b', 'scene1-point-c', 'scene1-point-d', 'scene1-point-o',
                'scene1-label-a', 'scene1-label-b', 'scene1-label-c', 'scene1-label-d', 'scene1-label-o',
                'scene1-right-angle-c',
                'scene1-anno-ac', 'scene1-anno-bc', 'scene1-anno-ab'
            ],
            2: [
                // 步骤 2：新增辅助线 CD、直角标记 D、角度说明
                'scene1-line-bc', 'scene1-line-ac', 'scene1-line-ab',
                'scene1-circle',
                'scene1-point-a', 'scene1-point-b', 'scene1-point-c', 'scene1-point-d', 'scene1-point-o',
                'scene1-label-a', 'scene1-label-b', 'scene1-label-c', 'scene1-label-d', 'scene1-label-o',
                'scene1-right-angle-c',
                'scene1-anno-ac', 'scene1-anno-bc', 'scene1-anno-ab',
                'scene1-line-cd', 'scene1-right-angle-d', 'scene1-angle-text-d'
            ],
            3: [
                // 步骤 3：新增三角形高亮、角度弧线、相似提示
                'scene1-line-bc', 'scene1-line-ac', 'scene1-line-ab',
                'scene1-circle',
                'scene1-point-a', 'scene1-point-b', 'scene1-point-c', 'scene1-point-d', 'scene1-point-o',
                'scene1-label-a', 'scene1-label-b', 'scene1-label-c', 'scene1-label-d', 'scene1-label-o',
                'scene1-right-angle-c',
                'scene1-anno-ac', 'scene1-anno-bc', 'scene1-anno-ab',
                'scene1-line-cd', 'scene1-right-angle-d', 'scene1-angle-text-d',
                'scene1-highlight-bac', 'scene1-highlight-bcd',
                'scene1-angle-b', 'scene1-angle-bca', 'scene1-angle-bdc',
                'scene1-similar-text'
            ],
            4: [
                // 步骤 4：新增比例式、结果、BD 标注
                'scene1-line-bc', 'scene1-line-ac', 'scene1-line-ab',
                'scene1-circle',
                'scene1-point-a', 'scene1-point-b', 'scene1-point-c', 'scene1-point-d', 'scene1-point-o',
                'scene1-label-a', 'scene1-label-b', 'scene1-label-c', 'scene1-label-d', 'scene1-label-o',
                'scene1-right-angle-c',
                'scene1-anno-ac', 'scene1-anno-bc', 'scene1-anno-ab',
                'scene1-line-cd', 'scene1-right-angle-d', 'scene1-angle-text-d',
                'scene1-highlight-bac', 'scene1-highlight-bcd',
                'scene1-angle-b', 'scene1-angle-bca', 'scene1-angle-bdc',
                'scene1-similar-text',
                'scene1-anno-bd', 'scene1-proportion', 'scene1-proportion-num', 'scene1-result'
            ]
        };

        // 显示元素的辅助函数
        function showElement(id) {
            var el = document.getElementById(id);
            if (el) {
                el.style.display = '';
                // 触发重绘以启动淡入动画
                setTimeout(function() {
                    el.classList.add('visible');
                }, 10);
            }
        }

        // 隐藏元素的辅助函数
        function hideElement(id) {
            var el = document.getElementById(id);
            if (el) {
                el.classList.remove('visible');
                el.style.display = 'none';
            }
        }

        // 更新步骤显示
        function updateStep() {
            // 获取当前步骤需要显示的元素
            var visibleIds = steps[currentStep];

            // 遍历所有可能的元素 ID
            var allIds = [];
            for (var step = 0; step <= totalSteps; step++) {
                allIds = allIds.concat(steps[step]);
            }
            // 去重
            allIds = allIds.filter(function(id, index, self) {
                return self.indexOf(id) === index;
            });

            // 更新每个元素的显示状态
            allIds.forEach(function(id) {
                if (visibleIds.indexOf(id) !== -1) {
                    showElement(id);
                } else {
                    hideElement(id);
                }
            });

            // 更新步骤指示器
            document.getElementById('scene1-step-indicator').textContent = '步骤 ' + currentStep + '/' + totalSteps;

            // 更新按钮状态
            var prevBtn = document.getElementById('scene1-btn-prev');
            var nextBtn = document.getElementById('scene1-btn-next');
            prevBtn.disabled = (currentStep === 0);
            nextBtn.disabled = (currentStep === totalSteps);
            if (currentStep === totalSteps) {
                nextBtn.textContent = '完成';
            } else {
                nextBtn.textContent = '下一步 ▶';
            }
        }

        // 绑定按钮事件
        document.getElementById('scene1-btn-prev').addEventListener('click', function() {
            if (currentStep > 0) {
                currentStep--;
                updateStep();
            }
        });

        document.getElementById('scene1-btn-next').addEventListener('click', function() {
            if (currentStep < totalSteps) {
                currentStep++;
                updateStep();
            }
        });

        // 初始化显示
        updateStep();
    })();
    </script>
