    <script>
    (function() {
        // Scene 2: v-t 图像交互实现
        
        // 物理常量
        const v_m = 4.9;  // 最大速度 (m/s)
        const tau = 1.0;  // 时间常数 (s)
        const a0 = 4.9;   // 初始加速度 (m/s²)
        
        // 坐标映射
        const tMin = 0, tMax = 5;
        const xMin = 80, xMax = 580;
        const vMin = 0, vMax = 6;
        const yMin = 80, yMax = 400;
        
        // 映射函数
        function tToX(t) {
            return xMin + (t - tMin) / (tMax - tMin) * (xMax - xMin);
        }
        
        function vToY(v) {
            return yMax - (v - vMin) / (vMax - vMin) * (yMax - yMin);
        }
        
        // 物理函数
        function vOfT(t) {
            return v_m * (1 - Math.exp(-t / tau));
        }
        
        function aOfT(t) {
            return a0 * Math.exp(-t / tau);
        }
        
        // 生成 v-t 曲线路径
        function generateCurvePath() {
            let d = `M ${tToX(0)} ${vToY(vOfT(0))}`;
            for (let t = 0.05; t <= 5; t += 0.05) {
                d += ` L ${tToX(t)} ${vToY(vOfT(t))}`;
            }
            return d;
        }
        
        // 更新当前状态
        function updateState(t) {
            const v = vOfT(t);
            const a = aOfT(t);
            const x = tToX(t);
            const y = vToY(v);
            
            // 更新当前点
            const point = document.getElementById('scene2-current-point');
            point.setAttribute('cx', x);
            point.setAttribute('cy', y);
            
            // 更新切线（长度为 60px，斜率为 a 的像素表示）
            const tangent = document.getElementById('scene2-tangent');
            const dvPerPixel = (vMax - vMin) / (yMax - yMin);
            const dtPerPixel = (tMax - tMin) / (xMax - xMin);
            const slopePixels = -a * (dvPerPixel / dtPerPixel) * 30; // 30px 半长
            tangent.setAttribute('x1', x - 30);
            tangent.setAttribute('y1', y + slopePixels);
            tangent.setAttribute('x2', x + 30);
            tangent.setAttribute('y2', y - slopePixels);
            
            // 更新数值面板
            document.getElementById('scene2-val-t').textContent = `t=${t.toFixed(2)}s`;
            document.getElementById('scene2-val-v').textContent = `v=${v.toFixed(2)}m/s`;
            document.getElementById('scene2-val-a').textContent = `a=${a.toFixed(2)}m/s²`;
            
            // 更新滑块手柄位置
            const handle = document.getElementById('scene2-slider-handle');
            handle.style.left = x + 'px';
        }
        
        // 初始化
        const curve = document.getElementById('scene2-curve-vt');
        curve.setAttribute('d', generateCurvePath());
        
        // 滑块交互
        let isDragging = false;
        const track = document.getElementById('scene2-slider-track');
        const handle = document.getElementById('scene2-slider-handle');
        
        function handleMove(clientX) {
            const rect = track.getBoundingClientRect();
            let x = clientX - rect.left;
            x = Math.max(0, Math.min(x, rect.width));
            const t = tMin + (x / rect.width) * (tMax - tMin);
            updateState(t);
        }
        
        handle.addEventListener('mousedown', (e) => {
            isDragging = true;
            e.preventDefault();
        });
        
        track.addEventListener('mousedown', (e) => {
            isDragging = true;
            handleMove(e.clientX);
        });
        
        document.addEventListener('mousemove', (e) => {
            if (isDragging) {
                handleMove(e.clientX);
            }
        });
        
        document.addEventListener('mouseup', () => {
            isDragging = false;
        });
        
        // 触摸支持
        handle.addEventListener('touchstart', (e) => {
            isDragging = true;
            e.preventDefault();
        });
        
        track.addEventListener('touchstart', (e) => {
            isDragging = true;
            handleMove(e.touches[0].clientX);
            e.preventDefault();
        });
        
        document.addEventListener('touchmove', (e) => {
            if (isDragging) {
                handleMove(e.touches[0].clientX);
                e.preventDefault();
            }
        });
        
        document.addEventListener('touchend', () => {
            isDragging = false;
        });
        
        // 初始状态
        updateState(0);
    })();
    </script>
