    <script>
    (function() {
        // Scene 1 实现：平行板电容器与三球系统初始状态
        
        // 状态变量
        let currentState = 0;
        const totalSteps = 4;
        
        // DOM 元素
        const btnPrev = document.getElementById('scene1-btn-prev');
        const btnNext = document.getElementById('scene1-btn-next');
        const stepIndicator = document.getElementById('scene1-step-indicator');
        
        // 各步骤的元素组
        const step1Elements = document.getElementById('scene1-plate-labels');
        const step2Elements = document.getElementById('scene1-distance-d');
        const step3Elements = document.getElementById('scene1-balls-rods');
        const step4Elements = document.getElementById('scene1-distance-l');
        
        // 显示/隐藏工具函数
        function showElement(el) {
            if (el) {
                el.style.display = 'block';
                el.style.opacity = '1';
                el.classList.add('fade-in');
            }
        }
        
        function hideElement(el) {
            if (el) {
                el.style.display = 'none';
                el.style.opacity = '0';
                el.classList.remove('fade-in');
            }
        }
        
        // 更新显示状态
        function updateDisplay() {
            // 根据当前状态显示对应元素
            if (currentState >= 1) {
                showElement(step1Elements);
            } else {
                hideElement(step1Elements);
            }
            
            if (currentState >= 2) {
                showElement(step2Elements);
            } else {
                hideElement(step2Elements);
            }
            
            if (currentState >= 3) {
                showElement(step3Elements);
            } else {
                hideElement(step3Elements);
            }
            
            if (currentState >= 4) {
                showElement(step4Elements);
            } else {
                hideElement(step4Elements);
            }
            
            // 更新按钮状态
            btnPrev.disabled = (currentState === 0);
            btnNext.disabled = (currentState === totalSteps - 1);
            
            // 更新步骤指示器
            stepIndicator.textContent = '步骤 ' + (currentState + 1) + '/' + totalSteps;
            
            // 重新渲染 MathJax
            if (window.MathJax) {
                MathJax.typesetPromise();
            }
        }
        
        // 按钮事件处理
        btnPrev.addEventListener('click', function() {
            if (currentState > 0) {
                currentState--;
                updateDisplay();
            }
        });
        
        btnNext.addEventListener('click', function() {
            if (currentState < totalSteps - 1) {
                currentState++;
                updateDisplay();
            }
        });
        
        // 初始化显示
        updateDisplay();
        
    })();
    </script>
