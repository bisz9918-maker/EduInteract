    <script>
    (function() {
        // Scene 1 实现 - 电磁感应装置结构与电路分析

        // 状态变量
        let currentStep = 0;
        const totalSteps = 3;

        // DOM 元素
        const btnPrev = document.getElementById('scene1-btn-prev');
        const btnNext = document.getElementById('scene1-btn-next');
        const stepIndicator = document.getElementById('scene1-step-indicator');

        // 元素组
        const magneticGroup = document.getElementById('scene1-magnetic');
        const paramsGroup = document.getElementById('scene1-params');
        const circuitGroup = document.getElementById('scene1-circuit');

        // 更新显示状态
        function updateStep() {
            // 更新步骤指示器
            stepIndicator.textContent = `步骤 ${currentStep}/${totalSteps}`;

            // 更新按钮状态
            btnPrev.disabled = (currentStep === 0);
            btnNext.disabled = (currentStep === totalSteps);

            // 根据步骤显示/隐藏元素组
            if (currentStep >= 1) {
                magneticGroup.classList.add('scene1-visible');
            } else {
                magneticGroup.classList.remove('scene1-visible');
            }

            if (currentStep >= 2) {
                paramsGroup.classList.add('scene1-visible');
            } else {
                paramsGroup.classList.remove('scene1-visible');
            }

            if (currentStep >= 3) {
                circuitGroup.classList.add('scene1-visible');
            } else {
                circuitGroup.classList.remove('scene1-visible');
            }
        }

        // 按钮事件处理
        btnPrev.addEventListener('click', function() {
            if (currentStep > 0) {
                currentStep--;
                updateStep();
            }
        });

        btnNext.addEventListener('click', function() {
            if (currentStep < totalSteps) {
                currentStep++;
                updateStep();
            }
        });

        // 初始化
        updateStep();
    })();
    </script>
