    <script>
    (function() {
        // Scene 2: 导体棒受力分析与匀速条件推导
        let currentStep = 0;
        const totalSteps = 4;

        // 获取所有元素
        const step1Group = document.getElementById('scene2-step1-group');
        const step2Group = document.getElementById('scene2-step2-group');
        const step3Group = document.getElementById('scene2-step3-group');
        const step4Group = document.getElementById('scene2-step4-group');
        const stepIndicator = document.getElementById('scene2-step-indicator');
        const btnPrevBg = document.getElementById('scene2-btn-prev-bg');
        const btnNextBg = document.getElementById('scene2-btn-next-bg');
        const btnPrevText = document.getElementById('scene2-btn-prev-text');
        const btnNextText = document.getElementById('scene2-btn-next-text');

        // 更新显示状态
        function updateDisplay() {
            // 根据当前步骤显示/隐藏各组
            step1Group.style.display = currentStep >= 1 ? 'block' : 'none';
            step2Group.style.display = currentStep >= 2 ? 'block' : 'none';
            step3Group.style.display = currentStep >= 3 ? 'block' : 'none';
            step4Group.style.display = currentStep >= 4 ? 'block' : 'none';

            // 更新步骤指示
            stepIndicator.textContent = '步骤 ' + currentStep + '/' + totalSteps;

            // 更新按钮状态
            if (currentStep === 0) {
                btnPrevBg.setAttribute('fill', '#cccccc');
                btnPrevText.setAttribute('fill', '#666666');
            } else {
                btnPrevBg.setAttribute('fill', '#1565c0');
                btnPrevText.setAttribute('fill', '#ffffff');
            }

            if (currentStep === totalSteps) {
                btnNextBg.setAttribute('fill', '#cccccc');
                btnNextText.setAttribute('fill', '#666666');
            } else {
                btnNextBg.setAttribute('fill', '#1565c0');
                btnNextText.setAttribute('fill', '#ffffff');
            }
        }

        // 按钮点击处理
        btnPrevBg.addEventListener('click', function() {
            if (currentStep > 0) {
                currentStep--;
                updateDisplay();
            }
        });

        btnNextBg.addEventListener('click', function() {
            if (currentStep < totalSteps) {
                currentStep++;
                updateDisplay();
            }
        });

        // 添加按钮悬停效果
        btnPrevBg.style.cursor = 'pointer';
        btnNextBg.style.cursor = 'pointer';

        // 初始化显示
        updateDisplay();
    })();
    </script>
