    <script>
    (function() {
        // Scene 1 实现 - 初始平衡状态受力分析

        // 状态变量
        let scene1CurrentStep = 0;
        const scene1TotalSteps = 4;

        // DOM 元素
        const scene1BtnPrev = document.getElementById('scene1-btn-prev');
        const scene1BtnNext = document.getElementById('scene1-btn-next');
        const scene1StepIndicator = document.getElementById('scene1-step-indicator');
        const scene1AFbdGroup = document.getElementById('scene1-a-fbd-group');
        const scene1BFbdGroup = document.getElementById('scene1-b-fbd-group');
        const scene1EquationGroup = document.getElementById('scene1-equation-group');

        // 更新按钮状态
        function scene1UpdateButtons() {
            // 上一步按钮
            if (scene1CurrentStep === 0) {
                scene1BtnPrev.className = 'scene1-btn scene1-btn-disabled';
                scene1BtnPrev.disabled = true;
            } else {
                scene1BtnPrev.className = 'scene1-btn scene1-btn-enabled';
                scene1BtnPrev.disabled = false;
            }

            // 下一步按钮
            if (scene1CurrentStep >= scene1TotalSteps - 1) {
                scene1BtnNext.className = 'scene1-btn scene1-btn-disabled';
                scene1BtnNext.disabled = true;
            } else {
                scene1BtnNext.className = 'scene1-btn scene1-btn-enabled';
                scene1BtnNext.disabled = false;
            }

            // 更新步骤指示
            scene1StepIndicator.textContent = '步骤 ' + (scene1CurrentStep + 1) + '/' + scene1TotalSteps;
        }

        // 更新显示状态
        function scene1UpdateDisplay() {
            // 步骤 0：只显示装置原图
            // 步骤 1：显示 A 受力图
            // 步骤 2：显示 B 受力图
            // 步骤 3：显示方程

            if (scene1CurrentStep >= 1) {
                scene1AFbdGroup.style.display = 'block';
                scene1AFbdGroup.classList.add('scene1-fade-in');
            } else {
                scene1AFbdGroup.style.display = 'none';
                scene1AFbdGroup.classList.remove('scene1-fade-in');
            }

            if (scene1CurrentStep >= 2) {
                scene1BFbdGroup.style.display = 'block';
                scene1BFbdGroup.classList.add('scene1-fade-in');
            } else {
                scene1BFbdGroup.style.display = 'none';
                scene1BFbdGroup.classList.remove('scene1-fade-in');
            }

            if (scene1CurrentStep >= 3) {
                scene1EquationGroup.style.display = 'block';
                scene1EquationGroup.classList.add('scene1-fade-in');
            } else {
                scene1EquationGroup.style.display = 'none';
                scene1EquationGroup.classList.remove('scene1-fade-in');
            }

            scene1UpdateButtons();
        }

        // 上一步按钮点击
        scene1BtnPrev.addEventListener('click', function() {
            if (scene1CurrentStep > 0) {
                scene1CurrentStep--;
                scene1UpdateDisplay();
            }
        });

        // 下一步按钮点击
        scene1BtnNext.addEventListener('click', function() {
            if (scene1CurrentStep < scene1TotalSteps - 1) {
                scene1CurrentStep++;
                scene1UpdateDisplay();
            }
        });

        // 初始化
        scene1UpdateDisplay();
    })();
    </script>
