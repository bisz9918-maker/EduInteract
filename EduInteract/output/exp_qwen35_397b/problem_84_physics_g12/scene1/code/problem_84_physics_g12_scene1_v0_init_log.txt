    <script>
    (function() {
        // Scene 1 实现：电磁感应装置结构与受力分析
        
        let currentStep = 0;
        const totalSteps = 4;

        // 获取 DOM 元素
        const btnPrev = document.getElementById('scene1-btn-prev-bg');
        const btnNext = document.getElementById('scene1-btn-next-bg');
        const stepIndicator = document.getElementById('scene1-step-indicator');
        
        const forceFGroup = document.getElementById('scene1-force-F-group');
        const forceFaGroup = document.getElementById('scene1-force-Fa-group');
        const forceFGroup_f = document.getElementById('scene1-force-f-group');
        const formulaGroup = document.getElementById('scene1-formula-group');
        
        const forceF = document.getElementById('scene1-force-F');
        const forceFa = document.getElementById('scene1-force-Fa');
        const forceF_f = document.getElementById('scene1-force-f');

        // 更新显示状态
        function updateDisplay() {
            // 更新步骤指示器
            stepIndicator.textContent = `步骤 ${currentStep}/${totalSteps}`;
            
            // 更新按钮状态
            btnPrev.style.opacity = currentStep === 0 ? '0.5' : '1';
            btnNext.style.opacity = currentStep === totalSteps ? '0.5' : '1';
            
            // 控制各步骤元素的显示
            forceFGroup.style.display = currentStep >= 1 ? 'block' : 'none';
            forceFaGroup.style.display = currentStep >= 2 ? 'block' : 'none';
            forceFGroup_f.style.display = currentStep >= 3 ? 'block' : 'none';
            formulaGroup.style.display = currentStep >= 4 ? 'block' : 'none';
            
            // 步骤 4 时高亮三个力箭头
            if (currentStep >= 4) {
                forceF.setAttribute('stroke-width', '7');
                forceFa.setAttribute('stroke-width', '7');
                forceF_f.setAttribute('stroke-width', '7');
            } else {
                forceF.setAttribute('stroke-width', '5');
                forceFa.setAttribute('stroke-width', '5');
                forceF_f.setAttribute('stroke-width', '5');
            }
        }

        // 按钮点击事件
        btnPrev.addEventListener('click', function() {
            if (currentStep > 0) {
                currentStep--;
                updateDisplay();
            }
        });

        btnNext.addEventListener('click', function() {
            if (currentStep < totalSteps) {
                currentStep++;
                updateDisplay();
            }
        });

        // 初始化显示
        updateDisplay();
    })();
    </script>
