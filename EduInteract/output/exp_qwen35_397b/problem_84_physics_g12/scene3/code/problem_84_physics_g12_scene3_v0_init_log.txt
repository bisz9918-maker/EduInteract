    <script>
    (function() {
        // Scene 3: 电磁感应中的能量转化与守恒
        // 布局坐标表（实现对照依据）：
        // | 元素 | 类型 | 位置/范围 | 大小 | 标签偏移方向 | 备注 |
        // | W_F 矩形 | rect | (50,180)-(200,320) | 150×140 | - | 左侧输入区，蓝色 |
        // | W_F 标签 | text | (125,245) | ~60×20 | 居中 | 矩形内部 |
        // | W_F 数值 | text | (125,275) | ~50×16 | 居中 | "8.4J" |
        // | ΔE_k 矩形 | rect | (350,120)-(500,180) | 150×60 | - | 右上，绿色 |
        // | Q_f 矩形 | rect | (350,220)-(500,280) | 150×60 | - | 右中，橙色 |
        // | Q_总 矩形 | rect | (350,320)-(500,380) | 150×60 | - | 右下，红色 |
        // | Q_R 矩形 | rect | (650,280)-(780,340) | 130×60 | - | 最右下，深红 |
        // | Q_r 矩形 | rect | (650,360)-(780,420) | 130×60 | - | 最右下，紫色 |

        let currentStep = 0;
        const totalSteps = 4;

        const titles = [
            "能量输入：外力做功",
            "能量去向：三分支",
            "焦耳热按电阻比例分配",
            "能量守恒验证"
        ];

        const stepIndicator = document.getElementById('scene3-step-indicator');
        const titleEl = document.getElementById('scene3-title');
        const btnPrev = document.getElementById('scene3-btn-prev');
        const btnNext = document.getElementById('scene3-btn-next');

        const step0El = document.getElementById('scene3-step0');
        const step1El = document.getElementById('scene3-step1');
        const step2El = document.getElementById('scene3-step2');
        const step3El = document.getElementById('scene3-step3');

        function updateUI() {
            // 更新标题
            titleEl.textContent = titles[currentStep];

            // 更新步骤指示器
            stepIndicator.textContent = `步骤 ${currentStep + 1}/${totalSteps}`;

            // 更新按钮状态
            btnPrev.disabled = (currentStep === 0);
            btnNext.disabled = (currentStep === totalSteps - 1);

            // 更新各步骤元素可见性
            hideElement(step0El);
            hideElement(step1El);
            hideElement(step2El);
            hideElement(step3El);

            // 根据当前步骤显示相应元素
            if (currentStep >= 0) {
                showElement(step0El);
            }
            if (currentStep >= 1) {
                showElement(step1El);
            }
            if (currentStep >= 2) {
                showElement(step2El);
            }
            if (currentStep >= 3) {
                showElement(step3El);
            }
        }

        function showElement(el) {
            el.style.display = 'block';
            el.style.opacity = '1';
        }

        function hideElement(el) {
            el.style.display = 'none';
            el.style.opacity = '0';
        }

        btnPrev.addEventListener('click', function() {
            if (currentStep > 0) {
                currentStep--;
                updateUI();
            }
        });

        btnNext.addEventListener('click', function() {
            if (currentStep < totalSteps - 1) {
                currentStep++;
                updateUI();
            }
        });

        // 初始化
        updateUI();
    })();
    </script>
