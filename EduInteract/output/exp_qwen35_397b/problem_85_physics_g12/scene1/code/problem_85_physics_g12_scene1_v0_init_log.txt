    <script>
    (function() {
        // Scene 1 实现：双气缸装置结构与初始状态

        // 状态变量
        let currentStep = 0;
        const totalSteps = 5;

        // 每步需要显示的元素 ID 列表
        const stepElements = {
            0: [
                // 基础结构
                'scene1-cylinder-a-left', 'scene1-cylinder-a-right', 'scene1-cylinder-a-top', 'scene1-cylinder-a-bottom',
                'scene1-cylinder-b-left', 'scene1-cylinder-b-right', 'scene1-cylinder-b-bottom',
                'scene1-connect-tube-h', 'scene1-connect-tube-a', 'scene1-connect-tube-b',
                'scene1-label-a', 'scene1-label-b',
                'scene1-resistor'
            ],
            1: [
                // 步骤 0 + 活塞
                'scene1-piston-a', 'scene1-label-piston-a',
                'scene1-piston-b', 'scene1-label-piston-b'
            ],
            2: [
                // 步骤 1 + 气体区域
                'scene1-oxygen-region', 'scene1-label-oxygen',
                'scene1-nitrogen-region', 'scene1-label-nitrogen'
            ],
            3: [
                // 步骤 2 + 尺寸标注
                'scene1-annot-h4-line', 'scene1-annot-h4-arrow1', 'scene1-annot-h4-arrow2', 'scene1-annot-h4-arrow3', 'scene1-annot-h4-arrow4', 'scene1-annot-h4',
                'scene1-annot-h-line', 'scene1-annot-h-arrow1', 'scene1-annot-h-arrow2', 'scene1-annot-h-arrow3', 'scene1-annot-h-arrow4', 'scene1-annot-h',
                'scene1-annot-h2-line', 'scene1-annot-h2-arrow1', 'scene1-annot-h2-arrow2', 'scene1-annot-h2-arrow3', 'scene1-annot-h2-arrow4', 'scene1-annot-h2',
                'scene1-label-sa'
            ],
            4: [
                // 步骤 3 + 状态参量
                'scene1-label-t1', 'scene1-label-p0'
            ]
        };

        // 工具函数：显示元素
        function showElement(id) {
            const el = document.getElementById(id);
            if (el) {
                el.classList.remove('scene1-hidden');
                el.classList.add('scene1-visible');
            }
        }

        // 工具函数：隐藏元素
        function hideElement(id) {
            const el = document.getElementById(id);
            if (el) {
                el.classList.remove('scene1-visible');
                el.classList.add('scene1-hidden');
            }
        }

        // 更新显示状态
        function updateDisplay() {
            // 隐藏所有元素
            for (let step = 0; step < totalSteps; step++) {
                stepElements[step].forEach(id => hideElement(id));
            }

            // 显示当前步骤及之前步骤的元素
            for (let step = 0; step <= currentStep; step++) {
                stepElements[step].forEach(id => showElement(id));
            }

            // 更新按钮状态
            const prevBtn = document.getElementById('scene1-btn-prev');
            const nextBtn = document.getElementById('scene1-btn-next');

            prevBtn.disabled = (currentStep === 0);

            if (currentStep === totalSteps - 1) {
                nextBtn.textContent = '重新开始';
            } else {
                nextBtn.textContent = '下一步 ▶';
            }

            // 触发 MathJax 重新渲染
            if (window.MathJax) {
                MathJax.typesetPromise();
            }
        }

        // 按钮事件处理
        document.getElementById('scene1-btn-prev').addEventListener('click', function() {
            if (currentStep > 0) {
                currentStep--;
                updateDisplay();
            }
        });

        document.getElementById('scene1-btn-next').addEventListener('click', function() {
            if (currentStep < totalSteps - 1) {
                currentStep++;
                updateDisplay();
            } else {
                // 重新开始
                currentStep = 0;
                updateDisplay();
            }
        });

        // 初始化显示
        updateDisplay();
    })();
    </script>
