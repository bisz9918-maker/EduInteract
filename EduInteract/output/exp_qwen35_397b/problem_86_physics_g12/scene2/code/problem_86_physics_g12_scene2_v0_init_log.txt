    <script>
    (function() {
        // Scene 2 实现
        // 布局坐标表（实现对照依据）：
        // | 元素 | 位置/范围 | 备注 |
        // | B 板 | (180,160)-(680,160) | 上极板，接地 |
        // | A 板 | (180,340)-(680,340) | 下极板 |
        // | 入射点 P | (180,250) | 两板中央 |
        // | 轨迹 1 | (180,250)-(680,340) | 红色，向下偏转 |
        // | 轨迹 2 | (180,250)-(680,160) | 绿色，向上偏转 |
        // | 电场线 | x=300,430,560 | 蓝色虚线 |
        // | mg 箭头 | (200,260)-(200,290) | 向下 |
        // | qE 箭头 | (220,260)-(220,230) | 向上 |
        // | v₀ 箭头 | (190,245)-(230,245) | 向右 |
        
        let currentState = 0;
        const totalSteps = 4;
        
        // 获取 DOM 元素
        const btnPrev = document.getElementById('scene2-btn-prev');
        const btnNext = document.getElementById('scene2-btn-next');
        const stepIndicator = document.getElementById('scene2-step-indicator');
        
        // 元素组
        const eFieldGroup = document.getElementById('scene2-e-field-group');
        const forceGroup = document.getElementById('scene2-force-group');
        const traj1 = document.getElementById('scene2-traj1');
        const traj2 = document.getElementById('scene2-traj2');
        const feasibleArea1 = document.getElementById('scene2-feasible-area');
        const feasibleArea2 = document.getElementById('scene2-feasible-area-2');
        const labelV0 = document.getElementById('scene2-label-v0');
        const labelMg = document.getElementById('scene2-label-mg');
        const labelQe = document.getElementById('scene2-label-qe');
        const labelTraj1 = document.getElementById('scene2-label-traj1');
        const labelTraj2 = document.getElementById('scene2-label-traj2');
        const conclusionBox = document.getElementById('scene2-conclusion-box');
        
        // 工具函数：显示元素
        function show(el) {
            if (el) {
                el.classList.remove('scene2-hidden');
                el.classList.add('scene2-visible');
                el.style.display = '';
                el.style.opacity = '1';
            }
        }
        
        // 工具函数：隐藏元素
        function hide(el) {
            if (el) {
                el.classList.remove('scene2-visible');
                el.classList.add('scene2-hidden');
                el.style.display = 'none';
                el.style.opacity = '0';
            }
        }
        
        // 更新状态显示
        function updateState() {
            // 更新步骤指示器
            stepIndicator.textContent = `步骤 ${currentState}/${totalSteps}`;
            
            // 更新按钮状态
            btnPrev.disabled = (currentState === 0);
            btnNext.disabled = (currentState === totalSteps);
            btnNext.textContent = (currentState === totalSteps) ? '完成' : '下一步 ▶';
            
            // 根据状态显示/隐藏元素
            hideAll();
            
            if (currentState >= 0) {
                // 始终显示：极板、接地、入射点、基本标签
                show(document.getElementById('scene2-plate-a'));
                show(document.getElementById('scene2-plate-b'));
                show(document.getElementById('scene2-ground'));
                show(document.getElementById('scene2-entry-point'));
                show(document.getElementById('scene2-label-a'));
                show(document.getElementById('scene2-label-b'));
                show(document.getElementById('scene2-label-l-group'));
                show(document.getElementById('scene2-label-d-group'));
                show(document.getElementById('scene2-label-phi-a'));
            }
            
            if (currentState >= 1) {
                // 步骤 1：显示电场线和受力箭头
                show(eFieldGroup);
                show(forceGroup);
                show(labelV0);
                show(labelMg);
                show(labelQe);
            }
            
            if (currentState >= 2) {
                // 步骤 2：显示轨迹 1
                show(traj1);
                show(labelTraj1);
                hide(traj2);
                hide(labelTraj2);
                hide(feasibleArea1);
                hide(feasibleArea2);
                hide(conclusionBox);
            }
            
            if (currentState >= 3) {
                // 步骤 3：显示轨迹 2（隐藏轨迹 1）
                hide(traj1);
                hide(labelTraj1);
                show(traj2);
                show(labelTraj2);
                hide(feasibleArea1);
                hide(feasibleArea2);
                hide(conclusionBox);
            }
            
            if (currentState >= 4) {
                // 步骤 4：同时显示两条轨迹和结论
                show(traj1);
                show(labelTraj1);
                show(traj2);
                show(labelTraj2);
                show(feasibleArea1);
                show(feasibleArea2);
                show(conclusionBox);
            }
        }
        
        // 隐藏所有动态元素
        function hideAll() {
            hide(eFieldGroup);
            hide(forceGroup);
            hide(traj1);
            hide(traj2);
            hide(feasibleArea1);
            hide(feasibleArea2);
            hide(labelV0);
            hide(labelMg);
            hide(labelQe);
            hide(labelTraj1);
            hide(labelTraj2);
            hide(conclusionBox);
        }
        
        // 事件绑定
        btnPrev.addEventListener('click', function() {
            if (currentState > 0) {
                currentState--;
                updateState();
            }
        });
        
        btnNext.addEventListener('click', function() {
            if (currentState < totalSteps) {
                currentState++;
                updateState();
            }
        });
        
        // 初始化
        updateState();
        
    })();
    </script>
