    <script>
    (function() {
        // Scene 3: 带电粒子在电场中的运动 - 参数探究
        
        // 物理常量 (归一化)
        const q_m = 1;  // 比荷 q/m = 1 C/kg
        const L = 0.4;  // 虚线 ab 距 y 轴距离 (m)
        
        // 参数范围
        const E1_MIN = 50, E1_MAX = 200;
        const E2_MIN = 50, E2_MAX = 250;
        const h_MIN = 0.2, h_MAX = 1.0;
        
        // 当前参数值
        let state = {
            E1: 100,
            E2: 150,
            h: 0.5
        };
        
        // 坐标映射函数
        function physToCanvasX(x) {
            // 物理 x (m) -> 画布 x (px)
            // x=0 -> 80px, x=1.2 -> 650px (比例尺)
            return 80 + x * 475;  // 475 px/m
        }
        
        function physToCanvasY(y) {
            // 物理 y (m) -> 画布 y (px)
            // y=0 -> 380px, y=1.0 -> 80px
            return 380 - y * 300;  // 300 px/m
        }
        
        // 计算轨迹
        function calculateTrajectory() {
            const { E1, E2, h } = state;
            
            // 在 E1 区域：粒子从 P(0,h) 水平加速到 (L, h)
            // v1 = sqrt(2 * q/m * E1 * L) = sqrt(2 * E1 * L) (因为 q/m=1)
            const v1 = Math.sqrt(2 * E1 * L);
            
            // 在 E2 区域：类平抛运动
            // 加速度 a_y = q*E2/m = E2 (因为 q/m=1)
            // 运动时间 t2 = sqrt(2*h/a_y) = sqrt(2*h/E2)
            const t2 = Math.sqrt(2 * h / E2);
            
            // Q 点 x 坐标：x_Q = L + v1 * t2
            const xQ = L + v1 * t2;
            
            return { v1, t2, xQ };
        }
        
        // 生成轨迹路径
        function generateTrajectoryPath() {
            const { E1, E2, h } = state;
            const { v1, t2, xQ } = calculateTrajectory();
            
            // E1 区域轨迹：从 P(0,h) 到 (L, h) 的水平线段
            const p1_start = { x: 0, y: h };
            const p1_end = { x: L, y: h };
            
            // E2 区域轨迹：抛物线
            // 参数方程：x(t) = L + v1*t, y(t) = h - 0.5*E2*t^2
            // 从 t=0 到 t=t2
            const numPoints = 50;
            let pathD = `M ${physToCanvasX(p1_start.x)} ${physToCanvasY(p1_start.y)}`;
            
            // E1 区域线段
            pathD += ` L ${physToCanvasX(p1_end.x)} ${physToCanvasY(p1_end.y)}`;
            
            // E2 区域抛物线
            for (let i = 1; i <= numPoints; i++) {
                const t = (i / numPoints) * t2;
                const x = L + v1 * t;
                const y = h - 0.5 * E2 * t * t;
                pathD += ` L ${physToCanvasX(x)} ${physToCanvasY(y)}`;
            }
            
            return pathD;
        }
        
        // 更新图示
        function updateDisplay() {
            const { E1, E2, h } = state;
            const { xQ } = calculateTrajectory();
            
            // 更新 P 点位置
            const pCanvasY = physToCanvasY(h);
            document.getElementById('scene3-point-p').setAttribute('cy', pCanvasY);
            document.getElementById('scene3-label-p').setAttribute('y', pCanvasY + 5);
            
            // 更新 h 标注线
            document.getElementById('scene3-h-line').setAttribute('y2', pCanvasY);
            document.getElementById('scene3-label-h').setAttribute('y', (380 + pCanvasY) / 2);
            
            // 更新轨迹
            const pathD = generateTrajectoryPath();
            document.getElementById('scene3-trajectory').setAttribute('d', pathD);
            
            // 更新 Q 点位置
            const qCanvasX = physToCanvasX(xQ);
            document.getElementById('scene3-point-q').setAttribute('cx', qCanvasX);
            document.getElementById('scene3-label-q').setAttribute('x', qCanvasX);
            
            // 更新坐标显示
            document.getElementById('scene3-q-coord').textContent = `Q 点：(${xQ.toFixed(2)}, 0) m`;
            
            // 更新滑块标签
            document.getElementById('scene3-slider-e1-value').textContent = `E₁ = ${E1} N/C`;
            document.getElementById('scene3-slider-e2-value').textContent = `E₂ = ${E2} N/C`;
            document.getElementById('scene3-slider-h-value').textContent = `h = ${h.toFixed(2)} m`;
            
            // 更新滑块手柄位置
            const e1HandleX = 50 + ((E1 - E1_MIN) / (E1_MAX - E1_MIN)) * 160;
            document.getElementById('scene3-slider-e1-handle').setAttribute('x', e1HandleX - 10);
            
            const e2HandleX = 280 + ((E2 - E2_MIN) / (E2_MAX - E2_MIN)) * 160;
            document.getElementById('scene3-slider-e2-handle').setAttribute('x', e2HandleX - 10);
            
            const hHandleX = 510 + ((h - h_MIN) / (h_MAX - h_MIN)) * 160;
            document.getElementById('scene3-slider-h-handle').setAttribute('x', hHandleX - 10);
        }
        
        // 滑块拖拽处理
        let draggingSlider = null;
        
        function setupSlider(sliderName, trackId, handleId, minVal, maxVal, paramName, displayFormat) {
            const track = document.getElementById(trackId);
            const handle = document.getElementById(handleId);
            const trackRect = track.getBoundingClientRect();
            const trackLeft = trackRect.left;
            const trackWidth = trackRect.width;
            
            function updateSlider(clientX) {
                let handleX = clientX - trackLeft;
                handleX = Math.max(10, Math.min(trackWidth - 10, handleX));
                
                const ratio = (handleX - 10) / (trackWidth - 20);
                const value = minVal + ratio * (maxVal - minVal);
                
                state[paramName] = value;
                handle.setAttribute('x', handleX - 10);
                updateDisplay();
            }
            
            handle.addEventListener('mousedown', (e) => {
                draggingSlider = { name: sliderName, update: updateSlider };
                e.preventDefault();
            });
            
            handle.addEventListener('touchstart', (e) => {
                draggingSlider = { name: sliderName, update: updateSlider };
                e.preventDefault();
            });
        }
        
        // 全局鼠标/触摸事件
        document.addEventListener('mousemove', (e) => {
            if (draggingSlider) {
                draggingSlider.update(e.clientX);
            }
        });
        
        document.addEventListener('touchmove', (e) => {
            if (draggingSlider) {
                draggingSlider.update(e.touches[0].clientX);
            }
        });
        
        document.addEventListener('mouseup', () => {
            draggingSlider = null;
        });
        
        document.addEventListener('touchend', () => {
            draggingSlider = null;
        });
        
        // 滑块悬停效果
        ['e1', 'e2', 'h'].forEach(name => {
            const handle = document.getElementById(`scene3-slider-${name}-handle`);
            handle.addEventListener('mouseenter', () => {
                handle.setAttribute('fill', '#1b5e20');
            });
            handle.addEventListener('mouseleave', () => {
                handle.setAttribute('fill', '#2e7d32');
            });
        });
        
        // 初始化滑块
        setupSlider('e1', 'scene3-slider-e1-track', 'scene3-slider-e1-handle', E1_MIN, E1_MAX, 'E1', 'E₁ = {} N/C');
        setupSlider('e2', 'scene3-slider-e2-track', 'scene3-slider-e2-handle', E2_MIN, E2_MAX, 'E2', 'E₂ = {} N/C');
        setupSlider('h', 'scene3-slider-h-track', 'scene3-slider-h-handle', h_MIN, h_MAX, 'h', 'h = {} m');
        
        // 初始显示
        updateDisplay();
        
    })();
    </script>
