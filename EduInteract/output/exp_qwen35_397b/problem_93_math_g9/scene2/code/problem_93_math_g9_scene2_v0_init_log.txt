    <script>
    (function() {
        // Scene 2 实现 - C为AB中点时的几何关系
        
        const currentState = { value: 0 };
        const TOTAL_STEPS = 4;
        
        // 元素可见性映射表
        const stepElements = {
            0: ['scene2-ab', 'scene2-point-a', 'scene2-point-b', 'scene2-point-c', 
                'scene2-label-a', 'scene2-label-b', 'scene2-label-c',
                'scene2-ab-label', 'scene2-ac-label'],
            1: ['scene2-triangle-acd', 'scene2-triangle-bce', 'scene2-point-d', 'scene2-point-e',
                'scene2-label-d', 'scene2-label-e', 'scene2-de', 'scene2-point-p', 'scene2-label-p',
                'scene2-cp', 'scene2-de-parallel', 'scene2-cp-perp'],
            2: ['scene2-cp-calc', 'scene2-cp-label'],
            3: ['scene2-point-f', 'scene2-label-f', 'scene2-af-ag', 'scene2-af-gh', 'scene2-af-hf',
                'scene2-point-g', 'scene2-point-h', 'scene2-label-g', 'scene2-label-h'],
            4: ['scene2-ratio-label']
        };
        
        // 显示元素函数
        function showElement(id) {
            const el = document.getElementById(id);
            if (el) {
                el.style.opacity = '1';
                el.style.display = '';
            }
        }
        
        // 隐藏元素函数
        function hideElement(id) {
            const el = document.getElementById(id);
            if (el) {
                el.style.opacity = '0';
                el.style.display = '';
            }
        }
        
        // 更新视图
        function updateView() {
            // 累积显示所有步骤的元素
            for (let step = 0; step <= currentState.value; step++) {
                const elements = stepElements[step] || [];
                elements.forEach(id => showElement(id));
            }
            
            // 隐藏未到达步骤的元素
            for (let step = currentState.value + 1; step <= TOTAL_STEPS; step++) {
                const elements = stepElements[step] || [];
                elements.forEach(id => hideElement(id));
            }
            
            // 更新步骤指示器
            document.getElementById('scene2-step-indicator').textContent = `步骤 ${currentState.value}/${TOTAL_STEPS}`;
            
            // 更新按钮状态
            document.getElementById('scene2-prev-btn').disabled = currentState.value === 0;
            document.getElementById('scene2-next-btn').disabled = currentState.value === TOTAL_STEPS;
        }
        
        // 初始化
        function init() {
            // 初始隐藏所有元素
            const allElements = [];
            Object.values(stepElements).forEach(arr => allElements.push(...arr));
            allElements.forEach(id => hideElement(id));
            
            // 显示初始状态
            updateView();
            
            // 绑定按钮事件
            document.getElementById('scene2-prev-btn').addEventListener('click', () => {
                if (currentState.value > 0) {
                    currentState.value--;
                    updateView();
                }
            });
            
            document.getElementById('scene2-next-btn').addEventListener('click', () => {
                if (currentState.value < TOTAL_STEPS) {
                    currentState.value++;
                    updateView();
                }
            });
        }
        
        // 启动
        init();
    })();
    </script>
