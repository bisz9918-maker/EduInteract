    <script>
    (function() {
        // 状态变量
        let currentStep = 0;
        const totalSteps = 4;

        // 获取 DOM 元素
        const btnPrev = document.getElementById('scene1-btn-prev');
        const btnNext = document.getElementById('scene1-btn-next');
        const stepIndicator = document.getElementById('scene1-step-indicator');

        // 立体三棱柱元素
        const prismTop = document.getElementById('scene1-prism-top');
        const prismBottom = document.getElementById('scene1-prism-bottom');
        const prismSide1 = document.getElementById('scene1-prism-side1');
        const prismSide2 = document.getElementById('scene1-prism-side2');
        const prismSide3 = document.getElementById('scene1-prism-side3');
        const labelTop = document.getElementById('scene1-label-top');
        const labelBottom = document.getElementById('scene1-label-bottom');
        const labelSide = document.getElementById('scene1-label-side');

        // 展开区元素
        const unfoldRect1 = document.getElementById('scene1-unfold-rect1');
        const unfoldRect2 = document.getElementById('scene1-unfold-rect2');
        const unfoldRect3 = document.getElementById('scene1-unfold-rect3');
        const unfoldTopTri = document.getElementById('scene1-unfold-top-tri');
        const unfoldBottomTri = document.getElementById('scene1-unfold-bottom-tri');
        const label1 = document.getElementById('scene1-label-1');
        const label2 = document.getElementById('scene1-label-2');
        const label3 = document.getElementById('scene1-label-3');

        // 箭头
        const arrow1 = document.getElementById('scene1-arrow1');
        const arrow2 = document.getElementById('scene1-arrow2');
        const arrow3 = document.getElementById('scene1-arrow3');

        // 选项 A 和结论
        const optionA = document.getElementById('scene1-option-a');
        const conclusion = document.getElementById('scene1-conclusion');

        // 辅助函数：显示元素
        function show(el) {
            el.style.display = '';
            el.style.opacity = '1';
        }

        // 辅助函数：隐藏元素
        function hide(el) {
            el.style.display = 'none';
            el.style.opacity = '0';
        }

        // 更新场景状态
        function updateScene() {
            // 更新步骤指示器
            stepIndicator.textContent = '步骤 ' + currentStep + '/' + totalSteps;

            // 更新按钮状态
            btnPrev.disabled = currentStep === 0;
            btnNext.disabled = currentStep === totalSteps;

            // 步骤 0：只显示立体三棱柱
            if (currentStep === 0) {
                show(prismTop);
                show(prismBottom);
                show(prismSide1);
                show(prismSide2);
                show(prismSide3);
                show(labelTop);
                show(labelBottom);
                show(labelSide);
                hide(arrow1);
                hide(arrow2);
                hide(arrow3);
                hide(unfoldRect1);
                hide(unfoldRect2);
                hide(unfoldRect3);
                hide(unfoldTopTri);
                hide(unfoldBottomTri);
                hide(label1);
                hide(label2);
                hide(label3);
                hide(optionA);
                hide(conclusion);
            }
            // 步骤 1：侧面展开
            else if (currentStep === 1) {
                prismTop.style.opacity = '0.5';
                prismBottom.style.opacity = '0.5';
                prismSide1.style.opacity = '0.5';
                prismSide2.style.opacity = '0.5';
                prismSide3.style.opacity = '0.5';
                show(labelTop);
                show(labelBottom);
                show(labelSide);
                show(arrow1);
                hide(arrow2);
                hide(arrow3);
                show(unfoldRect1);
                show(unfoldRect2);
                show(unfoldRect3);
                hide(unfoldTopTri);
                hide(unfoldBottomTri);
                show(label1);
                show(label2);
                show(label3);
                hide(optionA);
                hide(conclusion);
            }
            // 步骤 2：上底面展开
            else if (currentStep === 2) {
                prismTop.style.opacity = '0.3';
                prismBottom.style.opacity = '0.5';
                prismSide1.style.opacity = '0.5';
                prismSide2.style.opacity = '0.5';
                prismSide3.style.opacity = '0.5';
                show(labelTop);
                show(labelBottom);
                show(labelSide);
                show(arrow1);
                show(arrow2);
                hide(arrow3);
                show(unfoldRect1);
                show(unfoldRect2);
                show(unfoldRect3);
                show(unfoldTopTri);
                unfoldTopTri.classList.add('scene1-highlight');
                hide(unfoldBottomTri);
                show(label1);
                show(label2);
                show(label3);
                hide(optionA);
                hide(conclusion);
            }
            // 步骤 3：下底面展开
            else if (currentStep === 3) {
                prismTop.style.opacity = '0.3';
                prismBottom.style.opacity = '0.3';
                prismSide1.style.opacity = '0.5';
                prismSide2.style.opacity = '0.5';
                prismSide3.style.opacity = '0.5';
                show(labelTop);
                show(labelBottom);
                show(labelSide);
                show(arrow1);
                show(arrow2);
                show(arrow3);
                show(unfoldRect1);
                show(unfoldRect2);
                show(unfoldRect3);
                show(unfoldTopTri);
                unfoldTopTri.classList.add('scene1-highlight');
                show(unfoldBottomTri);
                unfoldBottomTri.classList.add('scene1-highlight');
                show(label1);
                show(label2);
                show(label3);
                hide(optionA);
                hide(conclusion);
            }
            // 步骤 4：完整展开图，与选项 A 对比
            else if (currentStep === 4) {
                hide(prismTop);
                hide(prismBottom);
                hide(prismSide1);
                hide(prismSide2);
                hide(prismSide3);
                hide(labelTop);
                hide(labelBottom);
                hide(labelSide);
                show(arrow1);
                show(arrow2);
                show(arrow3);
                show(unfoldRect1);
                show(unfoldRect2);
                show(unfoldRect3);
                show(unfoldTopTri);
                unfoldTopTri.classList.add('scene1-highlight');
                show(unfoldBottomTri);
                unfoldBottomTri.classList.add('scene1-highlight');
                show(label1);
                show(label2);
                show(label3);
                show(optionA);
                show(conclusion);
            }
        }

        // 事件监听
        btnPrev.addEventListener('click', function() {
            if (currentStep > 0) {
                currentStep--;
                updateScene();
            }
        });

        btnNext.addEventListener('click', function() {
            if (currentStep < totalSteps) {
                currentStep++;
                updateScene();
            }
        });

        // 初始化
        updateScene();
    })();
    </script>
