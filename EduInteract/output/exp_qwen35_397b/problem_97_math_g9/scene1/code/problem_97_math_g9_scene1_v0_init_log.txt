    <script>
    (function() {
        // Scene 1: 二次函数图像分析
        // 布局坐标表参考:
        // | 元素 | 类型 | 位置/范围 | 备注 |
        // | x 轴 | 线段 | (100,380)-(700,380) | 水平轴 |
        // | y 轴 | 线段 | (400,80)-(400,450) | 垂直轴 |
        // | 原点 O | text | (385,395) | 坐标轴交点左下 |
        // | x 标签 | text | (690,365) | x 轴末端 |
        // | y 标签 | text | (385,70) | y 轴顶端 |
        // | 抛物线 | path | 顶点 (280,150)，过 (400,280) | 开口向下 |
        // | 与 y 轴交点 | 圆点 | (400,280) | 实心圆 |
        // | 交点标注"1" | text | (415,275) | 右侧 |
        // | 对称轴虚线 | 虚线 | (280,80)-(280,380) | x=280 垂直虚线 |
        // | 对称轴标签 | text | (150,100) | 左侧 |
        // | 开口箭头 | 箭头 | (280,130) 向下 | 顶点下方 |
        // | a<0 标注框 | 矩形+text | (520,110) | 绿色半透明 |
        // | b<0 标注框 | 矩形+text | (520,160) | 绿色半透明 |
        // | 结论框 | 矩形+text | (500,220) | 绿色边框 |
        
        let currentStep = 0;
        const totalSteps = 3;
        
        // 获取 DOM 元素
        const btnPrev = document.getElementById('scene1-btn-prev');
        const btnNext = document.getElementById('scene1-btn-next');
        const stepIndicator = document.getElementById('scene1-step-indicator');
        
        // 分组元素
        const arrowGroup = document.getElementById('scene1-arrow-group');
        const symaxisGroup = document.getElementById('scene1-symaxis-group');
        const conclusionGroup = document.getElementById('scene1-conclusion-group');
        
        // 更新视图函数
        function updateView() {
            // 更新步骤指示器
            stepIndicator.textContent = '步骤 ' + currentStep + '/' + totalSteps;
            
            // 更新按钮状态
            btnPrev.disabled = (currentStep === 0);
            btnNext.disabled = (currentStep === totalSteps);
            
            // 根据当前步骤显示/隐藏元素
            if (currentStep >= 1) {
                arrowGroup.style.display = 'block';
            } else {
                arrowGroup.style.display = 'none';
            }
            
            if (currentStep >= 2) {
                symaxisGroup.style.display = 'block';
            } else {
                symaxisGroup.style.display = 'none';
            }
            
            if (currentStep >= 3) {
                conclusionGroup.style.display = 'block';
            } else {
                conclusionGroup.style.display = 'none';
            }
        }
        
        // 上一步按钮点击事件
        btnPrev.addEventListener('click', function() {
            if (currentStep > 0) {
                currentStep--;
                updateView();
            }
        });
        
        // 下一步按钮点击事件
        btnNext.addEventListener('click', function() {
            if (currentStep < totalSteps) {
                currentStep++;
                updateView();
            }
        });
        
        // 初始化视图
        updateView();
    })();
    </script>
