                  stroke="#c62828" stroke-width="2"/>
            
            <!-- DE 线段（蓝色，折叠后显示） -->
            <line id="scene2-line-de" x1="333" y1="400" x2="307" y2="215" 
                  stroke="#1565c0" stroke-width="2" class="scene2-hidden"/>
            
            <!-- DF 线段（红色，折叠后显示） -->
            <line id="scene2-line-df" x1="333" y1="400" x2="517" y2="256"

    <script>
    (function() {
        // 坐标常量
        const POINTS = {
            A: { x: 200, y: 400 },
            B: { x: 600, y: 400 },
            C: { x: 400, y: 54 },
            D: { x: 333, y: 400 },
            E: { x: 307, y: 215 },
            F: { x: 517, y: 256 },
            M: { x: 367, y: 227 }
        };
        
        // 状态管理
        let currentState = 0;
        const TOTAL_STEPS = 5;
        let isAnimating = false;
        
        // 获取元素
        const elements = {
            lineCE: document.getElementById('scene2-line-ce'),
            lineCF: document.getElementById('scene2-line-cf'),
            lineDE: document.getElementById('scene2-line-de'),
            lineDF: document.getElementById('scene2-line-df'),
            lineEF: document.getElementById('scene2-line-ef'),
            lineCD: document.getElementById('scene2-line-cd'),
            foldTrace: document.getElementById('scene2-fold-trace'),
            rightAngle: document.getElementById('scene2-right-angle'),
            labelC: document.getElementById('scene2-label-c'),
            labelM: document.getElementById('scene2-label-m'),
            labelCeDe: document.getElementById('scene2-label-ce-de'),
            labelCfDf: document.getElementById('scene2-label-cf-df'),
            labelPerp: document.getElementById('scene2-label-perp'),
            pointC: document.getElementById('scene2-point-c'),
            prevBtn: document.getElementById('scene2-prev-btn'),
            nextBtn: document.getElementById('scene2-next-btn'),
            stepIndicator: document.getElementById('scene2-step-indicator')
        };
        
        // 显示/隐藏元素
        function show(el) {
            if (el) {
                el.classList.remove('scene2-hidden');
                el.classList.add('scene2-visible');
                el.style.display = '';
                el.style.opacity = '1';
            }
        }
        
        function hide(el) {
            if (el) {
                el.classList.remove('scene2-visible');
                el.classList.add('scene2-hidden');
                el.style.display = 'none';
                el.style.opacity = '0';
            }
        }
        
        // 更新按钮状态
        function updateButtons() {
            elements.prevBtn.disabled = currentState === 0;
            elements.nextBtn.disabled = currentState === TOTAL_STEPS;
            elements.stepIndicator.textContent = `步骤 ${currentState}/${TOTAL_STEPS}`;
        }
        
        // 重置所有高亮
        function resetHighlights() {
            ['scene2-line-ce', 'scene2-line-de', 'scene2-line-cf', 'scene2-line-df'].forEach(id => {
                const el = document.getElementById(id);
                if (el) {
                    el.classList.remove('scene2-highlight');
                    el.style.strokeWidth = '2';
                }
            });
            ['scene2-label-ce-de', 'scene2-label-cf-df'].forEach(id => {
                const el = document.getElementById(id);
                if (el) {
                    el.classList.remove('scene2-label-highlight');
                }
            });
        }
        
        // 设置各步骤状态
        function setStep(step) {
            resetHighlights();
            
            // 步骤 0：初始状态
            if (step === 0) {
                show(elements.lineCE);
                show(elements.lineCF);
                hide(elements.lineDE);
                hide(elements.lineDF);
                hide(elements.lineCD);
                hide(elements.foldTrace);
                hide(elements.rightAngle);
                show(elements.labelC);
                hide(elements.labelM);
                hide(elements.labelCeDe);
                hide(elements.labelCfDf);
                hide(elements.labelPerp);
                show(elements.pointC);
                elements.pointC.setAttribute('cx', POINTS.C.x);
                elements.pointC.setAttribute('cy', POINTS.C.y);
            }
            // 步骤 1：开始折叠（C 点移动 50%）
            else if (step === 1) {
                show(elements.lineCE);
                show(elements.lineCF);
                hide(elements.lineDE);
                hide(elements.lineDF);
                hide(elements.lineCD);
                show(elements.foldTrace);
                elements.foldTrace.setAttribute('opacity', '0.1');
                hide(elements.rightAngle);
                show(elements.labelC);
                hide(elements.labelM);
                hide(elements.labelCeDe);
                hide(elements.labelCfDf);
                hide(elements.labelPerp);
                show(elements.pointC);
                // C 点移动到中间位置
                const midX = (POINTS.C.x + POINTS.D.x) / 2;
                const midY = (POINTS.C.y + POINTS.D.y) / 2;
                elements.pointC.setAttribute('cx', midX);
                elements.pointC.setAttribute('cy', midY);
            }
            // 步骤 2：折叠完成
            else if (step === 2) {
                hide(elements.lineCE);
                hide(elements.lineCF);
                show(elements.lineDE);
                show(elements.lineDF);
                hide(elements.lineCD);
                hide(elements.foldTrace);
                hide(elements.rightAngle);
                hide(elements.labelC);
                hide(elements.labelM);
                show(elements.labelCeDe);
                show(elements.labelCfDf);
                hide(elements.labelPerp);
                hide(elements.pointC);
            }
            // 步骤 3：高亮 CE=DE
            else if (step === 3) {
                hide(elements.lineCE);
                hide(elements.lineCF);
                show(elements.lineDE);
                show(elements.lineDF);
                hide(elements.lineCD);
                hide(elements.foldTrace);
                hide(elements.rightAngle);
                hide(elements.labelC);
                hide(elements.labelM);
                show(elements.labelCeDe);
                show(elements.labelCfDf);
                hide(elements.labelPerp);
                hide(elements.pointC);
                // 高亮 CE=DE
                document.getElementById('scene2-line-de').classList.add('scene2-highlight');
                document.getElementById('scene2-label-ce-de').classList.add('scene2-label-highlight');
            }
            // 步骤 4：高亮 CF=DF
            else if (step === 4) {
                hide(elements.lineCE);
                hide(elements.lineCF);
                show(elements.lineDE);
                show(elements.lineDF);
                hide(elements.lineCD);
                hide(elements.foldTrace);
                hide(elements.rightAngle);
                hide(elements.labelC);
                hide(elements.labelM);
                show(elements.labelCeDe);
                show(elements.labelCfDf);
                hide(elements.labelPerp);
                hide(elements.pointC);
                // 高亮 CF=DF
                document.getElementById('scene2-line-df').classList.add('scene2-highlight');
                document.getElementById('scene2-label-cf-df').classList.add('scene2-label-highlight');
            }
            // 步骤 5：显示垂直平分
            else if (step === 5) {
                hide(elements.lineCE);
                hide(elements.lineCF);
                show(elements.lineDE);
                show(elements.lineDF);
                show(elements.lineCD);
                hide(elements.foldTrace);
                show(elements.rightAngle);
                hide(elements.labelC);
                show(elements.labelM);
                show(elements.labelCeDe);
                show(elements.labelCfDf);
                show(elements.labelPerp);
                hide(elements.pointC);
            }
            
            updateButtons();
        }
        
        // 下一步
        function stepForward() {
            if (isAnimating || currentState >= TOTAL_STEPS) return;
            isAnimating = true;
            currentState++;
            
            if (currentState === 1) {
                // 动画：C 点移动到中间
                const duration = 800;
                const startTime = performance.now();
                const startX = POINTS.C.x, startY = POINTS.C.y;
                const endX = (POINTS.C.x + POINTS.D.x) / 2;
                const endY = (POINTS.C.y + POINTS.D.y) / 2;
                
                function animate(currentTime) {
                    const elapsed = currentTime - startTime;
                    const progress = Math.min(elapsed / duration, 1);
                    // easeInOut 二次曲线
                    const eased = progress < 0.5 ? 2 * progress * progress : 1 - Math.pow(-2 * progress + 2, 2) / 2;
                    
                    elements.pointC.setAttribute('cx', startX + (endX - startX) * eased);
                    elements.pointC.setAttribute('cy', startY + (endY - startY) * eased);
                    elements.foldTrace.setAttribute('opacity', 0.1 * eased);
                    
                    if (progress < 1) {
                        requestAnimationFrame(animate);
                    } else {
                        setStep(currentState);
                        isAnimating = false;
                    }
                }
                requestAnimationFrame(animate);
            } else {
                setStep(currentState);
                isAnimating = false;
            }
        }
        
        // 上一步
        function stepBackward() {
            if (isAnimating || currentState <= 0) return;
            isAnimating = true;
            currentState--;
            
            if (currentState === 0) {
                // 动画：C 点移回原位
                const duration = 800;
                const startTime = performance.now();
                const startX = (POINTS.C.x + POINTS.D.x) / 2;
                const startY = (POINTS.C.y + POINTS.D.y) / 2;
                const endX = POINTS.C.x, endY = POINTS.C.y;
                
                function animate(currentTime) {
                    const elapsed = currentTime - startTime;
                    const progress = Math.min(elapsed / duration, 1);
                    const eased = progress < 0.5 ? 2 * progress * progress : 1 - Math.pow(-2 * progress + 2, 2) / 2;
                    
                    elements.pointC.setAttribute('cx', startX + (endX - startX) * eased);
                    elements.pointC.setAttribute('cy', startY + (endY - startY) * eased);
                    
                    if (progress < 1) {
                        requestAnimationFrame(animate);
                    } else {
                        setStep(currentState);
                        isAnimating = false;
                    }
                }
                requestAnimationFrame(animate);
            } else {
                setStep(currentState);
                isAnimating = false;
            }
        }
        
        // 绑定事件
        elements.prevBtn.addEventListener('click', stepBackward);
        elements.nextBtn.addEventListener('click', stepForward);
        
        // 初始化
        setStep(0);
    })();
    </script>
