    <script>
    (function() {
        // Scene 3: 等边三角形折叠问题
        // 布局坐标表参考：
        // A(200,400), B(500,400), C(350,140), D(300,400), E(308,212), F(438,292)
        
        let currentStep = 0;
        const totalSteps = 5;
        
        // 获取 DOM 元素
        const btnPrev = document.getElementById('scene3-btn-prev');
        const btnNext = document.getElementById('scene3-btn-next');
        const stepIndicator = document.getElementById('scene3-step-indicator');
        
        // 步骤相关的元素
        const step1Elements = document.querySelectorAll('.scene3-step1');
        const step2Elements = document.querySelectorAll('.scene3-step2');
        const step3Elements = document.querySelectorAll('.scene3-step3');
        const step4Elements = document.querySelectorAll('.scene3-step4');
        const step5Elements = document.querySelectorAll('.scene3-step5');
        const step5SolveElements = document.querySelectorAll('.scene3-step5-solve');
        const step5ResultElements = document.querySelectorAll('.scene3-step5-result');
        
        // 更新按钮状态
        function updateButtons() {
            btnPrev.disabled = (currentStep === 0);
            btnNext.disabled = (currentStep === totalSteps);
            stepIndicator.textContent = '步骤 ' + currentStep + '/' + totalSteps;
        }
        
        // 显示元素
        function showElement(el) {
            if (el) {
                el.style.opacity = '1';
                el.style.display = '';
                el.classList.add('scene3-fade-in');
            }
        }
        
        // 隐藏元素
        function hideElement(el) {
            if (el) {
                el.style.opacity = '0';
                el.style.display = 'none';
                el.classList.remove('scene3-fade-in');
            }
        }
        
        // 更新步骤显示
        function updateStep() {
            // 步骤 1: △ADE 填充 + CE=x
            if (currentStep >= 1) {
                step1Elements.forEach(el => showElement(el));
            } else {
                step1Elements.forEach(el => hideElement(el));
            }
            
            // 步骤 2: AE=3-x + DE=x
            if (currentStep >= 2) {
                step2Elements.forEach(el => showElement(el));
            } else {
                step2Elements.forEach(el => hideElement(el));
            }
            
            // 步骤 3: ∠A=60° 高亮
            if (currentStep >= 3) {
                step3Elements.forEach(el => {
                    el.style.fontWeight = 'bold';
                    el.style.fill = '#000000';
                    showElement(el);
                });
            } else {
                step3Elements.forEach(el => hideElement(el));
            }
            
            // 步骤 4: 方程框出现
            if (currentStep >= 4) {
                step4Elements.forEach(el => showElement(el));
            } else {
                step4Elements.forEach(el => hideElement(el));
            }
            
            // 步骤 5: 方程求解过程
            if (currentStep >= 5) {
                step5Elements.forEach(el => showElement(el));
                // 延迟显示后续步骤
                setTimeout(() => {
                    step5SolveElements.forEach(el => showElement(el));
                }, 800);
                setTimeout(() => {
                    step5ResultElements.forEach(el => showElement(el));
                }, 1600);
            } else {
                step5Elements.forEach(el => hideElement(el));
                step5SolveElements.forEach(el => hideElement(el));
                step5ResultElements.forEach(el => hideElement(el));
            }
            
            updateButtons();
        }
        
        // 事件监听
        btnPrev.addEventListener('click', function() {
            if (currentStep > 0) {
                currentStep--;
                updateStep();
            }
        });
        
        btnNext.addEventListener('click', function() {
            if (currentStep < totalSteps) {
                currentStep++;
                updateStep();
            }
        });
        
        // 初始化
        updateButtons();
        
        // 设置按钮位置 (SVG 内使用 foreignObject 或直接用 HTML 绝对定位)
        // 这里我们用 HTML 绝对定位覆盖在 SVG 上
        const container = document.getElementById('scene3-container');
        const controlsHtml = document.getElementById('scene3-controls');
        
        // 将 SVG 内的 controls 移除，改用 HTML 绝对定位
        controlsHtml.remove();
        
        const controlsDiv = document.createElement('div');
        controlsDiv.id = 'scene3-controls-html';
        controlsDiv.style.cssText = 'position:absolute;bottom:10px;left:0;right:0;display:flex;justify-content:center;align-items:center;gap:20px;z-index:10;';
        controlsDiv.innerHTML = `
            <button id="scene3-btn-prev-html" class="scene3-btn scene3-btn-prev">◀ 上一步</button>
            <span id="scene3-step-indicator-html" style="font-size:14px;color:#424242;min-width:80px;text-align:center;">步骤 0/5</span>
            <button id="scene3-btn-next-html" class="scene3-btn scene3-btn-next">下一步 ▶</button>
        `;
        container.appendChild(controlsDiv);
        
        // 重新绑定事件
        document.getElementById('scene3-btn-prev-html').addEventListener('click', function() {
            if (currentStep > 0) {
                currentStep--;
                updateStep();
                document.getElementById('scene3-step-indicator-html').textContent = '步骤 ' + currentStep + '/' + totalSteps;
            }
        });
        
        document.getElementById('scene3-btn-next-html').addEventListener('click', function() {
            if (currentStep < totalSteps) {
                currentStep++;
                updateStep();
                document.getElementById('scene3-step-indicator-html').textContent = '步骤 ' + currentStep + '/' + totalSteps;
            }
        });
        
        // 移除 SVG 内的按钮元素
        const svgBtnPrev = document.getElementById('scene3-btn-prev');
        const svgBtnNext = document.getElementById('scene3-btn-next');
        const svgStepIndicator = document.getElementById('scene3-step-indicator');
        if (svgBtnPrev) svgBtnPrev.remove();
        if (svgBtnNext) svgBtnNext.remove();
        if (svgStepIndicator) svgStepIndicator.remove();
        
    })();
    </script>
