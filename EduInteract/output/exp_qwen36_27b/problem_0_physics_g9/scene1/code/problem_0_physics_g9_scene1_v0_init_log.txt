    <script>
    (function() {
        'use strict';

        var currentStep = 0;
        var totalSteps = 3;

        // Element references
        var btnPrev = document.getElementById('scene1-btn-prev');
        var btnNext = document.getElementById('scene1-btn-next');

        // Step 1 elements
        var gAdult = document.getElementById('scene1-g-adult');
        var gChild = document.getElementById('scene1-g-child');
        var lAdult = document.getElementById('scene1-l-adult');
        var lChild = document.getElementById('scene1-l-child');
        var lAdultTick1 = document.getElementById('scene1-l-adult-tick1');
        var lAdultTick2 = document.getElementById('scene1-l-adult-tick2');
        var lChildTick1 = document.getElementById('scene1-l-child-tick1');
        var lChildTick2 = document.getElementById('scene1-l-child-tick2');
        var labelGAdult = document.getElementById('scene1-label-g-adult');
        var labelGChild = document.getElementById('scene1-label-g-child');
        var labelGCompare = document.getElementById('scene1-label-g-compare');
        var labelLAdult = document.getElementById('scene1-label-l-adult');
        var labelLChild = document.getElementById('scene1-label-l-child');

        // Step 2 elements
        var eqBox = document.getElementById('scene1-eq-box');
        var eqText = document.getElementById('scene1-eq-text');
        var highlightAdult = document.getElementById('scene1-highlight-adult');
        var highlightChild = document.getElementById('scene1-highlight-child');

        // Step 3 elements
        var relBox = document.getElementById('scene1-rel-box');
        var relText = document.getElementById('scene1-rel-text');

        // Animation helper: interpolate an SVG attribute over time
        function animateAttr(el, attr, from, to, duration, easing) {
            var start = performance.now();
            function frame(now) {
                var t = Math.min((now - start) / duration, 1);
                var eased = easing ? easing(t) : t;
                el.setAttribute(attr, from + (to - from) * eased);
                if (t < 1) requestAnimationFrame(frame);
            }
            requestAnimationFrame(frame);
        }

        function easeOut(t) {
            return 1 - Math.pow(1 - t, 3);
        }

        // Unified show/hide utility (handles both display and opacity)
        function show(el) {
            el.style.display = '';
            el.style.opacity = '1';
        }

        function hide(el) {
            el.style.display = 'none';
            el.style.opacity = '0';
        }

        // Main render function: update all elements based on current step
        function renderStep(step) {
            // Update button states
            btnPrev.disabled = (step === 0);
            btnNext.disabled = (step === totalSteps);

            // ---- Step 1: Force arrows, arm lengths, labels ----
            if (step >= 1) {
                show(gAdult);
                show(gChild);
                show(lAdult);
                show(lChild);
                show(lAdultTick1);
                show(lAdultTick2);
                show(lChildTick1);
                show(lChildTick2);
                show(labelGAdult);
                show(labelGChild);
                show(labelGCompare);
                show(labelLAdult);
                show(labelLChild);

                // Animate force arrows from zero length to full length
                animateAttr(gAdult, 'y2', 210, 350, 400, easeOut);
                animateAttr(gChild, 'y2', 230, 330, 400, easeOut);

                // Animate arm lengths from fulcrum outward
                animateAttr(lAdult, 'x1', 300, 200, 400, easeOut);
                animateAttr(lChild, 'x2', 300, 550, 400, easeOut);
            } else {
                hide(gAdult);
                hide(gChild);
                hide(lAdult);
                hide(lChild);
                hide(lAdultTick1);
                hide(lAdultTick2);
                hide(lChildTick1);
                hide(lChildTick2);
                hide(labelGAdult);
                hide(labelGChild);
                hide(labelGCompare);
                hide(labelLAdult);
                hide(labelLChild);

                // Reset attributes to initial (zero-length) state
                gAdult.setAttribute('y2', 210);
                gChild.setAttribute('y2', 230);
                lAdult.setAttribute('x1', 300);
                lAdult.setAttribute('x2', 300);
                lChild.setAttribute('x1', 300);
                lChild.setAttribute('x2', 300);
            }

            // ---- Step 2: Equation box and highlight rectangles ----
            if (step >= 2) {
                show(eqBox);
                show(eqText);
                show(highlightAdult);
                show(highlightChild);
                // Set highlight fill-opacity to 20%
                highlightAdult.setAttribute('fill-opacity', '0.2');
                highlightChild.setAttribute('fill-opacity', '0.2');
            } else {
                hide(eqBox);
                hide(eqText);
                hide(highlightAdult);
                hide(highlightChild);
                highlightAdult.setAttribute('fill-opacity', '0');
                highlightChild.setAttribute('fill-opacity', '0');
            }

            // ---- Step 3: Relation box and arm color changes ----
            if (step >= 3) {
                show(relBox);
                show(relText);

                // Change arm colors to green, thicken lines
                lAdult.setAttribute('stroke', '#2e7d32');
                lAdult.setAttribute('stroke-width', '2');
                lChild.setAttribute('stroke', '#2e7d32');
                lChild.setAttribute('stroke-width', '2');
                lAdultTick1.setAttribute('stroke', '#2e7d32');
                lAdultTick2.setAttribute('stroke', '#2e7d32');
                lChildTick1.setAttribute('stroke', '#2e7d32');
                lChildTick2.setAttribute('stroke', '#2e7d32');
                labelLAdult.setAttribute('fill', '#2e7d32');
                labelLChild.setAttribute('fill', '#2e7d32');
            } else {
                hide(relBox);
                hide(relText);

                // Reset arm colors to gray
                lAdult.setAttribute('stroke', '#888888');
                lAdult.setAttribute('stroke-width', '1.5');
                lChild.setAttribute('stroke', '#888888');
                lChild.setAttribute('stroke-width', '1.5');
                lAdultTick1.setAttribute('stroke', '#888888');
                lAdultTick2.setAttribute('stroke', '#888888');
                lChildTick1.setAttribute('stroke', '#888888');
                lChildTick2.setAttribute('stroke', '#888888');
                labelLAdult.setAttribute('fill', '#888888');
                labelLChild.setAttribute('fill', '#888888');
            }
        }

        // Button event handlers
        btnPrev.addEventListener('click', function() {
            if (currentStep > 0) {
                currentStep--;
                renderStep(currentStep);
            }
        });

        btnNext.addEventListener('click', function() {
            if (currentStep < totalSteps) {
                currentStep++;
                renderStep(currentStep);
            }
        });

        // Initial render (step 0: only static elements visible)
        renderStep(0);
    })();
    </script>
