    <script>
    (function() { 'use strict';
        // ===== 布局坐标表（对照spec.json） =====
        // 杠杆: (100,247)-(700,253), 600×6
        // 支点: 顶点(350,250), 底边(330,280)-(370,280)
        // 大人(初): 中心(160,250) | 小孩(初): 中心(620,250)
        // 大人(新): 中心(230,250) | 小孩(新): 中心(550,250)
        // 移动箭头: 大人(170,210)→(220,210), 小孩(610,210)→(560,210)
        // ΔL标注: 左(195,235), 右(585,235)
        // 新力臂: 左(350,270)→(230,270), 右(350,270)→(550,270)
        // 力臂标签: 左(290,295), 右(450,295)
        // 柱状图: 基线(220,390)-(580,390), 蓝柱(250,310)60×80, 红柱(490,334)60×56
        // 结论框: (150,335)-(650,390), 500×55

        var currentStep = 0;
        var totalSteps = 4;

        // DOM引用
        var leverGroup = document.getElementById('scene2-lever-group');
        var step0People = document.getElementById('scene2-step0-people');
        var step0Labels = document.getElementById('scene2-step0-labels');
        var step1People = document.getElementById('scene2-step1-people');
        var step1Labels = document.getElementById('scene2-step1-labels');
        var step2 = document.getElementById('scene2-step2');
        var step3 = document.getElementById('scene2-step3');
        var step4 = document.getElementById('scene2-step4');
        var btnPrev = document.getElementById('scene2-btn-prev');
        var btnNext = document.getElementById('scene2-btn-next');
        var stepIndicator = document.getElementById('scene2-step-indicator');

        // 显隐工具函数（同时处理display和opacity）
        function show(el) {
            el.classList.add('visible');
            el.style.opacity = '';
        }

        function hide(el) {
            el.classList.remove('visible');
        }

        // 步骤更新
        function updateStep() {
            // 重置所有元素
            hide(step0People);
            hide(step0Labels);
            hide(step1People);
            hide(step1Labels);
            hide(step2);
            hide(step3);
            hide(step4);
            leverGroup.classList.remove('tilted');

            // 根据当前步骤显示对应元素
            if (currentStep === 0) {
                // 初始平衡状态
                show(step0People);
                show(step0Labels);
            } else if (currentStep === 1) {
                // 移动过程
                show(step1People);
                show(step1Labels);
            } else if (currentStep === 2) {
                // 新力臂标注
                show(step1People);
                show(step1Labels);
                show(step2);
            } else if (currentStep === 3) {
                // 力矩变化量对比（柱状图）
                show(step1People);
                show(step1Labels);
                show(step2);
                show(step3);
            } else if (currentStep === 4) {
                // 结论（柱状图淡出，杠杆倾斜，结论框显示）
                show(step1People);
                show(step1Labels);
                show(step2);
                show(step4);
                leverGroup.classList.add('tilted');
            }

            // 更新按钮状态
            btnPrev.disabled = (currentStep === 0);
            btnNext.disabled = (currentStep === totalSteps);
            stepIndicator.textContent = '步骤 ' + currentStep + ' / ' + totalSteps;
        }

        // 事件绑定
        btnPrev.addEventListener('click', function() {
            if (currentStep > 0) {
                currentStep--;
                updateStep();
            }
        });

        btnNext.addEventListener('click', function() {
            if (currentStep < totalSteps) {
                currentStep++;
                updateStep();
            }
        });

        // 初始化显示步骤0
        updateStep();
    })();
    </script>
