    <script>
    (function() { 'use strict';
        var currentStep = 0;
        var maxStep = 6;
        var stepElements = {
            1: ['scene1-square1', 'scene1-label-A1', 'scene1-label-B1', 'scene1-label-C1', 'scene1-label-D1'],
            2: ['scene1-angle-arc', 'scene1-angle-label', 'scene1-length-line', 'scene1-length-label', 'scene1-dot-B1', 'scene1-dot-C1'],
            3: ['scene1-rect1', 'scene1-square2', 'scene1-label-E1', 'scene1-label-E2', 'scene1-label-A2', 'scene1-label-B2', 'scene1-label-C2', 'scene1-label-D2'],
            4: ['scene1-rect2', 'scene1-square3', 'scene1-label-E3', 'scene1-label-E4', 'scene1-label-A3', 'scene1-label-B3', 'scene1-label-C3', 'scene1-label-D3'],
            5: ['scene1-parallel-marks', 'scene1-hl-B1C1', 'scene1-hl-B2C2', 'scene1-hl-B3C3'],
            6: ['scene1-ellipsis', 'scene1-question']
        };
        function showEl(id) {
            var el = document.getElementById(id);
            if (el) { el.style.opacity = '1'; el.style.display = ''; }
        }
        function hideEl(id) {
            var el = document.getElementById(id);
            if (el) { el.style.opacity = '0'; }
        }
        function renderStep() {
            for (var s = 1; s <= maxStep; s++) {
                var ids = stepElements[s];
                if (!ids) continue;
                for (var i = 0; i < ids.length; i++) {
                    if (s <= currentStep) { showEl(ids[i]); }
                    else { hideEl(ids[i]); }
                }
            }
            document.getElementById('scene1-prev-btn').disabled = (currentStep === 0);
            document.getElementById('scene1-next-btn').disabled = (currentStep === maxStep);
            document.getElementById('scene1-step-indicator').textContent = '步骤 ' + currentStep + '/' + maxStep;
        }
        document.getElementById('scene1-prev-btn').addEventListener('click', function() {
            if (currentStep > 0) { currentStep--; renderStep(); }
        });
        document.getElementById('scene1-next-btn').addEventListener('click', function() {
            if (currentStep < maxStep) { currentStep++; renderStep(); }
        });
        renderStep();
    })();
    </script>
