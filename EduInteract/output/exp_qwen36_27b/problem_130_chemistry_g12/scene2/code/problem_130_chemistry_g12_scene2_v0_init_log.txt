    <script>
    (function() { 'use strict';

        // ===== State =====
        var currentStep = 0;
        var TOTAL_STEPS = 5;

        // ===== DOM References =====
        var prevBtn = document.getElementById('scene2-prev-btn');
        var nextBtn = document.getElementById('scene2-next-btn');
        var stepIndicator = document.getElementById('scene2-step-indicator');
        var answerEl = document.getElementById('scene2-answer');

        var boxes = [
            document.getElementById('scene2-box-1'),
            document.getElementById('scene2-box-2'),
            document.getElementById('scene2-box-3'),
            document.getElementById('scene2-box-4')
        ];
        var contents = [
            document.getElementById('scene2-content-1'),
            document.getElementById('scene2-content-2'),
            document.getElementById('scene2-content-3'),
            document.getElementById('scene2-content-4')
        ];
        var hints = [
            document.getElementById('scene2-hint-1'),
            document.getElementById('scene2-hint-2'),
            document.getElementById('scene2-hint-3'),
            document.getElementById('scene2-hint-4')
        ];

        // ===== Helper functions =====
        function show(el) {
            el.style.display = '';
            el.style.opacity = '1';
        }
        function hide(el) {
            el.style.display = 'none';
            el.style.opacity = '0';
        }

        // ===== Render function =====
        function render() {
            // Update step indicator
            stepIndicator.textContent = currentStep + '/' + TOTAL_STEPS;

            // Update buttons
            prevBtn.disabled = (currentStep === 0);
            if (currentStep === TOTAL_STEPS) {
                nextBtn.textContent = '重新开始';
            } else {
                nextBtn.textContent = '下一步 ▶';
            }

            // Update answer visibility
            if (currentStep === TOTAL_STEPS) {
                show(answerEl);
            } else {
                hide(answerEl);
            }

            // Update each structure box
            for (var i = 0; i < 4; i++) {
                var box = boxes[i];
                var content = contents[i];
                var hint = hints[i];
                var structIndex = i + 1; // 1-based

                if (currentStep === 0) {
                    // Step 0: all boxes blank with hints
                    box.setAttribute('stroke', '#888888');
                    box.setAttribute('stroke-width', '1');
                    box.setAttribute('fill', '#ffffff');
                    hide(content);
                    show(hint);
                } else if (currentStep === TOTAL_STEPS) {
                    // Step 5: all green highlighted
                    box.setAttribute('stroke', '#2e7d32');
                    box.setAttribute('stroke-width', '3');
                    box.setAttribute('fill', 'rgba(46,125,50,0.08)');
                    show(content);
                    content.style.opacity = '1';
                    hide(hint);
                } else if (structIndex === currentStep) {
                    // Currently highlighted structure
                    box.setAttribute('stroke', '#1565c0');
                    box.setAttribute('stroke-width', '3');
                    box.setAttribute('fill', 'rgba(21,101,192,0.08)');
                    show(content);
                    content.style.opacity = '1';
                    hide(hint);
                } else if (structIndex < currentStep) {
                    // Previously shown structures (dimmed)
                    box.setAttribute('stroke', '#888888');
                    box.setAttribute('stroke-width', '1');
                    box.setAttribute('fill', '#ffffff');
                    show(content);
                    content.style.opacity = '0.6';
                    hide(hint);
                } else {
                    // Not yet shown
                    box.setAttribute('stroke', '#888888');
                    box.setAttribute('stroke-width', '1');
                    box.setAttribute('fill', '#ffffff');
                    hide(content);
                    show(hint);
                }
            }
        }

        // ===== Event handlers =====
        prevBtn.addEventListener('click', function() {
            if (currentStep > 0) {
                currentStep--;
                render();
            }
        });

        nextBtn.addEventListener('click', function() {
            if (currentStep === TOTAL_STEPS) {
                // Restart
                currentStep = 0;
            } else if (currentStep < TOTAL_STEPS) {
                currentStep++;
            }
            render();
        });

        // ===== Initial render =====
        render();

    })();
    </script>
