    <script>
    (function() { 'use strict';

        // ===== 布局坐标表（对照 spec.json） =====
        // 提示文本: (400, 55) 居中
        // A骨架顶点: (160,100)(120,220)(200,220)(160,260)
        // B骨架: 上三角(400,100)(360,140)(440,140) 下三角(400,220)(360,260)(440,260)
        // C骨架: 前(600,140)-(680,220) 后(640,100)-(720,180)
        // 标签A/B/C: y=305  分子式: y=338~370  最简式: y=382~414
        // 导航按钮: 底部居中

        var currentStep = 0;
        var maxStep = 4;

        // 提示文本内容
        var hints = [
            "观察三种笼状烃的骨架结构",
            "每个顶点 = 1个碳原子",
            "每个碳原子：3个C-C键 + 1个C-H键",
            "数清C、H原子个数，写出分子式",
            "三者最简式相同，均为 CH"
        ];

        // DOM 引用
        var hintEl = document.getElementById("scene1-hint");
        var carbonsEl = document.getElementById("scene1-carbons");
        var hydrogensEl = document.getElementById("scene1-hydrogens");
        var formulasEl = document.getElementById("scene1-formulas");
        var empiricalEl = document.getElementById("scene1-empirical");
        var btnPrev = document.getElementById("scene1-btn-prev");
        var btnNext = document.getElementById("scene1-btn-next");

        // 显示/隐藏工具函数（同时处理 display 和 opacity）
        function show(el) {
            el.style.display = "";
            el.style.opacity = "1";
        }
        function hide(el) {
            el.style.display = "none";
            el.style.opacity = "0";
        }

        // 渲染指定步骤
        function renderStep(step) {
            // 更新提示文本
            hintEl.textContent = hints[step];

            // 步骤1+: 显示碳原子
            if (step >= 1) {
                show(carbonsEl);
            } else {
                hide(carbonsEl);
            }

            // 步骤2+: 显示氢原子
            if (step >= 2) {
                show(hydrogensEl);
            } else {
                hide(hydrogensEl);
            }

            // 步骤3+: 显示分子式
            if (step >= 3) {
                show(formulasEl);
            } else {
                hide(formulasEl);
            }

            // 步骤4: 显示最简式
            if (step >= 4) {
                show(empiricalEl);
            } else {
                hide(empiricalEl);
            }

            // 按钮状态
            btnPrev.disabled = (step <= 0);
            btnNext.disabled = (step >= maxStep);
        }

        // 事件绑定
        btnNext.addEventListener("click", function() {
            if (currentStep < maxStep) {
                currentStep++;
                renderStep(currentStep);
            }
        });

        btnPrev.addEventListener("click", function() {
            if (currentStep > 0) {
                currentStep--;
                renderStep(currentStep);
            }
        });

        // 初始渲染
        renderStep(0);

    })();
    </script>
