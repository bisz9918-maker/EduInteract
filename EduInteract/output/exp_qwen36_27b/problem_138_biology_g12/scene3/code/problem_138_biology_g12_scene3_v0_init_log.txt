    <script>
    (function() { 'use strict';
        var currentStep = 0, totalSteps = 3;
        var stepElements = {
            0: ['scene3-plant-b'],
            1: ['scene3-arrow1', 'scene3-pollen', 'scene3-arrow2', 'scene3-haploid'],
            2: ['scene3-arrow3', 'scene3-diploid'],
            3: ['scene3-principle-box']
        };
        var btnPrev = document.getElementById('scene3-btn-prev');
        var btnNext = document.getElementById('scene3-btn-next');
        var stepIndicator = document.getElementById('scene3-step-indicator');
        function updateView() {
            for (var s = 0; s <= totalSteps; s++) {
                var show = s <= currentStep;
                stepElements[s].forEach(function(id) {
                    var el = document.getElementById(id);
                    if (el) { if (show) el.classList.add('visible'); else el.classList.remove('visible'); }
                });
            }
            btnPrev.disabled = (currentStep === 0);
            btnNext.disabled = (currentStep === totalSteps);
            stepIndicator.textContent = (currentStep + 1) + '/' + totalSteps;
        }
        btnNext.addEventListener('click', function() { if (currentStep < totalSteps) { currentStep++; updateView(); } });
        btnPrev.addEventListener('click', function() { if (currentStep > 0) { currentStep--; updateView(); } });
        updateView();
    })();
    </script>
