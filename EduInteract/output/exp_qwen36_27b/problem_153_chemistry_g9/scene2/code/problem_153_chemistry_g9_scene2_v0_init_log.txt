    <script>
    (function() {
        // ===== LAYOUT COORDINATE TABLE (from spec) =====
        // Col centers: H2=146, N2=219, HCl=292, H2O=365, CO=438, CO2=512
        // Row centers: 初始=98, 通过A后=154, 通过B后=210, 通过C后=266, 通过D后=322, 最终=370

        var gasColors = {
            'H2': '#1565c0', 'N2': '#2e7d32', 'HCl': '#c62828',
            'H2O': '#00838f', 'CO': '#6a1b9a', 'CO2': '#ef6c00'
        };
        var gasNames = ['H2', 'N2', 'HCl', 'H2O', 'CO', 'CO2'];
        var colCenters = [146, 219, 292, 365, 438, 512];
        var rowCenters = [98, 154, 210, 266, 322, 370];

        // ===== STEP DATA =====
        var stepData = [
            { rows: [[],[],[],[],[],[]], info: [], showAnswer: false },
            {
                rows: [['dot','dot','dot','dot','dot','dot'],[],[],[],[],[]],
                info: ['A装置：碳酸氢钠溶液', '除去气体：HCl', '反应：HCl + NaHCO₃ →', '     NaCl + H₂O + CO₂'],
                showAnswer: false
            },
            {
                rows: [['dot','dot','dot','dot','dot','dot'],['dot','dot','cross','dot-new','dot','dot-new'],[],[],[],[]],
                info: ['B装置：氧化铜（加热）', '除去气体：H₂、CO', '反应：H₂ + CuO → Cu + H₂O', '       CO + CuO → Cu + CO₂'],
                showAnswer: false
            },
            {
                rows: [['dot','dot','dot','dot','dot','dot'],['dot','dot','cross','dot-new','dot','dot-new'],['cross','dot','cross','dot-new','cross','dot-new'],[],[],[]],
                info: ['C装置：澄清石灰水', '除去气体：CO₂', '反应：CO₂ + Ca(OH)₂ →', '     CaCO₃↓ + H₂O'],
                showAnswer: false
            },
            {
                rows: [['dot','dot','dot','dot','dot','dot'],['dot','dot','cross','dot-new','dot','dot-new'],['cross','dot','cross','dot-new','cross','dot-new'],['cross','dot','cross','dot','cross','cross'],[],[]],
                info: ['D装置：浓硫酸', '除去气体：H₂O（干燥）', '浓硫酸具有强吸水性'],
                showAnswer: false
            },
            {
                rows: [['dot','dot','dot','dot','dot','dot'],['dot','dot','cross','dot-new','dot','dot-new'],['cross','dot','cross','dot-new','cross','dot-new'],['cross','dot','cross','dot','cross','cross'],['cross','dot','cross','cross','cross','cross'],['cross','dot','cross','cross','cross','cross']],
                info: ['总结', 'A：除去 HCl', 'B：除去 H₂、CO', 'C：除去 CO₂', 'D：除去 H₂O', '最终剩余：N₂'],
                showAnswer: true
            }
        ];

        var currentStep = 0;
        var cellMarksGroup = document.getElementById('scene2-cell-marks');
        var infoContent = document.getElementById('scene2-info-content');
        var answerBox = document.getElementById('scene2-answer-box');
        var prevBtn = document.getElementById('scene2-prev-btn');
        var nextBtn = document.getElementById('scene2-next-btn');
        var stepIndicator = document.getElementById('scene2-step-indicator');
        var finalHighlight = document.getElementById('scene2-final-highlight');

        function createDot(cx, cy, color, isNew) {
            var g = document.createElementNS('http://www.w3.org/2000/svg', 'g');
            g.setAttribute('class', 'cell-mark');
            g.setAttribute('opacity', '0');

            var circle = document.createElementNS('http://www.w3.org/2000/svg', 'circle');
            circle.setAttribute('cx', cx);
            circle.setAttribute('cy', cy);
            circle.setAttribute('r', '8');
            circle.setAttribute('fill', color);
            circle.setAttribute('stroke', '#333');
            circle.setAttribute('stroke-width', '1');
            g.appendChild(circle);

            if (isNew) {
                var label = document.createElementNS('http://www.w3.org/2000/svg', 'text');
                label.setAttribute('x', cx);
                label.setAttribute('y', cy + 18);
                label.setAttribute('text-anchor', 'middle');
                label.setAttribute('fill', '#ef6c00');
                label.setAttribute('font-size', '9');
                label.setAttribute('font-family', '"Microsoft YaHei", sans-serif');
                label.textContent = '含新生成';
                g.appendChild(label);
            }
            return g;
        }

        function createCross(cx, cy) {
            var g = document.createElementNS('http://www.w3.org/2000/svg', 'g');
            g.setAttribute('class', 'cell-mark');
            g.setAttribute('opacity', '0');
            var s = 6;
            var line1 = document.createElementNS('http://www.w3.org/2000/svg', 'line');
            line1.setAttribute('x1', cx - s); line1.setAttribute('y1', cy - s);
            line1.setAttribute('x2', cx + s); line1.setAttribute('y2', cy + s);
            line1.setAttribute('stroke', '#9e9e9e'); line1.setAttribute('stroke-width', '2');
            g.appendChild(line1);
            var line2 = document.createElementNS('http://www.w3.org/2000/svg', 'line');
            line2.setAttribute('x1', cx + s); line2.setAttribute('y1', cy - s);
            line2.setAttribute('x2', cx - s); line2.setAttribute('y2', cy + s);
            line2.setAttribute('stroke', '#9e9e9e'); line2.setAttribute('stroke-width', '2');
            g.appendChild(line2);
            return g;
        }

        function renderStep(step) {
            var data = stepData[step];
            cellMarksGroup.innerHTML = '';
            infoContent.innerHTML = '';

            // Render cell marks
            for (var r = 0; r < 6; r++) {
                var rowData = data.rows[r];
                if (!rowData) continue;
                for (var c = 0; c < 6; c++) {
                    var markType = rowData[c];
                    if (!markType) continue;
                    var cx = colCenters[c];
                    var cy = rowCenters[r];
                    var color = gasColors[gasNames[c]];
                    var mark;
                    if (markType === 'dot' || markType === 'dot-new') {
                        mark = createDot(cx, cy, color, markType === 'dot-new');
                    } else if (markType === 'cross') {
                        mark = createCross(cx, cy);
                    }
                    if (mark) {
                        cellMarksGroup.appendChild(mark);
                        (function(el, delay) {
                            setTimeout(function() { el.setAttribute('opacity', '1'); }, delay);
                        })(mark, (r * 6 + c) * 40);
                    }
                }
            }

            // Final row N2 highlight
            if (step === 5) {
                finalHighlight.style.display = 'block';
            } else {
                finalHighlight.style.display = 'none';
            }

            // Render info panel content
            var infoLines = data.info;
            for (var i = 0; i < infoLines.length; i++) {
                var line = document.createElement('div');
                line.textContent = infoLines[i];
                infoContent.appendChild(line);
            }

            // Show/hide answer box
            if (data.showAnswer) {
                answerBox.classList.add('visible');
            } else {
                answerBox.classList.remove('visible');
            }

            // Update buttons
            prevBtn.disabled = (step === 0);
            nextBtn.disabled = (step === 5);
            stepIndicator.textContent = step + '/5';
        }

        prevBtn.addEventListener('click', function() {
            if (currentStep > 0) { currentStep--; renderStep(currentStep); }
        });
        nextBtn.addEventListener('click', function() {
            if (currentStep < 5) { currentStep++; renderStep(currentStep); }
        });

        renderStep(0);
    })();
    </script>
