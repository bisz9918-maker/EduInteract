            {

        function renderStep(step) {
            var data = stepData[step];

            infoTitle.textContent = data.title;
            infoContent.innerHTML = data.content;

            if (data.showA) {
                showEl(groupA);
                groupA.style.opacity = data.dimA ? 0.6 : 1.0;
            } else {
                hideEl(groupA);
            }

            if (data.showB) {
                showEl(groupB);
                groupB.style.opacity = data.dimB ? 0.6 : 1.0;
            } else {
                hideEl(groupB);
            }

            if (data.showM) {
                showEl(groupM);
                groupM.style.opacity = data.dimM ? 0.6 : 1.0;
            } else {
                hideEl(groupM);
            }

            if (data.showCool) {
                showEl(groupCool);
            } else {
                hideEl(groupCool);
            }

            if (data.showMethods) {
                showEl(groupMethods);
            } else {
                hideEl(groupMethods);
            }

            if (data.showDilute) {
                showEl(groupDilute);
            } else {
                hideEl(groupDilute);
            }

            btnPrev.disabled = (step === 0);
            btnNext.disabled = (step === maxStep);
            stepIndicator.textContent = '步骤 ' + step + ' / ' + maxStep;
        }

        btnPrev.addEventListener('click', function() {
            if (currentStep > 0) {
                currentStep--;
                renderStep(currentStep);
            }
        });

        btnNext.addEventListener('click', function() {
            if (currentStep < maxStep) {
                currentStep++;
                renderStep(currentStep);
            }
        });

        renderStep(0);

    })();
