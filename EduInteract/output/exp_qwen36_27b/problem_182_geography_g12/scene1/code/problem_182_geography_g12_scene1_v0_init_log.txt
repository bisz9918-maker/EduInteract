    <script>
    (function() {
        'use strict';

        var currentStep = 0;
        var totalSteps = 3;

        var prevBtn = document.getElementById('scene1-prev-btn');
        var nextBtn = document.getElementById('scene1-next-btn');
        var stepIndicator = document.getElementById('scene1-step-indicator');

        var step1Group = document.getElementById('scene1-step1-group');
        var step2Group = document.getElementById('scene1-step2-group');
        var step3Group = document.getElementById('scene1-step3-group');

        function showElement(el) {
            el.style.display = '';
            el.style.opacity = '1';
        }

        function hideElement(el) {
            el.style.display = 'none';
            el.style.opacity = '0';
        }

        function updateView() {
            stepIndicator.textContent = '步骤 ' + currentStep + '/3';
            prevBtn.disabled = (currentStep === 0);
            nextBtn.disabled = (currentStep === totalSteps);

            if (currentStep >= 1) {
                showElement(step1Group);
            } else {
                hideElement(step1Group);
            }

            if (currentStep === 2) {
                showElement(step2Group);
                hideElement(step3Group);
            } else {
                hideElement(step2Group);
            }

            if (currentStep === 3) {
                showElement(step3Group);
                hideElement(step2Group);
            } else {
                hideElement(step3Group);
            }
        }

        prevBtn.addEventListener('click', function() {
            if (currentStep > 0) {
                currentStep--;
                updateView();
            }
        });

        nextBtn.addEventListener('click', function() {
            if (currentStep < totalSteps) {
                currentStep++;
                updateView();
            }
        });

        updateView();

    })();
    </script>
