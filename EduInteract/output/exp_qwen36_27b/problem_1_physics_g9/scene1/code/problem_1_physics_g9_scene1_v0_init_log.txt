    <script>
    (function() {
        "use strict";

        // ===== Layout coordinate table (from spec) =====
        // Lever: (100,200)-(500,200), Pivot: triangle at O, Weight G: rect at A
        // Forces at B(500,200): F3→(500,80), F2→(600,100), F4→(400,100), F1→(600,300)
        // Labels: O(82,195), A(233,186), B(500,186), G(233,238)
        // Force labels: F3(515,72), F2(615,92), F4(380,92), F1(615,318)

        var currentStep = 0;
        var totalSteps = 4;

        // Helper: show/hide element with both display and opacity
        function showEl(id) {
            var el = document.getElementById(id);
            el.style.display = 'block';
            el.style.opacity = '1';
        }
        function hideEl(id) {
            var el = document.getElementById(id);
            el.style.display = 'none';
            el.style.opacity = '0';
        }

        // Helper: set force color and marker
        function setForce(id, color, width) {
            var el = document.getElementById(id);
            el.setAttribute('stroke', color);
            el.setAttribute('stroke-width', width);
            var markerMap = {
                '#000': 'scene1-arr-black',
                '#c62828': 'scene1-arr-red',
                '#2e7d32': 'scene1-arr-green',
                '#1565c0': 'scene1-arr-blue'
            };
            el.setAttribute('marker-end', 'url(#' + markerMap[color] + ')');
        }

        // Main render function
        function renderStep() {
            // Button states
            document.getElementById('scene1-btn-prev').disabled = (currentStep === 0);
            document.getElementById('scene1-btn-next').disabled = (currentStep === totalSteps);

            // Step 1: 阻力与阻力臂
            if (currentStep >= 1) {
                showEl('scene1-step1');
                document.getElementById('scene1-weight').setAttribute('stroke', '#c62828');
                document.getElementById('scene1-weight').setAttribute('stroke-width', '3');
            } else {
                hideEl('scene1-step1');
                document.getElementById('scene1-weight').setAttribute('stroke', '#000');
                document.getElementById('scene1-weight').setAttribute('stroke-width', '2');
            }

            // Step 2: F₃力臂
            if (currentStep >= 2) {
                showEl('scene1-step2');
                setForce('scene1-f3', '#2e7d32', '3');
            } else {
                hideEl('scene1-step2');
                setForce('scene1-f3', '#000', '2');
            }

            // Step 3: F₁、F₂、F₄力臂
            if (currentStep >= 3) {
                showEl('scene1-step3');
                setForce('scene1-f1', '#1565c0', '3');
                setForce('scene1-f2', '#1565c0', '3');
                setForce('scene1-f4', '#1565c0', '3');
            } else {
                hideEl('scene1-step3');
                setForce('scene1-f1', '#000', '2');
                setForce('scene1-f2', '#000', '2');
                setForce('scene1-f4', '#000', '2');
            }

            // Step 4: 结论
            if (currentStep >= 4) {
                showEl('scene1-step4');
            } else {
                hideEl('scene1-step4');
            }
        }

        // Event listeners
        document.getElementById('scene1-btn-prev').addEventListener('click', function() {
            if (currentStep > 0) { currentStep--; renderStep(); }
        });
        document.getElementById('scene1-btn-next').addEventListener('click', function() {
            if (currentStep < totalSteps) { currentStep++; renderStep(); }
        });

        // Initial render
        renderStep();
    })();
    </script>
