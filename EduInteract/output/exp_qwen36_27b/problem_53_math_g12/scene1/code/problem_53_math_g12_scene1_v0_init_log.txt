    <script>
    (function() { 'use strict';

        var currentStep = 0;
        var totalSteps = 3;

        // 获取元素
        var btnPrev = document.getElementById('scene1-btn-prev');
        var btnNext = document.getElementById('scene1-btn-next');
        var stepIndicator = document.getElementById('scene1-step-indicator');

        // 各步骤需要显示的元素ID列表
        var stepElements = {
            1: [
                'scene1-point-e',
                'scene1-seg-me',
                'scene1-label-e',
                'scene1-label-me-ad',
                'scene1-edge-ad-highlight'
            ],
            2: [
                'scene1-seg-ne',
                'scene1-label-ne-pd',
                'scene1-plane-mne',
                'scene1-label-plane-parallel',
                'scene1-edge-pd-highlight'
            ],
            3: [
                'scene1-pc-np',
                'scene1-pc-nc',
                'scene1-cd-ce',
                'scene1-cd-ed',
                'scene1-label-nc-np-q',
                'scene1-label-ce-ed',
                'scene1-label-answer'
            ]
        };

        function showElement(id) {
            var el = document.getElementById(id);
            if (!el) return;
            el.style.display = '';
            el.classList.remove('hidden-step');
            // 强制重排后设置 opacity 以触发 transition
            requestAnimationFrame(function() {
                el.style.opacity = '1';
            });
        }

        function hideElement(id) {
            var el = document.getElementById(id);
            if (!el) return;
            el.style.opacity = '0';
            el.classList.add('hidden-step');
            // 等待 transition 结束后隐藏 display
            setTimeout(function() {
                if (el.classList.contains('hidden-step')) {
                    el.style.display = 'none';
                }
            }, 500);
        }

        function updateStep() {
            // 更新按钮状态
            btnPrev.disabled = (currentStep === 0);
            btnNext.disabled = (currentStep === totalSteps);
            stepIndicator.textContent = '步骤 ' + currentStep + ' / ' + totalSteps;

            // 更新各步骤元素的可见性
            for (var s = 1; s <= totalSteps; s++) {
                var ids = stepElements[s];
                for (var i = 0; i < ids.length; i++) {
                    if (s <= currentStep) {
                        showElement(ids[i]);
                    } else {
                        hideElement(ids[i]);
                    }
                }
            }
        }

        function stepForward() {
            if (currentStep < totalSteps) {
                currentStep++;
                updateStep();
            }
        }

        function stepBackward() {
            if (currentStep > 0) {
                currentStep--;
                updateStep();
            }
        }

        // 事件绑定
        btnNext.addEventListener('click', stepForward);
        btnPrev.addEventListener('click', stepBackward);

        // 初始化：确保所有步骤元素初始隐藏
        for (var s = 1; s <= totalSteps; s++) {
            var ids = stepElements[s];
            for (var i = 0; i < ids.length; i++) {
                var el = document.getElementById(ids[i]);
                if (el) {
                    el.style.display = 'none';
                    el.style.opacity = '0';
                }
            }
        }

    })();
    </script>
