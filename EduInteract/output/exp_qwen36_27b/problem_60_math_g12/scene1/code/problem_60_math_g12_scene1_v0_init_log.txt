    <script>
    (function() { 'use strict';
        // 布局坐标表（来自spec.json）：
        // 圆O: 中心(280,250), r=120
        // 直径AB: (160,250)-(400,250)
        // 切线AC: (160,250)-(160,20)
        // 切线BD: (400,250)-(400,20)
        // 点O: (280,250), 点A: (160,250), 点B: (400,250)
        // 点P: 初始θ=110°, P_x=280+120*cos(110°), P_y=250-120*sin(110°)
        // 切线CD: 过P的切线，与AC交于C，与BD交于D
        // 数值面板: (580,30)-(780,370)
        // 滑块: 底部y≈460, 范围10~170

        var CX = 280, CY = 250, R = 120;
        var AX = 160, BX = 400; // A和B的x坐标

        var slider = document.getElementById('scene1-slider');
        var conclusion = document.getElementById('scene1-conclusion');

        // SVG元素引用
        var elP = document.getElementById('scene1-point-p');
        var elC = document.getElementById('scene1-point-c');
        var elD = document.getElementById('scene1-point-d');
        var elCD = document.getElementById('scene1-tangent-cd');
        var elPO = document.getElementById('scene1-segment-po');
        var elLabelP = document.getElementById('scene1-label-p');
        var elLabelC = document.getElementById('scene1-label-c');
        var elLabelD = document.getElementById('scene1-label-d');
        var elPanelPC = document.getElementById('scene1-panel-pc');
        var elPanelPD = document.getElementById('scene1-panel-pd');
        var elPanelPO = document.getElementById('scene1-panel-po');
        var elPanelProd = document.getElementById('scene1-panel-prod');
        var elPanelSq = document.getElementById('scene1-panel-sq');
        var elPanelAngle = document.getElementById('scene1-panel-angle');

        function updateGeometry(angleDeg) {
            var theta = angleDeg * Math.PI / 180;

            // 点P坐标
            var Px = CX + R * Math.cos(theta);
            var Py = CY - R * Math.sin(theta);

            // 切线方程: (x-CX)*cos(θ) - (y-CY)*sin(θ) = R
            // 与x=AX交点C: y_C = CY + ((AX-CX)*cos(θ) - R) / sin(θ)
            var sinT = Math.sin(theta);
            var cosT = Math.cos(theta);
            var yC = CY + ((AX - CX) * cosT - R) / sinT;
            var yD = CY + ((BX - CX) * cosT - R) / sinT;

            // 距离计算
            var PC = Math.sqrt((Px - AX) * (Px - AX) + (Py - yC) * (Py - yC));
            var PD = Math.sqrt((Px - BX) * (Px - BX) + (Py - yD) * (Py - yD));
            var PO = R; // 半径

            // 更新SVG元素
            elP.setAttribute('cx', Px);
            elP.setAttribute('cy', Py);
            elC.setAttribute('cx', AX);
            elC.setAttribute('cy', yC);
            elD.setAttribute('cx', BX);
            elD.setAttribute('cy', yD);

            // 切线CD延长线（超出C和D一点）
            var extendFactor = 30;
            var dx = BX - AX;
            var dy = yD - yC;
            var len = Math.sqrt(dx * dx + dy * dy);
            var ux = dx / len, uy = dy / len;
            elCD.setAttribute('x1', AX - ux * extendFactor);
            elCD.setAttribute('y1', yC - uy * extendFactor);
            elCD.setAttribute('x2', BX + ux * extendFactor);
            elCD.setAttribute('y2', yD + uy * extendFactor);

            // 线段PO
            elPO.setAttribute('x1', CX);
            elPO.setAttribute('y1', CY);
            elPO.setAttribute('x2', Px);
            elPO.setAttribute('y2', Py);

            // 标签位置（动态调整避免重叠）
            var labelP_x = Px - 18;
            var labelP_y = Py - 12;
            elLabelP.setAttribute('x', labelP_x);
            elLabelP.setAttribute('y', labelP_y);

            elLabelC.setAttribute('x', AX - 18);
            elLabelC.setAttribute('y', yC + 5);

            elLabelD.setAttribute('x', BX + 18);
            elLabelD.setAttribute('y', yD + 5);

            // 更新面板数值
            var product = PC * PD;
            var sq = PO * PO;
            elPanelPC.textContent = 'PC = ' + PC.toFixed(1);
            elPanelPD.textContent = 'PD = ' + PD.toFixed(1);
            elPanelPO.textContent = 'PO = ' + PO.toFixed(1);
            elPanelProd.textContent = 'PC·PD = ' + Math.round(product);
            elPanelSq.textContent = 'PO² = ' + Math.round(sq);
            elPanelAngle.textContent = 'θ = ' + angleDeg + '°';
        }

        // 初始渲染
        updateGeometry(110);

        // 滑块事件
        slider.addEventListener('input', function(e) {
            updateGeometry(parseInt(e.target.value));
        });

    })();
    </script>
