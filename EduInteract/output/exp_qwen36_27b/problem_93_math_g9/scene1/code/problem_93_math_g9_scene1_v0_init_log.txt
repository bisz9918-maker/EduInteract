    <script>
    (function() {
        'use strict';
        var currentStep = 0;
        var maxStep = 3;

        var hints = [
            "基础图形：等边△ACD与△BCE，连线DE、AF及交点G、H、P、F",
            "第一对相似：△ADG ∽ △AFB（公共角+60°对应角）",
            "第二对相似：△CHE ∽ △CFA（公共角+60°对应角）",
            "第三对相似：△DCE ∽ △ACF（60°对应角+夹角相等）"
        ];

        var allArcs = [
            'scene1-arc-dag', 'scene1-arc-fab', 'scene1-arc-adg', 'scene1-arc-abf',
            'scene1-arc-fca', 'scene1-arc-caf',
            'scene1-arc-dce', 'scene1-arc-caf3', 'scene1-arc-cde', 'scene1-arc-acf'
        ];
        var allLabels = [
            'scene1-label-60-d', 'scene1-label-60-b', 'scene1-label-eq-a',
            'scene1-label-60-c2', 'scene1-label-60-a2',
            'scene1-label-60-c3', 'scene1-label-60-d3'
        ];
        var allHl = [
            'scene1-hl-adg', 'scene1-hl-afb',
            'scene1-hl-che', 'scene1-hl-cfa',
            'scene1-hl-dce', 'scene1-hl-acf'
        ];

        function showEl(id) {
            var el = document.getElementById(id);
            if (el) { el.style.display = ''; el.style.opacity = ''; }
        }
        function hideEl(id) {
            var el = document.getElementById(id);
            if (el) { el.style.display = 'none'; el.style.opacity = '0'; }
        }
        function showHl(id) {
            var el = document.getElementById(id);
            if (el) { el.style.display = ''; el.style.opacity = ''; }
        }
        function hideHl(id) {
            var el = document.getElementById(id);
            if (el) { el.style.display = 'none'; el.style.opacity = '0'; }
        }

        function updateScene(step) {
            document.getElementById('scene1-hint').textContent = hints[step];
            document.getElementById('scene1-prev').disabled = (step === 0);
            document.getElementById('scene1-next').disabled = (step === maxStep);

            allHl.forEach(hideHl);
            allArcs.forEach(hideEl);
            allLabels.forEach(hideEl);

            if (step === 1) {
                showHl('scene1-hl-adg');
                showHl('scene1-hl-afb');
                ['scene1-arc-dag','scene1-arc-fab','scene1-arc-adg','scene1-arc-abf'].forEach(showEl);
                ['scene1-label-60-d','scene1-label-60-b','scene1-label-eq-a'].forEach(showEl);
            } else if (step === 2) {
                showHl('scene1-hl-che');
                showHl('scene1-hl-cfa');
                ['scene1-arc-fca','scene1-arc-caf'].forEach(showEl);
                ['scene1-label-60-c2','scene1-label-60-a2'].forEach(showEl);
            } else if (step === 3) {
                showHl('scene1-hl-dce');
                showHl('scene1-hl-acf');
                ['scene1-arc-dce','scene1-arc-caf3','scene1-arc-cde','scene1-arc-acf'].forEach(showEl);
                ['scene1-label-60-c3','scene1-label-60-d3'].forEach(showEl);
            }
        }

        document.getElementById('scene1-prev').addEventListener('click', function() {
            if (currentStep > 0) { currentStep--; updateScene(currentStep); }
        });
        document.getElementById('scene1-next').addEventListener('click', function() {
            if (currentStep < maxStep) { currentStep++; updateScene(currentStep); }
        });
    })();
    </script>
