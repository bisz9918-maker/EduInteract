    <script>
    (function() {
        const slider = document.getElementById('scene3-slider-x');
        const L = 220;
        const M = { x: 100, y: 420 };
        const N = { x: 320, y: 420 };
        
        function update() {
            const xDeg = parseFloat(slider.value);
            const yDeg = 60 - xDeg;
            
            const x = xDeg * Math.PI / 180;
            const y = yDeg * Math.PI / 180;
            const x2 = 2 * x;
            const y2 = 2 * y;
            
            // OM, ON
            const OM = L * Math.sin(y2) / Math.sin(Math.PI / 3);
            const ON = L * Math.sin(x2) / Math.sin(Math.PI / 3);
            const O = {
                x: M.x + OM * Math.cos(x2),
                y: M.y - OM * Math.sin(x2)
            };
            
            // AM
            const AM = L * Math.sin(y) / Math.sin(x2 + y);
            const A = {
                x: M.x + AM * Math.cos(x2),
                y: M.y - AM * Math.sin(x2)
            };
            
            // BN
            const BN = L * Math.sin(x) / Math.sin(x + y2);
            const B = {
                x: N.x - BN * Math.cos(y2),
                y: N.y - BN * Math.sin(y2)
            };
            
            // CM
            const CM = L * Math.sin(y) / Math.sin(2 * Math.PI / 3);
            const C = {
                x: M.x + CM * Math.cos(x),
                y: M.y - CM * Math.sin(x)
            };
            
            // Update lines
            const setLine = (id, p1, p2) => {
                const el = document.getElementById(id);
                el.setAttribute('x1', p1.x);
                el.setAttribute('y1', p1.y);
                el.setAttribute('x2', p2.x);
                el.setAttribute('y2', p2.y);
            };
            
            setLine('scene3-line-mn', M, N);
            setLine('scene3-line-ma', M, A);
            setLine('scene3-line-na', N, A);
            setLine('scene3-line-mb', M, B);
            setLine('scene3-line-nb', N, B);
            setLine('scene3-line-ao', A, O);
            setLine('scene3-line-bo', B, O);
            
            // Update points
            const setPt = (id, p) => {
                const el = document.getElementById(id);
                el.setAttribute('cx', p.x);
                el.setAttribute('cy', p.y);
            };
            setPt('scene3-pt-m', M);
            setPt('scene3-pt-n', N);
            setPt('scene3-pt-o', O);
            setPt('scene3-pt-a', A);
            setPt('scene3-pt-b', B);
            setPt('scene3-pt-c', C);
            
            // Update labels
            const setLbl = (id, p, dx, dy) => {
                const el = document.getElementById(id);
                el.setAttribute('x', p.x + dx);
                el.setAttribute('y', p.y + dy);
            };
            setLbl('scene3-lbl-m', M, -20, 15);
            setLbl('scene3-lbl-n', N, 15, 15);
            setLbl('scene3-lbl-o', O, -5, -15);
            setLbl('scene3-lbl-a', A, -20, -10);
            setLbl('scene3-lbl-b', B, 15, -10);
            setLbl('scene3-lbl-c', C, -5, 25); // C below
            
            // Arcs
            const getArc = (cx, cy, r, startAngle, endAngle) => {
                const start = {
                    x: cx + r * Math.cos(startAngle),
                    y: cy - r * Math.sin(startAngle)
                };
                const end = {
                    x: cx + r * Math.cos(endAngle),
                    y: cy - r * Math.sin(endAngle)
                };
                const largeArcFlag = endAngle - startAngle <= Math.PI ? "0" : "1";
                return `M ${start.x} ${start.y} A ${r} ${r} 0 ${largeArcFlag} 0 ${end.x} ${end.y}`;
            };
            
            document.getElementById('scene3-arc-omn').setAttribute('d', getArc(M.x, M.y, 35, 0, x2));
            document.getElementById('scene3-arc-onm').setAttribute('d', getArc(N.x, N.y, 35, Math.PI - y2, Math.PI));
            
            const angleOM = Math.PI + x2;
            const angleON = 2 * Math.PI - y2;
            document.getElementById('scene3-arc-mon').setAttribute('d', getArc(O.x, O.y, 25, angleON, angleOM));
            
            const angleCM = Math.PI + x;
            const angleCN = 2 * Math.PI - y;
            document.getElementById('scene3-arc-mcn').setAttribute('d', getArc(C.x, C.y, 20, angleCN, angleCM));
            
            // Arc labels
            setLbl('scene3-lbl-arc-omn', M, 45 * Math.cos(x), -45 * Math.sin(x) + 5);
            setLbl('scene3-lbl-arc-onm', N, -45 * Math.cos(y), -45 * Math.sin(y) + 5);
            
            const midO = (angleOM + angleON) / 2;
            setLbl('scene3-lbl-arc-mon', O, 35 * Math.cos(midO) - 10, -35 * Math.sin(midO) + 5);
            
            const midC = (angleCM + angleCN) / 2;
            setLbl('scene3-lbl-arc-mcn', C, 30 * Math.cos(midC) - 15, -30 * Math.sin(midC) - 15); // 120 deg above
            
            // Lengths
            const OA = OM - AM;
            const OB = ON - BN;
            
            const barScale = 0.8;
            document.getElementById('scene3-bar-oa').setAttribute('width', OA * barScale);
            document.getElementById('scene3-bar-ob').setAttribute('width', OB * barScale);
            
            document.getElementById('scene3-val-oa').textContent = OA.toFixed(1);
            document.getElementById('scene3-val-oa').setAttribute('x', 520 + OA * barScale + 10);
            
            document.getElementById('scene3-val-ob').textContent = OB.toFixed(1);
            document.getElementById('scene3-val-ob').setAttribute('x', 520 + OB * barScale + 10);
            
            const isEq = Math.abs(OA - OB) < 0.1;
            const color = isEq ? '#2e7d32' : '#c62828';
            
            document.getElementById('scene3-line-ao').setAttribute('stroke', color);
            document.getElementById('scene3-line-bo').setAttribute('stroke', color);
            
            const conc = document.getElementById('scene3-lbl-conclusion');
            conc.textContent = isEq ? 'OA = OB' : 'OA ≠ OB';
            conc.setAttribute('fill', color);
        }
        
        slider.addEventListener('input', update);
        update();
    })();
    </script>
