    <script>
    (function() {
        let currentStep = 0;
        const totalSteps = 6;

        const btnPrev = document.getElementById('scene1-btn-prev');
        const btnNext = document.getElementById('scene1-btn-next');
        const stepIndicator = document.getElementById('scene1-step-indicator');

        function setOpacity(id, opacity) {
            const el = document.getElementById(id);
            if (el) el.style.opacity = opacity;
        }

        function setClass(id, className, add) {
            const el = document.getElementById(id);
            if (el) {
                if (add) el.classList.add(className);
                else el.classList.remove(className);
            }
        }

        function updateScene() {
            // Update buttons
            btnPrev.disabled = currentStep === 0;
            btnNext.disabled = currentStep === totalSteps - 1;
            stepIndicator.textContent = `${currentStep + 1} / ${totalSteps}`;

            // Update text
            for (let i = 0; i < totalSteps; i++) {
                const el = document.getElementById(`scene1-step${i}`);
                if (el) {
                    if (i === currentStep) {
                        el.classList.add('active');
                    } else {
                        el.classList.remove('active');
                    }
                }
            }

            // Reset all dynamic elements
            setOpacity('scene1-poly-abc', 0);
            setOpacity('scene1-poly-bce', 0);
            setOpacity('scene1-poly-ade', 0);
            
            setOpacity('scene1-angle-c', 0);
            setOpacity('scene1-angle-abc', 0);
            setOpacity('scene1-angle-a', 0);
            setOpacity('scene1-angle-abe', 0);
            setOpacity('scene1-angle-ebc', 0);
            setOpacity('scene1-angle-bec', 0);
            
            setOpacity('scene1-line-be', 0);
            
            setClass('scene1-line-de', 'highlight-green-stroke', false);
            setClass('scene1-outline-abc', 'highlight-blue-stroke', false);
            
            setOpacity('scene1-mark-ae-be', 0);
            setOpacity('scene1-mark-bc', 0);
            
            setOpacity('scene1-len-ab', 0);
            setOpacity('scene1-len-ac', 0);
            setOpacity('scene1-len-ad', 0);
            setOpacity('scene1-len-ae-x', 0);
            setOpacity('scene1-len-ae-val', 0);
            setOpacity('scene1-len-ec', 0);
            setOpacity('scene1-len-bc', 0);

            // Apply state based on currentStep
            if (currentStep >= 0) {
                setOpacity('scene1-len-ab', 1);
                setOpacity('scene1-len-ac', 1);
                setOpacity('scene1-angle-c', 1);
            }
            
            if (currentStep >= 1) {
                setOpacity('scene1-poly-abc', 1);
                setOpacity('scene1-angle-abc', 1);
                setOpacity('scene1-angle-a', 1);
            }
            
            if (currentStep >= 2) {
                setOpacity('scene1-poly-abc', 0);
                setClass('scene1-line-de', 'highlight-green-stroke', true);
                setOpacity('scene1-line-be', 1);
                setOpacity('scene1-mark-ae-be', 1);
            }
            
            if (currentStep >= 3) {
                setClass('scene1-line-de', 'highlight-green-stroke', false);
                setOpacity('scene1-angle-abc', 0); // Hide to prevent overlap
                setOpacity('scene1-angle-abe', 1);
                setOpacity('scene1-angle-ebc', 1);
                setOpacity('scene1-angle-bec', 1);
                setOpacity('scene1-mark-bc', 1);
            }
            
            if (currentStep >= 4) {
                setOpacity('scene1-poly-bce', 1);
                setClass('scene1-outline-abc', 'highlight-blue-stroke', true);
                setOpacity('scene1-len-ac', 0); // Hide to prevent overlap
                setOpacity('scene1-len-ae-x', 1);
                setOpacity('scene1-len-ec', 1);
                setOpacity('scene1-len-bc', 1);
            }
            
            if (currentStep >= 5) {
                setOpacity('scene1-poly-bce', 0);
                setClass('scene1-outline-abc', 'highlight-blue-stroke', false);
                setOpacity('scene1-poly-ade', 1);
                setOpacity('scene1-len-ab', 0);
                setOpacity('scene1-len-ad', 1);
                setOpacity('scene1-len-ae-x', 0);
                setOpacity('scene1-len-ae-val', 1);
            }
        }

        btnPrev.addEventListener('click', () => {
            if (currentStep > 0) {
                currentStep--;
                updateScene();
            }
        });

        btnNext.addEventListener('click', () => {
            if (currentStep < totalSteps - 1) {
                currentStep++;
                updateScene();
            }
        });

        // Initialize
        updateScene();
    })();
    </script>
