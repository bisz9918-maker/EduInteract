    <script>
    (function() {
        let currentStep = 0;
        const maxStep = 5;
        
        const btnPrev = document.getElementById('scene1-btn-prev');
        const btnNext = document.getElementById('scene1-btn-next');
        const textBox = document.getElementById('scene1-text-box');
        
        const texts = [
            "已知极地经纬网局部，P、Q所在经线及Q点经度15°E。",
            "条件一：P、Q经度差为30°。已知Q为15°E，P的度数为45°，因此P点经度必然是45°E（45°E - 15°E = 30°）。",
            "经度向东递增。从15°E到45°E为向东方向（图中向左）。顺着向东方向，地球自转呈顺时针，说明O为南极点。",
            "条件二：P、Q经度差为60°。已知Q为15°E，P的度数为45°，因此P点经度必然是45°W（45°W + 15°E = 60°）。",
            "从45°W向东到15°E（图中向右），跨越0°经线。顺着向东方向，地球自转呈逆时针，说明O为北极点。",
            "20°W以西为西半球，以东为东半球。如图划分可知，P(45°W)位于西半球，Q(15°E)位于东半球。"
        ];

        function show(id) {
            const el = document.getElementById(id);
            if (el) {
                el.style.display = 'block';
                // trigger reflow
                void el.offsetWidth;
                el.style.opacity = '1';
            }
        }

        function hide(id) {
            const el = document.getElementById(id);
            if (el) {
                el.style.opacity = '0';
                setTimeout(() => {
                    if (el.style.opacity === '0') {
                        el.style.display = 'none';
                    }
                }, 400);
            }
        }

        function updateStep() {
            btnPrev.disabled = currentStep === 0;
            btnNext.disabled = currentStep === maxStep;
            textBox.innerHTML = texts[currentStep];
            
            // Reset all dynamic elements
            hide('scene1-p-label-0');
            hide('scene1-p-label-1');
            hide('scene1-p-label-3');
            hide('scene1-step2-arrows');
            hide('scene1-step4-arrows');
            hide('scene1-aux-lines');
            hide('scene1-aux-lines-2');
            hide('scene1-hemispheres');
            
            if (currentStep === 0) {
                show('scene1-p-label-0');
            } else if (currentStep === 1) {
                show('scene1-p-label-1');
            } else if (currentStep === 2) {
                show('scene1-p-label-1');
                show('scene1-step2-arrows');
            } else if (currentStep === 3) {
                show('scene1-p-label-3');
            } else if (currentStep === 4) {
                show('scene1-p-label-3');
                show('scene1-step4-arrows');
                show('scene1-aux-lines');
            } else if (currentStep === 5) {
                show('scene1-p-label-3');
                show('scene1-step4-arrows');
                show('scene1-aux-lines');
                show('scene1-aux-lines-2');
                show('scene1-hemispheres');
            }
        }

        btnPrev.addEventListener('click', () => {
            if (currentStep > 0) {
                currentStep--;
                updateStep();
            }
        });

        btnNext.addEventListener('click', () => {
            if (currentStep < maxStep) {
                currentStep++;
                updateStep();
            }
        });

        // Initialize
        updateStep();
    })();
    </script>
