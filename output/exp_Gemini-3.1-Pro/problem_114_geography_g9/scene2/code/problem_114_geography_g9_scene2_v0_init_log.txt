    <script>
    (function() {
        let currentStep = 0;
        const maxStep = 3;
        
        const btnPrev = document.getElementById('scene2-btn-prev');
        const btnNext = document.getElementById('scene2-btn-next');
        const stepIndicator = document.getElementById('scene2-step-indicator');
        
        const step1 = document.getElementById('scene2-step1');
        const step2 = document.getElementById('scene2-step2');
        const step3 = document.getElementById('scene2-step3');
        
        // 动画相关
        const sunrayLine = document.getElementById('scene2-sunray-line');
        const sunrayHead = document.getElementById('scene2-sunray-head');
        const sunPoint = document.getElementById('scene2-sun-point');
        let animationId = null;
        let isAnimating = false;
        
        function show(el) {
            el.style.display = 'block';
            // 强制重绘以确保过渡生效
            el.offsetHeight;
            el.style.opacity = '1';
        }
        
        function hide(el) {
            el.style.opacity = '0';
            setTimeout(() => {
                if (el.style.opacity === '0') {
                    el.style.display = 'none';
                }
            }, 300);
        }
        
        function updateView() {
            stepIndicator.textContent = `步骤 ${currentStep} / ${maxStep}`;
            btnPrev.disabled = currentStep === 0;
            btnNext.disabled = currentStep === maxStep;
            
            if (currentStep >= 1) show(step1); else hide(step1);
            if (currentStep >= 2) {
                show(step2);
                if (!isAnimating) startAnimation();
            } else {
                hide(step2);
                stopAnimation();
            }
            if (currentStep >= 3) show(step3); else hide(step3);
        }
        
        function startAnimation() {
            isAnimating = true;
            const startTime = Date.now();
            
            function animate() {
                if (!isAnimating) return;
                
                const time = Date.now() - startTime;
                // y = 300 + 31 * Math.sin(time / 500)
                const y = 300 + 31 * Math.sin(time / 1000); // 减慢速度
                
                sunrayLine.setAttribute('y1', y);
                sunrayLine.setAttribute('y2', y);
                
                sunrayHead.setAttribute('points', `720,${y} 728,${y-4} 728,${y+4}`);
                
                sunPoint.setAttribute('cy', y);
                
                animationId = requestAnimationFrame(animate);
            }
            
            animate();
        }
        
        function stopAnimation() {
            isAnimating = false;
            if (animationId) {
                cancelAnimationFrame(animationId);
                animationId = null;
            }
        }
        
        btnPrev.addEventListener('click', () => {
            if (currentStep > 0) {
                currentStep--;
                updateView();
            }
        });
        
        btnNext.addEventListener('click', () => {
            if (currentStep < maxStep) {
                currentStep++;
                updateView();
            }
        });
        
        // 初始化
        updateView();
    })();
    </script>
