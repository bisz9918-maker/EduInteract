    <script>
    (function() {
        let currentStep = 0;
        const maxStep = 3;
        
        const btnPrev = document.getElementById('scene1-btn-prev');
        const btnNext = document.getElementById('scene1-btn-next');
        
        const s1Elements = ['scene1-s1-hl', 'scene1-s1-arrow', 'scene1-s1-text', 'scene1-s1-conc'];
        const s2Elements = ['scene1-s2-hl', 'scene1-s2-arrow', 'scene1-s2-text', 'scene1-s2-conc'];
        const s3Elements = ['scene1-s3-hl-lat', 'scene1-s3-hl-lon', 'scene1-s3-arrow-top', 'scene1-s3-text-top', 'scene1-s3-arrow-right', 'scene1-s3-text-right', 'scene1-s3-conc'];
        
        function setOpacity(id, opacity) {
            const el = document.getElementById(id);
            if (el) el.style.opacity = opacity;
        }
        
        function updateScene() {
            btnPrev.disabled = currentStep === 0;
            btnNext.disabled = currentStep === maxStep;
            
            // Step 1
            s1Elements.forEach(id => setOpacity(id, currentStep >= 1 ? 1 : 0));
            if (currentStep >= 2) setOpacity('scene1-s1-conc', 0); // Hide step 1 conclusion in step 2
            
            // Step 2
            s2Elements.forEach(id => setOpacity(id, currentStep >= 2 ? 1 : 0));
            
            // Step 3
            s3Elements.forEach(id => setOpacity(id, currentStep >= 3 ? 1 : 0));
        }
        
        btnPrev.addEventListener('click', () => {
            if (currentStep > 0) {
                currentStep--;
                updateScene();
            }
        });
        
        btnNext.addEventListener('click', () => {
            if (currentStep < maxStep) {
                currentStep++;
                updateScene();
            }
        });
        
        updateScene();
    })();
    </script>
