    <script>
    (function() {
        let currentStep = 0;
        const maxStep = 2;
        
        const btnPrev = document.getElementById('scene2-btn-prev');
        const btnNext = document.getElementById('scene2-btn-next');
        
        const dirLines = document.querySelectorAll('.scene2-dir-line');
        const dirLabels = document.querySelectorAll('.scene2-dir-label');
        const labelS = document.getElementById('scene2-label-S');
        const labelW = document.getElementById('scene2-label-W');
        
        const arrowLine = document.getElementById('scene2-arrow-line');
        const arrowHead = document.getElementById('scene2-arrow-head');
        const conclusionText = document.getElementById('scene2-conclusion-text');
        
        function updateScene() {
            btnPrev.disabled = currentStep === 0;
            btnNext.disabled = currentStep === maxStep;
            
            if (currentStep >= 1) {
                dirLines.forEach(el => el.style.opacity = '1');
                dirLabels.forEach(el => el.style.opacity = '1');
            } else {
                dirLines.forEach(el => el.style.opacity = '0');
                dirLabels.forEach(el => el.style.opacity = '0');
            }
            
            if (currentStep >= 2) {
                arrowLine.style.opacity = '1';
                arrowLine.style.strokeDashoffset = '0';
                
                setTimeout(() => {
                    if (currentStep >= 2) arrowHead.style.opacity = '1';
                }, 800);
                
                labelS.classList.add('scene2-highlight');
                labelW.classList.add('scene2-highlight');
                
                conclusionText.style.opacity = '1';
            } else {
                arrowLine.style.opacity = '0';
                arrowLine.style.strokeDashoffset = '260';
                arrowHead.style.opacity = '0';
                
                labelS.classList.remove('scene2-highlight');
                labelW.classList.remove('scene2-highlight');
                
                conclusionText.style.opacity = '0';
            }
        }
        
        btnPrev.addEventListener('click', () => {
            if (currentStep > 0) {
                currentStep--;
                updateScene();
            }
        });
        
        btnNext.addEventListener('click', () => {
            if (currentStep < maxStep) {
                currentStep++;
                updateScene();
            }
        });
        
        updateScene();
    })();
    </script>
