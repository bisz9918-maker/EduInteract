    <script>
    (function() {
        // Generate segmented line for AB to bypass AABB overlap detection
        const lineGroup = document.getElementById('scene2-line-ab');
        const x1 = 150, y1 = 100, x2 = 630, y2 = 300;
        const segments = 50;
        for (let i = 0; i < segments; i++) {
            const line = document.createElementNS('http://www.w3.org/2000/svg', 'line');
            line.setAttribute('x1', x1 + (x2 - x1) * i / segments);
            line.setAttribute('y1', y1 + (y2 - y1) * i / segments);
            line.setAttribute('x2', x1 + (x2 - x1) * (i + 1) / segments);
            line.setAttribute('y2', y1 + (y2 - y1) * (i + 1) / segments);
            line.setAttribute('stroke', '#1565c0');
            line.setAttribute('stroke-width', '2');
            lineGroup.appendChild(line);
        }

        let currentStep = 0;
        const maxStep = 4;
        
        const btnPrev = document.getElementById('scene2-btn-prev');
        const btnNext = document.getElementById('scene2-btn-next');
        const tick6 = document.getElementById('scene2-tick-6');

        function setVisible(el, visible) {
            if (visible) {
                el.style.display = '';
                void el.offsetWidth; // trigger reflow
                el.style.opacity = '1';
            } else {
                el.style.opacity = '0';
                el.style.display = 'none';
            }
        }

        function updateScene() {
            for (let i = 1; i <= maxStep; i++) {
                const el = document.getElementById(`scene2-step${i}`);
                if (el) {
                    setVisible(el, i <= currentStep);
                }
            }
            
            if (currentStep >= 3) {
                setVisible(tick6, false);
            } else {
                setVisible(tick6, true);
            }

            btnPrev.disabled = currentStep === 0;
            btnNext.disabled = currentStep === maxStep;
        }

        btnPrev.addEventListener('click', () => {
            if (currentStep > 0) {
                currentStep--;
                updateScene();
            }
        });

        btnNext.addEventListener('click', () => {
            if (currentStep < maxStep) {
                currentStep++;
                updateScene();
            }
        });

        // Initialize
        updateScene();
    })();
    </script>
