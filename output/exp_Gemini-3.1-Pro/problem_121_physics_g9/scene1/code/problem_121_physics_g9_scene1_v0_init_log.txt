    <script>
    (function() {
        let currentStep = 0;
        const maxStep = 4;

        const btnPrev = document.getElementById('scene1-btn-prev');
        const btnNext = document.getElementById('scene1-btn-next');
        
        const topText = document.getElementById('scene1-top-text');
        const boxAll = document.getElementById('scene1-box-all');
        const boxPart = document.getElementById('scene1-box-part');
        const group123 = document.getElementById('scene1-group-blocks123');
        const forceTotal = document.getElementById('scene1-force-total');
        const forceF123 = document.getElementById('scene1-force-f123');
        const forceF456 = document.getElementById('scene1-force-f456');
        const forceF34 = document.getElementById('scene1-force-f34');

        const stepTexts = [
            "",
            "整体分析：匀速直线运动，受力平衡",
            "摩擦力均摊：总摩擦力平分给6个物块",
            "隔离分析：将4-6号物块作为一个新整体",
            "局部平衡：F_34 = 3 × 1N = 3N"
        ];

        function setVisibility(el, visible, opacity = "1") {
            if (visible) {
                el.style.display = "";
                // Force reflow
                void el.offsetWidth;
                el.style.opacity = opacity;
            } else {
                el.style.opacity = "0";
                setTimeout(() => {
                    if (el.style.opacity === "0") {
                        el.style.display = "none";
                    }
                }, 400);
            }
        }

        function updateStep(step) {
            // Update buttons
            btnPrev.disabled = step === 0;
            btnNext.disabled = step === maxStep;

            // Update text
            topText.textContent = stepTexts[step];
            topText.style.opacity = step === 0 ? "0" : "1";

            // Step 0: Initial
            if (step === 0) {
                setVisibility(boxAll, false);
                setVisibility(boxPart, false);
                group123.style.opacity = "1";
                setVisibility(forceTotal, false);
                setVisibility(forceF123, false);
                setVisibility(forceF456, false);
                setVisibility(forceF34, false);
            }
            // Step 1: 整体分析
            else if (step === 1) {
                setVisibility(boxAll, true);
                setVisibility(boxPart, false);
                group123.style.opacity = "1";
                setVisibility(forceTotal, true);
                setVisibility(forceF123, false);
                setVisibility(forceF456, false);
                setVisibility(forceF34, false);
            }
            // Step 2: 摩擦力均摊
            else if (step === 2) {
                setVisibility(boxAll, true);
                setVisibility(boxPart, false);
                group123.style.opacity = "1";
                setVisibility(forceTotal, false);
                setVisibility(forceF123, true);
                setVisibility(forceF456, true);
                setVisibility(forceF34, false);
            }
            // Step 3: 隔离分析
            else if (step === 3) {
                setVisibility(boxAll, false);
                setVisibility(boxPart, true);
                group123.style.opacity = "0.3";
                setVisibility(forceTotal, false);
                setVisibility(forceF123, true, "0.3");
                setVisibility(forceF456, true);
                setVisibility(forceF34, false);
                
                // Reset F_34 text for step 3
                forceF34.querySelector('text').textContent = "F_34=?";
            }
            // Step 4: 局部平衡
            else if (step === 4) {
                setVisibility(boxAll, false);
                setVisibility(boxPart, true);
                group123.style.opacity = "0.3";
                setVisibility(forceTotal, false);
                setVisibility(forceF123, true, "0.3");
                setVisibility(forceF456, true);
                setVisibility(forceF34, true);
                
                // Update F_34 text for step 4
                forceF34.querySelector('text').textContent = "F_34=3N";
            }
        }

        btnPrev.addEventListener('click', () => {
            if (currentStep > 0) {
                currentStep--;
                updateStep(currentStep);
            }
        });

        btnNext.addEventListener('click', () => {
            if (currentStep < maxStep) {
                currentStep++;
                updateStep(currentStep);
            }
        });

        // Initialize
        updateStep(0);
    })();
    </script>
