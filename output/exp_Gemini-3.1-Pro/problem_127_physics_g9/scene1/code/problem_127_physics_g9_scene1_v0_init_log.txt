    <script>
    (function() {
        let currentStep = 0;
        const maxStep = 3;
        
        const btnPrev = document.getElementById('scene1-btn-prev');
        const btnNext = document.getElementById('scene1-btn-next');
        
        const groups = [
            document.getElementById('scene1-step0-group'), // 始终显示
            document.getElementById('scene1-step1-group'),
            document.getElementById('scene1-step2-group'),
            document.getElementById('scene1-step3-group')
        ];

        function show(el) {
            if (!el) return;
            el.style.display = '';
            // 强制重绘
            el.getBoundingClientRect();
            el.style.opacity = '1';
        }

        function hide(el) {
            if (!el) return;
            el.style.opacity = '0';
            setTimeout(() => {
                if (el.style.opacity === '0') {
                    el.style.display = 'none';
                }
            }, 300);
        }

        function updateView() {
            btnPrev.disabled = currentStep === 0;
            btnNext.disabled = currentStep === maxStep;
            
            for (let i = 1; i <= maxStep; i++) {
                if (i <= currentStep) {
                    show(groups[i]);
                } else {
                    hide(groups[i]);
                }
            }
        }

        btnPrev.addEventListener('click', () => {
            if (currentStep > 0) {
                currentStep--;
                updateView();
            }
        });

        btnNext.addEventListener('click', () => {
            if (currentStep < maxStep) {
                currentStep++;
                updateView();
            }
        });
        
        // 初始状态
        updateView();
    })();
    </script>
