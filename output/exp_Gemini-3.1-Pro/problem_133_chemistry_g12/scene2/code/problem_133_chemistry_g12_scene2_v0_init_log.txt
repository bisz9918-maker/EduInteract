    <script>
    (function() {
        // 获取 DOM 元素
        const rectRemain = document.getElementById('scene2-rect-remain');
        const textRemain = document.getElementById('scene2-text-remain');
        const panelCalc = document.getElementById('scene2-panel-calc');
        const btnState1 = document.getElementById('scene2-btn-state1');
        const btnState2 = document.getElementById('scene2-btn-state2');
        const btnState3 = document.getElementById('scene2-btn-state3');
        const buttons = [btnState1, btnState2, btnState3];

        // 状态数据
        const states = {
            1: {
                rectFill: '#888888',
                rectOpacity: '0.2',
                text: '未知 (1)',
                calcHtml: '反应消耗了 2 份体积的混合气体。<br>根据 \(2H_2 + O_2 = 2H_2O\)，消耗的 \(H_2\) 为 \(4/3\)，\(O_2\) 为 \(2/3\)。<br>剩余 1 份体积的气体，可能是 \(H_2\) 或 \(O_2\)。'
            },
            2: {
                rectFill: '#1565c0',
                rectOpacity: '1',
                text: 'H₂ (1)',
                calcHtml: '假设剩余气体为 \(H_2\)。<br>\(H_2\) 总量 = \(4/3\) (消耗) + 1 (剩余) = \(7/3\)<br>\(O_2\) 总量 = \(2/3\) (消耗)<br>初始体积比 \(H_2 : O_2 = 7/3 : 2/3 = 7 : 2\)'
            },
            3: {
                rectFill: '#c62828',
                rectOpacity: '1',
                text: 'O₂ (1)',
                calcHtml: '假设剩余气体为 \(O_2\)。<br>\(H_2\) 总量 = \(4/3\) (消耗)<br>\(O_2\) 总量 = \(2/3\) (消耗) + 1 (剩余) = \(5/3\)<br>初始体积比 \(H_2 : O_2 = 4/3 : 5/3 = 4 : 5\)'
            }
        };

        // 切换状态函数
        function setState(stateIndex) {
            // 更新按钮样式
            buttons.forEach((btn, index) => {
                if (index + 1 === stateIndex) {
                    btn.classList.add('active');
                } else {
                    btn.classList.remove('active');
                }
            });

            // 更新 SVG 元素
            const stateData = states[stateIndex];
            rectRemain.setAttribute('fill', stateData.rectFill);
            rectRemain.setAttribute('fill-opacity', stateData.rectOpacity);
            textRemain.textContent = stateData.text;

            // 更新计算面板并重新渲染 MathJax
            panelCalc.innerHTML = stateData.calcHtml;
            if (window.MathJax) {
                MathJax.typesetPromise([panelCalc]).catch(function (err) {
                    console.error('MathJax error:', err.message);
                });
            }
        }

        // 绑定事件
        btnState1.addEventListener('click', () => setState(1));
        btnState2.addEventListener('click', () => setState(2));
        btnState3.addEventListener('click', () => setState(3));

        // 初始化
        setState(1);
    })();
    </script>
