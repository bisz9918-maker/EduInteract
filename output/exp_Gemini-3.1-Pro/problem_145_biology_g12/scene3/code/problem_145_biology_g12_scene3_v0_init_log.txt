    <script>
    (function() {
        // Generate 10x10 grid
        const gridContainer = document.getElementById('grid-container');
        let html = '';
        for (let i = 0; i < 100; i++) {
            const row = Math.floor(i / 10);
            const col = i % 10;
            const x = col * 10;
            const y = row * 10;
            let fill = '#ffffff';
            if (i < 1) fill = '#1565c0'; // 1% AA
            else if (i < 19) fill = '#bbdefb'; // 18% Aa
            
            html += `<rect x="${x}" y="${y}" width="10" height="10" fill="${fill}" stroke="#ccc" stroke-width="0.5"/>`;
        }
        gridContainer.innerHTML = html;

        let currentStep = 0;
        const maxStep = 4;
        
        const btnPrev = document.getElementById('scene3-btn-prev');
        const btnNext = document.getElementById('scene3-btn-next');
        
        function updateSteps() {
            for (let i = 0; i <= maxStep; i++) {
                const el = document.getElementById(`scene3-step${i}`);
                if (el) {
                    if (i <= currentStep) {
                        el.classList.add('active');
                    } else {
                        el.classList.remove('active');
                    }
                }
            }
            btnPrev.disabled = currentStep === 0;
            btnNext.disabled = currentStep === maxStep;
        }
        
        btnPrev.addEventListener('click', () => {
            if (currentStep > 0) {
                currentStep--;
                updateSteps();
            }
        });
        
        btnNext.addEventListener('click', () => {
            if (currentStep < maxStep) {
                currentStep++;
                updateSteps();
            }
        });
        
        updateSteps();
    })();
    </script>
