    <script>
    (function() {
        let currentStep = 0;
        const maxStep = 5;
        
        const btnPrev = document.getElementById('scene2-btn-prev');
        const btnNext = document.getElementById('scene2-btn-next');
        const stepText = document.getElementById('scene2-step-text');
        const divider = document.getElementById('scene2-divider');
        
        const steps = [
            document.getElementById('scene2-step0'),
            document.getElementById('scene2-step1'),
            document.getElementById('scene2-step2'),
            document.getElementById('scene2-step3'),
            document.getElementById('scene2-step4'),
            document.getElementById('scene2-step5')
        ];

        function updateScene() {
            stepText.textContent = `步骤 ${currentStep} / ${maxStep}`;
            btnPrev.disabled = currentStep === 0;
            btnNext.disabled = currentStep === maxStep;
            
            for (let i = 0; i <= maxStep; i++) {
                if (i <= currentStep) {
                    steps[i].style.opacity = '1';
                } else {
                    steps[i].style.opacity = '0';
                }
            }
            
            if (currentStep >= 4) {
                divider.style.opacity = '1';
            } else {
                divider.style.opacity = '0';
            }
        }

        btnPrev.addEventListener('click', () => {
            if (currentStep > 0) {
                currentStep--;
                updateScene();
            }
        });

        btnNext.addEventListener('click', () => {
            if (currentStep < maxStep) {
                currentStep++;
                updateScene();
            }
        });
        
        // Initialize
        updateScene();
    })();
    </script>
