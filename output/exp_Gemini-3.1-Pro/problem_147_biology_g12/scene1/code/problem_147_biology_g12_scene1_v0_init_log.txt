    <script>
    (function() {
        let currentStep = 0;
        const maxStep = 3;
        
        const btnPrev = document.getElementById('scene1-btn-prev');
        const btnNext = document.getElementById('scene1-btn-next');
        
        // Elements
        const arrow1 = document.getElementById('scene1-arrow1');
        const row2 = document.getElementById('scene1-row2');
        const arrow2 = document.getElementById('scene1-arrow2');
        const row3 = document.getElementById('scene1-row3');
        const primer2Fill = document.getElementById('scene1-primer2-fill');
        const primer1Anim = document.getElementById('scene1-primer1-anim');
        const primer2Anim = document.getElementById('scene1-primer2-anim');
        const arrow3 = document.getElementById('scene1-arrow3');
        const row4 = document.getElementById('scene1-row4');
        
        // Extension elements
        const newLeft3 = document.getElementById('scene1-new-left-3');
        const newLeftBond = document.getElementById('scene1-new-left-bond');
        const newLeft5 = document.getElementById('scene1-new-left-5');
        const newRight5 = document.getElementById('scene1-new-right-5');
        const newRightBond = document.getElementById('scene1-new-right-bond');
        const newRight3 = document.getElementById('scene1-new-right-3');
        
        let extensionTimeout1, extensionTimeout2;
        
        function updateState() {
            btnPrev.disabled = currentStep === 0;
            btnNext.disabled = currentStep === maxStep;
            
            clearTimeout(extensionTimeout1);
            clearTimeout(extensionTimeout2);
            
            // Step 0: Initial
            if (currentStep === 0) {
                arrow1.style.opacity = 0;
                row2.style.opacity = 0;
                arrow2.style.opacity = 0;
                row3.style.opacity = 0;
                primer2Fill.style.opacity = 0;
                
                primer1Anim.style.opacity = 0;
                primer1Anim.setAttribute('x', '450');
                primer1Anim.setAttribute('y', '85');
                
                primer2Anim.style.opacity = 0;
                primer2Anim.setAttribute('x', '450');
                primer2Anim.setAttribute('y', '120');
                
                arrow3.style.opacity = 0;
                row4.style.opacity = 0;
                
                newLeft3.style.opacity = 0;
                newLeftBond.style.opacity = 0;
                newLeft5.style.opacity = 0;
                newRight5.style.opacity = 0;
                newRightBond.style.opacity = 0;
                newRight3.style.opacity = 0;
            }
            
            // Step 1: Denaturation
            if (currentStep >= 1) {
                arrow1.style.opacity = 1;
                row2.style.opacity = 1;
                
                if (currentStep === 1) {
                    arrow2.style.opacity = 0;
                    row3.style.opacity = 0;
                    primer2Fill.style.opacity = 0;
                    primer1Anim.style.opacity = 0;
                    primer2Anim.style.opacity = 0;
                    arrow3.style.opacity = 0;
                    row4.style.opacity = 0;
                }
            }
            
            // Step 2: Annealing
            if (currentStep >= 2) {
                arrow2.style.opacity = 1;
                row3.style.opacity = 1;
                primer2Fill.style.opacity = 1;
                
                primer1Anim.style.opacity = 1;
                primer1Anim.setAttribute('x', '450');
                primer1Anim.setAttribute('y', '320');
                
                primer2Anim.style.opacity = 1;
                primer2Anim.setAttribute('x', '260');
                primer2Anim.setAttribute('y', '280');
                
                if (currentStep === 2) {
                    arrow3.style.opacity = 0;
                    row4.style.opacity = 0;
                    newLeft3.style.opacity = 0;
                    newLeftBond.style.opacity = 0;
                    newLeft5.style.opacity = 0;
                    newRight5.style.opacity = 0;
                    newRightBond.style.opacity = 0;
                    newRight3.style.opacity = 0;
                }
            }
            
            // Step 3: Extension
            if (currentStep >= 3) {
                arrow3.style.opacity = 1;
                row4.style.opacity = 1;
                
                // Hide animated primers as they are now part of row4
                primer1Anim.style.opacity = 0;
                primer2Anim.style.opacity = 0;
                
                // Show initial primers in row4
                newLeft5.style.opacity = 1;
                newRight5.style.opacity = 1;
                
                // Animate extension
                extensionTimeout1 = setTimeout(() => {
                    newLeftBond.style.opacity = 1;
                    newRightBond.style.opacity = 1;
                }, 500);
                
                extensionTimeout2 = setTimeout(() => {
                    newLeft3.style.opacity = 1;
                    newRight3.style.opacity = 1;
                }, 1000);
            }
        }
        
        btnPrev.addEventListener('click', () => {
            if (currentStep > 0) {
                currentStep--;
                updateState();
            }
        });
        
        btnNext.addEventListener('click', () => {
            if (currentStep < maxStep) {
                currentStep++;
                updateState();
            }
        });
        
        // Initialize
        updateState();
    })();
    </script>
