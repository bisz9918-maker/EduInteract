    <script>
    (function() {
        let currentStep = 0;
        const maxStep = 3;

        const btnPrev = document.getElementById('scene2-btn-prev');
        const btnNext = document.getElementById('scene2-btn-next');
        const stat1 = document.getElementById('scene2-stat1');
        const stat2 = document.getElementById('scene2-stat2');

        const genElements = [
            null, // gen0 is always visible
            document.getElementById('scene2-gen1'),
            document.getElementById('scene2-gen2'),
            document.getElementById('scene2-gen3')
        ];

        const statsData = [
            "第0代：DNA分子数 1，总单链数 2，未标记单链数 2",
            "第1代：DNA分子数 2，总单链数 4，未标记单链数 2",
            "第2代：DNA分子数 4，总单链数 8，未标记单链数 2",
            "第3代：DNA分子数 8，总单链数 16，未标记单链数 2"
        ];

        function show(el) {
            if (!el) return;
            el.style.display = '';
            // trigger reflow
            void el.offsetWidth;
            el.style.opacity = '1';
        }

        function hide(el) {
            if (!el) return;
            el.style.opacity = '0';
            el.style.display = 'none';
        }

        function updateScene() {
            // Update buttons
            btnPrev.disabled = currentStep === 0;
            btnNext.disabled = currentStep === maxStep;

            // Update stats
            stat1.textContent = statsData[currentStep];
            if (currentStep === 3) {
                show(stat2);
            } else {
                hide(stat2);
            }

            // Update generations
            for (let i = 1; i <= maxStep; i++) {
                if (i <= currentStep) {
                    show(genElements[i]);
                } else {
                    hide(genElements[i]);
                }
            }
        }

        btnPrev.addEventListener('click', () => {
            if (currentStep > 0) {
                currentStep--;
                updateScene();
            }
        });

        btnNext.addEventListener('click', () => {
            if (currentStep < maxStep) {
                currentStep++;
                updateScene();
            }
        });

        // Initialize
        updateScene();
    })();
    </script>
