    <script>
    (function() {
        let currentStep = 0;
        const maxStep = 4;

        const btnPrev = document.getElementById('scene1-btn-prev');
        const btnNext = document.getElementById('scene1-btn-next');
        const promptText = document.getElementById('scene1-prompt');
        
        const equations = document.getElementById('scene1-equations');
        const out1 = document.getElementById('scene1-out1');
        const out2 = document.getElementById('scene1-out2');
        const out3 = document.getElementById('scene1-out3');
        
        const devA = document.getElementById('scene1-dev-a');
        const devB = document.getElementById('scene1-dev-b');
        const devC = document.getElementById('scene1-dev-c');
        
        const line1 = document.getElementById('scene1-line1');
        const line2 = document.getElementById('scene1-line2');
        const line3 = document.getElementById('scene1-line3');
        
        const liquidB = document.getElementById('scene1-liquid-b');
        const liquidC = document.getElementById('scene1-liquid-c');
        const flameA = document.getElementById('scene1-flame-a');

        const prompts = [
            "实验目的：测定相同条件下CO和H2还原氧化铁的量。请组装实验装置。",
            "加热通入混合气体，发生还原反应生成CO2和H2O。",
            "先连接装置c（浓硫酸）吸收水分，测定H2的还原量，避免后续溶液带来水分干扰。",
            "再连接装置b（NaOH浓溶液）吸收CO2，测定CO的还原量。",
            "最后连接装置a（酒精灯）点燃尾气，处理有毒的CO，防止污染。"
        ];

        function updateStep() {
            btnPrev.disabled = currentStep === 0;
            btnNext.disabled = currentStep === maxStep;
            promptText.textContent = prompts[currentStep];

            // Step 1
            equations.style.opacity = currentStep >= 1 ? "1" : "0";
            out1.style.opacity = currentStep >= 1 ? "1" : "0";

            // Step 2
            if (currentStep >= 2) {
                devC.setAttribute('transform', 'translate(300, 220)');
                line1.style.opacity = "1";
                liquidC.setAttribute('fill', '#1565c0');
                liquidC.setAttribute('fill-opacity', '0.2');
                out2.style.opacity = "1";
            } else {
                devC.setAttribute('transform', 'translate(490, 350)');
                line1.style.opacity = "0";
                liquidC.setAttribute('fill', 'transparent');
                liquidC.setAttribute('fill-opacity', '0');
                out2.style.opacity = "0";
            }

            // Step 3
            if (currentStep >= 3) {
                devB.setAttribute('transform', 'translate(450, 220)');
                line2.style.opacity = "1";
                liquidB.setAttribute('fill', '#c62828');
                liquidB.setAttribute('fill-opacity', '0.2');
                out3.style.opacity = "1";
            } else {
                devB.setAttribute('transform', 'translate(320, 350)');
                line2.style.opacity = "0";
                liquidB.setAttribute('fill', 'transparent');
                liquidB.setAttribute('fill-opacity', '0');
                out3.style.opacity = "0";
            }

            // Step 4
            if (currentStep >= 4) {
                devA.setAttribute('transform', 'translate(600, 220)');
                line3.style.opacity = "1";
                flameA.setAttribute('fill', '#2e7d32');
            } else {
                devA.setAttribute('transform', 'translate(150, 350)');
                line3.style.opacity = "0";
                flameA.setAttribute('fill', '#ff9800');
            }
        }

        btnPrev.addEventListener('click', () => {
            if (currentStep > 0) {
                currentStep--;
                updateStep();
            }
        });

        btnNext.addEventListener('click', () => {
            if (currentStep < maxStep) {
                currentStep++;
                updateStep();
            }
        });

        updateStep();
    })();
    </script>
