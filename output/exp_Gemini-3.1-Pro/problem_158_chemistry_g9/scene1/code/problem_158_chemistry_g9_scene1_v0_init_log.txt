    <script>
    (function() {
        let currentStep = 0;
        const maxStep = 4;

        const btnPrev = document.getElementById('scene1-btn-prev');
        const btnNext = document.getElementById('scene1-btn-next');

        const arr1_1 = document.getElementById('scene1-arr1-1');
        const arr1_2 = document.getElementById('scene1-arr1-2');
        const arr1_3 = document.getElementById('scene1-arr1-3');
        
        const arr2_1 = document.getElementById('scene1-arr2-1');
        const arr2_2 = document.getElementById('scene1-arr2-2');
        const arr2_3 = document.getElementById('scene1-arr2-3');
        
        const arr3_1 = document.getElementById('scene1-arr3-1');
        const arr3_2 = document.getElementById('scene1-arr3-2');
        const arr3_3 = document.getElementById('scene1-arr3-3');

        const cross = document.getElementById('scene1-cross');
        const eq1Group = document.getElementById('scene1-eq1-group');
        const eq2Group = document.getElementById('scene1-eq2-group');
        const eq3Group = document.getElementById('scene1-eq3-group');
        const hlGroup = document.getElementById('scene1-hl-group');
        const finalText = document.getElementById('scene1-final-text');

        function setArrow(el, color) {
            el.setAttribute('stroke', color);
            if (color === '#000000') el.setAttribute('marker-end', 'url(#scene1-arrow-black)');
            else if (color === '#2e7d32') el.setAttribute('marker-end', 'url(#scene1-arrow-green)');
            else if (color === '#c62828') el.setAttribute('marker-end', 'url(#scene1-arrow-red)');
        }

        function show(el) {
            el.style.display = '';
            // force reflow
            void el.offsetWidth;
            el.setAttribute('opacity', '1');
        }

        function hide(el) {
            el.setAttribute('opacity', '0');
        }

        function updateState() {
            btnPrev.disabled = currentStep === 0;
            btnNext.disabled = currentStep === maxStep;

            // Step 1
            if (currentStep >= 1) {
                setArrow(arr1_1, '#2e7d32');
                setArrow(arr1_2, '#2e7d32');
                setArrow(arr1_3, '#2e7d32');
                show(eq1Group);
            } else {
                setArrow(arr1_1, '#000000');
                setArrow(arr1_2, '#000000');
                setArrow(arr1_3, '#000000');
                hide(eq1Group);
            }

            // Step 2
            if (currentStep >= 2) {
                setArrow(arr2_1, '#2e7d32');
                setArrow(arr2_2, '#c62828');
                setArrow(arr2_3, '#000000');
                show(cross);
                show(eq2Group);
            } else {
                setArrow(arr2_1, '#000000');
                setArrow(arr2_2, '#000000');
                setArrow(arr2_3, '#000000');
                hide(cross);
                hide(eq2Group);
            }

            // Step 3
            if (currentStep >= 3) {
                setArrow(arr3_1, '#2e7d32');
                setArrow(arr3_2, '#2e7d32');
                setArrow(arr3_3, '#2e7d32');
                show(eq3Group);
            } else {
                setArrow(arr3_1, '#000000');
                setArrow(arr3_2, '#000000');
                setArrow(arr3_3, '#000000');
                hide(eq3Group);
            }

            // Step 4
            if (currentStep >= 4) {
                show(hlGroup);
                show(finalText);
            } else {
                hide(hlGroup);
                hide(finalText);
            }
        }

        btnPrev.addEventListener('click', () => {
            if (currentStep > 0) {
                currentStep--;
                updateState();
            }
        });

        btnNext.addEventListener('click', () => {
            if (currentStep < maxStep) {
                currentStep++;
                updateState();
            }
        });

        updateState();
    })();
    </script>
