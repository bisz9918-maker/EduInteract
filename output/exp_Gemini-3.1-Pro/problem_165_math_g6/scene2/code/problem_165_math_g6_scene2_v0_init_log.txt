    <script>
    (function() {
        let currentStep = 0;
        const maxStep = 3;
        
        const btnPrev = document.getElementById('scene2-btn-prev');
        const btnNext = document.getElementById('scene2-btn-next');
        
        const groupBlue = document.getElementById('scene2-group-blue');
        const blueTiles = document.getElementById('scene2-blue-tiles');
        const groupGreen = document.getElementById('scene2-group-green');
        
        const textStep1 = document.getElementById('scene2-text-step1');
        const textStep2 = document.getElementById('scene2-text-step2');
        const textEquals = document.getElementById('scene2-text-equals');
        const textLeft4 = document.getElementById('scene2-text-left-4');
        const textRight9 = document.getElementById('scene2-text-right-9');
        const textFormula = document.getElementById('scene2-text-formula');
        
        function show(el) {
            el.style.display = '';
            // small delay to allow display to apply before opacity transition
            setTimeout(() => { el.style.opacity = '1'; }, 10);
        }
        
        function hide(el) {
            el.style.opacity = '0';
            setTimeout(() => { 
                if(el.style.opacity === '0') el.style.display = 'none'; 
            }, 500);
        }
        
        function updateScene() {
            btnPrev.disabled = currentStep === 0;
            btnNext.disabled = currentStep === maxStep;
            
            if (currentStep === 0) {
                groupBlue.style.transform = 'translate(0px, 0px) scale(1)';
                show(groupBlue);
                hide(blueTiles);
                hide(groupGreen);
                hide(textStep1);
                hide(textStep2);
                hide(textEquals);
                hide(textLeft4);
                hide(textRight9);
                hide(textFormula);
            } else if (currentStep === 1) {
                groupBlue.style.transform = 'translate(0px, 0px) scale(1)';
                show(groupBlue);
                show(blueTiles);
                hide(groupGreen);
                show(textStep1);
                hide(textStep2);
                hide(textEquals);
                hide(textLeft4);
                hide(textRight9);
                hide(textFormula);
            } else if (currentStep === 2) {
                hide(groupBlue);
                groupGreen.style.transform = 'translate(0px, 0px) scale(1)';
                show(groupGreen);
                hide(textStep1);
                show(textStep2);
                hide(textEquals);
                hide(textLeft4);
                hide(textRight9);
                hide(textFormula);
            } else if (currentStep === 3) {
                show(groupBlue);
                show(blueTiles);
                show(groupGreen);
                
                groupBlue.style.transform = 'translate(-150px, -10px) scale(0.75)';
                groupGreen.style.transform = 'translate(150px, -10px) scale(0.75)';
                
                hide(textStep1);
                hide(textStep2);
                show(textEquals);
                show(textLeft4);
                show(textRight9);
                show(textFormula);
            }
        }
        
        btnPrev.addEventListener('click', () => {
            if (currentStep > 0) {
                currentStep--;
                updateScene();
            }
        });
        
        btnNext.addEventListener('click', () => {
            if (currentStep < maxStep) {
                currentStep++;
                updateScene();
            }
        });
        
        // Initialize
        updateScene();
    })();
    </script>
