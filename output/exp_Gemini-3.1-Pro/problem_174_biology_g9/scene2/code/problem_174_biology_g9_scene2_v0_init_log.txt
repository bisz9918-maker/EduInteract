    <script>
    (function() {
        const pointer = document.getElementById('scene2-pointer');
        const lightningB = document.getElementById('scene2-lightning-b');
        const lightningC = document.getElementById('scene2-lightning-c');
        const waveLeft = document.getElementById('scene2-wave-left');
        const waveRight = document.getElementById('scene2-wave-right');
        const blockX = document.getElementById('scene2-block-x');
        
        const btnRest = document.getElementById('scene2-btn-rest');
        const btnStimB = document.getElementById('scene2-btn-stim-b');
        const btnStimC = document.getElementById('scene2-btn-stim-c');

        let currentState = 'rest';
        let animationId = null;
        let startTime = 0;

        function setOpacity(el, val) {
            el.style.opacity = val;
            el.style.display = val > 0 ? 'block' : 'none';
        }

        function resetScene() {
            if (animationId) cancelAnimationFrame(animationId);
            pointer.setAttribute('transform', 'rotate(0, 400, 120)');
            setOpacity(lightningB, 0);
            setOpacity(lightningC, 0);
            setOpacity(waveLeft, 0);
            setOpacity(waveRight, 0);
            setOpacity(blockX, 0);
            
            btnRest.classList.remove('active');
            btnStimB.classList.remove('active');
            btnStimC.classList.remove('active');
        }

        function animate(timestamp) {
            if (!startTime) startTime = timestamp;
            const elapsed = (timestamp - startTime) / 1000; // seconds

            if (currentState === 'stim-b') {
                setOpacity(lightningB, elapsed < 0.5 ? 1 : 0);
                
                if (elapsed < 1.3) {
                    setOpacity(waveLeft, 0.5);
                    let cx = 330 - (130 / 1.3) * elapsed;
                    if (cx < 200) cx = 200;
                    waveLeft.setAttribute('cx', cx);
                } else {
                    setOpacity(waveLeft, 0);
                }

                if (elapsed < 3.5) {
                    setOpacity(waveRight, 0.5);
                    let cx = 330;
                    if (elapsed < 0.7) {
                        cx = 330 + (70 / 0.7) * elapsed;
                    } else if (elapsed < 1.2) {
                        cx = 400;
                    } else if (elapsed < 3.1) {
                        cx = 410 + (190 / 1.9) * (elapsed - 1.2);
                    } else {
                        cx = 600;
                    }
                    if (cx > 600) cx = 600;
                    waveRight.setAttribute('cx', cx);
                } else {
                    setOpacity(waveRight, 0);
                }

                let angle = 0;
                if (elapsed >= 1.3 && elapsed <= 1.8) {
                    const progress = (elapsed - 1.3) / 0.5;
                    angle = -30 * Math.sin(progress * Math.PI);
                }
                else if (elapsed >= 3.1 && elapsed <= 3.6) {
                    const progress = (elapsed - 3.1) / 0.5;
                    angle = 30 * Math.sin(progress * Math.PI);
                }
                pointer.setAttribute('transform', `rotate(${angle}, 400, 120)`);

                if (elapsed < 4.0) {
                    animationId = requestAnimationFrame(animate);
                }
            } else if (currentState === 'stim-c') {
                setOpacity(lightningC, elapsed < 0.5 ? 1 : 0);

                if (elapsed < 1.5) {
                    setOpacity(waveRight, 0.5);
                    let cx = 450 + (150 / 1.5) * elapsed;
                    if (cx > 600) cx = 600;
                    waveRight.setAttribute('cx', cx);
                } else {
                    setOpacity(waveRight, 0);
                }

                if (elapsed < 0.4) {
                    setOpacity(waveLeft, 0.5);
                    let cx = 450 - (40 / 0.4) * elapsed;
                    if (cx < 410) cx = 410;
                    waveLeft.setAttribute('cx', cx);
                } else if (elapsed < 2.0) {
                    setOpacity(waveLeft, 0);
                    setOpacity(blockX, 1);
                } else {
                    setOpacity(blockX, 0);
                }

                let angle = 0;
                if (elapsed >= 1.5 && elapsed <= 2.0) {
                    const progress = (elapsed - 1.5) / 0.5;
                    angle = 30 * Math.sin(progress * Math.PI);
                }
                pointer.setAttribute('transform', `rotate(${angle}, 400, 120)`);

                if (elapsed < 2.5) {
                    animationId = requestAnimationFrame(animate);
                }
            }
        }

        btnRest.addEventListener('click', () => {
            resetScene();
            currentState = 'rest';
            btnRest.classList.add('active');
        });

        btnStimB.addEventListener('click', () => {
            resetScene();
            currentState = 'stim-b';
            btnStimB.classList.add('active');
            startTime = 0;
            animationId = requestAnimationFrame(animate);
        });

        btnStimC.addEventListener('click', () => {
            resetScene();
            currentState = 'stim-c';
            btnStimC.classList.add('active');
            startTime = 0;
            animationId = requestAnimationFrame(animate);
        });

        resetScene();
        btnRest.classList.add('active');
    })();
    </script>
