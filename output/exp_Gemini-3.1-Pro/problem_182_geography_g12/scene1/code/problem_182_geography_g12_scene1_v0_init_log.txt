    <script>
    (function() {
        let currentStep = 0;
        const maxStep = 3;
        
        const btnPrev = document.getElementById('scene1-btn-prev');
        const btnNext = document.getElementById('scene1-btn-next');
        const textBox = document.getElementById('scene1-text-box');
        
        const step1Group = document.getElementById('scene1-step1');
        const step2Group = document.getElementById('scene1-step2');
        const step3Group = document.getElementById('scene1-step3');
        
        const texts = [
            "本题考查经纬网空间定位与距离计算。请点击「下一步」查看各小题解析。",
            "(1) A(30°N)与B(30°S)正午太阳高度相同，直射点在赤道(0°)。从B到A方向为西北。",
            "(2) 飞机从B向北飞到C，沿经线飞行。跨纬度30°，距离 = 30 × 111km = 3330km。",
            "(3) 飞机从D向东飞到E，沿60°N纬线飞行。距离 = 60 × 111km × cos60° = 3330km。飞行时间 = 3330 ÷ 1110 = 3小时。"
        ];
        
        function updateStep() {
            btnPrev.disabled = currentStep === 0;
            btnNext.disabled = currentStep === maxStep;
            
            textBox.innerHTML = texts[currentStep];
            
            // Hide all
            step1Group.classList.remove('step-active');
            step2Group.classList.remove('step-active');
            step3Group.classList.remove('step-active');
            
            // Show current
            if (currentStep === 1) step1Group.classList.add('step-active');
            if (currentStep === 2) step2Group.classList.add('step-active');
            if (currentStep === 3) step3Group.classList.add('step-active');
            
            if (window.MathJax) {
                MathJax.typesetPromise([textBox]).catch(function (err) {
                    console.error(err.message);
                });
            }
        }
        
        btnPrev.addEventListener('click', () => {
            if (currentStep > 0) {
                currentStep--;
                updateStep();
            }
        });
        
        btnNext.addEventListener('click', () => {
            if (currentStep < maxStep) {
                currentStep++;
                updateStep();
            }
        });
        
        updateStep();
    })();
    </script>
