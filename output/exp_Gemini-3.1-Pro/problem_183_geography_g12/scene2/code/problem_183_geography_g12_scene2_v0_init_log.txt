    <script>
    (function() {
        const btnSummer = document.getElementById('scene2-btn-summer');
        const btnWinter = document.getElementById('scene2-btn-winter');
        const summerElements = document.getElementById('scene2-summer-elements');
        const winterElements = document.getElementById('scene2-winter-elements');
        const nightMask = document.getElementById('scene2-night-mask');
        const arcticText = document.getElementById('scene2-arctic-text');
        const antarcticText = document.getElementById('scene2-antarctic-text');

        function setVisibility(el, isVisible) {
            if (isVisible) {
                el.style.display = '';
                el.style.opacity = '1';
            } else {
                el.style.display = 'none';
                el.style.opacity = '0';
            }
        }

        btnSummer.addEventListener('click', () => {
            btnSummer.classList.add('active');
            btnWinter.classList.remove('active');
            
            setVisibility(summerElements, true);
            setVisibility(winterElements, false);
            
            nightMask.setAttribute('d', 'M 400 100 A 150 150 0 0 1 400 400 Z');
            arcticText.textContent = '极昼';
            antarcticText.textContent = '极夜';
        });

        btnWinter.addEventListener('click', () => {
            btnWinter.classList.add('active');
            btnSummer.classList.remove('active');
            
            setVisibility(summerElements, false);
            setVisibility(winterElements, true);
            
            nightMask.setAttribute('d', 'M 400 100 A 150 150 0 0 0 400 400 Z');
            arcticText.textContent = '极夜';
            antarcticText.textContent = '极昼';
        });
    })();
    </script>
