    <script>
    (function() {
        let currentStep = 0;
        const maxStep = 4;
        
        const btnPrev = document.getElementById('scene1-btn-prev');
        const btnNext = document.getElementById('scene1-btn-next');
        
        const steps = [
            null, // step 0 is default
            document.getElementById('scene1-step1'),
            document.getElementById('scene1-step2'),
            document.getElementById('scene1-step3'),
            document.getElementById('scene1-step4')
        ];

        function updateStep() {
            btnPrev.disabled = currentStep === 0;
            btnNext.disabled = currentStep === maxStep;
            
            for (let i = 1; i <= maxStep; i++) {
                if (steps[i]) {
                    if (i <= currentStep) {
                        steps[i].style.opacity = '1';
                        steps[i].style.display = 'block';
                    } else {
                        steps[i].style.opacity = '0';
                        // Use setTimeout to allow fade out animation before display none
                        setTimeout(() => {
                            if (i > currentStep) {
                                steps[i].style.display = 'none';
                            }
                        }, 300);
                    }
                }
            }
        }

        btnPrev.addEventListener('click', () => {
            if (currentStep > 0) {
                currentStep--;
                updateStep();
            }
        });

        btnNext.addEventListener('click', () => {
            if (currentStep < maxStep) {
                // Ensure display is block before fading in
                if (steps[currentStep + 1]) {
                    steps[currentStep + 1].style.display = 'block';
                    // Force reflow
                    steps[currentStep + 1].offsetHeight;
                }
                currentStep++;
                updateStep();
            }
        });
        
        // Initialize
        updateStep();
    })();
    </script>
