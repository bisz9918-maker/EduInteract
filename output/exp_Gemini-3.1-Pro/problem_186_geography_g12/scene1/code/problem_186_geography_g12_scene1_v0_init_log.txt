    <script>
    (function() {
        let currentStep = 0;
        const maxStep = 5;
        
        const btnPrev = document.getElementById('scene1-btn-prev');
        const btnNext = document.getElementById('scene1-btn-next');
        
        const earthFeatures = document.getElementById('scene1-earth-features');
        const eclipticGroup = document.getElementById('scene1-ecliptic-group');
        const equatorGroup = document.getElementById('scene1-equator-group');
        const angleGroup = document.getElementById('scene1-angle-group');
        const tropicsGroup = document.getElementById('scene1-tropics-group');
        const observerGroup = document.getElementById('scene1-observer-group');
        const dataPanel = document.getElementById('scene1-data-panel');
        const subsolarPoint = document.getElementById('scene1-subsolar-point');
        const pHalo = document.getElementById('scene1-p-halo');
        
        const flowEq = document.getElementById('scene1-flow-eq');
        const flowEc = document.getElementById('scene1-flow-ec');
        const flowAngle = document.getElementById('scene1-flow-angle');
        const flowReturnRect = document.getElementById('scene1-flow-return-rect');
        const flowH = document.getElementById('scene1-flow-h');
        const flowDn = document.getElementById('scene1-flow-dn');
        const flowTemp = document.getElementById('scene1-flow-temp');
        const flowSeasonRect = document.getElementById('scene1-flow-season-rect');
        
        const dataH = document.getElementById('scene1-data-h');
        const dataDaynight = document.getElementById('scene1-data-daynight');
        
        let animationId = null;
        let animTime = 0;
        let isAnimating = false;
        
        function setOpacity(el, val) {
            el.style.opacity = val;
        }
        
        function updateScene() {
            // Reset all
            setOpacity(eclipticGroup, 0);
            setOpacity(equatorGroup, 0);
            setOpacity(angleGroup, 0);
            setOpacity(tropicsGroup, 0);
            setOpacity(observerGroup, 0);
            setOpacity(dataPanel, 0);
            setOpacity(subsolarPoint, 0);
            
            flowEq.textContent = '    ';
            flowEc.textContent = '    ';
            flowAngle.textContent = '    ';
            flowH.textContent = '        ';
            flowDn.textContent = '        ';
            flowTemp.textContent = '    ';
            
            flowReturnRect.setAttribute('fill', '#ffffff');
            flowSeasonRect.setAttribute('fill', '#ffffff');
            
            isAnimating = false;
            if (animationId) cancelAnimationFrame(animationId);
            
            if (currentStep === 0) {
                earthFeatures.setAttribute('transform', 'rotate(0, 220, 225)');
            }
            
            if (currentStep >= 1) {
                earthFeatures.setAttribute('transform', 'rotate(-23.5, 220, 225)');
                setOpacity(eclipticGroup, 1);
                setOpacity(equatorGroup, 1);
                flowEq.textContent = '赤道';
                flowEc.textContent = '黄道';
            }
            
            if (currentStep >= 2) {
                setOpacity(angleGroup, 1);
                flowAngle.textContent = '黄赤';
            }
            
            if (currentStep >= 3) {
                setOpacity(tropicsGroup, 1);
                setOpacity(subsolarPoint, 1);
                flowReturnRect.setAttribute('fill', '#ffcdd2');
                isAnimating = true;
                animTime = 0;
                animate();
            }
            
            if (currentStep >= 4) {
                setOpacity(observerGroup, 1);
                setOpacity(dataPanel, 1);
                flowH.textContent = '正午太阳高度';
                flowDn.textContent = '昼夜长短';
            }
            
            if (currentStep >= 5) {
                flowTemp.textContent = '气温';
                flowSeasonRect.setAttribute('fill', '#ffcdd2');
            }
            
            btnPrev.disabled = currentStep === 0;
            btnNext.disabled = currentStep === maxStep;
        }
        
        function animate() {
            if (!isAnimating) return;
            animTime += 0.02;
            // delta oscillates between -23.5 and 23.5
            const delta = 23.5 * Math.sin(animTime);
            
            // Rotate earth features
            earthFeatures.setAttribute('transform', `rotate(${-delta}, 220, 225)`);
            
            if (currentStep >= 4) {
                // Calculate H = 90 - |40 - delta|
                const H = 90 - Math.abs(40 - delta);
                dataH.textContent = H.toFixed(1) + '°';
                
                if (delta > 0.5) {
                    dataDaynight.textContent = '昼长夜短';
                } else if (delta < -0.5) {
                    dataDaynight.textContent = '昼短夜长';
                } else {
                    dataDaynight.textContent = '昼夜平分';
                }
            }
            
            if (currentStep >= 5) {
                // Color halo based on delta
                // delta from -23.5 to 23.5 -> hue from 240 (blue) to 0 (red)
                const hue = 240 - ((delta + 23.5) / 47) * 240;
                pHalo.setAttribute('fill', `hsl(${hue}, 100%, 50%)`);
                pHalo.setAttribute('opacity', '0.6');
            } else {
                pHalo.setAttribute('fill', 'none');
            }
            
            animationId = requestAnimationFrame(animate);
        }
        
        btnPrev.addEventListener('click', () => {
            if (currentStep > 0) {
                currentStep--;
                updateScene();
            }
        });
        
        btnNext.addEventListener('click', () => {
            if (currentStep < maxStep) {
                currentStep++;
                updateScene();
            }
        });
        
        // Init
        updateScene();
    })();
    </script>
