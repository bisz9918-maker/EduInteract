    <script>
    (function() {
        let currentStep = 0;
        const maxStep = 3;

        const btnPrev = document.getElementById('scene2-btn-prev');
        const btnNext = document.getElementById('scene2-btn-next');

        const textStep1 = document.getElementById('scene2-text-step1');
        const textStep2 = document.getElementById('scene2-text-step2');
        const labelEast = document.getElementById('scene2-label-east');
        const labelWest = document.getElementById('scene2-label-west');
        const nightMask = document.getElementById('scene2-night-mask');
        const pointE = document.getElementById('scene2-point-e');
        const sunIcon = document.getElementById('scene2-sun-icon');
        const timeAxis = document.getElementById('scene2-time-axis');

        function updateScene() {
            // Step 0
            if (currentStep === 0) {
                textStep1.style.opacity = '0';
                textStep2.style.opacity = '0';
                nightMask.style.opacity = '0';
                sunIcon.style.opacity = '0';
                timeAxis.style.opacity = '0';
                pointE.setAttribute('fill', '#000');
                labelEast.textContent = '东半球';
                labelWest.textContent = '西半球';
            }
            // Step 1
            if (currentStep >= 1) {
                textStep1.style.opacity = '1';
                nightMask.style.opacity = '0.3';
                labelEast.textContent = '东半球(黑夜)';
            } else {
                textStep1.style.opacity = '0';
                nightMask.style.opacity = '0';
                labelEast.textContent = '东半球';
            }
            // Step 2
            if (currentStep >= 2) {
                textStep2.style.opacity = '1';
                sunIcon.style.opacity = '1';
                pointE.setAttribute('fill', '#c62828');
                labelWest.textContent = '西半球(白昼)';
            } else {
                textStep2.style.opacity = '0';
                sunIcon.style.opacity = '0';
                pointE.setAttribute('fill', '#000');
                labelWest.textContent = '西半球';
            }
            // Step 3
            if (currentStep >= 3) {
                timeAxis.style.opacity = '1';
            } else {
                timeAxis.style.opacity = '0';
            }

            btnPrev.disabled = currentStep === 0;
            btnNext.disabled = currentStep === maxStep;
        }

        btnPrev.addEventListener('click', () => {
            if (currentStep > 0) {
                currentStep--;
                updateScene();
            }
        });

        btnNext.addEventListener('click', () => {
            if (currentStep < maxStep) {
                currentStep++;
                updateScene();
            }
        });

        updateScene();
    })();
    </script>
