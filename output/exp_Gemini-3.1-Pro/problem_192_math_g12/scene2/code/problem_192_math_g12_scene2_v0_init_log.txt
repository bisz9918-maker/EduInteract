    <script>
    (function() {
        let currentStep = 0;
        const maxStep = 4;
        
        const btnPrev = document.getElementById('scene2-btn-prev');
        const btnNext = document.getElementById('scene2-btn-next');
        
        const elements = {
            s1: {
                hl: document.getElementById('scene2-hl-col1'),
                panel: document.getElementById('scene2-panel-s1')
            },
            s2: {
                hl: document.getElementById('scene2-hl-col2'),
                arr1: document.getElementById('scene2-arrow-s2-1'),
                arr2: document.getElementById('scene2-arrow-s2-2'),
                panel: document.getElementById('scene2-panel-s2')
            },
            s3: {
                nodeQ: document.getElementById('scene2-node-q'),
                nodeAns: document.getElementById('scene2-node-ans'),
                arr1: document.getElementById('scene2-arrow-s3-1'),
                arr2: document.getElementById('scene2-arrow-s3-2'),
                panel: document.getElementById('scene2-panel-s3')
            },
            s4: {
                hl: document.getElementById('scene2-hl-col3'),
                arr1: document.getElementById('scene2-arrow-s4-1'),
                arr2: document.getElementById('scene2-arrow-s4-2'),
                panel: document.getElementById('scene2-panel-s4')
            }
        };

        function updateState() {
            // Step 1
            const s1Active = currentStep >= 1;
            elements.s1.hl.style.opacity = s1Active ? '1' : '0';
            elements.s1.panel.style.opacity = s1Active ? '1' : '0';
            
            // Step 2
            const s2Active = currentStep >= 2;
            elements.s2.hl.style.opacity = s2Active ? '1' : '0';
            elements.s2.arr1.style.opacity = s2Active ? '1' : '0';
            elements.s2.arr2.style.opacity = s2Active ? '1' : '0';
            elements.s2.panel.style.opacity = s2Active ? '1' : '0';
            
            // Step 3
            const s3Active = currentStep >= 3;
            elements.s3.nodeQ.style.opacity = s3Active ? '0' : '1';
            elements.s3.nodeAns.style.opacity = s3Active ? '1' : '0';
            elements.s3.arr1.style.opacity = s3Active ? '1' : '0';
            elements.s3.arr2.style.opacity = s3Active ? '1' : '0';
            elements.s3.panel.style.opacity = s3Active ? '1' : '0';
            
            // Step 4
            const s4Active = currentStep >= 4;
            elements.s4.hl.style.opacity = s4Active ? '1' : '0';
            elements.s4.arr1.style.opacity = s4Active ? '1' : '0';
            elements.s4.arr2.style.opacity = s4Active ? '1' : '0';
            elements.s4.panel.style.opacity = s4Active ? '1' : '0';
            
            // Buttons
            btnPrev.disabled = currentStep === 0;
            btnNext.disabled = currentStep === maxStep;
        }

        btnPrev.addEventListener('click', () => {
            if (currentStep > 0) {
                currentStep--;
                updateState();
            }
        });

        btnNext.addEventListener('click', () => {
            if (currentStep < maxStep) {
                currentStep++;
                updateState();
            }
        });

        // Initialize
        updateState();
    })();
    </script>
