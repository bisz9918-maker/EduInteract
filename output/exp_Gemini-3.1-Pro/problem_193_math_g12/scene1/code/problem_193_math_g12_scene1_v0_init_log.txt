    <script type="importmap">
    {
        "imports": {
            "three": "https://cdn.jsdelivr.net/npm/three@0.160.0/build/three.module.js",
            "three/addons/": "https://cdn.jsdelivr.net/npm/three@0.160.0/examples/jsm/"
        }

    <script type="module">
    import * as THREE from 'three';
    import { OrbitControls } from 'three/addons/controls/OrbitControls.js';

    (function() {
        const container = document.getElementById('scene1-3d-container');
        const width = 540;
        const height = 500;

        const scene = new THREE.Scene();
        scene.background = new THREE.Color(0xffffff);

        const camera = new THREE.PerspectiveCamera(45, width / height, 1, 1000);
        camera.position.set(200, 150, 200);

        const renderer = new THREE.WebGLRenderer({ antialias: true });
        renderer.setSize(width, height);
        container.appendChild(renderer.domElement);

        const controls = new OrbitControls(camera, renderer.domElement);
        controls.enableDamping = true;

        const ambientLight = new THREE.AmbientLight(0xffffff, 0.6);
        scene.add(ambientLight);
        const dirLight = new THREE.DirectionalLight(0xffffff, 0.8);
        dirLight.position.set(100, 200, 100);
        scene.add(dirLight);

        // 辅助坐标系
        // const axesHelper = new THREE.AxesHelper(100);
        // scene.add(axesHelper);

        // 材质
        const matCone = new THREE.MeshLambertMaterial({ color: 0x1565c0, transparent: true, opacity: 0.2, side: THREE.DoubleSide });
        const matPyramid = new THREE.MeshLambertMaterial({ color: 0x2e7d32, transparent: true, opacity: 0.2, side: THREE.DoubleSide });
        const matLine = new THREE.LineBasicMaterial({ color: 0x000000 });
        const matDash = new THREE.LineDashedMaterial({ color: 0x888888, dashSize: 3, gapSize: 3 });

        const groupCone = new THREE.Group();
        const groupPyramid = new THREE.Group();
        scene.add(groupCone);
        scene.add(groupPyramid);

        // 尺寸参数
        const R = 40;
        const H = 80;

        // 1. 半圆锥
        // 顶点: (0, H, 0)
        // 底面半圆: 半径R, XZ平面, Z>0 (或者根据需要调整)
        // 为了方便，我们让半圆在 Z<0 区域，三角形在 Z>0 区域
        // 半圆锥底面圆心 (0,0,0)
        const coneGeo = new THREE.ConeGeometry(R, H, 32, 1, false, Math.PI, Math.PI);
        // ConeGeometry 默认中心在高度一半，底面在 -H/2，顶点在 H/2
        // 我们需要将其平移，使底面在 y=0
        coneGeo.translate(0, H/2, 0);
        // 旋转使半圆朝向 -Z
        coneGeo.rotateY(Math.PI / 2);
        
        const coneMesh = new THREE.Mesh(coneGeo, matCone);
        groupCone.add(coneMesh);

        // 半圆锥轮廓线
        const coneEdges = new THREE.EdgesGeometry(coneGeo);
        const coneLine = new THREE.LineSegments(coneEdges, matLine);
        groupCone.add(coneLine);

        // 2. 三棱锥
        // 顶点: (0, H, 0)
        // 底面三角形: 顶点 (0,0,R), 底边 (-R,0,0) 到 (R,0,0)
        const pyramidGeo = new THREE.BufferGeometry();
        const vertices = new Float32Array([
            0, H, 0,   // 0: 顶点
            -R, 0, 0,  // 1: 左下
            R, 0, 0,   // 2: 右下
            0, 0, R    // 3: 前顶点
        ]);
        const indices = [
            0, 1, 2, // 后侧面
            0, 2, 3, // 右侧面
            0, 3, 1, // 左侧面
            1, 3, 2  // 底面
        ];
        pyramidGeo.setAttribute('position', new THREE.BufferAttribute(vertices, 3));
        pyramidGeo.setIndex(indices);
        pyramidGeo.computeVertexNormals();

        const pyramidMesh = new THREE.Mesh(pyramidGeo, matPyramid);
        groupPyramid.add(pyramidMesh);

        // 三棱锥轮廓线
        const pyramidEdges = new THREE.EdgesGeometry(pyramidGeo);
        const pyramidLine = new THREE.LineSegments(pyramidEdges, matLine);
        groupPyramid.add(pyramidLine);

        // 初始状态：只显示底面
        // 我们通过缩放 Y 轴来实现高度的拉伸
        let currentHScale = 0.001;
        groupCone.scale.y = currentHScale;
        groupPyramid.scale.y = currentHScale;

        // 状态变量
        let step = 0;
        const maxStep = 3;
        let targetHScale = 0.001;
        let targetZOffset = 0;
        let currentZOffset = 0;

        const btnPrev = document.getElementById('scene1-btn-prev');
        const btnNext = document.getElementById('scene1-btn-next');
        const labelA = document.getElementById('scene1-label-a');
        const labelS1 = document.getElementById('scene1-label-s1');
        const labelS2 = document.getElementById('scene1-label-s2');

        function updateButtons() {
            btnPrev.disabled = step === 0;
            btnNext.disabled = step === maxStep;
        }

        btnPrev.addEventListener('click', () => {
            if (step > 0) step--;
            applyStep();
        });

        btnNext.addEventListener('click', () => {
            if (step < maxStep) step++;
            applyStep();
        });

        function applyStep() {
            updateButtons();
            if (step === 0) {
                targetHScale = 0.001;
                targetZOffset = 0;
                labelA.style.opacity = 0;
                labelS1.style.opacity = 0;
                labelS2.style.opacity = 0;
            } else if (step === 1) {
                targetHScale = 1;
                targetZOffset = 0;
                labelA.style.opacity = 1;
                labelS1.style.opacity = 0;
                labelS2.style.opacity = 0;
            } else if (step === 2) {
                targetHScale = 1;
                targetZOffset = 40; // 分离距离
                labelA.style.opacity = 0;
                labelS1.style.opacity = 0;
                labelS2.style.opacity = 0;
            } else if (step === 3) {
                targetHScale = 1;
                targetZOffset = 40;
                labelA.style.opacity = 0;
                labelS1.style.opacity = 1;
                labelS2.style.opacity = 1;
            }
        }

        // 标签更新函数
        function updateLabels() {
            const updateLabelPos = (label, pos3d) => {
                const pos = pos3d.clone();
                pos.project(camera);
                const x = (pos.x * .5 + .5) * width;
                const y = (pos.y * -.5 + .5) * height;
                label.style.left = `${x}px`;
                label.style.top = `${y}px`;
            };

            if (step === 1) {
                updateLabelPos(labelA, new THREE.Vector3(0, H/2, 0));
            }
            if (step === 3) {
                updateLabelPos(labelS1, new THREE.Vector3(0, -10, -currentZOffset));
                updateLabelPos(labelS2, new THREE.Vector3(0, -10, currentZOffset));
            }
        }

        function animate() {
            requestAnimationFrame(animate);

            // 动画插值
            currentHScale += (targetHScale - currentHScale) * 0.1;
            currentZOffset += (targetZOffset - currentZOffset) * 0.1;

            groupCone.scale.y = currentHScale;
            groupPyramid.scale.y = currentHScale;

            groupCone.position.z = -currentZOffset;
            groupPyramid.position.z = currentZOffset;

            controls.update();
            updateLabels();
            renderer.render(scene, camera);
        }

        animate();
    })();
    </script>
