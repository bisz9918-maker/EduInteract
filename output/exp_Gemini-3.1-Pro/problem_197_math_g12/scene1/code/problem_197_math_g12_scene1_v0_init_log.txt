    <script>
    (function() {
        let currentStep = 0;
        const maxStep = 5;
        
        const btnPrev = document.getElementById('scene1-btn-prev');
        const btnNext = document.getElementById('scene1-btn-next');
        const stepText = document.getElementById('scene1-step-text');
        
        const stepDescriptions = [
            "步骤 0 / 5：初始状态",
            "步骤 1 / 5：还原原三角形",
            "步骤 2 / 5：计算 ∠B",
            "步骤 3 / 5：中位线平行性质",
            "步骤 4 / 5：折叠全等性质",
            "步骤 5 / 5：平角计算求目标角"
        ];

        function show(id) {
            const el = document.getElementById(id);
            if (el) el.style.opacity = '1';
        }

        function hide(id) {
            const el = document.getElementById(id);
            if (el) el.style.opacity = '0';
        }

        function setStroke(id, color, width) {
            const el = document.getElementById(id);
            if (el) {
                el.setAttribute('stroke', color);
                el.setAttribute('stroke-width', width);
            }
        }

        function updateScene() {
            // Update buttons and text
            btnPrev.disabled = currentStep === 0;
            btnNext.disabled = currentStep === maxStep;
            stepText.textContent = stepDescriptions[currentStep];

            // Reset all dynamic elements
            hide('scene1-line-ad');
            hide('scene1-line-ae');
            hide('scene1-point-a');
            hide('scene1-hl-abc');
            hide('scene1-angle-a');
            hide('scene1-angle-c');
            hide('scene1-angle-b');
            hide('scene1-parallel-arrows');
            hide('scene1-angle-ade');
            hide('scene1-hl-ade');
            hide('scene1-hl-a-prime-de');
            hide('scene1-angle-a-prime-de');
            hide('scene1-hl-adb');
            hide('scene1-angle-flat-d');
            hide('scene1-angle-a-prime-db');
            
            setStroke('scene1-line-de', '#000000', '2');
            setStroke('scene1-line-bc', '#000000', '2');

            // Apply state based on currentStep
            if (currentStep >= 1) {
                show('scene1-line-ad');
                show('scene1-line-ae');
                show('scene1-point-a');
            }
            
            if (currentStep === 2) {
                show('scene1-hl-abc');
            }
            
            if (currentStep >= 2) {
                show('scene1-angle-a');
                show('scene1-angle-c');
                show('scene1-angle-b');
            }
            
            if (currentStep === 3) {
                setStroke('scene1-line-de', '#2e7d32', '4');
                setStroke('scene1-line-bc', '#2e7d32', '4');
                show('scene1-parallel-arrows');
            }
            
            if (currentStep >= 3) {
                show('scene1-angle-ade');
            }
            
            if (currentStep === 4) {
                show('scene1-hl-ade');
                show('scene1-hl-a-prime-de');
            }
            
            if (currentStep >= 4) {
                show('scene1-angle-a-prime-de');
            }
            
            if (currentStep === 5) {
                show('scene1-hl-adb');
                show('scene1-angle-flat-d');
                show('scene1-angle-a-prime-db');
            }
        }

        btnPrev.addEventListener('click', () => {
            if (currentStep > 0) {
                currentStep--;
                updateScene();
            }
        });

        btnNext.addEventListener('click', () => {
            if (currentStep < maxStep) {
                currentStep++;
                updateScene();
            }
        });

        // Initialize
        updateScene();
    })();
    </script>
