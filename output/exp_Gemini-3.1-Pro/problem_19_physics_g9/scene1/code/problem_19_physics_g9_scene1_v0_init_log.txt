    <script>
    (function() {
        let currentStep = 0;
        const maxStep = 3;
        
        const btnPrev = document.getElementById('scene1-btn-prev');
        const btnNext = document.getElementById('scene1-btn-next');
        
        const groupOriginal = document.getElementById('scene1-group-original');
        const groupEquiv = document.getElementById('scene1-group-equiv');
        const groupChart = document.getElementById('scene1-group-chart');
        
        const currentArrows = document.getElementById('scene1-current-arrows');
        const wireLeft = document.getElementById('scene1-wire-left');
        const wireRight = document.getElementById('scene1-wire-right');
        
        const sliderContainer = document.getElementById('scene1-slider-container');
        const sliderP = document.getElementById('scene1-slider-p');
        const sliderPath = document.getElementById('scene1-slider-path');
        
        const rectRap = document.getElementById('scene1-rect-rap');
        const rectRbp = document.getElementById('scene1-rect-rbp');
        const rectI1 = document.getElementById('scene1-rect-i1');
        const rectI2 = document.getElementById('scene1-rect-i2');

        function updateStep() {
            btnPrev.disabled = currentStep === 0;
            btnNext.disabled = currentStep === maxStep;
            
            // Step 0: 原始电路
            if (currentStep === 0) {
                groupOriginal.classList.add('active');
                groupEquiv.classList.remove('active');
                groupChart.classList.remove('active');
                
                currentArrows.style.opacity = '0';
                currentArrows.style.display = 'none';
                wireLeft.classList.remove('wire-red');
                wireLeft.classList.add('wire');
                wireRight.classList.remove('wire-blue');
                wireRight.classList.add('wire');
                
                sliderContainer.style.display = 'none';
            }
            // Step 1: 原始电路 + 电流动画
            else if (currentStep === 1) {
                groupOriginal.classList.add('active');
                groupEquiv.classList.remove('active');
                groupChart.classList.remove('active');
                
                currentArrows.style.opacity = '1';
                currentArrows.style.display = 'inline';
                wireLeft.classList.remove('wire');
                wireLeft.classList.add('wire-red');
                wireRight.classList.remove('wire');
                wireRight.classList.add('wire-blue');
                
                sliderContainer.style.display = 'none';
            }
            // Step 2: 等效电路
            else if (currentStep === 2) {
                groupOriginal.classList.remove('active');
                groupEquiv.classList.add('active');
                groupChart.classList.remove('active');
                
                sliderContainer.style.display = 'none';
            }
            // Step 3: 等效电路 + 图表 + 滑块
            else if (currentStep === 3) {
                groupOriginal.classList.remove('active');
                groupEquiv.classList.add('active');
                groupChart.classList.add('active');
                
                sliderContainer.style.display = 'flex';
                updateChart();
            }
        }

        function updateChart() {
            const val = parseInt(sliderP.value); // 10 to 90
            
            // 更新原始电路中 P 的位置 (仅作逻辑同步，虽然此时不可见)
            // 铅笔芯范围: 180 到 320, 长度 140
            // val 10-90 映射到 180-320
            const px = 180 + (val / 100) * 140;
            sliderPath.setAttribute('transform', `translate(${px}, 220)`);
            
            // 计算电阻和电流高度
            // 假设总电阻 R = 100, Rap = val, Rbp = 100 - val
            const rap = val;
            const rbp = 100 - val;
            
            // 高度映射: R 最大 100 -> 高度 100
            const hRap = rap;
            const hRbp = rbp;
            
            rectRap.setAttribute('height', hRap);
            rectRap.setAttribute('y', 240 - hRap);
            
            rectRbp.setAttribute('height', hRbp);
            rectRbp.setAttribute('y', 240 - hRbp);
            
            // 电流 I = U / R, 假设 U = 1000
            // I1 = 1000 / rap, I2 = 1000 / rbp
            // 限制最大高度 120
            const i1 = Math.min(120, 1000 / rap);
            const i2 = Math.min(120, 1000 / rbp);
            
            rectI1.setAttribute('height', i1);
            rectI1.setAttribute('y', 380 - i1);
            
            rectI2.setAttribute('height', i2);
            rectI2.setAttribute('y', 380 - i2);
        }

        btnPrev.addEventListener('click', () => {
            if (currentStep > 0) {
                currentStep--;
                updateStep();
            }
        });

        btnNext.addEventListener('click', () => {
            if (currentStep < maxStep) {
                currentStep++;
                updateStep();
            }
        });

        sliderP.addEventListener('input', updateChart);

        // 初始化
        updateStep();
    })();
    </script>
