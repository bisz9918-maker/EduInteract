    <script>
    (function() {
        let currentStep = 0;
        const maxStep = 3;

        const btnPrev = document.getElementById('scene1-btn-prev');
        const btnNext = document.getElementById('scene1-btn-next');
        
        const piston = document.getElementById('scene1-piston');
        const mercury = document.getElementById('scene1-mercury');
        
        const step1Group = document.getElementById('scene1-step1-group');
        const step2Group = document.getElementById('scene1-step2-group');
        const step3Group = document.getElementById('scene1-step3-group');

        function show(el) {
            el.style.display = '';
            // trigger reflow
            void el.offsetWidth;
            el.style.opacity = '1';
        }

        function hide(el) {
            el.style.opacity = '0';
            setTimeout(() => {
                if (el.style.opacity === '0') {
                    el.style.display = 'none';
                }
            }, 300);
        }

        function updateView() {
            btnPrev.disabled = currentStep === 0;
            btnNext.disabled = currentStep === maxStep;

            // Step 1: p0
            if (currentStep >= 1) {
                show(step1Group);
            } else {
                hide(step1Group);
            }

            // Step 1 highlight piston
            if (currentStep === 1) {
                piston.setAttribute('stroke', '#c62828');
                piston.setAttribute('stroke-width', '3');
            } else {
                piston.setAttribute('stroke', '#000000');
                piston.setAttribute('stroke-width', '2');
            }

            // Step 2: h
            if (currentStep >= 2) {
                show(step2Group);
            } else {
                hide(step2Group);
            }

            // Step 2 highlight mercury
            if (currentStep === 2) {
                mercury.style.opacity = '0.8';
            } else {
                mercury.style.opacity = '1';
            }

            // Step 3: p1
            if (currentStep >= 3) {
                show(step3Group);
            } else {
                hide(step3Group);
            }
        }

        btnPrev.addEventListener('click', () => {
            if (currentStep > 0) {
                currentStep--;
                updateView();
            }
        });

        btnNext.addEventListener('click', () => {
            if (currentStep < maxStep) {
                currentStep++;
                updateView();
            }
        });

        updateView();
    })();
    </script>
