    <script>
    (function() {
        const slider = document.getElementById('scene2-slider-x');
        const sliderVal = document.getElementById('scene2-slider-val');
        
        const piston = document.getElementById('scene2-piston');
        const mercuryTop = document.getElementById('scene2-mercury-top');
        const mercuryBottom = document.getElementById('scene2-mercury-bottom');
        const forceArrow = document.getElementById('scene2-force-arrow');
        const forceLabel = document.getElementById('scene2-force-label');
        
        const xLine = document.getElementById('scene2-x-line');
        const xLabel = document.getElementById('scene2-x-label');
        const xRef2 = document.getElementById('scene2-x-ref2');
        
        const x4Line = document.getElementById('scene2-4x-line');
        const x4Label = document.getElementById('scene2-4x-label');
        const x4Ref2 = document.getElementById('scene2-4x-ref2');
        
        const hLine = document.getElementById('scene2-h-line');
        const hLabel = document.getElementById('scene2-h-label');
        const hRef1 = document.getElementById('scene2-h-ref1');
        const hRef2 = document.getElementById('scene2-h-ref2');
        
        const curve = document.getElementById('scene2-curve');
        const movingPoint = document.getElementById('scene2-moving-point');
        const guideX = document.getElementById('scene2-guide-x');
        const guideY = document.getElementById('scene2-guide-y');
        
        const statusText1 = document.getElementById('scene2-status-text1');
        const statusText2 = document.getElementById('scene2-status-text2');

        // 生成曲线
        let pathD = "";
        for (let x = 0; x <= 2.4; x += 0.02) {
            let F_val = 90 - 3 * x - 2520 / (28 + x);
            let px = 380 + x * 148;
            let py = 250 - F_val * 1500;
            if (x === 0) {
                pathD += `M ${px} ${py} `;
            } else {
                pathD += `L ${px} ${py} `;
            }
        }
        curve.setAttribute('d', pathD);

        function updateScene() {
            const x = parseFloat(slider.value);
            sliderVal.textContent = x.toFixed(2) + ' cm';
            
            const F_val = 90 - 3 * x - 2520 / (28 + x);
            
            // 左侧几何更新
            const pistonY = 145 - 15 * x;
            const mercBottomY = 370 - 60 * x;
            
            piston.setAttribute('y', pistonY);
            mercuryTop.setAttribute('y', pistonY + 5);
            mercuryTop.setAttribute('height', 220 - (pistonY + 5));
            mercuryBottom.setAttribute('height', mercBottomY - 220);
            
            // F 箭头
            if (Math.abs(F_val) > 0.001) {
                forceArrow.style.opacity = 1;
                forceLabel.style.opacity = 1;
                const arrowLen = Math.abs(F_val) * 400;
                if (F_val > 0) {
                    forceArrow.setAttribute('y1', pistonY);
                    forceArrow.setAttribute('y2', pistonY - arrowLen);
                    forceArrow.setAttribute('marker-end', 'url(#scene2-arrow-up)');
                    forceLabel.setAttribute('y', pistonY - arrowLen - 10);
                } else {
                    forceArrow.setAttribute('y1', pistonY);
                    forceArrow.setAttribute('y2', pistonY + arrowLen);
                    forceArrow.setAttribute('marker-end', 'url(#scene2-arrow-down)');
                    forceLabel.setAttribute('y', pistonY - 10);
                }
            } else {
                forceArrow.style.opacity = 0;
                forceLabel.style.opacity = 0;
            }
            
            // x 标注
            if (x > 0.01) {
                xLine.style.opacity = 1;
                xLabel.style.opacity = 1;
                xRef2.style.opacity = 1;
                xLine.setAttribute('y1', 145);
                xLine.setAttribute('y2', pistonY);
                xLabel.setAttribute('y', (145 + pistonY) / 2);
                xRef2.setAttribute('y1', pistonY);
                xRef2.setAttribute('y2', pistonY);
            } else {
                xLine.style.opacity = 0;
                xLabel.style.opacity = 0;
                xRef2.style.opacity = 0;
            }
            
            // 4x 标注
            if (x > 0.01) {
                x4Line.style.opacity = 1;
                x4Label.style.opacity = 1;
                x4Ref2.style.opacity = 1;
                x4Line.setAttribute('y1', 370);
                x4Line.setAttribute('y2', mercBottomY);
                x4Label.setAttribute('y', (370 + mercBottomY) / 2);
                x4Ref2.setAttribute('y1', mercBottomY);
                x4Ref2.setAttribute('y2', mercBottomY);
            } else {
                x4Line.style.opacity = 0;
                x4Label.style.opacity = 0;
                x4Ref2.style.opacity = 0;
            }
            
            // h' 标注
            hLine.setAttribute('y1', pistonY);
            hLine.setAttribute('y2', mercBottomY);
            hLabel.setAttribute('y', (pistonY + mercBottomY) / 2);
            hRef1.setAttribute('y1', pistonY);
            hRef1.setAttribute('y2', pistonY);
            hRef2.setAttribute('y1', mercBottomY);
            hRef2.setAttribute('y2', mercBottomY);
            
            // 右侧图表更新
            const px = 380 + x * 148;
            const py = 250 - F_val * 1500;
            
            movingPoint.setAttribute('cx', px);
            movingPoint.setAttribute('cy', py);
            
            guideX.setAttribute('x1', px);
            guideX.setAttribute('y1', py);
            guideX.setAttribute('x2', px);
            guideX.setAttribute('y2', 250);
            
            guideY.setAttribute('x1', px);
            guideY.setAttribute('y1', py);
            guideY.setAttribute('x2', 380);
            guideY.setAttribute('y2', py);
            
            statusText1.textContent = `x = ${x.toFixed(2)} cm`;
            statusText2.textContent = `F ∝ ${F_val.toFixed(4)}`;
        }

        slider.addEventListener('input', updateScene);
        updateScene();
    })();
    </script>
