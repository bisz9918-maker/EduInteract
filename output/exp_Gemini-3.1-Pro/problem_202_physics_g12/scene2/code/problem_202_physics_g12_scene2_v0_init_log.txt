    <script>
    (function() {
        // Generate B field crosses
        const bfieldGroup = document.getElementById('scene2-bfield');
        for (let x = 480; x <= 720; x += 40) {
            for (let y = 60; y <= 420; y += 40) {
                // Skip crosses that overlap with labels or paths
                if (x === 480 && (y === 140 || y === 180 || y === 220 || y === 260)) continue;
                if ((x === 480 || x === 520) && y === 100) continue;
                if ((x === 520 || x === 560 || x === 600) && y === 180) continue;
                if ((x === 480 || x === 520) && y === 220) continue;
                if (x === 520 && y === 260) continue; // A' label
                if (x === 720 && y === 60) continue; // B label
                if (x === 720 && y === 140) continue; // x label
                if (x === 520 && y === 140) continue; // R label
                if (x === 600 && y === 180) continue; // O1 label
                if (x === 640 && y === 180) continue; // O1 label
                if (x === 680 && y === 180) continue; // O1 label
                if (x === 600 && y === 220) continue; // O1 label
                if (x === 640 && y === 220) continue; // O1 label
                if (x === 720 && y === 180) continue; // O1 label
                if (x === 720 && y === 220) continue; // O1 label
                if (x === 720 && y === 260) continue; // O1 label
                if (x === 720 && y === 100) continue; // x label
                if (x === 680 && y === 220) continue; // O1 label
                if (x === 560 && y === 220) continue; // O1 label

                const text = document.createElementNS('http://www.w3.org/2000/svg', 'text');
                text.setAttribute('x', x);
                text.setAttribute('y', y);
                text.setAttribute('class', 'bfield-cross');
                text.textContent = '×';
                bfieldGroup.appendChild(text);
            }
        }

        // State management
        let currentStep = 0;
        const maxStep = 2;
        const btnPrev = document.getElementById('scene2-btn-prev');
        const btnNext = document.getElementById('scene2-btn-next');
        const stepText = document.getElementById('scene2-step-text');
        
        const step1Els = document.getElementById('scene2-step1-els');
        const step2Els = document.getElementById('scene2-step2-els');
        const trajMag = document.getElementById('scene2-traj-mag');
        const trajElec = document.getElementById('scene2-traj-elec');

        // Setup path lengths for animation
        const magLen = trajMag.getTotalLength();
        trajMag.style.strokeDasharray = magLen;
        trajMag.style.strokeDashoffset = magLen;

        function show(el) {
            el.style.display = '';
            // trigger reflow
            void el.offsetWidth;
            el.style.opacity = '1';
            el.style.transition = 'opacity 0.5s';
        }

        function hide(el) {
            el.style.opacity = '0';
            setTimeout(() => {
                if (el.style.opacity === '0') {
                    el.style.display = 'none';
                }
            }, 500);
        }

        function updateScene() {
            btnPrev.disabled = currentStep === 0;
            btnNext.disabled = currentStep === maxStep;
            
            if (currentStep === 0) {
                stepText.textContent = '初始状态';
                hide(step1Els);
                hide(step2Els);
                trajMag.style.transition = 'none';
                trajMag.style.strokeDashoffset = magLen;
            } else if (currentStep === 1) {
                stepText.textContent = '磁场偏转';
                show(step1Els);
                hide(step2Els);
                
                // Animate magnetic trajectory
                setTimeout(() => {
                    trajMag.style.transition = 'stroke-dashoffset 1s ease-in-out';
                    trajMag.style.strokeDashoffset = '0';
                }, 50);
            } else if (currentStep === 2) {
                stepText.textContent = '电场平抛';
                show(step1Els);
                show(step2Els);
                trajMag.style.strokeDashoffset = '0';
            }
        }

        btnPrev.addEventListener('click', () => {
            if (currentStep > 0) {
                currentStep--;
                updateScene();
            }
        });

        btnNext.addEventListener('click', () => {
            if (currentStep < maxStep) {
                currentStep++;
                updateScene();
            }
        });

        // Initialize
        updateScene();
    })();
    </script>
