    <script>
    (function() {
        // Generate B field crosses
        const bFieldGroup = document.getElementById('scene1-b-field');
        const xs = [220, 280, 340, 400, 460, 520, 580];
        const ys = [180, 220, 280, 320];
        
        xs.forEach(x => {
            ys.forEach(y => {
                const cross = document.createElementNS('http://www.w3.org/2000/svg', 'path');
                cross.setAttribute('d', `M ${x-4} ${y-4} L ${x+4} ${y+4} M ${x-4} ${y+4} L ${x+4} ${y-4}`);
                cross.setAttribute('stroke', '#888888');
                cross.setAttribute('stroke-width', '2');
                bFieldGroup.appendChild(cross);
            });
        });
        
        let currentStep = 0;
        const maxStep = 3;
        
        const btnPrev = document.getElementById('scene1-btn-prev');
        const btnNext = document.getElementById('scene1-btn-next');
        
        const feGroup = document.getElementById('scene1-fe-group');
        const bField = document.getElementById('scene1-b-field');
        const fbGroup = document.getElementById('scene1-fb-group');
        const ionGroup = document.getElementById('scene1-ion-group');
        
        let animationTimeout = null;
        
        function updateState() {
            btnPrev.disabled = currentStep === 0;
            btnNext.disabled = currentStep === maxStep;
            
            // Reset animation
            clearTimeout(animationTimeout);
            ionGroup.style.transition = 'none';
            ionGroup.style.transform = 'translate(290px, 250px)';
            
            // Step 1: FE
            if (currentStep >= 1) {
                feGroup.style.opacity = '1';
            } else {
                feGroup.style.opacity = '0';
            }
            
            // Step 2: B field
            if (currentStep >= 2) {
                bField.style.opacity = '1';
            } else {
                bField.style.opacity = '0';
            }
            
            // Step 3: FB and animation
            if (currentStep >= 3) {
                fbGroup.style.opacity = '1';
                
                // Trigger animation after a short delay
                animationTimeout = setTimeout(() => {
                    ionGroup.style.transition = 'transform 2s linear';
                    ionGroup.style.transform = 'translate(600px, 250px)';
                }, 500);
            } else {
                fbGroup.style.opacity = '0';
            }
        }
        
        btnPrev.addEventListener('click', () => {
            if (currentStep > 0) {
                currentStep--;
                updateState();
            }
        });
        
        btnNext.addEventListener('click', () => {
            if (currentStep < maxStep) {
                currentStep++;
                updateState();
            }
        });
        
        // Initial state
        updateState();
    })();
    </script>
