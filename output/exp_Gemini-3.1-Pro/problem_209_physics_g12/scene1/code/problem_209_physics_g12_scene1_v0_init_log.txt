    <script>
    (function() {
        let currentStep = 0;
        const maxStep = 2;

        const btnPrev = document.getElementById('scene1-btn-prev');
        const btnNext = document.getElementById('scene1-btn-next');
        const lineUp = document.getElementById('scene1-line-up');
        const lineDown = document.getElementById('scene1-line-down');
        const formulaUp = document.getElementById('scene1-formula-up');
        const formulaDown = document.getElementById('scene1-formula-down');

        function updateScene() {
            // 按钮状态
            btnPrev.disabled = currentStep === 0;
            btnNext.disabled = currentStep === maxStep;

            // 步骤 0: 初始状态
            if (currentStep === 0) {
                lineUp.setAttribute('stroke', '#000000');
                lineUp.setAttribute('stroke-width', '2');
                lineDown.setAttribute('stroke', '#000000');
                lineDown.setAttribute('stroke-width', '2');
                
                formulaUp.style.opacity = '0';
                formulaDown.style.opacity = '0';
            }
            // 步骤 1: 上滑阶段
            else if (currentStep === 1) {
                lineUp.setAttribute('stroke', '#c62828');
                lineUp.setAttribute('stroke-width', '4');
                lineDown.setAttribute('stroke', '#000000');
                lineDown.setAttribute('stroke-width', '2');
                
                formulaUp.style.opacity = '1';
                formulaDown.style.opacity = '0';
            }
            // 步骤 2: 下滑阶段
            else if (currentStep === 2) {
                lineUp.setAttribute('stroke', '#c62828');
                lineUp.setAttribute('stroke-width', '4');
                lineDown.setAttribute('stroke', '#1565c0');
                lineDown.setAttribute('stroke-width', '4');
                
                formulaUp.style.opacity = '1';
                formulaDown.style.opacity = '1';
            }
        }

        btnPrev.addEventListener('click', () => {
            if (currentStep > 0) {
                currentStep--;
                updateScene();
            }
        });

        btnNext.addEventListener('click', () => {
            if (currentStep < maxStep) {
                currentStep++;
                updateScene();
            }
        });

        // 初始化
        updateScene();
    })();
    </script>
