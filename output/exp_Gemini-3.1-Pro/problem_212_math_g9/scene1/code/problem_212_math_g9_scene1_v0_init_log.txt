    <script>
    (function() {
        let currentStep = 0;
        const maxStep = 6;
        
        const btnPrev = document.getElementById('scene1-btn-prev');
        const btnNext = document.getElementById('scene1-btn-next');
        
        const polys = [
            document.getElementById('scene1-poly-1'),
            document.getElementById('scene1-poly-2'),
            document.getElementById('scene1-poly-3'),
            document.getElementById('scene1-poly-4'),
            document.getElementById('scene1-poly-5')
        ];
        
        const listItems = [
            document.getElementById('scene1-list-1'),
            document.getElementById('scene1-list-2'),
            document.getElementById('scene1-list-3'),
            document.getElementById('scene1-list-4'),
            document.getElementById('scene1-list-5')
        ];
        
        const conclusion = document.getElementById('scene1-conclusion');
        
        let hoverActive = false;
        
        function updateStep() {
            btnPrev.disabled = currentStep === 0;
            btnNext.disabled = currentStep === maxStep;
            
            if (!hoverActive) {
                // Update polygons
                polys.forEach((poly, index) => {
                    if (currentStep > 0 && currentStep <= 5 && index === currentStep - 1) {
                        poly.style.opacity = '1';
                    } else {
                        poly.style.opacity = '0';
                    }
                });
            }
            
            // Update list items
            listItems.forEach((item, index) => {
                if (currentStep > index) {
                    item.style.opacity = '1';
                } else {
                    item.style.opacity = '0';
                }
            });
            
            // Update conclusion
            if (currentStep === 6) {
                conclusion.style.opacity = '1';
            } else {
                conclusion.style.opacity = '0';
            }
        }
        
        btnPrev.addEventListener('click', () => {
            if (currentStep > 0) {
                currentStep--;
                updateStep();
            }
        });
        
        btnNext.addEventListener('click', () => {
            if (currentStep < maxStep) {
                currentStep++;
                updateStep();
            }
        });
        
        // Hover interactions
        listItems.forEach((item, index) => {
            item.addEventListener('mouseenter', () => {
                if (item.style.opacity === '1') {
                    hoverActive = true;
                    polys.forEach((poly, pIndex) => {
                        poly.style.opacity = pIndex === index ? '1' : '0';
                    });
                }
            });
            
            item.addEventListener('mouseleave', () => {
                hoverActive = false;
                updateStep();
            });
        });
        
        // Initialize
        updateStep();
    })();
    </script>
