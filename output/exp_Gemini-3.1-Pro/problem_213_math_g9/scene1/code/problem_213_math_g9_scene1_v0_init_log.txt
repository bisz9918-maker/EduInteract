    <script>
    (function() {
        let currentStep = 0;
        const maxStep = 3;

        const btnPrev = document.getElementById('scene1-btn-prev');
        const btnNext = document.getElementById('scene1-btn-next');

        const elements = {
            link2: document.getElementById('scene1-link2'),
            link3: document.getElementById('scene1-link3'),
            ellipsis: document.getElementById('scene1-ellipsis'),
            link100: document.getElementById('scene1-link100'),
            hl1: document.getElementById('scene1-hl1'),
            hl2: document.getElementById('scene1-hl2'),
            hl3: document.getElementById('scene1-hl3'),
            dim08_1: document.getElementById('scene1-dim-0.8-1'),
            dim08_2: document.getElementById('scene1-dim-0.8-2'),
            dim08_3: document.getElementById('scene1-dim-0.8-3'),
            brace1: document.getElementById('scene1-brace-1'),
            brace2: document.getElementById('scene1-brace-2'),
            brace3: document.getElementById('scene1-brace-3'),
            brace100: document.getElementById('scene1-brace-100'),
            form1: document.getElementById('scene1-form-1'),
            form2: document.getElementById('scene1-form-2'),
            form3: document.getElementById('scene1-form-3'),
            form100: document.getElementById('scene1-form-100'),
            formN: document.getElementById('scene1-form-n')
        };

        function show(el) {
            if(el) {
                el.style.display = '';
                el.style.opacity = '1';
            }
        }

        function hide(el) {
            if(el) {
                el.style.display = 'none';
                el.style.opacity = '0';
            }
        }

        function updateScene() {
            btnPrev.disabled = currentStep === 0;
            btnNext.disabled = currentStep === maxStep;

            // Reset all
            Object.values(elements).forEach(hide);

            // Step 0
            if (currentStep >= 0) {
                if (currentStep === 0) {
                    show(elements.brace1);
                    show(elements.form1);
                }
            }

            // Step 1
            if (currentStep >= 1) {
                show(elements.link2);
                show(elements.hl1);
                show(elements.dim08_1);
                if (currentStep === 1) {
                    show(elements.brace2);
                    show(elements.form2);
                }
            }

            // Step 2
            if (currentStep >= 2) {
                show(elements.link3);
                show(elements.hl2);
                show(elements.dim08_2);
                if (currentStep === 2) {
                    show(elements.brace3);
                    show(elements.form3);
                }
            }

            // Step 3
            if (currentStep >= 3) {
                show(elements.ellipsis);
                show(elements.link100);
                show(elements.hl3);
                show(elements.dim08_3);
                show(elements.brace100);
                show(elements.form100);
                show(elements.formN);
            }
        }

        btnPrev.addEventListener('click', () => {
            if (currentStep > 0) {
                currentStep--;
                updateScene();
            }
        });

        btnNext.addEventListener('click', () => {
            if (currentStep < maxStep) {
                currentStep++;
                updateScene();
            }
        });

        // Init
        updateScene();
    })();
    </script>
