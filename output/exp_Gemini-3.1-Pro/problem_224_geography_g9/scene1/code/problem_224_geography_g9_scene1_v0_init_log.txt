    <script>
    (function() {
        let currentStep = 0;
        const maxStep = 3;
        
        const btnPrev = document.getElementById('scene1-btn-prev');
        const btnNext = document.getElementById('scene1-btn-next');
        
        const morningLine = document.getElementById('scene1-morning-line');
        const nightShadow = document.getElementById('scene1-night-shadow');
        const polarHighlight = document.getElementById('scene1-polar-highlight');
        
        const promptMorning = document.getElementById('scene1-prompt-morning');
        const promptNorth = document.getElementById('scene1-prompt-north');
        const promptSouth = document.getElementById('scene1-prompt-south');
        
        function updateScene() {
            btnPrev.disabled = currentStep === 0;
            btnNext.disabled = currentStep === maxStep;
            
            // Step 1: 晨线高亮
            const showStep1 = currentStep >= 1 ? '1' : '0';
            morningLine.style.opacity = showStep1;
            promptMorning.style.opacity = showStep1;
            
            // Step 2: 夜半球阴影
            const showStep2 = currentStep >= 2 ? '1' : '0';
            nightShadow.style.opacity = showStep2;
            
            // Step 3: 极圈高亮
            const showStep3 = currentStep >= 3 ? '1' : '0';
            polarHighlight.style.opacity = showStep3;
            promptNorth.style.opacity = showStep3;
            promptSouth.style.opacity = showStep3;
        }
        
        btnPrev.addEventListener('click', () => {
            if (currentStep > 0) {
                currentStep--;
                updateScene();
            }
        });
        
        btnNext.addEventListener('click', () => {
            if (currentStep < maxStep) {
                currentStep++;
                updateScene();
            }
        });
        
        updateScene();
    })();
    </script>
