    <script>
    (function() {
        let currentStep = 0;
        const maxStep = 3;
        
        const btnPrev = document.getElementById('scene1-btn-prev');
        const btnNext = document.getElementById('scene1-btn-next');
        
        const step1Els = document.querySelectorAll('.step1-el');
        const step2Els = document.querySelectorAll('.step2-el');
        const step3Els = document.querySelectorAll('.step3-el');
        
        function updateScene() {
            btnPrev.disabled = currentStep === 0;
            btnNext.disabled = currentStep === maxStep;
            
            step1Els.forEach(el => el.style.opacity = currentStep >= 1 ? '1' : '0');
            step2Els.forEach(el => el.style.opacity = currentStep >= 2 ? '1' : '0');
            step3Els.forEach(el => el.style.opacity = currentStep >= 3 ? '1' : '0');
        }
        
        btnPrev.addEventListener('click', () => {
            if (currentStep > 0) {
                currentStep--;
                updateScene();
            }
        });
        
        btnNext.addEventListener('click', () => {
            if (currentStep < maxStep) {
                currentStep++;
                updateScene();
            }
        });
        
        updateScene();
    })();
    </script>
