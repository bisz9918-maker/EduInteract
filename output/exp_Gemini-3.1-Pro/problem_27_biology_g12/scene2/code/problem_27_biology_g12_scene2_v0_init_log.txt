    <script>
    (function() {
        let currentStep = 0;
        const maxStep = 4;

        const btnPrev = document.getElementById('scene2-btn-prev');
        const btnNext = document.getElementById('scene2-btn-next');

        // Elements
        const chrLwOut = document.getElementById('scene2-chr-lw-out');
        const chrLbOut = document.getElementById('scene2-chr-lb-out');
        const chrSwOut = document.getElementById('scene2-chr-sw-out');
        const chrSbOut = document.getElementById('scene2-chr-sb-out');
        
        const geneA = document.getElementById('scene2-gene-A');
        const genea = document.getElementById('scene2-gene-a');
        const geneB = document.getElementById('scene2-gene-B');
        const geneSry = document.getElementById('scene2-gene-sry');
        
        const lineA = document.getElementById('scene2-line-A');
        const linea = document.getElementById('scene2-line-a');
        const lineB = document.getElementById('scene2-line-B');
        const lineSry = document.getElementById('scene2-line-sry');
        
        const sry = document.getElementById('scene2-sry');
        
        const panelLeft = document.getElementById('scene2-panel-left');
        const panelRight = document.getElementById('scene2-panel-right');
        const panelRightExtra1 = document.getElementById('scene2-panel-right-extra1');
        const panelRightExtra2 = document.getElementById('scene2-panel-right-extra2');
        const panelRightExtra3 = document.getElementById('scene2-panel-right-extra3');
        
        const conclusion = document.getElementById('scene2-conclusion');

        function updateStep() {
            btnPrev.disabled = currentStep === 0;
            btnNext.disabled = currentStep === maxStep;

            // Reset all
            chrLwOut.setAttribute('stroke', '#000000');
            chrLbOut.setAttribute('stroke', '#000000');
            chrSwOut.setAttribute('stroke', '#000000');
            chrSbOut.setAttribute('stroke', '#000000');
            
            geneA.setAttribute('fill', '#000000');
            geneA.setAttribute('font-size', '16');
            geneA.setAttribute('font-weight', 'normal');
            lineA.setAttribute('stroke', '#000000');
            
            genea.setAttribute('fill', '#000000');
            genea.setAttribute('font-size', '16');
            genea.setAttribute('font-weight', 'normal');
            linea.setAttribute('stroke', '#000000');
            
            geneB.setAttribute('fill', '#000000');
            geneB.setAttribute('font-size', '16');
            geneB.setAttribute('font-weight', 'normal');
            lineB.setAttribute('stroke', '#000000');
            
            geneSry.setAttribute('fill', '#000000');
            geneSry.setAttribute('font-size', '16');
            geneSry.setAttribute('font-weight', 'normal');
            lineSry.setAttribute('stroke', '#000000');
            
            sry.setAttribute('stroke', '#000000');
            sry.setAttribute('stroke-width', '2');
            
            panelLeft.style.opacity = '0';
            panelRight.style.opacity = '0';
            panelRightExtra1.style.opacity = '0';
            panelRightExtra2.style.opacity = '0';
            panelRightExtra3.style.opacity = '0';
            
            conclusion.style.opacity = '0';

            if (currentStep >= 1) {
                chrLwOut.setAttribute('stroke', '#2e7d32');
                chrLbOut.setAttribute('stroke', '#2e7d32');
                
                geneA.setAttribute('fill', '#c62828');
                geneA.setAttribute('font-size', '20');
                geneA.setAttribute('font-weight', 'bold');
                lineA.setAttribute('stroke', '#c62828');
                
                genea.setAttribute('fill', '#c62828');
                genea.setAttribute('font-size', '20');
                genea.setAttribute('font-weight', 'bold');
                linea.setAttribute('stroke', '#c62828');
                
                panelLeft.style.opacity = '1';
            }
            
            if (currentStep >= 2) {
                chrSwOut.setAttribute('stroke', '#1565c0');
                chrSbOut.setAttribute('stroke', '#1565c0');
                panelRight.style.opacity = '1';
            }
            
            if (currentStep >= 3) {
                geneB.setAttribute('fill', '#c62828');
                geneB.setAttribute('font-size', '20');
                geneB.setAttribute('font-weight', 'bold');
                lineB.setAttribute('stroke', '#c62828');
                
                geneSry.setAttribute('fill', '#c62828');
                geneSry.setAttribute('font-size', '20');
                geneSry.setAttribute('font-weight', 'bold');
                lineSry.setAttribute('stroke', '#c62828');
                
                sry.setAttribute('stroke', '#c62828');
                sry.setAttribute('stroke-width', '3');
                
                panelRightExtra1.style.opacity = '1';
                panelRightExtra2.style.opacity = '1';
                panelRightExtra3.style.opacity = '1';
            }
            
            if (currentStep >= 4) {
                conclusion.style.opacity = '1';
            }
        }

        btnPrev.addEventListener('click', () => {
            if (currentStep > 0) {
                currentStep--;
                updateStep();
            }
        });

        btnNext.addEventListener('click', () => {
            if (currentStep < maxStep) {
                currentStep++;
                updateStep();
            }
        });

        // Initialize
        updateStep();
    })();
    </script>
