    <script>
    (function() {
        let currentStep = 0;
        const maxStep = 4;
        
        const prevBtn = document.getElementById('scene3-prev-btn');
        const nextBtn = document.getElementById('scene3-next-btn');
        
        const step1 = document.getElementById('scene3-step1-group');
        const step2 = document.getElementById('scene3-step2-group');
        const step3 = document.getElementById('scene3-step3-group');
        const step4 = document.getElementById('scene3-step4-group');
        const mask = document.getElementById('scene3-mask');
        const finalText2 = document.getElementById('scene3-final-text2');
        
        function updateScene() {
            prevBtn.disabled = currentStep === 0;
            nextBtn.disabled = currentStep === maxStep;
            
            step1.style.opacity = currentStep >= 1 ? '1' : '0';
            step2.style.opacity = currentStep >= 2 ? '1' : '0';
            step3.style.opacity = currentStep >= 3 ? '1' : '0';
            step4.style.opacity = currentStep >= 4 ? '1' : '0';
            
            if (currentStep >= 4) {
                setTimeout(() => {
                    mask.style.opacity = '0.3';
                    finalText2.innerHTML = '1/4 aaDd <tspan fill="#c62828">(无毛)</tspan>';
                }, 300);
            } else {
                mask.style.opacity = '0';
                finalText2.innerHTML = '1/4 aaDd (无毛)';
            }
        }
        
        prevBtn.addEventListener('click', () => {
            if (currentStep > 0) {
                currentStep--;
                updateScene();
            }
        });
        
        nextBtn.addEventListener('click', () => {
            if (currentStep < maxStep) {
                currentStep++;
                updateScene();
            }
        });
        
        updateScene();
    })();
    </script>
