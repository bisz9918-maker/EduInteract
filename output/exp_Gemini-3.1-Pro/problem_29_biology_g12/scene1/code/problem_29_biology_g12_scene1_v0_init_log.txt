    <script>
    (function() {
        let currentStep = 0;
        const maxStep = 3;

        const btnPrev = document.getElementById('scene1-btn-prev');
        const btnNext = document.getElementById('scene1-btn-next');

        const step1Els = [document.getElementById('scene1-arrow95'), document.getElementById('scene1-step1')];
        const step2Els = [document.getElementById('scene1-arrow55'), document.getElementById('scene1-step2')];
        const step3Els = [document.getElementById('scene1-arrow72'), document.getElementById('scene1-step3')];
        
        const primer2Blank = document.getElementById('scene1-primer2-blank');
        const primer2Ans = document.getElementById('scene1-primer2-ans');

        function show(el) {
            if (!el) return;
            el.style.display = '';
            // small delay to allow display to apply before opacity transition
            setTimeout(() => { el.style.opacity = '1'; }, 10);
        }

        function hide(el) {
            if (!el) return;
            el.style.opacity = '0';
            setTimeout(() => { el.style.display = 'none'; }, 300);
        }

        function updateState() {
            btnPrev.disabled = currentStep === 0;
            btnNext.disabled = currentStep === maxStep;

            if (currentStep >= 1) {
                step1Els.forEach(show);
            } else {
                step1Els.forEach(hide);
            }

            if (currentStep >= 2) {
                step2Els.forEach(show);
                hide(primer2Blank);
                show(primer2Ans);
            } else {
                step2Els.forEach(hide);
                show(primer2Blank);
                hide(primer2Ans);
            }

            if (currentStep >= 3) {
                step3Els.forEach(show);
            } else {
                step3Els.forEach(hide);
            }
        }

        btnPrev.addEventListener('click', () => {
            if (currentStep > 0) {
                currentStep--;
                updateState();
            }
        });

        btnNext.addEventListener('click', () => {
            if (currentStep < maxStep) {
                currentStep++;
                updateState();
            }
        });

        // Initialize
        updateState();
    })();
    </script>
