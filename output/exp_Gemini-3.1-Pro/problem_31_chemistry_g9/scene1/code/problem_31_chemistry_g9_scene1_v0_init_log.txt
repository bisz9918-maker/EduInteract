            "three": "https://cdn.jsdelivr.net/npm/three@0.160.0/build/three.module.js",

        <div id="scene1-formula">\( (\sqrt{2}a_1)^2 + a_1^2 = (2d)^2 \Rightarrow \sqrt{3}a_1 = 2d \)</div>
        
        <!-- Labels -->
        <div id="scene1-label-center" class="scene1-label">中心原子</div>

        <div id="scene1-label-sqrt2a1" class="scene1-label">\( \sqrt{2}a_1 \)</div>

        <div id="scene1-controls">
            <button id="scene1-btn-prev" disabled>◀ 上一步</button>
            <button id="scene1-btn-next">下一步 ▶</button>
        </div>
    </div>

    <script type="module">
        import * as THREE from 'three';
        import { OrbitControls } from 'three/addons/controls/OrbitControls.js';

        const container = document.getElementById('scene1-container');
        const width = 800;
        const height = 500;

        // Scene, Camera, Renderer
        const scene = new THREE.Scene();
        scene.background = new THREE.Color(0xffffff);

        const camera = new THREE.PerspectiveCamera(45, width / height, 1, 2000);
        camera.position.set(350, 250, 450);
        camera.lookAt(0, 0, 0);

        const renderer = new THREE.WebGLRenderer({ antialias: true });
        renderer.setSize(width, height);
        // Insert canvas before controls
        container.insertBefore(renderer.domElement, document.getElementById('scene1-controls'));

        const controls = new OrbitControls(camera, renderer.domElement);
        controls.enableDamping = true;
        controls.dampingFactor = 0.05;

        // Lighting
        const ambientLight = new THREE.AmbientLight(0xffffff, 0.6);
        scene.add(ambientLight);
        const dirLight = new THREE.DirectionalLight(0xffffff, 0.8);
        dirLight.position.set(200, 500, 300);
        scene.add(dirLight);

        // Materials
        const matGrey = new THREE.MeshPhongMaterial({ color: 0x888888 });
        const matRed = new THREE.MeshPhongMaterial({ color: 0xc62828 });
        const matBlue = new THREE.MeshPhongMaterial({ color: 0x1565c0 });
        const matLineBlack = new THREE.LineBasicMaterial({ color: 0x000000, linewidth: 2 });
        const matLineBlueDashed = new THREE.LineDashedMaterial({ color: 0x1565c0, dashSize: 5, gapSize: 5 });
        const matLineRedThick = new THREE.LineBasicMaterial({ color: 0xc62828, linewidth: 4 }); // WebGL linewidth might not work, but we try
        const matPlaneBlue = new THREE.MeshBasicMaterial({ color: 0x1565c0, transparent: true, opacity: 0.2, side: THREE.DoubleSide });

        // Geometry
        const sphereGeo = new THREE.SphereGeometry(15, 32, 32);

        // Coordinates
        const a = 100;
        const vertices = [
            new THREE.Vector3(-a, -a, -a), // V0
            new THREE.Vector3(a, -a, -a),  // V1
            new THREE.Vector3(a, a, -a),   // V2
            new THREE.Vector3(-a, a, -a),  // V3
            new THREE.Vector3(-a, -a, a),  // V4
            new THREE.Vector3(a, -a, a),   // V5
            new THREE.Vector3(a, a, a),    // V6
            new THREE.Vector3(-a, a, a)    // V7
        ];
        const center = new THREE.Vector3(0, 0, 0);

        // Atoms
        const atoms = [];
        for (let i = 0; i < 8; i++) {
            const mesh = new THREE.Mesh(sphereGeo, matGrey);
            mesh.position.copy(vertices[i]);
            scene.add(mesh);
            atoms.push(mesh);
        }
        const centerAtom = new THREE.Mesh(sphereGeo, matGrey);
        centerAtom.position.copy(center);
        scene.add(centerAtom);

        // Wireframe (Cube edges)
        const edges = [
            [0,1], [1,2], [2,3], [3,0],
            [4,5], [5,6], [6,7], [7,4],
            [0,4], [1,5], [2,6], [3,7]
        ];
        const cubeLines = new THREE.Group();
        edges.forEach(edge => {
            const geo = new THREE.BufferGeometry().setFromPoints([vertices[edge[0]], vertices[edge[1]]]);
            const line = new THREE.Line(geo, matLineBlack);
            cubeLines.add(line);
        });
        scene.add(cubeLines);

        // Dashed lines to corners
        const dashedLines = new THREE.Group();
        for (let i = 0; i < 8; i++) {
            const geo = new THREE.BufferGeometry().setFromPoints([center, vertices[i]]);
            const line = new THREE.Line(geo, matLineBlueDashed);
            line.computeLineDistances();
            dashedLines.add(line);
        }
        scene.add(dashedLines);
        dashedLines.visible = false;

        // Diagonal Plane V0-V3-V6-V5
        const planeGeo = new THREE.BufferGeometry();
        const planeVertices = new Float32Array([
            -a, -a, -a,  // V0
            -a, a, -a,   // V3
            a, a, a,     // V6
            
            -a, -a, -a,  // V0
            a, a, a,     // V6
            a, -a, a     // V5
        ]);
        planeGeo.setAttribute('position', new THREE.BufferAttribute(planeVertices, 3));
        const diagPlane = new THREE.Mesh(planeGeo, matPlaneBlue);
        scene.add(diagPlane);
        diagPlane.visible = false;

        // Body diagonal V0-O-V6
        const bodyDiagGeo = new THREE.BufferGeometry().setFromPoints([vertices[0], vertices[6]]);
        const bodyDiagLine = new THREE.Line(bodyDiagGeo, matLineRedThick);
        scene.add(bodyDiagLine);
        bodyDiagLine.visible = false;

        // Right triangle V0-V5-V6
        const triangleLines = new THREE.Group();
        const triEdges = [[0,5], [5,6], [6,0]];
        triEdges.forEach(edge => {
            const geo = new THREE.BufferGeometry().setFromPoints([vertices[edge[0]], vertices[edge[1]]]);
            const line = new THREE.Line(geo, matLineRedThick);
            triangleLines.add(line);
        });
        scene.add(triangleLines);
        triangleLines.visible = false;

        // Labels Data
        const labelsData = [
            { id: 'scene1-label-center', pos: center, offset: {x: 20, y: -20}, step: 1 },
            { id: 'scene1-label-corner', pos: vertices[6], offset: {x: 20, y: -20}, step: 1 },
            { id: 'scene1-label-d', pos: new THREE.Vector3(50, 50, 50), offset: {x: -15, y: -15}, step: 2 },
            { id: 'scene1-label-2d', pos: new THREE.Vector3(-50, -50, -50), offset: {x: 15, y: 15}, step: 2 },
            { id: 'scene1-label-a1', pos: new THREE.Vector3(100, 0, 100), offset: {x: 20, y: 0}, step: 3 },
            { id: 'scene1-label-sqrt2a1', pos: new THREE.Vector3(0, -100, 0), offset: {x: 0, y: 20}, step: 3 }
        ];

        let currentStep = 0;
        const maxStep = 3;

        function updateStep() {
            // Reset all
            atoms.forEach(a => { a.material = matGrey; a.visible = true; });
            centerAtom.material = matGrey;
            dashedLines.visible = false;
            diagPlane.visible = false;
            bodyDiagLine.visible = false;
            triangleLines.visible = false;
            document.getElementById('scene1-formula').style.opacity = 0;

            if (currentStep >= 1) {
                centerAtom.material = matRed;
                atoms.forEach(a => a.material = matBlue);
                dashedLines.visible = true;
            }
            if (currentStep >= 2) {
                // Hide 6 corner atoms and their dashed lines
                atoms.forEach((a, i) => {
                    if (i !== 0 && i !== 6) {
                        a.visible = false;
                        dashedLines.children[i].visible = false;
                    }
                });
                diagPlane.visible = true;
                bodyDiagLine.visible = true;
            }
            if (currentStep >= 3) {
                triangleLines.visible = true;
                document.getElementById('scene1-formula').style.opacity = 1;
            }

            // Update buttons
            document.getElementById('scene1-btn-prev').disabled = currentStep === 0;
            document.getElementById('scene1-btn-next').disabled = currentStep === maxStep;
        }

        document.getElementById('scene1-btn-prev').addEventListener('click', () => {
            if (currentStep > 0) { currentStep--; updateStep(); }
        });
        document.getElementById('scene1-btn-next').addEventListener('click', () => {
            if (currentStep < maxStep) { currentStep++; updateStep(); }
        });

        function updateLabels() {
            labelsData.forEach(data => {
                const el = document.getElementById(data.id);
                if (currentStep >= data.step) {
                    const vector = data.pos.clone();
                    vector.project(camera);
                    
                    const x = (vector.x * .5 + .5) * width;
                    const y = (vector.y * -.5 + .5) * height;
                    
                    // Check if behind camera
                    if (vector.z > 1) {
                        el.style.opacity = 0;
                    } else {
                        el.style.opacity = 1;
                        el.style.left = (x + data.offset.x) + 'px';
                        el.style.top = (y + data.offset.y) + 'px';
                    }
                } else {
                    el.style.opacity = 0;
                }
            });
        }

        function animate() {
            requestAnimationFrame(animate);
            controls.update();
            updateLabels();
            renderer.render(scene, camera);
        }

        updateStep();
        animate();
    </script>
