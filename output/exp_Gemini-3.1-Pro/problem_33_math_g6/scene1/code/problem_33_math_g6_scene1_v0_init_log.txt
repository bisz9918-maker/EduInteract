    <script>
    (function() {
        // Draw grid
        const gridGroup = document.getElementById('scene1-grid');
        for (let i = 0; i <= 14; i++) {
            const x = 120 + i * 40;
            const line = document.createElementNS('http://www.w3.org/2000/svg', 'line');
            line.setAttribute('x1', x);
            line.setAttribute('y1', 80);
            line.setAttribute('x2', x);
            line.setAttribute('y2', 320);
            line.setAttribute('class', 'grid-line');
            if (i === 0 || i === 14) line.setAttribute('stroke-dasharray', '4,4'); // Optional border style
            gridGroup.appendChild(line);
        }
        for (let i = 0; i <= 6; i++) {
            const y = 80 + i * 40;
            const line = document.createElementNS('http://www.w3.org/2000/svg', 'line');
            line.setAttribute('x1', 120);
            line.setAttribute('y1', y);
            line.setAttribute('x2', 680);
            line.setAttribute('y2', y);
            line.setAttribute('class', 'grid-line');
            if (i === 0 || i === 6) line.setAttribute('stroke-dasharray', '4,4');
            gridGroup.appendChild(line);
        }

        let currentStep = 0;
        const maxStep = 4;
        
        const btnPrev = document.getElementById('scene1-btn-prev');
        const btnNext = document.getElementById('scene1-btn-next');
        
        function show(id) {
            const el = document.getElementById(id);
            if (el) el.style.opacity = 1;
        }
        function hide(id) {
            const el = document.getElementById(id);
            if (el) el.style.opacity = 0;
        }
        function setTransform(id, transform) {
            const el = document.getElementById(id);
            if (el) el.style.transform = transform;
        }
        function setClass(id, className, add) {
            const el = document.getElementById(id);
            if (el) {
                if (add) el.classList.add(className);
                else el.classList.remove(className);
            }
        }
        function setText(id, text) {
            const el = document.getElementById(id);
            if (el) el.innerHTML = text;
        }

        function updateState() {
            btnPrev.disabled = currentStep === 0;
            btnNext.disabled = currentStep === maxStep;
            
            // Reset all
            const allLabels = [
                'scene1-jia-l-left', 'scene1-jia-l-bottom', 'scene1-jia-l-right', 'scene1-jia-l-top-left', 'scene1-jia-l-top-right', 'scene1-jia-l-inner-left', 'scene1-jia-l-inner-right', 'scene1-jia-l-inner-bottom',
                'scene1-yi-l-left', 'scene1-yi-l-bottom', 'scene1-yi-l-right', 'scene1-yi-l-top', 'scene1-yi-l-top-left', 'scene1-yi-l-top-right', 'scene1-yi-l-inner-left', 'scene1-yi-l-inner-right'
            ];
            allLabels.forEach(hide);
            
            hide('scene1-jia-bbox');
            hide('scene1-yi-bbox');
            hide('scene1-jia-formula');
            hide('scene1-yi-formula');
            hide('scene1-conclusion');
            
            // Reset transforms and classes
            const allLines = [
                'scene1-jia-inner-bottom', 'scene1-jia-inner-left', 'scene1-jia-inner-right',
                'scene1-yi-top-left', 'scene1-yi-top-right', 'scene1-yi-inner-left', 'scene1-yi-inner-right'
            ];
            allLines.forEach(id => {
                setTransform(id, 'translate(0, 0)');
                setClass(id, 'highlight-blue', false);
                setClass(id, 'highlight-red', false);
            });
            
            if (currentStep >= 1) {
                if (currentStep === 1) {
                    allLabels.forEach(show);
                }
                show('scene1-jia-formula');
                show('scene1-yi-formula');
                setText('scene1-jia-formula', '甲的周长 = 4+5+4+1+2+3+2+1 = 22');
                setText('scene1-yi-formula', '乙的周长 = 2+5+2+1+2+3+2+1 = 18');
                setClass('scene1-yi-formula', 'text-blue', false);
            }
            
            if (currentStep >= 2) {
                show('scene1-yi-bbox');
                setClass('scene1-yi-top-left', 'highlight-blue', true);
                setClass('scene1-yi-top-right', 'highlight-blue', true);
                setClass('scene1-yi-inner-left', 'highlight-blue', true);
                setClass('scene1-yi-inner-right', 'highlight-blue', true);
                
                setTransform('scene1-yi-top-left', 'translateY(-80px)');
                setTransform('scene1-yi-top-right', 'translateY(-80px)');
                setTransform('scene1-yi-inner-left', 'translateX(-40px)');
                setTransform('scene1-yi-inner-right', 'translateX(40px)');
                
                setText('scene1-yi-formula', '乙的周长 = 外接矩形周长 = (5+4)×2 = 18');
                setClass('scene1-yi-formula', 'text-blue', true);
            }
            
            if (currentStep >= 3) {
                show('scene1-jia-bbox');
                setClass('scene1-jia-inner-bottom', 'highlight-blue', true);
                setTransform('scene1-jia-inner-bottom', 'translateY(-80px)');
                
                setText('scene1-jia-formula', '甲的周长 = 外接矩形周长 + <tspan class="text-red">2条竖直边</tspan>');
            }
            
            if (currentStep >= 4) {
                setClass('scene1-jia-inner-left', 'highlight-red', true);
                setClass('scene1-jia-inner-right', 'highlight-red', true);
                
                setText('scene1-jia-formula', '甲的周长 = 18 + <tspan class="text-red">2</tspan> + <tspan class="text-red">2</tspan> = <tspan class="text-red">22</tspan>');
                show('scene1-conclusion');
            }
        }

        btnPrev.addEventListener('click', () => {
            if (currentStep > 0) {
                currentStep--;
                updateState();
            }
        });
        
        btnNext.addEventListener('click', () => {
            if (currentStep < maxStep) {
                currentStep++;
                updateState();
            }
        });

        updateState();
    })();
    </script>
