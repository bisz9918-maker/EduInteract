    <style>
        body { margin: 0; padding: 0; background: #ffffff; font-family: sans-serif; }
        #scene1-container { position: relative; width: 800px; height: 500px; background: #ffffff; margin: 0 auto; overflow: hidden; }
        #scene1-controls { position: absolute; bottom: 10px; left: 0; right: 0; display: flex; justify-content: center; gap: 12px; z-index: 10; }
        button { padding: 8px 16px; font-size: 14px; cursor: pointer; border: 1px solid #ccc; background: #f9f9f9; border-radius: 4px; }
        button:disabled { opacity: 0.5; cursor: not-allowed; }
        button:hover:not(:disabled) { background: #e9e9e9; }

<body>
    <div id="scene1-container">
        <svg id="scene1-svg" width="800" height="500">
            <g id="scene1-ghost-circle" style="opacity: 0;">
                <circle cx="200" cy="240" r="100" fill="none" stroke="#888888" stroke-width="2" stroke-dasharray="5,5" />
            </g>
            <g id="scene1-sectors"></g>
            <g id="scene1-radius-group">
                <line id="scene1-radius-line" x1="200" y1="240" x2="129.29" y2="169.29" stroke="#000" stroke-width="2" />
                <text id="scene1-radius-label" x="80" y="140" font-size="16" fill="#000">r=10</text>
            </g>
            <g id="scene1-highlights">
                <path id="scene1-top-red" fill="none" stroke="#c62828" stroke-width="3" style="opacity: 0;" />
                <path id="scene1-bottom-red" fill="none" stroke="#c62828" stroke-width="3" style="opacity: 0;" />
                <line id="scene1-left-blue" stroke="#1565c0" stroke-width="3" style="opacity: 0;" />
                <line id="scene1-right-blue" stroke="#1565c0" stroke-width="3" style="opacity: 0;" />
            </g>
            <g id="scene1-labels">
                <text id="scene1-top-pir" x="577" y="175" font-size="16" fill="#c62828" text-anchor="middle" style="opacity: 0;">πr</text>
                <text id="scene1-bottom-pir" x="577" y="315" font-size="16" fill="#c62828" text-anchor="middle" style="opacity: 0;">πr</text>
                <text id="scene1-left-r" x="400" y="245" font-size="16" fill="#1565c0" text-anchor="end" style="opacity: 0;">r</text>
                <text id="scene1-right-r" x="760" y="245" font-size="16" fill="#1565c0" text-anchor="start" style="opacity: 0;">r</text>
                <text id="scene1-conclusion" x="400" y="420" font-size="20" font-weight="bold" fill="#000" text-anchor="middle" style="opacity: 0;">长方形周长 = 2πr + 2r</text>
            </g>
        </svg>
        <div id="scene1-controls">
            <button id="scene1-btn-prev" disabled>◀ 上一步</button>
            <button id="scene1-btn-next">下一步 ▶</button>
        </div>
    </div>
    <script>
    (function() {
        const svg = d3.select("#scene1-svg");
        const cx = 200, cy = 240, r = 100;
        const rectX = 420, rectY = 190;
        const numSectors = 16;
        const angle = 2 * Math.PI / numSectors;
        const arcLength = r * angle;
        
        const sectorsData = [];
        for (let i = 0; i < numSectors; i++) {
            const isTop = i >= 8;
            const color = isTop ? "#ffffff" : "#cccccc";
            
            // Initial state
            const initAngle = i * angle + angle / 2;
            const initDeg = initAngle * 180 / Math.PI;
            
            // Explode state
            const explodeDist = 10;
            const explodeX = cx + explodeDist * Math.cos(initAngle);
            const explodeY = cy + explodeDist * Math.sin(initAngle);
            
            // Target state
            let targetX, targetY, targetDeg;
            if (isTop) {
                // White sectors (top half) -> point down in rectangle
                const k = i - 8; // 0 to 7
                targetX = rectX + k * arcLength + arcLength / 2;
                targetY = rectY + 100; // 290
                targetDeg = -90;
            } else {
                // Gray sectors (bottom half) -> point up in rectangle
                const k = i; // 0 to 7
                targetX = rectX + k * arcLength + arcLength;
                targetY = rectY; // 190
                targetDeg = 90;
            }
            
            sectorsData.push({
                id: i,
                isTop: isTop,
                color: color,
                initX: cx,
                initY: cy,
                initDeg: initDeg,
                explodeX: explodeX,
                explodeY: explodeY,
                targetX: targetX,
                targetY: targetY,
                targetDeg: targetDeg
            });
        }
        
        // Draw sectors
        const sectorPath = `M 0 0 L ${r * Math.cos(-angle/2)} ${r * Math.sin(-angle/2)} A ${r} ${r} 0 0 1 ${r * Math.cos(angle/2)} ${r * Math.sin(angle/2)} Z`;
        
        const sectors = d3.select("#scene1-sectors")
            .selectAll("path")
            .data(sectorsData)
            .enter()
            .append("path")
            .attr("d", sectorPath)
            .attr("fill", d => d.color)
            .attr("stroke", "#000")
            .attr("stroke-width", 1)
            .attr("transform", d => `translate(${d.initX}, ${d.initY}) rotate(${d.initDeg})`);
            
        // Morphing paths for red lines
        const numPoints = 50;
        function getArcPoints(isTop) {
            const pts = [];
            const startAngle = isTop ? Math.PI : 0;
            const endAngle = isTop ? 2 * Math.PI : Math.PI;
            for (let i = 0; i <= numPoints; i++) {
                const a = startAngle + (endAngle - startAngle) * (i / numPoints);
                pts.push([cx + r * Math.cos(a), cy + r * Math.sin(a)]);
            }
            return pts;
        }
        
        function getLinePoints(isTop) {
            const pts = [];
            const y = isTop ? rectY : rectY + 100;
            const startX = rectX;
            const endX = rectX + 8 * arcLength;
            for (let i = 0; i <= numPoints; i++) {
                const x = startX + (endX - startX) * (i / numPoints);
                pts.push([x, y]);
            }
            return pts;
        }
        
        const lineGenerator = d3.line();
        const topArcPath = lineGenerator(getArcPoints(true));
        const bottomArcPath = lineGenerator(getArcPoints(false));
        const topLinePath = lineGenerator(getLinePoints(true));
        const bottomLinePath = lineGenerator(getLinePoints(false));
        
        d3.select("#scene1-top-red").attr("d", topArcPath);
        d3.select("#scene1-bottom-red").attr("d", bottomArcPath);
        
        // Blue lines initial
        d3.select("#scene1-left-blue").attr("x1", cx).attr("y1", cy).attr("x2", cx - r).attr("y2", cy);
        d3.select("#scene1-right-blue").attr("x1", cx).attr("y1", cy).attr("x2", cx + r).attr("y2", cy);
        
        let currentStep = 0;
        const maxStep = 5;
        
        const btnPrev = document.getElementById("scene1-btn-prev");
        const btnNext = document.getElementById("scene1-btn-next");
        
        function updateState(step) {
            const t = d3.transition().duration(800);
            
            // Step 0: Initial
            if (step === 0) {
                sectors.transition(t).attr("transform", d => `translate(${d.initX}, ${d.initY}) rotate(${d.initDeg})`);
                d3.select("#scene1-radius-group").transition(t).style("opacity", 1);
                d3.select("#scene1-ghost-circle").transition(t).style("opacity", 0);
            }
            // Step 1: Explode
            else if (step === 1) {
                sectors.transition(t).attr("transform", d => `translate(${d.explodeX}, ${d.explodeY}) rotate(${d.initDeg})`);
                d3.select("#scene1-radius-group").transition(t).style("opacity", 0);
                d3.select("#scene1-ghost-circle").transition(t).style("opacity", 0);
            }
            // Step 2: Rectangle
            else if (step >= 2) {
                sectors.transition(t).attr("transform", d => `translate(${d.targetX}, ${d.targetY}) rotate(${d.targetDeg})`);
                d3.select("#scene1-radius-group").transition(t).style("opacity", 0);
                d3.select("#scene1-ghost-circle").transition(t).style("opacity", 1);
            }
            
            // Step 3: Red lines
            if (step < 3) {
                d3.select("#scene1-top-red").transition(t).style("opacity", 0).attr("d", topArcPath);
                d3.select("#scene1-bottom-red").transition(t).style("opacity", 0).attr("d", bottomArcPath);
                d3.select("#scene1-top-pir").transition(t).style("opacity", 0);
                d3.select("#scene1-bottom-pir").transition(t).style("opacity", 0);
            } else if (step >= 3) {
                d3.select("#scene1-top-red").transition(t).style("opacity", 1).attr("d", topLinePath);
                d3.select("#scene1-bottom-red").transition(t).style("opacity", 1).attr("d", bottomLinePath);
                d3.select("#scene1-top-pir").transition(t).style("opacity", 1);
                d3.select("#scene1-bottom-pir").transition(t).style("opacity", 1);
            }
            
            // Step 4: Blue lines
            if (step < 4) {
                d3.select("#scene1-left-blue").transition(t).style("opacity", 0)
                    .attr("x1", cx).attr("y1", cy).attr("x2", cx - r).attr("y2", cy);
                d3.select("#scene1-right-blue").transition(t).style("opacity", 0)
                    .attr("x1", cx).attr("y1", cy).attr("x2", cx + r).attr("y2", cy);
                d3.select("#scene1-left-r").transition(t).style("opacity", 0);
                d3.select("#scene1-right-r").transition(t).style("opacity", 0);
            } else if (step >= 4) {
                // Left blue line: from (420, 190) to (439.64, 290)
                d3.select("#scene1-left-blue").transition(t).style("opacity", 1)
                    .attr("x1", rectX).attr("y1", rectY).attr("x2", rectX + arcLength/2).attr("y2", rectY + 100);
                // Right blue line: from (734.16, 190) to (753.79, 290)
                d3.select("#scene1-right-blue").transition(t).style("opacity", 1)
                    .attr("x1", rectX + 8*arcLength).attr("y1", rectY).attr("x2", rectX + 8*arcLength + arcLength/2).attr("y2", rectY + 100);
                d3.select("#scene1-left-r").transition(t).style("opacity", 1);
                d3.select("#scene1-right-r").transition(t).style("opacity", 1);
            }
            
            // Step 5: Conclusion
            if (step < 5) {
                d3.select("#scene1-conclusion").transition(t).style("opacity", 0);
            } else if (step === 5) {
                d3.select("#scene1-conclusion").transition(t).style("opacity", 1);
            }
            
            btnPrev.disabled = step === 0;
            btnNext.disabled = step === maxStep;
        }
        
        btnPrev.addEventListener("click", () => {
            if (currentStep > 0) {
                currentStep--;
                updateState(currentStep);
            }
        });
        
        btnNext.addEventListener("click", () => {
            if (currentStep < maxStep) {
                currentStep++;
                updateState(currentStep);
            }
        });
        
    })();
    </script>
</body>
