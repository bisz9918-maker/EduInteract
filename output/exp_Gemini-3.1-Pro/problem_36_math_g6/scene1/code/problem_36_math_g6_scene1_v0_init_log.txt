    <script>
    (function() {
        let currentStep = 0;
        const maxStep = 5;

        const texts = [
            "已知：三个圆面积均为 \(S\)，总覆盖面积为 \(2S+2\)，两处重叠面积相等。<br>直线 \(a\) 下方覆盖面积为 9。",
            "整体分析：直线 \(a\) 上方覆盖面积 = 总面积 - 下方面积 = \((2S+2) - 9 = 2S-7\)。",
            "聚焦上方：直线 \(a\) 上方区域由圆 A 的上半部和圆 B 的上半部组合而成。",
            "重叠分析：总面积 = \(3S - 2 	imes\) 单处重叠 = \(2S+2\)，解得单处重叠面积为 \(rac{S-2}{2}\)。",
            "上方重叠：圆 A 与圆 B 的重叠区被直线 \(a\) 平分，因此上方重叠面积为 \(rac{S-2}{4}\)。",
            "建立方程：上方覆盖面积 = 两个半圆面积 - 上方重叠面积。<br>即 \(rac{S}{2} + rac{S}{2} - rac{S-2}{4} = 2S-7\)，解得 \(S=6\)。"
        ];

        function updateState() {
            document.getElementById('scene1-btn-prev').disabled = currentStep === 0;
            document.getElementById('scene1-btn-next').disabled = currentStep === maxStep;

            const textEl = document.getElementById('scene1-text');
            textEl.innerHTML = texts[currentStep];
            if (window.MathJax) {
                MathJax.typesetPromise([textEl]);
            }

            const setOpacity = (id, val) => {
                const el = document.getElementById(id);
                if (el) {
                    el.style.opacity = val;
                    el.style.display = val > 0 ? 'block' : 'none';
                    // Force reflow for transition if changing display
                    if (val > 0) {
                        el.style.display = 'block';
                        setTimeout(() => el.style.opacity = val, 10);
                    } else {
                        el.style.opacity = 0;
                        setTimeout(() => el.style.display = 'none', 500);
                    }
                }
            };

            // Step 0: Base only
            // Step 1: Base + hl-below + hl-above
            // Step 2: Base + hl-above + mask-below
            // Step 3: Base + hl-overlap-all
            // Step 4: Base + hl-above + mask-below + hl-overlap-above
            // Step 5: Base + hl-above + mask-below + hl-overlap-above + labels-s2

            setOpacity('scene1-hl-below', currentStep === 1 ? 1 : 0);
            setOpacity('scene1-hl-above', [1, 2, 4, 5].includes(currentStep) ? 1 : 0);
            setOpacity('scene1-mask-below', [2, 4, 5].includes(currentStep) ? 0.8 : 0);
            setOpacity('scene1-hl-overlap-all', currentStep === 3 ? 1 : 0);
            setOpacity('scene1-hl-overlap-above', [4, 5].includes(currentStep) ? 1 : 0);
            setOpacity('scene1-labels-s2', currentStep === 5 ? 1 : 0);
            setOpacity('scene1-initial-overlaps', [0, 1, 2].includes(currentStep) ? 1 : 0);
        }

        document.getElementById('scene1-btn-prev').addEventListener('click', () => {
            if (currentStep > 0) {
                currentStep--;
                updateState();
            }
        });

        document.getElementById('scene1-btn-next').addEventListener('click', () => {
            if (currentStep < maxStep) {
                currentStep++;
                updateState();
            }
        });

        // Initial setup
        updateState();
    })();
    </script>
