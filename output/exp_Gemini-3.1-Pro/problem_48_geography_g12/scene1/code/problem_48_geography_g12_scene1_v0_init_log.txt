            <!-- 3D Labels (Step 1-3) -->
            <g id="scene1-3d-labels" style="display: none; opacity: 0; transition: opacity 0.5s;">

    <script type="module">
        import * as THREE from 'three';
        import { OrbitControls } from 'three/addons/controls/OrbitControls.js';

        let currentStep = 0;
        const maxStep = 3;
        
        // Elements
        const btnPrev = document.getElementById('scene1-btn-prev');
        const btnNext = document.getElementById('scene1-btn-next');
        const group2D = document.getElementById('scene1-2d-group');
        const container3D = document.getElementById('scene1-3d-container');
        const labels3D = document.getElementById('scene1-3d-labels');
        const labelM = document.getElementById('scene1-label-m');
        const labelN = document.getElementById('scene1-label-n');
        const labelMPrime = document.getElementById('scene1-label-m-prime');
        const labelO = document.getElementById('scene1-label-o');
        const titleText = document.getElementById('scene1-title-text');

        // Three.js Setup
        const width = 800;
        const height = 500;
        const scene = new THREE.Scene();
        const camera = new THREE.PerspectiveCamera(45, width / height, 1, 1000);
        const renderer = new THREE.WebGLRenderer({ alpha: true, antialias: true });
        renderer.setSize(width, height);
        container3D.appendChild(renderer.domElement);

        const controls = new OrbitControls(camera, renderer.domElement);
        controls.enableDamping = true;
        controls.enablePan = false;

        // Earth
        const R = 150;
        const earthGeo = new THREE.SphereGeometry(R, 32, 32);
        const earthMat = new THREE.MeshBasicMaterial({ color: 0x1565c0, transparent: true, opacity: 0.1 });
        const earth = new THREE.Mesh(earthGeo, earthMat);
        scene.add(earth);

        const wireMat = new THREE.MeshBasicMaterial({ color: 0x888888, wireframe: true, transparent: true, opacity: 0.3 });
        const wireEarth = new THREE.Mesh(earthGeo, wireMat);
        scene.add(wireEarth);

        // Equator and Prime Meridian
        const equatorGeo = new THREE.BufferGeometry().setFromPoints(
            new THREE.Path().absarc(0, 0, R, 0, Math.PI * 2).getPoints(64)
        );
        const equatorMat = new THREE.LineBasicMaterial({ color: 0x000000, linewidth: 2 });
        const equator = new THREE.Line(equatorGeo, equatorMat);
        equator.rotation.x = Math.PI / 2;
        equator.visible = false;
        scene.add(equator);

        const primeMeridianGeo = new THREE.BufferGeometry().setFromPoints(
            new THREE.Path().absarc(0, 0, R, 0, Math.PI * 2).getPoints(64)
        );
        const primeMeridian = new THREE.Line(primeMeridianGeo, equatorMat);
        primeMeridian.rotation.y = Math.PI / 2;
        primeMeridian.visible = false;
        scene.add(primeMeridian);

        // Coordinates to Vector3
        function latLonToVector3(lat, lon, radius) {
            const phi = (90 - lat) * (Math.PI / 180);
            const theta = (lon + 180) * (Math.PI / 180);
            const x = -(radius * Math.sin(phi) * Math.cos(theta));
            const z = (radius * Math.sin(phi) * Math.sin(theta));
            const y = (radius * Math.cos(phi));
            return new THREE.Vector3(x, y, z);
        }

        // Points
        const posM = latLonToVector3(60, 80, R);
        const posN = latLonToVector3(60, 120, R);
        const posMPrime = latLonToVector3(-60, -100, R);
        const posO = new THREE.Vector3(0, 0, 0);

        const pointGeo = new THREE.SphereGeometry(3, 16, 16);
        const pointMat = new THREE.MeshBasicMaterial({ color: 0x1565c0 });
        
        const meshM = new THREE.Mesh(pointGeo, pointMat);
        meshM.position.copy(posM);
        scene.add(meshM);

        const meshN = new THREE.Mesh(pointGeo, pointMat);
        meshN.position.copy(posN);
        scene.add(meshN);

        const meshMPrime = new THREE.Mesh(pointGeo, pointMat);
        meshMPrime.position.copy(posMPrime);
        meshMPrime.visible = false;
        scene.add(meshMPrime);

        const meshO = new THREE.Mesh(pointGeo, pointMat);
        meshO.position.copy(posO);
        meshO.visible = false;
        scene.add(meshO);

        // Lines
        // Latitude Arc M-N
        const latArcPoints = [];
        for (let i = 80; i <= 120; i++) {
            latArcPoints.push(latLonToVector3(60, i, R));
        }
        const latArcGeo = new THREE.BufferGeometry().setFromPoints(latArcPoints);
        const latArcMat = new THREE.LineDashedMaterial({ color: 0x888888, dashSize: 5, gapSize: 3 });
        const latArc = new THREE.Line(latArcGeo, latArcMat);
        latArc.computeLineDistances();
        latArc.visible = false;
        scene.add(latArc);

        // Great Circle Arc M-N
        const gcArcPoints = [];
        const segments = 50;
        for (let i = 0; i <= segments; i++) {
            const t = i / segments;
            const v = posM.clone().lerp(posN, t).normalize().multiplyScalar(R);
            gcArcPoints.push(v);
        }
        const gcArcGeo = new THREE.BufferGeometry().setFromPoints(gcArcPoints);
        const gcArcMat = new THREE.LineBasicMaterial({ color: 0xc62828, linewidth: 2 });
        const gcArc = new THREE.Line(gcArcGeo, gcArcMat);
        gcArc.visible = false;
        scene.add(gcArc);

        // Antipode Line M-O-M'
        const antipodeGeo = new THREE.BufferGeometry().setFromPoints([posM, posMPrime]);
        const antipodeMat = new THREE.LineDashedMaterial({ color: 0x2e7d32, dashSize: 5, gapSize: 3 });
        const antipodeLine = new THREE.Line(antipodeGeo, antipodeMat);
        antipodeLine.computeLineDistances();
        antipodeLine.visible = false;
        scene.add(antipodeLine);

        // Airplane/Arrow
        const arrowGeo = new THREE.ConeGeometry(4, 12, 16);
        arrowGeo.rotateX(Math.PI / 2);
        const arrowMat = new THREE.MeshBasicMaterial({ color: 0xc62828 });
        const arrow = new THREE.Mesh(arrowGeo, arrowMat);
        arrow.visible = false;
        scene.add(arrow);

        // Camera positions
        const camPos1 = new THREE.Vector3(400, 230, -70);
        const camPos3 = new THREE.Vector3(-80, 0, 450);
        camera.position.copy(camPos1);
        camera.lookAt(0, 0, 0);

        let targetCamPos = camPos1.clone();
        
        // Animation state
        let arrowT = 0;

        function updateLabels() {
            const updateLabel = (mesh, label, offsetX, offsetY) => {
                if (!mesh.visible && label.style.opacity === '0') {
                    return;
                }
                const pos = mesh.position.clone();
                pos.project(camera);
                const x = (pos.x + 1) / 2 * width;
                const y = -(pos.y - 1) / 2 * height;
                
                // Keep labels within bounds
                const margin = 30;
                // For M', avoid bottom controls
                const bottomMargin = label === labelMPrime ? 60 : margin;
                const boundedX = Math.max(margin, Math.min(width - margin, x + offsetX));
                const boundedY = Math.max(margin, Math.min(height - bottomMargin, y + offsetY));
                
                label.setAttribute('x', boundedX);
                label.setAttribute('y', boundedY);
            };

            updateLabel(meshM, labelM, -20, 20); // Changed to 20 to avoid top clipping
            updateLabel(meshN, labelN, 20, 20); // Changed to 20 to avoid top clipping
            updateLabel(meshMPrime, labelMPrime, -20, -20);
            updateLabel(meshO, labelO, 15, 15);
        }

        function animate() {
            requestAnimationFrame(animate);
            controls.update();

            // Camera interpolation
            camera.position.lerp(targetCamPos, 0.05);

            // Arrow animation
            if (currentStep === 2 && arrow.visible) {
                arrowT += 0.005;
                if (arrowT > 1) arrowT = 0;
                
                const t1 = arrowT;
                const t2 = Math.min(arrowT + 0.01, 1);
                
                const p1 = posM.clone().lerp(posN, t1).normalize().multiplyScalar(R);
                const p2 = posM.clone().lerp(posN, t2).normalize().multiplyScalar(R);
                
                arrow.position.copy(p1);
                if (t1 < 1) {
                    arrow.lookAt(p2);
                }
            }

            updateLabels();
            renderer.render(scene, camera);
        }
        animate();

        function show(el) {
            el.style.display = '';
            // small delay to allow display to apply before opacity transition
            setTimeout(() => el.style.opacity = 1, 10);
        }

        function hide(el) {
            el.style.opacity = 0;
            setTimeout(() => el.style.display = 'none', 500);
        }

        function updateStep() {
            btnPrev.disabled = currentStep === 0;
            btnNext.disabled = currentStep === maxStep;

            // Step 0: 2D Grid
            if (currentStep === 0) {
                show(group2D);
                hide(container3D);
                hide(labels3D);
            } else {
                hide(group2D);
                show(container3D);
                show(labels3D);
            }

            // Step 1: 3D Earth
            if (currentStep === 1) {
                targetCamPos.copy(camPos1);
                latArc.visible = false;
                gcArc.visible = false;
                arrow.visible = false;
                hide(titleText);
                
                meshMPrime.visible = false;
                meshO.visible = false;
                antipodeLine.visible = false;
                equator.visible = false;
                primeMeridian.visible = false;
                hide(labelMPrime);
                hide(labelO);
            }

            // Step 2: Shortest Route
            if (currentStep === 2) {
                targetCamPos.copy(camPos1);
                latArc.visible = true;
                gcArc.visible = true;
                arrow.visible = true;
                show(titleText);
                
                meshMPrime.visible = false;
                meshO.visible = false;
                antipodeLine.visible = false;
                equator.visible = false;
                primeMeridian.visible = false;
                hide(labelMPrime);
                hide(labelO);
            }

            // Step 3: Antipode
            if (currentStep === 3) {
                targetCamPos.copy(camPos3);
                latArc.visible = false;
                gcArc.visible = false;
                arrow.visible = false;
                hide(titleText);
                
                meshMPrime.visible = true;
                meshO.visible = true;
                antipodeLine.visible = true;
                equator.visible = true;
                primeMeridian.visible = true;
                show(labelMPrime);
                show(labelO);
            }
        }

        btnPrev.addEventListener('click', () => {
            if (currentStep > 0) {
                currentStep--;
                updateStep();
            }
        });

        btnNext.addEventListener('click', () => {
            if (currentStep < maxStep) {
                currentStep++;
                updateStep();
            }
        });

        // Init
        updateStep();
    </script>
