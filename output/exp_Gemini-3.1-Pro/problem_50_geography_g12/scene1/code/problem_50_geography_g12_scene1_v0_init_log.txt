    <script>
    (function() {
        let currentStep = 0;
        const maxStep = 4;
        
        const btnPrev = document.getElementById('scene1-btn-prev');
        const btnNext = document.getElementById('scene1-btn-next');
        
        // Elements
        const lineCab = document.getElementById('scene1-line-cab');
        const pointA = document.getElementById('scene1-point-a');
        const pointB = document.getElementById('scene1-point-b');
        const pointC = document.getElementById('scene1-point-c');
        const pointN = document.getElementById('scene1-point-n');
        const equator = document.getElementById('scene1-equator');
        
        const labelACoords1 = document.getElementById('scene1-label-a-coords1');
        const labelACoords2 = document.getElementById('scene1-label-a-coords2');
        const labelBCoords = document.getElementById('scene1-label-b-coords');
        const latDiff = document.getElementById('scene1-lat-diff');
        const labelN = document.getElementById('scene1-label-n');
        const labelCCoords = document.getElementById('scene1-label-c-coords');
        const timeAxis = document.getElementById('scene1-time-axis');
        
        function updateState() {
            // Reset all
            lineCab.setAttribute('class', 'scene1-element scene1-stroke-black');
            pointA.setAttribute('class', 'scene1-element scene1-fill-black');
            pointB.setAttribute('class', 'scene1-element scene1-fill-black');
            pointC.setAttribute('class', 'scene1-element scene1-fill-black');
            equator.setAttribute('class', 'scene1-element scene1-stroke-gray scene1-fill-none');
            
            hide(labelACoords1);
            hide(labelACoords2);
            hide(labelBCoords);
            hide(latDiff);
            hide(pointN);
            hide(labelN);
            hide(labelCCoords);
            hide(timeAxis);
            
            if (currentStep >= 1) {
                lineCab.setAttribute('class', 'scene1-element scene1-stroke-blue');
                pointA.setAttribute('class', 'scene1-element scene1-fill-blue');
                if (currentStep === 1) {
                    show(labelACoords1);
                }
            }
            
            if (currentStep >= 2) {
                pointA.setAttribute('class', 'scene1-element scene1-fill-red');
                pointB.setAttribute('class', 'scene1-element scene1-fill-red');
                equator.setAttribute('class', 'scene1-element scene1-stroke-red scene1-fill-none');
                show(labelACoords2);
                show(labelBCoords);
                show(latDiff);
            }
            
            if (currentStep >= 3) {
                pointC.setAttribute('class', 'scene1-element scene1-fill-red');
                show(pointN);
                show(labelN);
                show(labelCCoords);
            }
            
            if (currentStep >= 4) {
                show(timeAxis);
            }
            
            btnPrev.disabled = currentStep === 0;
            btnNext.disabled = currentStep === maxStep;
        }
        
        function show(el) {
            el.classList.remove('scene1-hidden');
            el.classList.add('scene1-visible');
        }
        
        function hide(el) {
            el.classList.remove('scene1-visible');
            el.classList.add('scene1-hidden');
        }
        
        btnPrev.addEventListener('click', () => {
            if (currentStep > 0) {
                currentStep--;
                updateState();
            }
        });
        
        btnNext.addEventListener('click', () => {
            if (currentStep < maxStep) {
                currentStep++;
                updateState();
            }
        });
        
        // Init
        updateState();
    })();
    </script>
