    <script>
    (function() {
        let currentStep = 0;
        const maxStep = 3;
        
        const btnPrev = document.getElementById('scene3-btn-prev');
        const btnNext = document.getElementById('scene3-btn-next');
        
        const step1 = document.getElementById('scene3-step1');
        const step2 = document.getElementById('scene3-step2');
        const step3 = document.getElementById('scene3-step3');
        
        function updateScene() {
            btnPrev.disabled = currentStep === 0;
            btnNext.disabled = currentStep === maxStep;
            
            if (currentStep >= 1) step1.classList.add('visible'); else step1.classList.remove('visible');
            if (currentStep >= 2) step2.classList.add('visible'); else step2.classList.remove('visible');
            if (currentStep >= 3) step3.classList.add('visible'); else step3.classList.remove('visible');
        }
        
        btnPrev.addEventListener('click', () => {
            if (currentStep > 0) {
                currentStep--;
                updateScene();
            }
        });
        
        btnNext.addEventListener('click', () => {
            if (currentStep < maxStep) {
                currentStep++;
                updateScene();
            }
        });
        
        updateScene();
    })();
    </script>
