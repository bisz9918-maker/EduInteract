    <script>
    (function() {
        let currentStep = 0;
        const maxStep = 4;
        
        const btnPrev = document.getElementById('scene1-btn-prev');
        const btnNext = document.getElementById('scene1-btn-next');
        
        const steps = [
            [], // Step 0
            ['scene1-step1'],
            ['scene1-step1', 'scene1-step2'],
            ['scene1-step1', 'scene1-step2', 'scene1-step3'],
            ['scene1-step1', 'scene1-step2', 'scene1-step3', 'scene1-step4']
        ];
        
        function show(el) {
            if (el.style.display === 'none' || el.getAttribute('display') === 'none') {
                el.style.display = '';
                el.removeAttribute('display');
                // force reflow
                void el.offsetWidth;
            }
            el.style.opacity = '1';
        }
        
        function hide(el) {
            el.style.opacity = '0';
            setTimeout(() => {
                if (el.style.opacity === '0') {
                    el.style.display = 'none';
                }
            }, 300);
        }
        
        function updateScene() {
            const activeIds = steps[currentStep];
            
            for (let i = 1; i <= 4; i++) {
                const id = 'scene1-step' + i;
                const el = document.getElementById(id);
                if (el) {
                    if (activeIds.includes(id)) {
                        show(el);
                    } else {
                        hide(el);
                    }
                }
            }
            
            btnPrev.disabled = currentStep === 0;
            btnNext.disabled = currentStep === maxStep;
        }
        
        btnPrev.addEventListener('click', () => {
            if (currentStep > 0) {
                currentStep--;
                updateScene();
            }
        });
        
        btnNext.addEventListener('click', () => {
            if (currentStep < maxStep) {
                currentStep++;
                updateScene();
            }
        });
        
        // Init
        updateScene();
    })();
    </script>
