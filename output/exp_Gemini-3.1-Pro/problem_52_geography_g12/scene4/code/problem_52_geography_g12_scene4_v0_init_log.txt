    <script>
    (function() {
        let currentStep = 0;
        const maxStep = 4;
        
        const btnPrev = document.getElementById('scene4-btn-prev');
        const btnNext = document.getElementById('scene4-btn-next');
        
        function updateSteps() {
            for (let i = 0; i <= maxStep; i++) {
                const group = document.getElementById('scene4-step' + i);
                if (group) {
                    if (i <= currentStep) {
                        group.classList.add('active');
                    } else {
                        group.classList.remove('active');
                    }
                }
            }
            
            btnPrev.disabled = currentStep === 0;
            btnNext.disabled = currentStep === maxStep;
        }
        
        btnPrev.addEventListener('click', () => {
            if (currentStep > 0) {
                currentStep--;
                updateSteps();
            }
        });
        
        btnNext.addEventListener('click', () => {
            if (currentStep < maxStep) {
                currentStep++;
                updateSteps();
            }
        });
        
        // 初始化
        updateSteps();
    })();
    </script>
