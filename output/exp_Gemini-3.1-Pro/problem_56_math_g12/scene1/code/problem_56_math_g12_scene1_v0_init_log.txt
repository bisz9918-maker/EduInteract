    <script>
    (function() {
        let currentStep = 0;
        const maxStep = 4;
        
        const btnPrev = document.getElementById('scene1-btn-prev');
        const btnNext = document.getElementById('scene1-btn-next');
        
        // Elements
        const c2 = document.getElementById('scene1-c2');
        const auxGroup = document.getElementById('scene1-aux-group');
        const triOAB = document.getElementById('scene1-triangle-oab');
        const lineOA = document.getElementById('scene1-line-oa');
        const lineOB = document.getElementById('scene1-line-ob');
        const lineAB = document.getElementById('scene1-line-ab');
        const pointO = document.getElementById('scene1-point-o');
        const pointA = document.getElementById('scene1-point-a');
        const pointB = document.getElementById('scene1-point-b');
        const labelO = document.getElementById('scene1-label-o');
        const labelA = document.getElementById('scene1-label-a');
        const labelB = document.getElementById('scene1-label-b');
        
        const lineOM = document.getElementById('scene1-line-om');
        const lineOARed = document.getElementById('scene1-line-oa-red');
        const angleArc = document.getElementById('scene1-angle-arc');
        const rightAngle = document.getElementById('scene1-right-angle');
        const pointM = document.getElementById('scene1-point-m');
        const labelM = document.getElementById('scene1-label-m');
        const labelR1 = document.getElementById('scene1-label-r1');
        const labelR2 = document.getElementById('scene1-label-r2');
        const label30deg = document.getElementById('scene1-label-30deg');
        
        const infiniteNesting = document.getElementById('scene1-infinite-nesting');
        
        let animationTimeouts = [];
        
        function clearAnimations() {
            animationTimeouts.forEach(t => clearTimeout(t));
            animationTimeouts = [];
            infiniteNesting.innerHTML = '';
        }
        
        function generateHexagonPoints(cx, cy, r) {
            let points = [];
            for (let i = 0; i < 6; i++) {
                let angle = Math.PI / 2 - i * Math.PI / 3;
                points.push(`${cx + r * Math.cos(angle)},${cy + r * Math.sin(angle)}`);
            }
            return points.join(' ');
        }
        
        function startInfiniteNesting() {
            clearAnimations();
            let r = 155.88;
            const cx = 400, cy = 220;
            
            for (let i = 2; i <= 6; i++) {
                let timeoutH = setTimeout(() => {
                    const hex = document.createElementNS('http://www.w3.org/2000/svg', 'polygon');
                    hex.setAttribute('points', generateHexagonPoints(cx, cy, r));
                    hex.setAttribute('stroke', 'black');
                    hex.setAttribute('stroke-width', '1.5');
                    hex.setAttribute('fill', 'none');
                    hex.style.opacity = '0';
                    hex.style.transition = 'opacity 0.3s ease';
                    infiniteNesting.appendChild(hex);
                    void hex.offsetWidth;
                    hex.style.opacity = '1';
                }, (i - 2) * 600 + 300);
                animationTimeouts.push(timeoutH);
                
                r = r * Math.cos(Math.PI / 6);
                let timeoutC = setTimeout(() => {
                    const circle = document.createElementNS('http://www.w3.org/2000/svg', 'circle');
                    circle.setAttribute('cx', cx);
                    circle.setAttribute('cy', cy);
                    circle.setAttribute('r', r);
                    circle.setAttribute('stroke', 'black');
                    circle.setAttribute('stroke-width', '1.5');
                    circle.setAttribute('fill', 'none');
                    circle.style.opacity = '0';
                    circle.style.transition = 'opacity 0.3s ease';
                    infiniteNesting.appendChild(circle);
                    void circle.offsetWidth;
                    circle.style.opacity = '1';
                }, (i - 2) * 600 + 600);
                animationTimeouts.push(timeoutC);
            }
        }
        
        function show(el, targetOpacity = '1') {
            if (!el) return;
            el.style.display = '';
            void el.offsetWidth;
            el.style.opacity = targetOpacity;
        }
        
        function hide(el) {
            if (!el) return;
            el.style.opacity = '0';
            setTimeout(() => {
                if (el.style.opacity === '0') {
                    el.style.display = 'none';
                }
            }, 300);
        }
        
        function updateScene() {
            btnPrev.disabled = currentStep === 0;
            btnNext.disabled = currentStep === maxStep;
            
            // Step 1: C2
            if (currentStep >= 1) show(c2); else hide(c2);
            
            // Step 2: Aux Group, OAB
            if (currentStep >= 2 && currentStep < 4) {
                show(auxGroup);
                show(triOAB, '0.2');
                show(lineOA);
                show(lineOB);
                show(lineAB);
                show(pointO);
                show(pointA);
                show(pointB);
                show(labelO);
                show(labelA);
                show(labelB);
            } else {
                hide(triOAB);
                hide(lineOA);
                hide(lineOB);
                hide(lineAB);
                hide(pointO);
                hide(pointA);
                hide(pointB);
                hide(labelO);
                hide(labelA);
                hide(labelB);
                if (currentStep < 2 || currentStep >= 4) hide(auxGroup);
            }
            
            // Step 3: OM, R1, R2, 30deg
            if (currentStep >= 3 && currentStep < 4) {
                show(lineOM);
                show(lineOARed);
                show(angleArc);
                show(rightAngle);
                show(pointM);
                show(labelM);
                show(labelR1);
                show(labelR2);
                show(label30deg);
            } else {
                hide(lineOM);
                hide(lineOARed);
                hide(angleArc);
                hide(rightAngle);
                hide(pointM);
                hide(labelM);
                hide(labelR1);
                hide(labelR2);
                hide(label30deg);
            }
            
            // Step 4: Infinite Nesting
            if (currentStep === 4) {
                if (animationTimeouts.length === 0) {
                    startInfiniteNesting();
                }
            } else {
                clearAnimations();
            }
        }
        
        btnPrev.addEventListener('click', () => {
            if (currentStep > 0) {
                currentStep--;
                updateScene();
            }
        });
        
        btnNext.addEventListener('click', () => {
            if (currentStep < maxStep) {
                currentStep++;
                updateScene();
            }
        });
        
        // Initialize
        updateScene();
    })();
    </script>
