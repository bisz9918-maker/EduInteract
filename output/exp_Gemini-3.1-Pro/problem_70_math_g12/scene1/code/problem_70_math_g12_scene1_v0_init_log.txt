    <script>
    (function() {
        let currentStep = 0;
        const maxStep = 6;
        
        const btnPrev = document.getElementById('scene1-btn-prev');
        const btnNext = document.getElementById('scene1-btn-next');
        
        // Elements
        const lineAC = document.getElementById('scene1-line-ac');
        const lineBD = document.getElementById('scene1-line-bd');
        const linePA = document.getElementById('scene1-line-pa');
        const linePC = document.getElementById('scene1-line-pc');
        const lineDM = document.getElementById('scene1-line-dm');
        const lineOQ = document.getElementById('scene1-line-oq');
        
        const planePAC = document.getElementById('scene1-plane-pac');
        const planeMBD = document.getElementById('scene1-plane-mbd');
        const planePCD = document.getElementById('scene1-plane-pcd');
        
        const ra1 = document.getElementById('scene1-ra1');
        const ra2 = document.getElementById('scene1-ra2');
        const ra3 = document.getElementById('scene1-ra3');
        const ra4 = document.getElementById('scene1-ra4');
        const ra5 = document.getElementById('scene1-ra5');
        
        const conclusion = document.getElementById('scene1-conclusion');
        
        function updateState() {
            // Reset all
            lineAC.style.opacity = 0;
            lineOQ.style.opacity = 0;
            planePAC.style.opacity = 0;
            planeMBD.style.opacity = 0;
            planePCD.style.opacity = 0;
            ra1.style.opacity = 0;
            ra2.style.opacity = 0;
            ra3.style.opacity = 0;
            ra4.style.opacity = 0;
            ra5.style.opacity = 0;
            conclusion.style.opacity = 0;
            
            lineAC.classList.remove('hl-red');
            lineBD.classList.remove('hl-red', 'hl-blue');
            linePA.classList.remove('hl-red');
            linePC.classList.remove('hl-blue');
            lineDM.classList.remove('hl-blue');
            
            switch(currentStep) {
                case 1:
                    lineAC.style.opacity = 1;
                    lineAC.classList.add('hl-red');
                    lineBD.classList.add('hl-red');
                    ra1.style.opacity = 1;
                    break;
                case 2:
                    linePA.classList.add('hl-red');
                    lineBD.classList.add('hl-red');
                    ra2.style.opacity = 1;
                    ra3.style.opacity = 1;
                    break;
                case 3:
                    planePAC.style.opacity = 1;
                    linePC.classList.add('hl-blue');
                    lineBD.classList.add('hl-blue');
                    lineOQ.style.opacity = 1;
                    ra4.style.opacity = 1;
                    break;
                case 4:
                    planeMBD.style.opacity = 1;
                    planePCD.style.opacity = 1;
                    break;
                case 5:
                    planeMBD.style.opacity = 1;
                    planePCD.style.opacity = 1;
                    linePC.classList.add('hl-blue');
                    lineDM.classList.add('hl-blue');
                    ra5.style.opacity = 1;
                    break;
                case 6:
                    planeMBD.style.opacity = 1;
                    planePCD.style.opacity = 1;
                    linePC.classList.add('hl-blue');
                    lineDM.classList.add('hl-blue');
                    ra5.style.opacity = 1;
                    conclusion.style.opacity = 1;
                    break;
            }
            
            btnPrev.disabled = currentStep === 0;
            btnNext.disabled = currentStep === maxStep;
        }
        
        btnPrev.addEventListener('click', () => {
            if (currentStep > 0) {
                currentStep--;
                updateState();
            }
        });
        
        btnNext.addEventListener('click', () => {
            if (currentStep < maxStep) {
                currentStep++;
                updateState();
            }
        });
        
        // Initial state
        updateState();
    })();
    </script>
