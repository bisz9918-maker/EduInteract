    <script>
    (function() {
        let currentStep = 0;
        const maxStep = 4;
        
        const btnPrev = document.getElementById('scene1-btn-prev');
        const btnNext = document.getElementById('scene1-btn-next');
        
        // Elements
        const el3dI = document.getElementById('scene1-3d-I');
        const el3dvGroup = document.getElementById('scene1-3d-v-group');
        const el3dv = document.getElementById('scene1-3d-v');
        
        const elSideForcesBase = document.getElementById('scene1-side-forces-base');
        const elSideFAGroup = document.getElementById('scene1-side-FA-group');
        const elSideFA = document.getElementById('scene1-side-FA');
        const elSideFALabel = document.getElementById('scene1-side-FA-label');
        
        const elSideVGroup = document.getElementById('scene1-side-v-group');
        const elSideV = document.getElementById('scene1-side-v');
        
        const elSideAGroup = document.getElementById('scene1-side-a-group');
        const elSideA = document.getElementById('scene1-side-a');
        const elSideALabel = document.getElementById('scene1-side-a-label');
        
        function updateState() {
            btnPrev.disabled = currentStep === 0;
            btnNext.disabled = currentStep === maxStep;
            
            // Step 0: Initial
            if (currentStep === 0) {
                el3dI.style.opacity = 0;
                el3dvGroup.style.opacity = 0;
                elSideForcesBase.style.opacity = 0;
                elSideFAGroup.style.opacity = 0;
                elSideVGroup.style.opacity = 0;
                elSideAGroup.style.opacity = 0;
            }
            // Step 1: Force analysis
            else if (currentStep === 1) {
                el3dI.style.opacity = 0;
                el3dvGroup.style.opacity = 0;
                elSideForcesBase.style.opacity = 1;
                elSideFAGroup.style.opacity = 0;
                elSideVGroup.style.opacity = 0;
                elSideAGroup.style.opacity = 0;
            }
            // Step 2: Motion & Ampere force (v=30, FA=20, a=60)
            else if (currentStep === 2) {
                el3dI.style.opacity = 1;
                el3dvGroup.style.opacity = 1;
                elSideForcesBase.style.opacity = 1;
                elSideFAGroup.style.opacity = 1;
                elSideVGroup.style.opacity = 1;
                elSideAGroup.style.opacity = 1;
                
                el3dv.setAttribute('x2', 290 - 21.2);
                el3dv.setAttribute('y2', 210 + 21.2);
                
                elSideFA.setAttribute('x2', 650 + 17.32);
                elSideFA.setAttribute('y2', 234.53 - 10);
                elSideFALabel.setAttribute('x', 650 + 17.32 + 5);
                elSideFALabel.setAttribute('y', 234.53 - 10 - 5);
                
                elSideV.setAttribute('x2', 640 - 25.98);
                elSideV.setAttribute('y2', 215 + 15);
                
                elSideA.setAttribute('x2', 660 - 51.96);
                elSideA.setAttribute('y2', 255 + 30);
                elSideALabel.setAttribute('x', 660 - 51.96 + 5);
                elSideALabel.setAttribute('y', 255 + 30 + 15);
            }
            // Step 3: Dynamic change (v=60, FA=35, a=30)
            else if (currentStep === 3) {
                el3dI.style.opacity = 1;
                el3dvGroup.style.opacity = 1;
                elSideForcesBase.style.opacity = 1;
                elSideFAGroup.style.opacity = 1;
                elSideVGroup.style.opacity = 1;
                elSideAGroup.style.opacity = 1;
                
                el3dv.setAttribute('x2', 290 - 42.4);
                el3dv.setAttribute('y2', 210 + 42.4);
                
                elSideFA.setAttribute('x2', 650 + 30.31);
                elSideFA.setAttribute('y2', 234.53 - 17.5);
                elSideFALabel.setAttribute('x', 650 + 30.31 + 5);
                elSideFALabel.setAttribute('y', 234.53 - 17.5 - 5);
                
                elSideV.setAttribute('x2', 640 - 51.96);
                elSideV.setAttribute('y2', 215 + 30);
                
                elSideA.setAttribute('x2', 660 - 25.98);
                elSideA.setAttribute('y2', 255 + 15);
                elSideALabel.setAttribute('x', 660 - 25.98 + 5);
                elSideALabel.setAttribute('y', 255 + 15 + 15);
            }
            // Step 4: Max velocity (v=90, FA=50, a=0)
            else if (currentStep === 4) {
                el3dI.style.opacity = 1;
                el3dvGroup.style.opacity = 1;
                elSideForcesBase.style.opacity = 1;
                elSideFAGroup.style.opacity = 1;
                elSideVGroup.style.opacity = 1;
                elSideAGroup.style.opacity = 0;
                
                el3dv.setAttribute('x2', 290 - 63.6);
                el3dv.setAttribute('y2', 210 + 63.6);
                
                elSideFA.setAttribute('x2', 650 + 43.3);
                elSideFA.setAttribute('y2', 234.53 - 25);
                elSideFALabel.setAttribute('x', 650 + 43.3 + 5);
                elSideFALabel.setAttribute('y', 234.53 - 25 - 5);
                
                elSideV.setAttribute('x2', 640 - 77.94);
                elSideV.setAttribute('y2', 215 + 45);
            }
            
            if (currentStep === 4) {
                document.getElementById('scene1-3d-v-label-normal').style.opacity = 0;
                document.getElementById('scene1-3d-v-label-max').style.opacity = 1;
                document.getElementById('scene1-side-v-label-normal').style.opacity = 0;
                document.getElementById('scene1-side-v-label-max').style.opacity = 1;
            } else {
                document.getElementById('scene1-3d-v-label-normal').style.opacity = 1;
                document.getElementById('scene1-3d-v-label-max').style.opacity = 0;
                document.getElementById('scene1-side-v-label-normal').style.opacity = 1;
                document.getElementById('scene1-side-v-label-max').style.opacity = 0;
            }
        }
        
        btnPrev.addEventListener('click', () => {
            if (currentStep > 0) {
                currentStep--;
                updateState();
            }
        });
        
        btnNext.addEventListener('click', () => {
            if (currentStep < maxStep) {
                currentStep++;
                updateState();
            }
        });
        
        updateState();
    })();
    </script>
