    <script>
    (function() {
        const state1Group = document.getElementById('scene2-state1');
        const state2Group = document.getElementById('scene2-state2');
        const btnState1 = document.getElementById('scene2-btn-state1');
        const btnState2 = document.getElementById('scene2-btn-state2');

        function showState(state) {
            if (state === 1) {
                state1Group.style.display = 'block';
                state2Group.style.display = 'none';
                btnState1.classList.add('active');
                btnState2.classList.remove('active');
            } else {
                state1Group.style.display = 'none';
                state2Group.style.display = 'block';
                btnState1.classList.remove('active');
                btnState2.classList.add('active');
            }
        }

        btnState1.addEventListener('click', () => showState(1));
        btnState2.addEventListener('click', () => showState(2));
    })();
