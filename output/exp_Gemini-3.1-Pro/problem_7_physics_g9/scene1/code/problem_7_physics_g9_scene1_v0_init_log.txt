    <script>
    (function() {
        let currentStep = 0;
        const maxStep = 4;

        const btnPrev = document.getElementById('scene1-btn-prev');
        const btnNext = document.getElementById('scene1-btn-next');
        
        const leftJia = document.getElementById('scene1-left-jia');
        const rightJia = document.getElementById('scene1-right-jia');
        
        const step1Vectors = document.getElementById('scene1-step1-vectors');
        const step2Vectors = document.getElementById('scene1-step2-vectors');
        const step3Vectors = document.getElementById('scene1-step3-vectors');
        
        const textTop = document.getElementById('scene1-text-top');
        const textLeft = document.getElementById('scene1-text-left');
        const textRight = document.getElementById('scene1-text-right');
        const textBottom = document.getElementById('scene1-text-bottom');
        
        const leftT = document.getElementById('scene1-left-t');
        const rightN = document.getElementById('scene1-right-n');

        function updateScene() {
            // Buttons
            btnPrev.disabled = currentStep === 0;
            btnNext.disabled = currentStep === maxStep;

            // Step 1: Highlight blocks, show F_float, show top text
            if (currentStep >= 1) {
                leftJia.style.fill = '#bbdefb';
                rightJia.style.fill = '#bbdefb';
                step1Vectors.style.opacity = 1;
                textTop.style.opacity = 1;
            } else {
                leftJia.style.fill = '#ffffff';
                rightJia.style.fill = '#ffffff';
                step1Vectors.style.opacity = 0;
                textTop.style.opacity = 0;
            }

            // Step 2: Show left G and T, show left text
            if (currentStep >= 2) {
                step2Vectors.style.opacity = 1;
                textLeft.style.opacity = 1;
            } else {
                step2Vectors.style.opacity = 0;
                textLeft.style.opacity = 0;
            }

            // Step 3: Show right G and N, show right text
            if (currentStep >= 3) {
                step3Vectors.style.opacity = 1;
                textRight.style.opacity = 1;
            } else {
                step3Vectors.style.opacity = 0;
                textRight.style.opacity = 0;
            }

            // Step 4: Show bottom text, highlight T and N
            if (currentStep >= 4) {
                textBottom.style.opacity = 1;
                leftT.style.strokeWidth = 4;
                rightN.style.strokeWidth = 4;
            } else {
                textBottom.style.opacity = 0;
                leftT.style.strokeWidth = 2;
                rightN.style.strokeWidth = 2;
            }
        }

        btnPrev.addEventListener('click', () => {
            if (currentStep > 0) {
                currentStep--;
                updateScene();
            }
        });

        btnNext.addEventListener('click', () => {
            if (currentStep < maxStep) {
                currentStep++;
                updateScene();
            }
        });

        // Initialize
        updateScene();
    })();
    </script>
