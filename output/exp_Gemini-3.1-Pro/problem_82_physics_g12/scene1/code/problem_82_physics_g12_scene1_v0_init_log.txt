    <script>
    (function() {
        let step = 0;
        const maxStep = 3;
        
        const btnPrev = document.getElementById('scene1-btn-prev');
        const btnNext = document.getElementById('scene1-btn-next');
        
        const groupA = document.getElementById('scene1-group-a');
        const groupB = document.getElementById('scene1-group-b');
        const groupC = document.getElementById('scene1-group-c');
        
        const labelA = document.getElementById('scene1-label-a');
        const labelB = document.getElementById('scene1-label-b');
        const labelC = document.getElementById('scene1-label-c');
        
        const stringAB = document.getElementById('scene1-string-ab');
        const stringBC = document.getElementById('scene1-string-bc');
        
        const velocity = document.getElementById('scene1-velocity');
        const velocityLabel = document.getElementById('scene1-velocity-label');
        
        const states = [
            {
                // Step 0: Initial
                a: { x: 550, y: 136, lx: 25, ly: -20, op: 1 },
                b: { x: 377, y: 236, lx: -25, ly: -20, op: 1 },
                c: { x: 204, y: 336, lx: -25, ly: -20, op: 1 },
                sab: { d: "M 550 136 Q 463.5 186 377 236", op: 1 },
                sbc: { d: "M 377 236 Q 290.5 286 204 336", op: 1 },
                v: { op: 0, label: "" }
            },
            {
                // Step 1: A lands
                a: { x: 550, y: 336, lx: 25, ly: 25, op: 1 },
                b: { x: 550, y: 136, lx: -25, ly: -20, op: 1 },
                c: { x: 377, y: 236, lx: -25, ly: -20, op: 1 },
                sab: { d: "M 550 336 Q 550 236 550 136", op: 1 },
                sbc: { d: "M 550 136 Q 463.5 186 377 236", op: 1 },
                v: { op: 1, label: "v_A" }
            },
            {
                // Step 2: B lands
                a: { x: 590, y: 336, lx: 0, ly: 30, op: 0.3 },
                b: { x: 550, y: 336, lx: 0, ly: 30, op: 1 },
                c: { x: 550, y: 136, lx: -25, ly: -20, op: 1 },
                sab: { d: "M 590 336 Q 570 350 550 336", op: 0.3 },
                sbc: { d: "M 550 336 Q 550 236 550 136", op: 1 },
                v: { op: 1, label: "v_B" }
            },
            {
                // Step 3: C lands
                a: { x: 630, y: 336, lx: 0, ly: 30, op: 0.3 },
                b: { x: 590, y: 336, lx: 0, ly: 30, op: 0.3 },
                c: { x: 550, y: 336, lx: 0, ly: 30, op: 1 },
                sab: { d: "M 630 336 Q 610 350 590 336", op: 0.3 },
                sbc: { d: "M 590 336 Q 570 350 550 336", op: 0.3 },
                v: { op: 1, label: "v_C" }
            }
        ];
        
        function updateScene() {
            const s = states[step];
            
            groupA.style.transform = `translate(${s.a.x}px, ${s.a.y}px)`;
            groupA.style.opacity = s.a.op;
            labelA.setAttribute('x', s.a.lx);
            labelA.setAttribute('y', s.a.ly);
            
            groupB.style.transform = `translate(${s.b.x}px, ${s.b.y}px)`;
            groupB.style.opacity = s.b.op;
            labelB.setAttribute('x', s.b.lx);
            labelB.setAttribute('y', s.b.ly);
            
            groupC.style.transform = `translate(${s.c.x}px, ${s.c.y}px)`;
            groupC.style.opacity = s.c.op;
            labelC.setAttribute('x', s.c.lx);
            labelC.setAttribute('y', s.c.ly);
            
            stringAB.setAttribute('d', s.sab.d);
            stringAB.style.opacity = s.sab.op;
            
            stringBC.setAttribute('d', s.sbc.d);
            stringBC.style.opacity = s.sbc.op;
            
            velocity.style.opacity = s.v.op;
            if (s.v.label) {
                velocityLabel.textContent = s.v.label;
            }
            
            btnPrev.disabled = step === 0;
            btnNext.disabled = step === maxStep;
        }
        
        btnPrev.addEventListener('click', () => {
            if (step > 0) {
                step--;
                updateScene();
            }
        });
        
        btnNext.addEventListener('click', () => {
            if (step < maxStep) {
                step++;
                updateScene();
            }
        });
        
        // Initialize
        updateScene();
    })();
    </script>
