    <script>
    (function() {
        let currentStep = 0;
        const maxStep = 3;

        const btnPrev = document.getElementById('scene1-btn-prev');
        const btnNext = document.getElementById('scene1-btn-next');

        const step1Group = document.getElementById('scene1-step1-group');
        const step1Force = document.getElementById('scene1-step1-force');
        const step2Group = document.getElementById('scene1-step2-group');
        const step2Force = document.getElementById('scene1-step2-force');
        const validRegion = document.getElementById('scene1-valid-region');

        function updateScene() {
            btnPrev.disabled = currentStep === 0;
            btnNext.disabled = currentStep === maxStep;

            const groups = [
                { el: step1Group, visible: currentStep >= 1 },
                { el: step1Force, visible: currentStep >= 1 && currentStep < 3 },
                { el: step2Group, visible: currentStep >= 2 },
                { el: step2Force, visible: currentStep >= 2 && currentStep < 3 },
                { el: validRegion, visible: currentStep >= 3 }
            ];

            groups.forEach(g => {
                if (g.visible) {
                    if (g.el.style.display === 'none') {
                        g.el.style.display = '';
                        // Force reflow to ensure transition works
                        void g.el.offsetWidth;
                    }
                    g.el.style.opacity = 1;
                } else {
                    g.el.style.opacity = 0;
                    // Wait for transition to finish before hiding
                    setTimeout(() => {
                        if (g.el.style.opacity == 0) {
                            g.el.style.display = 'none';
                        }
                    }, 400);
                }
            });
        }

        btnPrev.addEventListener('click', () => {
            if (currentStep > 0) {
                currentStep--;
                updateScene();
            }
        });

        btnNext.addEventListener('click', () => {
            if (currentStep < maxStep) {
                currentStep++;
                updateScene();
            }
        });

        // Initialize
        updateScene();
    })();
    </script>
