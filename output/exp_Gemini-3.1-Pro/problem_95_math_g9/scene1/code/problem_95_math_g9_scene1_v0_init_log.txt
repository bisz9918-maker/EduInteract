    <script>
    (function() {
        const btnPrev = document.getElementById('scene1-btn-prev');
        const btnNext = document.getElementById('scene1-btn-next');
        const stepText = document.getElementById('scene1-step-text');
        
        const jiaMap = document.getElementById('scene1-jia-map');
        const yiMap = document.getElementById('scene1-yi-map');
        const jiaDot = document.getElementById('scene1-jia-dot');
        const yiDot = document.getElementById('scene1-yi-dot');
        
        const jiaProjX = document.getElementById('scene1-jia-proj-x');
        const jiaProjY = document.getElementById('scene1-jia-proj-y');
        const yiProjX = document.getElementById('scene1-yi-proj-x');
        const yiProjY = document.getElementById('scene1-yi-proj-y');
        
        const jiaSpeed = document.getElementById('scene1-jia-speed');
        const yiSpeed = document.getElementById('scene1-yi-speed');
        const bcDistPath = document.getElementById('scene1-bc-dist-path');
        const bcDistText = document.getElementById('scene1-bc-dist-text');
        
        const distHighlight = document.getElementById('scene1-dist-highlight');
        const distPath = document.getElementById('scene1-dist-path');
        const distText = document.getElementById('scene1-dist-text');
        
        let currentStep = 0;
        const maxStep = 3;
        
        // Target times for each step: 0, 4, 6, 5
        const stepTimes = [0, 4, 6, 5];
        let currentT = 0;
        let targetT = 0;
        let animationFrameId = null;
        
        function updateElements(t) {
            // Map Jia
            let jiaMapX = 100;
            if (t <= 4) {
                jiaMapX = 100 + 100 * t;
            } else {
                jiaMapX = 500 + 100 * (t - 4);
            }
            jiaMap.setAttribute('transform', `translate(${jiaMapX}, 100)`);
            
            // Map Yi
            let yiMapX = 500;
            if (t > 1) {
                yiMapX = 500 - 80 * (t - 1);
                yiMap.setAttribute('opacity', '1');
            } else {
                yiMap.setAttribute('opacity', '0');
            }
            yiMap.setAttribute('transform', `translate(${yiMapX}, 100)`);
            
            // Graph Jia
            let jiaGraphX = 100 + 80 * t;
            let jiaGraphY = 240;
            if (t <= 4) {
                jiaGraphY = 240 + 40 * t;
            } else {
                jiaGraphY = 400 - 40 * (t - 4);
            }
            jiaDot.setAttribute('cx', jiaGraphX);
            jiaDot.setAttribute('cy', jiaGraphY);
            
            jiaProjX.setAttribute('x1', jiaGraphX);
            jiaProjX.setAttribute('x2', jiaGraphX);
            jiaProjX.setAttribute('y1', jiaGraphY);
            jiaProjX.setAttribute('opacity', t > 0 ? '1' : '0');
            
            jiaProjY.setAttribute('x1', jiaGraphX);
            jiaProjY.setAttribute('y1', jiaGraphY);
            jiaProjY.setAttribute('y2', jiaGraphY);
            jiaProjY.setAttribute('opacity', t > 0 ? '1' : '0');
            
            // Graph Yi
            let yiGraphX = 100 + 80 * t;
            let yiGraphY = 400;
            if (t > 1) {
                yiGraphY = 400 - 32 * (t - 1);
                yiDot.setAttribute('opacity', '1');
                yiProjX.setAttribute('opacity', '1');
                yiProjY.setAttribute('opacity', '1');
            } else {
                yiDot.setAttribute('opacity', '0');
                yiProjX.setAttribute('opacity', '0');
                yiProjY.setAttribute('opacity', '0');
            }
            yiDot.setAttribute('cx', yiGraphX);
            yiDot.setAttribute('cy', yiGraphY);
            
            yiProjX.setAttribute('x1', yiGraphX);
            yiProjX.setAttribute('x2', yiGraphX);
            yiProjX.setAttribute('y1', yiGraphY);
            
            yiProjY.setAttribute('x1', yiGraphX);
            yiProjY.setAttribute('y1', yiGraphY);
            yiProjY.setAttribute('y2', yiGraphY);
        }
        
        function updateVisibility() {
            // Step 1: Jia speed
            if (currentStep >= 1) {
                jiaSpeed.setAttribute('opacity', '1');
            } else {
                jiaSpeed.setAttribute('opacity', '0');
            }
            
            // Step 2: Yi speed, BC dist
            if (currentStep >= 2) {
                yiSpeed.setAttribute('opacity', '1');
                bcDistPath.setAttribute('opacity', '1');
                bcDistText.setAttribute('opacity', '1');
            } else {
                yiSpeed.setAttribute('opacity', '0');
                bcDistPath.setAttribute('opacity', '0');
                bcDistText.setAttribute('opacity', '0');
            }
            
            // Step 3: Dist highlight
            if (currentStep === 3) {
                distHighlight.setAttribute('opacity', '1');
                distPath.setAttribute('opacity', '1');
                distText.setAttribute('opacity', '1');
            } else {
                distHighlight.setAttribute('opacity', '0');
                distPath.setAttribute('opacity', '0');
                distText.setAttribute('opacity', '0');
            }
        }
        
        function animate() {
            const diff = targetT - currentT;
            if (Math.abs(diff) < 0.01) {
                currentT = targetT;
                updateElements(currentT);
                updateVisibility();
                return;
            }
            currentT += diff * 0.05;
            updateElements(currentT);
            animationFrameId = requestAnimationFrame(animate);
        }
        
        function setStep(step) {
            currentStep = step;
            targetT = stepTimes[currentStep];
            
            stepText.textContent = `${currentStep} / ${maxStep}`;
            btnPrev.disabled = currentStep === 0;
            btnNext.disabled = currentStep === maxStep;
            
            if (animationFrameId) {
                cancelAnimationFrame(animationFrameId);
            }
            animate();
        }
        
        btnPrev.addEventListener('click', () => {
            if (currentStep > 0) setStep(currentStep - 1);
        });
        
        btnNext.addEventListener('click', () => {
            if (currentStep < maxStep) setStep(currentStep + 1);
        });
        
        // Initial setup
        updateElements(0);
    })();
    </script>
