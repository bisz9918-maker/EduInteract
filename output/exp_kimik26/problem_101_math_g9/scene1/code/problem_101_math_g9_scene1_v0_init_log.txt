    <script>
    (function() {
        // Scene 1 实现
        let currentStep = 0;
        const totalSteps = 7;
        let isAnimating = false;

        const btnPrev = document.getElementById('scene1-btn-prev');
        const btnNext = document.getElementById('scene1-btn-next');

        // 元素引用
        const els = {
            square: document.getElementById('scene1-square'),
            labelA: document.getElementById('scene1-label-a'),
            labelB: document.getElementById('scene1-label-b'),
            labelC: document.getElementById('scene1-label-c'),
            labelD: document.getElementById('scene1-label-d'),
            diagonal: document.getElementById('scene1-diagonal'),
            creaseLabel: document.getElementById('scene1-crease-label'),
            foldTri: document.getElementById('scene1-fold-tri'),
            resultTri: document.getElementById('scene1-result-tri'),
            triA: document.getElementById('scene1-tri-a'),
            triC: document.getElementById('scene1-tri-c'),
            triDb: document.getElementById('scene1-tri-db'),
            angleA: document.getElementById('scene1-angle-a'),
            angleC: document.getElementById('scene1-angle-c'),
            angleDb: document.getElementById('scene1-angle-db'),
            triHoleA: document.getElementById('scene1-tri-hole-a'),
            triHoleC: document.getElementById('scene1-tri-hole-c'),
            triHoleDb: document.getElementById('scene1-tri-hole-db'),
            legendBlue: document.getElementById('scene1-legend-blue'),
            legendRed: document.getElementById('scene1-legend-red'),
            legendText1: document.getElementById('scene1-legend-text1'),
            legendText2: document.getElementById('scene1-legend-text2'),
            smallTri: document.getElementById('scene1-small-tri'),
            smallHoleA: document.getElementById('scene1-small-hole-a'),
            smallHoleC: document.getElementById('scene1-small-hole-c'),
            smallHoleDb: document.getElementById('scene1-small-hole-db'),
            arrow1: document.getElementById('scene1-arrow-1'),
            arrow2: document.getElementById('scene1-arrow-2'),
            arrow3: document.getElementById('scene1-arrow-3'),
            arrow4: document.getElementById('scene1-arrow-4'),
            arrowLabel1: document.getElementById('scene1-arrow-label-1'),
            arrowLabel2: document.getElementById('scene1-arrow-label-2'),
            arrowLabel3: document.getElementById('scene1-arrow-label-3'),
            arrowLabel4: document.getElementById('scene1-arrow-label-4'),
            holeA: document.getElementById('scene1-hole-a'),
            holeB: document.getElementById('scene1-hole-b'),
            holeC: document.getElementById('scene1-hole-c'),
            holeD: document.getElementById('scene1-hole-d'),
            optionA: document.getElementById('scene1-option-a'),
            optionB: document.getElementById('scene1-option-b'),
            optionC: document.getElementById('scene1-option-c'),
            optionD: document.getElementById('scene1-option-d'),
            optLabelA: document.getElementById('scene1-opt-label-a'),
            optLabelB: document.getElementById('scene1-opt-label-b'),
            optLabelC: document.getElementById('scene1-opt-label-c'),
            optLabelD: document.getElementById('scene1-opt-label-d'),
            correctLabel: document.getElementById('scene1-correct-label'),
        };

        function show(el) { if(el) { el.style.display = 'block'; el.style.opacity = '1'; } }
        function hide(el) { if(el) { el.style.display = 'none'; el.style.opacity = '0'; } }

        function render() {
            // 先隐藏所有元素
            hide(els.square);
            hide(els.labelA); hide(els.labelB); hide(els.labelC); hide(els.labelD);
            hide(els.diagonal);
            hide(els.creaseLabel);
            hide(els.foldTri);
            hide(els.resultTri);
            hide(els.triA); hide(els.triC); hide(els.triDb);
            hide(els.angleA); hide(els.angleC); hide(els.angleDb);
            hide(els.triHoleA); hide(els.triHoleC); hide(els.triHoleDb);
            hide(els.legendBlue); hide(els.legendRed);
            hide(els.legendText1); hide(els.legendText2);
            hide(els.smallTri);
            hide(els.smallHoleA); hide(els.smallHoleC); hide(els.smallHoleDb);
            hide(els.arrow1); hide(els.arrow2); hide(els.arrow3); hide(els.arrow4);
            hide(els.arrowLabel1); hide(els.arrowLabel2); hide(els.arrowLabel3); hide(els.arrowLabel4);
            hide(els.holeA); hide(els.holeB); hide(els.holeC); hide(els.holeD);
            hide(els.optionA); hide(els.optionB); hide(els.optionC); hide(els.optionD);
            hide(els.optLabelA); hide(els.optLabelB); hide(els.optLabelC); hide(els.optLabelD);
            hide(els.correctLabel);

            // 重置正方形洞颜色
            els.holeA.setAttribute('fill', '#1565c0');
            els.holeB.setAttribute('fill', '#c62828');
            els.holeC.setAttribute('fill', '#1565c0');
            els.holeD.setAttribute('fill', '#c62828');

            // 按钮状态
            btnPrev.disabled = currentStep === 0 || isAnimating;
            btnNext.disabled = currentStep === totalSteps || isAnimating;

            switch(currentStep) {
                case 0:
                    show(els.square);
                    show(els.labelA); show(els.labelB); show(els.labelC); show(els.labelD);
                    break;
                case 1:
                    show(els.square);
                    show(els.labelA); show(els.labelB); show(els.labelC); show(els.labelD);
                    show(els.diagonal);
                    show(els.creaseLabel);
                    break;
                case 2:
                    show(els.square);
                    show(els.labelA); show(els.labelB); show(els.labelC); show(els.labelD);
                    show(els.diagonal);
                    show(els.creaseLabel);
                    show(els.foldTri);
                    break;
                case 3:
                    show(els.resultTri);
                    show(els.triA); show(els.triC); show(els.triDb);
                    show(els.angleA); show(els.angleC); show(els.angleDb);
                    break;
                case 4:
                    show(els.resultTri);
                    show(els.triA); show(els.triC); show(els.triDb);
                    show(els.angleA); show(els.angleC); show(els.angleDb);
                    show(els.triHoleA); show(els.triHoleC); show(els.triHoleDb);
                    show(els.legendBlue); show(els.legendRed);
                    show(els.legendText1); show(els.legendText2);
                    break;
                case 5:
                    show(els.square);
                    show(els.labelA); show(els.labelB); show(els.labelC); show(els.labelD);
                    show(els.smallTri);
                    show(els.smallHoleA); show(els.smallHoleC); show(els.smallHoleDb);
                    show(els.arrow1); show(els.arrow2); show(els.arrow3); show(els.arrow4);
                    show(els.arrowLabel1); show(els.arrowLabel2); show(els.arrowLabel3); show(els.arrowLabel4);
                    show(els.holeA); show(els.holeB); show(els.holeC); show(els.holeD);
                    break;
                case 6:
                    show(els.square);
                    show(els.labelA); show(els.labelB); show(els.labelC); show(els.labelD);
                    show(els.holeA); show(els.holeB); show(els.holeC); show(els.holeD);
                    els.holeA.setAttribute('fill', '#2e7d32');
                    els.holeB.setAttribute('fill', '#2e7d32');
                    els.holeC.setAttribute('fill', '#2e7d32');
                    els.holeD.setAttribute('fill', '#2e7d32');
                    break;
                case 7:
                    show(els.square);
                    show(els.labelA); show(els.labelB); show(els.labelC); show(els.labelD);
                    show(els.holeA); show(els.holeB); show(els.holeC); show(els.holeD);
                    els.holeA.setAttribute('fill', '#2e7d32');
                    els.holeB.setAttribute('fill', '#2e7d32');
                    els.holeC.setAttribute('fill', '#2e7d32');
                    els.holeD.setAttribute('fill', '#2e7d32');
                    show(els.optionA); show(els.optionB); show(els.optionC); show(els.optionD);
                    show(els.optLabelA); show(els.optLabelB); show(els.optLabelC); show(els.optLabelD);
                    show(els.correctLabel);
                    break;
            }
        }

        function easeInOut(t) {
            return t < 0.5 ? 2 * t * t : -1 + (4 - 2 * t) * t;
        }

        function animateFold() {
            isAnimating = true;
            btnPrev.disabled = true;
            btnNext.disabled = true;
            const startTime = performance.now();
            const duration = 800;
            const startX = 500, startY = 120;
            const endX = 300, endY = 320;

            function frame(now) {
                const elapsed = now - startTime;
                const t = Math.min(elapsed / duration, 1);
                const e = easeInOut(t);
                const Bx = startX + (endX - startX) * e;
                const By = startY + (endY - startY) * e;
                els.foldTri.setAttribute('d', `M 300 120 L ${Bx} ${By} L 500 320 Z`);
                if (t < 1) {
                    requestAnimationFrame(frame);
                } else {
                    isAnimating = false;
                    btnPrev.disabled = currentStep === 0;
                    btnNext.disabled = currentStep === totalSteps;
                }
            }
            requestAnimationFrame(frame);
        }

        function nextStep() {
            if (currentStep < totalSteps && !isAnimating) {
                currentStep++;
                render();
                if (currentStep === 2) {
                    animateFold();
                }
            }
        }

        function prevStep() {
            if (currentStep > 0 && !isAnimating) {
                currentStep--;
                render();
            }
        }

        btnPrev.addEventListener('click', prevStep);
        btnNext.addEventListener('click', nextStep);

        // 初始渲染
        render();
    })();
    </script>
