    <script>
    (function() {
        // Scene 1 状态机：0~4 共5个步骤
        let state = 0;
        const maxState = 4;

        const btnPrev = document.getElementById('scene1-btn-prev');
        const btnNext = document.getElementById('scene1-btn-next');
        const stepText = document.getElementById('scene1-step-text');

        const groups = [
            null, // state 0 无额外组
            document.getElementById('scene1-step1-group'),
            document.getElementById('scene1-step2-group'),
            document.getElementById('scene1-step3-group'),
            document.getElementById('scene1-step4-group')
        ];

        function updateUI() {
            stepText.textContent = '步骤 ' + (state + 1) + '/5';
            btnPrev.disabled = (state === 0);
            btnNext.disabled = (state === maxState);

            for (let i = 1; i <= maxState; i++) {
                if (i <= state) {
                    groups[i].classList.remove('scene1-hidden');
                } else {
                    groups[i].classList.add('scene1-hidden');
                }
            }
        }

        btnPrev.addEventListener('click', function() {
            if (state > 0) {
                state--;
                updateUI();
            }
        });

        btnNext.addEventListener('click', function() {
            if (state < maxState) {
                state++;
                updateUI();
            }
        });

        // 初始化
        updateUI();
    })();
    </script>
