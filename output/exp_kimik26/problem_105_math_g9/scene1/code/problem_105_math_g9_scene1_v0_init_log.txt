    <script>
    (function() {
        // Scene 1 实现：旋转变换构造与全等关系建立
        // 坐标表（实现对照依据）：
        // A(400,120), B(180,340), C(620,340), D(460,340), E(274,340), F(180,180)
        // 标签：A(400,105), B(165,355), C(635,355), D(460,378), E(274,378), F(165,172)
        // 角度标注：∠BAC(435,110), ∠DAE(340,105), ∠DAF(295,110), ∠FBE(160,220)
        // 关系标注：AB=AC(130,125), AD=AF(80,165), △ABF≌△ACD(660,150), BF=DC(660,195)
        //           ∠ABF=∠C=45°(50,285), ∠ABC=45°(150,415), 推导式(300,415)

        let currentStep = 0;
        const totalSteps = 5;

        const btnPrev = document.getElementById('scene1-btn-prev');
        const btnNext = document.getElementById('scene1-btn-next');
        const indicator = document.getElementById('scene1-step-indicator');

        function setOpacity(id, opacity) {
            const el = document.getElementById(id);
            if (el) el.style.opacity = opacity;
        }

        function setStroke(id, color) {
            const el = document.getElementById(id);
            if (el) el.setAttribute('stroke', color);
        }

        function render() {
            // ========== 重置所有动态元素 ==========
            // 点与线段
            setOpacity('scene1-point-f', 0);
            setOpacity('scene1-line-af', 0);
            setOpacity('scene1-line-bf', 0);
            setOpacity('scene1-line-ef', 0);
            setOpacity('scene1-line-dc-blue', 0);

            // 三角形填充
            setOpacity('scene1-tri-adc', 0);
            setOpacity('scene1-tri-abf', 0);

            // 旋转相关
            setOpacity('scene1-arc-df', 0);
            setOpacity('scene1-line-af-dashed', 0);
            setOpacity('scene1-rotate-arrow', 0);

            // 角度弧线
            setOpacity('scene1-arc-daf', 0);

            // 直角符号
            setOpacity('scene1-right-angle-fbe', 0);

            // tick 标记
            setOpacity('scene1-tick-bf', 0);
            setOpacity('scene1-tick-dc', 0);

            // 标签
            setOpacity('scene1-label-f', 0);

            // 角度数值
            setOpacity('scene1-text-daf', 0);
            setOpacity('scene1-text-fbe', 0);

            // 关系与推导标注
            setOpacity('scene1-text-ab-ac', 0);
            setOpacity('scene1-text-ad-af', 0);
            setOpacity('scene1-text-congruent', 0);
            setOpacity('scene1-text-bf-dc', 0);
            setOpacity('scene1-text-abf-c', 0);
            setOpacity('scene1-text-abc', 0);
            setOpacity('scene1-text-derive', 0);

            // 重置线段颜色
            setStroke('scene1-line-ad', '#000000');

            // ========== 步骤0：基础图形 ==========
            if (currentStep >= 0) {
                setOpacity('scene1-text-ab-ac', 1);
            }

            // ========== 步骤1：高亮待旋转三角形 ==========
            if (currentStep >= 1) {
                // △ADC 蓝色填充
                setOpacity('scene1-tri-adc', 1);
                // AD 变蓝
                setStroke('scene1-line-ad', '#1565c0');
                // DC 变蓝
                setOpacity('scene1-line-dc-blue', 1);
                // AF 预告虚线
                setOpacity('scene1-line-af-dashed', 1);
                // 旋转方向箭头
                setOpacity('scene1-rotate-arrow', 1);
                // AD = AF 标注
                setOpacity('scene1-text-ad-af', 1);
            }

            // ========== 步骤2：旋转动画 ==========
            if (currentStep >= 2) {
                // 点 F
                setOpacity('scene1-point-f', 1);
                // AF、BF、EF 线段
                setOpacity('scene1-line-af', 1);
                setOpacity('scene1-line-bf', 1);
                setOpacity('scene1-line-ef', 1);
                // △ABF 红色填充
                setOpacity('scene1-tri-abf', 1);
                // 旋转轨迹
                setOpacity('scene1-arc-df', 1);
                // F 标签
                setOpacity('scene1-label-f', 1);
                // ∠DAF = 90°
                setOpacity('scene1-arc-daf', 1);
                setOpacity('scene1-text-daf', 1);
            }

            // ========== 步骤3：全等关系 ==========
            if (currentStep >= 3) {
                setOpacity('scene1-text-congruent', 1);
                setOpacity('scene1-text-bf-dc', 1);
                setOpacity('scene1-text-abf-c', 1);
                setOpacity('scene1-tick-bf', 1);
                setOpacity('scene1-tick-dc', 1);
            }

            // ========== 步骤4：直角推导 ==========
            if (currentStep >= 4) {
                setOpacity('scene1-text-abc', 1);
                setOpacity('scene1-right-angle-fbe', 1);
                setOpacity('scene1-text-fbe', 1);
                setOpacity('scene1-text-derive', 1);
            }

            // ========== 更新控件状态 ==========
            btnPrev.disabled = (currentStep === 0);
            btnNext.disabled = (currentStep === totalSteps - 1);
            indicator.textContent = '步骤 ' + (currentStep + 1) + '/' + totalSteps;
        }

        btnPrev.addEventListener('click', function() {
            if (currentStep > 0) {
                currentStep--;
                render();
            }
        });

        btnNext.addEventListener('click', function() {
            if (currentStep < totalSteps - 1) {
                currentStep++;
                render();
            }
        });

        // 初始渲染
        render();
    })();
    </script>
