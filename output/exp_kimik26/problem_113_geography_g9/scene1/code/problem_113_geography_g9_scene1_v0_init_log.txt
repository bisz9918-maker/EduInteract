        function setVisibility(id, visible) {
            var el = document.getElementById(id);
            if (!el) return;
            if (visible) {
                el.style.display = '';
                el.style.opacity = '1';
            } else {
                el.style.display = 'none';

        function renderStep() {
            var config = stepElements[currentStep];
            if (!config) return;

            config.show.forEach(function(id) { setVisibility(id, true); });
            config.hide.forEach(function(id) { setVisibility(id, false); });

            document.getElementById('scene1-step-indicator').textContent = '步骤 ' + (currentStep + 1) + '/' + totalSteps;

            document.getElementById('scene1-btn-next').disabled = (currentStep === totalSteps - 1);
        }

        document.getElementById('scene1-btn-prev').addEventListener('click', function() {
            if (currentStep > 0) {
                currentStep--;
                renderStep();
            }
        });

        document.getElementById('scene1-btn-next').addEventListener('click', function() {
            if (currentStep < totalSteps - 1) {
                currentStep++;
                renderStep();
            }
