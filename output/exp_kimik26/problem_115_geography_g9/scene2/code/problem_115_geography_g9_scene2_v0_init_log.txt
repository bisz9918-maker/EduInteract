    <script>
    (function() {
        // Scene 2 实现
        var currentStep = 0;
        var totalSteps = 5;

        var btnPrev = document.getElementById('scene2-btn-prev');
        var btnNext = document.getElementById('scene2-btn-next');
        var stepIndicator = document.getElementById('scene2-step-indicator');
        var infoText = document.getElementById('scene2-info-text');

        // 各步骤显示的元素ID列表
        var stepElements = [
            // 步骤0：基础地球模型 + A点
            {
                show: ['scene2-point-a', 'scene2-label-a', 'scene2-label-hero-a'],
                hide: ['scene2-point-b', 'scene2-label-b', 'scene2-label-hero-b',
                       'scene2-tunnel-line', 'scene2-radius-oa', 'scene2-radius-ob',
                       'scene2-label-lat-n', 'scene2-label-lat-s',
                       'scene2-label-lon-e', 'scene2-label-lon-w',
                       'scene2-formula-group']
            },
            // 步骤1：遁地线 + B点
            {
                show: ['scene2-point-a', 'scene2-label-a', 'scene2-label-hero-a',
                       'scene2-point-b', 'scene2-label-b', 'scene2-label-hero-b',
                       'scene2-tunnel-line'],
                hide: ['scene2-radius-oa', 'scene2-radius-ob',
                       'scene2-label-lat-n', 'scene2-label-lat-s',
                       'scene2-label-lon-e', 'scene2-label-lon-w',
                       'scene2-formula-group']
            },
            // 步骤2：纬度对称
            {
                show: ['scene2-point-a', 'scene2-label-a', 'scene2-label-hero-a',
                       'scene2-point-b', 'scene2-label-b', 'scene2-label-hero-b',
                       'scene2-tunnel-line', 'scene2-radius-oa', 'scene2-radius-ob',
                       'scene2-label-lat-n', 'scene2-label-lat-s'],
                hide: ['scene2-label-lon-e', 'scene2-label-lon-w',
                       'scene2-formula-group']
            },
            // 步骤3：经度互补
            {
                show: ['scene2-point-a', 'scene2-label-a', 'scene2-label-hero-a',
                       'scene2-point-b', 'scene2-label-b', 'scene2-label-hero-b',
                       'scene2-tunnel-line', 'scene2-radius-oa', 'scene2-radius-ob',
                       'scene2-label-lat-n', 'scene2-label-lat-s',
                       'scene2-label-lon-e', 'scene2-label-lon-w'],
                hide: ['scene2-formula-group']
            },
            // 步骤4：公式总结
            {
                show: ['scene2-point-a', 'scene2-label-a', 'scene2-label-hero-a',
                       'scene2-point-b', 'scene2-label-b', 'scene2-label-hero-b',
                       'scene2-tunnel-line', 'scene2-radius-oa', 'scene2-radius-ob',
                       'scene2-label-lat-n', 'scene2-label-lat-s',
                       'scene2-label-lon-e', 'scene2-label-lon-w',
                       'scene2-formula-group']
            }
        ];

        var stepDescriptions = [
            '咸蛋超人住在 (30°N, 120°E)。<br>观察地球截面模型，找到A点位置。',
            '从A点钻入地底，始终保持直线前进并穿过地心到达B点。<br>B点就是面包超人的家。',
            '观察纬度关系：A点在30°N，B点在30°S。<br>对跖点的纬度数值相等，南北纬相反。',
            '观察经度关系：A点为120°E，B点为60°W。<br>120° + 60° = 180°，东西经相反。',
            '总结对跖点通用公式，得出本题结论。<br>教师可随时回退复习。'
        ];

        function showElement(id) {
            var el = document.getElementById(id);
            if (el) {
                el.style.display = '';
                el.style.opacity = '1';
            }
        }

        function hideElement(id) {
            var el = document.getElementById(id);
            if (el) {
                el.style.display = 'none';
                el.style.opacity = '0';
            }
        }

        function animateTunnelLine() {
            // 使用stroke-dashoffset动画绘制遁地线
            var line = document.getElementById('scene2-tunnel-line');
            if (!line) return;
            line.style.display = '';
            line.style.opacity = '1';

            // 计算线段长度
            var dx = 397 - 103;
            var dy = 165 - 335;
            var length = Math.sqrt(dx * dx + dy * dy);

            line.style.strokeDasharray = length;
            line.style.strokeDashoffset = length;

            // 强制重排
            line.getBoundingClientRect();

            // CSS transition动画
            line.style.transition = 'stroke-dashoffset 800ms ease-out';
            line.style.strokeDashoffset = '0';
        }

        function resetTunnelLine() {
            var line = document.getElementById('scene2-tunnel-line');
            if (!line) return;
            line.style.transition = 'none';
            line.style.strokeDasharray = '';
            line.style.strokeDashoffset = '';
        }

        function updateStep(step) {
            currentStep = step;

            // 更新按钮状态
            btnPrev.disabled = (currentStep === 0);
            btnNext.disabled = (currentStep === totalSteps - 1);
            stepIndicator.textContent = '步骤 ' + (currentStep + 1) + '/' + totalSteps;

            // 更新说明文字
            infoText.innerHTML = stepDescriptions[currentStep];

            // 显示/隐藏元素
            var cfg = stepElements[currentStep];
            cfg.show.forEach(showElement);
            cfg.hide.forEach(hideElement);

            // 步骤1特殊处理：遁地线动画
            if (currentStep === 1) {
                animateTunnelLine();
            } else {
                resetTunnelLine();
            }
        }

        btnPrev.addEventListener('click', function() {
            if (currentStep > 0) {
                updateStep(currentStep - 1);
            }
        });

        btnNext.addEventListener('click', function() {
            if (currentStep < totalSteps - 1) {
                updateStep(currentStep + 1);
            }
        });

        // 初始化
        updateStep(0);
    })();
    </script>
