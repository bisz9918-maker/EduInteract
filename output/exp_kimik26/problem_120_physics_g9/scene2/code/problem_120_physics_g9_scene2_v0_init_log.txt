                <div>P_L = I² R_L（R_L 不变）</div>

                <div style="color:#1565c0; font-size:16px;">∴ I_max / I_min = 3 / 2</div>
                <div style="color:#2e7d32; font-weight:bold; font-size:16px;">I_max = 0.3 A</div>

    <script>
    (function() {
        // Scene 2 实现
        var currentStep = 0;
        var totalSteps = 5;

        var stepHtmls = [
            document.getElementById('scene2-step0-html'),
            document.getElementById('scene2-step1-html'),
            document.getElementById('scene2-step2-html'),
            document.getElementById('scene2-step3-html'),
            document.getElementById('scene2-step4-html')
        ];

        var prevBtn = document.getElementById('scene2-prev-btn');
        var nextBtn = document.getElementById('scene2-next-btn');
        var indicator = document.getElementById('scene2-step-indicator');

        // 电路图元素引用
        var voltmeter = document.getElementById('scene2-voltmeter');
        var rheostat = document.getElementById('scene2-rheostat');
        var lamp = document.getElementById('scene2-lamp');
        var circuitGroup = document.getElementById('scene2-circuit-group');
        var rmaxLabel = document.getElementById('scene2-rmax-label');

        // 电压表引线
        var voltLineLeft = document.getElementById('scene2-volt-line-left');
        var voltLineRight = document.getElementById('scene2-volt-line-right');

        function showStep(n) {
            // 切换HTML步骤内容
            stepHtmls.forEach(function(el, i) {
                if (i === n) {
                    el.style.display = 'block';
                    requestAnimationFrame(function() {
                        el.style.opacity = '1';
                    });
                } else {
                    el.style.opacity = '0';
                    setTimeout(function() {
                        if (currentStep !== i) {
                            el.style.display = 'none';
                        }
                    }, 200);
                }
            });

            // 更新指示器
            indicator.textContent = '步骤 ' + (n + 1) + '/' + totalSteps;

            // 更新按钮状态
            prevBtn.disabled = (n === 0);
            nextBtn.disabled = (n === totalSteps - 1);

            // 电路图高亮控制
            resetCircuitHighlight();
            if (n === 1) {
                // 步骤1：电压表和变阻器红色高亮
                setHighlight(voltmeter, '#c62828', '2.5');
                setHighlight(rheostat, '#c62828', '2.5');
                setHighlight(voltLineLeft, '#c62828', '2');
                setHighlight(voltLineRight, '#c62828', '2');
                rmaxLabel.style.display = 'block';
                requestAnimationFrame(function() {
                    rmaxLabel.style.opacity = '1';
                });
            } else if (n === 2) {
                // 步骤2：灯泡蓝色高亮
                setHighlight(lamp, '#1565c0', '2.5');
                rmaxLabel.style.opacity = '0';
                setTimeout(function() {
                    rmaxLabel.style.display = 'none';
                }, 200);
            } else if (n === 3) {
                // 步骤3：整体电路蓝色高亮
                circuitGroup.querySelectorAll('circle, rect, path, line').forEach(function(el) {
                    if (el.id !== 'scene2-rmax-label' && el.tagName.toLowerCase() !== 'text') {
                        setHighlight(el, '#1565c0', '2');
                    }
                });
                rmaxLabel.style.opacity = '0';
                setTimeout(function() {
                    rmaxLabel.style.display = 'none';
                }, 200);
            } else {
                rmaxLabel.style.opacity = '0';
                setTimeout(function() {
                    rmaxLabel.style.display = 'none';
                }, 200);
            }
        }

        function setHighlight(el, color, width) {
            if (!el) return;
            var tag = el.tagName.toLowerCase();
            if (tag === 'text') return;
            if (tag === 'line' || tag === 'path' || tag === 'circle' || tag === 'rect') {
                el.setAttribute('stroke', color);
                el.setAttribute('stroke-width', width);
            }
        }

        function resetCircuitHighlight() {
            // 恢复所有电路元素默认样式
            var wires = document.querySelectorAll('.scene2-wire');
            wires.forEach(function(w) {
                w.setAttribute('stroke', '#000');
                w.setAttribute('stroke-width', '1.5');
            });

            var voltWires = document.querySelectorAll('.scene2-volt-wire');
            voltWires.forEach(function(w) {
                w.setAttribute('stroke', '#888');
                w.setAttribute('stroke-width', '1');
            });

            if (voltmeter) {
                var vCircle = voltmeter.querySelector('circle');
                if (vCircle) {
                    vCircle.setAttribute('stroke', '#000');
                    vCircle.setAttribute('stroke-width', '1.5');
                }
            }

            if (rheostat) {
                rheostat.setAttribute('stroke', '#000');
                rheostat.setAttribute('stroke-width', '1.5');
            }

            if (lamp) {
                var lCircle = lamp.querySelector('circle');
                if (lCircle) {
                    lCircle.setAttribute('stroke', '#000');
                    lCircle.setAttribute('stroke-width', '1.5');
                }
                var lLines = lamp.querySelectorAll('line');
                lLines.forEach(function(ln) {
                    ln.setAttribute('stroke', '#000');
                    ln.setAttribute('stroke-width', '1.5');
                });
            }

            // 恢复电池和开关
            var batteryLines = document.querySelectorAll('#scene2-battery line');
            batteryLines.forEach(function(ln) {
                ln.setAttribute('stroke', '#000');
                ln.setAttribute('stroke-width', '2.5');
            });

            var switchGroup = document.getElementById('scene2-switch');
            if (switchGroup) {
                switchGroup.querySelectorAll('line, circle').forEach(function(el) {
                    el.setAttribute('stroke', '#000');
                    if (el.tagName.toLowerCase() !== 'circle') {
                        el.setAttribute('stroke-width', '1.5');
                    }
                });
            }

            var ammeter = document.getElementById('scene2-ammeter');
            if (ammeter) {
                var aCircle = ammeter.querySelector('circle');
                if (aCircle) {
                    aCircle.setAttribute('stroke', '#000');
                    aCircle.setAttribute('stroke-width', '1.5');
                }
            }

            var sliderArrow = document.getElementById('scene2-slider-arrow');
            if (sliderArrow) {
                sliderArrow.setAttribute('stroke', '#000');
                sliderArrow.setAttribute('stroke-width', '1.5');
            }
        }

        prevBtn.addEventListener('click', function() {
            if (currentStep > 0) {
                currentStep--;
                showStep(currentStep);
            }
        });

        nextBtn.addEventListener('click', function() {
            if (currentStep < totalSteps - 1) {
                currentStep++;
                showStep(currentStep);
            }
        });

        // 初始化
        showStep(0);
    })();
    </script>
