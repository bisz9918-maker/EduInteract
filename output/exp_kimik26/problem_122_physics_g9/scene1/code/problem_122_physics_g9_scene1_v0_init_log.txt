    <script>
    (function() {
        let currentStep = 0;
        const totalSteps = 5;

        const btnPrev = document.getElementById('scene1-btn-prev');
        const btnNext = document.getElementById('scene1-btn-next');
        const stepIndicatorText = document.getElementById('scene1-step-indicator-text');

        const step1Grid = document.getElementById('scene1-step1-grid');
        const step2Arrows = document.getElementById('scene1-step2-arrows');
        const step2Rlabels = document.getElementById('scene1-step2-rlabels');
        const step3Formula = document.getElementById('scene1-step3-formula');
        const step4Missing = document.getElementById('scene1-step4-missing');
        const r22 = document.getElementById('scene1-r-22');

        const formulaLines = [
            document.getElementById('scene1-formula-1'),
            document.getElementById('scene1-formula-2'),
            document.getElementById('scene1-formula-3'),
            document.getElementById('scene1-formula-4'),
            document.getElementById('scene1-formula-5'),
            document.getElementById('scene1-formula-6')
        ];

        function updateStepIndicator() {
            const text = '步骤 ' + (currentStep + 1) + '/' + totalSteps;
            stepIndicatorText.textContent = text;
            btnPrev.disabled = currentStep === 0;
            btnNext.disabled = currentStep === totalSteps - 1;
        }

        function showElement(el) {
            if (!el) return;
            el.style.display = '';
            el.classList.add('visible');
        }

        function hideElement(el) {
            if (!el) return;
            el.classList.remove('visible');
        }

        function renderStep() {
            if (currentStep >= 1) {
                showElement(step1Grid);
            } else {
                hideElement(step1Grid);
            }

            if (currentStep >= 2) {
                showElement(step2Arrows);
                showElement(step2Rlabels);
                const allRlabels = step2Rlabels.querySelectorAll('.scene1-r-label');
                allRlabels.forEach(function(lbl) {
                    lbl.style.display = '';
                    lbl.classList.add('visible');
                });
            } else {
                hideElement(step2Arrows);
                hideElement(step2Rlabels);
            }

            if (currentStep >= 3) {
                showElement(step3Formula);
                formulaLines.forEach(function(line, idx) {
                    if (line) {
                        line.style.display = '';
                        setTimeout(function() {
                            line.classList.add('visible');
                        }, idx * 150);
                    }
                });
            } else {
                hideElement(step3Formula);
                formulaLines.forEach(function(line) {
                    if (line) line.classList.remove('visible');
                });
            }

            if (currentStep >= 4) {
                showElement(step4Missing);
                if (r22) {
                    r22.classList.remove('visible');
                    setTimeout(function() {
                        r22.style.display = 'none';
                    }, 300);
                }
            } else {
                hideElement(step4Missing);
                if (r22) {
                    r22.style.display = '';
                    r22.classList.add('visible');
                }
            }

            updateStepIndicator();
        }

        btnPrev.addEventListener('click', function() {
            if (currentStep > 0) {
                currentStep--;
                renderStep();
            }
        });

        btnNext.addEventListener('click', function() {
            if (currentStep < totalSteps - 1) {
                currentStep++;
                renderStep();
            }
        });

        renderStep();
    })();
    </script>
