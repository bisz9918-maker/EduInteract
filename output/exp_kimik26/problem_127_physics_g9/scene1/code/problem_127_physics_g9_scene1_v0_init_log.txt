    <script>
    (function() {
        // Scene 1 实现：平行平面玻璃砖折射光路分步演示

        var currentStep = 0;
        var maxStep = 6;

        // 元素引用
        var els = {
            normal1: document.getElementById('scene1-normal-1'),
            normal2: document.getElementById('scene1-normal-2'),
            rayInternal: document.getElementById('scene1-ray-internal'),
            rayExit: document.getElementById('scene1-ray-exit'),
            rayExtended: document.getElementById('scene1-ray-extended'),
            displacement: document.getElementById('scene1-displacement'),
            angleI: document.getElementById('scene1-angle-i'),
            angleR1: document.getElementById('scene1-angle-r1'),
            angleR2: document.getElementById('scene1-angle-r2'),
            labelAngleI: document.getElementById('scene1-label-angle-i'),
            labelAngleR1: document.getElementById('scene1-label-angle-r1'),
            labelAngleR2: document.getElementById('scene1-label-angle-r2'),
            labelD: document.getElementById('scene1-label-d'),
            labelAirTop: document.getElementById('scene1-label-air-top'),
            labelAirBottom: document.getElementById('scene1-label-air-bottom'),
            labelGlass: document.getElementById('scene1-label-glass'),
            labelInterface1: document.getElementById('scene1-label-interface1'),
            labelInterface2: document.getElementById('scene1-label-interface2'),
            labelNAir: document.getElementById('scene1-label-n-air'),
            labelNGlass: document.getElementById('scene1-label-n-glass'),
            conclusionBox: document.getElementById('scene1-conclusion-box'),
            conclusionText: document.getElementById('scene1-conclusion-text'),
            btnPrev: document.getElementById('scene1-btn-prev'),
            btnNext: document.getElementById('scene1-btn-next'),
            btnReset: document.getElementById('scene1-btn-reset')
        };

        // 工具函数：显示元素
        function show(el) {
            if (!el) return;
            el.style.opacity = '1';
            el.style.display = '';
        }

        // 工具函数：隐藏元素
        function hide(el) {
            if (!el) return;
            el.style.opacity = '0';
            if (el.tagName.toLowerCase() !== 'line' && el.tagName.toLowerCase() !== 'path') {
                el.style.display = 'none';
            }
        }

        // 隐藏所有动态元素
        function hideAllDynamic() {
            hide(els.normal1);
            hide(els.normal2);
            hide(els.rayInternal);
            hide(els.rayExit);
            hide(els.rayExtended);
            hide(els.displacement);
            hide(els.angleI);
            hide(els.angleR1);
            hide(els.angleR2);
            hide(els.labelAngleI);
            hide(els.labelAngleR1);
            hide(els.labelAngleR2);
            hide(els.labelD);
            hide(els.labelAirTop);
            hide(els.labelAirBottom);
            hide(els.labelGlass);
            hide(els.labelInterface1);
            hide(els.labelInterface2);
            hide(els.labelNAir);
            hide(els.labelNGlass);
            hide(els.conclusionBox);
        }

        // 渲染指定步骤
        function renderStep(step) {
            hideAllDynamic();

            // 步骤0：基础（玻璃砖 + 入射光）已在HTML中默认显示
            if (step >= 0) {
                show(els.normal1);
            }

            // 步骤1：入射角 + 界面① + 空气(上) + n_空
            if (step >= 1) {
                show(els.angleI);
                show(els.labelAngleI);
                show(els.labelInterface1);
                show(els.labelAirTop);
                show(els.labelNAir);
            }

            // 步骤2：第一次折射 + 玻璃内光线 + ∠r₁ + 玻璃 + n_玻
            if (step >= 2) {
                show(els.rayInternal);
                show(els.angleR1);
                show(els.labelAngleR1);
                show(els.labelGlass);
                show(els.labelNGlass);
            }

            // 步骤3：下表面法线N2 + 界面②
            if (step >= 3) {
                show(els.normal2);
                show(els.labelInterface2);
            }

            // 步骤4：第二次折射 + 出射光 + ∠r₂ + 空气(下)
            if (step >= 4) {
                show(els.rayExit);
                show(els.angleR2);
                show(els.labelAngleR2);
                show(els.labelAirBottom);
            }

            // 步骤5：入射光延长线 + 侧向位移 + d + 结论框"入射光 ∥ 出射光"
            if (step >= 5) {
                show(els.rayExtended);
                show(els.displacement);
                show(els.labelD);
                show(els.conclusionBox);
                els.conclusionText.textContent = '入射光 ∥ 出射光';
                els.conclusionBox.style.borderWidth = '1.5px';
            }

            // 步骤6：完整结论公式
            if (step >= 6) {
                show(els.conclusionBox);
                els.conclusionText.textContent = 'n空·sin i = n玻·sin r₁ = n空·sin r₂';
                els.conclusionBox.style.borderWidth = '2.5px';
            }

            // 更新按钮状态
            els.btnPrev.disabled = (step <= 0);
            els.btnNext.disabled = (step >= maxStep);
        }

        // 事件绑定
        els.btnPrev.addEventListener('click', function() {
            if (currentStep > 0) {
                currentStep--;
                renderStep(currentStep);
            }
        });

        els.btnNext.addEventListener('click', function() {
            if (currentStep < maxStep) {
                currentStep++;
                renderStep(currentStep);
            }
        });

        els.btnReset.addEventListener('click', function() {
            currentStep = 0;
            renderStep(currentStep);
        });

        // 初始化
        renderStep(currentStep);
    })();
    </script>
