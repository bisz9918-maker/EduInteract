<script>
(function() {
  var currentStep = 0;
  var totalSteps = 4;
  
  var btnPrev = document.getElementById('scene2-btn-prev');
  var btnNext = document.getElementById('scene2-btn-next');
  var indicator = document.getElementById('scene2-step-indicator');
  var tooltip = document.getElementById('scene2-tooltip');
  var tableWrap = document.getElementById('scene2-table-wrap');
  var overview = document.getElementById('scene2-overview');
  var container = document.getElementById('scene2-container');
  
  function showStep(n) {
    // 隐藏所有SVG步骤
    for (var i = 0; i <= totalSteps; i++) {
      var g = document.getElementById('scene2-step-' + i);
      if (g) {
        g.style.opacity = '0';
        setTimeout((function(el) {
          return function() {
            if (el.style.opacity === '0') el.style.display = 'none';
          };
        })(g), 200);
      }
    }
    
    // 显示目标SVG步骤
    var target = document.getElementById('scene2-step-' + n);
    if (target) {
      target.style.display = 'block';
      requestAnimationFrame(function() {
        target.style.opacity = '1';
      });
    }
    
    // 验证表在步骤1-4显示
    if (n >= 1) {
      tableWrap.style.display = 'block';
      requestAnimationFrame(function() {
        tableWrap.style.opacity = '1';
      });
    } else {
      tableWrap.style.opacity = '0';
      setTimeout(function() {
        if (tableWrap.style.opacity === '0') tableWrap.style.display = 'none';
      }, 200);
    }
    
    // 概览按钮在步骤0显示
    if (n === 0) {
      overview.style.display = 'block';
    } else {
      overview.style.display = 'none';
    }
    
    // 更新按钮状态
    btnPrev.disabled = (n <= 0);
    btnNext.disabled = (n >= totalSteps);
    
    // 更新指示器
    indicator.textContent = '步骤 ' + n + '/' + totalSteps;
    
    currentStep = n;
  }
  
  btnPrev.addEventListener('click', function() {
    if (currentStep > 0) showStep(currentStep - 1);
  });
  
  btnNext.addEventListener('click', function() {
    if (currentStep < totalSteps) showStep(currentStep + 1);
  });
  
  // 步骤0按钮跳转
  var overviewBtns = overview.querySelectorAll('.scene2-overview-btn');
  overviewBtns.forEach(function(btn) {
    btn.addEventListener('click', function() {
      var step = parseInt(btn.getAttribute('data-step'));
      showStep(step);
    });
  });
  
  // 悬停高亮
  var groups = container.querySelectorAll('.scene2-group');
  groups.forEach(function(g) {
    var highlight = g.querySelector('.scene2-highlight');
    var tipText = g.getAttribute('data-tooltip');
    
    g.addEventListener('mouseenter', function(e) {
      if (highlight) highlight.style.opacity = '0.3';
      if (tipText) {
        tooltip.textContent = tipText;
        tooltip.style.display = 'block';
        var rect = g.getBoundingClientRect();
        var containerRect = container.getBoundingClientRect();
        tooltip.style.left = (rect.left - containerRect.left + rect.width / 2 - tooltip.offsetWidth / 2) + 'px';
        tooltip.style.top = (rect.top - containerRect.top - tooltip.offsetHeight - 8) + 'px';
      }
    });
    
    g.addEventListener('mouseleave', function() {
      if (highlight) highlight.style.opacity = '0';
      tooltip.style.display = 'none';
    });
