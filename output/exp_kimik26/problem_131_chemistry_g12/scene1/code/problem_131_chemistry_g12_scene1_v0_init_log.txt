    <script>
    (function() {
        // Scene 1 实现：原电池-电解池组合装置分步导航
        var currentStep = 0;
        var totalSteps = 6;

        var btnPrev = document.getElementById('scene1-btn-prev');
        var btnNext = document.getElementById('scene1-btn-next');
        var stepText = document.getElementById('scene1-step-text');

        var elWireConnect = document.getElementById('scene1-wire-mid-connect');
        var elSwitchBridge = document.getElementById('scene1-switch-bridge');
        var elHighlightRight = document.getElementById('scene1-highlight-right');
        var elLabelPrimary = document.getElementById('scene1-label-primary');
        var elElectrodeZn = document.getElementById('scene1-electrode-zn');
        var elElectrodeC = document.getElementById('scene1-electrode-c');
        var elLabelNegative = document.getElementById('scene1-label-negative');
        var elLabelPositive = document.getElementById('scene1-label-positive');
        var elHighlightLeft = document.getElementById('scene1-highlight-left');
        var elLabelElectrolytic = document.getElementById('scene1-label-electrolytic');
        var elElectrodePt = document.getElementById('scene1-electrode-pt');
        var elElectrodeCu = document.getElementById('scene1-electrode-cu');
        var elLabelAnode = document.getElementById('scene1-label-anode');
        var elLabelCathode = document.getElementById('scene1-label-cathode');
        var elArrowETop = document.getElementById('scene1-arrow-e-top');
        var elLabelETop = document.getElementById('scene1-label-e-top');
        var elArrowITop = document.getElementById('scene1-arrow-i-top');
        var elLabelITop = document.getElementById('scene1-label-i-top');
        var elArrowEBottom = document.getElementById('scene1-arrow-e-bottom');
        var elLabelEBottom = document.getElementById('scene1-label-e-bottom');
        var elArrowIBottom = document.getElementById('scene1-arrow-i-bottom');
        var elLabelIBottom = document.getElementById('scene1-label-i-bottom');

        function setVisible(el, visible) {
            if (!el) return;
            if (visible) {
                el.classList.add('scene1-visible');
            } else {
                el.classList.remove('scene1-visible');
            }
        }

        function updateScene() {
            stepText.textContent = '步骤 ' + (currentStep + 1) + '/' + totalSteps;
            btnPrev.disabled = (currentStep === 0);
            btnNext.disabled = (currentStep === totalSteps - 1);

            // 步骤0: 初始状态（开关断开，无动态元素）
            // 步骤1: 闭合开关，右侧高亮+原电池标签
            // 步骤2: Zn负极(红)，C正极(蓝)
            // 步骤3: 左侧高亮+电解池标签
            // 步骤4: Pt阳极(蓝)，Cu阴极(红)
            // 步骤5: 电子与电流箭头

            // 开关连通（步骤1+）
            setVisible(elWireConnect, currentStep >= 1);
            setVisible(elSwitchBridge, currentStep >= 1);

            // 右侧原电池（步骤1+）
            setVisible(elHighlightRight, currentStep >= 1);
            setVisible(elLabelPrimary, currentStep >= 1);

            // 原电池极性（步骤2+）
            if (currentStep >= 2) {
                elElectrodeZn.setAttribute('stroke', '#c62828');
                elElectrodeZn.setAttribute('stroke-width', '3');
                elElectrodeC.setAttribute('stroke', '#1565c0');
                elElectrodeC.setAttribute('stroke-width', '3');
            } else {
                elElectrodeZn.setAttribute('stroke', 'none');
                elElectrodeC.setAttribute('stroke', 'none');
            }
            setVisible(elLabelNegative, currentStep >= 2);
            setVisible(elLabelPositive, currentStep >= 2);

            // 左侧电解池（步骤3+）
            setVisible(elHighlightLeft, currentStep >= 3);
            setVisible(elLabelElectrolytic, currentStep >= 3);

            // 电解池极性（步骤4+）
            if (currentStep >= 4) {
                elElectrodePt.setAttribute('stroke', '#1565c0');
                elElectrodePt.setAttribute('stroke-width', '3');
                elElectrodeCu.setAttribute('stroke', '#c62828');
                elElectrodeCu.setAttribute('stroke-width', '3');
            } else {
                elElectrodePt.setAttribute('stroke', 'none');
                elElectrodeCu.setAttribute('stroke', 'none');
            }
            setVisible(elLabelAnode, currentStep >= 4);
            setVisible(elLabelCathode, currentStep >= 4);

            // 电子与电流方向（步骤5）
            setVisible(elArrowETop, currentStep >= 5);
            setVisible(elLabelETop, currentStep >= 5);
            setVisible(elArrowITop, currentStep >= 5);
            setVisible(elLabelITop, currentStep >= 5);
            setVisible(elArrowEBottom, currentStep >= 5);
            setVisible(elLabelEBottom, currentStep >= 5);
            setVisible(elArrowIBottom, currentStep >= 5);
            setVisible(elLabelIBottom, currentStep >= 5);
        }

        btnPrev.addEventListener('click', function() {
            if (currentStep > 0) { currentStep--; updateScene(); }
        });
        btnNext.addEventListener('click', function() {
            if (currentStep < totalSteps - 1) { currentStep++; updateScene(); }
        });

        updateScene();
    })();
    </script>
