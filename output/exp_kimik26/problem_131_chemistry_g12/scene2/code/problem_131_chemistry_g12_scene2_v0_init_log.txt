    <script>
    (function() {
        // Scene 2 状态机
        var currentStep = 0;
        var totalSteps = 8;

        var btnPrev = document.getElementById('scene2-btn-prev');
        var btnNext = document.getElementById('scene2-btn-next');
        var indicator = document.getElementById('scene2-step-indicator');

        function renderStep() {
            // 隐藏所有步骤分组
            for (var i = 0; i < totalSteps; i++) {
                var g = document.getElementById('scene2-step' + i);
                if (g) {
                    g.classList.remove('active');
                }
            }
            // 显示当前步骤
            var curG = document.getElementById('scene2-step' + currentStep);
            if (curG) {
                curG.classList.add('active');
            }

            // 更新按钮状态
            btnPrev.disabled = (currentStep === 0);
            btnNext.disabled = (currentStep === totalSteps - 1);

            // 更新指示器
            indicator.textContent = '步骤 ' + (currentStep + 1) + ' / ' + totalSteps;
        }

        btnPrev.addEventListener('click', function() {
            if (currentStep > 0) {
                currentStep--;
                renderStep();
            }
        });

        btnNext.addEventListener('click', function() {
            if (currentStep < totalSteps - 1) {
                currentStep++;
                renderStep();
            }
        });

        // 初始渲染
        renderStep();
    })();
    </script>
