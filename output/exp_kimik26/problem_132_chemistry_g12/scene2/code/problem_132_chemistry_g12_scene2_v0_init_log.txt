    <script>
    (function() {
        // Scene 2 实现
        let currentStep = 1;
        const totalSteps = 4;

        const btnPrev = document.getElementById('scene2-btn-prev');
        const btnNext = document.getElementById('scene2-btn-next');
        const stepIndicator = document.getElementById('scene2-step-indicator');

        // 获取所有需要控制的元素
        const formulaCard = document.getElementById('scene2-formula-card');

        const reactantText = document.getElementById('scene2-reactant-text');
        const co2Coef = document.getElementById('scene2-co2-coef');
        const h2oCoef = document.getElementById('scene2-h2o-coef');
        const n2Coef = document.getElementById('scene2-n2-coef');

        const cLabel = document.getElementById('scene2-c-label');
        const hLabel = document.getElementById('scene2-h-label');
        const nLabel = document.getElementById('scene2-n-label');

        const calcRow1 = document.getElementById('scene2-calc-row1');
        const calcRow2 = document.getElementById('scene2-calc-row2');
        const calcRow3 = document.getElementById('scene2-calc-row3');
        const calcResult = document.getElementById('scene2-calc-result');

        function setOpacity(el, val) {
            if (el) el.style.opacity = val;
        }

        function renderStep() {
            // 更新按钮状态
            btnPrev.disabled = (currentStep === 1);
            btnNext.disabled = (currentStep === totalSteps);
            stepIndicator.textContent = '步骤 ' + currentStep + '/' + totalSteps;

            // 步骤1：初始状态，反应物文本高亮（蓝色）
            if (currentStep === 1) {
                reactantText.setAttribute('fill', '#1565c0');
                setOpacity(co2Coef, '0');
                setOpacity(h2oCoef, '0');
                setOpacity(n2Coef, '0');
                co2Coef.setAttribute('fill', '#000000');
                h2oCoef.setAttribute('fill', '#000000');
                n2Coef.setAttribute('fill', '#000000');
                setOpacity(cLabel, '0');
                setOpacity(hLabel, '0');
                setOpacity(nLabel, '0');
                setOpacity(calcRow1, '0');
                setOpacity(calcRow2, '0');
                setOpacity(calcRow3, '0');
                if (calcResult) calcResult.setAttribute('fill', '#c62828');
                setOpacity(formulaCard, '0');
            }
            // 步骤2：产物系数出现，原子守恒标签，系数着色
            else if (currentStep === 2) {
                reactantText.setAttribute('fill', '#000000');
                setOpacity(co2Coef, '1');
                setOpacity(h2oCoef, '1');
                setOpacity(n2Coef, '1');
                co2Coef.setAttribute('fill', '#1565c0');
                h2oCoef.setAttribute('fill', '#2e7d32');
                n2Coef.setAttribute('fill', '#ef6c00');
                setOpacity(cLabel, '1');
                setOpacity(hLabel, '1');
                setOpacity(nLabel, '1');
                setOpacity(calcRow1, '0');
                setOpacity(calcRow2, '0');
                setOpacity(calcRow3, '0');
                if (calcResult) calcResult.setAttribute('fill', '#c62828');
                setOpacity(formulaCard, '0');
            }
            // 步骤3：计算式出现，结果高亮（蓝色）
            else if (currentStep === 3) {
                reactantText.setAttribute('fill', '#000000');
                setOpacity(co2Coef, '1');
                setOpacity(h2oCoef, '1');
                setOpacity(n2Coef, '1');
                co2Coef.setAttribute('fill', '#1565c0');
                h2oCoef.setAttribute('fill', '#2e7d32');
                n2Coef.setAttribute('fill', '#ef6c00');
                setOpacity(cLabel, '1');
                setOpacity(hLabel, '1');
                setOpacity(nLabel, '1');
                setOpacity(calcRow1, '1');
                setOpacity(calcRow2, '1');
                setOpacity(calcRow3, '1');
                if (calcResult) calcResult.setAttribute('fill', '#1565c0');
                setOpacity(formulaCard, '0');
            }
            // 步骤4：快捷公式卡片
            else if (currentStep === 4) {
                reactantText.setAttribute('fill', '#000000');
                setOpacity(co2Coef, '1');
                setOpacity(h2oCoef, '1');
                setOpacity(n2Coef, '1');
                co2Coef.setAttribute('fill', '#1565c0');
                h2oCoef.setAttribute('fill', '#2e7d32');
                n2Coef.setAttribute('fill', '#ef6c00');
                setOpacity(cLabel, '1');
                setOpacity(hLabel, '1');
                setOpacity(nLabel, '1');
                setOpacity(calcRow1, '1');
                setOpacity(calcRow2, '1');
                setOpacity(calcRow3, '1');
                if (calcResult) calcResult.setAttribute('fill', '#1565c0');
                setOpacity(formulaCard, '1');
            }
        }

        btnPrev.addEventListener('click', function() {
            if (currentStep > 1) {
                currentStep--;
                renderStep();
            }
        });

        btnNext.addEventListener('click', function() {
            if (currentStep < totalSteps) {
                currentStep++;
                renderStep();
            }
        });

        // 初始渲染
        renderStep();
    })();
    </script>
