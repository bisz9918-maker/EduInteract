    <script>
    (function() {
        // Scene 2 implementation
        const STATE_COUNT = 5;
        let currentStep = 0;

        // Groups visible in each step (in addition to parents-group which is always visible)
        const stepGroups = [
            [],                                      // step 0
            ['scene2-step1-group'],                  // step 1
            ['scene2-step1-group', 'scene2-step2-group'], // step 2
            ['scene2-step3-group'],                  // step 3
            ['scene2-step3-group', 'scene2-step4-group']  // step 4
        ];

        const allGroups = ['scene2-step1-group', 'scene2-step2-group', 'scene2-step3-group', 'scene2-step4-group'];

        const btnPrev = document.getElementById('scene2-btn-prev');
        const btnNext = document.getElementById('scene2-btn-next');
        const indicator = document.getElementById('scene2-step-indicator');
        const fatherX = document.getElementById('scene2-father-x');
        const fatherY = document.getElementById('scene2-father-y');

        function showGroup(id) {
            var el = document.getElementById(id);
            if (!el) return;
            el.classList.remove('scene2-hidden');
        }

        function hideGroup(id) {
            var el = document.getElementById(id);
            if (!el) return;
            el.classList.add('scene2-hidden');
        }

        function updateStep(step) {
            currentStep = step;

            // Hide all step groups
            allGroups.forEach(function(id) {
                hideGroup(id);
            });

            // Show groups for current step
            stepGroups[step].forEach(function(id) {
                showGroup(id);
            });

            // Chromosome highlight in abnormal steps
            if (step === 3 || step === 4) {
                fatherX.classList.add('highlight-chromosome');
                fatherY.classList.add('highlight-chromosome');
            } else {
                fatherX.classList.remove('highlight-chromosome');
                fatherY.classList.remove('highlight-chromosome');
            }

            // Button states
            btnPrev.disabled = (step === 0);
            btnNext.disabled = (step === STATE_COUNT - 1);
            indicator.textContent = '步骤 ' + (step + 1) + '/' + STATE_COUNT;
        }

        btnPrev.addEventListener('click', function() {
            if (currentStep > 0) updateStep(currentStep - 1);
        });

        btnNext.addEventListener('click', function() {
            if (currentStep < STATE_COUNT - 1) updateStep(currentStep + 1);
        });

        // Initialize
        updateStep(0);
    })();
    </script>
