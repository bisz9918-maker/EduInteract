    <script>
    (function() {
        var currentStep = 0;
        var totalSteps = 4;
        var titles = [
            "步骤0：正常男性减数分裂（对照）",
            "步骤1：父亲减数分裂Ⅰ不分离",
            "步骤2：Ⅱ-5（X^bO）的形成",
            "步骤3：另一种异常配子的受精结果"
        ];

        var btnPrev = document.getElementById("scene2-btn-prev");
        var btnNext = document.getElementById("scene2-btn-next");
        var stepIndicator = document.getElementById("scene2-step-indicator");
        var titleEl = document.getElementById("scene2-title");

        function updateScene() {
            for (var i = 0; i < totalSteps; i++) {
                var group = document.getElementById("scene2-step" + i + "-group");
                if (group) {
                    group.style.display = "none";
                }
            }
            var currentGroup = document.getElementById("scene2-step" + currentStep + "-group");
            if (currentGroup) {
                currentGroup.style.display = "block";
            }

            titleEl.innerHTML = titles[currentStep];
            btnPrev.disabled = (currentStep === 0);
            btnNext.disabled = (currentStep === totalSteps - 1);
            stepIndicator.textContent = "步骤 " + (currentStep + 1) + "/" + totalSteps;
        }

        btnPrev.addEventListener("click", function() {
            if (currentStep > 0) {
                currentStep--;
                updateScene();
            }
        });

        btnNext.addEventListener("click", function() {
            if (currentStep < totalSteps - 1) {
                currentStep++;
                updateScene();
            }
        });

        updateScene();
    })();
    </script>
