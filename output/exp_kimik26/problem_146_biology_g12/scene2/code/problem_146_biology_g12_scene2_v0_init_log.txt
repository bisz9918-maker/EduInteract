    <script>
    (function() {
        var currentStep = 0;
        var totalSteps = 5;

        var btnPrev = document.getElementById('scene2-btn-prev');
        var btnNext = document.getElementById('scene2-btn-next');
        var stepLabel = document.getElementById('scene2-step-label');
        var answerBox = document.getElementById('scene2-answer-box');
        var svg = document.getElementById('scene2-svg');

        var groups = [];
        for (var i = 0; i < totalSteps; i++) {
            groups.push(document.getElementById('scene2-group-step' + i));
        }

        function render() {
            stepLabel.textContent = '步骤 ' + (currentStep + 1) + '/' + totalSteps;
            btnPrev.disabled = (currentStep === 0);
            btnNext.disabled = (currentStep === totalSteps - 1);

            for (var i = 0; i < totalSteps; i++) {
                if (groups[i]) {
                    if (i <= currentStep) {
                        groups[i].style.display = 'block';
                    } else {
                        groups[i].style.display = 'none';
                    }
                }
            }

            // 答案框
            if (currentStep >= 4) {
                answerBox.classList.add('visible');
            } else {
                answerBox.classList.remove('visible');
            }

            // 步骤4：高亮目标格子，淡化非目标
            var allCellTexts = svg.querySelectorAll('[class*="s2-cell-"]');
            allCellTexts.forEach(function(el) {
                el.style.opacity = '1';
                el.style.fill = '';
                if (el.classList.contains('s2-pheno-r')) el.style.fill = '#1565c0';
                if (el.classList.contains('s2-pheno-w')) el.style.fill = '#c62828';
                if (el.classList.contains('s2-lethal')) el.style.fill = '#666666';
                el.style.fontWeight = '';
            });

            if (currentStep >= 4) {
                allCellTexts.forEach(function(el) {
                    if (el.classList.contains('s2-cell-11') || el.classList.contains('s2-cell-31')) {
                        el.style.fill = '#2e7d32';
                        el.style.fontWeight = 'bold';
                    } else {
                        el.style.opacity = '0.3';
                    }
                });
            }
        }

        btnPrev.addEventListener('click', function() {
            if (currentStep > 0) {
                currentStep--;
                render();
            }
        });

        btnNext.addEventListener('click', function() {
            if (currentStep < totalSteps - 1) {
                currentStep++;
                render();
            }
        });

        render();
    })();
    </script>
