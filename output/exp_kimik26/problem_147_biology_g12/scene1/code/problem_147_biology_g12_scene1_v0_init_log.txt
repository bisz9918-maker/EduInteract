    <script>
    (function() {
        // Scene 1 实现
        var currentStep = 0;
        var totalSteps = 5;
        var stepTitles = [
            "原始双链DNA",
            "95°C 变性 — 氢键断裂，双链解开",
            "55°C 复性 — 引物与模板互补配对",
            "72°C 延伸 — DNA聚合酶合成新链",
            "循环一次产物 — 半保留复制"
        ];

        var stepTitleEl = document.getElementById("scene1-step-title");
        var stepIndicatorEl = document.getElementById("scene1-step-indicator");
        var btnPrev = document.getElementById("scene1-btn-prev");
        var btnNext = document.getElementById("scene1-btn-next");

        function renderStep() {
            // 隐藏所有步骤组
            for (var i = 0; i < totalSteps; i++) {
                var group = document.getElementById("scene1-step" + i + "-group");
                if (group) {
                    group.style.display = "none";
                }
            }
            // 显示当前步骤组
            var currentGroup = document.getElementById("scene1-step" + currentStep + "-group");
            if (currentGroup) {
                currentGroup.style.display = "block";
            }
            // 更新标题
            stepTitleEl.textContent = stepTitles[currentStep];
            // 更新指示器
            stepIndicatorEl.textContent = "步骤 " + (currentStep + 1) + "/" + totalSteps;
            // 更新按钮状态
            btnPrev.disabled = (currentStep === 0);
            btnNext.disabled = (currentStep === totalSteps - 1);
            if (currentStep === totalSteps - 1) {
                btnNext.textContent = "完成";
            } else {
                btnNext.textContent = "下一步 ▶";
            }
        }

        btnPrev.addEventListener("click", function() {
            if (currentStep > 0) {
                currentStep--;
                renderStep();
            }
        });

        btnNext.addEventListener("click", function() {
            if (currentStep < totalSteps - 1) {
                currentStep++;
                renderStep();
            }
        });

        // 初始渲染
        renderStep();
    })();
    </script>
