    <script>
    (function() {
        // Scene 2 实现
        var currentN = 1;
        var svg = document.getElementById('scene2-svg');
        var dnaGroup = document.getElementById('scene2-dna-group');
        var pieBlue = document.getElementById('scene2-pie-blue');
        var pieRed = document.getElementById('scene2-pie-red');
        var pieText = document.getElementById('scene2-pie-text');
        var featureText = document.getElementById('scene2-feature-text');
        var formulaText = document.getElementById('scene2-formula-text');
        var sliderThumb = document.getElementById('scene2-slider-thumb');
        var sliderProgress = document.getElementById('scene2-slider-progress');
        var sliderValue = document.getElementById('scene2-slider-value');
        var statValues = [
            document.getElementById('scene2-stat-value-0'),
            document.getElementById('scene2-stat-value-1'),
            document.getElementById('scene2-stat-value-2'),
            document.getElementById('scene2-stat-value-3')
        ];

        var sliderMinX = 150;
        var sliderMaxX = 650;
        var sliderRange = sliderMaxX - sliderMinX;

        // 工具函数：创建SVG元素
        function createSVGElement(tag, attrs) {
            var el = document.createElementNS('http://www.w3.org/2000/svg', tag);
            for (var key in attrs) {
                el.setAttribute(key, attrs[key]);
            }
            return el;
        }

        // 绘制DNA图标
        // type: 'mixed' (蓝红) 或 'new' (红红)
        // x, y: 左上角坐标
        function drawDNAIcon(group, x, y, type) {
            var topColor = type === 'mixed' ? '#1565c0' : '#c62828';
            var bottomColor = '#c62828';
            // 上线
            group.appendChild(createSVGElement('line', {
                x1: x, y1: y, x2: x + 50, y2: y,
                stroke: topColor, 'stroke-width': 3
            }));
            // 下线
            group.appendChild(createSVGElement('line', {
                x1: x, y1: y + 8, x2: x + 50, y2: y + 8,
                stroke: bottomColor, 'stroke-width': 3
            }));
        }

        // 更新DNA示意图
        function updateDNAGroup(n) {
            // 清空
            while (dnaGroup.firstChild) {
                dnaGroup.removeChild(dnaGroup.firstChild);
            }

            if (n <= 4) {
                // 网格模式
                var total = Math.pow(2, n);
                var cols = Math.min(total, 4);
                var rows = Math.ceil(total / cols);
                var startX = 40;
                var startY = 50;
                var gapX = 60;
                var gapY = 24;

                for (var i = 0; i < total; i++) {
                    var col = i % cols;
                    var row = Math.floor(i / cols);
                    var x = startX + col * gapX;
                    var y = startY + row * gapY;
                    var type = i < 2 ? 'mixed' : 'new';
                    drawDNAIcon(dnaGroup, x, y, type);
                }
            } else {
                // 缩略模式
                // 蓝红图标 + 文字 "2个"
                drawDNAIcon(dnaGroup, 40, 50, 'mixed');
                dnaGroup.appendChild(createSVGElement('text', {
                    class: 'scene2-label scene2-medium scene2-left',
                    x: 100, y: 58
                })).textContent = '2个';

                // 红红图标 + 文字 "2ⁿ−2个"
                drawDNAIcon(dnaGroup, 40, 90, 'new');
                dnaGroup.appendChild(createSVGElement('text', {
                    class: 'scene2-label scene2-medium scene2-left',
                    x: 100, y: 98
                })).textContent = '2ⁿ−2个';

                // 总计文字
                dnaGroup.appendChild(createSVGElement('text', {
                    class: 'scene2-label scene2-medium scene2-left',
                    x: 40, y: 140
                })).textContent = '总计 2ⁿ 个DNA';
            }
        }

        // 更新饼图
        function updatePie(n) {
            var cx = 400, cy = 230, r = 40;
            var p = 1 / Math.pow(2, n); // 蓝色比例（不含³²P）
            var angle = p * 2 * Math.PI;

            // 蓝色扇形：从12点方向开始（角度 -PI/2）
            var x1 = cx + r * Math.sin(angle);
            var y1 = cy - r * Math.cos(angle);
            var largeArc = p > 0.5 ? 1 : 0;

            var dBlue = 'M ' + cx + ' ' + cy + ' L ' + cx + ' ' + (cy - r) +
                        ' A ' + r + ' ' + r + ' 0 ' + largeArc + ' 1 ' + x1 + ' ' + y1 + ' Z';
            pieBlue.setAttribute('d', dBlue);

            // 红色扇形：剩余部分
            var dRed = 'M ' + cx + ' ' + cy + ' L ' + x1 + ' ' + y1 +
                       ' A ' + r + ' ' + r + ' 0 ' + (1 - largeArc) + ' 1 ' + cx + ' ' + (cy - r) + ' Z';
            pieRed.setAttribute('d', dRed);

            // 饼图下方文字
            var fraction = '1/' + Math.pow(2, n);
            pieText.textContent = fraction;
        }

        // 更新统计数值
        function updateStats(n) {
            var dnaTotal = Math.pow(2, n);
            var strandTotal = Math.pow(2, n + 1);
            var radioactive = strandTotal - 2;
            var nonRadioactive = 2;

            statValues[0].textContent = dnaTotal;
            statValues[1].textContent = strandTotal;
            statValues[2].textContent = radioactive;
            statValues[3].textContent = nonRadioactive;
        }

        // 更新公式区
        function updateFormula(n) {
            if (n === 1) {
                featureText.style.opacity = '1';
            } else {
                featureText.style.opacity = '0';
            }
            var result = '1/' + Math.pow(2, n);
            formulaText.textContent = '不含³²P单链比例 = 2/2ⁿ⁺¹ = 1/2ⁿ = ' + result;
        }

        // 更新滑块位置
        function updateSlider(n) {
            var x = sliderMinX + (n - 1) * (sliderRange / 9);
            sliderThumb.setAttribute('cx', x);
            sliderProgress.setAttribute('width', x - sliderMinX);
            sliderValue.textContent = 'n = ' + n;
        }

        // 主更新函数
        function updateScene(n) {
            currentN = n;
            updateDNAGroup(n);
            updatePie(n);
            updateStats(n);
            updateFormula(n);
            updateSlider(n);
        }

        // 滑块交互
        var isDragging = false;

        function getNFromX(x) {
            var clamped = Math.max(sliderMinX, Math.min(sliderMaxX, x));
            var ratio = (clamped - sliderMinX) / sliderRange;
            var n = 1 + Math.round(ratio * 9);
            return Math.max(1, Math.min(10, n));
        }

        function handleStart(e) {
            isDragging = true;
            e.preventDefault();
        }

        function handleMove(e) {
            if (!isDragging) return;
            var clientX = e.touches ? e.touches[0].clientX : e.clientX;
            var rect = svg.getBoundingClientRect();
            var x = clientX - rect.left;
            var n = getNFromX(x);
            if (n !== currentN) {
                updateScene(n);
            }
        }

        function handleEnd() {
            isDragging = false;
        }

        sliderThumb.addEventListener('mousedown', handleStart);
        svg.addEventListener('mousemove', handleMove);
        svg.addEventListener('mouseup', handleEnd);
        svg.addEventListener('mouseleave', handleEnd);

        sliderThumb.addEventListener('touchstart', handleStart, {passive: false});
        svg.addEventListener('touchmove', handleMove, {passive: false});
        svg.addEventListener('touchend', handleEnd);

        // 点击轨道直接跳转
        svg.addEventListener('click', function(e) {
            if (e.target === sliderThumb) return;
            var rect = svg.getBoundingClientRect();
            var x = e.clientX - rect.left;
            var y = e.clientY - rect.top;
            // 只响应滑块区域附近的点击
            if (y >= 420 && y <= 470) {
                var n = getNFromX(x);
                updateScene(n);
            }
        });

        // 初始化
        updateScene(1);
    })();
    </script>
