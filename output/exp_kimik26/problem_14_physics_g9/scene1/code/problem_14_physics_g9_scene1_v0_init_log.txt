    <script>
    (function() {
        // Scene 1 实现
        let currentStep = 0;
        const totalSteps = 7;

        // 元素引用
        const btnPrev = document.getElementById('scene1-btn-prev');
        const btnNext = document.getElementById('scene1-btn-next');
        const stepIndicator = document.getElementById('scene1-step-indicator');
        const derivationBox = document.getElementById('scene1-derivation-box');
        const optionTable = document.getElementById('scene1-option-table');

        // SVG元素
        const ropeUp = document.getElementById('scene1-rope-up');
        const ropeMid = document.getElementById('scene1-rope-mid');
        const ropeDown = document.getElementById('scene1-rope-down');
        const arrowF = document.getElementById('scene1-arrow-f');
        const arrowFriction = document.getElementById('scene1-arrow-friction');
        const arrow3F = document.getElementById('scene1-arrow-3f');
        const arrowVRope = document.getElementById('scene1-arrow-v-rope');
        const labelN = document.getElementById('scene1-label-n');
        const label3F = document.getElementById('scene1-label-3f');
        const labelVRope = document.getElementById('scene1-label-v-rope');
        const labelFriction = document.getElementById('scene1-label-friction');

        // 推导文本内容
        const derivationContent = [
            // 步骤0：题目条件
            {
                title: '已知条件',
                html: '<div>G = 50N，F = 5N</div><div>v<sub>物</sub> = 1m/s，η = 80%</div><div>t = 10s</div>'
            },
            // 步骤1：确定n
            {
                title: '承担拉力的绳子段数',
                html: '<div style="font-size:20px; color:#1565c0; font-weight:bold; margin-top:10px;">n = 3</div>'
            },
            // 步骤2：求摩擦力
            {
                title: '求摩擦力 f',
                html: '<div class="formula">η = W<sub>有</sub> / W<sub>总</sub></div>' +
                      '<div class="formula">= (f · s<sub>物</sub>) / (F · s<sub>绳</sub>)</div>' +
                      '<div class="formula">= f / (n · F)</div>' +
                      '<div class="formula">f = η · n · F</div>' +
                      '<div class="formula">= 0.8 × 3 × 5 = <span class="highlight-green">12N</span></div>'
            },
            // 步骤3：求功率
            {
                title: '求拉力功率 P',
                html: '<div class="formula">v<sub>绳</sub> = n · v<sub>物</sub> = <span class="highlight-blue">3m/s</span></div>' +
                      '<div class="formula">P = F · v<sub>绳</sub> = 5 × 3 = <span class="highlight-blue">15W</span></div>' +
                      '<div style="margin-top:8px; color:#c62828;">选项C错误：错把 v<sub>物</sub> 当作 v<sub>绳</sub></div>'
            },
            // 步骤4：求有用功
            {
                title: '求有用功 W<sub>有</sub>',
                html: '<div class="formula">W<sub>有</sub> = f · s<sub>物</sub> = f · v<sub>物</sub> · t</div>' +
                      '<div class="formula">= 12 × 1 × 10 = <span class="highlight-green">120J</span></div>' +
                      '<div style="margin-top:8px; color:#c62828;">选项D错误：误用重力G算得500J</div>'
            },
            // 步骤5：合力分析
            {
                title: '合力分析',
                html: '<div>物体做匀速直线运动</div>' +
                      '<div style="font-size:18px; color:#2e7d32; font-weight:bold; margin:8px 0;">F<sub>合</sub> = 0</div>' +
                      '<div class="formula">向右：3F = 15N</div>' +
                      '<div class="formula">向左：f + 额外阻力</div>' +
                      '<div style="margin-top:8px; color:#c62828;">选项A错误：合力为0，不是5N</div>'
            },
            // 步骤6：选项总结
            {
                title: '正确答案',
                html: '<div style="font-size:24px; color:#2e7d32; font-weight:bold; margin-top:10px;">B</div>'
            }
        ];

        // 选项表格内容
        const optionTableContent = [
            null, // 步骤0无表格
            null, // 步骤1无表格
            { option: 'B', result: 'f = 12N', correct: true },
            { option: 'C', result: 'P = 15W', correct: false },
            { option: 'D', result: 'W<sub>有</sub> = 120J', correct: false },
            { option: 'A', result: 'F<sub>合</sub> = 0', correct: false },
            null // 步骤6用完整表格
        ];

        function show(el) {
            if (el) { el.style.display = ''; el.style.opacity = '1'; }
        }
        function hide(el) {
            if (el) { el.style.display = 'none'; }
        }

        function renderStep() {
            // 更新按钮状态
            btnPrev.disabled = currentStep === 0;
            btnNext.disabled = currentStep === totalSteps - 1;
            stepIndicator.textContent = '步骤 ' + (currentStep + 1) + ' / ' + totalSteps;

            // 更新推导区
            const content = derivationContent[currentStep];
            derivationBox.innerHTML = '<div class="title">' + content.title + '</div>' + content.html;

            // 调整推导区位置（步骤6缩小）
            if (currentStep === 6) {
                derivationBox.style.top = '60px';
                derivationBox.style.height = '80px';
            } else {
                derivationBox.style.top = '60px';
                derivationBox.style.height = '220px';
            }

            // 步骤0：初始状态
            if (currentStep === 0) {
                // 绳子黑色
                [ropeUp, ropeMid, ropeDown].forEach(r => {
                    r.setAttribute('stroke', '#000');
                    r.setAttribute('stroke-width', '1.5');
                    r.classList.remove('rope-blink');
                });
                hide(arrowF);
                hide(arrowFriction);
                hide(arrow3F);
                hide(arrowVRope);
                hide(labelN);
                hide(label3F);
                hide(labelVRope);
                hide(labelFriction);
                hide(optionTable);
            }

            // 步骤1：确定n
            if (currentStep >= 1) {
                show(labelN);
                // 绳子变蓝色
                [ropeUp, ropeMid, ropeDown].forEach(r => {
                    r.setAttribute('stroke', '#1565c0');
                    r.setAttribute('stroke-width', '3');
                });
                // 步骤1时添加闪烁动画
                if (currentStep === 1) {
                    [ropeUp, ropeMid, ropeDown].forEach(r => {
                        r.classList.add('rope-blink');
                        // 900ms后移除动画类
                        setTimeout(() => r.classList.remove('rope-blink'), 900);
                    });
                }
            }

            // 步骤2：摩擦力
            if (currentStep >= 2) {
                show(arrowFriction);
                show(labelFriction);
            } else {
                hide(arrowFriction);
                hide(labelFriction);
            }

            // 步骤3：v_绳
            if (currentStep >= 3) {
                show(arrowVRope);
                show(labelVRope);
            } else {
                hide(arrowVRope);
                hide(labelVRope);
            }

            // 步骤5：3F
            if (currentStep >= 5) {
                show(arrow3F);
                show(label3F);
            } else {
                hide(arrow3F);
                hide(label3F);
            }

            // 拉力F箭头始终显示（步骤0就显示）
            show(arrowF);

            // 选项表格
            if (currentStep >= 2 && currentStep <= 5) {
                const opt = optionTableContent[currentStep];
                if (opt) {
                    optionTable.style.display = 'block';
                    optionTable.style.top = '350px';
                    optionTable.innerHTML =
                        '<table><tr><th>选项</th><th>计算结果</th><th>正误</th></tr>' +
                        '<tr class="' + (opt.correct ? 'correct' : 'wrong') + '">' +
                        '<td>' + opt.option + '</td>' +
                        '<td>' + opt.result + '</td>' +
                        '<td>' + (opt.correct ? '✓ 正确' : '✗ 错误') + '</td>' +
                        '</tr></table>';
                }
            } else if (currentStep === 6) {
                // 完整选项表格
                optionTable.style.display = 'block';
                optionTable.style.top = '160px';
                optionTable.innerHTML =
                    '<table>' +
                    '<tr><th>选项</th><th>计算结果</th><th>正误</th></tr>' +
                    '<tr class="wrong"><td>A</td><td>F<sub>合</sub> = 0</td><td>✗</td></tr>' +
                    '<tr class="correct"><td>B</td><td>f = 12N</td><td>✓</td></tr>' +
                    '<tr class="wrong"><td>C</td><td>P = 15W</td><td>✗</td></tr>' +
                    '<tr class="wrong"><td>D</td><td>W<sub>有</sub> = 120J</td><td>✗</td></tr>' +
                    '</table>';
            } else {
                hide(optionTable);
            }
        }

        // 按钮事件
        btnPrev.addEventListener('click', function() {
            if (currentStep > 0) {
                currentStep--;
                renderStep();
            }
        });
        btnNext.addEventListener('click', function() {
            if (currentStep < totalSteps - 1) {
                currentStep++;
                renderStep();
            }
        });

        // 初始渲染
        renderStep();
    })();
    </script>
</body>
