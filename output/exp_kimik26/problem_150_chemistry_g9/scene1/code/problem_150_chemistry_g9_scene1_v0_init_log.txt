    <script>
    (function() {
        // Scene 1 实现
        let currentStep = 0;
        const totalSteps = 4;

        const btnPrev = document.getElementById('scene1-btn-prev');
        const btnNext = document.getElementById('scene1-btn-next');
        const dots = document.querySelectorAll('.scene1-dot');

        // 各步骤需要显示的元素 ID 列表
        const stepElements = {
            0: [], // 初始状态只显示装置基础元素
            1: [
                'scene1-label-fe', 'scene1-label-heat',
                'scene1-label-co', 'scene1-label-co2',
                'scene1-co-in-arrow', 'scene1-co2-out-arrow',
                'scene1-eq-co'
            ],
            2: [
                'scene1-label-fe', 'scene1-label-heat',
                'scene1-label-h2', 'scene1-label-h2o',
                'scene1-h2-in-arrow', 'scene1-h2o-out-arrow',
                'scene1-eq-h2'
            ],
            3: [
                'scene1-label-fe', 'scene1-label-heat',
                'scene1-compare'
            ]
        };

        function show(el) {
            if (!el) return;
            el.classList.remove('scene1-hidden');
        }
        function hide(el) {
            if (!el) return;
            el.classList.add('scene1-hidden');
        }

        function renderStep() {
            // 先隐藏所有步骤相关元素
            const allEls = document.querySelectorAll('.scene1-step-el');
            allEls.forEach(el => hide(el));

            // 显示当前步骤需要的元素
            const ids = stepElements[currentStep] || [];
            ids.forEach(id => {
                const el = document.getElementById(id);
                show(el);
            });

            // 更新按钮状态
            btnPrev.disabled = currentStep === 0;
            if (currentStep === totalSteps - 1) {
                btnNext.textContent = '完成';
                btnNext.disabled = true;
            } else {
                btnNext.textContent = '下一步 ▶';
                btnNext.disabled = false;
            }

            // 更新步骤指示器
            dots.forEach((dot, idx) => {
                dot.classList.toggle('active', idx === currentStep);
            });
        }

        btnPrev.addEventListener('click', function() {
            if (currentStep > 0) {
                currentStep--;
                renderStep();
            }
        });

        btnNext.addEventListener('click', function() {
            if (currentStep < totalSteps - 1) {
                currentStep++;
                renderStep();
            }
        });

        // 初始化
        renderStep();
    })();
    </script>
