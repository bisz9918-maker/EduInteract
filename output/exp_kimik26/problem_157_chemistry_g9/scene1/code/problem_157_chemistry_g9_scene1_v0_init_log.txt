    <script>
    (function() {
        // Scene 1 实现
        var currentStep = 0;
        var totalSteps = 4;

        // DOM 元素引用
        var btnPrev = document.getElementById('scene1-btn-prev');
        var btnNext = document.getElementById('scene1-btn-next');
        var stepIndicator = document.getElementById('scene1-step-indicator');

        var pointA = document.getElementById('scene1-point-a');
        var pointB = document.getElementById('scene1-point-b');
        var pointM = document.getElementById('scene1-point-m');

        var projAV = document.getElementById('scene1-proj-a-v');
        var projAH = document.getElementById('scene1-proj-a-h');
        var projBV = document.getElementById('scene1-proj-b-v');
        var projBH = document.getElementById('scene1-proj-b-h');
        var projMV = document.getElementById('scene1-proj-m-v');
        var projMH = document.getElementById('scene1-proj-m-h');

        var infoGroups = [];
        for (var i = 0; i <= totalSteps; i++) {
            infoGroups.push(document.getElementById('scene1-info-step' + i));
        }

        // 显示/隐藏工具函数
        function show(el) {
            if (!el) return;
            el.style.display = 'block';
            // 强制重排以确保过渡生效
            void el.offsetWidth;
            el.style.opacity = '1';
        }

        function hide(el) {
            if (!el) return;
            el.style.opacity = '0';
            // 等待过渡完成后隐藏
            setTimeout(function() {
                if (el.style.opacity === '0') {
                    el.style.display = 'none';
                }
            }, 300);
        }

        // 设置投影线状态
        function setProjLine(el, color, visible) {
            if (!el) return;
            if (visible) {
                el.setAttribute('stroke', color);
                el.style.opacity = '1';
            } else {
                el.setAttribute('stroke', 'none');
                el.style.opacity = '0';
            }
        }

        // 设置点状态
        function setPoint(el, color, radius) {
            if (!el) return;
            el.setAttribute('fill', color);
            el.setAttribute('r', String(radius));
        }

        // 渲染当前步骤
        function render() {
            // 更新按钮状态
            btnPrev.disabled = currentStep === 0;
            btnNext.disabled = currentStep === totalSteps;
            stepIndicator.textContent = '步骤 ' + currentStep + '/' + totalSteps;

            // 更新信息面板
            for (var i = 0; i <= totalSteps; i++) {
                if (i === currentStep) {
                    show(infoGroups[i]);
                } else {
                    hide(infoGroups[i]);
                }
            }

            // 根据步骤更新点和投影线
            if (currentStep === 0) {
                // 步骤0：所有点为灰色，无投影
                setPoint(pointA, '#999999', 3);
                setPoint(pointB, '#999999', 3);
                setPoint(pointM, '#999999', 3);
                setProjLine(projAV, '#1565c0', false);
                setProjLine(projAH, '#1565c0', false);
                setProjLine(projBV, '#c62828', false);
                setProjLine(projBH, '#c62828', false);
                setProjLine(projMV, '#c62828', false);
                setProjLine(projMH, '#c62828', false);
            } else if (currentStep === 1) {
                // 步骤1：A点蓝色，投影蓝色
                setPoint(pointA, '#1565c0', 4);
                setPoint(pointB, '#999999', 3);
                setPoint(pointM, '#999999', 3);
                setProjLine(projAV, '#1565c0', true);
                setProjLine(projAH, '#1565c0', true);
                setProjLine(projBV, '#c62828', false);
                setProjLine(projBH, '#c62828', false);
                setProjLine(projMV, '#c62828', false);
                setProjLine(projMH, '#c62828', false);
            } else if (currentStep === 2) {
                // 步骤2：B点红色，投影红色
                setPoint(pointA, '#999999', 3);
                setPoint(pointB, '#c62828', 4);
                setPoint(pointM, '#999999', 3);
                setProjLine(projAV, '#1565c0', false);
                setProjLine(projAH, '#1565c0', false);
                setProjLine(projBV, '#c62828', true);
                setProjLine(projBH, '#c62828', true);
                setProjLine(projMV, '#c62828', false);
                setProjLine(projMH, '#c62828', false);
            } else if (currentStep === 3) {
                // 步骤3：M点红色，投影红色
                setPoint(pointA, '#999999', 3);
                setPoint(pointB, '#999999', 3);
                setPoint(pointM, '#c62828', 4);
                setProjLine(projAV, '#1565c0', false);
                setProjLine(projAH, '#1565c0', false);
                setProjLine(projBV, '#c62828', false);
                setProjLine(projBH, '#c62828', false);
                setProjLine(projMV, '#c62828', true);
                setProjLine(projMH, '#c62828', true);
            } else if (currentStep === 4) {
                // 步骤4：A点蓝色，投影蓝色
                setPoint(pointA, '#1565c0', 4);
                setPoint(pointB, '#999999', 3);
                setPoint(pointM, '#999999', 3);
                setProjLine(projAV, '#1565c0', true);
                setProjLine(projAH, '#1565c0', true);
                setProjLine(projBV, '#c62828', false);
                setProjLine(projBH, '#c62828', false);
                setProjLine(projMV, '#c62828', false);
                setProjLine(projMH, '#c62828', false);
            }
        }

        // 事件绑定
        btnPrev.addEventListener('click', function() {
            if (currentStep > 0) {
                currentStep--;
                render();
            }
        });

        btnNext.addEventListener('click', function() {
            if (currentStep < totalSteps) {
                currentStep++;
                render();
            }
        });

        // 初始渲染
        render();
    })();
    </script>
