    <script>
    (function() {
        // Scene 2 实现
        let currentStep = 0;
        const totalSteps = 6;

        // 面板内容数据
        const panelContents = [
            // 步骤0：初始状态
            `<h3>初始状态</h3>
            <p><span style="color:#1565c0;font-weight:bold;">A点</span>：饱和溶液（t₂℃, S₂ g/100g水）</p>
            <p><span style="color:#c62828;font-weight:bold;">B点</span>：不饱和溶液（t₃℃, S₁ g/100g水）</p>
            <p><span style="color:#c62828;font-weight:bold;">M点</span>：不饱和溶液（t₂℃, S₁ g/100g水）</p>
            <p style="margin-top:10px;color:#666;">A在曲线上 = 饱和；B、M在曲线下方 = 不饱和</p>`,
            // 步骤1：恒温加水
            `<h3>步骤1：恒温加水</h3>
            <p>保持温度不变，A、B各加20g水</p>
            <p>溶剂：100g → 120g</p>
            <p>A状态：<span style="color:#c62828;">不饱和</span></p>
            <p class="formula">wₐ = S₂/(120+S₂)</p>
            <p class="formula">wₑ = S₁/(120+S₁)</p>
            <p class="conclusion">结论：前者大</p>`,
            // 步骤2：A点降温
            `<h3>步骤2：A点降温</h3>
            <p>A点从 t₂℃ 降温至 t₁℃</p>
            <p>沿溶解度曲线移动</p>
            <p>到达新位置：饱和溶液（t₁℃, S₁）</p>
            <p style="color:#888;">♦ 晶体析出</p>
            <p class="formula">w' = S₁/(100+S₁)</p>
            <p class="conclusion">结论：减小</p>`,
            // 步骤3：M降温至饱和
            `<h3>步骤3：M降温至饱和</h3>
            <p>M点不饱和溶液变为饱和溶液</p>
            <p>方法①：<span style="color:#c62828;font-weight:bold;">降温</span></p>
            <p>从 t₂℃ 降温至 <span class="conclusion">t₁℃</span></p>
            <p>水平移动至曲线上的对应点</p>
            <p class="conclusion">恰好饱和</p>`,
            // 步骤4：M增加溶质
            `<h3>步骤4：M增加溶质</h3>
            <p>方法②：<span style="color:#c62828;font-weight:bold;">增加溶质</span></p>
            <p>保持温度 t₂℃ 不变</p>
            <p>垂直向上移动至A点</p>
            <p class="formula">需加溶质 = (S₂-S₁) g</p>
            <p class="conclusion">恰好饱和</p>`,
            // 步骤5：M蒸发溶剂
            `<h3>步骤5：M蒸发溶剂</h3>
            <p>方法③：<span style="color:#c62828;font-weight:bold;">蒸发溶剂</span></p>
            <p>保持温度 t₂℃ 不变</p>
            <p>垂直向上移动至A点</p>
            <p class="formula">需蒸发水 = 100(S₂-S₁)/S₂ g</p>
            <p class="conclusion">恰好饱和</p>`
        ];

        // 获取元素
        const btnPrev = document.getElementById('scene2-btn-prev');
        const btnNext = document.getElementById('scene2-btn-next');
        const indicator = document.getElementById('scene2-step-indicator');
        const panelContent = document.getElementById('scene2-panel-content');

        // 步骤元素组
        const stepGroups = {
            2: document.getElementById('scene2-step2-group'),
            3: document.getElementById('scene2-step3-group'),
            4: document.getElementById('scene2-step4-group'),
            5: document.getElementById('scene2-step5-group')
        };

        // 工具函数
        function show(el) {
            if (!el) return;
            el.style.display = 'block';
            el.style.opacity = '1';
        }
        function hide(el) {
            if (!el) return;
            el.style.display = 'none';
            el.style.opacity = '0';
        }

        // 动画函数：A点沿曲线移动
        function animateAPoint(fromX, fromY, toX, toY, duration, callback) {
            const movingA = document.getElementById('scene2-step2-moving-a');
            const startTime = performance.now();

            function step(now) {
                const elapsed = now - startTime;
                const t = Math.min(elapsed / duration, 1);
                // 使用 ease-in-out
                const ease = t < 0.5 ? 2*t*t : -1+(4-2*t)*t;

                const cx = fromX + (toX - fromX) * ease;
                const cy = fromY + (toY - fromY) * ease;
                movingA.setAttribute('cx', cx);
                movingA.setAttribute('cy', cy);

                if (t < 1) {
                    requestAnimationFrame(step);
                } else if (callback) {
                    callback();
                }
            }
            requestAnimationFrame(step);
        }

        // 切换到指定步骤
        function goToStep(step) {
            if (step < 0 || step >= totalSteps) return;
            currentStep = step;

            // 更新指示器
            indicator.textContent = (step + 1) + '/' + totalSteps;
            btnPrev.disabled = (step === 0);
            btnNext.disabled = (step === totalSteps - 1);

            // 更新面板
            panelContent.innerHTML = panelContents[step];

            // 隐藏所有步骤特效
            Object.values(stepGroups).forEach(hide);

            // 重置A点位置
            const movingA = document.getElementById('scene2-step2-moving-a');
            if (movingA) {
                movingA.setAttribute('cx', '300');
                movingA.setAttribute('cy', '240');
            }

            // 根据步骤显示对应元素
            if (step === 2) {
                // 步骤2：A点降温动画
                show(stepGroups[2]);
                animateAPoint(300, 240, 220, 320, 1000);
            } else if (step === 3) {
                show(stepGroups[3]);
            } else if (step === 4) {
                show(stepGroups[4]);
            } else if (step === 5) {
                show(stepGroups[5]);
            }
        }

        // 事件绑定
        btnPrev.addEventListener('click', function() {
            goToStep(currentStep - 1);
        });
        btnNext.addEventListener('click', function() {
            goToStep(currentStep + 1);
        });

        // 初始化
        goToStep(0);
    })();
    </script>
