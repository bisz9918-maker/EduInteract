    <script>
    (function() {
        // Scene 1 实现
        let currentStep = 0;
        const totalSteps = 4; // 步骤0~3

        const step1Group = document.getElementById('scene1-step1-group');
        const step2Group = document.getElementById('scene1-step2-group');
        const conclusionGroup = document.getElementById('scene1-conclusion-group');
        const btnPrev = document.getElementById('scene1-btn-prev');
        const btnNext = document.getElementById('scene1-btn-next');
        const stepIndicator = document.getElementById('scene1-step-indicator');

        function show(el) {
            if (!el) return;
            el.style.display = 'block';
            el.style.opacity = '1';
        }
        function hide(el) {
            if (!el) return;
            el.style.display = 'none';
            el.style.opacity = '0';
        }

        function renderStep() {
            // 更新指示器
            stepIndicator.textContent = '步骤 ' + (currentStep + 1) + ' / ' + totalSteps;
            // 更新按钮状态
            btnPrev.disabled = (currentStep === 0);
            btnNext.disabled = (currentStep === totalSteps - 1);

            // 隐藏所有步骤专属元素
            hide(step1Group);
            hide(step2Group);
            hide(conclusionGroup);

            // 根据当前步骤显示对应元素
            if (currentStep >= 1) {
                show(step1Group);
            }
            if (currentStep >= 2) {
                show(step2Group);
            }
            if (currentStep >= 3) {
                show(conclusionGroup);
            }
        }

        btnPrev.addEventListener('click', function() {
            if (currentStep > 0) {
                currentStep--;
                renderStep();
            }
        });

        btnNext.addEventListener('click', function() {
            if (currentStep < totalSteps - 1) {
                currentStep++;
                renderStep();
            }
        });

        // 初始渲染
        renderStep();
    })();
    </script>
