    <script>
    (function() {
        // Scene 2 实现：长方形计数规律探究

        // 坐标表（实现对照依据）
        // 大长方形外框: (220, 40), 360×120
        // 小长方形填充区: (220+i·w, 40), w=360/n, w×120, i=0~n-1
        // 内部竖直分割线: (220+i·w, 40)-(220+i·w, 160), i=1~n-1
        // 竖线短标记: (220+i·w, 175)-(220+i·w, 185), i=0~n
        // 竖线编号: (220+i·w, 188), i=0~n
        // 滑块标签: (400, 228)
        // 滑块数值: (cx, 262), cx = 250 + (n-2)*75
        // 滑块轨道: (250, 285), 300×6
        // 滑块按钮: (cx, 295), r=10
        // 特殊标记: (400, 330)
        // 竖线数公式: (400, 368)
        // 总数公式+结果: (400, 408)
        // 本题算式: (400, 446)

        const slider = document.getElementById('scene2-slider');
        const fillGroup = document.getElementById('scene2-fill-group');
        const dividerGroup = document.getElementById('scene2-divider-group');
        const tickGroup = document.getElementById('scene2-tick-group');
        const tickLabelGroup = document.getElementById('scene2-tick-label-group');
        const sliderValue = document.getElementById('scene2-slider-value');
        const formulaSpecial = document.getElementById('scene2-formula-special');
        const formulaLines = document.getElementById('scene2-formula-lines');
        const formulaResult = document.getElementById('scene2-formula-result');
        const formulaExample = document.getElementById('scene2-formula-example');

        function updateScene(n) {
            const nVal = parseInt(n, 10);
            const w = 360 / nVal;

            // 清空动态组
            fillGroup.innerHTML = '';
            dividerGroup.innerHTML = '';
            tickGroup.innerHTML = '';
            tickLabelGroup.innerHTML = '';

            // 小长方形填充区
            for (let i = 0; i < nVal; i++) {
                const rect = document.createElementNS('http://www.w3.org/2000/svg', 'rect');
                rect.setAttribute('x', 220 + i * w);
                rect.setAttribute('y', 40);
                rect.setAttribute('width', w);
                rect.setAttribute('height', 120);
                rect.setAttribute('fill', '#1565c0');
                rect.setAttribute('opacity', '0.15');
                fillGroup.appendChild(rect);
            }

            // 内部竖直分割线（i=1~n-1）
            for (let i = 1; i < nVal; i++) {
                const line = document.createElementNS('http://www.w3.org/2000/svg', 'line');
                line.setAttribute('x1', 220 + i * w);
                line.setAttribute('y1', 40);
                line.setAttribute('x2', 220 + i * w);
                line.setAttribute('y2', 160);
                line.setAttribute('stroke', '#000000');
                line.setAttribute('stroke-width', '2');
                dividerGroup.appendChild(line);
            }

            // 竖线短标记（i=0~n）
            for (let i = 0; i <= nVal; i++) {
                const line = document.createElementNS('http://www.w3.org/2000/svg', 'line');
                line.setAttribute('x1', 220 + i * w);
                line.setAttribute('y1', 175);
                line.setAttribute('x2', 220 + i * w);
                line.setAttribute('y2', 185);
                line.setAttribute('stroke', '#000000');
                line.setAttribute('stroke-width', '1');
                tickGroup.appendChild(line);
            }

            // 竖线编号（i=0~n）
            for (let i = 0; i <= nVal; i++) {
                const text = document.createElementNS('http://www.w3.org/2000/svg', 'text');
                text.setAttribute('x', 220 + i * w);
                text.setAttribute('y', 188);
                text.setAttribute('text-anchor', 'middle');
                text.setAttribute('font-size', '12');
                text.setAttribute('fill', '#000000');
                text.textContent = String(i + 1);
                tickLabelGroup.appendChild(text);
            }

            // 更新滑块数值位置
            const cx = 250 + (nVal - 2) * 75;
            sliderValue.setAttribute('x', cx);
            sliderValue.textContent = 'n = ' + nVal;

            // 更新公式
            const totalLines = nVal + 1;
            const totalRects = totalLines * nVal / 2;
            formulaLines.textContent = '竖线总数 = n + 1 = ' + totalLines;
            formulaResult.textContent = totalRects;

            // n=4 特殊标记
            if (nVal === 4) {
                formulaSpecial.style.display = 'block';
                formulaExample.style.display = 'block';
            } else {
                formulaSpecial.style.display = 'none';
                formulaExample.style.display = 'none';
            }
        }

        // 绑定滑块事件
        slider.addEventListener('input', function() {
            updateScene(this.value);
        });

        // 初始状态
        updateScene(4);
    })();
    </script>
