    <script>
    (function() {
        // Scene 1 实现
        var currentStep = 0;
        var maxStep = 4;

        var btnPrev = document.getElementById('scene1-btn-prev');
        var btnNext = document.getElementById('scene1-btn-next');
        var indicator = document.getElementById('scene1-step-indicator');
        var square3dm = document.getElementById('scene1-square-3dm');
        var square2dm = document.getElementById('scene1-square-2dm');

        function setDisplay(selector, display) {
            var els = document.querySelectorAll(selector);
            for (var i = 0; i < els.length; i++) {
                els[i].style.display = display;
            }
        }

        function renderStep() {
            // 更新指示器
            indicator.textContent = '步骤 ' + (currentStep + 1) + ' / 5';

            // 按钮状态
            btnPrev.disabled = currentStep === 0;
            btnNext.disabled = currentStep === maxStep;

            // 重置所有步骤相关元素
            setDisplay('.scene1-step1', 'none');
            setDisplay('.scene1-step1-only', 'none');
            setDisplay('.scene1-step2', 'none');
            setDisplay('.scene1-step3', 'none');
            setDisplay('.scene1-step4', 'none');

            // 重置边框粗细
            square3dm.setAttribute('stroke-width', '2');
            square2dm.setAttribute('stroke-width', '2');

            // 根据步骤显示元素
            if (currentStep >= 1) {
                setDisplay('.scene1-step1', 'block');
            }
            if (currentStep === 1) {
                setDisplay('.scene1-step1-only', 'block');
            }
            if (currentStep >= 2) {
                setDisplay('.scene1-step2', 'block');
                square3dm.setAttribute('stroke-width', '3');
            }
            if (currentStep >= 3) {
                setDisplay('.scene1-step3', 'block');
                square2dm.setAttribute('stroke-width', '3');
            }
            if (currentStep === 4) {
                setDisplay('.scene1-step4', 'block');
                // 步骤4隐藏网格、面积标签、块数标签、所有算式、选项
                setDisplay('.scene1-step1', 'none');
                setDisplay('.scene1-step2', 'none');
                setDisplay('.scene1-step3', 'none');
            }
        }

        btnPrev.addEventListener('click', function() {
            if (currentStep > 0) {
                currentStep--;
                renderStep();
            }
        });

        btnNext.addEventListener('click', function() {
            if (currentStep < maxStep) {
                currentStep++;
                renderStep();
            }
        });

        // 初始渲染
        renderStep();
    })();
    </script>
