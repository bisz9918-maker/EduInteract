    <script>
    (function() {
        // Scene 1 实现
        var currentStep = 0; // 0~3
        var totalSteps = 4;

        // 元素引用
        var triJia = document.getElementById('scene1-tri-jia');
        var triYi = document.getElementById('scene1-tri-yi');
        var btnPrev = document.getElementById('scene1-btn-prev');
        var btnNext = document.getElementById('scene1-btn-next');
        var stepIndicator = document.getElementById('scene1-step-indicator');
        var optionC = document.getElementById('scene1-option-c');

        // 甲相关元素
        var arrowJiaBase = document.getElementById('scene1-arrow-jia-base');
        var numJiaBase = document.getElementById('scene1-num-jia-base');
        var arrowJiaHeight = document.getElementById('scene1-arrow-jia-height');
        var numJiaHeight = document.getElementById('scene1-num-jia-height');
        var formulaJia = document.getElementById('scene1-formula-jia');

        // 乙相关元素
        var arrowYiBase = document.getElementById('scene1-arrow-yi-base');
        var numYiBase = document.getElementById('scene1-num-yi-base');
        var arrowYiHeight = document.getElementById('scene1-arrow-yi-height');
        var numYiHeight = document.getElementById('scene1-num-yi-height');
        var formulaYi = document.getElementById('scene1-formula-yi');

        // 结论相关元素
        var conclusion1 = document.getElementById('scene1-conclusion-1');
        var conclusion2 = document.getElementById('scene1-conclusion-2');

        function show(el) {
            if (el) { el.style.display = 'block'; el.style.opacity = '1'; }
        }
        function hide(el) {
            if (el) { el.style.display = 'none'; }
        }
        function showFlex(el) {
            if (el) { el.style.display = 'flex'; el.style.opacity = '1'; }
        }

        function updateScene() {
            // 更新步骤指示器
            stepIndicator.textContent = (currentStep + 1) + '/' + totalSteps;

            // 更新按钮状态
            btnPrev.disabled = (currentStep === 0);
            btnNext.disabled = (currentStep === totalSteps - 1);

            // 重置所有高亮和标注
            triJia.setAttribute('stroke', '#000000');
            triJia.setAttribute('stroke-width', '1.5');
            triYi.setAttribute('stroke', '#000000');
            triYi.setAttribute('stroke-width', '1.5');

            hide(arrowJiaBase); hide(numJiaBase);
            hide(arrowJiaHeight); hide(numJiaHeight);
            hide(formulaJia);
            hide(arrowYiBase); hide(numYiBase);
            hide(arrowYiHeight); hide(numYiHeight);
            hide(formulaYi);
            hide(conclusion1); hide(conclusion2);
            hide(optionC);

            switch (currentStep) {
                case 0:
                    // 步骤0：观察原图，只显示基本图形和标签
                    break;
                case 1:
                    // 步骤1：分析甲
                    triJia.setAttribute('stroke', '#1565c0');
                    triJia.setAttribute('stroke-width', '2.5');
                    show(arrowJiaBase); show(numJiaBase);
                    show(arrowJiaHeight); show(numJiaHeight);
                    show(formulaJia);
                    break;
                case 2:
                    // 步骤2：分析乙
                    triYi.setAttribute('stroke', '#2e7d32');
                    triYi.setAttribute('stroke-width', '2.5');
                    show(arrowYiBase); show(numYiBase);
                    show(arrowYiHeight); show(numYiHeight);
                    show(formulaYi);
                    break;
                case 3:
                    // 步骤3：得出结论
                    triJia.setAttribute('stroke', '#1565c0');
                    triJia.setAttribute('stroke-width', '2.5');
                    triYi.setAttribute('stroke', '#2e7d32');
                    triYi.setAttribute('stroke-width', '2.5');
                    show(arrowJiaBase); show(numJiaBase);
                    show(arrowJiaHeight); show(numJiaHeight);
                    show(formulaJia);
                    show(arrowYiBase); show(numYiBase);
                    show(arrowYiHeight); show(numYiHeight);
                    show(formulaYi);
                    show(conclusion1); show(conclusion2);
                    showFlex(optionC);
                    break;
            }
        }

        btnPrev.addEventListener('click', function() {
            if (currentStep > 0) {
                currentStep--;
                updateScene();
            }
        });

        btnNext.addEventListener('click', function() {
            if (currentStep < totalSteps - 1) {
                currentStep++;
                updateScene();
            }
        });

        // 初始化
        updateScene();
    })();
    </script>
