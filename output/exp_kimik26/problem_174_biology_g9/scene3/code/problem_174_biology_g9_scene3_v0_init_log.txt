    <script>
    (function() {
        // Scene 3 实现：突触结构与抑制性递质作用机制

        var currentStep = 0;
        var totalSteps = 4;

        var btnPrev = document.getElementById('scene3-btn-prev');
        var btnNext = document.getElementById('scene3-btn-next');
        var indicator = document.getElementById('scene3-step-indicator');
        var leftGroup = document.getElementById('scene3-left-group');
        var rightTitleGroup = document.getElementById('scene3-right-title-group');
        var step1Group = document.getElementById('scene3-step1-group');
        var step2Group = document.getElementById('scene3-step2-group');
        var step3Group = document.getElementById('scene3-step3-group');
        var step4Group = document.getElementById('scene3-step4-group');

        function updateIndicator() {
            indicator.textContent = '步骤 ' + currentStep + '/' + totalSteps;
        }

        function setDisplay(el, show) {
            if (!el) return;
            if (show) {
                el.style.display = 'block';
                el.style.opacity = '1';
            } else {
                el.style.display = 'none';
                el.style.opacity = '0';
            }
        }

        function setOpacity(el, opacity) {
            if (!el) return;
            el.style.opacity = String(opacity);
        }

        function renderStep() {
            // 更新按钮状态
            btnPrev.disabled = (currentStep === 0);
            btnNext.disabled = (currentStep === totalSteps);
            updateIndicator();

            // 步骤0：初始状态
            if (currentStep === 0) {
                setDisplay(leftGroup, true);
                setOpacity(leftGroup, 1);
                setDisplay(rightTitleGroup, true);
                setDisplay(step1Group, false);
                setDisplay(step2Group, false);
                setDisplay(step3Group, false);
                setDisplay(step4Group, false);
            }
            // 步骤1：兴奋传递
            else if (currentStep === 1) {
                setDisplay(leftGroup, true);
                setOpacity(leftGroup, 1);
                setDisplay(rightTitleGroup, true);
                setDisplay(step1Group, true);
                setDisplay(step2Group, false);
                setDisplay(step3Group, false);
                setDisplay(step4Group, false);
            }
            // 步骤2：抑制性递质释放
            else if (currentStep === 2) {
                setDisplay(leftGroup, true);
                setOpacity(leftGroup, 1);
                setDisplay(rightTitleGroup, true);
                setDisplay(step1Group, true);
                setDisplay(step2Group, true);
                setDisplay(step3Group, false);
                setDisplay(step4Group, false);
            }
            // 步骤3：超极化与兴奋阻断
            else if (currentStep === 3) {
                setDisplay(leftGroup, true);
                setOpacity(leftGroup, 1);
                setDisplay(rightTitleGroup, true);
                setDisplay(step1Group, true);
                setDisplay(step2Group, true);
                setDisplay(step3Group, true);
                setDisplay(step4Group, false);
            }
            // 步骤4：受体分子结构放大
            else if (currentStep === 4) {
                setDisplay(leftGroup, true);
                setOpacity(leftGroup, 0.4);
                setDisplay(rightTitleGroup, false);
                setDisplay(step1Group, true);
                setDisplay(step2Group, true);
                setDisplay(step3Group, true);
                setDisplay(step4Group, true);
            }
        }

        btnPrev.addEventListener('click', function() {
            if (currentStep > 0) {
                currentStep--;
                renderStep();
            }
        });

        btnNext.addEventListener('click', function() {
            if (currentStep < totalSteps) {
                currentStep++;
                renderStep();
            }
        });

        // 初始化
        renderStep();
    })();
    </script>
