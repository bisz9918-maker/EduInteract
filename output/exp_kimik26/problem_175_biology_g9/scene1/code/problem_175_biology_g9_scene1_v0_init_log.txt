    <script>
    (function() {
        // Scene 1 实现
        let currentStep = 0;
        const totalSteps = 4;

        const btnPrev = document.getElementById('scene1-btn-prev');
        const btnNext = document.getElementById('scene1-btn-next');
        const stepIndicator = document.getElementById('scene1-step-indicator');

        const labelsGroup = document.getElementById('scene1-labels-group');
        const cardsGroup = document.getElementById('scene1-cards-group');
        const optionsGroup = document.getElementById('scene1-options-group');

        const labelGymnosperm = document.getElementById('scene1-label-gymnosperm');
        const labelAlgae = document.getElementById('scene1-label-algae');
        const labelFern = document.getElementById('scene1-label-fern');
        const labelMoss = document.getElementById('scene1-label-moss');

        const end1 = document.getElementById('scene1-tree-end-1');
        const end2 = document.getElementById('scene1-tree-end-2');
        const end3 = document.getElementById('scene1-tree-end-3');
        const end4 = document.getElementById('scene1-tree-end-4');

        const pathRootSeed = document.getElementById('scene1-path-root-seed');
        const pathSeedEnd1 = document.getElementById('scene1-path-seed-end1');
        const pathRootSeedless = document.getElementById('scene1-path-root-seedless');
        const pathSeedlessNostem = document.getElementById('scene1-path-seedless-nostem');
        const pathNostemEnd2 = document.getElementById('scene1-path-nostem-end2');
        const pathSeedlessStem = document.getElementById('scene1-path-seedless-stem');
        const pathStemVascular = document.getElementById('scene1-path-stem-vascular');
        const pathVascularEnd3 = document.getElementById('scene1-path-vascular-end3');
        const pathStemNovascular = document.getElementById('scene1-path-stem-novascular');
        const pathNovascularEnd4 = document.getElementById('scene1-path-novascular-end4');

        const cardAlgae = document.getElementById('scene1-card-algae');
        const cardMoss = document.getElementById('scene1-card-moss');
        const cardFern = document.getElementById('scene1-card-fern');
        const cardGymnosperm = document.getElementById('scene1-card-gymnosperm');

        function sleep(ms) {
            return new Promise(resolve => setTimeout(resolve, ms));
        }

        function show(el) {
            el.style.display = '';
            el.style.opacity = '1';
            el.classList.remove('scene1-hidden');
        }

        function hide(el) {
            el.style.opacity = '0';
            el.classList.add('scene1-hidden');
        }

        function hideDisplay(el) {
            el.style.display = 'none';
            el.classList.add('scene1-hidden-display');
        }

        function showDisplay(el) {
            el.style.display = '';
            el.classList.remove('scene1-hidden-display');
        }

        // 清除所有高亮
        function clearHighlights() {
            [end1, end2, end3, end4].forEach(el => el.classList.remove('scene1-highlight-blue'));
            [pathRootSeed, pathSeedEnd1, pathRootSeedless, pathSeedlessNostem, pathNostemEnd2,
             pathSeedlessStem, pathStemVascular, pathVascularEnd3, pathStemNovascular, pathNovascularEnd4]
                .forEach(el => el.classList.remove('scene1-path-highlight'));
            [cardAlgae, cardMoss, cardFern, cardGymnosperm].forEach(el => {
                const rect = el.querySelector('rect');
                if (rect) rect.classList.remove('scene1-card-highlight');
            });
        }

        // 高亮分支
        function highlightBranch(endEl, pathEls) {
            endEl.classList.add('scene1-highlight-blue');
            pathEls.forEach(p => p.classList.add('scene1-path-highlight'));
        }

        // 步骤1：依次高亮四个末端节点
        async function runStep1() {
            showDisplay(labelsGroup);
            // ① 裸子植物
            highlightBranch(end1, [pathRootSeed, pathSeedEnd1]);
            show(labelGymnosperm);
            await sleep(700);
            // ② 藻类植物
            highlightBranch(end2, [pathRootSeedless, pathSeedlessNostem, pathNostemEnd2]);
            show(labelAlgae);
            await sleep(700);
            // ③ 蕨类植物
            highlightBranch(end3, [pathRootSeedless, pathSeedlessStem, pathStemVascular, pathVascularEnd3]);
            show(labelFern);
            await sleep(700);
            // ④ 苔藓植物
            highlightBranch(end4, [pathRootSeedless, pathSeedlessStem, pathStemNovascular, pathNovascularEnd4]);
            show(labelMoss);
            await sleep(700);
        }

        // 步骤2：卡片淡入并依次高亮
        async function runStep2() {
            showDisplay(cardsGroup);
            // 触发reflow
            void cardsGroup.offsetWidth;
            cardsGroup.classList.add('scene1-fade-in');
            await sleep(500);

            // 依次高亮卡片：藻类 → 苔藓 → 蕨类 → 裸子
            const cardOrder = [cardAlgae, cardMoss, cardFern, cardGymnosperm];
            for (const card of cardOrder) {
                const rect = card.querySelector('rect');
                if (rect) rect.classList.add('scene1-card-highlight');
                await sleep(800);
                if (rect) rect.classList.remove('scene1-card-highlight');
            }
        }

        // 步骤3：选项滑入
        async function runStep3() {
            showDisplay(optionsGroup);
            void optionsGroup.offsetWidth;
            optionsGroup.classList.add('scene1-slide-up');
        }

        // 隐藏所有步骤元素
        function hideAllStepElements() {
            hideDisplay(labelsGroup);
            hideDisplay(cardsGroup);
            hideDisplay(optionsGroup);
            hide(labelGymnosperm);
            hide(labelAlgae);
            hide(labelFern);
            hide(labelMoss);
            cardsGroup.classList.remove('scene1-fade-in');
            optionsGroup.classList.remove('scene1-slide-up');
            clearHighlights();
        }

        // 跳转到指定步骤
        async function goToStep(step) {
            currentStep = step;
            stepIndicator.textContent = '步骤 ' + (step + 1) + '/' + totalSteps;
            btnPrev.disabled = step === 0;
            btnNext.textContent = step === totalSteps - 1 ? '完成' : '下一步 ▶';

            hideAllStepElements();

            if (step === 0) {
                // 步骤0：只显示分类树
            } else if (step === 1) {
                await runStep1();
            } else if (step === 2) {
                // 先显示步骤1的内容（保持显示）
                showDisplay(labelsGroup);
                show(labelGymnosperm);
                show(labelAlgae);
                show(labelFern);
                show(labelMoss);
                end1.classList.add('scene1-highlight-blue');
                end2.classList.add('scene1-highlight-blue');
                end3.classList.add('scene1-highlight-blue');
                end4.classList.add('scene1-highlight-blue');
                await runStep2();
            } else if (step === 3) {
                // 显示之前步骤的内容
                showDisplay(labelsGroup);
                show(labelGymnosperm);
                show(labelAlgae);
                show(labelFern);
                show(labelMoss);
                end1.classList.add('scene1-highlight-blue');
                end2.classList.add('scene1-highlight-blue');
                end3.classList.add('scene1-highlight-blue');
                end4.classList.add('scene1-highlight-blue');
                showDisplay(cardsGroup);
                cardsGroup.style.opacity = '1';
                await runStep3();
            }
        }

        btnPrev.addEventListener('click', () => {
            if (currentStep > 0) goToStep(currentStep - 1);
        });

        btnNext.addEventListener('click', () => {
            if (currentStep < totalSteps - 1) goToStep(currentStep + 1);
        });

        // 初始化
        goToStep(0);
    })();
    </script>
