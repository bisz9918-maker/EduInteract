    <script>
    (function() {
        // Scene 1 实现
        let currentStep = 0;
        const totalSteps = 3;

        const btnPrev = document.getElementById('scene1-btn-prev');
        const btnNext = document.getElementById('scene1-btn-next');
        const stepIndicator = document.getElementById('scene1-step-indicator');

        // 获取所有需要控制的元素
        const vesselLeftUp = document.getElementById('scene1-vessel-left-up');
        const vesselLeftDown = document.getElementById('scene1-vessel-left-down');
        const vesselRightUp = document.getElementById('scene1-vessel-right-up');
        const vesselRightDown = document.getElementById('scene1-vessel-right-down');

        const fillLeftDown = document.getElementById('scene1-fill-left-down');
        const fillRightUp = document.getElementById('scene1-fill-right-up');

        const arrowLeft = document.getElementById('scene1-arrow-left');
        const arrowRight = document.getElementById('scene1-arrow-right');

        const veinLeader = document.getElementById('scene1-vein-leader');
        const arteryLeader = document.getElementById('scene1-artery-leader');
        const typeVein = document.getElementById('scene1-type-vein');
        const typeArtery = document.getElementById('scene1-type-artery');

        function updateButtons() {
            btnPrev.disabled = currentStep === 0;
            btnNext.disabled = currentStep === totalSteps;
            stepIndicator.textContent = '步骤 ' + currentStep + '/' + totalSteps;
        }

        function renderStep() {
            // 步骤0: 初始状态，只显示血管、丝带、编号
            // 步骤1: 显示血流方向箭头
            // 步骤2: 显示扎紧效应（膨胀/变扁）
            // 步骤3: 显示类型标签

            // 重置所有血管线宽
            vesselLeftUp.setAttribute('stroke-width', '2');
            vesselLeftDown.setAttribute('stroke-width', '2');
            vesselRightUp.setAttribute('stroke-width', '2');
            vesselRightDown.setAttribute('stroke-width', '2');

            // 隐藏所有动态元素
            arrowLeft.style.opacity = '0';
            arrowRight.style.opacity = '0';
            fillLeftDown.style.opacity = '0';
            fillRightUp.style.opacity = '0';
            veinLeader.style.opacity = '0';
            arteryLeader.style.opacity = '0';
            typeVein.style.opacity = '0';
            typeArtery.style.opacity = '0';

            if (currentStep >= 1) {
                // 步骤1+: 显示箭头
                arrowLeft.style.opacity = '1';
                arrowRight.style.opacity = '1';
            }

            if (currentStep >= 2) {
                // 步骤2+: 扎紧效应
                // 左血管下段(①区)膨胀: 线宽6px + 蓝色填充
                vesselLeftDown.setAttribute('stroke-width', '6');
                fillLeftDown.style.opacity = '1';
                // 左血管上段(④区)变扁: 线宽1px
                vesselLeftUp.setAttribute('stroke-width', '1');

                // 右血管上段(③区)膨胀: 线宽6px + 红色填充
                vesselRightUp.setAttribute('stroke-width', '6');
                fillRightUp.style.opacity = '1';
                // 右血管下段(②区)变扁: 线宽1px
                vesselRightDown.setAttribute('stroke-width', '1');
            }

            if (currentStep >= 3) {
                // 步骤3+: 显示类型标签和引线
                veinLeader.style.opacity = '1';
                arteryLeader.style.opacity = '1';
                typeVein.style.opacity = '1';
                typeArtery.style.opacity = '1';
            }

            updateButtons();
        }

        btnPrev.addEventListener('click', function() {
            if (currentStep > 0) {
                currentStep--;
                renderStep();
            }
        });

        btnNext.addEventListener('click', function() {
            if (currentStep < totalSteps) {
                currentStep++;
                renderStep();
            }
        });

        // 初始渲染
        renderStep();
    })();
    </script>
