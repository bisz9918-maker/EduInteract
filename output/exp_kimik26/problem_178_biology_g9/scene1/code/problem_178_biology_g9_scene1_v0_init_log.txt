<script>
(function() {
  let currentStep = 0;
  const totalSteps = 5;

  const els = {
    prevBtn: document.getElementById('scene1-btn-prev'),
    nextBtn: document.getElementById('scene1-btn-next'),
    indicator: document.getElementById('scene1-step-indicator'),
    optionA: document.getElementById('scene1-option-a'),
    optionB: document.getElementById('scene1-option-b'),
    optionC: document.getElementById('scene1-option-c'),
    optionD: document.getElementById('scene1-option-d'),
    f1WhiteBox: document.getElementById('scene1-f1-white-box'),
    f2White3Box: document.getElementById('scene1-f2-white3-box'),
    pWhiteBox: document.getElementById('scene1-p-white-box'),
    panels: [
      document.getElementById('scene1-panel-step0'),
      document.getElementById('scene1-panel-step1'),
      document.getElementById('scene1-panel-step2'),
      document.getElementById('scene1-panel-step3'),
      document.getElementById('scene1-panel-step4')
    ],
    genotypes: [
      document.getElementById('scene1-genotype-p-yellow'),
      document.getElementById('scene1-genotype-p-white'),
      document.getElementById('scene1-genotype-f1-yellow'),
      document.getElementById('scene1-genotype-f1-white'),
      document.getElementById('scene1-genotype-f2-yellow2'),
      document.getElementById('scene1-genotype-f2-yellow3'),
      document.getElementById('scene1-genotype-f2-white3')
    ],
    ratios: [
      document.getElementById('scene1-ratio-f1'),
      document.getElementById('scene1-ratio-f2')
    ]
  };

  function setArrow3Color(color, width, markerId) {
    ['scene1-arrow-3-v', 'scene1-arrow-3-l', 'scene1-arrow-3-r'].forEach(function(id) {
      var el = document.getElementById(id);
      if (el) {
        el.setAttribute('stroke', color);
        el.setAttribute('stroke-width', width);
        if (el.hasAttribute('marker-end')) {
          el.setAttribute('marker-end', 'url(#' + markerId + ')');
        }
      }
    });
  }

  function updateScene(step) {
    // Switch panel
    els.panels.forEach(function(panel, i) {
      panel.classList.toggle('active', i === step);
    });

    // Update nav buttons
    els.prevBtn.disabled = (step === 0);
    els.nextBtn.disabled = (step === totalSteps - 1);
    els.indicator.textContent = '步骤 ' + (step + 1) + '/' + totalSteps;

    // Reset arrow ③
    setArrow3Color('#000', '2', 'scene1-arrowhead');

    // Reset box highlights
    els.f1WhiteBox.classList.remove('scene1-box-highlight-green', 'scene1-box-highlight-red');
    els.f2White3Box.classList.remove('scene1-box-highlight-green', 'scene1-box-highlight-red');
    els.pWhiteBox.classList.remove('scene1-box-highlight-red');

    // Reset options
    els.optionA.classList.remove('correct', 'wrong');
    els.optionB.classList.remove('correct', 'wrong');
    els.optionC.classList.remove('correct', 'wrong');
    els.optionD.classList.remove('correct', 'wrong');

    // Hide genotypes and ratios
    els.genotypes.forEach(function(g) { g.classList.remove('show'); });
    els.ratios.forEach(function(r) { r.classList.remove('show'); });

    // Step 1+: trait segregation analysis
    if (step >= 1) {
      setArrow3Color('#c62828', '3', 'scene1-arrowhead-red');
      els.f1WhiteBox.classList.add('scene1-box-highlight-green');
      els.f2White3Box.classList.add('scene1-box-highlight-green');
      els.optionB.classList.add('correct');
    }

    // Step 2+: genotype annotation & test cross
    if (step >= 2) {
      els.genotypes.forEach(function(g) { g.classList.add('show'); });
      document.getElementById('scene1-ratio-f1').classList.add('show');
      els.optionA.classList.add('correct');
    }

    // Step 3+: F2 ratio calculation
    if (step >= 3) {
      document.getElementById('scene1-ratio-f2').classList.add('show');
      els.optionC.classList.add('correct');
    }

    // Step 4: genotype comparison & conclusion
    if (step >= 4) {
      els.f1WhiteBox.classList.remove('scene1-box-highlight-green');
      els.f2White3Box.classList.remove('scene1-box-highlight-green');
      els.pWhiteBox.classList.add('scene1-box-highlight-red');
      els.f1WhiteBox.classList.add('scene1-box-highlight-red');
      els.f2White3Box.classList.add('scene1-box-highlight-red');
      els.optionD.classList.add('wrong');
    }
  }

  els.prevBtn.addEventListener('click', function() {
    if (currentStep > 0) {
      currentStep--;
      updateScene(currentStep);
    }
  });

  els.nextBtn.addEventListener('click', function() {
    if (currentStep < totalSteps - 1) {
      currentStep++;
      updateScene(currentStep);
    }
  });

  // Initialize
  updateScene(currentStep);
})();
</script>
