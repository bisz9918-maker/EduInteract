    <script>
    (function() {
        let currentStep = 0;
        const totalSteps = 9;

        const btnPrev = document.getElementById('scene1-btn-prev');
        const btnNext = document.getElementById('scene1-btn-next');
        const stepDisplay = document.getElementById('scene1-step-display');

        const wires = {
            mainTop: ['scene1-wire-main-top-1'],
            branch1: ['scene1-wire-branch1-1', 'scene1-wire-branch1-2', 'scene1-wire-branch1-3'],
            branch2: ['scene1-wire-branch2-1', 'scene1-wire-branch2-2'],
            mainBottom: ['scene1-wire-main-bottom-1', 'scene1-wire-main-bottom-2']
        };

        const compBattery = document.getElementById('scene1-comp-battery');
        const compSwitch = document.getElementById('scene1-comp-switch');
        const compAmmeter = document.getElementById('scene1-comp-ammeter');
        const compL1 = document.getElementById('scene1-comp-l1');
        const compL2 = document.getElementById('scene1-comp-l2');

        const nodeJ1 = document.getElementById('scene1-node-j1');
        const nodeJ2 = document.getElementById('scene1-node-j2');

        const arrowsMainTop = document.getElementById('scene1-arrows-main-top');
        const arrowsBranch1 = document.getElementById('scene1-arrows-branch1');
        const arrowsBranch2 = document.getElementById('scene1-arrows-branch2');
        const arrowsMainBottom = document.getElementById('scene1-arrows-main-bottom');

        const labelJ1 = document.getElementById('scene1-label-j1');
        const labelJ2 = document.getElementById('scene1-label-j2');
        const labelBranch1 = document.getElementById('scene1-label-branch1');
        const labelBranch2 = document.getElementById('scene1-label-branch2');
        const labelMain = document.getElementById('scene1-label-main');
        const labelConclusion = document.getElementById('scene1-label-conclusion');
        const labelConclusion7 = document.getElementById('scene1-label-conclusion7');
        const labelConclusion8 = document.getElementById('scene1-label-conclusion8');

        const glowL1 = document.getElementById('scene1-glow-l1');
        const glowL2 = document.getElementById('scene1-glow-l2');

        const optionA = document.getElementById('scene1-option-a');
        const optionB = document.getElementById('scene1-option-b');
        const optionC = document.getElementById('scene1-option-c');
        const optionD = document.getElementById('scene1-option-d');
        const optionAMark = optionA.querySelector('.opt-mark');
        const optionBMark = optionB.querySelector('.opt-mark');
        const optionCMark = optionC.querySelector('.opt-mark');
        const optionDMark = optionD.querySelector('.opt-mark');

        function show(el) {
            if (!el) return;
            el.classList.remove('scene1-hidden');
            if (el.style && el.style.display === 'none') el.style.display = '';
        }

        function hide(el) {
            if (!el) return;
            el.classList.add('scene1-hidden');
        }

        function showCard(el) {
            if (!el) return;
            el.style.display = 'flex';
            el.classList.remove('scene1-hidden');
        }

        function hideCard(el) {
            if (!el) return;
            el.style.display = 'none';
            el.classList.add('scene1-hidden');
        }

        function setWireColor(wireIds, color, width) {
            wireIds.forEach(id => {
                const el = document.getElementById(id);
                if (el) {
                    el.setAttribute('stroke', color);
                    el.setAttribute('stroke-width', width || 2);
                }
            });
        }

        function setAllWiresBlack() {
            Object.values(wires).flat().forEach(id => {
                const el = document.getElementById(id);
                if (el) {
                    el.setAttribute('stroke', '#000000');
                    el.setAttribute('stroke-width', '2');
                }
            });
        }

        function setAllWiresGray() {
            Object.values(wires).flat().forEach(id => {
                const el = document.getElementById(id);
                if (el) {
                    el.setAttribute('stroke', '#888888');
                    el.setAttribute('stroke-width', '1');
                }
            });
        }

        function setCompStroke(el, color, width) {
            if (!el) return;
            el.querySelectorAll('line, circle').forEach(line => {
                if (line.hasAttribute('stroke')) {
                    line.setAttribute('stroke', color);
                    line.setAttribute('stroke-width', width || 2);
                }
            });
        }

        function resetCompStroke(el) {
            setCompStroke(el, '#000000', 2);
        }

        function renderStep() {
            btnPrev.disabled = currentStep === 0;
            btnNext.disabled = currentStep === totalSteps;
            stepDisplay.textContent = '步骤 ' + currentStep + '/' + totalSteps;

            setAllWiresBlack();
            resetCompStroke(compBattery);
            resetCompStroke(compSwitch);
            resetCompStroke(compAmmeter);
            resetCompStroke(compL1);
            resetCompStroke(compL2);

            hide(nodeJ1); hide(nodeJ2);
            hide(labelJ1); hide(labelJ2);
            hide(labelBranch1); hide(labelBranch2); hide(labelMain);
            hide(labelConclusion); hide(labelConclusion7); hide(labelConclusion8);

            hide(arrowsMainTop); hide(arrowsBranch1); hide(arrowsBranch2); hide(arrowsMainBottom);
            hide(glowL1); hide(glowL2);

            hideCard(optionA); hideCard(optionB); hideCard(optionC); hideCard(optionD);
            optionAMark.textContent = '';
            optionBMark.textContent = '';
            optionCMark.textContent = '';
            optionDMark.textContent = '';

            switch (currentStep) {
                case 0:
                    break;

                case 1:
                    setCompStroke(compBattery, '#1565c0', 3);
                    setCompStroke(compSwitch, '#1565c0', 3);
                    setCompStroke(compAmmeter, '#1565c0', 3);
                    setCompStroke(compL1, '#1565c0', 3);
                    setCompStroke(compL2, '#1565c0', 3);
                    break;

                case 2:
                    setAllWiresGray();
                    setWireColor(wires.mainTop, '#1565c0', 3);
                    show(nodeJ1); show(labelJ1);
                    show(arrowsMainTop); show(labelMain);
                    break;

                case 3:
                    setAllWiresGray();
                    setWireColor(wires.mainTop, '#1565c0', 3);
                    setWireColor(wires.branch1, '#1565c0', 3);
                    show(nodeJ1); show(labelJ1);
                    show(arrowsMainTop); show(arrowsBranch1); show(labelBranch1); show(labelMain);
                    break;

                case 4:
                    setAllWiresGray();
                    setWireColor(wires.mainTop, '#1565c0', 3);
                    setWireColor(wires.branch1, '#1565c0', 3);
                    setWireColor(wires.branch2, '#1565c0', 3);
                    show(nodeJ1); show(labelJ1);
                    show(arrowsMainTop); show(arrowsBranch1); show(arrowsBranch2);
                    show(labelBranch1); show(labelBranch2); show(labelMain);
                    break;

                case 5:
                    setAllWiresGray();
                    setWireColor(wires.mainTop, '#1565c0', 3);
                    setWireColor(wires.branch1, '#1565c0', 3);
                    setWireColor(wires.branch2, '#1565c0', 3);
                    setWireColor(wires.mainBottom, '#1565c0', 3);
                    show(nodeJ1); show(nodeJ2); show(labelJ1); show(labelJ2);
                    show(arrowsMainTop); show(arrowsBranch1); show(arrowsBranch2); show(arrowsMainBottom);
                    show(labelBranch1); show(labelBranch2); show(labelMain);
                    setCompStroke(compSwitch, '#1565c0', 3);
                    break;

                case 6:
                    setAllWiresGray();
                    setWireColor(wires.mainTop, '#1565c0', 3);
                    setWireColor(wires.branch1, '#1565c0', 3);
                    setWireColor(wires.branch2, '#1565c0', 3);
                    setWireColor(wires.mainBottom, '#1565c0', 3);
                    show(nodeJ1); show(nodeJ2); show(labelJ1); show(labelJ2);
                    show(arrowsMainTop); show(arrowsBranch1); show(arrowsBranch2); show(arrowsMainBottom);
                    show(labelBranch1); show(labelBranch2); show(labelMain);
                    show(glowL1); show(glowL2);
                    show(labelConclusion);
                    labelConclusion.textContent = 'L₁与L₂并联';
                    labelConclusion.setAttribute('fill', '#2e7d32');
                    break;

                case 7:
                    setAllWiresGray();
                    setWireColor(wires.mainTop, '#1565c0', 3);
                    setWireColor(wires.branch2, '#1565c0', 3);
                    setWireColor(wires.mainBottom, '#1565c0', 3);
                    setWireColor(wires.branch1, '#ed6c02', 3);
                    show(nodeJ1); show(nodeJ2); show(labelJ1); show(labelJ2);
                    show(arrowsMainTop); show(arrowsBranch1); show(arrowsBranch2); show(arrowsMainBottom);
                    show(labelBranch1); show(labelBranch2); show(labelMain);
                    show(glowL1); show(glowL2);
                    show(labelConclusion);
                    labelConclusion.textContent = 'L₁与L₂并联';
                    labelConclusion.setAttribute('fill', '#2e7d32');
                    show(labelConclusion7);
                    labelConclusion7.textContent = '电流表与L₁串联，测L₁电流';
                    setCompStroke(compAmmeter, '#ed6c02', 3);
                    setCompStroke(compL1, '#ed6c02', 3);
                    break;

                case 8:
                    setAllWiresGray();
                    setWireColor(wires.branch1, '#1565c0', 3);
                    setWireColor(wires.branch2, '#1565c0', 3);
                    setWireColor(wires.mainTop, '#7b1fa2', 3);
                    setWireColor(wires.mainBottom, '#7b1fa2', 3);
                    show(nodeJ1); show(nodeJ2); show(labelJ1); show(labelJ2);
                    show(arrowsMainTop); show(arrowsBranch1); show(arrowsBranch2); show(arrowsMainBottom);
                    show(labelBranch1); show(labelBranch2); show(labelMain);
                    show(glowL1); show(glowL2);
                    show(labelConclusion);
                    labelConclusion.textContent = 'L₁与L₂并联';
                    labelConclusion.setAttribute('fill', '#2e7d32');
                    show(labelConclusion7);
                    labelConclusion7.textContent = '电流表与L₁串联，测L₁电流';
                    show(labelConclusion8);
                    labelConclusion8.textContent = '开关在干路，控制整个电路';
                    setCompStroke(compAmmeter, '#ed6c02', 3);
                    setCompStroke(compL1, '#ed6c02', 3);
                    setCompStroke(compSwitch, '#7b1fa2', 3);
                    break;

                case 9:
                    setAllWiresBlack();
                    resetCompStroke(compBattery);
                    resetCompStroke(compSwitch);
                    resetCompStroke(compAmmeter);
                    resetCompStroke(compL1);
                    resetCompStroke(compL2);
                    show(nodeJ1); show(nodeJ2); show(labelJ1); show(labelJ2);
                    show(labelBranch1); show(labelBranch2); show(labelMain);
                    show(glowL1); show(glowL2);
                    show(labelConclusion);
                    labelConclusion.textContent = 'L₁与L₂并联';
                    labelConclusion.setAttribute('fill', '#2e7d32');
                    show(labelConclusion7);
                    labelConclusion7.textContent = '电流表与L₁串联，测L₁电流';
                    show(labelConclusion8);
                    labelConclusion8.textContent = '开关在干路，控制整个电路';

                    showCard(optionA); showCard(optionB); showCard(optionC); showCard(optionD);
                    optionAMark.textContent = '×  串联→错误';
                    optionBMark.textContent = '√  测L₁电流→正确';
                    optionCMark.textContent = '×  测L₂电流→错误';
                    optionDMark.textContent = '×  只控制L₂→错误';
                    break;
            }
        }

        btnPrev.addEventListener('click', function() {
            if (currentStep > 0) { currentStep--; renderStep(); }
        });

        btnNext.addEventListener('click', function() {
            if (currentStep < totalSteps) { currentStep++; renderStep(); }
        });

        renderStep();
    })();
    </script>
</body>
