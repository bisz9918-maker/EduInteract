    <script>
    (function() {
        // Scene 1 实现
        let currentStep = 0;
        const totalSteps = 4;

        const btnPrev = document.getElementById('scene1-btn-prev');
        const btnNext = document.getElementById('scene1-btn-next');
        const stepIndicator = document.getElementById('scene1-step-indicator');

        const step1Group = document.getElementById('scene1-step1-group');
        const step2Group = document.getElementById('scene1-step2-group');
        const step3Group = document.getElementById('scene1-step3-group');

        function updateUI() {
            // 更新步骤指示器
            stepIndicator.textContent = '步骤 ' + (currentStep + 1) + '/' + totalSteps;

            // 更新按钮状态
            btnPrev.disabled = currentStep === 0;
            btnNext.disabled = currentStep === totalSteps - 1;

            // 控制各步骤组的显隐
            step1Group.style.display = (currentStep >= 1) ? 'block' : 'none';
            step2Group.style.display = (currentStep >= 2) ? 'block' : 'none';
            step3Group.style.display = (currentStep >= 3) ? 'block' : 'none';
        }

        btnPrev.addEventListener('click', function() {
            if (currentStep > 0) {
                currentStep--;
                updateUI();
            }
        });

        btnNext.addEventListener('click', function() {
            if (currentStep < totalSteps - 1) {
                currentStep++;
                updateUI();
            }
        });

        // 初始化
        updateUI();
    })();
    </script>
