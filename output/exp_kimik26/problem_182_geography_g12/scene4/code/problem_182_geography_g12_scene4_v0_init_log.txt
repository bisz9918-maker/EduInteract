<script>
(function() {
let currentStep = 0;
const totalSteps = 4;
const btnPrev = document.getElementById('scene4-btn-prev');
const btnNext = document.getElementById('scene4-btn-next');
const stepIndicator = document.getElementById('scene4-step-indicator');
const step1Group = document.getElementById('scene4-step1-group');
const step2Group = document.getElementById('scene4-step2-group');
const step3Group = document.getElementById('scene4-step3-group');
const hintEquator = document.getElementById('scene4-hint-equator');
function updateUI() {
stepIndicator.textContent = '步骤 ' + (currentStep + 1) + ' / ' + totalSteps;
btnPrev.disabled = currentStep === 0;
btnNext.disabled = currentStep === totalSteps - 1;
step1Group.style.display = currentStep >= 1 ? 'block' : 'none';
step2Group.style.display = currentStep >= 2 ? 'block' : 'none';
step3Group.style.display = currentStep >= 3 ? 'block' : 'none';
hintEquator.style.display = currentStep >= 2 ? 'none' : 'block';
}
btnPrev.addEventListener('click', function() {
if (currentStep > 0) { currentStep--; updateUI(); }
});
btnNext.addEventListener('click', function() {
if (currentStep < totalSteps - 1) { currentStep++; updateUI(); }
});
updateUI();
})();
</script>
