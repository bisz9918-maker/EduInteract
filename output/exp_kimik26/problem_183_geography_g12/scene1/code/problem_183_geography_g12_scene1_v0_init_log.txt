    <script>
    (function() {
        // Scene 1 实现
        var currentStep = 0;
        var totalSteps = 5;

        // 步骤文字
        var stepTexts = [
            "步骤 0/5：整体概览",
            "步骤 1/5：A点·夏至 — 北极倾向太阳，直射北回归线",
            "步骤 2/5：D点·春分 — 直射赤道，昼夜平分",
            "步骤 3/5：C点·冬至 — 北极背离太阳，直射南回归线",
            "步骤 4/5：B点·秋分 — 直射赤道，昼夜平分",
            "步骤 5/5：直射点回归运动 — 南北回归线之间周期性移动"
        ];

        // 元素引用
        var elStepText = document.getElementById('scene1-step-text');
        var btnPrev = document.getElementById('scene1-btn-prev');
        var btnNext = document.getElementById('scene1-btn-next');

        // 地球组元素
        var groups = {
            a: document.getElementById('scene1-group-a'),
            d: document.getElementById('scene1-group-d'),
            c: document.getElementById('scene1-group-c'),
            b: document.getElementById('scene1-group-b')
        };

        var highlights = {
            a: document.getElementById('scene1-highlight-a'),
            d: document.getElementById('scene1-highlight-d'),
            c: document.getElementById('scene1-highlight-c'),
            b: document.getElementById('scene1-highlight-b')
        };

        var earths = {
            a: document.getElementById('scene1-earth-a'),
            d: document.getElementById('scene1-earth-d'),
            c: document.getElementById('scene1-earth-c'),
            b: document.getElementById('scene1-earth-b')
        };

        var axes = {
            a: document.getElementById('scene1-axis-a'),
            d: document.getElementById('scene1-axis-d'),
            c: document.getElementById('scene1-axis-c'),
            b: document.getElementById('scene1-axis-b')
        };

        var equators = {
            a: document.getElementById('scene1-equator-a'),
            d: document.getElementById('scene1-equator-d'),
            c: document.getElementById('scene1-equator-c'),
            b: document.getElementById('scene1-equator-b')
        };

        // 步骤专属组
        var stepGroups = [
            null, // step0 无专属组
            document.getElementById('scene1-step1-group'),
            document.getElementById('scene1-step2-group'),
            document.getElementById('scene1-step3-group'),
            document.getElementById('scene1-step4-group'),
            document.getElementById('scene1-step5-group')
        ];

        // 节气标签（步骤5保留）
        var seasonLabels = {
            a: document.getElementById('scene1-summer-label'),
            d: document.getElementById('scene1-spring-label'),
            c: document.getElementById('scene1-winter-label'),
            b: document.getElementById('scene1-autumn-label')
        };

        function setOpacity(el, val) {
            if (el) el.setAttribute('opacity', val);
        }

        function setDisplay(el, val) {
            if (el) el.style.display = val;
        }

        function setStroke(el, color, width) {
            if (el) {
                el.setAttribute('stroke', color);
                el.setAttribute('stroke-width', width);
            }
        }

        function renderStep() {
            // 更新步骤文字
            elStepText.textContent = stepTexts[currentStep];

            // 更新按钮状态
            btnPrev.disabled = (currentStep === 0);
            btnNext.disabled = (currentStep === totalSteps);

            // 重置所有地球样式
            ['a','d','c','b'].forEach(function(k) {
                setOpacity(groups[k], 1);
                setOpacity(highlights[k], 0);
                setStroke(earths[k], '#888', 1);
                setStroke(axes[k], '#888', 1.5);
                setStroke(equators[k], '#888', 1);
            });

            // 隐藏所有步骤专属组
            for (var i = 1; i <= totalSteps; i++) {
                setOpacity(stepGroups[i], 0);
                setDisplay(stepGroups[i], 'none');
            }

            // 隐藏所有节气标签
            ['a','d','c','b'].forEach(function(k) {
                setOpacity(seasonLabels[k], 0);
            });

            // 根据当前步骤设置样式
            if (currentStep === 0) {
                // 步骤0：整体概览，所有地球等权重
                // 默认样式已设置
            } else if (currentStep === 1) {
                // 步骤1：A点高亮，其余淡化
                setOpacity(groups.d, 0.35);
                setOpacity(groups.c, 0.35);
                setOpacity(groups.b, 0.35);
                setOpacity(highlights.a, 0.1);
                setStroke(earths.a, '#2e7d32', 3);
                setStroke(axes.a, '#000', 2.5);
                setStroke(equators.a, '#000', 1.5);
                setOpacity(stepGroups[1], 1);
                setDisplay(stepGroups[1], 'block');
            } else if (currentStep === 2) {
                // 步骤2：D点高亮，其余淡化
                setOpacity(groups.a, 0.35);
                setOpacity(groups.c, 0.35);
                setOpacity(groups.b, 0.35);
                setOpacity(highlights.d, 0.1);
                setStroke(earths.d, '#2e7d32', 3);
                setStroke(axes.d, '#000', 2.5);
                setStroke(equators.d, '#000', 1.5);
                setOpacity(stepGroups[2], 1);
                setDisplay(stepGroups[2], 'block');
            } else if (currentStep === 3) {
                // 步骤3：C点高亮，其余淡化
                setOpacity(groups.a, 0.35);
                setOpacity(groups.d, 0.35);
                setOpacity(groups.b, 0.35);
                setOpacity(highlights.c, 0.1);
                setStroke(earths.c, '#2e7d32', 3);
                setStroke(axes.c, '#000', 2.5);
                setStroke(equators.c, '#000', 1.5);
                setOpacity(stepGroups[3], 1);
                setDisplay(stepGroups[3], 'block');
            } else if (currentStep === 4) {
                // 步骤4：B点高亮，其余淡化
                setOpacity(groups.a, 0.35);
                setOpacity(groups.d, 0.35);
                setOpacity(groups.c, 0.35);
                setOpacity(highlights.b, 0.1);
                setStroke(earths.b, '#2e7d32', 3);
                setStroke(axes.b, '#000', 2.5);
                setStroke(equators.b, '#000', 1.5);
                setOpacity(stepGroups[4], 1);
                setDisplay(stepGroups[4], 'block');
            } else if (currentStep === 5) {
                // 步骤5：回归运动，所有地球正常，显示所有节气标签
                setOpacity(stepGroups[5], 1);
                setDisplay(stepGroups[5], 'block');
                setOpacity(seasonLabels.a, 1);
                setOpacity(seasonLabels.d, 1);
                setOpacity(seasonLabels.c, 1);
                setOpacity(seasonLabels.b, 1);
            }
        }

        window.scene1NextStep = function() {
            if (currentStep < totalSteps) {
                currentStep++;
                renderStep();
            }
        };

        window.scene1PrevStep = function() {
            if (currentStep > 0) {
                currentStep--;
                renderStep();
            }
        };

        window.scene1ResetStep = function() {
            currentStep = 0;
            renderStep();
        };

        // 初始化
        renderStep();
    })();
    </script>
