    <script>
    (function() {
        // Scene 2 实现
        var currentState = 0; // 0=夏至, 1=冬至, 2=春秋分

        var curves = [
            document.getElementById('scene2-curve-summer'),
            document.getElementById('scene2-curve-winter'),
            document.getElementById('scene2-curve-equinox')
        ];
        var markers = [
            document.getElementById('scene2-marker-summer'),
            document.getElementById('scene2-marker-winter'),
            document.getElementById('scene2-marker-equinox')
        ];
        var labels = [
            document.getElementById('scene2-label-summer'),
            document.getElementById('scene2-label-winter'),
            document.getElementById('scene2-label-equinox')
        ];
        var buttons = [
            document.getElementById('scene2-btn-summer'),
            document.getElementById('scene2-btn-winter'),
            document.getElementById('scene2-btn-equinox')
        ];

        function renderState() {
            for (var i = 0; i < 3; i++) {
                if (i === currentState) {
                    // 当前选中状态
                    curves[i].style.opacity = '1';
                    curves[i].setAttribute('stroke-width', '3');
                    markers[i].style.opacity = '1';
                    labels[i].style.opacity = '1';
                    buttons[i].classList.add('active');
                } else {
                    // 非选中状态
                    curves[i].style.opacity = '0.15';
                    curves[i].setAttribute('stroke-width', '2');
                    markers[i].style.opacity = '0';
                    labels[i].style.opacity = '0';
                    buttons[i].classList.remove('active');
                }
            }
        }

        // 初始状态：夏至
        renderState();

        // 按钮事件绑定
        buttons[0].addEventListener('click', function() {
            currentState = 0;
            renderState();
        });
        buttons[1].addEventListener('click', function() {
            currentState = 1;
            renderState();
        });
        buttons[2].addEventListener('click', function() {
            currentState = 2;
            renderState();
        });
    })();
    </script>
