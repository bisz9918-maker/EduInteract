    <script>
    (function() {
        // Scene 1 实现
        let currentStep = 0;
        const totalSteps = 5;

        const btnPrev = document.getElementById('scene1-btn-prev');
        const btnNext = document.getElementById('scene1-btn-next');
        const stepIndicator = document.getElementById('scene1-step-indicator');

        const stepGroups = [
            null, // step0 没有专属group，装置图始终显示
            document.getElementById('scene1-step1-group'),
            document.getElementById('scene1-step2-group'),
            document.getElementById('scene1-step3-group'),
            document.getElementById('scene1-step4-group')
        ];

        function updateButtons() {
            btnPrev.disabled = currentStep === 0;
            btnNext.disabled = currentStep === totalSteps - 1;
            stepIndicator.textContent = '步骤 ' + (currentStep + 1) + '/' + totalSteps;
        }

        function renderStep() {
            // 隐藏所有步骤专属group
            for (let i = 1; i < stepGroups.length; i++) {
                if (stepGroups[i]) {
                    stepGroups[i].setAttribute('display', 'none');
                }
            }
            // 显示当前步骤的group
            if (currentStep > 0 && stepGroups[currentStep]) {
                stepGroups[currentStep].setAttribute('display', 'block');
            }
            updateButtons();
        }

        btnPrev.addEventListener('click', function() {
            if (currentStep > 0) {
                currentStep--;
                renderStep();
            }
        });

        btnNext.addEventListener('click', function() {
            if (currentStep < totalSteps - 1) {
                currentStep++;
                renderStep();
            }
        });

        // 初始化
        renderStep();
    })();
    </script>
