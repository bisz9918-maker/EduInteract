    <script>
    (function() {
        // Scene 2 实现
        let currentStep = 0;
        const totalSteps = 6;

        // 元素引用
        const elOp = document.getElementById('scene2-op');
        const elPb = document.getElementById('scene2-pb');
        const elPa = document.getElementById('scene2-pa');
        const elPc = document.getElementById('scene2-pc');
        const elPd = document.getElementById('scene2-pd');
        const elConeFill = document.getElementById('scene2-cone-fill');
        const elPyramidFillPbd = document.getElementById('scene2-pyramid-fill-pbd');
        const elPyramidFillPad = document.getElementById('scene2-pyramid-fill-pad');
        const elPyramidFillPab = document.getElementById('scene2-pyramid-fill-pab');
        const elSemicircle = document.getElementById('scene2-semicircle');
        const elTriangle = document.getElementById('scene2-triangle');
        const elShadow = document.getElementById('scene2-shadow');
        const elLabelP = document.getElementById('scene2-label-p');
        const elLabelAHeight = document.getElementById('scene2-label-a-height');
        const elLabelCone = document.getElementById('scene2-label-cone');
        const elLabelPyramid = document.getElementById('scene2-label-pyramid');
        const elViews = document.getElementById('scene2-views');
        const elProjectionLines = document.getElementById('scene2-projection-lines');
        const btnPrev = document.getElementById('scene2-btn-prev');
        const btnNext = document.getElementById('scene2-btn-next');
        const indicator = document.getElementById('scene2-step-indicator');

        function setOpacity(el, val) { if(el) el.style.opacity = String(val); }
        function setStroke(el, color, width, dash) {
            if(!el) return;
            if(color) el.setAttribute('stroke', color);
            if(width !== undefined) el.setAttribute('stroke-width', String(width));
            if(dash !== undefined) el.setAttribute('stroke-dasharray', dash);
        }

        function applyStep(step) {
            // 重置所有元素到默认状态
            setOpacity(elOp, 0);
            setOpacity(elPb, 0);
            setOpacity(elPa, 0);
            setOpacity(elPc, 0);
            setOpacity(elPd, 0);
            setOpacity(elConeFill, 0);
            setOpacity(elPyramidFillPbd, 0);
            setOpacity(elPyramidFillPad, 0);
            setOpacity(elPyramidFillPab, 0);
            setOpacity(elShadow, 0);
            setOpacity(elLabelP, 0);
            setOpacity(elLabelAHeight, 0);
            setOpacity(elLabelCone, 0);
            setOpacity(elLabelPyramid, 0);
            setOpacity(elViews, 0);
            setOpacity(elProjectionLines, 0);

            // 恢复底面默认样式
            setStroke(elSemicircle, '#1565c0', 2, 'none');
            setStroke(elTriangle, '#c62828', 2, 'none');
            setOpacity(elSemicircle, 1);
            setOpacity(elTriangle, 1);

            // 恢复母线/侧棱默认样式
            setStroke(elPb, '#2e7d32', 1.5, 'none');
            setStroke(elPa, '#2e7d32', 1.5, 'none');
            setStroke(elPc, '#2e7d32', 1.5, 'none');
            setStroke(elPd, '#2e7d32', 1.5, 'none');

            switch(step) {
                case 0:
                    // 步骤0：只显示底面
                    break;
                case 1:
                    // 步骤1：拔高，显示OP、P、a，绿色虚线
                    setOpacity(elOp, 1);
                    setOpacity(elPb, 1);
                    setOpacity(elPa, 1);
                    setOpacity(elPc, 1);
                    setOpacity(elPd, 1);
                    setOpacity(elLabelP, 1);
                    setOpacity(elLabelAHeight, 1);
                    setStroke(elPb, '#2e7d32', 1.5, '4,4');
                    setStroke(elPa, '#2e7d32', 1.5, '4,4');
                    setStroke(elPc, '#2e7d32', 1.5, '4,4');
                    setStroke(elPd, '#2e7d32', 1.5, '4,4');
                    break;
                case 2:
                    // 步骤2：高亮半圆锥，淡化三棱锥
                    setOpacity(elOp, 1);
                    setOpacity(elPb, 1);
                    setOpacity(elPa, 1);
                    setOpacity(elPc, 1);
                    setOpacity(elPd, 0.3);
                    setOpacity(elLabelP, 1);
                    setOpacity(elLabelAHeight, 1);
                    setOpacity(elConeFill, 0.2);
                    setOpacity(elLabelCone, 1);

                    // 半圆锥高亮
                    setStroke(elSemicircle, '#1565c0', 3, 'none');
                    setStroke(elPb, '#1565c0', 2, 'none');
                    setStroke(elPa, '#1565c0', 2, 'none');
                    setStroke(elPc, '#1565c0', 2, 'none');

                    // 三棱锥淡化
                    setStroke(elTriangle, '#888888', 1.5, '4,4');
                    setOpacity(elTriangle, 0.3);
                    break;
                case 3:
                    // 步骤3：高亮三棱锥，淡化半圆锥
                    setOpacity(elOp, 1);
                    setOpacity(elPb, 1);
                    setOpacity(elPa, 1);
                    setOpacity(elPd, 1);
                    setOpacity(elPc, 0.3);
                    setOpacity(elLabelP, 1);
                    setOpacity(elLabelAHeight, 1);
                    setOpacity(elPyramidFillPbd, 0.2);
                    setOpacity(elPyramidFillPad, 0.2);
                    setOpacity(elPyramidFillPab, 0.2);
                    setOpacity(elLabelPyramid, 1);

                    // 三棱锥高亮
                    setStroke(elTriangle, '#c62828', 3, 'none');
                    setStroke(elPb, '#c62828', 2, 'none');
                    setStroke(elPa, '#c62828', 2, 'none');
                    setStroke(elPd, '#c62828', 2, 'none');

                    // 半圆锥淡化
                    setStroke(elSemicircle, '#888888', 1.5, '4,4');
                    setOpacity(elSemicircle, 0.3);
                    break;
                case 4:
                    // 步骤4：组合体
                    setOpacity(elOp, 1);
                    setOpacity(elPb, 1);
                    setOpacity(elPa, 1);
                    setOpacity(elPc, 1);
                    setOpacity(elPd, 1);
                    setOpacity(elLabelP, 1);
                    setOpacity(elLabelAHeight, 1);
                    setOpacity(elConeFill, 0.15);
                    setOpacity(elPyramidFillPbd, 0.15);
                    setOpacity(elPyramidFillPad, 0.15);
                    setOpacity(elPyramidFillPab, 0.15);
                    setOpacity(elShadow, 1);

                    setStroke(elSemicircle, '#1565c0', 2, 'none');
                    setStroke(elTriangle, '#c62828', 2, 'none');
                    setStroke(elPb, '#1565c0', 1.5, 'none');
                    setStroke(elPa, '#1565c0', 1.5, 'none');
                    setStroke(elPc, '#1565c0', 1.5, 'none');
                    setStroke(elPd, '#c62828', 1.5, 'none');
                    break;
                case 5:
                    // 步骤5：投影验证
                    setOpacity(elOp, 1);
                    setOpacity(elPb, 1);
                    setOpacity(elPa, 1);
                    setOpacity(elPc, 1);
                    setOpacity(elPd, 1);
                    setOpacity(elLabelP, 1);
                    setOpacity(elLabelAHeight, 1);
                    setOpacity(elConeFill, 0.15);
                    setOpacity(elPyramidFillPbd, 0.15);
                    setOpacity(elPyramidFillPad, 0.15);
                    setOpacity(elPyramidFillPab, 0.15);
                    setOpacity(elShadow, 1);
                    setOpacity(elViews, 1);
                    setOpacity(elProjectionLines, 1);

                    setStroke(elSemicircle, '#1565c0', 2, 'none');
                    setStroke(elTriangle, '#c62828', 2, 'none');
                    setStroke(elPb, '#1565c0', 1.5, 'none');
                    setStroke(elPa, '#1565c0', 1.5, 'none');
                    setStroke(elPc, '#1565c0', 1.5, 'none');
                    setStroke(elPd, '#c62828', 1.5, 'none');
                    break;
            }

            indicator.textContent = '步骤 ' + (step + 1) + '/' + totalSteps;
            btnPrev.disabled = step === 0;
            btnNext.disabled = step === totalSteps - 1;
        }

        btnPrev.addEventListener('click', function() {
            if(currentStep > 0) { currentStep--; applyStep(currentStep); }
        });
        btnNext.addEventListener('click', function() {
            if(currentStep < totalSteps - 1) { currentStep++; applyStep(currentStep); }
        });

        // 初始化
        applyStep(0);
    })();
    </script>
