    <script>
    (function() {
        // Scene 1 实现
        var currentStep = 0;
        var totalSteps = 5;

        var stepDescs = [
            "已知：AP是圆O的切线，P为切点；AC是割线，交圆O于B、C两点；M是BC的中点",
            "切线性质：OP ⊥ AP",
            "垂径定理：OM ⊥ AC",
            "O、P、A、M 四点共圆（以OA为直径）",
            "同弧OM所对圆周角相等：∠OAM = ∠OPM",
            "∠OAM + ∠APM = 90°"
        ];

        var stepDescEl = document.getElementById("scene1-step-desc");
        var stepIndicatorEl = document.getElementById("scene1-step-indicator");
        var btnPrev = document.getElementById("scene1-btn-prev");
        var btnNext = document.getElementById("scene1-btn-next");

        function setDisplay(id, show) {
            var el = document.getElementById(id);
            if (el) el.style.display = show ? "block" : "none";
        }

        function render() {
            // 更新步骤说明
            stepDescEl.textContent = stepDescs[currentStep];
            stepDescEl.setAttribute("fill", currentStep === 5 ? "#2e7d32" : (currentStep === 0 ? "#000000" : "#1565c0"));
            stepIndicatorEl.textContent = currentStep + " / " + totalSteps;

            // 按钮状态
            btnPrev.disabled = currentStep === 0;
            btnNext.disabled = currentStep === totalSteps;

            // 步骤1：显示OP和P处直角
            setDisplay("scene1-line-op", currentStep >= 1);
            setDisplay("scene1-rightangle-p", currentStep >= 1);

            // 步骤2：显示OM和M处直角
            setDisplay("scene1-line-om", currentStep >= 2);
            setDisplay("scene1-rightangle-m", currentStep >= 2);

            // 步骤3：显示辅助圆，四点放大变蓝；隐藏B标签避免与辅助圆重叠
            setDisplay("scene1-aux-circle", currentStep >= 3);
            setDisplay("scene1-label-b", currentStep < 3);
            var pointIds = ["scene1-point-o", "scene1-point-p", "scene1-point-a", "scene1-point-m"];
            pointIds.forEach(function(pid) {
                var el = document.getElementById(pid);
                if (el) {
                    if (currentStep >= 3) {
                        el.setAttribute("r", "5");
                        el.setAttribute("fill", "#1565c0");
                    } else {
                        el.setAttribute("r", "3");
                        el.setAttribute("fill", "#000000");
                    }
                }
            });

            // 步骤4：显示角弧和角标签
            setDisplay("scene1-angle-oam", currentStep >= 4);
            setDisplay("scene1-angle-opm", currentStep >= 4);
            setDisplay("scene1-label-angle-oam", currentStep >= 4);
            setDisplay("scene1-label-angle-opm", currentStep >= 4);

            // 步骤5：显示结论
            setDisplay("scene1-conclusion-1", currentStep >= 5);
            setDisplay("scene1-conclusion-2", currentStep >= 5);
        }

        btnPrev.addEventListener("click", function() {
            if (currentStep > 0) {
                currentStep--;
                render();
            }
        });

        btnNext.addEventListener("click", function() {
            if (currentStep < totalSteps) {
                currentStep++;
                render();
            }
        });

        // 初始渲染
        render();
    })();
    </script>
