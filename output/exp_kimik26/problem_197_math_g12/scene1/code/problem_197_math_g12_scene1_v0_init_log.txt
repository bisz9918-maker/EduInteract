    <script>
    (function() {
        // Scene 1 实现
        let currentStep = 0;
        const totalSteps = 5;

        const stepHints = [
            "步骤0：原题条件——已知∠A=26°，∠C=120°",
            "步骤1：求∠B=34°，标出中位线DE",
            "步骤2：DE ∥ BC，同位角∠ADE=34°",
            "步骤3：沿DE折叠，A→A′，∠A′DE=∠ADE=34°",
            "步骤4：平角180°−34°−34°=112°，即∠A′DB",
            "步骤5：A′在直线BC上，△A′DB为等腰三角形，底角34°验证112°"
        ];

        // 每步新增显示的元素 ID
        const stepElements = {
            0: ['scene1-line-ab','scene1-line-bc','scene1-line-ca','scene1-label-a','scene1-label-b','scene1-label-c','scene1-angle-a-arc','scene1-angle-a-text','scene1-angle-c-arc','scene1-angle-c-text'],
            1: ['scene1-angle-b-arc','scene1-angle-b-text','scene1-label-d','scene1-label-e','scene1-line-de','scene1-tick-ad','scene1-tick-db'],
            2: ['scene1-parallel','scene1-angle-ade-arc','scene1-angle-ade-text'],
            3: ['scene1-tri-a-de','scene1-line-a-d','scene1-line-a-e','scene1-label-a-prime','scene1-fold-arc','scene1-angle-a-de-arc','scene1-angle-a-de-text','scene1-line-bc-ext'],
            4: ['scene1-angle-a-db-arc','scene1-angle-a-db-text','scene1-formula'],
            5: ['scene1-tick-a-d','scene1-angle-dba-arc','scene1-angle-dba-text','scene1-angle-da-b-arc','scene1-angle-da-b-text','scene1-text-a-on-bc','scene1-tri-a-db-hl']
        };

        const btnPrev = document.getElementById('scene1-btn-prev');
        const btnNext = document.getElementById('scene1-btn-next');
        const hintEl = document.getElementById('scene1-step-hint');

        function setVisible(id, visible) {
            const el = document.getElementById(id);
            if (!el) return;
            if (visible) {
                el.classList.remove('scene1-hidden');
                el.style.display = '';
                el.style.opacity = '1';
            } else {
                el.classList.add('scene1-hidden');
                el.style.display = 'none';
                el.style.opacity = '0';
            }
        }

        function updateScene(step) {
            currentStep = step;
            hintEl.textContent = stepHints[step];
            btnPrev.disabled = (step === 0);
            btnNext.disabled = (step === totalSteps);

            // 收集所有动态元素
            const allDynamic = new Set();
            for (let s = 1; s <= totalSteps; s++) {
                if (stepElements[s]) stepElements[s].forEach(id => allDynamic.add(id));
            }
            // 先全部隐藏
            allDynamic.forEach(id => setVisible(id, false));

            // 恢复基础元素（步骤0）
            stepElements[0].forEach(id => setVisible(id, true));

            // 显示当前步骤及之前的元素
            for (let s = 1; s <= step; s++) {
                if (stepElements[s]) {
                    stepElements[s].forEach(id => setVisible(id, true));
                }
            }

            // 步骤3+：隐藏原 BC，显示延长线
            if (step >= 3) {
                setVisible('scene1-line-bc', false);
                setVisible('scene1-line-bc-ext', true);
            } else {
                setVisible('scene1-line-bc', true);
                setVisible('scene1-line-bc-ext', false);
            }
        }

        btnPrev.addEventListener('click', function() {
            if (currentStep > 0) updateScene(currentStep - 1);
        });

        btnNext.addEventListener('click', function() {
            if (currentStep < totalSteps) updateScene(currentStep + 1);
        });

        // 初始化
        updateScene(0);
    })();
    </script>
