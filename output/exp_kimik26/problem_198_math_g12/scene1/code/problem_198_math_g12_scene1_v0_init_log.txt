    <script>
    (function() {
        // Scene 1 实现
        var currentStep = 0;
        var totalSteps = 6; // 步骤0~5

        var btnPrev = document.getElementById('scene1-btn-prev');
        var btnNext = document.getElementById('scene1-btn-next');
        var indicator = document.getElementById('scene1-step-indicator');

        function showStep(n) {
            currentStep = n;
            // 更新指示器
            indicator.textContent = '步骤 ' + (n + 1) + '/' + totalSteps;
            // 更新按钮状态
            btnPrev.disabled = (n === 0);
            btnNext.disabled = (n === totalSteps - 1);

            // 控制各步骤组的显示/隐藏
            for (var i = 1; i <= 5; i++) {
                var group = document.getElementById('scene1-step-' + i);
                if (!group) continue;
                if (i <= n) {
                    group.style.display = 'block';
                    // 重新触发动画：移除再添加动画类
                    var animated = group.querySelectorAll('.scene1-animate-draw, .scene1-fade-in');
                    animated.forEach(function(el) {
                        var classes = el.classList;
                        if (classes.contains('scene1-animate-draw')) {
                            classes.remove('scene1-animate-draw');
                            void el.offsetWidth; // 强制重排
                            classes.add('scene1-animate-draw');
                        }
                        if (classes.contains('scene1-fade-in')) {
                            classes.remove('scene1-fade-in');
                            void el.offsetWidth;
                            classes.add('scene1-fade-in');
                        }
                    });
                } else {
                    group.style.display = 'none';
                }
            }

            // 特殊处理：步骤5时隐藏步骤4的AB绿色标注，显示红色标注
            var abGreen = document.getElementById('scene1-label-ab-val');
            var abRed = document.getElementById('scene1-label-ab-red');
            if (abGreen && abRed) {
                if (n >= 5) {
                    abGreen.style.display = 'none';
                    abRed.style.display = 'block';
                } else {
                    abGreen.style.display = 'block';
                    abRed.style.display = 'none';
                }
            }
        }

        btnPrev.addEventListener('click', function() {
            if (currentStep > 0) showStep(currentStep - 1);
        });

        btnNext.addEventListener('click', function() {
            if (currentStep < totalSteps - 1) showStep(currentStep + 1);
        });

        // 初始状态
        showStep(0);
    })();
    </script>
