    <script>
    (function() {
        // Scene 1 实现
        var currentStep = 0;
        var totalSteps = 3;

        var btnPrev = document.getElementById('scene1-btn-prev');
        var btnNext = document.getElementById('scene1-btn-next');
        var indicator = document.getElementById('scene1-step-indicator');

        var step0Group = document.getElementById('scene1-step0-group');
        var step1Group = document.getElementById('scene1-step1-group');
        var step2Group = document.getElementById('scene1-step2-group');

        function show(el) {
            el.style.display = 'block';
            // 强制重排后设置 opacity，确保 transition 生效
            requestAnimationFrame(function() {
                el.style.opacity = '1';
            });
        }

        function hide(el) {
            el.style.opacity = '0';
            setTimeout(function() {
                el.style.display = 'none';
            }, 300);
        }

        function renderStep() {
            // 更新指示器
            indicator.textContent = '步骤 ' + (currentStep + 1) + ' / ' + totalSteps;

            // 更新按钮状态
            btnPrev.disabled = (currentStep === 0);
            btnNext.disabled = (currentStep === totalSteps - 1);

            // 控制各组显隐
            if (currentStep >= 1) {
                show(step1Group);
            } else {
                hide(step1Group);
            }
            if (currentStep >= 2) {
                show(step2Group);
            } else {
                hide(step2Group);
            }
        }

        btnPrev.addEventListener('click', function() {
            if (currentStep > 0) {
                currentStep--;
                renderStep();
            }
        });

        btnNext.addEventListener('click', function() {
            if (currentStep < totalSteps - 1) {
                currentStep++;
                renderStep();
            }
        });

        // 初始渲染
        renderStep();
    })();
    </script>
