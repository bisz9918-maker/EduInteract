    <script>
    (function() {
        // Scene 1 实现
        let currentStep = 0;
        const totalSteps = 6;

        const stepTitle = document.getElementById('scene1-step-title');
        const btnPrev = document.getElementById('scene1-btn-prev');
        const btnNext = document.getElementById('scene1-btn-next');

        // 所有可选元素ID
        const allOptionalIds = [
            'scene1-action-f3', 'scene1-action-f2', 'scene1-action-f1', 'scene1-action-f4',
            'scene1-arm-d3', 'scene1-arm-d2', 'scene1-arm-d1', 'scene1-arm-d4',
            'scene1-right-angle',
            'scene1-torque-resist', 'scene1-torque-f3', 'scene1-torque-f1',
            'scene1-oa-label', 'scene1-d3-label', 'scene1-d2-label', 'scene1-d1-label', 'scene1-d4-label',
            'scene1-formula-resist', 'scene1-torque-m3', 'scene1-f1-wrong',
            'scene1-d3-highlight', 'scene1-conclusion'
        ];

        const forceIds = ['scene1-force-f1', 'scene1-force-f2', 'scene1-force-f3', 'scene1-force-f4'];
        const highlightClasses = ['highlight-f1', 'highlight-f2', 'highlight-f3', 'highlight-f4'];

        // 步骤配置
        const stepConfig = [
            { // 步骤0: 题目原图
                title: '题目原图',
                show: [],
                highlight: null
            },
            { // 步骤1: 阻力矩分析
                title: '步骤1：阻力矩分析',
                show: ['scene1-oa-label', 'scene1-torque-resist', 'scene1-formula-resist'],
                highlight: null
            },
            { // 步骤2: F3力臂
                title: '步骤2：F₃ 力臂分析',
                show: ['scene1-action-f3', 'scene1-arm-d3', 'scene1-right-angle', 'scene1-d3-label', 'scene1-torque-f3', 'scene1-torque-m3'],
                highlight: 'scene1-force-f3'
            },
            { // 步骤3: F2力臂
                title: '步骤3：F₂ 力臂分析',
                show: ['scene1-action-f2', 'scene1-arm-d2', 'scene1-d2-label'],
                highlight: 'scene1-force-f2'
            },
            { // 步骤4: F4力臂
                title: '步骤4：F₄ 力臂分析',
                show: ['scene1-action-f4', 'scene1-arm-d4', 'scene1-d4-label'],
                highlight: 'scene1-force-f4'
            },
            { // 步骤5: F1力臂
                title: '步骤5：F₁ 力臂分析',
                show: ['scene1-action-f1', 'scene1-arm-d1', 'scene1-d1-label', 'scene1-torque-f1', 'scene1-f1-wrong'],
                highlight: 'scene1-force-f1'
            },
            { // 步骤6: 结论
                title: '步骤6：结论',
                show: ['scene1-arm-d1', 'scene1-arm-d2', 'scene1-arm-d3', 'scene1-arm-d4',
                       'scene1-d1-label', 'scene1-d2-label', 'scene1-d4-label',
                       'scene1-d3-highlight', 'scene1-conclusion'],
                highlight: null
            }
        ];

        function hideAllOptional() {
            allOptionalIds.forEach(id => {
                const el = document.getElementById(id);
                if (el) el.style.display = 'none';
            });
        }

        function removeAllHighlights() {
            forceIds.forEach((id, idx) => {
                const el = document.getElementById(id);
                if (el) el.classList.remove(highlightClasses[idx]);
            });
        }

        function renderStep() {
            // 隐藏所有可选元素
            hideAllOptional();
            // 移除所有高亮
            removeAllHighlights();

            const config = stepConfig[currentStep];
            stepTitle.textContent = config.title;

            // 显示当前步骤元素
            config.show.forEach(id => {
                const el = document.getElementById(id);
                if (el) el.style.display = 'block';
            });

            // 高亮当前力
            if (config.highlight) {
                const idx = forceIds.indexOf(config.highlight);
                if (idx >= 0) {
                    const el = document.getElementById(config.highlight);
                    if (el) el.classList.add(highlightClasses[idx]);
                }
            }

            // 更新按钮状态
            btnPrev.disabled = currentStep === 0;
            if (currentStep === totalSteps) {
                btnNext.textContent = '完成';
            } else {
                btnNext.textContent = '下一步 ▶';
            }
        }

        btnPrev.addEventListener('click', function() {
            if (currentStep > 0) {
                currentStep--;
                renderStep();
            }
        });

        btnNext.addEventListener('click', function() {
            if (currentStep < totalSteps) {
                currentStep++;
                renderStep();
            }
        });

        // 初始渲染
        renderStep();
    })();
    </script>
