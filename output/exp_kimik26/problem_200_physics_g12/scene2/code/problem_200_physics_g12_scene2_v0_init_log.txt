    <script>
    (function() {
        // Scene 2 实现
        var currentX = 0;
        var showExtreme = false;
        var isDragging = false;
        
        var sliderThumb = document.getElementById('scene2-slider-thumb');
        var sliderValue = document.getElementById('scene2-slider-value');
        var statusHint = document.getElementById('scene2-status-hint');
        var boundaryHint = document.getElementById('scene2-boundary-hint');
        var btnExtreme = document.getElementById('scene2-btn-extreme');
        
        // 装置元素
        var piston = document.getElementById('scene2-piston');
        var mercuryA = document.getElementById('scene2-mercury-a');
        var mercuryB = document.getElementById('scene2-mercury-b');
        var gas = document.getElementById('scene2-gas');
        var arrowF = document.getElementById('scene2-arrow-f');
        var labelF = document.getElementById('scene2-label-f');
        var labelPa = document.getElementById('scene2-label-pa');
        var labelPgas = document.getElementById('scene2-label-pgas');
        
        // 尺寸标注
        var dimH1Line = document.getElementById('scene2-dim-h1-line');
        var dimH1Text = document.getElementById('scene2-dim-h1-text');
        var dimH2Line = document.getElementById('scene2-dim-h2-line');
        var dimH2Text = document.getElementById('scene2-dim-h2-text');
        var dimLLine = document.getElementById('scene2-dim-l-line');
        var dimLText = document.getElementById('scene2-dim-l-text');
        
        // 标记线
        var markX1 = document.getElementById('scene2-mark-x1');
        var markX2 = document.getElementById('scene2-mark-x2');
        var markExtreme = document.getElementById('scene2-mark-extreme');
        var fxMarkExtreme = document.getElementById('scene2-fx-mark-extreme');
        
        // 数据
        var dataH1 = document.getElementById('scene2-data-h1');
        var dataH2 = document.getElementById('scene2-data-h2');
        var dataL = document.getElementById('scene2-data-l');
        var dataPgas = document.getElementById('scene2-data-pgas');
        var dataPa = document.getElementById('scene2-data-pa');
        var dataF = document.getElementById('scene2-data-f');
        var dataFArrow = document.getElementById('scene2-data-f-arrow');
        
        // F-x 曲线
        var fxPath = document.getElementById('scene2-fx-path');
        var fxCurrentPoint = document.getElementById('scene2-fx-current-point');
        
        // 预计算 F-x 曲线
        function computeFxPath() {
            var points = [];
            for (var i = 0; i <= 100; i++) {
                var x = i * 0.025;
                var Lp = 112 + 4 * x;
                var pGas = 10080 / Lp;
                var pA = pGas - 15 + 3 * x;
                var fPerS = 75 - pA;
                var svgX = 510 + 100 * x;
                var svgY = 340 - 300 * fPerS;
                points.push((i === 0 ? 'M' : 'L') + svgX + ',' + svgY);
            }
            return points.join(' ');
        }
        
        fxPath.setAttribute('d', computeFxPath());
        
        function update(x) {
            x = Math.max(0, Math.min(2.5, x));
            currentX = x;
            
            // 物理计算
            var h1 = 5 + x;
            var h2 = Math.max(0, 10 - 4 * x);
            var Lp = 112 + 4 * x;
            var pGas = 10080 / Lp;
            var pA = pGas - h1 - h2;
            var fPerS = 75 - pA;
            
            // 坐标计算 (1 cm = 2 px)
            var pistonY = 154 - 2 * x;
            var mercuryAY = 166 - 2 * x;
            var mercuryAHeight = 10 + 2 * x;
            var mercuryBHeight = Math.max(0, 20 - 8 * x);
            var gasY = 196 - 8 * x;
            var gasHeight = 224 + 8 * x;
            
            // 更新装置
            piston.setAttribute('y', pistonY);
            mercuryA.setAttribute('y', mercuryAY);
            mercuryA.setAttribute('height', mercuryAHeight);
            mercuryB.setAttribute('height', mercuryBHeight);
            gas.setAttribute('y', gasY);
            gas.setAttribute('height', gasHeight);
            
            // 更新 F 箭头
            var arrowColor, arrowEndY;
            if (fPerS > 0.001) {
                arrowColor = '#1565c0';
                arrowEndY = pistonY - 24;
            } else if (fPerS < -0.001) {
                arrowColor = '#c62828';
                arrowEndY = pistonY + 16;
            } else {
                arrowColor = '#888888';
                arrowEndY = pistonY - 24;
            }
            arrowF.setAttribute('stroke', arrowColor);
            arrowF.setAttribute('y1', pistonY);
            arrowF.setAttribute('y2', arrowEndY);
            labelF.setAttribute('fill', arrowColor);
            labelF.setAttribute('y', (pistonY + arrowEndY) / 2 + 4);
            
            // 更新标签位置
            labelPa.setAttribute('y', 171 - x);
            labelPgas.setAttribute('y', 308 - 4 * x);
            
            // 更新尺寸标注
            dimH1Line.setAttribute('y1', 176);
            dimH1Line.setAttribute('y2', mercuryAY);
            dimH1Text.setAttribute('y', (176 + mercuryAY) / 2 + 4);
            
            dimH2Line.setAttribute('y1', 176);
            dimH2Line.setAttribute('y2', 196 - 8 * x);
            dimH2Text.setAttribute('y', (176 + 196 - 8 * x) / 2 + 4);
            
            dimLLine.setAttribute('y1', 196 - 8 * x);
            dimLLine.setAttribute('y2', 420);
            dimLText.setAttribute('y', (196 - 8 * x + 420) / 2 + 4);
            
            // 更新数据面板
            dataH1.textContent = h1.toFixed(1);
            dataH2.textContent = h2.toFixed(1);
            dataL.textContent = Lp.toFixed(1);
            dataPgas.textContent = pGas.toFixed(1);
            dataPa.textContent = pA.toFixed(1);
            dataF.textContent = fPerS.toFixed(2);
            
            // F 趋势箭头
            var trend = 40320 / Math.pow(112 + 4 * x, 2) - 3;
            if (trend > 0.01) {
                dataFArrow.textContent = '↑';
            } else if (trend < -0.01) {
                dataFArrow.textContent = '↓';
            } else {
                dataFArrow.textContent = '→';
            }
            
            // F 颜色
            if (fPerS > 0.001) {
                dataF.setAttribute('fill', '#1565c0');
            } else if (fPerS < -0.001) {
                dataF.setAttribute('fill', '#c62828');
            } else {
                dataF.setAttribute('fill', '#888888');
            }
            
            // 更新 F-x 当前点
            var fxX = 510 + 100 * x;
            var fxY = 340 - 300 * fPerS;
            fxCurrentPoint.setAttribute('cx', fxX);
            fxCurrentPoint.setAttribute('cy', fxY);
            
            // 更新滑块
            var thumbX = 150 + 200 * x;
            sliderThumb.setAttribute('cx', thumbX);
            sliderValue.textContent = 'x = ' + x.toFixed(1);
            
            // 更新标记线高亮
            if (Math.abs(x - 1.0) < 0.05) {
                markX1.setAttribute('opacity', '1');
                markX1.setAttribute('stroke-width', '2');
            } else {
                markX1.setAttribute('opacity', '0.3');
                markX1.setAttribute('stroke-width', '1');
            }
            
            if (Math.abs(x - 2.0) < 0.05) {
                markX2.setAttribute('opacity', '1');
                markX2.setAttribute('stroke-width', '2');
            } else {
                markX2.setAttribute('opacity', '0.3');
                markX2.setAttribute('stroke-width', '1');
            }
            
            // 边界提示
            if (x >= 2.48) {
                boundaryHint.setAttribute('opacity', '1');
            } else {
                boundaryHint.setAttribute('opacity', '0');
            }
        }
        
        // 状态提示动画
        function showStatus(text, color) {
            statusHint.textContent = text;
            statusHint.setAttribute('fill', color);
            statusHint.setAttribute('opacity', '1');
            setTimeout(function() {
                statusHint.setAttribute('opacity', '0');
            }, 1500);
        }
        
        // 吸附逻辑
        function snap() {
            if (Math.abs(currentX - 1.0) < 0.08) {
                animateTo(1.0);
                showStatus('第(2)问状态：x = 1.0 cm', '#1565c0');
            } else if (Math.abs(currentX - 2.0) < 0.08) {
                animateTo(2.0);
                showStatus('第(3)问状态：F = 0', '#c62828');
            }
        }
        
        function animateTo(targetX) {
            var startX = currentX;
            var startTime = Date.now();
            var duration = 200;
            
            function step() {
                var elapsed = Date.now() - startTime;
                var t = Math.min(1, elapsed / duration);
                var eased = 1 - Math.pow(1 - t, 3);
                var newX = startX + (targetX - startX) * eased;
                update(newX);
                if (t < 1) {
                    requestAnimationFrame(step);
                }
            }
            requestAnimationFrame(step);
        }
        
        // 滑块事件
        sliderThumb.addEventListener('mousedown', function(e) {
            isDragging = true;
            e.preventDefault();
        });
        
        sliderThumb.addEventListener('touchstart', function(e) {
            isDragging = true;
            e.preventDefault();
        });
        
        document.addEventListener('mousemove', function(e) {
            if (!isDragging) return;
            var rect = document.getElementById('scene2-svg').getBoundingClientRect();
            var mouseX = e.clientX - rect.left;
            var x = (mouseX - 150) / 200;
            update(x);
        });
        
        document.addEventListener('touchmove', function(e) {
            if (!isDragging) return;
            var rect = document.getElementById('scene2-svg').getBoundingClientRect();
            var mouseX = e.touches[0].clientX - rect.left;
            var x = (mouseX - 150) / 200;
            update(x);
        });
        
        document.addEventListener('mouseup', function() {
            if (isDragging) {
                isDragging = false;
                snap();
            }
        });
        
        document.addEventListener('touchend', function() {
            if (isDragging) {
                isDragging = false;
                snap();
            }
        });
        
        // 极值点按钮
        btnExtreme.addEventListener('click', function() {
            showExtreme = !showExtreme;
            if (showExtreme) {
                btnExtreme.classList.add('active');
                btnExtreme.textContent = '隐藏极值点';
                fxMarkExtreme.setAttribute('opacity', '1');
                markExtreme.setAttribute('opacity', '1');
            } else {
                btnExtreme.classList.remove('active');
                btnExtreme.textContent = '显示极值点';
                fxMarkExtreme.setAttribute('opacity', '0');
                markExtreme.setAttribute('opacity', '0');
            }
        });
        
        // 初始化
        update(0);
    })();
    </script>
