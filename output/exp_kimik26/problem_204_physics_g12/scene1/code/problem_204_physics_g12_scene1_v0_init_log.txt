    <script>
    (function() {
        // Scene 1 实现
        let currentStep = 0;
        const totalSteps = 3;

        // 获取控件元素
        const btnPrev = document.getElementById('scene1-btn-prev');
        const btnNext = document.getElementById('scene1-btn-next');
        const stepText = document.getElementById('scene1-step-text');
        const omegaText = document.getElementById('scene1-omega-text');

        // 获取圆盘区元素
        const massA = document.getElementById('scene1-mass-a');
        const massB = document.getElementById('scene1-mass-b');
        const ropeAB = document.getElementById('scene1-rope-ab');
        const ropeOB = document.getElementById('scene1-rope-ob');

        // 获取A框力元素
        const groupFa = document.getElementById('scene1-group-fa');
        const groupArrowTabA = document.getElementById('scene1-group-arrow-tab-a');
        const labelTabA = document.getElementById('scene1-label-tab-a');
        const textTabAZero = document.getElementById('scene1-text-tab-a-zero');

        // 获取B框力元素
        const groupFb = document.getElementById('scene1-group-fb');
        const groupArrowTob = document.getElementById('scene1-group-arrow-tob');
        const labelTob = document.getElementById('scene1-label-tob');
        const textTobZero = document.getElementById('scene1-text-tob-zero');
        const groupArrowTabB = document.getElementById('scene1-group-arrow-tab-b');
        const labelTabB = document.getElementById('scene1-label-tab-b');

        // 工具函数
        function show(el) {
            el.classList.remove('hidden');
        }
        function hide(el) {
            el.classList.add('hidden');
        }

        function renderStep() {
            // 更新步骤文本
            stepText.textContent = '步骤 ' + currentStep + ' / 3';

            // 更新按钮状态
            btnPrev.disabled = (currentStep === 0);
            btnNext.disabled = (currentStep === totalSteps);
            btnPrev.classList.toggle('disabled', currentStep === 0);
            btnNext.classList.toggle('disabled', currentStep === totalSteps);

            // 重置高亮
            massA.classList.remove('highlight-a');
            massB.classList.remove('highlight-b');

            // 重置绳样式为灰色虚线（无张力）
            ropeAB.setAttribute('stroke', '#888888');
            ropeAB.setAttribute('stroke-width', '1');
            ropeAB.setAttribute('stroke-dasharray', '4,4');
            ropeOB.setAttribute('stroke', '#888888');
            ropeOB.setAttribute('stroke-width', '1');
            ropeOB.setAttribute('stroke-dasharray', '4,4');

            // 隐藏所有力元素和ω值
            hide(groupFa);
            hide(groupArrowTabA);
            hide(labelTabA);
            hide(textTabAZero);
            hide(groupFb);
            hide(groupArrowTob);
            hide(labelTob);
            hide(textTobZero);
            hide(groupArrowTabB);
            hide(labelTabB);
            hide(omegaText);

            // 重置标签文本为默认值
            labelTabA.textContent = 'T_AB = 0';
            labelTob.textContent = 'T_OB = 0';
            labelTabB.textContent = 'T_AB = 0';

            switch(currentStep) {
                case 0:
                    // 初始状态：仅装置可见，所有力隐藏
                    break;

                case 1:
                    // ===== 步骤1：ω₀ 临界（AB绳刚有弹力） =====
                    // 圆盘区：A高亮
                    massA.classList.add('highlight-a');

                    // A框：f_A 出现；T_AB = 0（只显示文本，无箭头）
                    show(groupFa);
                    show(textTabAZero);

                    // ω值
                    omegaText.textContent = 'ω₀ = √(5g/2)';
                    show(omegaText);
                    break;

                case 2:
                    // ===== 步骤2：ω₁ 临界（OB绳刚有弹力） =====
                    // 圆盘区：B高亮；AB绳有张力（蓝色实线）
                    massB.classList.add('highlight-b');
                    ropeAB.setAttribute('stroke', '#1565c0');
                    ropeAB.setAttribute('stroke-width', '2');
                    ropeAB.removeAttribute('stroke-dasharray');

                    // A框：f_A 保持；T_AB 箭头出现
                    show(groupFa);
                    show(groupArrowTabA);
                    labelTabA.textContent = 'T_AB = mg/6';
                    show(labelTabA);

                    // B框：f_B 出现；T_AB(B) 出现；T_OB = 0（只显示文本，无箭头）
                    show(groupFb);
                    show(groupArrowTabB);
                    labelTabB.textContent = 'T_AB = mg/6';
                    show(labelTabB);
                    show(textTobZero);

                    // ω值
                    omegaText.textContent = 'ω₁ = √(10g/3)';
                    show(omegaText);
                    break;

                case 3:
                    // ===== 步骤3：ω₂ 临界（A刚滑动） =====
                    // 圆盘区：A高亮；AB绳蓝色实线；OB绳红色实线
                    massA.classList.add('highlight-a');
                    ropeAB.setAttribute('stroke', '#1565c0');
                    ropeAB.setAttribute('stroke-width', '2');
                    ropeAB.removeAttribute('stroke-dasharray');
                    ropeOB.setAttribute('stroke', '#c62828');
                    ropeOB.setAttribute('stroke-width', '2');
                    ropeOB.removeAttribute('stroke-dasharray');

                    // A框：f_A 保持；T_AB 箭头保持，值更新
                    show(groupFa);
                    show(groupArrowTabA);
                    labelTabA.textContent = 'T_AB = 1.5mg';
                    show(labelTabA);

                    // B框：f_B 保持；T_AB(B) 保持，值更新；T_OB 箭头出现
                    show(groupFb);
                    show(groupArrowTabB);
                    labelTabB.textContent = 'T_AB = 1.5mg';
                    show(labelTabB);
                    show(groupArrowTob);
                    labelTob.textContent = 'T_OB = 2mg';
                    show(labelTob);

                    // ω值
                    omegaText.textContent = 'ω₂ = √(10g)';
                    show(omegaText);
                    break;
            }
        }

        // 事件绑定
        btnPrev.addEventListener('click', function() {
            if (currentStep > 0) {
                currentStep--;
                renderStep();
            }
        });

        btnNext.addEventListener('click', function() {
            if (currentStep < totalSteps) {
                currentStep++;
                renderStep();
            }
        });

        // 初始渲染
        renderStep();
    })();
    </script>
