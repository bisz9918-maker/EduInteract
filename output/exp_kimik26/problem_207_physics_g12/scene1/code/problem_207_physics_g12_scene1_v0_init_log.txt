    <script>
    (function() {
        // Scene 1 实现
        let currentStep = 0;
        
        // 元素引用
        const groupA = document.getElementById('scene1-group-a');
        const groupB = document.getElementById('scene1-group-b');
        const labelA = document.getElementById('scene1-label-a');
        const stepIndicator = document.getElementById('scene1-step-indicator');
        const btnPrev = document.getElementById('scene1-btn-prev');
        const btnNext = document.getElementById('scene1-btn-next');
        
        // 步骤配置：每个步骤定义 A、B、标签的位置，以及需要显示/隐藏的元素
        const stepConfig = [
            // 步骤 0：初始状态
            {
                aTranslate: { x: 0, y: 0 },
                bTranslate: { x: 0, y: 0 },
                labelATranslate: { x: 0, y: 0 },
                show: [],
                hide: ['label-a1', 'arrow-a1', 'label-v0', 'arrow-v0', 'label-aA', 'arrow-aA', 'label-aB', 'arrow-aB', 'label-v', 'arrow-v', 'label-xA', 'disp-xA', 'label-xB', 'disp-xB', 'label-dx']
            },
            // 步骤 1：斜面下滑（A 移动到斜面中点）
            {
                aTranslate: { x: 90, y: 67 },
                bTranslate: { x: 0, y: 0 },
                labelATranslate: { x: 0, y: 0 },
                show: ['label-a1', 'arrow-a1'],
                hide: ['label-v0', 'arrow-v0', 'label-aA', 'arrow-aA', 'label-aB', 'arrow-aB', 'label-v', 'arrow-v', 'label-xA', 'disp-xA', 'label-xB', 'disp-xB', 'label-dx']
            },
            // 步骤 2：到达底端（A 移动到斜面底端）
            {
                aTranslate: { x: 180, y: 135 },
                bTranslate: { x: 0, y: 0 },
                labelATranslate: { x: 0, y: 0 },
                show: ['label-v0', 'arrow-v0'],
                hide: ['label-a1', 'arrow-a1', 'label-aA', 'arrow-aA', 'label-aB', 'arrow-aB', 'label-v', 'arrow-v', 'label-xA', 'disp-xA', 'label-xB', 'disp-xB', 'label-dx']
            },
            // 步骤 3：滑上 B（A 在 B 中部，B 示意性右移）
            {
                aTranslate: { x: 215, y: 145 },
                bTranslate: { x: 8, y: 0 },
                labelATranslate: { x: 250, y: 135 },
                show: ['label-aA', 'arrow-aA', 'label-aB', 'arrow-aB'],
                hide: ['label-a1', 'arrow-a1', 'label-v0', 'arrow-v0', 'label-v', 'arrow-v', 'label-xA', 'disp-xA', 'label-xB', 'disp-xB', 'label-dx']
            },
            // 步骤 4：共速临界（A 在 B 右端，B 右移更多）
            {
                aTranslate: { x: 261, y: 145 },
                bTranslate: { x: 13, y: 0 },
                labelATranslate: { x: 296, y: 135 },
                show: ['label-v', 'arrow-v', 'label-xA', 'disp-xA', 'label-xB', 'disp-xB', 'label-dx'],
                hide: ['label-a1', 'arrow-a1', 'label-v0', 'arrow-v0', 'label-aA', 'arrow-aA', 'label-aB', 'arrow-aB']
            }
        ];
        
        function setTranslate(el, x, y) {
            el.style.transform = 'translate(' + x + 'px, ' + y + 'px)';
        }
        
        function setOpacity(id, value) {
            const el = document.getElementById('scene1-' + id);
            if (el) el.style.opacity = value;
        }
        
        function goToStep(n) {
            if (n < 0 || n > 4) return;
            currentStep = n;
            
            const config = stepConfig[n];
            
            // 移动 A、B 和标签
            setTranslate(groupA, config.aTranslate.x, config.aTranslate.y);
            setTranslate(groupB, config.bTranslate.x, config.bTranslate.y);
            setTranslate(labelA, config.labelATranslate.x, config.labelATranslate.y);
            
            // 显示元素
            config.show.forEach(function(id) {
                setOpacity(id, '1');
            });
            
            // 隐藏元素
            config.hide.forEach(function(id) {
                setOpacity(id, '0');
            });
            
            // 更新控件状态
            stepIndicator.textContent = '步骤 ' + n + '/4';
            btnPrev.disabled = (n === 0);
            btnNext.disabled = (n === 4);
        }
        
        // 事件绑定
        btnPrev.addEventListener('click', function() {
            goToStep(currentStep - 1);
        });
        btnNext.addEventListener('click', function() {
            goToStep(currentStep + 1);
        });
        
        // 初始化
        goToStep(0);
    })();
    </script>
