    <script>
    (function() {
        // Scene 2 隔离体受力分析 - 分步导航
        var currentStep = 0;
        var totalSteps = 3;
        var stepTitles = [
            "步骤 1/3：A 在斜面上的受力",
            "步骤 2/3：A 在 B 上表面滑行（隔离 A）",
            "步骤 3/3：木板 B 的受力分析"
        ];

        var btnPrev = document.getElementById("scene2-btn-prev");
        var btnNext = document.getElementById("scene2-btn-next");
        var stepIndicator = document.getElementById("scene2-step-indicator");
        var stepTitle = document.getElementById("scene2-step-title");
        var groups = [
            document.getElementById("scene2-step1-group"),
            document.getElementById("scene2-step2-group"),
            document.getElementById("scene2-step3-group")
        ];

        function showStep(index) {
            // 更新标题
            stepTitle.textContent = stepTitles[index];
            stepIndicator.textContent = (index + 1) + "/" + totalSteps;

            // 更新按钮状态
            btnPrev.disabled = (index === 0);
            btnNext.disabled = (index === totalSteps - 1);

            // 切换分组显示（带淡入淡出）
            for (var i = 0; i < groups.length; i++) {
                if (i === index) {
                    groups[i].style.display = "block";
                    // 强制重排后设置opacity实现过渡
                    requestAnimationFrame(function(g) {
                        return function() {
                            g.style.opacity = "1";
                        };
                    }(groups[i]));
                } else {
                    groups[i].style.opacity = "0";
                    // 等待过渡完成后隐藏
                    (function(g) {
                        setTimeout(function() {
                            if (g.style.opacity === "0") {
                                g.style.display = "none";
                            }
                        }, 300);
                    })(groups[i]);
                }
            }
        }

        btnPrev.addEventListener("click", function() {
            if (currentStep > 0) {
                currentStep--;
                showStep(currentStep);
            }
        });

        btnNext.addEventListener("click", function() {
            if (currentStep < totalSteps - 1) {
                currentStep++;
                showStep(currentStep);
            }
        });

        // 初始化
        showStep(0);
    })();
    </script>
