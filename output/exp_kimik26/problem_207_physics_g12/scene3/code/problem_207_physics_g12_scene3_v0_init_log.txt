    <script>
    (function() {
        // Scene 3 实现
        var currentStep = 0;
        var totalSteps = 5;

        var btnPrev = document.getElementById('scene3-btn-prev');
        var btnNext = document.getElementById('scene3-btn-next');
        var stepIndicator = document.getElementById('scene3-step-indicator');

        // 各步骤控制的元素
        var stepElements = [
            // 步骤0: 坐标系 + v₀=6 (始终显示)
            {
                show: [],
                hide: ['scene3-line-a','scene3-label-a','scene3-aa-label',
                       'scene3-line-b','scene3-label-b','scene3-ab-label',
                       'scene3-coincident-point','scene3-dash-h','scene3-dash-v','scene3-t1-label','scene3-v1-label',
                       'scene3-area-a','scene3-xa-label','scene3-area-b','scene3-xb-label','scene3-area-delta','scene3-dx-label',
                       'scene3-legend']
            },
            // 步骤1: + A图线
            {
                show: ['scene3-line-a','scene3-label-a','scene3-aa-label'],
                hide: ['scene3-line-b','scene3-label-b','scene3-ab-label',
                       'scene3-coincident-point','scene3-dash-h','scene3-dash-v','scene3-t1-label','scene3-v1-label',
                       'scene3-area-a','scene3-xa-label','scene3-area-b','scene3-xb-label','scene3-area-delta','scene3-dx-label',
                       'scene3-legend']
            },
            // 步骤2: + B图线
            {
                show: ['scene3-line-a','scene3-label-a','scene3-aa-label',
                       'scene3-line-b','scene3-label-b','scene3-ab-label'],
                hide: ['scene3-coincident-point','scene3-dash-h','scene3-dash-v','scene3-t1-label','scene3-v1-label',
                       'scene3-area-a','scene3-xa-label','scene3-area-b','scene3-xb-label','scene3-area-delta','scene3-dx-label',
                       'scene3-legend']
            },
            // 步骤3: + 共速点
            {
                show: ['scene3-line-a','scene3-label-a','scene3-aa-label',
                       'scene3-line-b','scene3-label-b','scene3-ab-label',
                       'scene3-coincident-point','scene3-dash-h','scene3-dash-v','scene3-t1-label','scene3-v1-label'],
                hide: ['scene3-area-a','scene3-xa-label','scene3-area-b','scene3-xb-label','scene3-area-delta','scene3-dx-label',
                       'scene3-legend']
            },
            // 步骤4: + 面积与图例
            {
                show: ['scene3-line-a','scene3-label-a','scene3-aa-label',
                       'scene3-line-b','scene3-label-b','scene3-ab-label',
                       'scene3-coincident-point','scene3-dash-h','scene3-dash-v','scene3-t1-label','scene3-v1-label',
                       'scene3-area-a','scene3-xa-label','scene3-area-b','scene3-xb-label','scene3-area-delta','scene3-dx-label',
                       'scene3-legend'],
                hide: []
            }
        ];

        function setVisible(id, visible) {
            var el = document.getElementById(id);
            if (!el) return;
            if (visible) {
                el.style.display = '';
                // 对于 fade-in 元素，添加 visible 类触发 transition
                if (el.classList.contains('scene3-fade-in')) {
                    // 强制重排确保 transition 生效
                    void el.offsetWidth;
                    el.classList.add('visible');
                }
                if (el.classList.contains('scene3-line-a') || el.classList.contains('scene3-line-b')) {
                    el.classList.add('visible');
                }
            } else {
                if (el.classList.contains('scene3-fade-in')) {
                    el.classList.remove('visible');
                }
                if (el.classList.contains('scene3-line-a') || el.classList.contains('scene3-line-b')) {
                    el.classList.remove('visible');
                }
                el.style.display = 'none';
            }
        }

        function updateScene() {
            var cfg = stepElements[currentStep];
            // 先隐藏
            cfg.hide.forEach(function(id) { setVisible(id, false); });
            // 再显示
            cfg.show.forEach(function(id) { setVisible(id, true); });

            stepIndicator.textContent = '步骤 ' + (currentStep + 1) + ' / ' + totalSteps;
            btnPrev.disabled = (currentStep === 0);
            btnNext.disabled = (currentStep === totalSteps - 1);
        }

        btnPrev.addEventListener('click', function() {
            if (currentStep > 0) {
                currentStep--;
                updateScene();
            }
        });

        btnNext.addEventListener('click', function() {
            if (currentStep < totalSteps - 1) {
                currentStep++;
                updateScene();
            }
        });

        // 初始化
        updateScene();
    })();
    </script>
