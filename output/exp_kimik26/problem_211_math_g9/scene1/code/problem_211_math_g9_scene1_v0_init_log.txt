    <script>
    (function() {
        // Scene 1 实现
        var currentStep = 0;
        var maxStep = 5;

        var btnPrev = document.getElementById('scene1-btn-prev');
        var btnNext = document.getElementById('scene1-btn-next');
        var stepText = document.getElementById('scene1-step-text');
        var judgeText = document.getElementById('scene1-judge-text');

        // 获取所有需要控制的元素
        var lineAB = document.getElementById('scene1-line-ab');
        var lineCD = document.getElementById('scene1-line-cd');
        var lineAE = document.getElementById('scene1-line-ae');
        var lineCF = document.getElementById('scene1-line-cf');
        var fillABE = document.getElementById('scene1-fill-abe');
        var fillCDF = document.getElementById('scene1-fill-cdf');

        var arcABE = document.getElementById('scene1-arc-abe');
        var arcCDF = document.getElementById('scene1-arc-cdf');
        var arcBAE = document.getElementById('scene1-arc-bae');
        var arcDCF = document.getElementById('scene1-arc-dcf');

        var labelAngle1 = document.getElementById('scene1-label-angle1');
        var labelAngle2 = document.getElementById('scene1-label-angle2');
        var labelEF = document.getElementById('scene1-label-ef');

        var tickAB = document.getElementById('scene1-tick-ab');
        var tickCD = document.getElementById('scene1-tick-cd');
        var tickBE = document.getElementById('scene1-tick-be');
        var tickFD = document.getElementById('scene1-tick-fd');
        var tickBF = document.getElementById('scene1-tick-bf');
        var tickDE = document.getElementById('scene1-tick-de');
        var tickAE = document.getElementById('scene1-tick-ae');
        var tickCF = document.getElementById('scene1-tick-cf');

        // HTML 选项框
        var optA = document.getElementById('scene1-option-a-div');
        var optB = document.getElementById('scene1-option-b-div');
        var optC = document.getElementById('scene1-option-c-div');
        var optD = document.getElementById('scene1-option-d-div');

        function setOpacity(el, val) { if(el) el.setAttribute('opacity', val); }
        function setStroke(el, val) { if(el) el.setAttribute('stroke', val); }
        function setStrokeWidth(el, val) { if(el) el.setAttribute('stroke-width', val); }
        function setFill(el, val) { if(el) el.setAttribute('fill', val); }
        function setText(el, val) { if(el) el.textContent = val; }

        function clearOptionHighlights() {
            optA.classList.remove('scene1-option-green', 'scene1-option-red');
            optB.classList.remove('scene1-option-green', 'scene1-option-red');
            optC.classList.remove('scene1-option-green', 'scene1-option-red');
            optD.classList.remove('scene1-option-green', 'scene1-option-red');
        }

        function resetToBase() {
            // 恢复所有线条为黑色
            setStroke(lineAB, '#000000');
            setStroke(lineCD, '#000000');
            setStroke(lineAE, '#000000');
            setStroke(lineCF, '#000000');
            setStrokeWidth(lineAB, '2');
            setStrokeWidth(lineCD, '2');
            setStrokeWidth(lineAE, '2');
            setStrokeWidth(lineCF, '2');

            // 隐藏所有额外元素
            setOpacity(arcABE, '0');
            setOpacity(arcCDF, '0');
            setOpacity(arcBAE, '0');
            setOpacity(arcDCF, '0');
            setOpacity(labelAngle1, '0');
            setOpacity(labelAngle2, '0');
            setOpacity(labelEF, '0');
            setOpacity(tickAB, '0');
            setOpacity(tickCD, '0');
            setOpacity(tickBE, '0');
            setOpacity(tickFD, '0');
            setOpacity(tickBF, '0');
            setOpacity(tickDE, '0');
            setOpacity(tickAE, '0');
            setOpacity(tickCF, '0');
            setOpacity(fillABE, '0');
            setOpacity(fillCDF, '0');

            // 恢复填充颜色
            setFill(fillABE, '#2e7d32');
            setFill(fillCDF, '#2e7d32');

            // 清除选项高亮
            clearOptionHighlights();

            // 恢复判定文字颜色
            judgeText.setAttribute('fill', '#000000');
        }

        function updateScene(step) {
            resetToBase();

            if (step === 0) {
                setText(judgeText, '');
                setText(stepText, '初始');
            } else if (step === 1) {
                // 步骤1: 已知条件与隐含条件
                setStroke(lineAB, '#1565c0');
                setStroke(lineCD, '#1565c0');
                setStrokeWidth(lineAB, '2.5');
                setStrokeWidth(lineCD, '2.5');
                setOpacity(tickAB, '1');
                setOpacity(tickCD, '1');
                setOpacity(arcABE, '1');
                setOpacity(arcCDF, '1');
                setText(judgeText, '已知：AB = CD，AB ∥ CD → 内错角 ∠ABE = ∠CDF');
                setText(stepText, '步骤 1/5');
            } else if (step === 2) {
                // 步骤2: 选项B - BE = FD, SAS
                setStroke(lineAB, '#1565c0');
                setStroke(lineCD, '#1565c0');
                setStrokeWidth(lineAB, '2.5');
                setStrokeWidth(lineCD, '2.5');
                setOpacity(tickAB, '1');
                setOpacity(tickCD, '1');
                setOpacity(arcABE, '1');
                setOpacity(arcCDF, '1');

                // BE, FD 等长标记
                setOpacity(tickBE, '1');
                setOpacity(tickFD, '1');

                // 三角形填充浅绿
                setOpacity(fillABE, '0.2');
                setOpacity(fillCDF, '0.2');

                // 选项B高亮
                optB.classList.add('scene1-option-green');

                setText(judgeText, 'SAS：两边及其夹角对应相等 → △ABE ≌ △CDF');
                setText(stepText, '步骤 2/5');
            } else if (step === 3) {
                // 步骤3: 选项C - BF = DE, 转化为 BE = FD
                setStroke(lineAB, '#1565c0');
                setStroke(lineCD, '#1565c0');
                setStrokeWidth(lineAB, '2.5');
                setStrokeWidth(lineCD, '2.5');
                setOpacity(tickAB, '1');
                setOpacity(tickCD, '1');
                setOpacity(arcABE, '1');
                setOpacity(arcCDF, '1');

                // 保留 BE=FD 标记（推导结论）
                setOpacity(tickBE, '1');
                setOpacity(tickFD, '1');

                // BF, DE 绿色高亮
                setOpacity(tickBF, '1');
                setOpacity(tickDE, '1');

                // EF 标注
                setOpacity(labelEF, '1');

                // 三角形填充浅绿
                setOpacity(fillABE, '0.2');
                setOpacity(fillCDF, '0.2');

                // 选项C高亮
                optC.classList.add('scene1-option-green');

                setText(judgeText, 'BF = DE → BF − EF = DE − EF → BE = FD → SAS');
                setText(stepText, '步骤 3/5');
            } else if (step === 4) {
                // 步骤4: 选项D - ∠1 = ∠2, ASA
                setStroke(lineAB, '#1565c0');
                setStroke(lineCD, '#1565c0');
                setStrokeWidth(lineAB, '2.5');
                setStrokeWidth(lineCD, '2.5');
                setOpacity(tickAB, '1');
                setOpacity(tickCD, '1');
                setOpacity(arcABE, '1');
                setOpacity(arcCDF, '1');

                // 保留 BE=FD 标记
                setOpacity(tickBE, '1');
                setOpacity(tickFD, '1');

                // ∠1, ∠2 角弧线
                setOpacity(arcBAE, '1');
                setOpacity(arcDCF, '1');
                setOpacity(labelAngle1, '1');
                setOpacity(labelAngle2, '1');

                // 三角形填充浅绿
                setOpacity(fillABE, '0.2');
                setOpacity(fillCDF, '0.2');

                // 选项D高亮
                optD.classList.add('scene1-option-green');

                setText(judgeText, 'ASA：两角及其夹边对应相等 → △ABE ≌ △CDF');
                setText(stepText, '步骤 4/5');
            } else if (step === 5) {
                // 步骤5: 选项A - AE = CF, SSA不能判定
                setStroke(lineAB, '#1565c0');
                setStroke(lineCD, '#1565c0');
                setStrokeWidth(lineAB, '2.5');
                setStrokeWidth(lineCD, '2.5');
                setOpacity(tickAB, '1');
                setOpacity(tickCD, '1');
                setOpacity(arcABE, '1');
                setOpacity(arcCDF, '1');

                // AE, CF 红色高亮
                setStroke(lineAE, '#c62828');
                setStroke(lineCF, '#c62828');
                setStrokeWidth(lineAE, '2.5');
                setStrokeWidth(lineCF, '2.5');
                setOpacity(tickAE, '1');
                setOpacity(tickCF, '1');

                // 三角形填充浅红
                setFill(fillABE, '#c62828');
                setFill(fillCDF, '#c62828');
                setOpacity(fillABE, '0.2');
                setOpacity(fillCDF, '0.2');

                // 选项A高亮
                optA.classList.add('scene1-option-red');

                // 判定文字红色
                judgeText.setAttribute('fill', '#c62828');
                setText(judgeText, 'SSA 不能判定全等 ⚠');
                setText(stepText, '步骤 5/5');
            }

            // 更新按钮状态
            btnPrev.disabled = (step === 0);
            btnNext.disabled = (step === maxStep);
        }

        btnPrev.addEventListener('click', function() {
            if (currentStep > 0) {
                currentStep--;
                updateScene(currentStep);
            }
        });

        btnNext.addEventListener('click', function() {
            if (currentStep < maxStep) {
                currentStep++;
                updateScene(currentStep);
            }
        });

        // 初始化
        updateScene(0);
    })();
    </script>
