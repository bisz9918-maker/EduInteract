    <script>
    (function() {
        // Scene 1 实现：自行车链条长度计算分步导航
        let currentStep = 0;
        const totalSteps = 5;

        const prevBtn = document.getElementById('scene1-btn-prev');
        const nextBtn = document.getElementById('scene1-btn-next');
        const indicator = document.getElementById('scene1-step-indicator');

        function updateDisplay() {
            // 隐藏所有步骤（同时设置 opacity 和 display，避免 getBoundingClientRect 仍返回非零值）
            for (let i = 0; i < totalSteps; i++) {
                const group = document.getElementById('scene1-step-' + i);
                if (group) {
                    group.classList.remove('active');
                    group.style.display = 'none';
                }
            }
            // 显示当前步骤
            const currentGroup = document.getElementById('scene1-step-' + currentStep);
            if (currentGroup) {
                currentGroup.style.display = 'block';
                // 强制重排确保 display 生效后再加 active 类触发 opacity 过渡
                void currentGroup.offsetWidth;
                currentGroup.classList.add('active');
            }

            // 更新按钮状态
            prevBtn.disabled = currentStep === 0;
            nextBtn.disabled = currentStep === totalSteps - 1;
            if (currentStep === totalSteps - 1) {
                nextBtn.textContent = '完成';
            } else {
                nextBtn.textContent = '下一步 ▶';
            }

            // 更新指示器
            indicator.textContent = '步骤 ' + (currentStep + 1) + '/5';
        }

        prevBtn.addEventListener('click', function() {
            if (currentStep > 0) {
                currentStep--;
                updateDisplay();
            }
        });

        nextBtn.addEventListener('click', function() {
            if (currentStep < totalSteps - 1) {
                currentStep++;
                updateDisplay();
            }
        });

        // 初始化
        updateDisplay();
    })();
    </script>
