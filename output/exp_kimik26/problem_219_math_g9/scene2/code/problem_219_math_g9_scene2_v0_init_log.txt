    <script>
    (function() {
        // Scene 2 实现
        // 坐标映射系统：
        // dataToPixel(x, y): px = 80 + x * 80, py = 420 - y * 60
        // pixelToData(px, py): x = (px - 80) / 80, y = (420 - py) / 60

        const ORIGIN_X = 80, ORIGIN_Y = 420;
        const SCALE_X = 80, SCALE_Y = 60;
        const MAX_Y_DATA = 6;
        const MIN_X_DATA = 0.1;

        let k = 4;
        let pointA = { x: 1, y: 4 };
        let pointB = { x: 4, y: 1 };
        let dragging = null; // 'A' or 'B' or 'slider'
        let sliderMinK = 1, sliderMaxK = 8;

        // DOM 元素引用 - SVG
        const svg = document.getElementById('scene2-svg');
        const elCurve = document.getElementById('scene2-curve-path');
        const elTrap = document.getElementById('scene2-trapezoid-abcd');
        const elTriOAB = document.getElementById('scene2-triangle-oab');
        const elTriOAD = document.getElementById('scene2-triangle-oad');
        const elTriOBC = document.getElementById('scene2-triangle-obc');
        const elLineAD = document.getElementById('scene2-line-ad');
        const elLineBC = document.getElementById('scene2-line-bc');
        const elLineAB = document.getElementById('scene2-line-ab');
        const elLineOA = document.getElementById('scene2-line-oa');
        const elLineOB = document.getElementById('scene2-line-ob');
        const elPointA = document.getElementById('scene2-point-a');
        const elPointB = document.getElementById('scene2-point-b');
        const elPointC = document.getElementById('scene2-point-c');
        const elPointD = document.getElementById('scene2-point-d');
        const elValKSlider = document.getElementById('scene2-value-kslider');
        const elSliderThumb = document.getElementById('scene2-slider-thumb');

        // DOM 元素引用 - HTML 标签
        const elLabelA = document.getElementById('scene2-html-label-a');
        const elLabelB = document.getElementById('scene2-html-label-b');
        const elLabelC = document.getElementById('scene2-html-label-c');
        const elLabelD = document.getElementById('scene2-html-label-d');

        // DOM 元素引用 - HTML 面板
        const elPanelS1 = document.getElementById('scene2-panel-s1');
        const elPanelS2 = document.getElementById('scene2-panel-s2');
        const elPanelSOAD = document.getElementById('scene2-panel-soad');
        const elPanelSOBC = document.getElementById('scene2-panel-sobc');
        const elPanelRatio = document.getElementById('scene2-panel-ratio');
        const elPanelK = document.getElementById('scene2-panel-k');
        const elBtnReset = document.getElementById('scene2-btn-reset');

        function dataToPixel(dx, dy) {
            return {
                x: ORIGIN_X + dx * SCALE_X,
                y: ORIGIN_Y - dy * SCALE_Y
            };
        }

        function pixelToData(px, py) {
            return {
                x: (px - ORIGIN_X) / SCALE_X,
                y: (ORIGIN_Y - py) / SCALE_Y
            };
        }

        function updateCurve() {
            const minX = Math.max(MIN_X_DATA, k / MAX_Y_DATA);
            const maxX = 5.5;
            const steps = 60;
            let d = '';
            for (let i = 0; i <= steps; i++) {
                const t = i / steps;
                const dx = minX + (maxX - minX) * t;
                const dy = k / dx;
                const p = dataToPixel(dx, dy);
                if (i === 0) d += `M ${p.x} ${p.y}`;
                else d += ` L ${p.x} ${p.y}`;
            }
            elCurve.setAttribute('d', d);
        }

        function updateGraphics() {
            const pA = dataToPixel(pointA.x, pointA.y);
            const pB = dataToPixel(pointB.x, pointB.y);
            const pO = dataToPixel(0, 0);
            const pD = dataToPixel(0, pointA.y);
            const pC = dataToPixel(0, pointB.y);

            // 更新点位置
            elPointA.setAttribute('cx', pA.x);
            elPointA.setAttribute('cy', pA.y);
            elPointB.setAttribute('cx', pB.x);
            elPointB.setAttribute('cy', pB.y);
            elPointC.setAttribute('cx', pC.x);
            elPointC.setAttribute('cy', pC.y);
            elPointD.setAttribute('cx', pD.x);
            elPointD.setAttribute('cy', pD.y);

            // 更新 HTML 标签位置
            // A 在 y 轴左侧，B 在曲线右侧，C/D 在 y 轴左侧
            elLabelA.style.left = '55px';
            elLabelA.style.top = (pA.y - 30) + 'px';
            elLabelB.style.left = '525px';
            elLabelB.style.top = (pB.y - 30) + 'px';
            elLabelC.style.left = '55px';
            elLabelC.style.top = (pC.y - 10) + 'px';
            elLabelD.style.left = '55px';
            elLabelD.style.top = (pD.y - 10) + 'px';

            // 更新线段
            elLineAD.setAttribute('x1', pD.x);
            elLineAD.setAttribute('y1', pD.y);
            elLineAD.setAttribute('x2', pA.x);
            elLineAD.setAttribute('y2', pA.y);

            elLineBC.setAttribute('x1', pC.x);
            elLineBC.setAttribute('y1', pC.y);
            elLineBC.setAttribute('x2', pB.x);
            elLineBC.setAttribute('y2', pB.y);

            elLineAB.setAttribute('x1', pA.x);
            elLineAB.setAttribute('y1', pA.y);
            elLineAB.setAttribute('x2', pB.x);
            elLineAB.setAttribute('y2', pB.y);

            elLineOA.setAttribute('x1', pO.x);
            elLineOA.setAttribute('y1', pO.y);
            elLineOA.setAttribute('x2', pA.x);
            elLineOA.setAttribute('y2', pA.y);

            elLineOB.setAttribute('x1', pO.x);
            elLineOB.setAttribute('y1', pO.y);
            elLineOB.setAttribute('x2', pB.x);
            elLineOB.setAttribute('y2', pB.y);

            // 更新多边形
            elTrap.setAttribute('points', `${pD.x},${pD.y} ${pC.x},${pC.y} ${pB.x},${pB.y} ${pA.x},${pA.y}`);
            elTriOAB.setAttribute('points', `${pO.x},${pO.y} ${pA.x},${pA.y} ${pB.x},${pB.y}`);
            elTriOAD.setAttribute('points', `${pO.x},${pO.y} ${pA.x},${pA.y} ${pD.x},${pD.y}`);
            elTriOBC.setAttribute('points', `${pO.x},${pO.y} ${pB.x},${pB.y} ${pC.x},${pC.y}`);

            // 计算面积
            const s1 = (pointA.x + pointB.x) * (pointA.y - pointB.y) / 2;
            const s2 = Math.abs(pointA.x * pointB.y - pointB.x * pointA.y) / 2;
            const sOAD = pointA.x * pointA.y / 2;
            const sOBC = pointB.x * pointB.y / 2;

            // 更新 HTML 面板
            elPanelS1.textContent = s1.toFixed(2);
            elPanelS2.textContent = s2.toFixed(2);
            elPanelSOAD.textContent = sOAD.toFixed(2);
            elPanelSOBC.textContent = sOBC.toFixed(2);
            elPanelRatio.textContent = '1 : 1';
            elPanelK.textContent = k;
            elValKSlider.textContent = k;
        }

        function updateSlider() {
            const trackX = 150, trackW = 150;
            const t = (k - sliderMinK) / (sliderMaxK - sliderMinK);
            const cx = trackX + t * trackW;
            elSliderThumb.setAttribute('cx', cx);
        }

        function constrainPointA(newX) {
            const minX = Math.max(0.8, k / MAX_Y_DATA + 0.05);
            const maxX = pointB.x - 0.3;
            let x = Math.max(minX, Math.min(maxX, newX));
            return { x: x, y: k / x };
        }

        function constrainPointB(newX) {
            const minX = pointA.x + 0.3;
            const maxX = 5.5;
            let x = Math.max(minX, Math.min(maxX, newX));
            return { x: x, y: k / x };
        }

        function adjustPointsForK() {
            const maxY = MAX_Y_DATA;
            if (k / pointA.x > maxY) {
                pointA.x = Math.max(k / maxY, 0.8);
            }
            pointA.y = k / pointA.x;
            if (k / pointB.x > maxY) {
                pointB.x = Math.max(k / maxY, pointA.x + 0.3);
            }
            pointB.y = k / pointB.x;
            if (pointB.x > 5.5) {
                pointB.x = 5.5;
                pointB.y = k / 5.5;
            }
            if (pointA.x >= pointB.x - 0.3) {
                pointA.x = Math.max(0.8, pointB.x - 0.5);
                pointA.y = k / pointA.x;
            }
        }

        function onMouseDown(e) {
            const target = e.target;
            if (target === elPointA) {
                dragging = 'A';
                elPointA.setAttribute('r', '6');
                elPointA.style.cursor = 'grabbing';
                e.preventDefault();
            } else if (target === elPointB) {
                dragging = 'B';
                elPointB.setAttribute('r', '6');
                elPointB.style.cursor = 'grabbing';
                e.preventDefault();
            } else if (target === elSliderThumb) {
                dragging = 'slider';
                e.preventDefault();
            }
        }

        function onMouseMove(e) {
            if (!dragging) return;
            const rect = svg.getBoundingClientRect();
            const px = e.clientX - rect.left;
            const py = e.clientY - rect.top;

            if (dragging === 'A') {
                const data = pixelToData(px, py);
                pointA = constrainPointA(data.x);
                updateGraphics();
            } else if (dragging === 'B') {
                const data = pixelToData(px, py);
                pointB = constrainPointB(data.x);
                updateGraphics();
            } else if (dragging === 'slider') {
                const trackX = 150, trackW = 150;
                let t = (px - trackX) / trackW;
                t = Math.max(0, Math.min(1, t));
                k = Math.round(sliderMinK + t * (sliderMaxK - sliderMinK));
                adjustPointsForK();
                updateCurve();
                updateGraphics();
                updateSlider();
            }
        }

        function onMouseUp() {
            if (dragging === 'A') {
                elPointA.setAttribute('r', '4');
                elPointA.style.cursor = 'grab';
            } else if (dragging === 'B') {
                elPointB.setAttribute('r', '4');
                elPointB.style.cursor = 'grab';
            }
            dragging = null;
        }

        function reset() {
            k = 4;
            pointA = { x: 1, y: 4 };
            pointB = { x: 4, y: 1 };
            updateCurve();
            updateGraphics();
            updateSlider();
        }

        // 事件绑定
        svg.addEventListener('mousedown', onMouseDown);
        svg.addEventListener('mousemove', onMouseMove);
        svg.addEventListener('mouseup', onMouseUp);
        svg.addEventListener('mouseleave', onMouseUp);

        // 触摸支持
        svg.addEventListener('touchstart', function(e) {
            if (e.touches.length === 1) {
                const touch = e.touches[0];
                const target = document.elementFromPoint(touch.clientX, touch.clientY);
                if (target === elPointA) {
                    dragging = 'A';
                    elPointA.setAttribute('r', '6');
                    e.preventDefault();
                } else if (target === elPointB) {
                    dragging = 'B';
                    elPointB.setAttribute('r', '6');
                    e.preventDefault();
                } else if (target === elSliderThumb) {
                    dragging = 'slider';
                    e.preventDefault();
                }
            }
        }, { passive: false });

        svg.addEventListener('touchmove', function(e) {
            if (dragging && e.touches.length === 1) {
                e.preventDefault();
                const touch = e.touches[0];
                const rect = svg.getBoundingClientRect();
                const px = touch.clientX - rect.left;
                const py = touch.clientY - rect.top;
                if (dragging === 'A') {
                    const data = pixelToData(px, py);
                    pointA = constrainPointA(data.x);
                    updateGraphics();
                } else if (dragging === 'B') {
                    const data = pixelToData(px, py);
                    pointB = constrainPointB(data.x);
                    updateGraphics();
                } else if (dragging === 'slider') {
                    const trackX = 150, trackW = 150;
                    let t = (px - trackX) / trackW;
                    t = Math.max(0, Math.min(1, t));
                    k = Math.round(sliderMinK + t * (sliderMaxK - sliderMinK));
                    adjustPointsForK();
                    updateCurve();
                    updateGraphics();
                    updateSlider();
                }
            }
        }, { passive: false });

        svg.addEventListener('touchend', onMouseUp);
        svg.addEventListener('touchcancel', onMouseUp);

        // 重置按钮事件
        elBtnReset.addEventListener('click', reset);

        // 初始化
        updateCurve();
        updateGraphics();
        updateSlider();
    })();
    </script>
</body>
