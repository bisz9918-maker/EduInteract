<script>
(function(){
var currentStep=0;
var totalSteps=7;
var stepTitles=["初始状态","晨昏线判读","划分昼夜半球","节气判断","太阳直射点定位","地方时标注","北京时间计算","正午太阳高度与线速度"];
var stepElements={
1:["scene1-step1-arrow","scene1-step1-arrow-text","scene1-step1-legend-div","scene1-step1-ed-highlight","scene1-step1-df-highlight"],
2:["scene1-step2-night-fill","scene1-step2-polar-day","scene1-step2-polar-night"],
3:["scene1-step3-solar-term","scene1-step3-tropic-highlight"],
4:["scene1-step4-prime-meridian","scene1-step4-direct-star","scene1-step4-direct-label"],
5:["scene1-step5-time-a","scene1-step5-time-b","scene1-step5-time-c","scene1-step5-time-d","scene1-step5-time-e","scene1-step5-time-f"],
6:["scene1-step6-longitude-120","scene1-step6-arrow","scene1-step6-arrow-text","scene1-step6-bjt-label"],
7:["scene1-step7-formula-div"]
};
var btnPrev=document.getElementById("scene1-btn-prev");
var btnNext=document.getElementById("scene1-btn-next");
var stepText=document.getElementById("scene1-step-text");
function updateScene(step){
for(var i=1;i<=totalSteps;i++){
var ids=stepElements[i]||[];
ids.forEach(function(id){
var el=document.getElementById(id);
if(el){el.classList.remove("scene1-active");}
});
}
for(var i=1;i<=step;i++){
var ids=stepElements[i]||[];
ids.forEach(function(id){
var el=document.getElementById(id);
if(el){el.classList.add("scene1-active");}
});
}
btnPrev.disabled=(step<=0);
btnNext.disabled=(step>=totalSteps);
btnNext.textContent=(step>=totalSteps)?"完成":"下一步 ▶";
stepText.textContent="步骤 "+step+"/"+totalSteps+"："+stepTitles[step];
}
btnPrev.addEventListener("click",function(){
if(currentStep>0){currentStep--;updateScene(currentStep);}
});
btnNext.addEventListener("click",function(){
if(currentStep<totalSteps){currentStep++;updateScene(currentStep);}
});
updateScene(0);
})();
</script>
