    <script>
    (function() {
        // Scene 2 实现
        let currentStep = 0;
        const totalSteps = 6;
        let animating = false;
        let animReqId = null;

        // 贝塞尔曲线控制点
        const p0 = { x: 250, y: 280 };
        const p1 = { x: 400, y: 50 };
        const p2 = { x: 550, y: 280 };

        // 获取元素
        const btnPrev = document.getElementById('scene2-btn-prev');
        const btnNext = document.getElementById('scene2-btn-next');
        const stepIndicator = document.getElementById('scene2-step-indicator');

        const elLat20Highlight = document.getElementById('scene2-lat-20-highlight');
        const elEastArrowGroup = document.getElementById('scene2-east-arrow-group');
        const elArcShortest = document.getElementById('scene2-arc-shortest');
        const elDirArrows = document.getElementById('scene2-dir-arrows');
        const elAnimGroup = document.getElementById('scene2-anim-group');
        const elAnimDot = document.getElementById('scene2-anim-dot');
        const elAnimArrowLine = document.getElementById('scene2-anim-arrow-line');
        const elAnimArrowHead = document.getElementById('scene2-anim-arrow-head');

        const elLabelLatPath = document.getElementById('scene2-label-lat-path');
        const elLabelEastLat = document.getElementById('scene2-label-east-lat');
        const elLabelShortest = document.getElementById('scene2-label-shortest');
        const elLabelNortheast = document.getElementById('scene2-label-northeast');
        const elLabelEastArc = document.getElementById('scene2-label-east-arc');
        const elLabelSoutheast = document.getElementById('scene2-label-southeast');
        const elConclusion = document.getElementById('scene2-conclusion');

        function show(el) { if (el) { el.style.display = ''; el.style.opacity = '1'; } }
        function hide(el) { if (el) { el.style.display = 'none'; } }

        // 二次贝塞尔曲线计算
        function bezierPoint(t, p0, p1, p2) {
            const mt = 1 - t;
            return {
                x: mt * mt * p0.x + 2 * mt * t * p1.x + t * t * p2.x,
                y: mt * mt * p0.y + 2 * mt * t * p1.y + t * t * p2.y
            };
        }

        function bezierTangent(t, p0, p1, p2) {
            const mt = 1 - t;
            return {
                x: 2 * mt * (p1.x - p0.x) + 2 * t * (p2.x - p1.x),
                y: 2 * mt * (p1.y - p0.y) + 2 * t * (p2.y - p1.y)
            };
        }

        function normalize(v) {
            const len = Math.sqrt(v.x * v.x + v.y * v.y);
            if (len === 0) return { x: 0, y: 0 };
            return { x: v.x / len, y: v.y / len };
        }

        // 动画播放
        function playAnimation() {
            if (animating) return;
            animating = true;
            btnNext.disabled = true;
            show(elAnimGroup);

            const duration = 2000;
            const startTime = performance.now();

            function frame(now) {
                const elapsed = now - startTime;
                let t = elapsed / duration;
                if (t > 1) t = 1;

                const pt = bezierPoint(t, p0, p1, p2);
                const tan = normalize(bezierTangent(t, p0, p1, p2));

                // 更新移动点
                elAnimDot.setAttribute('cx', pt.x);
                elAnimDot.setAttribute('cy', pt.y);

                // 更新方向箭头（长度20px）
                const arrowLen = 20;
                const ax = pt.x + tan.x * arrowLen;
                const ay = pt.y + tan.y * arrowLen;
                elAnimArrowLine.setAttribute('x1', pt.x);
                elAnimArrowLine.setAttribute('y1', pt.y);
                elAnimArrowLine.setAttribute('x2', ax);
                elAnimArrowLine.setAttribute('y2', ay);

                // 箭头翼
                const wingLen = 8;
                const wingAngle = Math.PI / 6;
                const angle = Math.atan2(tan.y, tan.x);
                const a1x = ax - wingLen * Math.cos(angle - wingAngle);
                const a1y = ay - wingLen * Math.sin(angle - wingAngle);
                const a2x = ax - wingLen * Math.cos(angle + wingAngle);
                const a2y = ay - wingLen * Math.sin(angle + wingAngle);
                elAnimArrowHead.setAttribute('points', `${ax},${ay} ${a1x},${a1y} ${a2x},${a2y}`);

                if (t < 1) {
                    animReqId = requestAnimationFrame(frame);
                } else {
                    animating = false;
                    btnNext.disabled = false;
                }
            }

            animReqId = requestAnimationFrame(frame);
        }

        function stopAnimation() {
            if (animReqId) {
                cancelAnimationFrame(animReqId);
                animReqId = null;
            }
            animating = false;
        }

        function resetAnimation() {
            stopAnimation();
            hide(elAnimGroup);
            elAnimDot.setAttribute('cx', p0.x);
            elAnimDot.setAttribute('cy', p0.y);
        }

        function renderStep(step) {
            // 先隐藏所有步骤专属元素
            hide(elLat20Highlight);
            hide(elEastArrowGroup);
            hide(elArcShortest);
            hide(elDirArrows);
            resetAnimation();
            hide(elLabelLatPath);
            hide(elLabelEastLat);
            hide(elLabelShortest);
            hide(elLabelNortheast);
            hide(elLabelEastArc);
            hide(elLabelSoutheast);
            hide(elConclusion);

            // 根据步骤显示对应元素
            if (step >= 1) {
                show(elLat20Highlight);
                show(elEastArrowGroup);
                show(elLabelLatPath);
                show(elLabelEastLat);
            }
            if (step >= 2) {
                show(elArcShortest);
                show(elLabelShortest);
            }
            if (step >= 3) {
                show(elDirArrows);
                show(elLabelNortheast);
                show(elLabelEastArc);
                show(elLabelSoutheast);
            }
            if (step >= 4) {
                // 步骤4自动播放动画
                setTimeout(() => {
                    if (currentStep === 4) {
                        playAnimation();
                    }
                }, 50);
            }
            if (step >= 5) {
                show(elAnimGroup);
                // 动画结束状态：点在D点
                elAnimDot.setAttribute('cx', p2.x);
                elAnimDot.setAttribute('cy', p2.y);
                // 终点方向箭头（东南）
                const tan = normalize(bezierTangent(1, p0, p1, p2));
                const arrowLen = 20;
                const ax = p2.x + tan.x * arrowLen;
                const ay = p2.y + tan.y * arrowLen;
                elAnimArrowLine.setAttribute('x1', p2.x);
                elAnimArrowLine.setAttribute('y1', p2.y);
                elAnimArrowLine.setAttribute('x2', ax);
                elAnimArrowLine.setAttribute('y2', ay);
                const wingLen = 8;
                const wingAngle = Math.PI / 6;
                const angle = Math.atan2(tan.y, tan.x);
                const a1x = ax - wingLen * Math.cos(angle - wingAngle);
                const a1y = ay - wingLen * Math.sin(angle - wingAngle);
                const a2x = ax - wingLen * Math.cos(angle + wingAngle);
                const a2y = ay - wingLen * Math.sin(angle + wingAngle);
                elAnimArrowHead.setAttribute('points', `${ax},${ay} ${a1x},${a1y} ${a2x},${a2y}`);
                show(elConclusion);
            }

            // 更新按钮状态
            btnPrev.disabled = step <= 0;
            if (step === 4 && animating) {
                btnNext.disabled = true;
            } else {
                btnNext.disabled = step >= totalSteps - 1;
            }

            stepIndicator.textContent = '步骤 ' + (step + 1) + '/' + totalSteps;
        }

        function setStep(step) {
            if (step < 0 || step >= totalSteps) return;
            currentStep = step;
            renderStep(currentStep);
        }

        btnPrev.addEventListener('click', function() {
            setStep(currentStep - 1);
        });

        btnNext.addEventListener('click', function() {
            setStep(currentStep + 1);
        });

        // 初始渲染
        renderStep(0);
    })();
    </script>
