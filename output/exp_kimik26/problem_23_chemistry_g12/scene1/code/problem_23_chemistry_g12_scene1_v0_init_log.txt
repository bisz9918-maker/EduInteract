    <script>
    (function() {
        let currentStep = 0;
        const totalSteps = 4;
        
        const btnPrev = document.getElementById('scene1-btn-prev');
        const btnNext = document.getElementById('scene1-btn-next');
        const stepIndicator = document.getElementById('scene1-step-indicator');
        
        // 元素引用
        const bondOO = document.getElementById('scene1-bond-oo');
        const bondO1H1 = document.getElementById('scene1-bond-o1h1');
        const bondO2H2 = document.getElementById('scene1-bond-o2h2');
        
        const labelNonpolar = document.getElementById('scene1-label-nonpolar');
        const labelPolar1 = document.getElementById('scene1-label-polar1');
        const labelPolar2 = document.getElementById('scene1-label-polar2');
        const deltaO1 = document.getElementById('scene1-delta-o1');
        const deltaO2 = document.getElementById('scene1-delta-o2');
        const deltaH1 = document.getElementById('scene1-delta-h1');
        const deltaH2 = document.getElementById('scene1-delta-h2');
        const arrow1 = document.getElementById('scene1-arrow1');
        const arrow2 = document.getElementById('scene1-arrow2');
        
        const centerPlus = document.getElementById('scene1-center-plus');
        const centerMinus = document.getElementById('scene1-center-minus');
        const centerLine = document.getElementById('scene1-center-line');
        const labelNocoincide = document.getElementById('scene1-label-nocoincide');
        const labelPolarMolecule = document.getElementById('scene1-label-polar-molecule');
        
        const molecule1 = document.getElementById('scene1-molecule1');
        const molecule2 = document.getElementById('scene1-molecule2');
        const hbond1 = document.getElementById('scene1-hbond1');
        const hbond2 = document.getElementById('scene1-hbond2');
        const vdw1 = document.getElementById('scene1-vdw1');
        const vdw2 = document.getElementById('scene1-vdw2');
        const labelHbond = document.getElementById('scene1-label-hbond');
        const labelVdw = document.getElementById('scene1-label-vdw');
        
        function setDisplay(el, show) {
            if (el) el.style.display = show ? 'block' : 'none';
        }
        
        function renderStep() {
            stepIndicator.textContent = '步骤 ' + (currentStep + 1) + '/' + totalSteps;
            btnPrev.disabled = currentStep === 0;
            btnNext.disabled = currentStep === totalSteps - 1;
            
            // 键颜色
            if (currentStep >= 1) {
                bondOO.setAttribute('stroke', '#c62828');
                bondO1H1.setAttribute('stroke', '#1565c0');
                bondO2H2.setAttribute('stroke', '#1565c0');
            } else {
                bondOO.setAttribute('stroke', '#000000');
                bondO1H1.setAttribute('stroke', '#000000');
                bondO2H2.setAttribute('stroke', '#000000');
            }
            
            // 步骤1
            setDisplay(labelNonpolar, currentStep >= 1);
            setDisplay(labelPolar1, currentStep >= 1);
            setDisplay(labelPolar2, currentStep >= 1);
            setDisplay(deltaO1, currentStep >= 1);
            setDisplay(deltaO2, currentStep >= 1);
            setDisplay(deltaH1, currentStep >= 1);
            setDisplay(deltaH2, currentStep >= 1);
            setDisplay(arrow1, currentStep >= 1);
            setDisplay(arrow2, currentStep >= 1);
            
            // 步骤2
            setDisplay(centerPlus, currentStep >= 2);
            setDisplay(centerMinus, currentStep >= 2);
            setDisplay(centerLine, currentStep >= 2);
            setDisplay(labelNocoincide, currentStep >= 2);
            setDisplay(labelPolarMolecule, currentStep >= 2);
            
            // 步骤3
            setDisplay(molecule1, currentStep >= 3);
            setDisplay(molecule2, currentStep >= 3);
            setDisplay(hbond1, currentStep >= 3);
            setDisplay(hbond2, currentStep >= 3);
            setDisplay(vdw1, currentStep >= 3);
            setDisplay(vdw2, currentStep >= 3);
            setDisplay(labelHbond, currentStep >= 3);
            setDisplay(labelVdw, currentStep >= 3);
        }
        
        btnPrev.addEventListener('click', function() {
            if (currentStep > 0) {
                currentStep--;
                renderStep();
            }
        });
        
        btnNext.addEventListener('click', function() {
            if (currentStep < totalSteps - 1) {
                currentStep++;
                renderStep();
            }
        });
        
        renderStep();
    })();
    </script>
