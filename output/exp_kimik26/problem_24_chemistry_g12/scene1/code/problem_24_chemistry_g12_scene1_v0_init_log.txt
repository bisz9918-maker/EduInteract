    <script>
    (function() {
        var currentStep = 0;
        var totalSteps = 4;
        var btnPrev = document.getElementById('scene1-btn-prev');
        var btnNext = document.getElementById('scene1-btn-next');
        var stepIndicator = document.getElementById('scene1-step-indicator');
        var stepElements = {
            1: ['scene1-condition-text', 'scene1-condition-line1', 'scene1-condition-line2', 'scene1-condition-line3'],
            2: ['scene1-equation'],
            3: ['scene1-type-jia', 'scene1-type-yi', 'scene1-type-bing']
        };
        function getEl(id) { return document.getElementById(id); }
        function showElement(id) {
            var el = getEl(id);
            if (!el) return;
            el.style.display = '';
            void el.offsetWidth;
            el.classList.add('visible');
        }
        function hideElement(id) {
            var el = getEl(id);
            if (!el) return;
            el.classList.remove('visible');
            setTimeout(function() {
                if (!el.classList.contains('visible')) el.style.display = 'none';
            }, 300);
        }
        function renderStep() {
            btnPrev.disabled = (currentStep === 0);
            btnNext.disabled = (currentStep === totalSteps - 1);
            stepIndicator.textContent = '步骤 ' + (currentStep + 1) + ' / ' + totalSteps;
            for (var step = 1; step < totalSteps; step++) {
                var ids = stepElements[step] || [];
                if (step <= currentStep) {
                    ids.forEach(function(id) { showElement(id); });
                } else {
                    ids.forEach(function(id) { hideElement(id); });
                }
            }
            var yiContainer = getEl('scene1-container-yi');
            if (currentStep === 3) {
                yiContainer.classList.remove('highlight-stroke');
                void yiContainer.offsetWidth;
                yiContainer.classList.add('highlight-stroke');
            } else {
                yiContainer.classList.remove('highlight-stroke');
                var lines = yiContainer.querySelectorAll('line, path');
                lines.forEach(function(l) {
                    l.setAttribute('stroke', '#000000');
                    l.setAttribute('stroke-width', l.tagName.toLowerCase() === 'line' ? '2' : '2');
                });
            }
        }
        function init() {
            for (var step = 1; step < totalSteps; step++) {
                var ids = stepElements[step] || [];
                ids.forEach(function(id) {
                    var el = getEl(id);
                    if (el) { el.style.display = 'none'; el.classList.remove('visible'); }
                });
            }
            renderStep();
        }
        btnPrev.addEventListener('click', function() {
            if (currentStep > 0) { currentStep--; renderStep(); }
        });
        btnNext.addEventListener('click', function() {
            if (currentStep < totalSteps - 1) { currentStep++; renderStep(); }
        });
        init();
    })();
    </script>
