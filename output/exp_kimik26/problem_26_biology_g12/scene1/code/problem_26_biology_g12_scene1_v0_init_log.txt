    <script>
    (function() {
        // Scene 1 实现
        var currentStep = 0;
        var totalSteps = 4;

        var btnPrev = document.getElementById('scene1-btn-prev');
        var btnNext = document.getElementById('scene1-btn-next');
        var indicator = document.getElementById('scene1-step-indicator');

        var step1Group = document.getElementById('scene1-step1-group');
        var step2Group = document.getElementById('scene1-step2-group');
        var step3Group = document.getElementById('scene1-step3-group');

        function updateUI() {
            // 更新指示器
            indicator.textContent = (currentStep + 1) + ' / ' + totalSteps;

            // 更新按钮状态
            btnPrev.disabled = (currentStep === 0);
            btnNext.disabled = (currentStep === totalSteps - 1);

            // 隐藏所有分析区
            step1Group.style.display = 'none';
            step2Group.style.display = 'none';
            step3Group.style.display = 'none';

            // 显示当前步骤对应的分析区
            if (currentStep === 1) {
                step1Group.style.display = 'block';
            } else if (currentStep === 2) {
                step2Group.style.display = 'block';
            } else if (currentStep === 3) {
                step3Group.style.display = 'block';
            }
        }

        btnPrev.addEventListener('click', function() {
            if (currentStep > 0) {
                currentStep--;
                updateUI();
            }
        });

        btnNext.addEventListener('click', function() {
            if (currentStep < totalSteps - 1) {
                currentStep++;
                updateUI();
            }
        });

        // 初始状态
        updateUI();
    })();
    </script>
