    <script>
    (function() {
        // Scene 1 实现
        var currentStep = 0;
        var totalSteps = 5;

        var btnPrev = document.getElementById('scene1-btn-prev');
        var btnNext = document.getElementById('scene1-btn-next');

        // 列高亮元素 (HTML divs)
        var colHighlights = [];
        for (var i = 1; i <= 6; i++) {
            colHighlights.push(document.getElementById('scene1-col-highlight-' + i));
        }

        // 结论区元素
        var conclusionGroups = [];
        for (var i = 0; i < totalSteps; i++) {
            conclusionGroups.push(document.getElementById('scene1-conclusion-step' + i));
        }

        // 功能框元素
        var funcBox1 = document.getElementById('scene1-func-box-1');
        var funcBox2 = document.getElementById('scene1-func-box-2');

        // 指示器圆点
        var dots = [];
        for (var i = 0; i < totalSteps; i++) {
            dots.push(document.getElementById('scene1-dot-' + i));
        }

        // 所有表格文字元素 (用于淡化非聚焦列)
        var allTableTexts = document.querySelectorAll('.table-text');

        // 步骤配置：每步高亮的列索引 (1-based)，空数组表示不高亮任何列
        var stepHighlightCols = [
            [],      // 步骤0: 无高亮
            [1],     // 步骤1: 高亮第1列
            [4, 6],  // 步骤2: 高亮第4、6列
            [2, 4],  // 步骤3: 高亮第2、4列
            []       // 步骤4: 无高亮
        ];

        function show(el) {
            el.style.display = 'block';
            requestAnimationFrame(function() {
                el.style.opacity = '1';
            });
        }

        function hide(el) {
            el.style.opacity = '0';
            setTimeout(function() {
                if (el.style.opacity === '0') {
                    el.style.display = 'none';
                }
            }, 300);
        }

        function renderStep() {
            // 1. 更新按钮状态
            btnPrev.disabled = (currentStep === 0);
            btnNext.disabled = (currentStep === totalSteps - 1);

            // 2. 更新指示器
            for (var i = 0; i < totalSteps; i++) {
                dots[i].classList.toggle('active', i === currentStep);
            }

            // 3. 更新列高亮 (HTML divs)
            var highlightSet = stepHighlightCols[currentStep];
            for (var i = 0; i < 6; i++) {
                var colIdx = i + 1;
                var shouldHighlight = highlightSet.indexOf(colIdx) !== -1;
                colHighlights[i].style.opacity = shouldHighlight ? '1' : '0';
            }

            // 4. 更新表格文字淡化 (非聚焦列 opacity 0.35)
            allTableTexts.forEach(function(txt) {
                var x = parseFloat(txt.getAttribute('x'));
                if (x < 120) {
                    // 左侧标签列始终清晰
                    txt.style.opacity = '1';
                    return;
                }
                var inHighlight = false;
                for (var j = 0; j < highlightSet.length; j++) {
                    var col = highlightSet[j];
                    var colLeft = 120 + (col - 1) * 55;
                    var colRight = colLeft + 55;
                    if (x >= colLeft && x <= colRight) {
                        inHighlight = true;
                        break;
                    }
                }
                // 步骤0和步骤4不高亮任何列，所有文字清晰
                if (highlightSet.length === 0) {
                    txt.style.opacity = '1';
                } else {
                    txt.style.opacity = inHighlight ? '1' : '0.35';
                }
            });

            // 5. 更新结论区显示
            for (var i = 0; i < totalSteps; i++) {
                if (i === currentStep) {
                    show(conclusionGroups[i]);
                } else {
                    hide(conclusionGroups[i]);
                }
            }

            // 6. 更新功能框显示 (仅在步骤4显示)
            if (currentStep === 4) {
                funcBox1.style.opacity = '1';
                funcBox2.style.opacity = '1';
            } else {
                funcBox1.style.opacity = '0';
                funcBox2.style.opacity = '0';
            }
        }

        btnPrev.addEventListener('click', function() {
            if (currentStep > 0) {
                currentStep--;
                renderStep();
            }
        });

        btnNext.addEventListener('click', function() {
            if (currentStep < totalSteps - 1) {
                currentStep++;
                renderStep();
            }
        });

        // 初始化
        renderStep();
    })();
