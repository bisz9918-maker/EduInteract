    <script>
    (function() {
        // Scene 1 实现
        let currentStep = 0;
        const totalSteps = 5; // 步骤0~4

        // 获取按钮和指示器
        const btnPrev = document.getElementById('scene1-btn-prev');
        const btnNext = document.getElementById('scene1-btn-next');
        const stepIndicator = document.getElementById('scene1-step-indicator');

        // 左侧节点 div
        const divFe = document.getElementById('scene1-div-fe');
        const divFe2o3 = document.getElementById('scene1-div-fe2o3');
        const divHcl = document.getElementById('scene1-div-hcl');
        const divCaoh2 = document.getElementById('scene1-div-caoh2');
        const divNa2co3 = document.getElementById('scene1-div-na2co3');

        // 右侧节点 div
        const divX = document.getElementById('scene1-div-x');
        const divY = document.getElementById('scene1-div-y');
        const divFeR = document.getElementById('scene1-div-fe-r');
        const divM = document.getElementById('scene1-div-m');
        const divN = document.getElementById('scene1-div-n');

        // 左侧连线
        const lineFeHcl = document.getElementById('scene1-line-fe-hcl');
        const lineFe2o3Hcl = document.getElementById('scene1-line-fe2o3-hcl');
        const lineCaoh2Hcl = document.getElementById('scene1-line-caoh2-hcl');
        const lineNa2co3Hcl = document.getElementById('scene1-line-na2co3-hcl');
        const lineCaoh2Na2co3 = document.getElementById('scene1-line-caoh2-na2co3');

        // 右侧连线
        const lineXY = document.getElementById('scene1-line-x-y');
        const lineYFe = document.getElementById('scene1-line-y-fe');
        const lineXM = document.getElementById('scene1-line-x-m');
        const lineYM = document.getElementById('scene1-line-y-m');
        const lineYN = document.getElementById('scene1-line-y-n');
        const lineNFe = document.getElementById('scene1-line-n-fe');

        // 提示文字组
        const hintStep1 = document.getElementById('scene1-g-hint-step1');
        const hintStep2 = document.getElementById('scene1-g-hint-step2');
        const hintStep3 = document.getElementById('scene1-g-hint-step3');
        const hintExclude = document.getElementById('scene1-g-hint-exclude');
        const hintStep4a = document.getElementById('scene1-g-hint-step4a');
        const hintStep4b = document.getElementById('scene1-g-hint-step4b');

        // 工具函数：显示/隐藏元素
        function show(el) { if(el) { el.style.display = 'block'; } }
        function hide(el) { if(el) { el.style.display = 'none'; } }

        // 清除所有高亮和提示
        function clearState() {
            // 清除节点高亮
            const allDivs = [divFe, divFe2o3, divHcl, divCaoh2, divNa2co3, divX, divY, divFeR, divM, divN];
            allDivs.forEach(d => {
                d.classList.remove('hl-blue', 'hl-green', 'hl-red');
                d.style.opacity = '1';
                d.style.fontSize = '';
                d.style.borderColor = '';
            });

            // 恢复节点文字
            divFe.querySelector('.node-label').textContent = '铁';
            divFe2o3.querySelector('.node-label').textContent = '氧化铁';
            divHcl.querySelector('.node-label').textContent = '稀盐酸';
            divCaoh2.querySelector('.node-label').textContent = '氢氧化钙';
            divNa2co3.querySelector('.node-label').textContent = '碳酸钠';
            divX.querySelector('.node-label').textContent = 'X';
            divY.querySelector('.node-label').textContent = 'Y';
            divFeR.querySelector('.node-label').textContent = '铁';
            divM.querySelector('.node-label').textContent = 'M';
            divN.querySelector('.node-label').textContent = 'N';

            // 清除连线高亮
            const allLines = [lineFeHcl, lineFe2o3Hcl, lineCaoh2Hcl, lineNa2co3Hcl, lineCaoh2Na2co3,
                              lineXY, lineYFe, lineXM, lineYM, lineYN, lineNFe];
            allLines.forEach(l => {
                l.classList.remove('hl-blue', 'hl-green');
                if (l.id === 'scene1-line-n-fe') {
                    l.setAttribute('marker-end', 'url(#scene1-arrow-black)');
                }
            });

            // 隐藏所有提示
            hide(hintStep1); hide(hintStep2); hide(hintStep3);
            hide(hintExclude); hide(hintStep4a); hide(hintStep4b);
        }

        // 渲染步骤
        function renderStep() {
            clearState();
            stepIndicator.textContent = '步骤 ' + (currentStep + 1) + ' / ' + totalSteps;
            btnPrev.disabled = currentStep === 0;
            btnNext.disabled = currentStep === totalSteps - 1;

            switch(currentStep) {
                case 0:
                    // 初始状态：全部默认，无提示
                    break;

                case 1:
                    // 步骤1：稀盐酸和Y高亮（蓝色）
                    divHcl.classList.add('hl-blue');
                    divY.classList.add('hl-blue');

                    // 稀盐酸的4条连线
                    lineFeHcl.classList.add('hl-blue');
                    lineFe2o3Hcl.classList.add('hl-blue');
                    lineCaoh2Hcl.classList.add('hl-blue');
                    lineNa2co3Hcl.classList.add('hl-blue');

                    // Y的4条连线
                    lineXY.classList.add('hl-blue');
                    lineYFe.classList.add('hl-blue');
                    lineYM.classList.add('hl-blue');
                    lineYN.classList.add('hl-blue');

                    show(hintStep1);
                    break;

                case 2:
                    // 步骤2：N→铁和氧化铁高亮（蓝色）
                    divFe2o3.classList.add('hl-blue');
                    divN.classList.add('hl-blue');

                    // 氧化铁-稀盐酸连线
                    lineFe2o3Hcl.classList.add('hl-blue');
                    // N→铁箭头
                    lineNFe.classList.add('hl-blue');
                    lineNFe.setAttribute('marker-end', 'url(#scene1-arrow-blue)');

                    // 稀盐酸保持淡关联（通过直接设置边框色）
                    divHcl.style.borderColor = '#1565c0';

                    show(hintStep2);
                    break;

                case 3:
                    // 步骤3：X-Y-M三角形、氢氧化钙、碳酸钠高亮（绿色），铁排除（红色）
                    divCaoh2.classList.add('hl-green');
                    divNa2co3.classList.add('hl-green');

                    // X-Y-M三角形连线
                    lineXY.classList.add('hl-green');
                    lineXM.classList.add('hl-green');
                    lineYM.classList.add('hl-green');

                    // 氢氧化钙-碳酸钠连线
                    lineCaoh2Na2co3.classList.add('hl-green');

                    // 铁排除
                    divFe.classList.add('hl-red');

                    show(hintStep3);
                    show(hintExclude);
                    break;

                case 4:
                    // 步骤4：全部高亮绿色，右侧节点文字替换
                    // 左侧全部节点绿色
                    [divFe, divFe2o3, divHcl, divCaoh2, divNa2co3].forEach(d => d.classList.add('hl-green'));

                    // 左侧全部连线绿色
                    [lineFeHcl, lineFe2o3Hcl, lineCaoh2Hcl, lineNa2co3Hcl, lineCaoh2Na2co3].forEach(l => l.classList.add('hl-green'));

                    // 右侧全部节点绿色
                    [divX, divY, divFeR, divM, divN].forEach(d => d.classList.add('hl-green'));

                    // 右侧全部连线绿色
                    [lineXY, lineYFe, lineXM, lineYM, lineYN, lineNFe].forEach(l => l.classList.add('hl-green'));
                    lineNFe.setAttribute('marker-end', 'url(#scene1-arrow-green)');

                    // 右侧节点文字替换
                    divY.querySelector('.node-label').textContent = '稀盐酸';
                    divN.querySelector('.node-label').textContent = '氧化铁';
                    divX.querySelector('.node-label').textContent = '氢氧化钙/碳酸钠';
                    divX.style.fontSize = '10px';
                    divM.querySelector('.node-label').textContent = '碳酸钠/氢氧化钙';
                    divM.style.fontSize = '10px';

                    show(hintStep4a);
                    show(hintStep4b);
                    break;
            }
        }

        // 按钮事件
        btnPrev.addEventListener('click', function() {
            if (currentStep > 0) {
                currentStep--;
                renderStep();
            }
        });

        btnNext.addEventListener('click', function() {
            if (currentStep < totalSteps - 1) {
                currentStep++;
                renderStep();
            }
        });

        // 初始渲染
        renderStep();
    })();
    </script>
