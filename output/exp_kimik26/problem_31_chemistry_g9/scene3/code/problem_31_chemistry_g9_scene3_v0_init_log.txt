<script>
(function() {
var currentStep = 0;
var totalSteps = 5;
var btnPrev = document.getElementById('scene3-btn-prev');
var btnNext = document.getElementById('scene3-btn-next');
var indicator = document.getElementById('scene3-step-indicator');
var groups = [];
for (var i = 0; i < totalSteps; i++) {
groups.push(document.getElementById('scene3-step' + i + '-group'));
}
function updateUI() {
for (var i = 0; i < totalSteps; i++) {
if (groups[i]) {
groups[i].style.display = (i === currentStep) ? 'block' : 'none';
groups[i].style.opacity = (i === currentStep) ? '1' : '0';
}
}
indicator.textContent = '步骤 ' + (currentStep + 1) + '/' + totalSteps;
btnPrev.disabled = (currentStep === 0);
btnNext.disabled = (currentStep === totalSteps - 1);
}
btnPrev.addEventListener('click', function() {
if (currentStep > 0) { currentStep--; updateUI(); }
});
btnNext.addEventListener('click', function() {
if (currentStep < totalSteps - 1) { currentStep++; updateUI(); }
});
updateUI();
})();
</script>
