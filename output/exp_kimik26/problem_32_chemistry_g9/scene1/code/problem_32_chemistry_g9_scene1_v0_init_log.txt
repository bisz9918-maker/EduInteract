    <script>
    (function() {
        // Scene 1 实现
        let currentStep = 0;
        const totalSteps = 5;
        let animationId = null;
        let molecules = [];
        let frameCount = 0;
        let smokePulseId = null;

        const btnPrev = document.getElementById('scene1-btn-prev');
        const btnNext = document.getElementById('scene1-btn-next');
        const stepIndicator = document.getElementById('scene1-step-indicator');
        const stepTitle = document.getElementById('scene1-step-title');
        const moleculesGroup = document.getElementById('scene1-molecules');
        const smoke = document.getElementById('scene1-smoke');
        const infoPanel = document.getElementById('scene1-info-panel');

        // 元素引用
        const els = {
            statusC: document.getElementById('scene1-status-c'),
            statusA: document.getElementById('scene1-status-a'),
            statusB: document.getElementById('scene1-status-b'),
            trackHcl: document.getElementById('scene1-track-hcl'),
            trackNh3: document.getElementById('scene1-track-nh3'),
            smoke: document.getElementById('scene1-smoke'),
            labelSmoke: document.getElementById('scene1-label-smoke'),
            labelNearA: document.getElementById('scene1-label-near-a'),
            arrow: document.getElementById('scene1-arrow'),
            pistonC: document.getElementById('scene1-piston-c'),
            pistonA: document.getElementById('scene1-piston-a'),
            pistonB: document.getElementById('scene1-piston-b')
        };

        function show(el) {
            if (!el) return;
            el.classList.remove('hidden');
            el.style.display = '';
            el.style.opacity = '1';
        }

        function hide(el) {
            if (!el) return;
            el.classList.add('hidden');
            el.style.display = 'none';
            el.style.opacity = '0';
        }

        function setOpacity(el, val) {
            if (!el) return;
            el.style.opacity = String(val);
        }

        function stopAnimation() {
            if (animationId) {
                cancelAnimationFrame(animationId);
                animationId = null;
            }
            if (smokePulseId) {
                cancelAnimationFrame(smokePulseId);
                smokePulseId = null;
            }
            molecules = [];
            frameCount = 0;
            moleculesGroup.innerHTML = '';
        }

        function createMolecule(type, x, y) {
            const circle = document.createElementNS('http://www.w3.org/2000/svg', 'circle');
            circle.setAttribute('cx', x);
            circle.setAttribute('cy', y);
            circle.setAttribute('r', 3);
            circle.setAttribute('fill', type === 'hcl' ? '#1565c0' : '#c62828');
            circle.setAttribute('stroke', 'none');
            moleculesGroup.appendChild(circle);
            return {
                element: circle,
                x: x,
                y: y,
                vx: type === 'hcl' ? 1 : -2,
                vy: 0,
                type: type,
                opacity: 1
            };
        }

        function animateMolecules() {
            frameCount++;
            // 每30帧释放一组分子
            if (frameCount % 30 === 0 && frameCount <= 150) {
                molecules.push(createMolecule('hcl', 220, 210));
                molecules.push(createMolecule('nh3', 580, 210));
            }

            for (let i = molecules.length - 1; i >= 0; i--) {
                const m = molecules[i];
                m.x += m.vx;
                m.y += m.vy;
                m.element.setAttribute('cx', m.x);
                m.element.setAttribute('cy', m.y);

                // 到达相遇点附近淡出
                const targetX = 340;
                const dist = Math.abs(m.x - targetX);
                if (dist < 20) {
                    m.opacity -= 0.05;
                    m.element.setAttribute('opacity', Math.max(0, m.opacity));
                }
                if (m.opacity <= 0 || (m.type === 'hcl' && m.x > 340) || (m.type === 'nh3' && m.x < 340)) {
                    moleculesGroup.removeChild(m.element);
                    molecules.splice(i, 1);
                }
            }

            if (frameCount < 200 || molecules.length > 0) {
                animationId = requestAnimationFrame(animateMolecules);
            }
        }

        function pulseSmoke() {
            let start = null;
            function step(timestamp) {
                if (!start) start = timestamp;
                const elapsed = timestamp - start;
                const scale = 1 + 0.05 * Math.sin(elapsed / 300);
                smoke.setAttribute('transform', `translate(340,210) scale(${scale}) translate(-340,-210)`);
                smokePulseId = requestAnimationFrame(step);
            }
            smokePulseId = requestAnimationFrame(step);
        }

        function renderStep() {
            stopAnimation();

            // 更新按钮状态
            btnPrev.disabled = currentStep === 0;
            btnNext.disabled = currentStep === totalSteps - 1;
            stepIndicator.textContent = '步骤 ' + (currentStep + 1) + '/' + totalSteps;

            // 重置所有步骤相关元素
            hide(els.statusC);
            hide(els.statusA);
            hide(els.statusB);
            hide(els.trackHcl);
            hide(els.trackNh3);
            hide(moleculesGroup);
            hide(els.smoke);
            hide(els.labelSmoke);
            hide(els.labelNearA);
            hide(els.arrow);
            if (infoPanel) infoPanel.style.display = 'none';

            // 重置活塞样式
            els.pistonC.setAttribute('stroke', '#000');
            els.pistonC.setAttribute('stroke-width', '1.5');
            els.pistonA.setAttribute('stroke', '#000');
            els.pistonA.setAttribute('stroke-width', '1.5');
            els.pistonB.setAttribute('fill', '#fff');

            switch (currentStep) {
                case 0:
                    stepTitle.textContent = '装置认知';
                    break;
                case 1:
                    stepTitle.textContent = '操作示意：打开活塞a、c，关闭活塞b';
                    show(els.statusC);
                    show(els.statusA);
                    show(els.statusB);
                    els.pistonC.setAttribute('stroke', '#2e7d32');
                    els.pistonC.setAttribute('stroke-width', '2');
                    els.pistonA.setAttribute('stroke', '#2e7d32');
                    els.pistonA.setAttribute('stroke-width', '2');
                    els.pistonB.setAttribute('fill', '#bdbdbd');
                    break;
                case 2:
                    stepTitle.textContent = '分子运动：NH₃与HCl相向扩散';
                    show(els.statusC);
                    show(els.statusA);
                    show(els.statusB);
                    show(els.trackHcl);
                    show(els.trackNh3);
                    show(moleculesGroup);
                    els.pistonC.setAttribute('stroke', '#2e7d32');
                    els.pistonC.setAttribute('stroke-width', '2');
                    els.pistonA.setAttribute('stroke', '#2e7d32');
                    els.pistonA.setAttribute('stroke-width', '2');
                    els.pistonB.setAttribute('fill', '#bdbdbd');
                    frameCount = 0;
                    animateMolecules();
                    break;
                case 3:
                    stepTitle.textContent = '白烟生成：NH₄Cl白烟靠近A端';
                    show(els.smoke);
                    show(els.labelSmoke);
                    show(els.labelNearA);
                    setOpacity(els.smoke, 0.4);
                    pulseSmoke();
                    break;
                case 4:
                    stepTitle.textContent = '原理总结：分子运动速率与质量成反比';
                    show(els.smoke);
                    show(els.labelSmoke);
                    show(els.labelNearA);
                    show(els.arrow);
                    if (infoPanel) infoPanel.style.display = 'block';
                    setOpacity(els.smoke, 0.4);
                    break;
            }
        }

        btnPrev.addEventListener('click', function() {
            if (currentStep > 0) {
                currentStep--;
                renderStep();
            }
        });

        btnNext.addEventListener('click', function() {
            if (currentStep < totalSteps - 1) {
                currentStep++;
                renderStep();
            }
        });

        // 初始渲染
        renderStep();
    })();
    </script>
