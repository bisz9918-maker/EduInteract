    <script>
    (function() {
        // Scene 1 实现：三视图对应关系与信息提取

        // 状态变量
        var currentStep = 0;
        var MAX_STEP = 4;

        // 获取元素引用
        var mainFrame = document.getElementById('scene1-main-frame');
        var leftFrame = document.getElementById('scene1-left-frame');
        var topFrame = document.getElementById('scene1-top-frame');

        var mainLabel = document.getElementById('scene1-main-label');
        var leftLabel = document.getElementById('scene1-left-label');
        var topLabel = document.getElementById('scene1-top-label');

        var step1Group = document.getElementById('scene1-step1-group');
        var step2Group = document.getElementById('scene1-step2-group');
        var step3Group = document.getElementById('scene1-step3-group');
        var step4Group = document.getElementById('scene1-step4-group');

        var topCells = [
            document.getElementById('scene1-top-cell1'),
            document.getElementById('scene1-top-cell2'),
            document.getElementById('scene1-top-cell3'),
            document.getElementById('scene1-top-cell4')
        ];

        var prevBtn = document.getElementById('scene1-prev-btn');
        var nextBtn = document.getElementById('scene1-next-btn');
        var stepIndicator = document.getElementById('scene1-step-indicator');

        // 工具函数：显示元素（带淡入效果）
        function show(el) {
            if (!el) return;
            // 清除可能存在的隐藏定时器
            if (el._hideTimeout) {
                clearTimeout(el._hideTimeout);
                el._hideTimeout = null;
            }
            el.style.display = '';
            el.style.opacity = '0';
            requestAnimationFrame(function() {
                el.style.transition = 'opacity 200ms ease-out';
                el.style.opacity = '1';
            });
        }

        // 工具函数：隐藏元素（带淡出效果）
        function hide(el) {
            if (!el) return;
            el.style.opacity = '0';
            el.style.transition = 'opacity 200ms ease-out';
            // 清除之前的定时器，避免冲突
            if (el._hideTimeout) {
                clearTimeout(el._hideTimeout);
            }
            el._hideTimeout = setTimeout(function() {
                el.style.display = 'none';
                el._hideTimeout = null;
            }, 200);
        }

        // 重置所有高亮和标签
        function resetAll() {
            // 恢复外框颜色
            mainFrame.setAttribute('stroke', '#000000');
            mainFrame.setAttribute('stroke-width', '2');
            leftFrame.setAttribute('stroke', '#000000');
            leftFrame.setAttribute('stroke-width', '2');
            topFrame.setAttribute('stroke', '#000000');
            topFrame.setAttribute('stroke-width', '2');

            // 恢复标签颜色
            mainLabel.setAttribute('fill', '#000000');
            leftLabel.setAttribute('fill', '#000000');
            topLabel.setAttribute('fill', '#000000');

            // 恢复俯视图填充格外框
            topCells.forEach(function(cell) {
                cell.setAttribute('stroke', '#000000');
                cell.setAttribute('stroke-width', '1');
            });

            // 隐藏所有步骤组
            hide(step1Group);
            hide(step2Group);
            hide(step3Group);
            hide(step4Group);
        }

        // 跳转到指定步骤
        function goToStep(step) {
            currentStep = step;

            // 更新按钮状态
            prevBtn.disabled = (currentStep === 0);
            nextBtn.disabled = (currentStep === MAX_STEP);

            // 更新步骤指示器
            stepIndicator.textContent = '步骤 ' + (currentStep + 1) + ' / ' + (MAX_STEP + 1);

            // 先重置所有
            resetAll();

            // 根据步骤显示对应内容
            switch (currentStep) {
                case 0:
                    // 初始状态：仅显示三个视图，无高亮无标注
                    break;
                case 1:
                    // 步骤1：主视图高亮（蓝色），显示列高标签
                    mainFrame.setAttribute('stroke', '#1565c0');
                    mainFrame.setAttribute('stroke-width', '3');
                    mainLabel.setAttribute('fill', '#1565c0');
                    show(step1Group);
                    break;
                case 2:
                    // 步骤2：左视图高亮（绿色），显示行高标签
                    leftFrame.setAttribute('stroke', '#2e7d32');
                    leftFrame.setAttribute('stroke-width', '3');
                    leftLabel.setAttribute('fill', '#2e7d32');
                    show(step2Group);
                    break;
                case 3:
                    // 步骤3：俯视图高亮（红色），显示行列标签，填充格加红框
                    topFrame.setAttribute('stroke', '#c62828');
                    topFrame.setAttribute('stroke-width', '3');
                    topLabel.setAttribute('fill', '#c62828');
                    topCells.forEach(function(cell) {
                        cell.setAttribute('stroke', '#c62828');
                        cell.setAttribute('stroke-width', '2');
                    });
                    show(step3Group);
                    break;
                case 4:
                    // 步骤4：保持俯视图红色高亮，显示连线
                    topFrame.setAttribute('stroke', '#c62828');
                    topFrame.setAttribute('stroke-width', '3');
                    topLabel.setAttribute('fill', '#c62828');
                    topCells.forEach(function(cell) {
                        cell.setAttribute('stroke', '#c62828');
                        cell.setAttribute('stroke-width', '2');
                    });
                    show(step3Group);
                    show(step4Group);
                    break;
            }
        }

        // 事件绑定
        prevBtn.addEventListener('click', function() {
            if (currentStep > 0) {
                goToStep(currentStep - 1);
            }
        });

        nextBtn.addEventListener('click', function() {
            if (currentStep < MAX_STEP) {
                goToStep(currentStep + 1);
            }
        });

        // 初始化
        goToStep(0);
    })();
    </script>
