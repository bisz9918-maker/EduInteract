    <script>
    (function() {
        // Scene 1 实现
        var currentStep = 0;
        var totalSteps = 4;

        var btnPrev = document.getElementById('scene1-btn-prev');
        var btnNext = document.getElementById('scene1-btn-next');
        var indicator = document.getElementById('scene1-step-indicator');

        function showStep(n) {
            // 隐藏所有步骤组
            for (var i = 0; i <= totalSteps; i++) {
                var g = document.getElementById('scene1-step-' + i);
                if (g) {
                    g.style.display = 'none';
                }
            }
            // 显示当前步骤组
            var currentG = document.getElementById('scene1-step-' + n);
            if (currentG) {
                currentG.style.display = 'block';
            }

            // 更新按钮状态
            btnPrev.disabled = (n === 0);
            btnNext.disabled = (n === totalSteps);

            // 更新步骤指示器
            indicator.textContent = '步骤 ' + n + '/' + totalSteps;
        }

        btnPrev.addEventListener('click', function() {
            if (currentStep > 0) {
                currentStep--;
                showStep(currentStep);
            }
        });

        btnNext.addEventListener('click', function() {
            if (currentStep < totalSteps) {
                currentStep++;
                showStep(currentStep);
            }
        });

        // 初始显示步骤0
        showStep(0);
    })();
    </script>
