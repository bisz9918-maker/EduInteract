<script>
(function() {
let currentStep = 0;
const totalSteps = 7;
const btnPrev = document.getElementById('scene1-btn-prev');
const btnNext = document.getElementById('scene1-btn-next');
const stepIndicator = document.getElementById('scene1-step-indicator');
const axis = document.getElementById('scene1-axis');
const terminator = document.getElementById('scene1-terminator');
const meridian = document.getElementById('scene1-meridian');
const nightHemisphere = document.getElementById('scene1-night-hemisphere');
const polarNight = document.getElementById('scene1-polar-night');
const arcticCircle = document.getElementById('scene1-arctic-circle');
const tropic = document.getElementById('scene1-tropic');
const rotationArrow = document.getElementById('scene1-rotation-arrow');
const sun = document.getElementById('scene1-sun');
const sunRays = document.getElementById('scene1-sun-rays');
const labelNorthPole = document.getElementById('scene1-label-north-pole');
const labelEarthCore = document.getElementById('scene1-label-earth-core');
const labelN = document.getElementById('scene1-label-n');
const labelRotation = document.getElementById('scene1-label-rotation');
const labelArcticIntersect = document.getElementById('scene1-label-arctic-intersect');
const labelTropicIntersect = document.getElementById('scene1-label-tropic-intersect');
const labelNight = document.getElementById('scene1-label-night');
const labelDay = document.getElementById('scene1-label-day');
const labelArctic = document.getElementById('scene1-label-arctic');
const labelTropicS = document.getElementById('scene1-label-tropic-s');
const labelWinter = document.getElementById('scene1-label-winter');
const labelDirectSun = document.getElementById('scene1-label-direct-sun');
function setOpacity(el, value) { if (el) el.setAttribute('opacity', value); }
function setFillOpacity(el, value) { if (el) el.setAttribute('fill-opacity', value); }
function setStroke(el, color, width) { if (el) { if (color !== undefined) el.setAttribute('stroke', color); if (width !== undefined) el.setAttribute('stroke-width', width); } }
function renderStep() {
stepIndicator.textContent = '步骤 ' + currentStep + '/' + totalSteps;
btnPrev.disabled = currentStep === 0;
btnNext.disabled = currentStep === totalSteps;
setStroke(axis, '#000000', '2');
setStroke(terminator, '#000000', '2');
setStroke(meridian, '#000000', '1.5');
setFillOpacity(nightHemisphere, '0');
setFillOpacity(polarNight, '0');
setOpacity(arcticCircle, '0');
setOpacity(tropic, '0');
setOpacity(rotationArrow, '0');
setOpacity(sun, '0');
setOpacity(sunRays, '0');
setOpacity(labelNorthPole, '0');
setOpacity(labelEarthCore, '0');
setOpacity(labelN, '0');
setOpacity(labelRotation, '0');
setOpacity(labelArcticIntersect, '0');
setOpacity(labelTropicIntersect, '0');
setOpacity(labelNight, '0');
setOpacity(labelDay, '0');
setOpacity(labelArctic, '0');
setOpacity(labelTropicS, '0');
setOpacity(labelWinter, '0');
setOpacity(labelDirectSun, '0');
if (currentStep >= 1) { setStroke(axis, '#1565c0', '3'); setStroke(meridian, '#cccccc', '1.5'); setOpacity(labelNorthPole, '1'); setOpacity(labelEarthCore, '1'); }
if (currentStep >= 2) { setOpacity(rotationArrow, '1'); setOpacity(labelN, '1'); setOpacity(labelRotation, '1'); }
if (currentStep >= 3) { setStroke(meridian, '#000000', '2'); setStroke(terminator, '#c62828', '2.5'); setOpacity(labelArcticIntersect, '1'); setOpacity(labelTropicIntersect, '1'); }
if (currentStep >= 4) { setFillOpacity(nightHemisphere, '0.2'); setOpacity(labelNight, '1'); setOpacity(labelDay, '1'); }
if (currentStep >= 5) { setOpacity(arcticCircle, '1'); setOpacity(tropic, '1'); setOpacity(labelArctic, '1'); setOpacity(labelTropicS, '1'); }
if (currentStep >= 6) { setFillOpacity(polarNight, '0.3'); }
if (currentStep >= 7) { setOpacity(sun, '1'); setOpacity(sunRays, '1'); setOpacity(labelWinter, '1'); setOpacity(labelDirectSun, '1'); }
}
btnPrev.addEventListener('click', function() { if (currentStep > 0) { currentStep--; renderStep(); } });
btnNext.addEventListener('click', function() { if (currentStep < totalSteps) { currentStep++; renderStep(); } });
renderStep();
})();
</script>
