    <script>
    (function() {
        // Scene 1 实现
        // 状态机
        var state = {
            theta: 0,
            showProj: false,
            showSection: false,
            showPyramid: false
        };

        // DOM 元素缓存
        var els = {};

        function cacheElements() {
            els.slider = document.getElementById('scene1-slider-theta');
            els.thetaDisplay = document.getElementById('scene1-theta-display');
            els.checkProj = document.getElementById('scene1-check-proj');
            els.checkSection = document.getElementById('scene1-check-section');
            els.checkPyramid = document.getElementById('scene1-check-pyramid');

            els.pointAp = document.getElementById('scene1-point-ap');
            els.labelAp = document.getElementById('scene1-label-ap');
            els.bgAp = document.getElementById('scene1-bg-ap');
            els.lineAg = document.getElementById('scene1-line-ag');
            els.groupProj = document.getElementById('scene1-group-proj');
            els.lineApProj = document.getElementById('scene1-line-ap-proj');
            els.pointP = document.getElementById('scene1-point-p');
            els.labelP = document.getElementById('scene1-label-p');
            els.bgP = document.getElementById('scene1-bg-p');
            els.groupSection = document.getElementById('scene1-group-section');
            els.polyAgf = document.getElementById('scene1-poly-agf');
            els.groupPyramid = document.getElementById('scene1-group-pyramid');
            els.lineApD = document.getElementById('scene1-line-ap-d');
            els.lineApE = document.getElementById('scene1-line-ap-e');
            els.lineApF = document.getElementById('scene1-line-ap-f');
            els.polyPyramidFd = document.getElementById('scene1-poly-pyramid-fd');
            els.polyPyramidFe = document.getElementById('scene1-poly-pyramid-fe');
            els.polyPyramidEd = document.getElementById('scene1-poly-pyramid-ed');
            els.groupOrbit = document.getElementById('scene1-group-orbit');
        }

        // 坐标计算
        function getAprime(thetaDeg) {
            var rad = thetaDeg * Math.PI / 180;
            return {
                x: 400 + 94 * Math.cos(rad),
                y: 300 - 42 * Math.cos(rad) - 114 * Math.sin(rad)
            };
        }

        function getP(thetaDeg) {
            var rad = thetaDeg * Math.PI / 180;
            return {
                x: 400 + 94 * Math.cos(rad),
                y: 300 - 42 * Math.cos(rad)
            };
        }

        // 生成轨迹圆路径（使用短线段，避免单一路径的超大bounding box导致误报）
        function generateOrbitPath() {
            var fragment = document.createDocumentFragment();
            var prev = getAprime(0);
            for (var i = 1; i <= 72; i++) {
                var theta = i * 5;
                var p = getAprime(theta);
                var line = document.createElementNS('http://www.w3.org/2000/svg', 'line');
                line.setAttribute('x1', prev.x.toFixed(2));
                line.setAttribute('y1', prev.y.toFixed(2));
                line.setAttribute('x2', p.x.toFixed(2));
                line.setAttribute('y2', p.y.toFixed(2));
                line.setAttribute('stroke', '#1565c0');
                line.setAttribute('stroke-width', '1.5');
                line.setAttribute('stroke-dasharray', '6,4');
                fragment.appendChild(line);
                prev = p;
            }
            els.groupOrbit.appendChild(fragment);
        }

        // 更新场景
        var rafId = null;
        function updateScene() {
            if (rafId) cancelAnimationFrame(rafId);
            rafId = requestAnimationFrame(function() {
                var ap = getAprime(state.theta);
                var p = getP(state.theta);

                // A′点
                els.pointAp.setAttribute('cx', ap.x);
                els.pointAp.setAttribute('cy', ap.y);

                // A′标签（θ=0°时隐藏，避免与A重叠）
                if (state.theta < 3) {
                    els.labelAp.style.display = 'none';
                    els.bgAp.style.display = 'none';
                } else {
                    els.labelAp.style.display = 'block';
                    els.bgAp.style.display = 'block';
                    els.labelAp.setAttribute('x', ap.x - 28);
                    els.labelAp.setAttribute('y', ap.y - 14);
                    els.bgAp.setAttribute('x', ap.x - 30);
                    els.bgAp.setAttribute('y', ap.y - 26);
                }

                // A′G线段
                els.lineAg.setAttribute('x1', ap.x);
                els.lineAg.setAttribute('y1', ap.y);

                // 投影系统
                if (state.showProj) {
                    els.lineApProj.setAttribute('x1', ap.x);
                    els.lineApProj.setAttribute('y1', ap.y);
                    els.lineApProj.setAttribute('x2', p.x);
                    els.lineApProj.setAttribute('y2', p.y);
                    els.pointP.setAttribute('cx', p.x);
                    els.pointP.setAttribute('cy', p.y);
                    els.labelP.setAttribute('x', p.x + 15);
                    els.labelP.setAttribute('y', p.y + 15);
                    els.bgP.setAttribute('x', p.x + 13);
                    els.bgP.setAttribute('y', p.y + 2);
                }

                // 截面A′GF
                if (state.showSection) {
                    els.polyAgf.setAttribute('points', ap.x + ',' + ap.y + ' 400,300 306,342');
                }

                // 三棱锥
                if (state.showPyramid) {
                    els.lineApD.setAttribute('x1', ap.x);
                    els.lineApD.setAttribute('y1', ap.y);
                    els.lineApE.setAttribute('x1', ap.x);
                    els.lineApE.setAttribute('y1', ap.y);
                    els.lineApF.setAttribute('x1', ap.x);
                    els.lineApF.setAttribute('y1', ap.y);
                    els.polyPyramidFd.setAttribute('points', ap.x + ',' + ap.y + ' 306,342 370,300');
                    els.polyPyramidFe.setAttribute('points', ap.x + ',' + ap.y + ' 306,342 430,300');
                    els.polyPyramidEd.setAttribute('points', ap.x + ',' + ap.y + ' 430,300 370,300');
                }

                // 角度显示
                els.thetaDisplay.textContent = 'θ = ' + state.theta + '°';
            });
        }

        // 事件绑定
        function bindEvents() {
            els.slider.addEventListener('input', function(e) {
                state.theta = parseInt(e.target.value, 10);
                updateScene();
            });

            els.checkProj.addEventListener('change', function(e) {
                state.showProj = e.target.checked;
                els.groupProj.style.display = state.showProj ? 'block' : 'none';
                updateScene();
            });

            els.checkSection.addEventListener('change', function(e) {
                state.showSection = e.target.checked;
                els.groupSection.style.display = state.showSection ? 'block' : 'none';
                updateScene();
            });

            els.checkPyramid.addEventListener('change', function(e) {
                state.showPyramid = e.target.checked;
                els.groupPyramid.style.display = state.showPyramid ? 'block' : 'none';
                updateScene();
            });
        }

        // 初始化
        function init() {
            cacheElements();
            generateOrbitPath();
            bindEvents();
            updateScene();
        }

        init();
    })();
    </script>
