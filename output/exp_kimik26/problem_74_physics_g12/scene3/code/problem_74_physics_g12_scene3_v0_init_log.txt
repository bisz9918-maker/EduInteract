    <script>
    (function() {
        // Scene 3 实现
        let currentStep = 0;
        const totalSteps = 6;

        const btnPrev = document.getElementById('scene3-btn-prev');
        const btnNext = document.getElementById('scene3-btn-next');
        const indicator = document.getElementById('scene3-step-indicator');

        // 获取所有按步骤分组的元素
        function getStepElements(step) {
            return document.querySelectorAll('.scene3-step-' + step);
        }

        function showElement(el) {
            el.style.display = '';
            el.style.opacity = '1';
        }

        function hideElement(el) {
            el.style.display = 'none';
            el.style.opacity = '0';
        }

        function updateStep() {
            // 更新按钮状态
            btnPrev.disabled = currentStep === 0;
            btnNext.disabled = currentStep === totalSteps - 1;
            indicator.textContent = (currentStep + 1) + ' / ' + totalSteps;

            // 隐藏所有右侧步骤内容
            for (let i = 0; i < totalSteps; i++) {
                const rightGroup = document.getElementById('scene3-right-step-' + i);
                if (rightGroup) {
                    if (i === currentStep) {
                        showElement(rightGroup);
                    } else {
                        hideElement(rightGroup);
                    }
                }
            }

            // 处理左侧步骤相关元素的显隐
            for (let step = 1; step <= 5; step++) {
                const els = getStepElements(step);
                els.forEach(function(el) {
                    if (currentStep >= step) {
                        showElement(el);
                    } else {
                        hideElement(el);
                    }
                });
            }

            // 处理初态线圈/砝码的淡化效果
            const coilInitial = document.getElementById('scene3-coil-initial');
            const weightInitialRect = document.getElementById('scene3-weight-initial-rect');
            if (currentStep >= 1) {
                // 淡化初态
                if (coilInitial) {
                    coilInitial.querySelectorAll('line').forEach(function(line) {
                        line.setAttribute('stroke', '#888');
                        line.setAttribute('stroke-width', '1');
                    });
                }
                if (weightInitialRect) {
                    weightInitialRect.setAttribute('stroke', '#888');
                    weightInitialRect.setAttribute('stroke-width', '1');
                }
            } else {
                // 恢复初态
                if (coilInitial) {
                    coilInitial.querySelectorAll('line').forEach(function(line) {
                        line.setAttribute('stroke', '#000');
                        line.setAttribute('stroke-width', '2');
                    });
                }
                if (weightInitialRect) {
                    weightInitialRect.setAttribute('stroke', '#000');
                    weightInitialRect.setAttribute('stroke-width', '2');
                }
            }

            // 步骤5进一步淡化初态
            if (currentStep === 5) {
                if (coilInitial) {
                    coilInitial.querySelectorAll('line').forEach(function(line) {
                        line.setAttribute('stroke', '#bbb');
                    });
                }
                if (weightInitialRect) {
                    weightInitialRect.setAttribute('stroke', '#bbb');
                }
            } else if (currentStep >= 1) {
                if (coilInitial) {
                    coilInitial.querySelectorAll('line').forEach(function(line) {
                        line.setAttribute('stroke', '#888');
                    });
                }
                if (weightInitialRect) {
                    weightInitialRect.setAttribute('stroke', '#888');
                }
            }
        }

        btnPrev.addEventListener('click', function() {
            if (currentStep > 0) {
                currentStep--;
                updateStep();
            }
        });

        btnNext.addEventListener('click', function() {
            if (currentStep < totalSteps - 1) {
                currentStep++;
                updateStep();
            }
        });

        // 初始化
        updateStep();
    })();
    </script>
