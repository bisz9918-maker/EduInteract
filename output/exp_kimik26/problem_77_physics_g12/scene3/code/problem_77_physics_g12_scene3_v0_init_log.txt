    <script>
    (function() {
        // Scene 3 实现
        var currentStep = 1;
        var totalSteps = 3;

        var prevBtn = document.getElementById('scene3-prev-btn');
        var nextBtn = document.getElementById('scene3-next-btn');
        var indicator = document.getElementById('scene3-step-indicator');

        var groups = [
            document.getElementById('scene3-step1-group'),
            document.getElementById('scene3-step2-group'),
            document.getElementById('scene3-step3-group')
        ];

        function updateUI() {
            // 更新步骤组显示
            for (var i = 0; i < groups.length; i++) {
                if (groups[i]) {
                    if (i + 1 === currentStep) {
                        groups[i].style.display = 'block';
                    } else {
                        groups[i].style.display = 'none';
                    }
                }
            }
            // 更新指示器
            indicator.textContent = '步骤 ' + currentStep + ' / ' + totalSteps;
            // 更新按钮状态
            prevBtn.disabled = (currentStep === 1);
            nextBtn.disabled = (currentStep === totalSteps);
        }

        prevBtn.addEventListener('click', function() {
            if (currentStep > 1) {
                currentStep--;
                updateUI();
            }
        });

        nextBtn.addEventListener('click', function() {
            if (currentStep < totalSteps) {
                currentStep++;
                updateUI();
            }
        });

        // 初始化
        updateUI();
    })();
    </script>
