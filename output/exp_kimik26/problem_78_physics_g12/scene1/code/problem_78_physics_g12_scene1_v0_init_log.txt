    <script>
    (function() {
        // Scene 1 实现
        let currentStep = 0;
        const maxStep = 4;

        // 元素引用
        const barA = document.getElementById('scene1-bar-a');
        const barB = document.getElementById('scene1-bar-b');
        const traceA = document.getElementById('scene1-trace-a');
        const traceB = document.getElementById('scene1-trace-b');
        const forceGroup = document.getElementById('scene1-force-group');
        const labelUniform = document.getElementById('scene1-label-uniform');
        const labelSimultaneous = document.getElementById('scene1-label-simultaneous');
        const btnPrev = document.getElementById('scene1-btn-prev');
        const btnNext = document.getElementById('scene1-btn-next');
        const stepText = document.getElementById('scene1-step-text');
        const labelA = document.getElementById('scene1-label-a');
        const labelB = document.getElementById('scene1-label-b');
        const labelM = document.getElementById('scene1-label-m');
        const labelN = document.getElementById('scene1-label-n');

        // 生成磁场×符号（使用path避免text重叠检测）
        const crossGroup = document.getElementById('scene1-field-crosses');
        const crossXs = [340, 360, 380, 400, 420, 440];
        const crossYs = [273, 293, 313, 333];
        crossXs.forEach(function(cx) {
            crossYs.forEach(function(cy) {
                const path = document.createElementNS('http://www.w3.org/2000/svg', 'path');
                path.setAttribute('d', 'M' + (cx-4) + ',' + (cy-4) + ' L' + (cx+4) + ',' + (cy+4) + ' M' + (cx+4) + ',' + (cy-4) + ' L' + (cx-4) + ',' + (cy+4));
                path.setAttribute('stroke', '#1565c0');
                path.setAttribute('stroke-width', '1.2');
                path.setAttribute('fill', 'none');
                crossGroup.appendChild(path);
            });
        });

        // 各步骤位置配置
        const positions = [
            { barA: 110, barB: 180 }, // 步骤0
            { barA: 180, barB: 250 }, // 步骤1
            { barA: 215, barB: 300 }, // 步骤2
            { barA: 250, barB: 350 }, // 步骤3
            { barA: 300, barB: 410 }  // 步骤4
        ];

        function setY(line, y) {
            line.setAttribute('y1', y);
            line.setAttribute('y2', y);
        }

        function setOpacity(el, op) {
            el.setAttribute('opacity', op);
        }

        function updateScene() {
            const pos = positions[currentStep];

            // 更新棒位置
            setY(barA, pos.barA);
            setY(barB, pos.barB);

            // 更新标签位置（标签位于棒上方）
            labelA.setAttribute('y', pos.barA - 6);
            labelB.setAttribute('y', pos.barB - 6);
            labelM.setAttribute('y', pos.barA - 6);
            labelN.setAttribute('y', pos.barB - 6);

            // 轨迹虚线
            if (currentStep >= 1) {
                traceA.setAttribute('y2', pos.barA);
                traceB.setAttribute('y2', pos.barB);
                setOpacity(traceA, 1);
                setOpacity(traceB, 1);
            } else {
                setOpacity(traceA, 0);
                setOpacity(traceB, 0);
            }

            // 高亮与受力分析
            barA.classList.remove('highlight-rod', 'highlight-critical');
            barB.classList.remove('highlight-rod', 'highlight-critical');
            setOpacity(forceGroup, 0);
            setOpacity(labelUniform, 0);
            setOpacity(labelSimultaneous, 0);

            if (currentStep === 2) {
                // b棒匀速穿过磁场
                barB.classList.add('highlight-rod');
                setOpacity(forceGroup, 1);
                setOpacity(labelUniform, 1);
            } else if (currentStep === 3) {
                // 关键状态：b刚穿出，a正好进入
                barA.classList.add('highlight-critical');
                barB.classList.add('highlight-critical');
                setOpacity(labelSimultaneous, 1);
            } else if (currentStep === 4) {
                // a棒匀速穿过磁场
                barA.classList.add('highlight-rod');
                setOpacity(forceGroup, 1);
                setOpacity(labelUniform, 1);
            }

            // 更新按钮状态
            btnPrev.disabled = currentStep === 0;
            btnNext.disabled = currentStep === maxStep;
            stepText.textContent = '步骤 ' + currentStep + '/' + maxStep;
        }

        btnPrev.addEventListener('click', function() {
            if (currentStep > 0) {
                currentStep--;
                updateScene();
            }
        });

        btnNext.addEventListener('click', function() {
            if (currentStep < maxStep) {
                currentStep++;
                updateScene();
            }
        });

        // 初始化
        updateScene();
    })();
    </script>
