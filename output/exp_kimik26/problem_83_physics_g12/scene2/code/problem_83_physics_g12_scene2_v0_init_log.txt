    <script>
    (function() {
        // Scene 2 实现
        var currentStep = 0;
        var totalSteps = 4;

        var btnPrev = document.getElementById('scene2-btn-prev');
        var btnNext = document.getElementById('scene2-btn-next');
        var indicator = document.getElementById('scene2-step-indicator');

        var blockGroup = document.getElementById('scene2-block-group');
        var spring = document.getElementById('scene2-spring');
        var groupStep0 = document.getElementById('scene2-group-step0');
        var groupStep1 = document.getElementById('scene2-group-step1');
        var groupStep2 = document.getElementById('scene2-group-step2');
        var groupStep3 = document.getElementById('scene2-group-step3');
        var labelMass = document.getElementById('scene2-label-mass');
        var groundLabels = document.getElementById('scene2-ground-labels');

        // 弹簧绘制函数
        function drawSpring(startX, endX, y, amp, period) {
            var d = 'M' + startX + ',' + y;
            var x = startX;
            var up = true;
            while (x + period <= endX + 0.1) {
                if (up) {
                    d += ' q' + (period/4) + ',' + (-amp) + ' ' + (period/2) + ',0';
                    d += ' t' + period + ',0';
                } else {
                    d += ' q' + (period/4) + ',' + amp + ' ' + (period/2) + ',0';
                    d += ' t' + period + ',0';
                }
                x += period;
                up = !up;
            }
            // 收尾到endX
            var remaining = endX - x;
            if (remaining > 2) {
                var half = remaining / 2;
                if (up) {
                    d += ' q' + (half/2) + ',' + (-amp) + ' ' + half + ',0';
                    d += ' t' + remaining + ',0';
                } else {
                    d += ' q' + (half/2) + ',' + amp + ' ' + half + ',0';
                    d += ' t' + remaining + ',0';
                }
            }
            spring.setAttribute('d', d);
        }

        // 步骤配置
        var stepConfigs = [
            { // 步骤0：子弹射入后瞬间
                blockX: 330,
                springStart: 160, springEnd: 330, springAmp: 8, springPeriod: 20,
                showStep0: true, showStep1: false, showStep2: false, showStep3: false,
                massLabelX: 350, showGroundLabels: true
            },
            { // 步骤1：弹簧压缩中
                blockX: 310,
                springStart: 160, springEnd: 310, springAmp: 8, springPeriod: 18,
                showStep0: true, showStep1: true, showStep2: false, showStep3: false,
                massLabelX: 330, showGroundLabels: true
            },
            { // 步骤2：弹簧压缩到最短
                blockX: 280,
                springStart: 160, springEnd: 280, springAmp: 8, springPeriod: 15,
                showStep0: false, showStep1: false, showStep2: true, showStep3: false,
                massLabelX: 300, showGroundLabels: true
            },
            { // 步骤3：能量转化分析
                blockX: 280,
                springStart: 160, springEnd: 280, springAmp: 8, springPeriod: 15,
                showStep0: false, showStep1: false, showStep2: true, showStep3: true,
                massLabelX: 300, showGroundLabels: false
            }
        ];

        function updateStep(step) {
            var cfg = stepConfigs[step];

            // 滑块位置动画（使用CSS transition）
            blockGroup.style.transition = 'transform 300ms ease-out';
            blockGroup.setAttribute('transform', 'translate(' + cfg.blockX + ',0)');

            // 弹簧
            drawSpring(cfg.springStart, cfg.springEnd, 235, cfg.springAmp, cfg.springPeriod);

            // 质量标签位置
            labelMass.setAttribute('x', cfg.massLabelX);

            // 显隐控制
            groupStep0.style.display = cfg.showStep0 ? 'block' : 'none';
            groupStep1.style.display = cfg.showStep1 ? 'block' : 'none';
            groupStep2.style.display = cfg.showStep2 ? 'block' : 'none';
            groupStep3.style.display = cfg.showStep3 ? 'block' : 'none';
            groundLabels.style.display = cfg.showGroundLabels ? 'block' : 'none';

            // 更新步骤1中的相对运动高亮位置
            if (cfg.showStep1) {
                var highlight = document.getElementById('scene2-highlight-relative');
                if (highlight) highlight.setAttribute('x', cfg.blockX);
            }

            // 按钮状态
            btnPrev.disabled = (step === 0);
            btnNext.disabled = (step === totalSteps - 1);
            indicator.textContent = '步骤 ' + (step + 1) + '/' + totalSteps;
        }

        btnPrev.addEventListener('click', function() {
            if (currentStep > 0) {
                currentStep--;
                updateStep(currentStep);
            }
        });

        btnNext.addEventListener('click', function() {
            if (currentStep < totalSteps - 1) {
                currentStep++;
                updateStep(currentStep);
            }
        });

        // 初始化
        updateStep(0);
    })();
    </script>
