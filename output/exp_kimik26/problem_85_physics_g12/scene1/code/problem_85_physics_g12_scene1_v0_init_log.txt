    <script>
    (function() {
        // Scene 1 实现

        // ===== 状态变量 =====
        let currentStep = 0;
        let isAnimating = false;
        const totalSteps = 6;

        // ===== DOM 元素引用 =====
        const btnPrev = document.getElementById('scene1-btn-prev');
        const btnNext = document.getElementById('scene1-btn-next');
        const stepIndicator = document.getElementById('scene1-step-indicator');

        // 活塞
        const pistonA = document.getElementById('scene1-piston-a');
        const pistonB = document.getElementById('scene1-piston-b');

        // 气体区域
        const gasO2 = document.getElementById('scene1-gas-o2');
        const gasAtm = document.getElementById('scene1-gas-atm');
        const gasN2A = document.getElementById('scene1-gas-n2-a');
        const gasN2B = document.getElementById('scene1-gas-n2-b');

        // 电阻丝
        const heaterA = document.getElementById('scene1-heater-a');
        const heaterB = document.getElementById('scene1-heater-b');

        // 标签
        const processLabel = document.getElementById('scene1-process-label');
        const dimH16Group = document.getElementById('scene1-dim-h16-group');

        // 状态面板 - HTML元素
        const n2P = document.getElementById('scene1-n2-p');
        const n2V = document.getElementById('scene1-n2-v');
        const n2T = document.getElementById('scene1-n2-t');
        const o2P = document.getElementById('scene1-o2-p');
        const o2V = document.getElementById('scene1-o2-v');
        const o2T = document.getElementById('scene1-o2-t');
        const conclusion = document.getElementById('scene1-conclusion');

        // ===== 工具函数 =====
        function setText(el, text) {
            if (el) el.textContent = text;
        }
        function show(el) {
            if (el) el.style.display = '';
        }
        function hide(el) {
            if (el) el.style.display = 'none';
        }
        function setY(el, y) {
            if (el) el.setAttribute('y', y);
        }
        function setHeight(el, h) {
            if (el) el.setAttribute('height', h);
        }
        function addClass(el, cls) {
            if (el) el.classList.add(cls);
        }
        function removeClass(el, cls) {
            if (el) el.classList.remove(cls);
        }

        // 动画：ease-out 插值
        function animateValue(start, end, duration, callback, onDone) {
            const startTime = performance.now();
            isAnimating = true;
            updateButtons();

            function frame(now) {
                const elapsed = now - startTime;
                const t = Math.min(elapsed / duration, 1);
                // ease-out cubic
                const ease = 1 - Math.pow(1 - t, 3);
                const val = start + (end - start) * ease;
                callback(val);
                if (t < 1) {
                    requestAnimationFrame(frame);
                } else {
                    isAnimating = false;
                    updateButtons();
                    if (onDone) onDone();
                }
            }
            requestAnimationFrame(frame);
        }

        // 更新按钮状态
        function updateButtons() {
            btnPrev.disabled = (currentStep === 0) || isAnimating;
            btnNext.disabled = (currentStep === totalSteps - 1) || isAnimating;
            stepIndicator.textContent = '步骤 ' + (currentStep + 1) + ' / ' + totalSteps;
        }

        // ===== 步骤配置 =====
        function applyStep(step) {
            // 清除高亮
            removeClass(n2T, 'highlight-green');
            removeClass(n2P, 'highlight-green');
            removeClass(o2P, 'highlight-green');
            removeClass(n2T, 'highlight-text');
            removeClass(n2P, 'highlight-text');
            removeClass(o2P, 'highlight-text');

            // 清除加热动画
            removeClass(heaterA, 'heating');
            removeClass(heaterB, 'heating');

            // 清除过程标注
            hide(processLabel);
            setText(processLabel, '');

            // 清除H/16标注
            hide(dimH16Group);

            // 清除结论
            hide(conclusion);
            setText(conclusion, '');

            switch(step) {
                case 0:
                    // 装置结构：初始位置，空面板
                    setY(pistonA, 192);
                    setY(pistonB, 267);
                    setHeight(gasO2, 72);
                    setHeight(gasAtm, 147);
                    setY(gasN2A, 198);
                    setHeight(gasN2A, 222);
                    setY(gasN2B, 273);
                    setHeight(gasN2B, 147);
                    show(gasAtm);
                    setText(n2P, '');
                    setText(n2V, '');
                    setText(n2T, '');
                    setText(o2P, '');
                    setText(o2V, '');
                    setText(o2T, '');
                    break;

                case 1:
                    // 初始平衡态
                    setY(pistonA, 192);
                    setY(pistonB, 267);
                    setHeight(gasO2, 72);
                    setHeight(gasAtm, 147);
                    setY(gasN2A, 198);
                    setHeight(gasN2A, 222);
                    setY(gasN2B, 273);
                    setHeight(gasN2B, 147);
                    show(gasAtm);
                    setText(n2P, 'P = P₀');
                    setText(n2V, 'V = 7SH/2');
                    setText(n2T, 'T = 280 K');
                    setText(o2P, 'P = P₀');
                    setText(o2V, 'V = SH');
                    setText(o2T, 'T = 280 K');
                    break;

                case 2:
                    // 过程(1)加热 - 活塞b上升动画
                    setY(pistonA, 192);
                    // 活塞b动画到顶部
                    animateValue(267, 117, 800, function(y) {
                        setY(pistonB, y);
                        // 更新B部氮气
                        setY(gasN2B, y + 6);
                        setHeight(gasN2B, 420 - (y + 6));
                        // 更新大气区
                        setHeight(gasAtm, y - 120);
                    }, null);
                    setHeight(gasO2, 72);
                    setY(gasN2A, 198);
                    setHeight(gasN2A, 222);
                    show(gasAtm);
                    setText(n2P, 'P = P₀');
                    setText(n2V, 'V = 7SH/2');
                    setText(n2T, 'T = 280 K ↑');
                    setText(o2P, 'P = P₀');
                    setText(o2V, 'V = SH');
                    setText(o2T, 'T = 280 K');
                    show(processLabel);
                    setText(processLabel, '氮气等压膨胀');
                    addClass(heaterA, 'heating');
                    addClass(heaterB, 'heating');
                    break;

                case 3:
                    // 过程(1)末态
                    setY(pistonA, 192);
                    setY(pistonB, 117);
                    setHeight(gasO2, 72);
                    setHeight(gasAtm, 0);
                    hide(gasAtm);
                    setY(gasN2A, 198);
                    setHeight(gasN2A, 222);
                    setY(gasN2B, 123);
                    setHeight(gasN2B, 297);
                    setText(n2P, 'P = P₀');
                    setText(n2V, 'V = 4SH');
                    setText(n2T, 'T = 320 K');
                    addClass(n2T, 'highlight-text');
                    setText(o2P, 'P = P₀');
                    setText(o2V, 'V = SH');
                    setText(o2T, 'T = 280 K（未变）');
                    show(processLabel);
                    setText(processLabel, '活塞b到达顶部');
                    break;

                case 4:
                    // 过程(2)加热 - 活塞a上升动画
                    setY(pistonB, 117);
                    setHeight(gasAtm, 0);
                    hide(gasAtm);
                    setY(gasN2B, 123);
                    setHeight(gasN2B, 297);
                    // 活塞a动画上升H/16 (从192到173)
                    animateValue(192, 173, 600, function(y) {
                        setY(pistonA, y);
                        // 更新氧气区
                        setHeight(gasO2, y - 120);
                        // 更新氮气A部
                        setY(gasN2A, y + 6);
                        setHeight(gasN2A, 420 - (y + 6));
                    }, null);
                    setText(n2P, 'P = P₀');
                    setText(n2V, 'V = 4SH');
                    setText(n2T, 'T = 320 K ↑');
                    setText(o2P, 'P = P₀');
                    setText(o2V, 'V = SH ↓');
                    setText(o2T, 'T = 280 K');
                    show(processLabel);
                    setText(processLabel, '氧气等温压缩');
                    show(dimH16Group);
                    addClass(heaterA, 'heating');
                    addClass(heaterB, 'heating');
                    break;

                case 5:
                    // 过程(2)末态
                    setY(pistonA, 173);
                    setY(pistonB, 117);
                    setHeight(gasO2, 53);
                    setHeight(gasAtm, 0);
                    hide(gasAtm);
                    setY(gasN2A, 179);
                    setHeight(gasN2A, 241);
                    setY(gasN2B, 123);
                    setHeight(gasN2B, 297);
                    setText(n2P, 'P = 4P₀/3');
                    addClass(n2P, 'highlight-text');
                    setText(n2V, 'V = 17SH/4');
                    setText(n2T, 'T = 320 K');
                    setText(o2P, 'P = 4P₀/3');
                    addClass(o2P, 'highlight-text');
                    setText(o2V, 'V = 3SH/4');
                    setText(o2T, 'T = 280 K');
                    show(processLabel);
                    setText(processLabel, '氧气等温压缩');
                    show(dimH16Group);
                    show(conclusion);
                    setText(conclusion, '(1) 氮气温度 T = 320 K\n(2) 氧气压强 Pₒ = 4P₀/3');
                    break;
            }
        }

        // ===== 事件绑定 =====
        btnPrev.addEventListener('click', function() {
            if (currentStep > 0 && !isAnimating) {
                currentStep--;
                applyStep(currentStep);
                updateButtons();
            }
        });

        btnNext.addEventListener('click', function() {
            if (currentStep < totalSteps - 1 && !isAnimating) {
                currentStep++;
                applyStep(currentStep);
                updateButtons();
            }
        });

        // ===== 初始化 =====
        applyStep(0);
        updateButtons();

    })();
    </script>
