    <script>
    (function() {
        // Scene 3 实现：功能关系与拉力做功分析
        
        let currentStep = 0;
        const totalSteps = 3;
        
        const btnPrev = document.getElementById('scene3-btn-prev');
        const btnNext = document.getElementById('scene3-btn-next');
        const indicator = document.getElementById('scene3-step-indicator');
        
        const flowHtml = document.getElementById('scene3-flow-html');
        const step2Group = document.getElementById('scene3-step2-group');
        const step3Group = document.getElementById('scene3-step3-group');
        
        function updateStep() {
            // 更新指示器
            indicator.textContent = '步骤 ' + currentStep + '/' + totalSteps;
            
            // 更新按钮状态
            btnPrev.disabled = currentStep === 0;
            btnNext.disabled = currentStep === totalSteps;
            
            // 控制元素显示
            flowHtml.style.display = currentStep >= 1 ? 'block' : 'none';
            step2Group.style.display = currentStep >= 2 ? 'block' : 'none';
            step3Group.style.display = currentStep >= 3 ? 'block' : 'none';
            
            // 步骤3高亮动画：公式行5短暂放大后回弹
            if (currentStep === 3) {
                const formula5 = document.getElementById('scene3-formula5');
                if (formula5) {
                    formula5.setAttribute('font-size', '17');
                    setTimeout(function() {
                        formula5.setAttribute('font-size', '15');
                    }, 300);
                }
            }
        }
        
        btnPrev.addEventListener('click', function() {
            if (currentStep > 0) {
                currentStep--;
                updateStep();
            }
        });
        
        btnNext.addEventListener('click', function() {
            if (currentStep < totalSteps) {
                currentStep++;
                updateStep();
            }
        });
        
        // 初始化
        updateStep();
    })();
    </script>
