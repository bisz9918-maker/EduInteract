    <script>
    (function() {
        // Scene 1 实现
        const STEP_COUNT = 5;
        let currentStep = 0;

        // 元素引用
        const elCurrentPath = document.getElementById('scene1-current-path');
        const elL1Bulb = document.getElementById('scene1-l1-bulb');
        const elL1Glow = document.getElementById('scene1-l1-glow');
        const elL2Bulb = document.getElementById('scene1-l2-bulb');
        const elL2Glow = document.getElementById('scene1-l2-glow');
        const elShortL1 = document.getElementById('scene1-short-l1');
        const elShortL2 = document.getElementById('scene1-short-l2');
        const elBreakL1 = document.getElementById('scene1-break-l1');
        const elBreakL2 = document.getElementById('scene1-break-l2');
        const elVoltReading = document.getElementById('scene1-voltmeter-reading');
        const elValFault = document.getElementById('scene1-val-fault');
        const elValL1 = document.getElementById('scene1-val-l1');
        const elValL2 = document.getElementById('scene1-val-l2');
        const elValVolt = document.getElementById('scene1-val-volt');
        const elValMatch = document.getElementById('scene1-val-match');
        const btnPrev = document.getElementById('scene1-btn-prev');
        const btnNext = document.getElementById('scene1-btn-next');
        const elStepIndicator = document.getElementById('scene1-step-indicator');
        const rowFault = document.getElementById('scene1-row-fault');
        const rowL1 = document.getElementById('scene1-row-l1');
        const rowL2 = document.getElementById('scene1-row-l2');
        const rowVolt = document.getElementById('scene1-row-volt');
        const rowMatch = document.getElementById('scene1-row-match');

        // 步骤配置
        const steps = [
            { // 步骤0: 正常电路
                l1On: true, l2On: true,
                current: true,
                shortL1: false, shortL2: false,
                breakL1: false, breakL2: false,
                voltText: '有示数',
                fault: '正常', l1Text: '亮', l2Text: '亮', volt: '有示数', match: '—',
                faultColor: '#000000', matchColor: '#000000',
                highlightRow: null
            },
            { // 步骤1: L₁短路
                l1On: false, l2On: true,
                current: true,
                shortL1: true, shortL2: false,
                breakL1: false, breakL2: false,
                voltText: '有示数（U≈U电源）',
                fault: 'L₁短路', l1Text: '不亮', l2Text: '亮', volt: '有示数', match: '符合题意',
                faultColor: '#2e7d32', matchColor: '#2e7d32',
                highlightRow: 'fault'
            },
            { // 步骤2: L₁断路
                l1On: false, l2On: false,
                current: false,
                shortL1: false, shortL2: false,
                breakL1: true, breakL2: false,
                voltText: '0V',
                fault: 'L₁断路', l1Text: '不亮', l2Text: '不亮', volt: '0V', match: '不符合',
                faultColor: '#c62828', matchColor: '#c62828',
                highlightRow: 'fault'
            },
            { // 步骤3: L₂短路
                l1On: true, l2On: false,
                current: true,
                shortL1: false, shortL2: true,
                breakL1: false, breakL2: false,
                voltText: '0V',
                fault: 'L₂短路', l1Text: '亮', l2Text: '不亮', volt: '0V', match: '不符合',
                faultColor: '#c62828', matchColor: '#c62828',
                highlightRow: 'fault'
            },
            { // 步骤4: L₂断路
                l1On: false, l2On: false,
                current: false,
                shortL1: false, shortL2: false,
                breakL1: false, breakL2: true,
                voltText: '有示数（U≈U电源）',
                fault: 'L₂断路', l1Text: '不亮', l2Text: '不亮', volt: '有示数', match: '不符合',
                faultColor: '#c62828', matchColor: '#c62828',
                highlightRow: 'fault'
            }
        ];

        function setDisplay(el, show) {
            if (!el) return;
            el.style.display = show ? 'block' : 'none';
        }

        function clearHighlight() {
            [rowFault, rowL1, rowL2, rowVolt, rowMatch].forEach(r => r.classList.remove('active'));
        }

        function updateScene() {
            const cfg = steps[currentStep];

            // 灯泡状态
            elL1Bulb.setAttribute('fill', cfg.l1On ? '#ffeb3b' : '#e0e0e0');
            elL1Glow.setAttribute('fill-opacity', cfg.l1On ? '0.2' : '0');
            elL2Bulb.setAttribute('fill', cfg.l2On ? '#ffeb3b' : '#e0e0e0');
            elL2Glow.setAttribute('fill-opacity', cfg.l2On ? '0.2' : '0');

            // 电流路径
            setDisplay(elCurrentPath, cfg.current);

            // 故障标记
            setDisplay(elShortL1, cfg.shortL1);
            setDisplay(elShortL2, cfg.shortL2);
            setDisplay(elBreakL1, cfg.breakL1);
            setDisplay(elBreakL2, cfg.breakL2);

            // 电压表示数
            elVoltReading.textContent = cfg.voltText;

            // 面板值
            elValFault.textContent = cfg.fault;
            elValFault.style.color = cfg.faultColor;
            elValL1.textContent = cfg.l1Text;
            elValL2.textContent = cfg.l2Text;
            elValVolt.textContent = cfg.volt;
            elValMatch.textContent = cfg.match;
            elValMatch.style.color = cfg.matchColor;

            // 面板高亮
            clearHighlight();
            if (cfg.highlightRow === 'fault') rowFault.classList.add('active');

            // 按钮状态
            btnPrev.disabled = currentStep === 0;
            btnNext.disabled = currentStep === STEP_COUNT - 1;
            elStepIndicator.textContent = '步骤 ' + (currentStep + 1) + '/' + STEP_COUNT;
        }

        btnPrev.addEventListener('click', function() {
            if (currentStep > 0) {
                currentStep--;
                updateScene();
            }
        });

        btnNext.addEventListener('click', function() {
            if (currentStep < STEP_COUNT - 1) {
                currentStep++;
                updateScene();
            }
        });

        // 初始化
        updateScene();
    })();
    </script>
