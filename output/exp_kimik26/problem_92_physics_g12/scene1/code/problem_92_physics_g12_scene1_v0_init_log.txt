    <script>
    (function() {
        // Scene 1 实现 - 带电粒子在组合电场中的运动
        let currentStep = 0;
        const totalSteps = 5;

        // 元素引用
        const btnPrev = document.getElementById('scene1-btn-prev');
        const btnNext = document.getElementById('scene1-btn-next');
        const stepText = document.getElementById('scene1-step-text');

        // SVG元素
        const els = {
            particleGroup: document.getElementById('scene1-particle-group'),
            v0Arrow: document.getElementById('scene1-v0-arrow'),
            plateFieldLines: document.getElementById('scene1-plate-field-lines'),
            chargeFieldLines: document.getElementById('scene1-charge-field-lines'),
            highlightPlate: document.getElementById('scene1-highlight-plate'),
            highlightFree: document.getElementById('scene1-highlight-free'),
            highlightCharge: document.getElementById('scene1-highlight-charge'),
            traj1: document.getElementById('scene1-trajectory-1'),
            traj2: document.getElementById('scene1-trajectory-2'),
            traj3: document.getElementById('scene1-trajectory-3'),
            exitPoint: document.getElementById('scene1-exit-point'),
            yAnnotation: document.getElementById('scene1-y-annotation'),
            thetaAnnotation: document.getElementById('scene1-theta-annotation'),
            labelQReq: document.getElementById('scene1-label-q-req')
        };

        // 显示/隐藏工具函数
        function show(el) { if(el) { el.style.display = ''; el.style.opacity = '1'; } }
        function hide(el) { if(el) { el.style.display = 'none'; } }
        function setOpacity(el, val) { if(el) { el.style.opacity = String(val); } }

        // 轨迹画出动画
        function animatePath(pathEl, duration) {
            if (!pathEl) return;
            show(pathEl);
            const length = pathEl.getTotalLength();
            pathEl.style.strokeDasharray = length + ' ' + length;
            pathEl.style.strokeDashoffset = length;
            const startTime = performance.now();
            function step(now) {
                const elapsed = now - startTime;
                const progress = Math.min(elapsed / duration, 1);
                // ease-out
                const ease = 1 - Math.pow(1 - progress, 3);
                pathEl.style.strokeDashoffset = length * (1 - ease);
                if (progress < 1) {
                    requestAnimationFrame(step);
                } else {
                    pathEl.style.strokeDasharray = '';
                    pathEl.style.strokeDashoffset = '';
                }
            }
            requestAnimationFrame(step);
        }

        function animateLine(lineEl, duration) {
            if (!lineEl) return;
            show(lineEl);
            const x1 = parseFloat(lineEl.getAttribute('x1'));
            const y1 = parseFloat(lineEl.getAttribute('y1'));
            const x2 = parseFloat(lineEl.getAttribute('x2'));
            const y2 = parseFloat(lineEl.getAttribute('y2'));
            const length = Math.sqrt((x2-x1)*(x2-x1) + (y2-y1)*(y2-y1));
            lineEl.style.strokeDasharray = length + ' ' + length;
            lineEl.style.strokeDashoffset = length;
            const startTime = performance.now();
            function step(now) {
                const elapsed = now - startTime;
                const progress = Math.min(elapsed / duration, 1);
                const ease = 1 - Math.pow(1 - progress, 3);
                lineEl.style.strokeDashoffset = length * (1 - ease);
                if (progress < 1) {
                    requestAnimationFrame(step);
                } else {
                    lineEl.style.strokeDasharray = '';
                    lineEl.style.strokeDashoffset = '';
                }
            }
            requestAnimationFrame(step);
        }

        // 渲染当前步骤
        function render() {
            stepText.textContent = '步骤 ' + currentStep + '/5';
            btnPrev.classList.toggle('disabled', currentStep === 0);
            btnNext.classList.toggle('disabled', currentStep === totalSteps);

            // 步骤0: 初始装置与已知条件
            if (currentStep === 0) {
                hide(els.particleGroup);
                hide(els.v0Arrow);
                hide(els.plateFieldLines);
                hide(els.chargeFieldLines);
                setOpacity(els.highlightPlate, 0);
                setOpacity(els.highlightFree, 0);
                setOpacity(els.highlightCharge, 0);
                hide(els.traj1);
                hide(els.traj2);
                hide(els.traj3);
                hide(els.exitPoint);
                hide(els.yAnnotation);
                hide(els.thetaAnnotation);
                hide(els.labelQReq);
                // 重置轨迹样式
                [els.traj1, els.traj2, els.traj3].forEach(t => {
                    if(t) { t.style.strokeDasharray = ''; t.style.strokeDashoffset = ''; t.style.strokeWidth = '1.5px'; }
                });
            }
            // 步骤1: 高亮平行板，电场线，粒子，v₀
            else if (currentStep === 1) {
                show(els.particleGroup);
                show(els.v0Arrow);
                show(els.plateFieldLines);
                hide(els.chargeFieldLines);
                setOpacity(els.highlightPlate, 0.15);
                setOpacity(els.highlightFree, 0);
                setOpacity(els.highlightCharge, 0);
                hide(els.traj1);
                hide(els.traj2);
                hide(els.traj3);
                hide(els.exitPoint);
                hide(els.yAnnotation);
                hide(els.thetaAnnotation);
                hide(els.labelQReq);
            }
            // 步骤2: 轨迹1画出
            else if (currentStep === 2) {
                show(els.particleGroup);
                show(els.v0Arrow);
                show(els.plateFieldLines);
                hide(els.chargeFieldLines);
                setOpacity(els.highlightPlate, 0.15);
                setOpacity(els.highlightFree, 0);
                setOpacity(els.highlightCharge, 0);
                show(els.exitPoint);
                hide(els.traj2);
                hide(els.traj3);
                hide(els.yAnnotation);
                hide(els.thetaAnnotation);
                hide(els.labelQReq);
                // 轨迹1动画
                animatePath(els.traj1, 600);
            }
            // 步骤3: 轨迹2画出，高亮无电场区
            else if (currentStep === 3) {
                show(els.particleGroup);
                show(els.v0Arrow);
                show(els.plateFieldLines);
                hide(els.chargeFieldLines);
                setOpacity(els.highlightPlate, 0.15);
                setOpacity(els.highlightFree, 0.15);
                setOpacity(els.highlightCharge, 0);
                show(els.exitPoint);
                show(els.traj1);
                hide(els.traj3);
                hide(els.yAnnotation);
                hide(els.thetaAnnotation);
                hide(els.labelQReq);
                // 确保轨迹1已显示
                els.traj1.style.strokeDasharray = '';
                els.traj1.style.strokeDashoffset = '';
                // 轨迹2动画
                animateLine(els.traj2, 600);
            }
            // 步骤4: 轨迹3画出，点电荷电场线，高亮点电荷区
            else if (currentStep === 4) {
                show(els.particleGroup);
                show(els.v0Arrow);
                show(els.plateFieldLines);
                show(els.chargeFieldLines);
                setOpacity(els.highlightPlate, 0.15);
                setOpacity(els.highlightFree, 0.15);
                setOpacity(els.highlightCharge, 0.15);
                show(els.exitPoint);
                show(els.traj1);
                show(els.traj2);
                hide(els.yAnnotation);
                hide(els.thetaAnnotation);
                hide(els.labelQReq);
                // 确保前两条轨迹已显示
                els.traj1.style.strokeDasharray = '';
                els.traj1.style.strokeDashoffset = '';
                els.traj2.style.strokeDasharray = '';
                els.traj2.style.strokeDashoffset = '';
                // 轨迹3动画
                animatePath(els.traj3, 800);
            }
            // 步骤5: 待求量标注，轨迹加粗，高亮淡出
            else if (currentStep === 5) {
                show(els.particleGroup);
                show(els.v0Arrow);
                show(els.plateFieldLines);
                show(els.chargeFieldLines);
                setOpacity(els.highlightPlate, 0.05);
                setOpacity(els.highlightFree, 0.05);
                setOpacity(els.highlightCharge, 0.05);
                show(els.exitPoint);
                show(els.traj1);
                show(els.traj2);
                show(els.traj3);
                show(els.yAnnotation);
                show(els.thetaAnnotation);
                show(els.labelQReq);
                // 轨迹加粗
                [els.traj1, els.traj2, els.traj3].forEach(t => {
                    if(t) t.style.strokeWidth = '2px';
                });
                // 确保轨迹无dash残留
                els.traj1.style.strokeDasharray = '';
                els.traj1.style.strokeDashoffset = '';
                els.traj2.style.strokeDasharray = '';
                els.traj2.style.strokeDashoffset = '';
                els.traj3.style.strokeDasharray = '';
                els.traj3.style.strokeDashoffset = '';
            }
        }

        // 按钮事件
        btnPrev.addEventListener('click', function() {
            if (currentStep > 0) {
                currentStep--;
                render();
            }
        });
        btnNext.addEventListener('click', function() {
            if (currentStep < totalSteps) {
                currentStep++;
                render();
            }
        });

        // 初始渲染
        render();
    })();
    </script>
</body>
