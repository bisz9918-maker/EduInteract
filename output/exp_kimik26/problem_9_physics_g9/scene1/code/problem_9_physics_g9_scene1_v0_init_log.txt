    <script>
    (function() {
        // Scene 1 实现
        let currentStep = 0;
        const totalSteps = 3;

        const btnPrev = document.getElementById('scene1-btn-prev');
        const btnNext = document.getElementById('scene1-btn-next');
        const s1Blade = document.getElementById('scene1-s1-blade');
        const s2Blade = document.getElementById('scene1-s2-blade');
        const hintText = document.getElementById('scene1-hint');
        const panelContent = document.getElementById('scene1-panel-content');
        const l1Glow = document.getElementById('scene1-l1-glow');
        const l2Glow = document.getElementById('scene1-l2-glow');

        // L2相关元素（步骤1需要淡化）
        const l2Group = document.getElementById('scene1-l2-group');
        const wireL2Right = document.getElementById('scene1-wire-l2-right');
        const wireL2Down = document.getElementById('scene1-wire-l2-down');

        // 选项元素
        const optA = document.getElementById('scene1-opt-a');
        const optB = document.getElementById('scene1-opt-b');
        const optC = document.getElementById('scene1-opt-c');
        const optD = document.getElementById('scene1-opt-d');
        const optAWrong = optA.querySelector('.scene1-wrong-mark');
        const optBWrong = optB.querySelector('.scene1-wrong-mark');
        const optCWrong = optC.querySelector('.scene1-wrong-mark');
        const optDRight = optD.querySelector('.scene1-right-mark');

        // 箭头组
        const arrowsStep1 = document.getElementById('scene1-arrows-step1');
        const arrowsStep2 = document.getElementById('scene1-arrows-step2');
        const arrowsStep3 = document.getElementById('scene1-arrows-step3');

        // 面板内容模板
        const panelHTML = [
            // 步骤0
            '<div>R₁ = 6Ω</div><div>R₂ = 3Ω</div><div style="margin-top:10px">电源电压 U = 3V</div><div>L₁额定：6V 6W</div><div>L₂额定：3V 3W</div>',
            // 步骤1
            '<div>U₁ = 3V</div><div>I₁ = 0.5A</div><div>P₁ = 1.5W</div><div style="margin-top:15px" class="red">U₁ &lt; 6V</div><div class="red">不能正常发光</div>',
            // 步骤2
            '<div>I = 1/3 A</div><div>U₁ = 2V</div><div>U₂ = 1V</div><div>P₁ = 2/3 W</div><div class="red">P₂ = 1/3 W</div><div style="margin-top:15px" class="blue">P₁ &gt; P₂</div><div class="blue">L₁更亮</div>',
            // 步骤3
            '<div>U₁ = U₂ = 3V</div><div>I₁ = 0.5A</div><div class="red">I₂ = 1A</div><div>P₁ = 1.5W</div><div class="red">P₂ = 3W</div><div style="margin-top:15px" class="green">L₂正常发光</div><div class="green">且更亮</div>'
        ];

        function resetAll() {
            // 重置开关
            s2Blade.setAttribute('x1', '200');
            s2Blade.setAttribute('y1', '120');
            s2Blade.setAttribute('x2', '230');
            s2Blade.setAttribute('y2', '90');
            s2Blade.setAttribute('stroke', '#000');
            s2Blade.setAttribute('stroke-width', '2');

            s1Blade.setAttribute('x1', '460');
            s1Blade.setAttribute('y1', '340');
            s1Blade.setAttribute('x2', '460');
            s1Blade.setAttribute('y2', '300');
            s1Blade.setAttribute('stroke', '#000');
            s1Blade.setAttribute('stroke-width', '2');

            // 重置箭头
            arrowsStep1.style.display = 'none';
            arrowsStep2.style.display = 'none';
            arrowsStep3.style.display = 'none';

            // 重置选项样式
            [optA, optB, optC, optD].forEach(opt => {
                opt.classList.remove('wrong', 'right');
            });
            optAWrong.style.display = 'none';
            optBWrong.style.display = 'none';
            optCWrong.style.display = 'none';
            optDRight.style.display = 'none';

            // 重置发光
            l1Glow.setAttribute('opacity', '0');
            l2Glow.setAttribute('opacity', '0');

            // 重置L2电路显示
            l2Group.style.opacity = '1';
            l2Group.querySelectorAll('line, circle').forEach(el => {
                if (el.id !== 'scene1-l2-glow') {
                    el.style.strokeDasharray = 'none';
                }
            });
            wireL2Right.style.opacity = '1';
            wireL2Right.style.strokeDasharray = 'none';
            wireL2Down.style.opacity = '1';
            wireL2Down.style.strokeDasharray = 'none';
        }

        function dimL2Circuit() {
            l2Group.style.opacity = '0.3';
            l2Group.querySelectorAll('line, circle').forEach(el => {
                if (el.id !== 'scene1-l2-glow') {
                    el.style.strokeDasharray = '4,4';
                }
            });
            wireL2Right.style.opacity = '0.3';
            wireL2Right.style.strokeDasharray = '4,4';
            wireL2Down.style.opacity = '0.3';
            wireL2Down.style.strokeDasharray = '4,4';
        }

        function updateScene(step) {
            currentStep = step;

            // 更新按钮状态
            btnPrev.disabled = (step === 0);
            btnNext.disabled = (step === totalSteps);

            // 重置所有
            resetAll();

            // 更新面板内容
            panelContent.innerHTML = panelHTML[step];

            // 根据步骤设置
            if (step === 0) {
                hintText.textContent = '点击下一步开始分析各选项';
                hintText.style.color = '#666';
            } else if (step === 1) {
                // S1接b
                s1Blade.setAttribute('x1', '460');
                s1Blade.setAttribute('y1', '340');
                s1Blade.setAttribute('x2', '420');
                s1Blade.setAttribute('y2', '260');

                // 淡化L2电路
                dimL2Circuit();

                // 显示箭头
                arrowsStep1.style.display = 'block';

                // 选项A错误
                optA.classList.add('wrong');
                optAWrong.style.display = 'flex';

                hintText.textContent = '选项A分析：S₂断开，S₁接b';
                hintText.style.color = '#666';
            } else if (step === 2) {
                // S1接a
                s1Blade.setAttribute('x1', '460');
                s1Blade.setAttribute('y1', '340');
                s1Blade.setAttribute('x2', '480');
                s1Blade.setAttribute('y2', '260');

                // 显示箭头
                arrowsStep2.style.display = 'block';

                // 选项B、C错误
                optB.classList.add('wrong');
                optBWrong.style.display = 'flex';
                optC.classList.add('wrong');
                optCWrong.style.display = 'flex';

                hintText.textContent = '选项B/C分析：S₂断开，S₁接a';
                hintText.style.color = '#666';
            } else if (step === 3) {
                // S2闭合
                s2Blade.setAttribute('x1', '200');
                s2Blade.setAttribute('y1', '120');
                s2Blade.setAttribute('x2', '380');
                s2Blade.setAttribute('y2', '120');
                s2Blade.setAttribute('stroke', '#1565c0');
                s2Blade.setAttribute('stroke-width', '3');

                // S1接b
                s1Blade.setAttribute('x1', '460');
                s1Blade.setAttribute('y1', '340');
                s1Blade.setAttribute('x2', '420');
                s1Blade.setAttribute('y2', '260');

                // 显示箭头
                arrowsStep3.style.display = 'block';

                // 发光效果
                l1Glow.setAttribute('opacity', '0.2');
                l2Glow.setAttribute('opacity', '0.2');

                // 选项D正确
                optD.classList.add('right');
                optDRight.style.display = 'flex';

                hintText.textContent = '选项D分析：S₂闭合，S₁接b';
                hintText.style.color = '#666';
            }
        }

        btnPrev.addEventListener('click', function() {
            if (currentStep > 0) {
                updateScene(currentStep - 1);
            }
        });

        btnNext.addEventListener('click', function() {
            if (currentStep < totalSteps) {
                updateScene(currentStep + 1);
            }
        });

        // 初始化
        updateScene(0);
    })();
    </script>
</body>
