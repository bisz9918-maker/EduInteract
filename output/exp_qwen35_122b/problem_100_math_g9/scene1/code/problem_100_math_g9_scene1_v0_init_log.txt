    <script>
    (function() {
        // Scene 1 实现
        // 布局坐标表：
        // | 元素 | 类型 | 位置/范围 | 大小 | 标签偏移方向 | 备注 |
        // |------|------|-----------|------|-------------|------|
        // | 标题 | text | (400,45) | ~160×20 | - | 顶部居中 |
        // | 立体图区域 | 分组 | (80,80)-(320,380) | 240×300 | - | 左侧 |
        // | 立体图 - 上底面 | 正三角形 | (160,80) 为中心 | 边长 80 | - | 透视效果 |
        // | 立体图 - 下底面 | 正三角形 | (160,300) 为中心 | 边长 80 | - | 透视效果 |
        // | 立体图 - 侧面 1 | 矩形 | (120,120)-(160,280) | 40×160 | - | 前侧面 |
        // | 立体图 - 侧面 2 | 矩形 | (160,120)-(200,280) | 40×160 | - | 右侧面 |
        // | 立体图 - 侧面 3 | 矩形 | (120,120)-(160,280) | 40×160 | - | 左侧面（虚线） |
        // | 立体图标签 - 上底面 | text | (150,65) | ~60×16 | 上方 | "上底面" |
        // | 立体图标签 - 下底面 | text | (150,320) | ~60×16 | 下方 | "下底面" |
        // | 立体图标签 - 侧面 | text | (220,200) | ~60×16 | 右侧 | "侧面" |
        // | 展开箭头 | 箭头 | (340,200)-(380,200) | 40×20 | - | 中间连接 |
        // | 展开图区域 | 分组 | (420,80)-(720,380) | 300×300 | - | 右侧 |
        // | 展开图 - 中间矩形 1 | 矩形 | (460,180)-(520,280) | 60×100 | - | 中间 |
        // | 展开图 - 中间矩形 2 | 矩形 | (520,180)-(580,280) | 60×100 | - | 右侧 |
        // | 展开图 - 中间矩形 3 | 矩形 | (580,180)-(640,280) | 60×100 | - | 最右 |
        // | 展开图 - 上三角形 | 正三角形 | (520,180) 为底边 | 边长 60 | - | 上方 |
        // | 展开图 - 下三角形 | 正三角形 | (520,280) 为底边 | 边长 60 | - | 下方 |
        // | 展开图标签 - 矩形 | text | (550,160) | ~60×16 | 上方 | "侧面×3" |
        // | 展开图标签 - 三角形 | text | (550,110) | ~60×16 | 上方 | "底面" |
        // | 标注线 - 边长 a | 标注线 | (460,230)-(440,230) | 20×2 | 左侧 | "a" |
        // | 标注线 - 高 h | 标注线 | (640,180)-(660,180) | 20×2 | 右侧 | "h" |
        // | 折叠线 1 | 虚线 | (520,180)-(520,280) | 2×100 | - | 矩形间 |
        // | 折叠线 2 | 虚线 | (580,180)-(580,280) | 2×100 | - | 矩形间 |
        // | 控件栏 | div | (250,420)-(550,480) | 300×60 | - | 底部居中 |
        // | 上一步按钮 | button | (280,435) | 100×30 | - | scene1-prev |
        // | 下一步按钮 | button | (420,435) | 100×30 | - | scene1-next |
        
        const currentStep = { value: 0 };
        const totalSteps = 4;
        
        // 工具函数：显示元素
        function show(elementId) {
            const el = document.getElementById(elementId);
            if (el) {
                el.style.display = 'block';
                el.style.opacity = '0';
                setTimeout(() => {
                    el.style.opacity = '1';
                }, 50);
            }
        }
        
        // 工具函数：隐藏元素
        function hide(elementId) {
            const el = document.getElementById(elementId);
            if (el) {
                el.style.opacity = '0';
                setTimeout(() => {
                    el.style.display = 'none';
                }, 300);
            }
        }
        
        // 工具函数：高亮元素（增加透明度）
        function highlight(elementId, opacity) {
            const el = document.getElementById(elementId);
            if (el) {
                el.style.opacity = opacity;
            }
        }
        
        function updateScene() {
            const step = currentStep.value;
            
            // 步骤 0：显示立体图，隐藏展开图
            // 步骤 1：显示展开图的三个矩形和折叠线
            // 步骤 2：显示展开图的上下三角形和标注
            // 步骤 3：高亮所有元素，显示结论文字
            
            // 重置所有展开图元素
            const expandElements = [
                'scene1-rect-1', 'scene1-rect-2', 'scene1-rect-3',
                'scene1-tri-top-2d', 'scene1-tri-bottom-2d',
                'scene1-fold-1', 'scene1-fold-2',
                'scene1-arrow-group',
                'scene1-label-sides', 'scene1-label-base',
                'scene1-labels-group', 'scene1-conclusion'
            ];
            expandElements.forEach(id => hide(id));
            
            // 根据步骤显示相应元素
            if (step >= 1) {
                // 显示展开箭头
                show('scene1-arrow-group');
                // 显示三个矩形
                show('scene1-rect-1');
                show('scene1-rect-2');
                show('scene1-rect-3');
                // 显示折叠线
                show('scene1-fold-1');
                show('scene1-fold-2');
                // 显示侧面标签
                show('scene1-label-sides');
            }
            
            if (step >= 2) {
                // 显示上下三角形
                show('scene1-tri-top-2d');
                show('scene1-tri-bottom-2d');
                // 显示底面标签
                show('scene1-label-base');
                // 显示标注线
                show('scene1-labels-group');
            }
            
            if (step >= 3) {
                // 高亮所有元素（增加透明度）
                highlight('scene1-tri-top', '0.4');
                highlight('scene1-tri-bottom', '0.4');
                highlight('scene1-rect-front', '0.4');
                highlight('scene1-rect-right', '0.4');
                highlight('scene1-rect-1', '0.4');
                highlight('scene1-rect-2', '0.4');
                highlight('scene1-rect-3', '0.4');
                highlight('scene1-tri-top-2d', '0.4');
                highlight('scene1-tri-bottom-2d', '0.4');
                // 显示结论文字
                show('scene1-conclusion');
            }
            
            // 更新按钮状态
            document.getElementById('scene1-prev').disabled = step === 0;
            document.getElementById('scene1-next').disabled = step === totalSteps - 1;
        }
        
        function changeStep(delta) {
            currentStep.value += delta;
            if (currentStep.value < 0) currentStep.value = 0;
            if (currentStep.value >= totalSteps) currentStep.value = totalSteps - 1;
            updateScene();
        }
        
        // 事件绑定
        document.getElementById('scene1-prev').addEventListener('click', () => changeStep(-1));
        document.getElementById('scene1-next').addEventListener('click', () => changeStep(1));
        
        // 初始化
        updateScene();
    })();
    </script>
