    <script>
    (function() {
        // Scene 1 实现：正方形折叠挖洞展开过程演示
        // 布局坐标表（实现对照依据）：
        // | 元素 | 位置/范围 | 备注 |
        // | 正方形轮廓 | (150,100)-(350,300) 200×200 | 步骤 0 主图形 |
        // | 对角线折痕 | (150,300)-(350,100) | 步骤 1 显示 |
        // | 折叠三角形 | (150,100)-(350,100)-(350,300) | 步骤 1 显示 |
        // | 洞 1（直角顶点） | (170,280) r=12 | 步骤 2 显示 |
        // | 洞 2（上锐角） | (170,120) r=12 | 步骤 2 显示 |
        // | 洞 3（右锐角） | (330,280) r=12 | 步骤 2 显示 |
        // | 展开洞 A | (170,280) r=12 | 步骤 3 显示 |
        // | 展开洞 B | (170,120) r=12 | 步骤 3 显示 |
        // | 展开洞 C | (330,280) r=12 | 步骤 3 显示 |
        // | 展开洞 D | (330,120) r=12 | 步骤 3 显示（对称位置） |
        // | 步骤标签 | (30,80) | 显示当前步骤名称 |
        
        // 状态变量
        var currentStep = 0;
        var totalSteps = 4;
        
        // DOM 元素引用
        var elements = {
            stepLabel: document.getElementById('scene1-step-label'),
            square: document.getElementById('scene1-square'),
            foldLine: document.getElementById('scene1-fold-line'),
            foldLabel: document.getElementById('scene1-fold-label'),
            foldTriangle: document.getElementById('scene1-fold-triangle'),
            hole1: document.getElementById('scene1-hole1'),
            hole2: document.getElementById('scene1-hole2'),
            hole3: document.getElementById('scene1-hole3'),
            holeA: document.getElementById('scene1-holeA'),
            holeB: document.getElementById('scene1-holeB'),
            holeC: document.getElementById('scene1-holeC'),
            holeD: document.getElementById('scene1-holeD'),
            highlight1: document.getElementById('scene1-highlight1'),
            highlight2: document.getElementById('scene1-highlight2'),
            prevBtn: document.getElementById('scene1-prev-btn'),
            nextBtn: document.getElementById('scene1-next-btn')
        };
        
        // 步骤标签文本
        var stepLabels = ['原始正方形', '折叠后', '挖洞', '展开后'];
        
        // 工具函数：显示元素
        function show(el) {
            if (el) {
                el.classList.remove('scene1-hidden');
                el.style.opacity = '1';
            }
        }
        
        // 工具函数：隐藏元素
        function hide(el) {
            if (el) {
                el.style.opacity = '0';
                setTimeout(function() {
                    el.classList.add('scene1-hidden');
                    el.style.opacity = '1'; // 恢复为下次显示做准备
                }, 300);
            }
        }
        
        // 更新场景显示
        function updateScene() {
            // 更新步骤标签
            elements.stepLabel.textContent = stepLabels[currentStep];
            
            // 重置所有元素
            hide(elements.foldLine);
            hide(elements.foldLabel);
            hide(elements.foldTriangle);
            hide(elements.hole1);
            hide(elements.hole2);
            hide(elements.hole3);
            hide(elements.holeA);
            hide(elements.holeB);
            hide(elements.holeC);
            hide(elements.holeD);
            hide(elements.highlight1);
            hide(elements.highlight2);
            
            // 根据步骤显示对应元素
            switch(currentStep) {
                case 0: // 原始正方形
                    show(elements.square);
                    break;
                    
                case 1: // 折叠后
                    show(elements.square);
                    show(elements.foldLine);
                    show(elements.foldLabel);
                    show(elements.foldTriangle);
                    break;
                    
                case 2: // 挖洞
                    show(elements.square);
                    show(elements.foldLine);
                    show(elements.foldLabel);
                    show(elements.foldTriangle);
                    show(elements.hole1);
                    show(elements.hole2);
                    show(elements.hole3);
                    break;
                    
                case 3: // 展开后
                    show(elements.square);
                    show(elements.holeA);
                    show(elements.holeB);
                    show(elements.holeC);
                    show(elements.holeD);
                    show(elements.highlight1);
                    show(elements.highlight2);
                    break;
            }
            
            // 更新按钮状态
            elements.prevBtn.disabled = (currentStep === 0);
            elements.nextBtn.disabled = (currentStep === totalSteps - 1);
        }
        
        // 事件绑定
        elements.prevBtn.addEventListener('click', function() {
            if (currentStep > 0) {
                currentStep--;
                updateScene();
            }
        });
        
        elements.nextBtn.addEventListener('click', function() {
            if (currentStep < totalSteps - 1) {
                currentStep++;
                updateScene();
            }
        });
        
        // 初始化场景
        updateScene();
    })();
    </script>
