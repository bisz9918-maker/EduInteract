    <script>
    (function() {
        // Scene 1: 正方形序列几何布局
        // 坐标表对照：
        // B₁(60,316), A₁(164,256), D₁(224,360), C₁(120,420)
        // B₂(284,360), A₂(344,325), D₂(379,385), C₂(319,420)
        // B₃(414,385), A₃(448,365), D₃(468,400), C₃(433,420)
        // D₁E₁E₂B₂: (224,360)-(224,420)-(284,420)-(284,360)
        // D₂E₃E₄B₃: (379,385)-(379,420)-(414,420)-(414,385)

        var currentStep = 0;
        var totalSteps = 4;

        // 获取 DOM 元素
        var btnPrev = document.getElementById('scene1-btn-prev');
        var btnNext = document.getElementById('scene1-btn-next');
        var stepDisplay = document.getElementById('scene1-step-display');
        var step1Group = document.getElementById('scene1-step1');
        var step2Group = document.getElementById('scene1-step2');
        var step3Group = document.getElementById('scene1-step3');
        var step4Group = document.getElementById('scene1-step4');
        var vertexLabels = document.getElementById('scene1-vertex-labels');

        // 更新显示状态
        function updateDisplay() {
            // 更新按钮状态
            btnPrev.disabled = (currentStep === 0);
            btnNext.disabled = (currentStep === totalSteps);

            // 更新步骤显示
            stepDisplay.textContent = '步骤 ' + currentStep + '/' + totalSteps;

            // 控制各步骤元素显示
            if (currentStep >= 1) {
                step1Group.style.display = 'block';
                vertexLabels.style.display = 'block';
            } else {
                step1Group.style.display = 'none';
                vertexLabels.style.display = 'none';
            }

            if (currentStep >= 2) {
                step2Group.style.display = 'block';
            } else {
                step2Group.style.display = 'none';
            }

            if (currentStep >= 3) {
                step3Group.style.display = 'block';
            } else {
                step3Group.style.display = 'none';
            }

            if (currentStep >= 4) {
                step4Group.style.display = 'block';
            } else {
                step4Group.style.display = 'none';
            }
        }

        // 按钮事件绑定
        btnPrev.addEventListener('click', function() {
            if (currentStep > 0) {
                currentStep--;
                updateDisplay();
            }
        });

        btnNext.addEventListener('click', function() {
            if (currentStep < totalSteps) {
                currentStep++;
                updateDisplay();
            }
        });

        // 初始化显示
        updateDisplay();
    })();
    </script>
