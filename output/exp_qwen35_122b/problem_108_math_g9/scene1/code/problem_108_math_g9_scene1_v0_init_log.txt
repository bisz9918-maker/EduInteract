    <script>
    (function() {
        // Scene 1: 角平分线交点的角度关系分析
        // 布局坐标表:
        // | 元素 | 类型 | 位置/范围 | 大小 | 标签偏移方向 | 备注 |
        // |------|------|-----------|------|-------------|------|
        // | 底边 MN | 线段 | (150,380)-(650,380) | 500×3 | - | 加粗黑色，水平居中 |
        // | 边 AM | 线段 | (150,380)-(350,150) | - | - | 黑色 |
        // | 边 AN | 线段 | (350,150)-(650,380) | - | - | 蓝色角平分线 |
        // | 边 BN | 线段 | (580,220)-(650,380) | - | - | 黑色 |
        // | 边 BM | 线段 | (150,380)-(580,220) | - | - | 蓝色角平分线 |
        // | 顶点 M | 圆点 | (150,380) | 4px 直径 | - | 黑色实心 |
        // | 顶点 N | 圆点 | (650,380) | 4px 直径 | - | 黑色实心 |
        // | 顶点 A | 圆点 | (350,150) | 4px 直径 | - | 黑色实心 |
        // | 顶点 B | 圆点 | (580,220) | 4px 直径 | - | 黑色实心 |
        // | 交点 C | 圆点 | (465,238) | 6px 直径 | - | 红色实心 |
        
        const TOTAL_STEPS = 5;
        let currentStep = 0;
        
        // 工具函数：显示元素
        function show(elId) {
            const el = document.getElementById(elId);
            if (el) {
                el.style.display = 'block';
                el.style.opacity = '1';
            }
        }
        
        // 工具函数：隐藏元素
        function hide(elId) {
            const el = document.getElementById(elId);
            if (el) {
                el.style.display = 'none';
                el.style.opacity = '0';
            }
        }
        
        // 更新场景状态
        function updateScene() {
            const prevBtn = document.getElementById('scene1-btn-prev');
            const nextBtn = document.getElementById('scene1-btn-next');
            const indicator = document.getElementById('scene1-step-indicator');
            
            // 重置所有动态元素
            hide('scene1-line-an');
            hide('scene1-line-bm');
            hide('scene1-arc-amb');
            hide('scene1-arc-nmb');
            hide('scene1-arc-bna');
            hide('scene1-arc-bna-2');
            hide('scene1-arc-mna');
            hide('scene1-arc-mna-2');
            hide('scene1-point-c');
            hide('scene1-label-c');
            hide('scene1-arc-mcn');
            hide('scene1-perp-bisector');
            hide('scene1-final-conclusion');
            hide('scene1-opt-a-mark');
            hide('scene1-opt-b-mark');
            hide('scene1-opt-c-mark');
            hide('scene1-opt-d-mark');
            
            // 步骤 0: 初始状态 - 基础图形
            if (currentStep >= 0) {
                // 基础图形已显示
            }
            
            // 步骤 1: 高亮角平分线 BM，显示∠AMB 和∠NMB 的单弧线
            if (currentStep >= 1) {
                show('scene1-line-bm');
                show('scene1-arc-amb');
                show('scene1-arc-nmb');
            }
            
            // 步骤 2: 高亮角平分线 AN，显示∠BNA 和∠MNA 的双弧线
            if (currentStep >= 2) {
                show('scene1-line-an');
                show('scene1-arc-bna');
                show('scene1-arc-bna-2');
                show('scene1-arc-mna');
                show('scene1-arc-mna-2');
            }
            
            // 步骤 3: 显示交点 C 和∠MCN
            if (currentStep >= 3) {
                show('scene1-point-c');
                show('scene1-label-c');
                show('scene1-arc-mcn');
            }
            
            // 步骤 4: 验证选项 A (α=β 时，C 在垂直平分线上)
            if (currentStep >= 4) {
                show('scene1-perp-bisector');
                show('scene1-opt-a');
                show('scene1-opt-a-mark');
            }
            
            // 步骤 5: 显示所有选项验证结果
            if (currentStep >= 5) {
                show('scene1-final-conclusion');
                show('scene1-opt-b-mark');
                show('scene1-opt-c-mark');
                show('scene1-opt-d-mark');
            }
            
            // 更新按钮状态
            prevBtn.disabled = (currentStep === 0);
            nextBtn.disabled = (currentStep === TOTAL_STEPS);
            
            // 更新步骤指示器
            if (currentStep === TOTAL_STEPS) {
                indicator.textContent = '步骤 5/5 (完成)';
            } else {
                indicator.textContent = '步骤 ' + (currentStep + 1) + '/' + TOTAL_STEPS;
            }
        }
        
        // 绑定按钮事件
        document.getElementById('scene1-btn-prev').addEventListener('click', function() {
            if (currentStep > 0) {
                currentStep--;
                updateScene();
            }
        });
        
        document.getElementById('scene1-btn-next').addEventListener('click', function() {
            if (currentStep < TOTAL_STEPS) {
                currentStep++;
                updateScene();
            }
        });
        
        // 初始化场景
        updateScene();
    })();
    </script>
