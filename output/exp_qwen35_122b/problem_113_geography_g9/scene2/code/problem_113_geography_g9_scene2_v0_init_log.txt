    <script>
    (function() {
        // Scene 2 实现
        // 布局坐标表（实现对照依据）：
        // | 元素 | 位置/范围 | 备注 |
        // | 极点 O | (400,80) | 黑色实心圆，标签在 (397,65) |
        // | 经线 OP | (400,80)-(280,220) | 黑色实线 |
        // | 经线 OQ | (400,80)-(520,220) | 黑色实线 |
        // | 纬线 L | 弧线 (260,220)-(540,220) | 黑色弧线，标签在 (250,200) |
        // | P 点 | (280,220) | 黑色实心圆 |
        // | Q 点 | (520,220) | 黑色实心圆 |
        // | P 点经度初始 | (240,240) | 黑色"45°" |
        // | Q 点经度 | (530,240) | 蓝色加粗"15°E" |
        // | 经度差弧线 | (460,155)-(520,155) | 绿色 - 移到右侧避免与经线重叠 |
        // | 经度差标签 | (540,155) | 绿色 - 移到三角形右侧外部 |
        // | P 点经度结果 | (220,275) | 红色加粗 |
        // | 20°W 分界线 | (200,350)-(200,420) | 灰色虚线 |
        // | 160°E 分界线 | (600,350)-(600,420) | 灰色虚线 |
        // | 东半球区域 | (200,350)-(600,420) | 蓝色 15% |
        // | 西半球区域 | (100,350)-(200,420) 和 (600,350)-(700,420) | 红色 15% |
        // | 半球标签 | y=340 | 在矩形区域上方 |
        // | 方向结论 | (400,445) | 黑色 - 在矩形区域下方 |

        const scene2 = {
            currentStep: 0,
            totalSteps: 6,

            // 步骤配置
            steps: [
                // 步骤 0：初始状态 - 基础图形
                {
                    name: "初始状态",
                    show: ['base', 'labels-initial'],
                    hide: ['longdiff', 'P-long-result', 'hemisphere', 'boundary', 'hemisphere-labels', 'boundary-labels', 'direction']
                },
                // 步骤 1：情况一（经度差 30°）
                {
                    name: "情况一：经度差 30°",
                    show: ['base', 'longdiff-30', 'P-long-result-45E'],
                    hide: ['hemisphere', 'boundary', 'hemisphere-labels', 'boundary-labels', 'direction']
                },
                // 步骤 2：情况一（东西半球判断）
                {
                    name: "情况一：半球判断",
                    show: ['base', 'longdiff-30', 'P-long-result-45E', 'hemisphere', 'boundary', 'hemisphere-labels', 'boundary-labels'],
                    hide: ['direction']
                },
                // 步骤 3：情况二（经度差 60°）
                {
                    name: "情况二：经度差 60°",
                    show: ['base', 'longdiff-60', 'P-long-result-45W'],
                    hide: ['hemisphere', 'boundary', 'hemisphere-labels', 'boundary-labels', 'direction']
                },
                // 步骤 4：情况二（东西半球判断）
                {
                    name: "情况二：半球判断",
                    show: ['base', 'longdiff-60', 'P-long-result-45W', 'hemisphere', 'boundary', 'hemisphere-labels', 'boundary-labels'],
                    hide: ['direction']
                },
                // 步骤 5：方向判断
                {
                    name: "方向判断",
                    show: ['base', 'longdiff-60', 'P-long-result-45W', 'hemisphere', 'boundary', 'hemisphere-labels', 'boundary-labels', 'direction'],
                    hide: []
                }
            ],

            // 工具函数：显示元素
            show: function(elId) {
                const el = document.getElementById(elId);
                if (el) {
                    el.style.opacity = '1';
                    el.style.display = 'block';
                }
            },

            // 工具函数：隐藏元素
            hide: function(elId) {
                const el = document.getElementById(elId);
                if (el) {
                    el.style.opacity = '0';
                    // 对于 g 元素，需要处理子元素
                    if (el.tagName === 'g') {
                        el.style.display = 'none';
                    } else {
                        el.style.display = 'block';
                    }
                }
            },

            // 渲染当前步骤
            renderStep: function() {
                const config = this.steps[this.currentStep];

                // 隐藏所有
                this.hide('longdiff-group');
                this.hide('scene2-P-long-result');
                this.hide('scene2-hemisphere-group');
                this.hide('scene2-boundary-group');
                this.hide('scene2-hemisphere-labels');
                this.hide('scene2-boundary-labels');
                this.hide('scene2-direction-conclusion');
                this.hide('scene2-label-P-long-initial');

                // 根据配置显示元素
                config.show.forEach(item => {
                    switch(item) {
                        case 'base':
                            this.show('scene2-base-group');
                            break;
                        case 'longdiff-30':
                            this.show('scene2-longdiff-group');
                            const arc30 = document.getElementById('scene2-longdiff-arc');
                            const label30 = document.getElementById('scene2-longdiff-label');
                            if (arc30) { arc30.style.opacity = '1'; }
                            if (label30) { label30.textContent = '经度差 = 30°'; label30.style.opacity = '1'; }
                            break;
                        case 'longdiff-60':
                            this.show('scene2-longdiff-group');
                            const arc60 = document.getElementById('scene2-longdiff-arc');
                            const label60 = document.getElementById('scene2-longdiff-label');
                            if (arc60) { arc60.style.opacity = '1'; }
                            if (label60) { label60.textContent = '经度差 = 60°'; label60.style.opacity = '1'; }
                            break;
                        case 'P-long-result-45E':
                            const result45E = document.getElementById('scene2-P-long-result');
                            if (result45E) { result45E.textContent = 'P 点经度 = 45°E'; result45E.style.opacity = '1'; }
                            break;
                        case 'P-long-result-45W':
                            const result45W = document.getElementById('scene2-P-long-result');
                            if (result45W) { result45W.textContent = 'P 点经度 = 45°W'; result45W.style.opacity = '1'; }
                            break;
                        case 'hemisphere':
                            this.show('scene2-hemisphere-group');
                            this.show('scene2-west-left');
                            this.show('scene2-east');
                            this.show('scene2-west-right');
                            break;
                        case 'boundary':
                            this.show('scene2-boundary-group');
                            this.show('scene2-boundary-20W');
                            this.show('scene2-boundary-160E');
                            break;
                        case 'hemisphere-labels':
                            this.show('scene2-hemisphere-labels');
                            break;
                        case 'boundary-labels':
                            this.show('scene2-boundary-labels');
                            break;
                        case 'direction':
                            this.show('scene2-direction-conclusion');
                            break;
                    }
                });

                // 更新步骤指示器
                document.getElementById('scene2-step-indicator').textContent = `步骤 ${this.currentStep + 1}/${this.totalSteps}`;

                // 更新按钮状态
                document.getElementById('scene2-btn-prev').disabled = (this.currentStep === 0);
                document.getElementById('scene2-btn-next').disabled = (this.currentStep === this.totalSteps - 1);
            },

            // 下一步
            nextStep: function() {
                if (this.currentStep < this.totalSteps - 1) {
                    this.currentStep++;
                    this.renderStep();
                }
            },

            // 上一步
            prevStep: function() {
                if (this.currentStep > 0) {
                    this.currentStep--;
                    this.renderStep();
                }
            },

            // 初始化
            init: function() {
                // 绑定按钮事件
                document.getElementById('scene2-btn-prev').addEventListener('click', () => this.prevStep());
                document.getElementById('scene2-btn-next').addEventListener('click', () => this.nextStep());

                // 初始渲染
                this.renderStep();
            }
        };

        // 启动
        scene2.init();
    })();
    </script>
