    <script>
    (function() {
        // Scene 1 实现 - 经纬网坐标读取与方向判断
        const scene1 = {
            currentStep: 0,
            totalSteps: 5,
            
            // 步骤元素分组
            stepElements: {
                0: ['longitude-lines', 'latitude-lines', 'longitude-labels', 'latitude-labels'],
                1: ['points', 'coordinate-labels'],
                2: ['direction-arrow'],
                3: ['highlight-group', 'condition-boxes'],
                4: ['answer-highlight']
            },
            
            // 工具函数：显示元素
            show: function(elementIds) {
                elementIds.forEach(id => {
                    const el = document.getElementById('scene1-' + id);
                    if (el) {
                        el.style.display = 'block';
                        el.style.opacity = '1';
                    }
                });
            },
            
            // 工具函数：隐藏元素
            hide: function(elementIds) {
                elementIds.forEach(id => {
                    const el = document.getElementById('scene1-' + id);
                    if (el) {
                        el.style.opacity = '0';
                        setTimeout(() => {
                            if (el.style.opacity === '0') {
                                el.style.display = 'none';
                            }
                        }, 300);
                    }
                });
            },
            
            // 更新按钮状态
            updateButtons: function() {
                const prevBtn = document.getElementById('scene1-prev-btn');
                const nextBtn = document.getElementById('scene1-next-btn');
                
                prevBtn.disabled = (this.currentStep === 0);
                nextBtn.disabled = (this.currentStep === this.totalSteps - 1);
                
                if (this.currentStep === this.totalSteps - 1) {
                    nextBtn.textContent = '完成';
                } else {
                    nextBtn.textContent = '下一步 ▶';
                }
            },
            
            // 跳转到指定步骤
            goToStep: function(step) {
                if (step < 0 || step >= this.totalSteps) return;
                
                // 隐藏之前步骤新增的元素（从当前步骤往前）
                for (let i = this.currentStep; i > step; i--) {
                    const elements = this.stepElements[i] || [];
                    this.hide(elements);
                }
                
                // 显示目标步骤及之前步骤的元素
                for (let i = 0; i <= step; i++) {
                    const elements = this.stepElements[i] || [];
                    this.show(elements);
                }
                
                this.currentStep = step;
                this.updateButtons();
            },
            
            // 下一步
            nextStep: function() {
                if (this.currentStep < this.totalSteps - 1) {
                    this.goToStep(this.currentStep + 1);
                }
            },
            
            // 上一步
            prevStep: function() {
                if (this.currentStep > 0) {
                    this.goToStep(this.currentStep - 1);
                }
            },
            
            // 初始化
            init: function() {
                // 绑定按钮事件
                document.getElementById('scene1-prev-btn').addEventListener('click', () => {
                    this.prevStep();
                });
                
                document.getElementById('scene1-next-btn').addEventListener('click', () => {
                    this.nextStep();
                });
                
                // 初始化：显示步骤 0 的元素
                this.goToStep(0);
            }
        };
        
        // 启动场景
        scene1.init();
    })();
    </script>
