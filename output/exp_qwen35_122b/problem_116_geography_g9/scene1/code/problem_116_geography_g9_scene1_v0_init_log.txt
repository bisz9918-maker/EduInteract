    <script>
    (function() {
        // Scene 1: 经纬网方向判断
        // 坐标表对照：
        // 左图边框：(50,60)-(330,380) | 右图边框：(470,60)-(750,380)
        // 点 A: (600,380) | 点 B: (200,120)
        // A 坐标框：(350,300)-(470,360) | B 坐标框：(350,160)-(470,220)
        // 经度比较：(410,240) | 纬度比较：(410,280)
        // 结论框：(350,80)-(470,140)

        const MAX_STEP = 5;
        let currentStep = 0;

        // 获取 DOM 元素
        const btnPrev = document.getElementById('scene1-btnPrev');
        const btnNext = document.getElementById('scene1-btnNext');

        // 元素分组（按步骤显示）
        const groups = {
            step1: document.getElementById('scene1-groupA'),
            step2: document.getElementById('scene1-groupB'),
            step2b: document.getElementById('scene1-groupBoxA'),
            step2c: document.getElementById('scene1-groupBoxB'),
            step3: document.getElementById('scene1-groupLon'),
            step4: document.getElementById('scene1-groupLat'),
            step5: document.getElementById('scene1-groupConclusion')
        };

        // 虚线元素（步骤 3、4 显示）
        const dashedLines = {
            lonA: document.getElementById('scene1-dashedLonA'),
            lonB: document.getElementById('scene1-dashedLonB'),
            latA: document.getElementById('scene1-dashedLatA'),
            latB: document.getElementById('scene1-dashedLatB')
        };

        // 点元素（步骤 5 闪烁）
        const points = {
            A: document.getElementById('scene1-pointA'),
            B: document.getElementById('scene1-pointB')
        };

        // 更新按钮状态
        function updateButtons() {
            btnPrev.disabled = (currentStep === 0);
            btnNext.disabled = (currentStep === MAX_STEP);
            btnNext.textContent = (currentStep === MAX_STEP) ? '完成' : '下一步 ▶';
        }

        // 显示元素（同时设置 display 和 opacity）
        function show(el) {
            if (el) {
                el.classList.remove('scene1-hidden');
                el.style.opacity = '1';
            }
        }

        // 隐藏元素
        function hide(el) {
            if (el) {
                el.style.opacity = '0';
                setTimeout(() => {
                    el.classList.add('scene1-hidden');
                }, 300);
            }
        }

        // 闪烁效果（步骤 5）
        function flashPoint(el, times = 3) {
            let count = 0;
            const interval = setInterval(() => {
                el.style.stroke = (count % 2 === 0) ? '#2e7d32' : 'transparent';
                el.style.strokeWidth = (count % 2 === 0) ? '3' : '0';
                count++;
                if (count >= times * 2) {
                    clearInterval(interval);
                    el.style.stroke = 'transparent';
                    el.style.strokeWidth = '0';
                }
            }, 200);
        }

        // 更新显示状态
        function updateDisplay() {
            // 步骤 0：只显示框架（初始状态，所有 group 都隐藏）
            // 步骤 1：显示 A 点
            if (currentStep >= 1) {
                show(groups.step1);
            }
            // 步骤 2：显示 B 点 + 坐标框
            if (currentStep >= 2) {
                show(groups.step2);
                show(groups.step2b);
                show(groups.step2c);
            }
            // 步骤 3：显示经度比较
            if (currentStep >= 3) {
                show(groups.step3);
                dashedLines.lonA.style.opacity = '1';
                dashedLines.lonB.style.opacity = '1';
            }
            // 步骤 4：显示纬度比较
            if (currentStep >= 4) {
                show(groups.step4);
                dashedLines.latA.style.opacity = '1';
                dashedLines.latB.style.opacity = '1';
            }
            // 步骤 5：显示结论 + 闪烁
            if (currentStep >= 5) {
                show(groups.step5);
                flashPoint(points.A);
                setTimeout(() => flashPoint(points.B), 300);
            }
        }

        // 下一步
        function nextStep() {
            if (currentStep < MAX_STEP) {
                currentStep++;
                updateDisplay();
                updateButtons();
            }
        }

        // 上一步
        function prevStep() {
            if (currentStep > 0) {
                currentStep--;
                updateDisplay();
                updateButtons();
            }
        }

        // 事件绑定
        btnNext.addEventListener('click', nextStep);
        btnPrev.addEventListener('click', prevStep);

        // 初始化
        updateButtons();
    })();
    </script>
