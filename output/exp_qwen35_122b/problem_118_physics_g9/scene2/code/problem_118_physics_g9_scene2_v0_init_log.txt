    <script>
    (function() {
        // Scene 2 实现 - 滑动变阻器电路动态变化
        
        // 物理参数（假设值）
        const U = 12;      // 电源电压 12V
        const R1 = 10;     // R₁固定电阻 10Ω
        const R2Max = 20;  // R₂最大阻值 20Ω
        
        // 状态变量
        let sliderValue = 50;  // 滑块值 0-100
        let prevSliderValue = 50;  // 用于判断变化方向
        
        // DOM 元素
        const slider = document.getElementById('scene2-slider');
        const sliderP = document.getElementById('scene2-slider-p');
        const displayI = document.getElementById('scene2-display-i');
        const displayItext = document.getElementById('scene2-display-i-text');
        const displayU1 = document.getElementById('scene2-display-u1');
        const displayU1text = document.getElementById('scene2-display-u1-text');
        const displayU2 = document.getElementById('scene2-display-u2');
        const displayU2text = document.getElementById('scene2-display-u2-text');
        const arrowI = document.getElementById('scene2-arrow-i');
        const arrowU1 = document.getElementById('scene2-arrow-u1');
        const arrowU2 = document.getElementById('scene2-arrow-u2');
        const r2Value = document.getElementById('scene2-r2-value');
        
        // 计算物理量
        function calculateValues(sliderVal) {
            const R2 = (sliderVal / 100) * R2Max;
            const R = R1 + R2;
            const I = U / R;
            const U1 = I * R2;  // R₂两端电压
            const U2 = U;       // 电源电压恒定
            return { R2, I, U1, U2 };
        }
        
        // 更新滑片位置
        function updateSliderPosition(sliderVal) {
            const x = 200 + (sliderVal / 100) * 300;
            sliderP.setAttribute('transform', `translate(${x - 350}, 0)`);
        }
        
        // 更新趋势箭头
        function updateTrendArrows(currentVal, prevVal, element) {
            const diff = currentVal - prevVal;
            if (Math.abs(diff) < 0.01) {
                element.textContent = '→';
                element.setAttribute('fill', '#888888');
            } else if (diff > 0) {
                element.textContent = '↑';
                element.setAttribute('fill', '#2e7d32');  // 绿色 - 增大
            } else {
                element.textContent = '↓';
                element.setAttribute('fill', '#c62828');  // 红色 - 减小
            }
        }
        
        // 更新示数框边框颜色
        function updateDisplayBorder(element, trend) {
            if (trend === 'up') {
                element.setAttribute('stroke', '#2e7d32');
            } else if (trend === 'down') {
                element.setAttribute('stroke', '#c62828');
            } else {
                element.setAttribute('stroke', '#000000');
            }
        }
        
        // 更新所有显示
        function updateDisplay() {
            const values = calculateValues(sliderValue);
            
            // 更新滑片位置
            updateSliderPosition(sliderValue);
            
            // 更新示数
            displayItext.textContent = `I = ${values.I.toFixed(2)}A`;
            displayU1text.textContent = `U₁ = ${values.U1.toFixed(2)}V`;
            displayU2text.textContent = `U₂ = ${values.U2.toFixed(2)}V`;
            
            // 更新 R₂阻值显示
            const percentage = Math.round(sliderValue);
            r2Value.textContent = `R₂ = ${values.R2.toFixed(1)}Ω (${percentage}%)`;
            
            // 更新趋势箭头（与上一状态比较）
            const prevValues = calculateValues(prevSliderValue);
            updateTrendArrows(values.I, prevValues.I, arrowI);
            updateTrendArrows(values.U1, prevValues.U1, arrowU1);
            updateTrendArrows(values.U2, prevValues.U2, arrowU2);
            
            // 更新示数框边框颜色
            const iTrend = values.I > prevValues.I ? 'up' : (values.I < prevValues.I ? 'down' : 'none');
            const u1Trend = values.U1 > prevValues.U1 ? 'up' : (values.U1 < prevValues.U1 ? 'down' : 'none');
            const u2Trend = 'none';  // U₂始终不变
            
            updateDisplayBorder(displayI, iTrend);
            updateDisplayBorder(displayU1, u1Trend);
            updateDisplayBorder(displayU2, u2Trend);
            
            // 更新上一状态
            prevSliderValue = sliderValue;
        }
        
        // 滑块事件
        slider.addEventListener('input', function(e) {
            sliderValue = parseInt(e.target.value);
            updateDisplay();
        });
        
        // 滑片拖拽支持
        let isDragging = false;
        
        sliderP.addEventListener('mousedown', function(e) {
            isDragging = true;
            e.preventDefault();
        });
        
        document.addEventListener('mousemove', function(e) {
            if (!isDragging) return;
            
            const svg = document.getElementById('scene2-svg');
            const rect = svg.getBoundingClientRect();
            const x = e.clientX - rect.left;
            
            // 限制在 R₂范围内
            const clampedX = Math.max(200, Math.min(500, x));
            const newValue = Math.round((clampedX - 200) / 300 * 100);
            
            if (newValue !== sliderValue) {
                sliderValue = newValue;
                slider.value = newValue;
                updateDisplay();
            }
        });
        
        document.addEventListener('mouseup', function() {
            isDragging = false;
        });
        
        // 触摸支持
        sliderP.addEventListener('touchstart', function(e) {
            isDragging = true;
            e.preventDefault();
        });
        
        document.addEventListener('touchmove', function(e) {
            if (!isDragging) return;
            
            const svg = document.getElementById('scene2-svg');
            const rect = svg.getBoundingClientRect();
            const touch = e.touches[0];
            const x = touch.clientX - rect.left;
            
            const clampedX = Math.max(200, Math.min(500, x));
            const newValue = Math.round((clampedX - 200) / 300 * 100);
            
            if (newValue !== sliderValue) {
                sliderValue = newValue;
                slider.value = newValue;
                updateDisplay();
            }
        });
        
        document.addEventListener('touchend', function() {
            isDragging = false;
        });
        
        // 初始化
        updateDisplay();
    })();
    </script>
