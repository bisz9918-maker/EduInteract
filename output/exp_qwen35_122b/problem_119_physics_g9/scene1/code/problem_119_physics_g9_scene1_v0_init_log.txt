    <script>
    (function() {
        // Scene 1 实现：并联电路中滑动变阻器的影响
        // 布局坐标表（参考）：
        // | 元素 | 位置 | 备注 |
        // | 电源 | (350,70)-(450,110) | 电池符号 |
        // | 开关 S | (200,90)-(280,90) | 闭合状态 |
        // | 滑动变阻器 | (200,180)-(380,200) | 红色电阻丝 |
        // | 滑片 P | (290,160)-(310,200) | 可拖动 |
        // | 灯 L₂ | (450,175)-(490,215) | 绿色发光 |
        // | 灯 L₁ | (280,295)-(320,335) | 绿色发光 |
        // | 电阻指示条 | (620,150)-(750,250) | 红色渐变 |
        
        // 状态变量
        let sliderX = 290;  // 滑片 P 的 x 坐标，范围 [200, 380]
        const sliderMin = 200;
        const sliderMax = 380;
        const sliderWidth = 20;
        
        // 获取 DOM 元素
        const svg = document.getElementById('scene1-svg');
        const sliderP = document.getElementById('scene1-slider-p');
        const resetBtn = document.getElementById('scene1-reset-btn');
        const resistorBar = document.getElementById('scene1-resistor-bar');
        const lampL1 = document.getElementById('scene1-lamp-l1');
        const lampL2 = document.getElementById('scene1-lamp-l2');
        const currentI = document.getElementById('scene1-current-i');
        const currentI1 = document.getElementById('scene1-current-i1');
        const currentI2 = document.getElementById('scene1-current-i2');
        
        // 拖拽状态
        let isDragging = false;
        let dragOffsetX = 0;
        
        // 更新视觉元素
        function updateVisuals() {
            // 计算电阻比例（滑片越靠右，电阻越小）
            const resistanceRatio = (sliderMax - sliderX) / (sliderMax - sliderMin);
            
            // 更新滑片位置
            sliderP.setAttribute('transform', `translate(${sliderX - 290}, 0)`);
            
            // 更新电阻指示条填充宽度
            const barWidth = 130 * resistanceRatio;
            resistorBar.setAttribute('width', barWidth);
            
            // 更新 L₂ 亮度（电阻越小，L₂ 越亮）
            // 透明度范围：10%（最暗）~ 30%（最亮）
            const l2Opacity = 0.10 + 0.20 * resistanceRatio;
            lampL2.querySelector('circle').setAttribute('opacity', l2Opacity);
            
            // L₁ 亮度恒定（并联支路独立）
            // 透明度保持 20%
            
            // 更新电流箭头长度
            // I₂ 电流与电阻成反比，电阻越小电流越大
            const i2Length = 40 + 40 * resistanceRatio;  // 40~80px
            const i2Line = currentI2.querySelector('line');
            const i2Poly = currentI2.querySelector('polygon');
            i2Line.setAttribute('x2', i2Line.getAttribute('x1') + i2Length);
            i2Poly.setAttribute('points', `${i2Line.getAttribute('x1') + i2Length},180 ${i2Line.getAttribute('x1') + i2Length - 5},177 ${i2Line.getAttribute('x1') + i2Length - 5},183`);
            
            // I₁ 电流恒定
            // I 干路电流 = I₁ + I₂，随 I₂ 增大而增大
            const iLength = 40 + 20 * resistanceRatio;  // 40~60px
            const iLine = currentI.querySelector('line');
            const iPoly = currentI.querySelector('polygon');
            iLine.setAttribute('x2', iLine.getAttribute('x1') + iLength);
            iPoly.setAttribute('points', `${iLine.getAttribute('x1') + iLength},80 ${iLine.getAttribute('x1') + iLength - 5},77 ${iLine.getAttribute('x1') + iLength - 5},83`);
        }
        
        // 滑片拖拽事件处理
        sliderP.addEventListener('mousedown', function(e) {
            isDragging = true;
            const rect = sliderP.getBoundingClientRect();
            dragOffsetX = e.clientX - rect.left;
            e.preventDefault();
        });
        
        document.addEventListener('mousemove', function(e) {
            if (!isDragging) return;
            
            const svgRect = svg.getBoundingClientRect();
            let newX = e.clientX - svgRect.left - dragOffsetX;
            
            // 限制滑片在有效范围内
            newX = Math.max(sliderMin, Math.min(sliderMax - sliderWidth, newX));
            
            sliderX = newX;
            updateVisuals();
        });
        
        document.addEventListener('mouseup', function() {
            isDragging = false;
        });
        
        // 触摸事件支持
        sliderP.addEventListener('touchstart', function(e) {
            isDragging = true;
            const rect = sliderP.getBoundingClientRect();
            dragOffsetX = e.touches[0].clientX - rect.left;
            e.preventDefault();
        });
        
        document.addEventListener('touchmove', function(e) {
            if (!isDragging) return;
            
            const svgRect = svg.getBoundingClientRect();
            let newX = e.touches[0].clientX - svgRect.left - dragOffsetX;
            
            newX = Math.max(sliderMin, Math.min(sliderMax - sliderWidth, newX));
            
            sliderX = newX;
            updateVisuals();
            e.preventDefault();
        });
        
        document.addEventListener('touchend', function() {
            isDragging = false;
        });
        
        // 重置按钮
        resetBtn.addEventListener('click', function() {
            sliderX = 290;
            updateVisuals();
        });
        
        // 初始化
        updateVisuals();
    })();
    </script>
