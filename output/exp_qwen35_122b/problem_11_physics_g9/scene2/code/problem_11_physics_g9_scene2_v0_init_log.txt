    <script>
    (function() {
        const COLORS = {
            main: '#000000',
            blue: '#1565c0',
            red: '#c62828',
            green: '#2e7d32',
            gray: '#888888'
        };
        
        let currentStep = 1;
        const totalSteps = 7;
        
        const btnPrev = document.getElementById('scene2-btn-prev');
        const btnNext = document.getElementById('scene2-btn-next');
        const stepIndicator = document.getElementById('scene2-step-indicator');
        
        const stepElements = {
            1: { show: [], hide: ['scene2-right-hand', 'scene2-magnet-s', 'scene2-magnet-n', 
                                   'scene2-solenoid-n', 'scene2-solenoid-s', 'scene2-field-lines',
                                   'scene2-repulsion-force', 'scene2-repulsion-text', 'scene2-change-hint',
                                   'scene2-magnetic-glow', 'scene2-conclusion'] },
            2: { show: ['scene2-right-hand'], hide: ['scene2-magnet-s', 'scene2-magnet-n', 
                                                     'scene2-solenoid-n', 'scene2-solenoid-s', 'scene2-field-lines',
                                                     'scene2-repulsion-force', 'scene2-repulsion-text', 'scene2-change-hint',
                                                     'scene2-magnetic-glow', 'scene2-conclusion'] },
            3: { show: ['scene2-solenoid-n', 'scene2-solenoid-s'], hide: ['scene2-right-hand', 'scene2-magnet-s', 'scene2-magnet-n',
                                                                           'scene2-field-lines', 'scene2-repulsion-force', 'scene2-repulsion-text',
                                                                           'scene2-change-hint', 'scene2-magnetic-glow', 'scene2-conclusion'] },
            4: { show: ['scene2-magnet-s', 'scene2-magnet-n'], hide: ['scene2-right-hand', 'scene2-field-lines', 'scene2-repulsion-force',
                                                                      'scene2-repulsion-text', 'scene2-change-hint', 'scene2-magnetic-glow',
                                                                      'scene2-conclusion'] },
            5: { show: ['scene2-field-lines', 'scene2-repulsion-force', 'scene2-repulsion-text'], hide: ['scene2-right-hand',
                                                                                                        'scene2-change-hint', 'scene2-magnetic-glow', 'scene2-conclusion'] },
            6: { show: ['scene2-change-hint', 'scene2-magnetic-glow'], hide: ['scene2-right-hand', 'scene2-repulsion-force', 'scene2-repulsion-text',
                                                                              'scene2-conclusion'] },
            7: { show: ['scene2-conclusion'], hide: ['scene2-right-hand', 'scene2-change-hint', 'scene2-magnetic-glow',
                                                     'scene2-repulsion-force', 'scene2-repulsion-text'] }
        };
        
        function showElement(id) {
            const el = document.getElementById(id);
            if (el) {
                el.classList.remove('scene2-hidden');
                el.classList.add('scene2-visible');
                el.style.display = 'block';
                el.style.opacity = '1';
            }
        }
        
        function hideElement(id) {
            const el = document.getElementById(id);
            if (el) {
                el.classList.remove('scene2-visible');
                el.classList.add('scene2-hidden');
                el.style.display = 'none';
                el.style.opacity = '0';
            }
        }
        
        function renderStep() {
            const elements = stepElements[currentStep];
            
            Object.values(stepElements).forEach(step => {
                step.hide.forEach(id => hideElement(id));
            });
            
            elements.show.forEach(showElement);
            
            btnPrev.disabled = (currentStep === 1);
            btnNext.disabled = (currentStep === totalSteps);
            
            stepIndicator.textContent = `步骤 ${currentStep}/${totalSteps}`;
            
            if (currentStep === 6) {
                const sliderArrow = document.getElementById('scene2-slider-arrow');
                if (sliderArrow) {
                    sliderArrow.style.transition = 'transform 0.5s ease';
                    sliderArrow.style.transform = 'translateX(30px)';
                }
                const springPath = document.getElementById('scene2-spring-path');
                if (springPath) {
                    springPath.setAttribute('d', 'M610,50 Q605,65 610,80 Q615,95 610,110 Q605,125 615,140');
                }
            } else if (currentStep < 6) {
                const sliderArrow = document.getElementById('scene2-slider-arrow');
                if (sliderArrow) {
                    sliderArrow.style.transform = 'translateX(0)';
                }
                const springPath = document.getElementById('scene2-spring-path');
                if (springPath) {
                    springPath.setAttribute('d', 'M610,50 Q605,60 610,70 Q615,80 610,90 Q605,100 610,110 Q615,120 610,130 Q605,140 615,150');
                }
            }
        }
        
        btnPrev.addEventListener('click', function() {
            if (currentStep > 1) {
                currentStep--;
                renderStep();
            }
        });
        
        btnNext.addEventListener('click', function() {
            if (currentStep < totalSteps) {
                currentStep++;
                renderStep();
            }
        });
        
        renderStep();
    })();
    </script>
