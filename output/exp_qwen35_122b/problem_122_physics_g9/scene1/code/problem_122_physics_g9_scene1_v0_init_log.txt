    <script>
    (function() {
        // Scene 1 实现
        // 布局坐标表对照：
        // 金属板外框：(200,130)-(600,370) 400×240
        // 接线柱 A：(150,250) 圆圈 + 引线
        // 接线柱 B：(650,250) 圆圈 + 引线
        // 列分隔线：x=333, x=467
        // 列标签：(267,110), (400,110), (533,110)
        // r 标签：每小块中心 (267,165),(267,245),(267,325) 等
        
        const MAX_STEP = 3;
        let currentStep = 1;
        
        // 步骤元素显示映射
        const stepElements = {
            1: [
                'scene1-rect-board', 'scene1-line-grid-h1', 'scene1-line-grid-h2',
                'scene1-line-grid-v1', 'scene1-line-grid-v2',
                'scene1-circle-term-A', 'scene1-line-term-A', 'scene1-circle-term-B', 'scene1-line-term-B',
                'scene1-label-A', 'scene1-label-B', 'scene1-label-R'
            ],
            2: [
                'scene1-line-col-sep-1', 'scene1-line-col-sep-2',
                'scene1-label-col-1', 'scene1-label-col-2', 'scene1-label-col-3',
                'scene1-arrow-current'
            ],
            3: [
                'scene1-arc-parallel-1', 'scene1-arc-parallel-2', 'scene1-arc-parallel-3',
                'scene1-label-r-1', 'scene1-label-r-2', 'scene1-label-r-3',
                'scene1-label-r-4', 'scene1-label-r-5', 'scene1-label-r-6',
                'scene1-label-r-7', 'scene1-label-r-8', 'scene1-label-r-9'
            ],
            4: [
                'scene1-text-formula'
            ]
        };
        
        function showElement(id) {
            const el = document.getElementById(id);
            if (el) {
                el.style.display = 'block';
                el.style.opacity = '1';
            }
        }
        
        function hideElement(id) {
            const el = document.getElementById(id);
            if (el) {
                el.style.opacity = '0';
                setTimeout(() => {
                    if (el.style.opacity === '0') {
                        el.style.display = 'none';
                    }
                }, 300);
            }
        }
        
        function updateStep() {
            const indicator = document.getElementById('scene1-indicator');
            const prevBtn = document.getElementById('scene1-btn-prev');
            const nextBtn = document.getElementById('scene1-btn-next');
            
            // 更新指示器
            indicator.textContent = `步骤 ${currentStep}/${MAX_STEP}`;
            
            // 更新按钮状态
            prevBtn.disabled = currentStep <= 1;
            nextBtn.disabled = currentStep >= MAX_STEP;
            
            // 显示当前步骤及之前步骤的所有元素
            for (let step = 1; step <= currentStep; step++) {
                const elements = stepElements[step];
                if (elements) {
                    elements.forEach(id => showElement(id));
                }
            }
        }
        
        function prevStep() {
            if (currentStep > 1) {
                // 隐藏当前步骤的元素
                const currentElements = stepElements[currentStep];
                if (currentElements) {
                    currentElements.forEach(id => hideElement(id));
                }
                currentStep--;
                updateStep();
            }
        }
        
        function nextStep() {
            if (currentStep < MAX_STEP) {
                currentStep++;
                updateStep();
            }
        }
        
        // 绑定按钮事件
        document.getElementById('scene1-btn-prev').addEventListener('click', prevStep);
        document.getElementById('scene1-btn-next').addEventListener('click', nextStep);
        
        // 初始化
        updateStep();
    })();
    </script>
