    <script>
    (function() {
        // Scene 2: 凸透镜成像规律 - 物距变化对成像的影响
        // 布局坐标表对照：
        // 主光轴：(50,250)-(750,250)
        // 凸透镜：(390,150)-(410,350), 中心 (400,250), ry=100
        // O 点：(400,250), F 左：(320,250), F 右：(480,250)
        // 2F 左：(240,250), 2F 右：(560,250)
        // 蜡烛：(235,200)-(265,280), 火焰：(240,160)-(260,200)
        // 拖拽手柄：(250,240), 半径 12
        // 光屏：(520,180)-(560,240), 支架：(540,250)-(540,320)
        // 数值框：(600,40)-(760,110), 成像性质框：(600,420)-(760,485)
        // 标记点标签 y 坐标调整为 300，"O"标签 x 坐标调整为 410（原 405），避免与凸透镜椭圆重叠
        
        const f = 80; // 焦距 80px
        const lensX = 400; // 透镜光心 x 坐标
        const initialCandleX = 220; // 初始蜡烛位置（2F 左侧）
        
        let candleX = initialCandleX;
        let isDragging = false;
        
        // 获取 DOM 元素
        const svg = document.getElementById('scene2-svg');
        const handle = document.getElementById('scene2-handle');
        const candleBody = document.getElementById('scene2-candle-body');
        const candleFlame = document.getElementById('scene2-candle-flame');
        const screen = document.getElementById('scene2-screen');
        const screenStand = document.getElementById('scene2-screen-stand');
        const imageFlame = document.getElementById('scene2-image-flame');
        const uText = document.getElementById('scene2-u-text');
        const vText = document.getElementById('scene2-v-text');
        const imageProperty = document.getElementById('scene2-image-property');
        const resetBtn = document.getElementById('scene2-reset-btn');
        
        // 计算像的位置和大小
        function calculateImage(cX) {
            const u = lensX - cX; // 物距
            if (Math.abs(u - f) < 1) {
                return { v: Infinity, scale: Infinity, type: 'parallel', imageX: Infinity };
            }
            const v = (f * u) / (u - f); // 像距
            const scale = v / u; // 放大倍数
            const imageX = lensX + v; // 像的 x 坐标
            
            let type;
            if (u > 2 * f) type = '倒立、缩小、实像';
            else if (Math.abs(u - 2 * f) < 1) type = '倒立、等大、实像';
            else if (u > f) type = '倒立、放大、实像';
            else type = '正立、放大、虚像';
            
            return { v, scale, type, imageX };
        }
        
        // 更新图示
        function updateDisplay() {
            const result = calculateImage(candleX);
            
            // 更新蜡烛位置
            candleBody.setAttribute('x', candleX - 15);
            candleFlame.setAttribute('d', `M ${candleX - 10} ${200} Q ${candleX} ${160} ${candleX + 10} ${200}`);
            handle.setAttribute('cx', candleX);
            
            // 更新像的位置和大小
            if (result.type === 'parallel') {
                imageFlame.setAttribute('d', 'M 700 250 L 750 250'); // 平行光线
                imageFlame.setAttribute('opacity', '0.3');
            } else if (result.imageX > 750) {
                // 像在画布外
                imageFlame.setAttribute('d', 'M 700 250 Q 725 290 750 250');
                imageFlame.setAttribute('opacity', '0.3');
            } else {
                const imageHeight = 40 * Math.abs(result.scale);
                
                // 倒立像（实像）
                imageFlame.setAttribute('d', `M ${result.imageX - 15} ${250} Q ${result.imageX} ${250 + imageHeight} ${result.imageX + 15} ${250}`);
                imageFlame.setAttribute('opacity', '0.5');
                
                // 更新光屏位置
                const screenX = Math.min(result.imageX - 20, 730);
                screen.setAttribute('x', screenX);
                screenStand.setAttribute('x1', screenX + 20);
                screenStand.setAttribute('x2', screenX + 20);
            }
            
            // 更新数值显示
            const u = lensX - candleX;
            const v = result.v === Infinity ? '∞' : result.v.toFixed(1);
            uText.textContent = `u = ${u.toFixed(1)} cm`;
            vText.textContent = `v = ${v} cm`;
            
            // 更新成像性质
            if (result.type === 'parallel') {
                imageProperty.textContent = '成像：光线平行，不成像';
            } else {
                imageProperty.textContent = `成像：${result.type}`;
            }
        }
        
        // 拖拽事件处理
        function handleMouseDown(e) {
            const rect = svg.getBoundingClientRect();
            const mouseX = (e.clientX || e.touches[0].clientX) - rect.left;
            const mouseY = (e.clientY || e.touches[0].clientY) - rect.top;
            
            // 检查是否点击了拖拽手柄
            const dx = mouseX - candleX;
            const dy = mouseY - 240;
            if (Math.sqrt(dx * dx + dy * dy) <= 20) {
                isDragging = true;
                e.preventDefault();
            }
        }
        
        function handleMouseMove(e) {
            if (!isDragging) return;
            
            const rect = svg.getBoundingClientRect();
            const mouseX = (e.clientX || e.touches[0].clientX) - rect.left;
            
            // 限制蜡烛位置（不能越过透镜）
            candleX = Math.max(100, Math.min(380, mouseX));
            updateDisplay();
        }
        
        function handleMouseUp() {
            isDragging = false;
        }
        
        // 事件绑定
        handle.addEventListener('mousedown', handleMouseDown);
        handle.addEventListener('touchstart', handleMouseDown);
        window.addEventListener('mousemove', handleMouseMove);
        window.addEventListener('touchmove', handleMouseMove, { passive: false });
        window.addEventListener('mouseup', handleMouseUp);
        window.addEventListener('touchend', handleMouseUp);
        
        // 重置按钮
        resetBtn.addEventListener('click', function() {
            candleX = initialCandleX;
            updateDisplay();
        });
        
        // 初始化
        updateDisplay();
    })();
    </script>
