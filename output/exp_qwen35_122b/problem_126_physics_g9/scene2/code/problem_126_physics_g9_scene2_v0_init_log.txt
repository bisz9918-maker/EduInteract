    <script>
    (function() {
        // Scene 2 实现
        // 布局坐标表对照：
        // 左侧已知条件框：(30,60)-(180,240)
        // 有用功框：(260,60)-(540,135)
        // 总功框：(260,155)-(540,230)
        // 功率框：(260,250)-(540,325)
        // 机械效率框：(260,345)-(540,420)
        // 选项 A/B/C/D：右侧垂直排列 (610,60)-(770,290)
        // 结论框：(610,300)-(770,380)

        const TOTAL_STEPS = 5;
        let currentStep = 0;

        // 工具函数：显示元素（同时设置 display 和 opacity）
        function show(el) {
            if (el) {
                el.style.display = '';
                el.style.opacity = '1';
            }
        }

        // 工具函数：隐藏元素
        function hide(el) {
            if (el) {
                el.style.display = 'none';
                el.style.opacity = '0';
            }
        }

        // 工具函数：高亮闪烁
        function highlight(el) {
            if (el) {
                el.classList.remove('scene2-highlight');
                void el.offsetWidth; // 触发重绘
                el.classList.add('scene2-highlight');
            }
        }

        // 更新显示状态
        function updateDisplay() {
            // 步骤 0：初始状态 - 只显示已知条件和选项
            if (currentStep === 0) {
                hide(document.getElementById('scene2-wyou-box'));
                hide(document.getElementById('scene2-wyou-title'));
                hide(document.getElementById('scene2-wyou-formula'));
                hide(document.getElementById('scene2-wyou-desc'));
                hide(document.getElementById('scene2-wzong-box'));
                hide(document.getElementById('scene2-wzong-title'));
                hide(document.getElementById('scene2-wzong-formula'));
                hide(document.getElementById('scene2-wzong-desc'));
                hide(document.getElementById('scene2-power-box'));
                hide(document.getElementById('scene2-power-title'));
                hide(document.getElementById('scene2-power-formula'));
                hide(document.getElementById('scene2-power-desc'));
                hide(document.getElementById('scene2-eff-box'));
                hide(document.getElementById('scene2-eff-title'));
                hide(document.getElementById('scene2-eff-formula'));
                hide(document.getElementById('scene2-eff-desc'));
                hide(document.getElementById('scene2-conclusion-box'));
                hide(document.getElementById('scene2-conclusion-title'));
                hide(document.getElementById('scene2-conclusion-correct'));
                hide(document.getElementById('scene2-conclusion-wrong'));
                hide(document.getElementById('scene2-arrow1'));
                hide(document.getElementById('scene2-arrow1-text'));
                hide(document.getElementById('scene2-arrow2'));
                hide(document.getElementById('scene2-arrow2-text'));
                hide(document.getElementById('scene2-arrow3a'));
                hide(document.getElementById('scene2-arrow3b'));
                hide(document.getElementById('scene2-arrow3-text'));
            }

            // 步骤 1：显示有用功框
            if (currentStep >= 1) {
                show(document.getElementById('scene2-wyou-box'));
                show(document.getElementById('scene2-wyou-title'));
                show(document.getElementById('scene2-wyou-formula'));
                show(document.getElementById('scene2-wyou-desc'));
                if (currentStep === 1) highlight(document.getElementById('scene2-wyou-box'));
            }

            // 步骤 2：显示总功框 + 箭头 1
            if (currentStep >= 2) {
                show(document.getElementById('scene2-wzong-box'));
                show(document.getElementById('scene2-wzong-title'));
                show(document.getElementById('scene2-wzong-formula'));
                show(document.getElementById('scene2-wzong-desc'));
                show(document.getElementById('scene2-arrow1'));
                show(document.getElementById('scene2-arrow1-text'));
                if (currentStep === 2) highlight(document.getElementById('scene2-wzong-box'));
            }

            // 步骤 3：显示功率框 + 箭头 2
            if (currentStep >= 3) {
                show(document.getElementById('scene2-power-box'));
                show(document.getElementById('scene2-power-title'));
                show(document.getElementById('scene2-power-formula'));
                show(document.getElementById('scene2-power-desc'));
                show(document.getElementById('scene2-arrow2'));
                show(document.getElementById('scene2-arrow2-text'));
                if (currentStep === 3) highlight(document.getElementById('scene2-power-box'));
            }

            // 步骤 4：显示机械效率框 + 箭头 3
            if (currentStep >= 4) {
                show(document.getElementById('scene2-eff-box'));
                show(document.getElementById('scene2-eff-title'));
                show(document.getElementById('scene2-eff-formula'));
                show(document.getElementById('scene2-eff-desc'));
                show(document.getElementById('scene2-arrow3a'));
                show(document.getElementById('scene2-arrow3b'));
                show(document.getElementById('scene2-arrow3-text'));
                if (currentStep === 4) highlight(document.getElementById('scene2-eff-box'));
            }

            // 步骤 5：显示结论框
            if (currentStep >= 5) {
                show(document.getElementById('scene2-conclusion-box'));
                show(document.getElementById('scene2-conclusion-title'));
                show(document.getElementById('scene2-conclusion-correct'));
                show(document.getElementById('scene2-conclusion-wrong'));
                if (currentStep === 5) {
                    highlight(document.getElementById('scene2-wyou-box'));
                    highlight(document.getElementById('scene2-wzong-box'));
                    highlight(document.getElementById('scene2-power-box'));
                    highlight(document.getElementById('scene2-eff-box'));
                }
            }

            // 更新按钮状态
            document.getElementById('scene2-prev-btn').disabled = (currentStep === 0);
            document.getElementById('scene2-next-btn').disabled = (currentStep === TOTAL_STEPS);

            // 更新步骤指示器
            document.getElementById('scene2-step-indicator').textContent = '步骤 ' + currentStep + '/' + TOTAL_STEPS;
        }

        // 绑定按钮事件
        document.getElementById('scene2-prev-btn').addEventListener('click', function() {
            if (currentStep > 0) {
                currentStep--;
                updateDisplay();
            }
        });

        document.getElementById('scene2-next-btn').addEventListener('click', function() {
            if (currentStep < TOTAL_STEPS) {
                currentStep++;
                updateDisplay();
            }
        });

        // 初始化显示
        updateDisplay();
    })();
    </script>
