    <script>
    (function() {
        // Scene 2 实现 - 均摊法计算晶体结构
        const currentStepMax = 4;
        let currentStep = 0;
        
        // 获取 DOM 元素
        const prevBtn = document.getElementById('scene2-prev-btn');
        const nextBtn = document.getElementById('scene2-next-btn');
        const correctBox = document.getElementById('scene2-correct-box');
        
        // 微粒元素分组
        const vertexMicrobes = document.querySelectorAll('[data-type="A"]');
        const edgeMicrobes = document.querySelectorAll('[data-type="B"]');
        const bodyMicrobe = document.querySelectorAll('[data-type="C"]');
        
        // 步骤文本元素
        const stepTexts = {
            0: document.getElementById('scene2-step0-text'),
            1: document.getElementById('scene2-step1-text'),
            1desc: document.getElementById('scene2-step1-desc'),
            2: document.getElementById('scene2-step2-text'),
            2desc: document.getElementById('scene2-step2-desc'),
            3: document.getElementById('scene2-step3-text'),
            3desc: document.getElementById('scene2-step3-desc'),
            4: document.getElementById('scene2-step4-text')
        };
        
        // 工具函数：显示元素
        function show(el) {
            if (el) {
                el.style.display = 'block';
                el.style.opacity = '1';
            }
        }
        
        // 工具函数：隐藏元素
        function hide(el) {
            if (el) {
                el.style.display = 'none';
                el.style.opacity = '0';
            }
        }
        
        // 工具函数：高亮微粒
        function highlightMicrobes(microbes, color, opacity = 0.2) {
            microbes.forEach(m => {
                m.style.fill = color;
                m.style.fillOpacity = opacity;
            });
        }
        
        // 工具函数：恢复微粒为黑色
        function resetMicrobes(microbes) {
            microbes.forEach(m => {
                if (m.getAttribute('data-type') === 'C') {
                    m.style.fill = 'none';
                    m.style.stroke = '#000000';
                } else {
                    m.style.fill = '#000000';
                    m.style.fillOpacity = '1';
                }
            });
        }
        
        // 更新场景显示
        function updateScene() {
            // 隐藏所有步骤文本
            Object.values(stepTexts).forEach(el => hide(el));
            
            // 恢复所有微粒为黑色
            resetMicrobes(vertexMicrobes);
            resetMicrobes(edgeMicrobes);
            resetMicrobes(bodyMicrobe);
            
            // 隐藏正确答案框
            hide(correctBox);
            
            // 根据当前步骤更新显示
            switch(currentStep) {
                case 0:
                    show(stepTexts[0]);
                    prevBtn.disabled = true;
                    nextBtn.textContent = '下一步 ▶';
                    break;
                    
                case 1:
                    show(stepTexts[1]);
                    show(stepTexts[1desc]);
                    highlightMicrobes(vertexMicrobes, '#1565c0', 0.2);
                    prevBtn.disabled = false;
                    nextBtn.textContent = '下一步 ▶';
                    break;
                    
                case 2:
                    show(stepTexts[1]);
                    show(stepTexts[1desc]);
                    show(stepTexts[2]);
                    show(stepTexts[2desc]);
                    highlightMicrobes(vertexMicrobes, '#1565c0', 0.2);
                    highlightMicrobes(edgeMicrobes, '#2e7d32', 0.2);
                    prevBtn.disabled = false;
                    nextBtn.textContent = '下一步 ▶';
                    break;
                    
                case 3:
                    show(stepTexts[1]);
                    show(stepTexts[1desc]);
                    show(stepTexts[2]);
                    show(stepTexts[2desc]);
                    show(stepTexts[3]);
                    show(stepTexts[3desc]);
                    highlightMicrobes(vertexMicrobes, '#1565c0', 0.2);
                    highlightMicrobes(edgeMicrobes, '#2e7d32', 0.2);
                    highlightMicrobes(bodyMicrobe, '#c62828', 0.2);
                    prevBtn.disabled = false;
                    nextBtn.textContent = '下一步 ▶';
                    break;
                    
                case 4:
                    show(stepTexts[4]);
                    show(correctBox);
                    resetMicrobes(vertexMicrobes);
                    resetMicrobes(edgeMicrobes);
                    resetMicrobes(bodyMicrobe);
                    prevBtn.disabled = false;
                    nextBtn.textContent = '完成';
                    nextBtn.disabled = true;
                    break;
            }
        }
        
        // 事件监听
        prevBtn.addEventListener('click', () => {
            if (currentStep > 0) {
                currentStep--;
                updateScene();
            }
        });
        
        nextBtn.addEventListener('click', () => {
            if (currentStep < currentStepMax) {
                currentStep++;
                updateScene();
            }
        });
        
        // 初始化
        updateScene();
    })();
    </script>
