    <script>
    (function() {
        // Scene 2: C₅H₁₁Cl 同分异构体结构展示
        // 状态机：0=全部，1=直链，2=支链
        var currentState = 0;
        
        // 获取 DOM 元素
        var struct1 = document.getElementById('scene2-struct-1');
        var struct2 = document.getElementById('scene2-struct-2');
        var struct3 = document.getElementById('scene2-struct-3');
        var struct4 = document.getElementById('scene2-struct-4');
        
        var btnAll = document.getElementById('scene2-btn-all');
        var btnChain = document.getElementById('scene2-btn-chain');
        var btnBranch = document.getElementById('scene2-btn-branch');
        
        // 更新按钮状态
        function updateButtonStates(activeState) {
            btnAll.classList.remove('active-all');
            btnChain.classList.remove('active-chain');
            btnBranch.classList.remove('active-branch');
            
            if (activeState === 0) {
                btnAll.classList.add('active-all');
            } else if (activeState === 1) {
                btnChain.classList.add('active-chain');
            } else if (activeState === 2) {
                btnBranch.classList.add('active-branch');
            }
        }
        
        // 更新结构显示状态
        function updateStructures(state) {
            if (state === 0) {
                // 显示全部
                struct1.setAttribute('opacity', '1');
                struct2.setAttribute('opacity', '1');
                struct3.setAttribute('opacity', '1');
                struct4.setAttribute('opacity', '1');
            } else if (state === 1) {
                // 仅直链（结构 1、2 正常显示，结构 3、4 半透明）
                struct1.setAttribute('opacity', '1');
                struct2.setAttribute('opacity', '1');
                struct3.setAttribute('opacity', '0.2');
                struct4.setAttribute('opacity', '0.2');
            } else if (state === 2) {
                // 仅支链（结构 3、4 正常显示，结构 1、2 半透明）
                struct1.setAttribute('opacity', '0.2');
                struct2.setAttribute('opacity', '0.2');
                struct3.setAttribute('opacity', '1');
                struct4.setAttribute('opacity', '1');
            }
        }
        
        // 按钮事件绑定
        btnAll.addEventListener('click', function() {
            currentState = 0;
            updateButtonStates(currentState);
            updateStructures(currentState);
        });
        
        btnChain.addEventListener('click', function() {
            currentState = 1;
            updateButtonStates(currentState);
            updateStructures(currentState);
        });
        
        btnBranch.addEventListener('click', function() {
            currentState = 2;
            updateButtonStates(currentState);
            updateStructures(currentState);
        });
        
        // 初始化：默认显示全部
        updateButtonStates(0);
        updateStructures(0);
    })();
    </script>
