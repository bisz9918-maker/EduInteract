    <script>
    (function() {
        'use strict';
        
        let currentStep = 0;
        const totalSteps = 7;
        
        function show(el) {
            if (el) {
                el.style.display = '';
                el.style.opacity = '1';
                el.classList.add('scene1-fade-in');
            }
        }
        
        function hide(el) {
            if (el) {
                el.style.opacity = '0';
                setTimeout(() => {
                    el.style.display = 'none';
                }, 300);
                el.classList.remove('scene1-fade-in');
            }
        }
        
        function updateScene() {
            const indicator = document.getElementById('scene1-step-indicator');
            if (indicator) {
                indicator.textContent = '步骤 ' + (currentStep + 1) + '/' + totalSteps;
            }
            
            const prevBtn = document.getElementById('scene1-prev-btn');
            const nextBtn = document.getElementById('scene1-next-btn');
            if (prevBtn) {
                prevBtn.disabled = (currentStep === 0);
            }
            if (nextBtn) {
                nextBtn.disabled = (currentStep === totalSteps - 1);
                if (currentStep === totalSteps - 1) {
                    nextBtn.textContent = '完成';
                } else {
                    nextBtn.textContent = '下一步 ▶';
                }
            }
            
            if (currentStep >= 1) {
                show(document.getElementById('scene1-highlight-right'));
                show(document.getElementById('scene1-electrode-zn-highlight'));
                show(document.getElementById('scene1-electrode-c-highlight'));
                show(document.getElementById('scene1-galvanic-group'));
                const switchPath = document.getElementById('scene1-switch');
                if (switchPath) {
                    switchPath.setAttribute('d', 'M 254 110 L 320 110');
                }
            }
            
            if (currentStep >= 2) {
                show(document.getElementById('scene1-highlight-left'));
                show(document.getElementById('scene1-electrode-cu-highlight'));
                show(document.getElementById('scene1-electrode-pt-highlight'));
                show(document.getElementById('scene1-electrolytic-group'));
            }
            
            if (currentStep >= 3) {
                show(document.getElementById('scene1-galvanic-reactions-group'));
            }
            
            if (currentStep >= 4) {
                show(document.getElementById('scene1-electrolytic-reactions-group'));
            }
            
            if (currentStep >= 5) {
                show(document.getElementById('scene1-ions-group'));
            }
            
            if (currentStep >= 6) {
                show(document.getElementById('scene1-option-highlight'));
                show(document.getElementById('scene1-cu-oh-notice'));
            }
        }
        
        document.getElementById('scene1-prev-btn').addEventListener('click', function() {
            if (currentStep > 0) {
                const groups = ['galvanic-group', 'electrolytic-group', 'galvanic-reactions-group', 'electrolytic-reactions-group', 'ions-group'];
                const highlights = ['highlight-left', 'highlight-right', 'electrode-pt-highlight', 'electrode-cu-highlight', 'electrode-zn-highlight', 'electrode-c-highlight'];
                const others = ['option-highlight', 'cu-oh-notice'];
                
                groups.forEach(id => {
                    const el = document.getElementById('scene1-' + id);
                    if (el) { el.style.display = 'none'; el.style.opacity = '0'; }
                });
                highlights.forEach(id => {
                    const el = document.getElementById('scene1-' + id);
                    if (el) { el.style.display = 'none'; el.style.opacity = '0'; }
                });
                others.forEach(id => {
                    const el = document.getElementById('scene1-' + id);
                    if (el) { el.style.display = 'none'; el.style.opacity = '0'; }
                });
                
                const switchPath = document.getElementById('scene1-switch');
                if (switchPath) {
                    switchPath.setAttribute('d', 'M 254 110 L 280 110 M 290 110 L 320 110');
                }
                
                currentStep--;
                updateScene();
            }
        });
        
        document.getElementById('scene1-next-btn').addEventListener('click', function() {
            if (currentStep < totalSteps - 1) {
                currentStep++;
                updateScene();
            }
        });
        
        updateScene();
    })();
    </script>
