    <script>
    (function() {
        // Scene 1: 溶液导电性测定实验装置
        // 布局坐标表（参考）：
        // | 元素 | 位置/范围 | 备注 |
        // | 电池组 | (320,60)-(480,110) | 两节电池串联 |
        // | 灯泡 | (140,300)-(240,370) | 椭圆玻璃泡 + 矩形底座 |
        // | 烧杯 | (550,280)-(700,420) | 梯形轮廓 |
        // | 溶液区域 | (560,320)-(690,410) | 蓝色填充 |
        // | 电极左 | (580,220)-(600,400) | 垂直矩形 |
        // | 电极右 | (660,220)-(680,400) | 垂直矩形 |

        const TOTAL_STEPS = 5;
        let currentStep = 0;

        // 获取 DOM 元素
        const svg = document.getElementById('scene1-svg');
        const bulbGlass = document.getElementById('scene1-bulb-glass');
        const circuitText = document.getElementById('scene1-circuit-text');
        const electrodeText = document.getElementById('scene1-electrode-text');
        const strongText = document.getElementById('scene1-strong-text');
        const weakText = document.getElementById('scene1-weak-text');
        const conclusionText = document.getElementById('scene1-conclusion-text');
        const stepIndicator = document.getElementById('scene1-step-indicator');
        const btnPrev = document.getElementById('scene1-btn-prev');
        const btnNext = document.getElementById('scene1-btn-next');

        // 步骤配置
        const stepConfig = [
            {
                // 步骤 0：初始状态 - 显示完整实验装置
                bulbOpacity: 0.1,
                showCircuitText: false,
                showElectrodeText: false,
                showStrongText: false,
                showWeakText: false,
                showConclusion: false,
                label: '步骤 1/5：实验装置全貌'
            },
            {
                // 步骤 1：标注电路连接
                bulbOpacity: 0.1,
                showCircuitText: true,
                showElectrodeText: false,
                showStrongText: false,
                showWeakText: false,
                showConclusion: false,
                label: '步骤 2/5：电路连接'
            },
            {
                // 步骤 2：标注溶液与电极
                bulbOpacity: 0.15,
                showCircuitText: false,
                showElectrodeText: true,
                showStrongText: false,
                showWeakText: false,
                showConclusion: false,
                label: '步骤 3/5：电极与溶液'
            },
            {
                // 步骤 3：灯泡亮度与导电性关系
                bulbOpacity: 0.3,
                showCircuitText: false,
                showElectrodeText: false,
                showStrongText: true,
                showWeakText: true,
                showConclusion: false,
                label: '步骤 4/5：灯泡亮度与导电性'
            },
            {
                // 步骤 4：结论
                bulbOpacity: 0.1,
                showCircuitText: false,
                showElectrodeText: false,
                showStrongText: false,
                showWeakText: false,
                showConclusion: true,
                label: '步骤 5/5：结论'
            }
        ];

        // 更新步骤显示
        function renderStep() {
            const config = stepConfig[currentStep];

            // 更新灯泡亮度
            bulbGlass.style.fillOpacity = config.bulbOpacity;

            // 更新文字显示
            circuitText.style.display = config.showCircuitText ? 'block' : 'none';
            electrodeText.style.display = config.showElectrodeText ? 'block' : 'none';
            strongText.style.display = config.showStrongText ? 'block' : 'none';
            weakText.style.display = config.showWeakText ? 'block' : 'none';
            conclusionText.style.display = config.showConclusion ? 'block' : 'none';

            // 更新步骤指示器
            stepIndicator.textContent = config.label;

            // 更新按钮状态
            btnPrev.disabled = currentStep === 0;
            btnNext.disabled = currentStep === TOTAL_STEPS - 1;
        }

        // 事件绑定
        btnPrev.addEventListener('click', function() {
            if (currentStep > 0) {
                currentStep--;
                renderStep();
            }
        });

        btnNext.addEventListener('click', function() {
            if (currentStep < TOTAL_STEPS - 1) {
                currentStep++;
                renderStep();
            }
        });

        // 初始化
        renderStep();
    })();
    </script>
