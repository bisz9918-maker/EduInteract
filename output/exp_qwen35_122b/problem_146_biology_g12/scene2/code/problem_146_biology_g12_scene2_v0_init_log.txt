    <script>
    (function() {
        // Scene 2: 果蝇减数分裂染色体行为
        // 布局坐标表（实现对照依据）：
        // | 元素 | 类型 | 位置/范围 | 大小 | 备注 |
        // |------|------|-----------|------|------|
        // | 细胞轮廓 | 椭圆 | 中心 (400,240) | 300×260 | 主细胞边界 |
        // | 赤道板 | 虚线 | (250,240)-(550,240) | 300×1 | 水平居中 |
        // | 纺锤体左极 | 点 | (400,100) | 4×4 | 纺锤体上极 |
        // | 纺锤体右极 | 点 | (400,380) | 4×4 | 纺锤体下极 |
        // | 常染色体 1 对 | X 形 | 中心 (350,240) | 20×40 | 减Ⅰ中期 |
        // | 常染色体 2 对 | X 形 | 中心 (390,230) | 20×40 | 减Ⅰ中期 |
        // | 常染色体 3 对 | X 形 | 中心 (430,240) | 20×40 | 减Ⅰ中期 |
        // | X 染色体 | X 形 | 中心 (370,240) | 20×20 | 蓝色 |
        // | Y 染色体 | X 形 | 中心 (430,240) | 20×20 | 红色 |
        
        // 状态变量
        var currentStep = 0;
        var totalSteps = 4;
        
        // DOM 元素引用
        var titleEl = document.getElementById('scene2-title');
        var prevBtn = document.getElementById('scene2-prev-btn');
        var nextBtn = document.getElementById('scene2-next-btn');
        var equatorLabelEl = document.getElementById('scene2-equator-label');
        var spindleFibersEl = document.getElementById('scene2-spindle-fibers');
        var chromo1El = document.getElementById('scene2-chromo-1');
        var chromo2El = document.getElementById('scene2-chromo-2');
        var chromo3El = document.getElementById('scene2-chromo-3');
        var xChromoEl = document.getElementById('scene2-x-chromo');
        var yChromoEl = document.getElementById('scene2-y-chromo');
        var secondaryLeftEl = document.getElementById('scene2-secondary-left');
        var secondaryRightEl = document.getElementById('scene2-secondary-right');
        var meiosis2LeftEl = document.getElementById('scene2-meiosis2-left');
        var meiosis2RightEl = document.getElementById('scene2-meiosis2-right');
        var chromoCountEl = document.getElementById('scene2-chromo-count');
        var chromoSetEl = document.getElementById('scene2-chromo-set');
        var summaryEl = document.getElementById('scene2-summary');
        
        // 工具函数：显示元素（同时设置 display 和 opacity）
        function show(el) {
            if (!el) return;
            el.style.display = '';
            el.style.opacity = '1';
        }
        
        // 工具函数：隐藏元素（同时设置 display 和 opacity）
        function hide(el) {
            if (!el) return;
            el.style.display = 'none';
            el.style.opacity = '0';
        }
        
        // 更新场景状态
        function updateScene() {
            // 重置所有元素
            hide(spindleFibersEl);
            hide(chromo1El);
            hide(chromo2El);
            hide(chromo3El);
            hide(xChromoEl);
            hide(yChromoEl);
            hide(secondaryLeftEl);
            hide(secondaryRightEl);
            hide(meiosis2LeftEl);
            hide(meiosis2RightEl);
            hide(chromoCountEl);
            hide(chromoSetEl);
            hide(summaryEl);
            
            switch(currentStep) {
                case 0: // 初始状态
                    titleEl.textContent = '减数分裂过程';
                    equatorLabelEl.style.display = 'none';
                    prevBtn.disabled = true;
                    nextBtn.disabled = false;
                    break;
                    
                case 1: // 减数第一次分裂中期
                    titleEl.textContent = '减数第一次分裂中期';
                    show(spindleFibersEl);
                    show(chromo1El);
                    show(chromo2El);
                    show(chromo3El);
                    show(xChromoEl);
                    show(yChromoEl);
                    show(chromoCountEl);
                    show(chromoSetEl);
                    chromoCountEl.textContent = '染色体数：8';
                    chromoSetEl.textContent = '染色体组数：2';
                    equatorLabelEl.style.display = '';
                    prevBtn.disabled = false;
                    nextBtn.disabled = false;
                    break;
                    
                case 2: // 减数第二次分裂后期
                    titleEl.textContent = '减数第二次分裂后期';
                    show(secondaryLeftEl);
                    show(secondaryRightEl);
                    show(meiosis2LeftEl);
                    show(meiosis2RightEl);
                    show(chromoCountEl);
                    show(chromoSetEl);
                    chromoCountEl.textContent = '染色体数：8（每个细胞 4 条）';
                    chromoSetEl.textContent = '染色体组数：1（每个细胞）';
                    equatorLabelEl.style.display = '';
                    prevBtn.disabled = false;
                    nextBtn.disabled = false;
                    break;
                    
                case 3: // 结论总结
                    titleEl.textContent = '减数分裂染色体行为总结';
                    show(secondaryLeftEl);
                    show(secondaryRightEl);
                    show(meiosis2LeftEl);
                    show(meiosis2RightEl);
                    show(summaryEl);
                    equatorLabelEl.style.display = '';
                    prevBtn.disabled = false;
                    nextBtn.disabled = true;
                    break;
            }
        }
        
        // 事件绑定
        prevBtn.addEventListener('click', function() {
            if (currentStep > 0) {
                currentStep--;
                updateScene();
            }
        });
        
        nextBtn.addEventListener('click', function() {
            if (currentStep < totalSteps - 1) {
                currentStep++;
                updateScene();
            }
        });
        
        // 初始化
        updateScene();
    })();
    </script>
