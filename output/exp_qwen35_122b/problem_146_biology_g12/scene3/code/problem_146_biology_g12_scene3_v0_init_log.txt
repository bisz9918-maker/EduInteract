    <script>
    (function() {
        // Scene 3: X^rX^rY 雌果蝇减数分裂与配子形成
        // 布局坐标表对照：
        // 主细胞：(400,200) 半径 100×80
        // X^r 染色体 1: (370,180) 蓝色
        // X^r 染色体 2: (400,180) 蓝色
        // Y 染色体：(430,185) 红色
        // 纺锤体：(400,140)-(400,260) 垂直虚线
        // 配子细胞：(150,340), (280,340), (410,340), (540,340)
        // 子代细胞：(730,220)

        const currentStepMax = 5;
        let currentStep = 0;

        // 获取 DOM 元素
        const stepLabel = document.getElementById('scene3-step-label');
        const mainCell = document.getElementById('scene3-main-cell');
        const spindle = document.getElementById('scene3-spindle');
        const spindleFiberLeft = document.getElementById('scene3-spindle-fiber-left');
        const spindleFiberRight = document.getElementById('scene3-spindle-fiber-right');
        const chromX1 = document.getElementById('scene3-chrom-x1');
        const chromX2 = document.getElementById('scene3-chrom-x2');
        const chromY = document.getElementById('scene3-chrom-y');
        const chromX1Label = document.getElementById('scene3-chrom-x1-label');
        const chromX2Label = document.getElementById('scene3-chrom-x2-label');
        const chromYLabel = document.getElementById('scene3-chrom-y-label');
        const subcellLeft = document.getElementById('scene3-subcell-left');
        const subcellRight = document.getElementById('scene3-subcell-right');
        const gametes = [
            document.getElementById('scene3-gamete-1'),
            document.getElementById('scene3-gamete-2'),
            document.getElementById('scene3-gamete-3'),
            document.getElementById('scene3-gamete-4')
        ];
        const gameteLabels = [
            document.getElementById('scene3-gamete-label-1'),
            document.getElementById('scene3-gamete-label-2'),
            document.getElementById('scene3-gamete-label-3'),
            document.getElementById('scene3-gamete-label-4')
        ];
        const hybridArrow = document.getElementById('scene3-hybrid-arrow');
        const hybridArrowhead = document.getElementById('scene3-hybrid-arrowhead');
        const offspringCell = document.getElementById('scene3-offspring-cell');
        const offspringGenotype = document.getElementById('scene3-offspring-genotype');
        const prevBtn = document.getElementById('scene3-prev');
        const nextBtn = document.getElementById('scene3-next');

        // 步骤说明文本
        const stepDescriptions = [
            '初始：X^rX^rY 雌果蝇细胞',
            '减数第一次分裂：两条 X 联会，Y 单独',
            '分离方式 1：X^r 与 X^rY 分离',
            '分离方式 2：X^rX^r 与 Y 分离',
            '四种配子类型：X^r、X^rX^r、X^rY、Y',
            '与 X^R Y 杂交，红眼雌果蝇基因型'
        ];

        // 工具函数：显示/隐藏元素
        function show(el) {
            if (el) {
                el.style.display = 'block';
                el.style.opacity = '1';
            }
        }

        function hide(el) {
            if (el) {
                el.style.display = 'none';
                el.style.opacity = '0';
            }
        }

        // 更新场景状态
        function updateScene() {
            // 更新步骤标签
            stepLabel.textContent = stepDescriptions[currentStep];

            // 重置所有元素状态
            spindleFiberLeft.style.display = 'none';
            spindleFiberRight.style.display = 'none';
            subcellLeft.style.display = 'none';
            subcellRight.style.display = 'none';
            gametes.forEach(g => g.style.display = 'none');
            gameteLabels.forEach(l => l.style.display = 'none');
            hybridArrow.style.display = 'none';
            hybridArrowhead.style.display = 'none';
            offspringCell.style.display = 'none';
            offspringGenotype.style.display = 'none';

            // 重置染色体位置到初始状态
            chromX1.setAttribute('transform', 'translate(370,180)');
            chromX2.setAttribute('transform', 'translate(400,180)');
            chromY.setAttribute('transform', 'translate(430,185)');
            chromX1Label.setAttribute('x', '340');
            chromX1Label.setAttribute('y', '205');
            chromX2Label.setAttribute('x', '415');
            chromX2Label.setAttribute('y', '205');
            chromYLabel.setAttribute('x', '460');
            chromYLabel.setAttribute('y', '205');

            switch(currentStep) {
                case 0:
                    // 初始状态：显示主细胞、三条染色体、纺锤体
                    show(mainCell);
                    show(spindle);
                    chromX1.setAttribute('transform', 'translate(370,180)');
                    chromX2.setAttribute('transform', 'translate(400,180)');
                    chromY.setAttribute('transform', 'translate(430,185)');
                    break;

                case 1:
                    // 减数第一次分裂：X 联会，Y 单独
                    show(mainCell);
                    show(spindle);
                    show(spindleFiberLeft);
                    show(spindleFiberRight);
                    chromX1.setAttribute('transform', 'translate(350,190)');
                    chromX2.setAttribute('transform', 'translate(365,190)');
                    chromY.setAttribute('transform', 'translate(450,190)');
                    chromX1Label.setAttribute('x', '330');
                    chromX1Label.setAttribute('y', '215');
                    chromX2Label.setAttribute('x', '380');
                    chromX2Label.setAttribute('y', '215');
                    chromYLabel.setAttribute('x', '475');
                    chromYLabel.setAttribute('y', '215');
                    break;

                case 2:
                    // 分离方式 1: X 与 XY 分离
                    show(mainCell);
                    show(spindle);
                    show(subcellLeft);
                    show(subcellRight);
                    chromX1.setAttribute('transform', 'translate(320,160)');
                    chromX2.setAttribute('transform', 'translate(470,180)');
                    chromY.setAttribute('transform', 'translate(485,180)');
                    chromX1Label.setAttribute('x', '300');
                    chromX1Label.setAttribute('y', '185');
                    chromX2Label.setAttribute('x', '485');
                    chromX2Label.setAttribute('y', '205');
                    chromYLabel.setAttribute('x', '510');
                    chromYLabel.setAttribute('y', '205');
                    break;

                case 3:
                    // 分离方式 2: XX 与 Y 分离
                    show(mainCell);
                    show(spindle);
                    show(subcellLeft);
                    show(subcellRight);
                    chromX1.setAttribute('transform', 'translate(310,170)');
                    chromX2.setAttribute('transform', 'translate(325,170)');
                    chromY.setAttribute('transform', 'translate(480,190)');
                    chromX1Label.setAttribute('x', '290');
                    chromX1Label.setAttribute('y', '195');
                    chromX2Label.setAttribute('x', '345');
                    chromX2Label.setAttribute('y', '195');
                    chromYLabel.setAttribute('x', '505');
                    chromYLabel.setAttribute('y', '215');
                    break;

                case 4:
                    // 四种配子类型
                    hide(mainCell);
                    hide(spindle);
                    show(gametes[0]);
                    show(gametes[1]);
                    show(gametes[2]);
                    show(gametes[3]);
                    show(gameteLabels[0]);
                    show(gameteLabels[1]);
                    show(gameteLabels[2]);
                    show(gameteLabels[3]);
                    break;

                case 5:
                    // 杂交结果
                    show(gametes[0]);
                    show(gametes[1]);
                    show(gametes[2]);
                    show(gametes[3]);
                    show(gameteLabels[0]);
                    show(gameteLabels[1]);
                    show(gameteLabels[2]);
                    show(gameteLabels[3]);
                    show(hybridArrow);
                    show(hybridArrowhead);
                    show(offspringCell);
                    show(offspringGenotype);
                    break;
            }

            // 更新按钮状态
            prevBtn.disabled = (currentStep === 0);
            nextBtn.disabled = (currentStep === currentStepMax);

            // 触发 MathJax 重新渲染公式
            if (window.MathJax) {
                MathJax.typesetPromise();
            }
        }

        // 按钮事件处理
        prevBtn.addEventListener('click', function() {
            if (currentStep > 0) {
                currentStep--;
                updateScene();
            }
        });

        nextBtn.addEventListener('click', function() {
            if (currentStep < currentStepMax) {
                currentStep++;
                updateScene();
            }
        });

        // 初始化场景
        updateScene();
    })();
    </script>
</body>
