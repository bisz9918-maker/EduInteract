    <script>
    (function() {
        // Scene 2 实现
        // 布局坐标表：
        // | 元素 | 类型 | 位置/范围 | 大小 | 备注 |
        // |------|------|-----------|------|------|
        // | 已知条件框 | rect | (20,40)-(270,180) | 250×140 | 左上区域 |
        // | 公式推导区标题 | text | (290,62) | - | 右上区域顶部 |
        // | 公式框 | rect+text | (290,85)-(540,125) | 250×40 | 动态显示 |
        // | 滑轮组示意图 | svg | (570,40)-(780,180) | 210×140 | 右上区域右侧 |
        // | 计算过程区标题 | text | (20,222) | - | 左下区域顶部 |
        // | 计算结果框 | rect+text | (20,245)-(270,345) | 250×100 | 动态显示 |
        // | 选项判断区标题 | text | (290,222) | - | 右下区域顶部 |
        // | 选项 A 框 | rect+text | (290,245)-(410,295) | 120×50 | 2×2 网格左上 |
        // | 选项 B 框 | rect+text | (430,245)-(550,295) | 120×50 | 2×2 网格右上 |
        // | 选项 C 框 | rect+text | (290,310)-(410,360) | 120×50 | 2×2 网格左下 |
        // | 选项 D 框 | rect+text | (430,310)-(550,360) | 120×50 | 2×2 网格右下 |
        // | 控件栏 | div | (20,440)-(780,490) | 760×50 | SVG 内底部 |

        const MAX_STEP = 5;
        let currentStep = 0;

        // 工具函数：显示元素
        function show(el) {
            if (el) {
                el.style.display = 'block';
                el.style.opacity = '1';
            }
        }

        // 工具函数：隐藏元素
        function hide(el) {
            if (el) {
                el.style.display = 'none';
                el.style.opacity = '0';
            }
        }

        // 获取元素
        const formulaBox = document.getElementById('scene2-formula-box');
        const formulaBorder = document.getElementById('scene2-formula-border');
        const formulaContent = document.getElementById('scene2-formula-content');
        const calcBox = document.getElementById('scene2-calc-box');
        const calcBorder = document.getElementById('scene2-calc-border');
        const calcContent = document.getElementById('scene2-calc-content');
        const optionAMark = document.getElementById('scene2-option-a-mark');
        const optionBMark = document.getElementById('scene2-option-b-mark');
        const optionCMark = document.getElementById('scene2-option-c-mark');
        const optionDMark = document.getElementById('scene2-option-d-mark');
        const btnPrev = document.getElementById('scene2-btn-prev');
        const btnNext = document.getElementById('scene2-btn-next');

        // 步骤内容配置
        const stepConfig = [
            // 步骤 0：初始状态
            {
                formula: '',
                calc: '',
                marks: { a: false, b: false, c: false, d: false }
            },
            // 步骤 1：显示 η = f/(nF)
            {
                formula: 'η = f/(nF)',
                calc: '',
                marks: { a: false, b: false, c: false, d: false }
            },
            // 步骤 2：显示 f = η·n·F，计算 f=12N，标记 B 正确
            {
                formula: 'f = η·n·F',
                calc: 'f = 0.8×3×5 = 12N',
                marks: { a: false, b: true, c: false, d: false }
            },
            // 步骤 3：显示 P = F·n·v_物，计算 P=15W，标记 C 错误
            {
                formula: 'P = F·n·v_物',
                calc: 'P = 5×3×1 = 15W',
                marks: { a: false, b: true, c: false, d: false }
            },
            // 步骤 4：显示 W_有 = f·v·t，计算 W=120J，标记 D 错误
            {
                formula: 'W_有 = f·v·t',
                calc: 'W_有 = 12×1×10 = 120J',
                marks: { a: false, b: true, c: false, d: false }
            },
            // 步骤 5：最终判断
            {
                formula: '选项判断',
                calc: 'A: 匀速运动合力=0×\nB: f=12N√\nC: P=15W≠5W×\nD: W=120J≠500J×',
                marks: { a: false, b: true, c: false, d: false }
            }
        ];

        // 更新页面显示
        function updateDisplay() {
            const config = stepConfig[currentStep];

            // 更新公式框
            if (config.formula) {
                show(formulaBox);
                show(formulaBorder);
                show(formulaContent);
                formulaContent.textContent = config.formula;
                formulaBorder.setAttribute('stroke-width', '3');
            } else {
                hide(formulaBox);
                hide(formulaBorder);
                hide(formulaContent);
            }

            // 更新计算框
            if (config.calc) {
                show(calcBox);
                show(calcBorder);
                show(calcContent);
                // 处理多行文本
                if (config.calc.includes('\n')) {
                    const lines = config.calc.split('\n');
                    calcContent.innerHTML = '';
                    let y = 270;
                    lines.forEach((line, i) => {
                        const tspan = document.createElementNS('http://www.w3.org/2000/svg', 'tspan');
                        tspan.setAttribute('x', '35');
                        tspan.setAttribute('y', y + i * 20);
                        tspan.setAttribute('font-size', '12');
                        tspan.setAttribute('fill', '#2e7d32');
                        tspan.textContent = line;
                        calcContent.appendChild(tspan);
                    });
                } else {
                    calcContent.textContent = config.calc;
                }
            } else {
                hide(calcBox);
                hide(calcBorder);
                hide(calcContent);
            }

            // 更新选项标记
            optionAMark.style.display = config.marks.a ? 'block' : 'none';
            optionBMark.style.display = config.marks.b ? 'block' : 'none';
            optionCMark.style.display = config.marks.c ? 'block' : 'none';
            optionDMark.style.display = config.marks.d ? 'block' : 'none';

            // 更新按钮状态
            btnPrev.disabled = currentStep === 0;
            btnNext.disabled = currentStep === MAX_STEP;
        }

        // 下一步按钮点击
        btnNext.addEventListener('click', function() {
            if (currentStep < MAX_STEP) {
                currentStep++;
                updateDisplay();
            }
        });

        // 上一步按钮点击
        btnPrev.addEventListener('click', function() {
            if (currentStep > 0) {
                currentStep--;
                updateDisplay();
            }
        });

        // 初始化显示
        updateDisplay();
    })();
    </script>
