    <script>
    (function() {
        // 布局坐标表对照：
        // 集气瓶瓶身：(80,80)-(320,380) 240×300
        // 集气瓶瓶口：(160,60)-(240,80) 80×20
        // 橡皮塞：(150,55)-(250,75) 100×20
        // 燃烧匙杆：(200,75)-(200,150) 4×75
        // 燃烧匙勺：(185,150)-(215,170) 30×20
        // 红磷：(188,155)-(212,168) 24×13
        // 刻度线 1-4：y=320,260,200,140
        // 初始水位：y=350
        // 上升水位目标：y=320（刻度 1 处）
        // 导管水平段：(240,90)-(500,90)
        // 导管垂直段：(500,90)-(500,320)
        // 止水夹：(420,85)-(450,95)
        // 烧杯：(550,200)-(720,380)
        // 烧杯内水：(555,240)-(715,375)

        // 状态变量
        var currentStep = 0;
        var totalSteps = 7;

        // 获取 DOM 元素
        var elements = {
            prevBtn: document.getElementById('scene2-btn-prev'),
            nextBtn: document.getElementById('scene2-btn-next'),
            stepIndicator: document.getElementById('scene2-step-indicator'),
            flame: document.getElementById('scene2-flame'),
            redPhosphorus: document.getElementById('scene2-red-phosphorus'),
            smoke: document.getElementById('scene2-smoke'),
            smokeLabel: document.getElementById('scene2-label-smoke'),
            clip: document.getElementById('scene2-clip'),
            clipLabel: document.getElementById('scene2-label-clip'),
            pressureArrow: document.getElementById('scene2-pressure-arrow'),
            pressureLabel: document.getElementById('scene2-label-pressure'),
            waterFlow: document.getElementById('scene2-water-flow'),
            initialWater: document.getElementById('scene2-initial-water'),
            risingWater: document.getElementById('scene2-rising-water'),
            waterRising: document.getElementById('scene2-water-rising'),
            waterLineRising: document.getElementById('scene2-water-line-rising'),
            mark1Highlight: document.getElementById('scene2-mark-1-highlight'),
            conclusion: document.getElementById('scene2-conclusion'),
            phosphorusLabel: document.getElementById('scene2-label-phosphorus')
        };

        // 显示/隐藏工具函数
        function show(el) {
            if (el) {
                el.style.display = 'block';
                el.style.opacity = '1';
                el.classList.add('fade-in');
            }
        }

        function hide(el) {
            if (el) {
                el.style.display = 'none';
                el.style.opacity = '0';
                el.classList.remove('fade-in');
            }
        }

        // 更新显示状态
        function updateDisplay() {
            // 更新步骤指示器
            elements.stepIndicator.textContent = '步骤 ' + (currentStep + 1) + '/7';

            // 更新按钮状态
            elements.prevBtn.disabled = (currentStep === 0);
            elements.nextBtn.disabled = (currentStep === 6);

            // 步骤 0：初始状态
            if (currentStep === 0) {
                hide(elements.flame);
                show(elements.redPhosphorus);
                hide(elements.smoke);
                hide(elements.smokeLabel);
                elements.clip.setAttribute('fill', '#c62828');
                hide(elements.clipLabel);
                hide(elements.pressureArrow);
                hide(elements.pressureLabel);
                hide(elements.waterFlow);
                show(elements.initialWater);
                hide(elements.risingWater);
                hide(elements.mark1Highlight);
                hide(elements.conclusion);
                show(elements.phosphorusLabel);
            }
            // 步骤 1：点燃红磷
            else if (currentStep === 1) {
                show(elements.flame);
                hide(elements.redPhosphorus);
                hide(elements.smoke);
                hide(elements.smokeLabel);
                elements.clip.setAttribute('fill', '#c62828');
                hide(elements.clipLabel);
                hide(elements.pressureArrow);
                hide(elements.pressureLabel);
                hide(elements.waterFlow);
                show(elements.initialWater);
                hide(elements.risingWater);
                hide(elements.mark1Highlight);
                hide(elements.conclusion);
                show(elements.phosphorusLabel);
            }
            // 步骤 2：燃烧结束，白烟产生
            else if (currentStep === 2) {
                hide(elements.flame);
                hide(elements.redPhosphorus);
                show(elements.smoke);
                show(elements.smokeLabel);
                elements.clip.setAttribute('fill', '#c62828');
                hide(elements.clipLabel);
                hide(elements.pressureArrow);
                hide(elements.pressureLabel);
                hide(elements.waterFlow);
                show(elements.initialWater);
                hide(elements.risingWater);
                hide(elements.mark1Highlight);
                hide(elements.conclusion);
                hide(elements.phosphorusLabel);
            }
            // 步骤 3：冷却至室温
            else if (currentStep === 3) {
                hide(elements.flame);
                hide(elements.redPhosphorus);
                show(elements.smoke);
                show(elements.smokeLabel);
                elements.clip.setAttribute('fill', '#c62828');
                hide(elements.clipLabel);
                show(elements.pressureArrow);
                show(elements.pressureLabel);
                hide(elements.waterFlow);
                show(elements.initialWater);
                hide(elements.risingWater);
                hide(elements.mark1Highlight);
                hide(elements.conclusion);
                hide(elements.phosphorusLabel);
            }
            // 步骤 4：打开止水夹
            else if (currentStep === 4) {
                hide(elements.flame);
                hide(elements.redPhosphorus);
                show(elements.smoke);
                show(elements.smokeLabel);
                elements.clip.setAttribute('fill', '#2e7d32');
                show(elements.clipLabel);
                elements.clipLabel.textContent = '止水夹（开）';
                show(elements.pressureArrow);
                show(elements.pressureLabel);
                hide(elements.waterFlow);
                show(elements.initialWater);
                hide(elements.risingWater);
                hide(elements.mark1Highlight);
                hide(elements.conclusion);
                hide(elements.phosphorusLabel);
            }
            // 步骤 5：水倒吸入集气瓶
            else if (currentStep === 5) {
                hide(elements.flame);
                hide(elements.redPhosphorus);
                show(elements.smoke);
                show(elements.smokeLabel);
                elements.clip.setAttribute('fill', '#2e7d32');
                show(elements.clipLabel);
                elements.clipLabel.textContent = '止水夹（开）';
                show(elements.pressureArrow);
                show(elements.pressureLabel);
                show(elements.waterFlow);
                hide(elements.initialWater);
                show(elements.risingWater);
                // 触发动画：水位上升到刻度 1（y=320）
                setTimeout(function() {
                    elements.waterRising.setAttribute('y', '320');
                    elements.waterRising.setAttribute('height', '60');
                    elements.waterLineRising.setAttribute('y1', '320');
                    elements.waterLineRising.setAttribute('y2', '320');
                }, 50);
                hide(elements.mark1Highlight);
                hide(elements.conclusion);
                hide(elements.phosphorusLabel);
            }
            // 步骤 6：结论
            else if (currentStep === 6) {
                hide(elements.flame);
                hide(elements.redPhosphorus);
                hide(elements.smoke);
                hide(elements.smokeLabel);
                elements.clip.setAttribute('fill', '#2e7d32');
                show(elements.clipLabel);
                elements.clipLabel.textContent = '止水夹（开）';
                hide(elements.pressureArrow);
                hide(elements.pressureLabel);
                show(elements.waterFlow);
                hide(elements.initialWater);
                show(elements.risingWater);
                elements.waterRising.setAttribute('y', '320');
                elements.waterRising.setAttribute('height', '60');
                elements.waterLineRising.setAttribute('y1', '320');
                elements.waterLineRising.setAttribute('y2', '320');
                show(elements.mark1Highlight);
                show(elements.conclusion);
                hide(elements.phosphorusLabel);
            }
        }

        // 事件绑定
        elements.prevBtn.addEventListener('click', function() {
            if (currentStep > 0) {
                currentStep--;
                updateDisplay();
            }
        });

        elements.nextBtn.addEventListener('click', function() {
            if (currentStep < totalSteps - 1) {
                currentStep++;
                updateDisplay();
            }
        });

        // 初始化显示
        updateDisplay();
    })();
    </script>
