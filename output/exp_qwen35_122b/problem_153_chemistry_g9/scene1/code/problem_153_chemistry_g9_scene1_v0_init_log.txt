    <script>
    (function() {
        // Scene 1: 气体除杂实验装置串联全景图
        // 布局坐标表（参考）：
        // | 元素 | 类型 | 位置/范围 | 备注 |
        // |------|------|-----------|------|
        // | 装置 A 瓶身 | 矩形 | (80,180)-(180,380) | 洗气瓶主体 |
        // | 装置 B 玻璃管 | 矩形 | (220,140)-(380,180) | 硬质玻璃管 |
        // | 装置 C 瓶身 | 矩形 | (420,180)-(520,380) | 洗气瓶主体 |
        // | 装置 D 瓶身 | 矩形 | (560,180)-(660,380) | 洗气瓶主体 |
        // | 入口标注 | text | (20,180) | 混合气体→ |
        // | 出口标注 | text | (700,180) | →N₂ |

        // 状态变量
        let currentStep = 0;
        const totalSteps = 5;

        // 获取 DOM 元素
        const btnPrev = document.getElementById('scene1-btn-prev');
        const btnNext = document.getElementById('scene1-btn-next');
        const stepIndicator = document.getElementById('scene1-step-indicator');

        // 装置元素集合
        const devices = {
            a: {
                body: document.getElementById('scene1-device-a-body'),
                removeLabel: document.getElementById('scene1-label-a-remove')
            },
            b: {
                body: document.getElementById('scene1-device-b-tube'),
                flame: document.getElementById('scene1-device-b-flame'),
                removeLabel: document.getElementById('scene1-label-b-remove')
            },
            c: {
                body: document.getElementById('scene1-device-c-body'),
                removeLabel: document.getElementById('scene1-label-c-remove')
            },
            d: {
                body: document.getElementById('scene1-device-d-body'),
                removeLabel: document.getElementById('scene1-label-d-remove')
            }
        };

        // 箭头元素集合
        const arrows = [
            { line: document.getElementById('scene1-arrow-0'), triangle: document.getElementById('scene1-triangle-0') },
            { line: document.getElementById('scene1-arrow-1'), triangle: document.getElementById('scene1-triangle-1') },
            { line: document.getElementById('scene1-arrow-2'), triangle: document.getElementById('scene1-triangle-2') },
            { line: document.getElementById('scene1-arrow-3'), triangle: document.getElementById('scene1-triangle-3') },
            { line: document.getElementById('scene1-arrow-4'), triangle: document.getElementById('scene1-triangle-4') }
        ];

        // 标签元素
        const labelInput = document.getElementById('scene1-label-input');
        const labelOutput = document.getElementById('scene1-label-output');
        const labelFinal = document.getElementById('scene1-label-final');

        // 更新步骤指示器
        function updateStepIndicator() {
            stepIndicator.textContent = `步骤 ${currentStep}/${totalSteps}`;
            btnPrev.disabled = currentStep === 0;
            btnNext.disabled = currentStep === totalSteps;
        }

        // 高亮箭头（蓝色）
        function highlightArrow(index) {
            if (arrows[index]) {
                arrows[index].line.setAttribute('stroke', '#1565c0');
                arrows[index].line.setAttribute('stroke-width', '5');
                arrows[index].triangle.setAttribute('fill', '#1565c0');
            }
        }

        // 重置箭头（灰色）
        function resetArrow(index) {
            if (arrows[index]) {
                arrows[index].line.setAttribute('stroke', '#888888');
                arrows[index].line.setAttribute('stroke-width', '4');
                arrows[index].triangle.setAttribute('fill', '#888888');
            }
        }

        // 显示元素
        function show(el) {
            if (el) {
                el.style.display = 'block';
                el.style.opacity = '1';
            }
        }

        // 隐藏元素
        function hide(el) {
            if (el) {
                el.style.display = 'none';
                el.style.opacity = '0';
            }
        }

        // 更新场景状态
        function updateScene() {
            // 重置所有高亮
            Object.values(devices).forEach(dev => {
                if (dev.body) {
                    dev.body.setAttribute('stroke', '#000000');
                    dev.body.setAttribute('stroke-width', '2');
                }
                if (dev.flame) {
                    dev.flame.setAttribute('opacity', '0.8');
                }
                if (dev.removeLabel) {
                    hide(dev.removeLabel);
                }
            });

            // 重置所有箭头
            for (let i = 0; i < arrows.length; i++) {
                resetArrow(i);
            }

            // 重置标签
            labelInput.setAttribute('class', 'label-gray');
            labelOutput.setAttribute('class', 'label-gray');
            labelOutput.setAttribute('stroke-width', '1');
            hide(labelFinal);

            // 根据当前步骤更新状态
            switch (currentStep) {
                case 0:
                    // 初始状态：显示全景，灰色箭头
                    break;

                case 1:
                    // 高亮装置 A，除去 HCl
                    devices.a.body.setAttribute('stroke', '#1565c0');
                    devices.a.body.setAttribute('stroke-width', '4');
                    show(devices.a.removeLabel);
                    // 高亮入口到 A 的箭头
                    highlightArrow(0);
                    labelInput.setAttribute('class', 'label-blue');
                    break;

                case 2:
                    // 高亮装置 B，除去 H₂、CO
                    devices.b.body.setAttribute('stroke', '#1565c0');
                    devices.b.body.setAttribute('stroke-width', '4');
                    devices.b.flame.setAttribute('opacity', '1');
                    show(devices.b.removeLabel);
                    // 高亮 A 到 B 的箭头
                    highlightArrow(1);
                    break;

                case 3:
                    // 高亮装置 C，除去 CO₂
                    devices.c.body.setAttribute('stroke', '#1565c0');
                    devices.c.body.setAttribute('stroke-width', '4');
                    show(devices.c.removeLabel);
                    // 高亮 B 到 C 的箭头
                    highlightArrow(2);
                    break;

                case 4:
                    // 高亮装置 D，除去 H₂O
                    devices.d.body.setAttribute('stroke', '#1565c0');
                    devices.d.body.setAttribute('stroke-width', '4');
                    show(devices.d.removeLabel);
                    // 高亮 C 到 D 的箭头
                    highlightArrow(3);
                    break;

                case 5:
                    // 最终状态：高亮出口 N₂
                    labelOutput.setAttribute('class', 'label-green');
                    labelOutput.setAttribute('stroke-width', '3');
                    show(labelFinal);
                    // 高亮出口箭头
                    highlightArrow(4);
                    break;
            }

            updateStepIndicator();
        }

        // 事件绑定
        btnPrev.addEventListener('click', function() {
            if (currentStep > 0) {
                currentStep--;
                updateScene();
            }
        });

        btnNext.addEventListener('click', function() {
            if (currentStep < totalSteps) {
                currentStep++;
                updateScene();
            }
        });

        // 初始化
        updateScene();
    })();
    </script>
