    <script>
    (function() {
        let currentStep = 0;
        function showSvg(el) { if (el) el.style.opacity = '1'; }
        function hideSvg(el) { if (el) el.style.opacity = '0'; }
        function updateView() {
            const elements = {
                curveOriginal: document.getElementById('scene2-curve-original'),
                labelOriginal: document.getElementById('scene2-label-original'),
                curveCorrect: document.getElementById('scene2-curve-correct'),
                labelCorrect: document.getElementById('scene2-label-correct'),
                dashedLine: document.getElementById('scene2-dashed-line'),
                equationTitle: document.getElementById('scene2-equation-title'),
                equation: document.getElementById('scene2-equation'),
                equationNote: document.getElementById('scene2-equation-note'),
                errorTitle: document.getElementById('scene2-error-title'),
                errorText1: document.getElementById('scene2-error-text1'),
                errorText2: document.getElementById('scene2-error-text2'),
                errorText3: document.getElementById('scene2-error-text3'),
                errorText4: document.getElementById('scene2-error-text4'),
                highlightHorizontal: document.getElementById('scene2-highlight-horizontal')
            };
            switch (currentStep) {
                case 0:
                    showSvg(elements.curveOriginal); showSvg(elements.labelOriginal);
                    hideSvg(elements.curveCorrect); hideSvg(elements.labelCorrect); hideSvg(elements.dashedLine);
                    hideSvg(elements.equationTitle); hideSvg(elements.equation); hideSvg(elements.equationNote);
                    hideSvg(elements.errorTitle); hideSvg(elements.errorText1); hideSvg(elements.errorText2);
                    hideSvg(elements.errorText3); hideSvg(elements.errorText4); hideSvg(elements.highlightHorizontal);
                    break;
                case 1:
                    hideSvg(elements.curveOriginal); hideSvg(elements.labelOriginal);
                    hideSvg(elements.curveCorrect); hideSvg(elements.labelCorrect); hideSvg(elements.dashedLine);
                    showSvg(elements.equationTitle); showSvg(elements.equation); showSvg(elements.equationNote);
                    hideSvg(elements.errorTitle); hideSvg(elements.errorText1); hideSvg(elements.errorText2);
                    hideSvg(elements.errorText3); hideSvg(elements.errorText4); hideSvg(elements.highlightHorizontal);
                    break;
                case 2:
                    hideSvg(elements.curveOriginal); hideSvg(elements.labelOriginal);
                    showSvg(elements.curveCorrect); showSvg(elements.labelCorrect); showSvg(elements.dashedLine);
                    hideSvg(elements.equationTitle); hideSvg(elements.equation); hideSvg(elements.equationNote);
                    hideSvg(elements.errorTitle); hideSvg(elements.errorText1); hideSvg(elements.errorText2);
                    hideSvg(elements.errorText3); hideSvg(elements.errorText4); hideSvg(elements.highlightHorizontal);
                    break;
                case 3:
                    hideSvg(elements.curveOriginal); hideSvg(elements.labelOriginal);
                    showSvg(elements.curveCorrect); showSvg(elements.labelCorrect); showSvg(elements.dashedLine);
                    hideSvg(elements.equationTitle); hideSvg(elements.equation); hideSvg(elements.equationNote);
                    showSvg(elements.errorTitle); showSvg(elements.errorText1); showSvg(elements.errorText2);
                    showSvg(elements.errorText3); showSvg(elements.errorText4); showSvg(elements.highlightHorizontal);
                    break;
            }
            for (let i = 0; i <= 3; i++) {
                const btn = document.getElementById('scene2-btn-' + i);
                if (btn) btn.classList.toggle('active', i === currentStep);
            }
        }
        function setupButtons() {
            for (let i = 0; i <= 3; i++) {
                const btn = document.getElementById('scene2-btn-' + i);
                if (btn) btn.addEventListener('click', function() { currentStep = i; updateView(); });
            }
        }
        function init() { setupButtons(); updateView(); }
        if (document.readyState === 'loading') document.addEventListener('DOMContentLoaded', init);
        else init();
    })();
    </script>
