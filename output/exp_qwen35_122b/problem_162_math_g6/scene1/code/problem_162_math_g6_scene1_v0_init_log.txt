    <script>
    (function() {
        // Scene 1 实现 - 图形周长对比
        // 布局坐标表：
        // | 元素 | 位置/范围 | 大小 |
        // |------|-----------|------|
        // | 图形 A 轮廓 | (80,150)-(220,270) | 140×120 |
        // | 图形 B 轮廓 | (320,150)-(460,270) | 140×120 |
        // | 图形 C 轮廓 | (560,150)-(700,270) | 140×120 |
        // | 标题 | (400,55) | 居中 |
        
        const currentStepMax = 4;
        let currentStep = 0;
        
        // DOM 元素引用
        const graphA = document.getElementById('scene1-graph-a');
        const graphB = document.getElementById('scene1-graph-b');
        const graphC = document.getElementById('scene1-graph-c');
        const perimeterA = document.getElementById('scene1-perimeter-a');
        const perimeterB = document.getElementById('scene1-perimeter-b');
        const perimeterC = document.getElementById('scene1-perimeter-c');
        const conclusion = document.getElementById('scene1-conclusion');
        const prevBtn = document.getElementById('scene1-prev-btn');
        const nextBtn = document.getElementById('scene1-next-btn');
        const stepIndicator = document.getElementById('scene1-step-indicator');
        
        // 辅助线元素
        const auxA_H = document.getElementById('scene1-aux-a-h');
        const auxA_V = document.getElementById('scene1-aux-a-v');
        const auxB_H = document.getElementById('scene1-aux-b-h');
        const auxB_V = document.getElementById('scene1-aux-b-v');
        const auxCExtra = document.getElementById('scene1-aux-c-extra');
        const auxCDepth1 = document.getElementById('scene1-aux-c-depth1');
        const auxCDepth2 = document.getElementById('scene1-aux-c-depth2');
        
        // 工具函数：显示元素
        function show(el) {
            if (el) { el.style.opacity = '1'; el.style.display = 'block'; }
        }
        
        // 工具函数：隐藏元素
        function hide(el) {
            if (el) { el.style.opacity = '0'; el.style.display = 'none'; }
        }
        
        // 工具函数：设置边框颜色
        function setBorderColor(graph, color) {
            if (graph) graph.style.stroke = color;
        }
        
        // 更新步骤显示
        function updateStepDisplay() {
            stepIndicator.textContent = '步骤 ' + (currentStep + 1) + '/5';
            prevBtn.disabled = (currentStep === 0);
            nextBtn.disabled = (currentStep === currentStepMax);
        }
        
        // 重置所有图形状态
        function resetAll() {
            // 重置边框颜色
            setBorderColor(graphA, '#000000');
            setBorderColor(graphB, '#000000');
            setBorderColor(graphC, '#000000');
            
            // 隐藏所有周长标签
            hide(perimeterA);
            hide(perimeterB);
            hide(perimeterC);
            
            // 隐藏所有辅助线
            hide(auxA_H); hide(auxA_V);
            hide(auxB_H); hide(auxB_V);
            hide(auxCExtra); hide(auxCDepth1); hide(auxCDepth2);
            
            // 隐藏结论
            hide(conclusion);
        }
        
        // 步骤 0：初始状态
        function renderStep0() {
            resetAll();
        }
        
        // 步骤 1：计算图形 A 周长
        function renderStep1() {
            resetAll();
            setBorderColor(graphA, '#1565c0');
            show(perimeterA);
            show(auxA_H);
            show(auxA_V);
        }
        
        // 步骤 2：计算图形 B 周长
        function renderStep2() {
            resetAll();
            setBorderColor(graphB, '#1565c0');
            show(perimeterB);
            show(auxB_H);
            show(auxB_V);
        }
        
        // 步骤 3：计算图形 C 周长
        function renderStep3() {
            resetAll();
            setBorderColor(graphC, '#1565c0');
            show(perimeterC);
            show(auxCExtra);
            show(auxCDepth1);
            show(auxCDepth2);
        }
        
        // 步骤 4：比较得出结论
        function renderStep4() {
            resetAll();
            setBorderColor(graphC, '#2e7d32');
            show(perimeterA);
            show(perimeterB);
            show(perimeterC);
            show(conclusion);
        }
        
        // 渲染当前步骤
        function renderStep() {
            switch(currentStep) {
                case 0: renderStep0(); break;
                case 1: renderStep1(); break;
                case 2: renderStep2(); break;
                case 3: renderStep3(); break;
                case 4: renderStep4(); break;
            }
            updateStepDisplay();
        }
        
        // 事件绑定
        prevBtn.addEventListener('click', function() {
            if (currentStep > 0) {
                currentStep--;
                requestAnimationFrame(renderStep);
            }
        });
        
        nextBtn.addEventListener('click', function() {
            if (currentStep < currentStepMax) {
                currentStep++;
                requestAnimationFrame(renderStep);
            }
        });
        
        // 初始化
        renderStep();
    })();
    </script>
