    <script>
    (function() {
        // Scene 1: 三视图解析与立体图形构建
        // 坐标表对照：
        // 正面视图框：rect (80,120)-(180,220) 100×100
        // 侧面视图框：rect (220,120)-(320,220) 100×100
        // 上面视图框：polygon (360,140)-(460,140)-(440,220)-(380,220) 梯形
        // 立体图形：四棱柱 (520,150)-(700,320)
        // 标签：正面 (130,245) 侧面 (270,245) 上面 (410,245) 立体图 (610,355) 结论 (610,390)
        // 提示文字：y=285（与视图标签间距 40px）
        // 连接线 2：终点从 (520,250) 改为 (520,220)，避免穿过"上面"标签
        // 连接线 3：起点从 (410,220) 改为 (410,140)，终点从 (520,280) 改为 (520,170)，从梯形上方出发到立体图顶部

        const TOTAL_STEPS = 6;
        let currentStep = 0;

        // 获取 DOM 元素
        const viewFront = document.getElementById('scene1-view-front');
        const viewSide = document.getElementById('scene1-view-side');
        const viewTop = document.getElementById('scene1-view-top');
        const solid = document.getElementById('scene1-solid');
        const solidFill = document.getElementById('scene1-solid-fill');
        const line1 = document.getElementById('scene1-line1');
        const line2 = document.getElementById('scene1-line2');
        const line3 = document.getElementById('scene1-line3');
        const hintText = document.getElementById('scene1-hint-text');
        const labelConclusion = document.getElementById('scene1-label-conclusion');
        const labelSolid = document.getElementById('scene1-label-solid');
        const btnPrev = document.getElementById('scene1-btn-prev');
        const btnNext = document.getElementById('scene1-btn-next');
        const stepDisplay = document.getElementById('scene1-step-display');

        // 工具函数：高亮视图
        function highlightView(viewElement, isHighlighted) {
            if (isHighlighted) {
                viewElement.setAttribute('stroke', '#1565c0');
                viewElement.setAttribute('stroke-width', '5');
            } else {
                viewElement.setAttribute('stroke', '#000000');
                viewElement.setAttribute('stroke-width', '3');
            }
        }

        // 工具函数：显示/隐藏
        function show(el) {
            el.style.opacity = '1';
            el.style.display = 'block';
        }

        function hide(el) {
            el.style.opacity = '0';
            // 对于 SVG 元素，保持 display 但用 opacity 控制
        }

        // 更新场景状态
        function updateScene(step) {
            // 重置所有视图高亮
            highlightView(viewFront, false);
            highlightView(viewSide, false);
            highlightView(viewTop, false);

            // 隐藏连接线
            hide(line1);
            hide(line2);
            hide(line3);

            // 隐藏立体图形
            hide(solid);
            hide(solidFill);
            hide(labelSolid);

            // 隐藏结论
            hide(labelConclusion);

            // 隐藏提示文字
            hide(hintText);

            // 根据步骤更新状态
            switch(step) {
                case 0:
                    // 初始状态：仅显示三个视图
                    hintText.textContent = '观察三个视图';
                    show(hintText);
                    break;

                case 1:
                    // 高亮上面视图
                    highlightView(viewTop, true);
                    hintText.textContent = '俯视图决定底面形状';
                    show(hintText);
                    break;

                case 2:
                    // 高亮正面视图
                    highlightView(viewFront, true);
                    hintText.textContent = '正视图显示正面轮廓';
                    show(hintText);
                    break;

                case 3:
                    // 高亮侧面视图
                    highlightView(viewSide, true);
                    hintText.textContent = '侧视图显示侧面轮廓';
                    show(hintText);
                    break;

                case 4:
                    // 三个视图同时高亮
                    highlightView(viewFront, true);
                    highlightView(viewSide, true);
                    highlightView(viewTop, true);
                    // 显示连接线
                    show(line1);
                    show(line2);
                    show(line3);
                    hintText.textContent = '综合推断：梯形底面的柱体';
                    show(hintText);
                    break;

                case 5:
                    // 三个视图保持高亮
                    highlightView(viewFront, true);
                    highlightView(viewSide, true);
                    highlightView(viewTop, true);
                    // 连接线保持显示
                    show(line1);
                    show(line2);
                    show(line3);
                    // 显示立体图形
                    show(solid);
                    show(solidFill);
                    show(labelSolid);
                    // 显示结论
                    show(labelConclusion);
                    // 隐藏提示文字
                    hide(hintText);
                    break;
            }

            // 更新步骤显示
            stepDisplay.textContent = '步骤 ' + (step + 1) + '/' + TOTAL_STEPS;

            // 更新按钮状态
            btnPrev.disabled = (step === 0);
            btnNext.disabled = (step === TOTAL_STEPS - 1);
        }

        // 按钮事件
        btnPrev.addEventListener('click', function() {
            if (currentStep > 0) {
                currentStep--;
                updateScene(currentStep);
            }
        });

        btnNext.addEventListener('click', function() {
            if (currentStep < TOTAL_STEPS - 1) {
                currentStep++;
                updateScene(currentStep);
            }
        });

        // 初始化
        updateScene(currentStep);
    })();
    </script>
