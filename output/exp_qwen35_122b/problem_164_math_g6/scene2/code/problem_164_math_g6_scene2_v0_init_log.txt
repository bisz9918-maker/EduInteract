    <script>
    (function() {
        // Scene 2 实现
        // 坐标表对照：
        // 标题 (320,45) | 选项① (120,100)-(200,190) | 选项② (360,100)-(440,190) | 选项③ (600,100)-(680,190)
        // 三视图区域 (250,240)-(450,400) | 正面 (270,260)-(320,360) | 侧面 (350,260)-(400,360) | 上面 (430,260)-(490,320)
        // 判断标记 (580,290) | 判断原因 (580,330)
        // 按钮 (180,450), (360,450), (540,450)
        
        const state = {
            currentOption: 1  // 当前选中选项 1/2/3
        };
        
        // 各选项的三视图数据
        const viewData = {
            1: {
                topPoints: "140,115 180,100 200,115 180,130 140,130",  // 六边形
                mark: "×",
                markColor: "#c62828",
                reason: "俯视图=六边形≠梯形",
                reasonColor: "#c62828"
            },
            2: {
                topPoints: "380,110 400,95 420,110 440,110 400,125",  // 五边形
                mark: "×",
                markColor: "#c62828",
                reason: "俯视图=五边形≠梯形",
                reasonColor: "#c62828"
            },
            3: {
                topPoints: "620,110 640,95 680,95 700,110",  // 梯形
                mark: "✓",
                markColor: "#2e7d32",
                reason: "俯视图=梯形✓ 正确",
                reasonColor: "#2e7d32"
            }
        };
        
        // 获取 DOM 元素
        const els = {
            opt1Btn: document.getElementById('scene2-opt1-btn'),
            opt2Btn: document.getElementById('scene2-opt2-btn'),
            opt3Btn: document.getElementById('scene2-opt3-btn'),
            opt1Shape: document.getElementById('scene2-opt1-shape'),
            opt2Shape: document.getElementById('scene2-opt2-shape'),
            opt3Shape: document.getElementById('scene2-opt3-shape'),
            topView: document.getElementById('scene2-top-view'),
            judgeMark: document.getElementById('scene2-judge-mark'),
            judgeReason: document.getElementById('scene2-judge-reason')
        };
        
        // 更新视图
        function updateView(option) {
            state.currentOption = option;
            
            // 更新按钮状态
            els.opt1Btn.classList.toggle('active', option === 1);
            els.opt2Btn.classList.toggle('active', option === 2);
            els.opt3Btn.classList.toggle('active', option === 3);
            
            // 更新立体图高亮边框
            els.opt1Shape.setAttribute('stroke-width', option === 1 ? '3' : '2');
            els.opt1Shape.setAttribute('stroke', option === 1 ? '#1565c0' : '#000');
            els.opt2Shape.setAttribute('stroke-width', option === 2 ? '3' : '2');
            els.opt2Shape.setAttribute('stroke', option === 2 ? '#1565c0' : '#000');
            els.opt3Shape.setAttribute('stroke-width', option === 3 ? '3' : '2');
            els.opt3Shape.setAttribute('stroke', option === 3 ? '#1565c0' : '#000');
            
            // 更新上面视图形状
            els.topView.setAttribute('points', viewData[option].topPoints);
            
            // 更新判断标记和原因（带淡入淡出动画）
            els.judgeMark.style.opacity = '0';
            els.judgeReason.style.opacity = '0';
            
            setTimeout(() => {
                els.judgeMark.textContent = viewData[option].mark;
                els.judgeMark.setAttribute('fill', viewData[option].markColor);
                els.judgeReason.textContent = viewData[option].reason;
                els.judgeReason.setAttribute('fill', viewData[option].reasonColor);
                els.judgeMark.style.opacity = '1';
                els.judgeReason.style.opacity = '1';
            }, 150);
        }
        
        // 绑定按钮事件
        els.opt1Btn.addEventListener('click', () => updateView(1));
        els.opt2Btn.addEventListener('click', () => updateView(2));
        els.opt3Btn.addEventListener('click', () => updateView(3));
        
        // 初始化显示选项①
        updateView(1);
    })();
    </script>
