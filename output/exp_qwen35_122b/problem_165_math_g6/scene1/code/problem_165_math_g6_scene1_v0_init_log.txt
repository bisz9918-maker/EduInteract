    <script>
    (function() {
        // Scene 1 实现
        // 布局坐标表：
        // | 元素 | 类型 | 位置/范围 | 大小 | 备注 |
        // |------|------|-----------|------|------|
        // | 左侧正方形 | rect | (150,200) | 100×100 | 2dm 地砖 |
        // | 右侧正方形 | rect | (550,200) | 150×150 | 3dm 地砖 |
        // | 左侧边长标签 | text | (130,250) | - | "2dm" |
        // | 右侧边长标签 | text | (530,275) | - | "3dm" |
        // | 左侧面积标签 | text | (200,320) | - | "4dm²" |
        // | 右侧面积标签 | text | (625,365) | - | "9dm²" |
        // | 总面积标签 | text | (400,80) | - | "总面积=3600dm²" |
        // | 块数信息标签 | text | (400,390) | - | "需要 900 块" |
        // | 反比例等式 | text | (400,420) | - | "9×400=4×900=3600" |
        
        const TOTAL_STEPS = 4;
        let currentStep = 0;
        
        // 获取 DOM 元素
        const elements = {
            leftSquare: document.getElementById('scene1-left-square'),
            rightSquare: document.getElementById('scene1-right-square'),
            labelTotalArea: document.getElementById('scene1-label-total-area'),
            labelArea2dm: document.getElementById('scene1-label-area-2dm'),
            labelArea3dm: document.getElementById('scene1-label-area-3dm'),
            labelBlocks: document.getElementById('scene1-label-blocks'),
            labelEquation: document.getElementById('scene1-label-equation'),
            btnPrev: document.getElementById('scene1-btn-prev'),
            btnNext: document.getElementById('scene1-btn-next')
        };
        
        // 显示/隐藏工具函数
        function show(el) {
            el.style.display = 'block';
            el.style.opacity = '1';
        }
        
        function hide(el) {
            el.style.display = 'none';
            el.style.opacity = '0';
        }
        
        // 边框闪烁高亮动画
        function flashBorder(element) {
            element.classList.remove('flash-animation');
            void element.offsetWidth; // 触发重绘
            element.classList.add('flash-animation');
        }
        
        // 更新显示状态
        function updateDisplay() {
            // 步骤 0：仅显示正方形和边长标签（默认显示）
            // 步骤 1：显示右侧面积标签、总面积标签
            // 步骤 2：显示左侧面积标签
            // 步骤 3：显示块数信息标签、反比例等式标签
            
            switch(currentStep) {
                case 0:
                    // 隐藏所有动态标签
                    hide(elements.labelTotalArea);
                    hide(elements.labelArea2dm);
                    hide(elements.labelArea3dm);
                    hide(elements.labelBlocks);
                    hide(elements.labelEquation);
                    // 按钮状态
                    elements.btnPrev.disabled = true;
                    elements.btnNext.disabled = false;
                    break;
                    
                case 1:
                    // 显示右侧面积标签、总面积标签
                    show(elements.labelArea3dm);
                    show(elements.labelTotalArea);
                    // 右侧正方形闪烁高亮
                    flashBorder(elements.rightSquare);
                    // 按钮状态
                    elements.btnPrev.disabled = false;
                    elements.btnNext.disabled = false;
                    break;
                    
                case 2:
                    // 显示左侧面积标签
                    show(elements.labelArea2dm);
                    // 左侧正方形闪烁高亮
                    flashBorder(elements.leftSquare);
                    // 按钮状态
                    elements.btnPrev.disabled = false;
                    elements.btnNext.disabled = false;
                    break;
                    
                case 3:
                    // 显示块数信息标签、反比例等式标签
                    show(elements.labelBlocks);
                    show(elements.labelEquation);
                    // 两个正方形同时闪烁高亮
                    flashBorder(elements.leftSquare);
                    flashBorder(elements.rightSquare);
                    // 按钮状态
                    elements.btnPrev.disabled = false;
                    elements.btnNext.disabled = true;
                    break;
            }
        }
        
        // 下一步按钮点击
        elements.btnNext.addEventListener('click', function() {
            if (currentStep < TOTAL_STEPS - 1) {
                currentStep++;
                updateDisplay();
            }
        });
        
        // 上一步按钮点击
        elements.btnPrev.addEventListener('click', function() {
            if (currentStep > 0) {
                currentStep--;
                updateDisplay();
            }
        });
        
        // 初始化显示
        updateDisplay();
    })();
    </script>
