    <script>
    (function() {
        // ===== Scene 1 实现 =====
        // 布局坐标表对照：
        // 大圆：圆心 (400,140)，半径 70
        // 左小圆：圆心 (330,140)，半径 30
        // 右小圆：圆心 (470,140)，半径 40
        // 水平直径线：(300,140)-(540,140)
        // d₁标注线：(300,210)-(360,210)，标签 (330,228)
        // d₂标注线：(360,230)-(540,230)，标签 (450,248)
        // D 标注线：(300,250)-(540,250)，标签 (420,268)
        // 公式区域：右侧垂直排列
        
        const currentStepMax = 5;
        let currentStep = 0;
        
        // 获取 DOM 元素
        const btnPrev = document.getElementById('scene1-btn-prev');
        const btnNext = document.getElementById('scene1-btn-next');
        
        // 元素分组
        const groupDiameterLabels = document.getElementById('scene1-group-diameter-labels');
        const groupFormulas = document.getElementById('scene1-group-formulas');
        const groupRelation = document.getElementById('scene1-group-relation');
        const groupDerivation = document.getElementById('scene1-group-derivation');
        const groupConclusion = document.getElementById('scene1-group-conclusion');
        
        // 高亮元素
        const lineD1 = document.getElementById('scene1-line-d1');
        const lineD2 = document.getElementById('scene1-line-d2');
        const lineD = document.getElementById('scene1-line-D');
        const textD1 = document.getElementById('scene1-text-d1');
        const textD2 = document.getElementById('scene1-text-d2');
        const textD = document.getElementById('scene1-text-D');
        const textC1 = document.getElementById('scene1-text-c1');
        const textC2 = document.getElementById('scene1-text-c2');
        const textC = document.getElementById('scene1-text-C');
        const textRelation = document.getElementById('scene1-text-relation');
        const circleLeft = document.getElementById('scene1-circle-left');
        const circleRight = document.getElementById('scene1-circle-right');
        const circleLarge = document.getElementById('scene1-circle-large');
        const lineDiameter = document.getElementById('scene1-line-diameter');
        
        // 工具函数：显示元素
        function show(el) {
            if (el) {
                el.style.display = 'block';
                el.style.opacity = '1';
                el.classList.remove('scene1-hidden');
                el.classList.add('scene1-visible');
            }
        }
        
        // 工具函数：隐藏元素
        function hide(el) {
            if (el) {
                el.style.display = 'none';
                el.style.opacity = '0';
                el.classList.remove('scene1-visible');
                el.classList.add('scene1-hidden');
            }
        }
        
        // 工具函数：清除高亮
        function clearHighlights() {
            // 清除圆的样式
            [circleLeft, circleRight, circleLarge].forEach(circle => {
                circle.style.stroke = '#000000';
                circle.style.strokeWidth = '2';
                circle.classList.remove('scene1-blink');
            });
            // 清除标注线样式
            [lineD1, lineD2, lineD].forEach(line => {
                line.style.stroke = '#888888';
                line.style.strokeWidth = '1';
            });
            // 清除文字高亮
            [textD1, textD2, textD, textC1, textC2, textC, textRelation].forEach(text => {
                text.style.fill = '#000000';
                text.style.fontWeight = 'normal';
            });
            // 清除直径线高亮
            lineDiameter.style.stroke = '#000000';
            lineDiameter.style.strokeWidth = '2';
        }
        
        // 更新步骤显示
        function updateStep() {
            // 隐藏所有组
            hide(groupDiameterLabels);
            hide(groupFormulas);
            hide(groupRelation);
            hide(groupDerivation);
            hide(groupConclusion);
            
            // 清除所有高亮
            clearHighlights();
            
            // 根据当前步骤显示对应内容
            if (currentStep >= 1) {
                show(groupDiameterLabels);
            }
            if (currentStep >= 2) {
                show(groupFormulas);
            }
            if (currentStep >= 3) {
                show(groupRelation);
            }
            if (currentStep >= 4) {
                show(groupDerivation);
            }
            if (currentStep >= 5) {
                show(groupConclusion);
            }
            
            // 步骤 1：高亮直径标注
            if (currentStep === 1) {
                lineD1.style.stroke = '#1565c0';
                lineD1.style.strokeWidth = '2';
                lineD2.style.stroke = '#1565c0';
                lineD2.style.strokeWidth = '2';
                lineD.style.stroke = '#1565c0';
                lineD.style.strokeWidth = '2';
                textD1.style.fill = '#1565c0';
                textD1.style.fontWeight = 'bold';
                textD2.style.fill = '#1565c0';
                textD2.style.fontWeight = 'bold';
                textD.style.fill = '#1565c0';
                textD.style.fontWeight = 'bold';
            }
            
            // 步骤 2：高亮公式
            if (currentStep === 2) {
                textC1.style.fill = '#1565c0';
                textC1.style.fontWeight = 'bold';
                textC2.style.fill = '#1565c0';
                textC2.style.fontWeight = 'bold';
                textC.style.fill = '#1565c0';
                textC.style.fontWeight = 'bold';
            }
            
            // 步骤 3：高亮直径关系
            if (currentStep === 3) {
                lineDiameter.style.stroke = '#1565c0';
                lineDiameter.style.strokeWidth = '3';
                lineD1.style.stroke = '#1565c0';
                lineD1.style.strokeWidth = '2';
                lineD2.style.stroke = '#1565c0';
                lineD2.style.strokeWidth = '2';
                lineD.style.stroke = '#1565c0';
                lineD.style.strokeWidth = '2';
                textRelation.style.fill = '#c62828';
                textRelation.style.fontWeight = 'bold';
            }
            
            // 步骤 4：高亮推导过程，圆闪烁
            if (currentStep === 4) {
                textRelation.style.fill = '#c62828';
                textRelation.style.fontWeight = 'bold';
                document.getElementById('scene1-text-derivation1').style.fill = '#c62828';
                document.getElementById('scene1-text-derivation1').style.fontWeight = 'bold';
                document.getElementById('scene1-text-derivation2').style.fill = '#c62828';
                document.getElementById('scene1-text-derivation2').style.fontWeight = 'bold';
                document.getElementById('scene1-text-derivation3').style.fill = '#c62828';
                document.getElementById('scene1-text-derivation3').style.fontWeight = 'bold';
                document.getElementById('scene1-text-derivation4').style.fill = '#c62828';
                document.getElementById('scene1-text-derivation4').style.fontWeight = 'bold';
                
                // 圆闪烁动画
                circleLeft.classList.add('scene1-blink');
                setTimeout(() => circleRight.classList.add('scene1-blink'), 250);
                setTimeout(() => circleLarge.classList.add('scene1-blink'), 500);
            }
            
            // 步骤 5：高亮结论
            if (currentStep === 5) {
                circleLeft.style.stroke = '#2e7d32';
                circleLeft.style.strokeWidth = '3';
                circleRight.style.stroke = '#2e7d32';
                circleRight.style.strokeWidth = '3';
                circleLarge.style.stroke = '#2e7d32';
                circleLarge.style.strokeWidth = '3';
            }
            
            // 更新按钮状态
            btnPrev.disabled = (currentStep === 0);
            btnNext.disabled = (currentStep === currentStepMax);
        }
        
        // 按钮事件
        btnPrev.addEventListener('click', () => {
            if (currentStep > 0) {
                currentStep--;
                updateStep();
            }
        });
        
        btnNext.addEventListener('click', () => {
            if (currentStep < currentStepMax) {
                currentStep++;
                updateStep();
            }
        });
        
        // 初始化
        updateStep();
    })();
    </script>
