    <script>
    (function() {
        // Scene 1: DNA 组成层次结构概念图
        // 坐标表对照：
        // 方框 A: (80,110)-(140,170) | 方框 B: (170,110)-(230,210) | 方框 C: (270,110)-(330,210)
        // 方框 D: (420,110)-(540,210) | 方框 E: (620,110)-(740,210)
        // 加号 1: (155,145) | 加号 2: (245,155)
        // 箭头 1: (330,160)-(420,160) | 箭头 2: (540,160)-(620,160)

        const currentStepMax = 5;
        let currentStep = 0;

        // 工具函数：显示元素
        function show(el) {
            if (el) {
                el.style.display = 'block';
                el.style.opacity = '1';
            }
        }

        // 工具函数：隐藏元素
        function hide(el) {
            if (el) {
                el.style.display = 'none';
                el.style.opacity = '0';
            }
        }

        // 工具函数：移除高亮类
        function removeHighlight(el) {
            if (el) {
                el.classList.remove('highlight-blue');
                el.classList.remove('highlight-green');
            }
        }

        // 工具函数：添加蓝色高亮
        function addHighlightBlue(el) {
            if (el) {
                el.classList.remove('highlight-green');
                el.classList.add('highlight-blue');
            }
        }

        // 工具函数：添加绿色高亮
        function addHighlightGreen(el) {
            if (el) {
                el.classList.remove('highlight-blue');
                el.classList.add('highlight-green');
            }
        }

        // 更新步骤状态
        function updateStep(step) {
            currentStep = step;

            // 重置所有高亮
            ['A', 'B', 'C', 'D', 'E'].forEach(letter => {
                removeHighlight(document.getElementById('scene1-box-' + letter));
            });

            // 隐藏所有标注
            ['1', '2', '3', '4A', '4B', '4C', '4D'].forEach(id => {
                hide(document.getElementById('scene1-label-' + id));
            });

            // 隐藏所有选项标记
            ['A', 'B', 'C', 'D'].forEach(letter => {
                hide(document.getElementById('scene1-option-' + letter + '-mark'));
            });

            // 恢复 D 框的括号文本
            const dParen = document.getElementById('scene1-text-d-paren');
            dParen.textContent = '( )';

            // 根据步骤设置状态
            switch(step) {
                case 0:
                    // 初始状态：什么都不做，保持基础显示
                    break;

                case 1:
                    // 高亮方框 E，显示标注 1
                    addHighlightBlue(document.getElementById('scene1-box-E'));
                    show(document.getElementById('scene1-label-1'));
                    break;

                case 2:
                    // 高亮方框 D，显示标注 2，D 框下方显示"脱氧核苷酸"
                    addHighlightBlue(document.getElementById('scene1-box-D'));
                    show(document.getElementById('scene1-label-2'));
                    dParen.textContent = '(脱氧核苷酸)';
                    break;

                case 3:
                    // 高亮方框 A、B、C，显示标注 3
                    addHighlightBlue(document.getElementById('scene1-box-A'));
                    addHighlightBlue(document.getElementById('scene1-box-B'));
                    addHighlightBlue(document.getElementById('scene1-box-C'));
                    show(document.getElementById('scene1-label-3'));
                    break;

                case 4:
                    // 显示标注 4A、4B、4C、4D，D 框下方显示"脱氧核苷酸（4 种）"
                    show(document.getElementById('scene1-label-4A'));
                    show(document.getElementById('scene1-label-4B'));
                    show(document.getElementById('scene1-label-4C'));
                    show(document.getElementById('scene1-label-4D'));
                    dParen.textContent = '脱氧核苷酸（4 种）';
                    break;

                case 5:
                    // 显示选项标记，高亮正确选项 B
                    const markA = document.getElementById('scene1-option-A-mark');
                    const markB = document.getElementById('scene1-option-B-mark');
                    const markC = document.getElementById('scene1-option-C-mark');
                    const markD = document.getElementById('scene1-option-D-mark');

                    markA.textContent = '✗';
                    markA.style.fill = '#c62828';
                    show(markA);

                    markB.textContent = '✓';
                    markB.style.fill = '#2e7d32';
                    show(markB);

                    markC.textContent = '✗';
                    markC.style.fill = '#c62828';
                    show(markC);

                    markD.textContent = '✗';
                    markD.style.fill = '#c62828';
                    show(markD);

                    // 高亮正确选项 B 对应的方框 D（绿色）
                    addHighlightGreen(document.getElementById('scene1-box-D'));
                    break;
            }

            // 更新按钮状态
            const prevBtn = document.getElementById('scene1-btn-prev');
            const nextBtn = document.getElementById('scene1-btn-next');

            prevBtn.disabled = (step === 0);

            if (step === currentStepMax) {
                nextBtn.textContent = '完成';
                nextBtn.disabled = true;
            } else {
                nextBtn.textContent = '下一步 ▶';
                nextBtn.disabled = false;
            }
        }

        // 按钮事件绑定
        document.getElementById('scene1-btn-prev').addEventListener('click', function() {
            if (currentStep > 0) {
                updateStep(currentStep - 1);
            }
        });

        document.getElementById('scene1-btn-next').addEventListener('click', function() {
            if (currentStep < currentStepMax) {
                updateStep(currentStep + 1);
            }
        });

        // 初始化
        updateStep(0);
    })();
    </script>
