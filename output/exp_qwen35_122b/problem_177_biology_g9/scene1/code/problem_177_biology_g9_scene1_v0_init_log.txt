    <script>
    (function() {
        const scene1 = {
            currentStep: 1,
            totalSteps: 4,

            elements: {
                // 图甲元素
                chromosomeGroup: document.getElementById('scene1-chromosome-group'),
                chromosome: document.getElementById('scene1-chromosome'),
                label1: document.getElementById('scene1-label-1'),
                nameChromosome: document.getElementById('scene1-name-chromosome'),
                dnaGroup: document.getElementById('scene1-dna-group'),
                dna: document.getElementById('scene1-dna'),
                label3: document.getElementById('scene1-label-3'),
                nameDna: document.getElementById('scene1-name-dna'),
                proteinGroup: document.getElementById('scene1-protein-group'),
                protein: document.getElementById('scene1-protein'),
                label2: document.getElementById('scene1-label-2'),
                nameProtein: document.getElementById('scene1-name-protein'),
                geneGroup: document.getElementById('scene1-gene-group'),
                gene: document.getElementById('scene1-gene'),
                label4: document.getElementById('scene1-label-4'),
                nameGene: document.getElementById('scene1-name-gene'),

                // 图乙元素
                circlesGroup: document.getElementById('scene1-circles-group'),
                circle1: document.getElementById('scene1-circle-1'),
                circle2: document.getElementById('scene1-circle-2'),
                circle3: document.getElementById('scene1-circle-3'),
                labelCircle1: document.getElementById('scene1-label-circle-1'),
                nameCircle1: document.getElementById('scene1-name-circle-1'),
                labelCircle2: document.getElementById('scene1-label-circle-2'),
                nameCircle2: document.getElementById('scene1-name-circle-2'),
                labelCircle3: document.getElementById('scene1-label-circle-3'),
                nameCircle3: document.getElementById('scene1-name-circle-3'),

                // 箭头
                arrowsGroup: document.getElementById('scene1-arrows-group'),
                arrow1: document.getElementById('scene1-arrow-1'),
                arrow3: document.getElementById('scene1-arrow-3'),
                arrow4: document.getElementById('scene1-arrow-4'),

                // 高亮框
                highlightGroup: document.getElementById('scene1-highlight-group'),
                highlightChromosome: document.getElementById('scene1-highlight-chromosome'),
                highlightText1: document.getElementById('scene1-highlight-text-1'),
                highlightDna: document.getElementById('scene1-highlight-dna'),
                highlightText2: document.getElementById('scene1-highlight-text-2'),

                // 选项
                optionsGroup: document.getElementById('scene1-options-group'),
                optionsBg: document.getElementById('scene1-options-bg'),
                optionA: document.getElementById('scene1-option-a'),
                optionB: document.getElementById('scene1-option-b'),
                optionC: document.getElementById('scene1-option-c'),
                optionD: document.getElementById('scene1-option-d'),
                optionBBorder: document.getElementById('scene1-option-b-border'),
                optionBCorrect: document.getElementById('scene1-option-b-correct'),
                proteinMark: document.getElementById('scene1-protein-mark'),
                proteinNote: document.getElementById('scene1-protein-note'),

                // 控件
                stepIndicator: document.getElementById('scene1-step-indicator'),
                btnPrev: document.getElementById('scene1-btn-prev'),
                btnNext: document.getElementById('scene1-btn-next')
            },

            show: function(el) {
                if (el) {
                    el.style.display = 'block';
                    el.style.opacity = '1';
                }
            },

            hide: function(el) {
                if (el) {
                    el.style.opacity = '0';
                    setTimeout(() => {
                        if (el) el.style.display = 'none';
                    }, 300);
                }
            },

            updateScene: function() {
                const el = this.elements;
                
                // 重置所有元素
                this.hide(el.chromosomeGroup);
                this.hide(el.dnaGroup);
                this.hide(el.proteinGroup);
                this.hide(el.geneGroup);
                this.hide(el.circlesGroup);
                this.hide(el.arrowsGroup);
                this.hide(el.highlightGroup);
                this.hide(el.optionsGroup);

                // 根据步骤显示对应元素
                switch(this.currentStep) {
                    case 1:
                        // 步骤 1：显示图甲各结构标注
                        this.show(el.chromosomeGroup);
                        this.show(el.dnaGroup);
                        this.show(el.proteinGroup);
                        this.show(el.geneGroup);
                        break;
                    case 2:
                        // 步骤 2：显示图乙同心圆和连接箭头
                        this.show(el.chromosomeGroup);
                        this.show(el.dnaGroup);
                        this.show(el.proteinGroup);
                        this.show(el.geneGroup);
                        this.show(el.circlesGroup);
                        this.show(el.arrowsGroup);
                        break;
                    case 3:
                        // 步骤 3：高亮包含关系
                        this.show(el.chromosomeGroup);
                        this.show(el.dnaGroup);
                        this.show(el.proteinGroup);
                        this.show(el.geneGroup);
                        this.show(el.circlesGroup);
                        this.show(el.arrowsGroup);
                        this.show(el.highlightGroup);
                        break;
                    case 4:
                        // 步骤 4：显示选项分析
                        this.show(el.chromosomeGroup);
                        this.show(el.dnaGroup);
                        this.show(el.proteinGroup);
                        this.show(el.geneGroup);
                        this.show(el.circlesGroup);
                        this.show(el.arrowsGroup);
                        this.show(el.highlightGroup);
                        this.show(el.optionsGroup);
                        break;
                }

                // 更新步骤指示器
                el.stepIndicator.textContent = `步骤 ${this.currentStep}/${this.totalSteps}`;
                
                // 更新按钮状态
                el.btnPrev.disabled = this.currentStep === 1;
                el.btnNext.disabled = this.currentStep === this.totalSteps;
            },

            next: function() {
                if (this.currentStep < this.totalSteps) {
                    this.currentStep++;
                    this.updateScene();
                }
            },

            prev: function() {
                if (this.currentStep > 1) {
                    this.currentStep--;
                    this.updateScene();
                }
            },

            init: function() {
                const el = this.elements;
                el.btnPrev.addEventListener('click', () => this.prev());
                el.btnNext.addEventListener('click', () => this.next());
                this.updateScene();
            }
        };

        scene1.init();
    })();
    </script>
