    <script>
    (function() {
        // Scene 1: 并联电路结构与电流路径分析
        // 布局坐标表（参考）：
        // | 元素 | 位置/范围 | 备注 |
        // | 电源正极 | (470,55) | 顶部中央偏右 |
        // | 电源负极 | (310,55) | 顶部中央偏左 |
        // | 开关 S | (580,110)-(650,160) | 右上角 |
        // | 分支点 O1 | (650,170) | 开关下方 |
        // | L₁灯泡 | (320,280) 半径 35 | 中下偏左 |
        // | L₂灯泡 | (550,280) 半径 35 | 中下偏右 |
        // | 汇合点 O2 | (435,350) | 两灯泡下方中央 |
        // | 电流表 A | (120,300) 半径 40 | 左侧 |

        const scene1 = {
            currentStep: 0,
            totalSteps: 5,

            // 获取元素
            getEl: function(id) {
                return document.getElementById('scene1-' + id);
            },

            // 更新步骤显示
            updateStep: function() {
                // 重置所有高亮和标签
                this.resetHighlights();

                // 根据当前步骤设置显示
                switch(this.currentStep) {
                    case 0:
                        // 初始状态：显示完整电路图，无高亮
                        this.updateIndicator('步骤 1/5');
                        break;

                    case 1:
                        // 识别连接方式：高亮两条支路
                        this.highlightWire('wire-junction1-lamp1', 'blue');
                        this.highlightWire('wire-lamp1-junction2', 'blue');
                        this.highlightWire('wire-junction1-lamp2', 'green');
                        this.highlightWire('wire-lamp2-junction2', 'green');
                        this.showLabel('label-junction1', true);
                        this.showLabel('label-junction2', true);
                        this.showArrows(true);
                        this.updateIndicator('步骤 2/5');
                        break;

                    case 2:
                        // 电流表位置：高亮电流表
                        this.highlightWire('wire-junction2-ammeter', 'blue');
                        this.highlightEl('ammeter-circle', 'highlight-blue');
                        this.showLabel('ammeter-info', true);
                        this.showLabel('ammeter-formula', true);
                        this.updateIndicator('步骤 3/5');
                        break;

                    case 3:
                        // 开关控制范围：高亮开关
                        this.highlightWire('wire-battery-switch', 'red');
                        this.highlightEl('switch-arm', 'highlight-red');
                        this.showLabel('switch-info', true);
                        this.showLabel('dry-circuit', true);
                        this.updateIndicator('步骤 4/5');
                        break;

                    case 4:
                        // 结论总结：显示所有高亮和结论框
                        this.highlightWire('wire-junction1-lamp1', 'blue');
                        this.highlightWire('wire-lamp1-junction2', 'blue');
                        this.highlightWire('wire-junction1-lamp2', 'green');
                        this.highlightWire('wire-lamp2-junction2', 'green');
                        this.highlightWire('wire-junction2-ammeter', 'blue');
                        this.highlightEl('ammeter-circle', 'highlight-blue');
                        this.highlightEl('switch-arm', 'highlight-red');
                        this.showLabel('junction1', true);
                        this.showLabel('junction2', true);
                        this.showLabel('ammeter-info', true);
                        this.showLabel('switch-info', true);
                        this.showArrows(true);
                        this.getEl('conclusion-box').classList.add('visible');
                        this.updateIndicator('步骤 5/5');
                        break;
                }

                // 更新按钮状态
                this.getEl('btn-prev').disabled = (this.currentStep === 0);
                this.getEl('btn-next').disabled = (this.currentStep === this.totalSteps - 1);
            },

            // 重置所有高亮
            resetHighlights: function() {
                // 重置导线高亮
                const wires = ['wire-junction1-lamp1', 'wire-lamp1-junction2', 
                              'wire-junction1-lamp2', 'wire-lamp2-junction2',
                              'wire-junction2-ammeter', 'wire-battery-switch', 'switch-arm'];
                wires.forEach(w => {
                    const el = this.getEl(w);
                    if (el) {
                        el.classList.remove('highlight-blue', 'highlight-green', 'highlight-red');
                    }
                });

                // 重置组件高亮
                const comps = ['ammeter-circle'];
                comps.forEach(c => {
                    const el = this.getEl(c);
                    if (el) {
                        el.classList.remove('highlight-blue', 'highlight-green', 'highlight-red');
                    }
                });

                // 隐藏所有信息标签
                const labels = ['junction1', 'junction2', 'ammeter-info', 'ammeter-formula', 
                               'switch-info', 'dry-circuit'];
                labels.forEach(l => this.showLabel(l, false));

                // 隐藏电流箭头
                this.showArrows(false);

                // 隐藏结论框
                this.getEl('conclusion-box').classList.remove('visible');
            },

            // 高亮导线
            highlightWire: function(wireId, color) {
                const el = this.getEl(wireId);
                if (el) {
                    el.classList.add('wire', 'highlight-' + color);
                }
            },

            // 高亮组件
            highlightEl: function(elId, className) {
                const el = this.getEl(elId);
                if (el) {
                    el.classList.add(className);
                }
            },

            // 显示/隐藏标签
            showLabel: function(labelId, show) {
                const el = this.getEl('label-' + labelId);
                if (el) {
                    el.classList.toggle('visible', show);
                }
            },

            // 显示/隐藏电流箭头
            showArrows: function(show) {
                const arrows = ['arrow-battery', 'arrow-switch', 'arrow-split1', 
                               'arrow-split2', 'arrow-merge1', 'arrow-merge2', 'arrow-ammeter'];
                arrows.forEach(a => {
                    const el = this.getEl(a);
                    if (el) {
                        el.style.opacity = show ? '1' : '0';
                    }
                });
            },

            // 更新步骤指示器
            updateIndicator: function(text) {
                this.getEl('step-indicator').textContent = text;
            },

            // 下一步
            nextStep: function() {
                if (this.currentStep < this.totalSteps - 1) {
                    this.currentStep++;
                    this.updateStep();
                }
            },

            // 上一步
            prevStep: function() {
                if (this.currentStep > 0) {
                    this.currentStep--;
                    this.updateStep();
                }
            },

            // 初始化
            init: function() {
                this.getEl('btn-next').addEventListener('click', () => this.nextStep());
                this.getEl('btn-prev').addEventListener('click', () => this.prevStep());
                this.updateStep();
            }
        };

        // 启动
        scene1.init();
    })();
    </script>
