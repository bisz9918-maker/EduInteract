    <script>
    (function() {
        // Scene 1 实现 - 地球光照图基础结构认知
        // 坐标表对照：
        // 地球圆心 (400,250)，半径 180px
        // A 点 (400,250)，B 点 (340,150)，C 点 (580,250)
        // N (395,35)，S (395,450)，73°41' (420,35)
        // A 标签 (415,285)，B 标签 (290,125)，C 标签 (595,265)
        // 昼半球标签 (660,195)，夜半球标签 (40,315)
        // 晨昏线标签 (270,65)，赤道标签 (660,265)
        
        const sceneId = 'scene1';
        let currentStep = 0;
        const totalSteps = 5;
        
        // 步骤元素配置：每个步骤应显示的元素 ID 列表
        const steps = [
            // 步骤 0：基础地球网格
            ['scene1-earth', 'scene1-equator', 'scene1-central-meridian', 
             'scene1-meridian-left1', 'scene1-meridian-left2', 
             'scene1-meridian-right1', 'scene1-meridian-right2',
             'scene1-latitude-north', 'scene1-latitude-tropic-north',
             'scene1-latitude-tropic-south', 'scene1-latitude-south'],
            // 步骤 1：昼夜半球 + 阳光
            ['scene1-night-hemisphere', 'scene1-day-hemisphere', 'scene1-sunlight',
             'scene1-label-day', 'scene1-label-night'],
            // 步骤 2：晨昏线
            ['scene1-terminator', 'scene1-label-terminator'],
            // 步骤 3：A/B/C 三点
            ['scene1-point-a', 'scene1-point-b', 'scene1-point-c',
             'scene1-label-a', 'scene1-label-b', 'scene1-label-c'],
            // 步骤 4：极点及纬线标注
            ['scene1-label-n', 'scene1-label-s', 'scene1-label-7341', 'scene1-label-equator']
        ];
        
        // 工具函数：显示元素
        function show(elId) {
            const el = document.getElementById(elId);
            if (el) {
                el.style.display = 'block';
                el.style.opacity = '1';
            }
        }
        
        // 工具函数：隐藏元素
        function hide(elId) {
            const el = document.getElementById(elId);
            if (el) {
                el.style.display = 'none';
                el.style.opacity = '0';
            }
        }
        
        // 更新显示状态
        function updateDisplay() {
            // 先隐藏所有步骤元素
            for (let i = 0; i < totalSteps; i++) {
                steps[i].forEach(id => hide(id));
            }
            
            // 显示当前步骤及之前的所有元素
            for (let i = 0; i <= currentStep; i++) {
                steps[i].forEach(id => show(id));
            }
            
            // 更新按钮状态
            const prevBtn = document.getElementById(sceneId + '-prev-btn');
            const nextBtn = document.getElementById(sceneId + '-next-btn');
            prevBtn.disabled = currentStep === 0;
            nextBtn.disabled = currentStep === totalSteps - 1;
            nextBtn.textContent = currentStep === totalSteps - 1 ? '完成' : '下一步 ▶';
        }
        
        // 事件绑定
        document.getElementById(sceneId + '-prev-btn').addEventListener('click', function() {
            if (currentStep > 0) {
                currentStep--;
                updateDisplay();
            }
        });
        
        document.getElementById(sceneId + '-next-btn').addEventListener('click', function() {
            if (currentStep < totalSteps - 1) {
                currentStep++;
                updateDisplay();
            }
        });
        
        // 初始化
        updateDisplay();
    })();
    </script>
