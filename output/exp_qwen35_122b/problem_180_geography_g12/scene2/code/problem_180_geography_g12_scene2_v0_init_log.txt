    <script>
    (function() {
        // Scene 2 实现：太阳高度与昼夜长短分析
        // 布局坐标表对照：
        // 地球中心 (400,250)，半径 180
        // A 点 (320,250) - 赤道与晨昏线交点，标签 (380,265) - 地球外右侧
        // B 点 (340,180) - 北半球晨昏线附近，标签 (200,175) - 地球外左侧
        // C 点 (580,250) - 赤道最右侧，标签 (595,255)
        // N 点 (390,50), S 点 (390,455)
        // 73°41'标注 (450,45) - 地球外上方
        // A 太阳高度标注 (180,275) - 地球左侧外
        // B 昼长标注 (180,140) - 地球左侧外，与 B 标签错开
        // C 太阳高度标注 (620,220) - 地球右侧外
        // 南半球标注 (620,350) - 地球右侧外
        
        const scene2 = {
            currentStep: 0,
            totalSteps: 5,
            
            // 步骤配置
            stepConfig: [
                // 步骤 0：初始状态
                {
                    showHighlightA: false,
                    showHighlightC: false,
                    showLabelAHeight: false,
                    showLabelCHeight: false,
                    showLatitudeB: false,
                    showBDayArc: false,
                    showBNightArc: false,
                    showLabelBDaylength: false,
                    showSouthDayArc: false,
                    showSouthNightArc: false,
                    showLabelSouth: false
                },
                // 步骤 1：A 点高亮
                {
                    showHighlightA: true,
                    showHighlightC: false,
                    showLabelAHeight: true,
                    showLabelCHeight: false,
                    showLatitudeB: false,
                    showBDayArc: false,
                    showBNightArc: false,
                    showLabelBDaylength: false,
                    showSouthDayArc: false,
                    showSouthNightArc: false,
                    showLabelSouth: false
                },
                // 步骤 2：C 点高亮
                {
                    showHighlightA: true,
                    showHighlightC: true,
                    showLabelAHeight: true,
                    showLabelCHeight: true,
                    showLatitudeB: false,
                    showBDayArc: false,
                    showBNightArc: false,
                    showLabelBDaylength: false,
                    showSouthDayArc: false,
                    showSouthNightArc: false,
                    showLabelSouth: false
                },
                // 步骤 3：B 点昼长
                {
                    showHighlightA: true,
                    showHighlightC: true,
                    showLabelAHeight: true,
                    showLabelCHeight: true,
                    showLatitudeB: true,
                    showBDayArc: true,
                    showBNightArc: true,
                    showLabelBDaylength: true,
                    showSouthDayArc: false,
                    showSouthNightArc: false,
                    showLabelSouth: false
                },
                // 步骤 4：南半球对比
                {
                    showHighlightA: true,
                    showHighlightC: true,
                    showLabelAHeight: true,
                    showLabelCHeight: true,
                    showLatitudeB: true,
                    showBDayArc: true,
                    showBNightArc: true,
                    showLabelBDaylength: true,
                    showSouthDayArc: true,
                    showSouthNightArc: true,
                    showLabelSouth: true
                },
                // 步骤 5：结论汇总
                {
                    showHighlightA: true,
                    showHighlightC: true,
                    showLabelAHeight: true,
                    showLabelCHeight: true,
                    showLatitudeB: true,
                    showBDayArc: true,
                    showBNightArc: true,
                    showLabelBDaylength: true,
                    showSouthDayArc: true,
                    showSouthNightArc: true,
                    showLabelSouth: true
                }
            ],
            
            // 初始化
            init: function() {
                this.bindEvents();
                this.renderStep();
            },
            
            // 绑定事件
            bindEvents: function() {
                const btnPrev = document.getElementById('scene2-btn-prev');
                const btnNext = document.getElementById('scene2-btn-next');
                
                btnPrev.addEventListener('click', () => {
                    if (this.currentStep > 0) {
                        this.currentStep--;
                        this.renderStep();
                    }
                });
                
                btnNext.addEventListener('click', () => {
                    if (this.currentStep < this.totalSteps) {
                        this.currentStep++;
                        this.renderStep();
                    }
                });
            },
            
            // 渲染当前步骤
            renderStep: function() {
                const config = this.stepConfig[this.currentStep];
                
                // 更新步骤指示器
                document.getElementById('scene2-step-indicator').textContent = '步骤 ' + this.currentStep + '/' + this.totalSteps;
                
                // 更新按钮状态
                document.getElementById('scene2-btn-prev').disabled = (this.currentStep === 0);
                document.getElementById('scene2-btn-next').disabled = (this.currentStep === this.totalSteps);
                
                // 根据配置显示/隐藏元素
                this.toggleElement('scene2-highlight-a', config.showHighlightA);
                this.toggleElement('scene2-highlight-c', config.showHighlightC);
                this.toggleElement('scene2-label-a-height', config.showLabelAHeight);
                this.toggleElement('scene2-label-c-height', config.showLabelCHeight);
                this.toggleElement('scene2-latitude-b', config.showLatitudeB);
                this.toggleElement('scene2-b-day-arc', config.showBDayArc);
                this.toggleElement('scene2-b-night-arc', config.showBNightArc);
                this.toggleElement('scene2-label-b-daylength', config.showLabelBDaylength);
                this.toggleElement('scene2-south-day-arc', config.showSouthDayArc);
                this.toggleElement('scene2-south-night-arc', config.showSouthNightArc);
                this.toggleElement('scene2-label-south', config.showLabelSouth);
            },
            
            // 切换元素显示状态
            toggleElement: function(id, show) {
                const el = document.getElementById(id);
                if (show) {
                    el.style.display = 'block';
                    el.style.opacity = '1';
                } else {
                    el.style.display = 'none';
                    el.style.opacity = '0';
                }
            }
        };
        
        // 初始化场景
        scene2.init();
    })();
    </script>
