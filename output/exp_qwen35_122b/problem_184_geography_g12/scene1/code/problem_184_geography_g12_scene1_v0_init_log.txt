    <script>
    (function() {
        // Scene 1: 地球公转轨道与两分两至位置判断
        // 布局坐标表：
        // | 元素 | 类型 | 位置/范围 | 大小 | 标签偏移方向 | 备注 |
        // |------|------|-----------|------|-------------|------|
        // | 椭圆轨道 | path | 中心 (400,250)，长轴 500，短轴 280 | 500×280 | - | 黑色实线，线宽 2 |
        // | 太阳 | circle | (550,250) | r=30 | 下方 (550,330) | 黄色填充，标注"太阳" |
        // | 地球 A | circle | (400,110) | r=20 | 上方 (400,45) | 蓝色填充 20% 透明度，标注"A" |
        // | 地球 B | circle | (150,250) | r=20 | 左方 (115,255) | 蓝色填充 20% 透明度，标注"B" |
        // | 地球 C | circle | (400,390) | r=20 | 下方 (400,455) | 蓝色填充 20% 透明度，标注"C" |
        // | 地球 D | circle | (650,250) | r=20 | 右方 (685,255) | 蓝色填充 20% 透明度，标注"D" |
        // | 地轴 A | line | (385,55)-(415,165) | 30×110 | - | 蓝色，穿过地球 A，倾斜 |
        // | 地轴 B | line | (135,195)-(165,305) | 30×110 | - | 蓝色，穿过地球 B，倾斜 |
        // | 地轴 C | line | (385,335)-(415,445) | 30×110 | - | 蓝色，穿过地球 C，倾斜 |
        // | 地轴 D | line | (635,195)-(665,305) | 30×110 | - | 蓝色，穿过地球 D，倾斜 |
        
        const currentStepMax = 4;
        let currentStep = 0;
        
        const prevBtn = document.getElementById('scene1-prev-btn');
        const nextBtn = document.getElementById('scene1-next-btn');
        const stepIndicator = document.getElementById('scene1-step-indicator');
        const conclusion = document.getElementById('scene1-conclusion');
        
        // 元素 ID 映射
        const elements = {
            // 步骤 1：地轴
            axisA: ['scene1-axis-a', 'scene1-axis-b', 'scene1-axis-c', 'scene1-axis-d', 'scene1-north-pole-a', 'scene1-south-pole-a'],
            // 步骤 2：太阳光线和直射点
            ray: ['scene1-ray-b', 'scene1-ray-d', 'scene1-subpoint-b', 'scene1-subpoint-d'],
            // 步骤 3：节气标签
            solarTerm: ['scene1-solar-term-a', 'scene1-solar-term-b', 'scene1-solar-term-c', 'scene1-solar-term-d'],
            // 步骤 4：高亮
            highlight: ['scene1-highlight-b', 'scene1-highlight-d']
        };
        
        function showElement(id) {
            const el = document.getElementById(id);
            if (el) {
                el.style.opacity = '1';
                el.style.display = 'block';
            }
        }
        
        function hideElement(id) {
            const el = document.getElementById(id);
            if (el) {
                el.style.opacity = '0';
                if (el.tagName === 'line' || el.tagName === 'circle' || el.tagName === 'path') {
                    el.style.display = 'block'; // 保持 SVG 元素显示，只改 opacity
                } else {
                    el.style.display = 'none';
                }
            }
        }
        
        function updateStep() {
            // 更新按钮状态
            prevBtn.disabled = (currentStep === 0);
            if (currentStep === currentStepMax) {
                nextBtn.textContent = '完成';
                nextBtn.disabled = true;
            } else {
                nextBtn.textContent = '下一步 ▶';
                nextBtn.disabled = false;
            }
            
            // 更新步骤指示器
            stepIndicator.textContent = `步骤 ${currentStep + 1}/${currentStepMax + 1}`;
            
            // 根据步骤显示/隐藏元素
            // 步骤 0：基础结构（已默认显示）
            // 步骤 1：显示地轴
            if (currentStep >= 1) {
                elements.axisA.forEach(showElement);
            } else {
                elements.axisA.forEach(hideElement);
            }
            
            // 步骤 2：显示太阳光线和直射点
            if (currentStep >= 2) {
                elements.ray.forEach(showElement);
            } else {
                elements.ray.forEach(hideElement);
            }
            
            // 步骤 3：显示节气标签
            if (currentStep >= 3) {
                elements.solarTerm.forEach(showElement);
            } else {
                elements.solarTerm.forEach(hideElement);
            }
            
            // 步骤 4：显示高亮和结论
            if (currentStep >= 4) {
                elements.highlight.forEach(showElement);
                conclusion.style.opacity = '1';
            } else {
                elements.highlight.forEach(hideElement);
                conclusion.style.opacity = '0';
            }
        }
        
        prevBtn.addEventListener('click', () => {
            if (currentStep > 0) {
                currentStep--;
                updateStep();
            }
        });
        
        nextBtn.addEventListener('click', () => {
            if (currentStep < currentStepMax) {
                currentStep++;
                updateStep();
            }
        });
        
        // 初始化
        updateStep();
    })();
    </script>
