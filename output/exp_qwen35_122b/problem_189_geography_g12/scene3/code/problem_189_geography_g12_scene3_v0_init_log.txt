    <script>
    (function() {
        // Scene 3: 东半球与地方时计算
        // 坐标表对照：
        // 地球圆：(150,250) r=60
        // 经线 0°: (150,190)-(150,310)
        // 经线 180°: (90,250)-(210,250)
        // 20°W 边界：弧线 (130,210)-(130,290)
        // 160°E 边界：弧线 (170,210)-(170,290)
        // 信息框：(320,55)-(770,380)
        
        const currentStepMax = 5;
        let currentStep = 0;
        
        // DOM 元素引用
        const els = {
            eastHemisphere: document.getElementById('scene3-east-hemisphere'),
            boundary20w: document.getElementById('scene3-boundary-20w'),
            boundary160e: document.getElementById('scene3-boundary-160e'),
            terminatorMorning: document.getElementById('scene3-terminator-morning'),
            terminatorEvening: document.getElementById('scene3-terminator-evening'),
            labelMorning: document.getElementById('scene3-label-morning'),
            labelEvening: document.getElementById('scene3-label-evening'),
            labelEastHemisphere: document.getElementById('scene3-label-east-hemisphere'),
            sunPoint: document.getElementById('scene3-sun-point'),
            labelSunPoint: document.getElementById('scene3-label-sun-point'),
            labelSun110w: document.getElementById('scene3-label-sun-110w'),
            labelSunTime: document.getElementById('scene3-label-sun-time'),
            point120e: document.getElementById('scene3-120e-point'),
            label120e: document.getElementById('scene3-label-120e'),
            infoContent: document.getElementById('scene3-info-content'),
            calc1: document.getElementById('scene3-calc-1'),
            calc2: document.getElementById('scene3-calc-2'),
            calc3: document.getElementById('scene3-calc-3'),
            calc4: document.getElementById('scene3-calc-4'),
            calc5: document.getElementById('scene3-calc-5'),
            summary1: document.getElementById('scene3-summary-1'),
            summary2: document.getElementById('scene3-summary-2'),
            summary3: document.getElementById('scene3-summary-3'),
            pointD: document.getElementById('scene3-point-d'),
            labelD: document.getElementById('scene3-label-d'),
            btnPrev: document.getElementById('scene3-btn-prev'),
            btnNext: document.getElementById('scene3-btn-next'),
            btnSummary: document.getElementById('scene3-btn-summary')
        };
        
        // 工具函数：显示元素
        function show(el) {
            if (el) {
                el.style.display = 'block';
                el.style.opacity = '1';
            }
        }
        
        // 工具函数：隐藏元素
        function hide(el) {
            if (el) {
                el.style.opacity = '0';
                setTimeout(() => {
                    if (el.style.opacity === '0') {
                        el.style.display = 'none';
                    }
                }, 300);
            }
        }
        
        // 更新显示状态
        function updateDisplay() {
            // 重置所有动态元素
            hide(els.eastHemisphere);
            hide(els.terminatorMorning);
            hide(els.terminatorEvening);
            hide(els.labelMorning);
            hide(els.labelEvening);
            hide(els.labelEastHemisphere);
            hide(els.sunPoint);
            hide(els.labelSunPoint);
            hide(els.labelSun110w);
            hide(els.labelSunTime);
            hide(els.point120e);
            hide(els.label120e);
            hide(els.calc1);
            hide(els.calc2);
            hide(els.calc3);
            hide(els.calc4);
            hide(els.calc5);
            hide(els.summary1);
            hide(els.summary2);
            hide(els.summary3);
            hide(els.pointD);
            hide(els.labelD);
            
            // 重置边界线宽
            els.boundary20w.style.strokeWidth = '2';
            els.boundary160e.style.strokeWidth = '2';
            
            // 根据步骤显示对应元素
            switch(currentStep) {
                case 0:
                    els.infoContent.textContent = '点击"下一步"查看东半球范围详解';
                    els.btnPrev.disabled = true;
                    els.btnNext.disabled = false;
                    els.btnSummary.disabled = true;
                    break;
                    
                case 1:
                    show(els.eastHemisphere);
                    show(els.labelEastHemisphere);
                    els.boundary20w.style.strokeWidth = '3';
                    els.boundary160e.style.strokeWidth = '3';
                    els.infoContent.textContent = '东半球范围：20°W → 0° → 160°E，共 180°经度跨度';
                    els.btnPrev.disabled = false;
                    els.btnNext.disabled = false;
                    els.btnSummary.disabled = true;
                    break;
                    
                case 2:
                    show(els.eastHemisphere);
                    els.eastHemisphere.style.fill = '#0d47a1'; // 深蓝色表示黑夜
                    show(els.labelEastHemisphere);
                    show(els.terminatorMorning);
                    show(els.terminatorEvening);
                    show(els.labelMorning);
                    show(els.labelEvening);
                    els.boundary20w.style.strokeWidth = '3';
                    els.boundary160e.style.strokeWidth = '3';
                    els.infoContent.textContent = '东半球全部黑夜时：20°W 为晨线 (6:00)，160°E 为昏线 (18:00)';
                    els.btnPrev.disabled = false;
                    els.btnNext.disabled = false;
                    els.btnSummary.disabled = false;
                    break;
                    
                case 3:
                    show(els.eastHemisphere);
                    els.eastHemisphere.style.fill = '#0d47a1';
                    show(els.labelEastHemisphere);
                    show(els.terminatorMorning);
                    show(els.terminatorEvening);
                    show(els.labelMorning);
                    show(els.labelEvening);
                    show(els.sunPoint);
                    show(els.labelSunPoint);
                    show(els.labelSun110w);
                    show(els.labelSunTime);
                    els.sunPoint.setAttribute('cx', '50');
                    els.sunPoint.setAttribute('cy', '250');
                    els.boundary20w.style.strokeWidth = '3';
                    els.boundary160e.style.strokeWidth = '3';
                    els.infoContent.textContent = '太阳直射点位于 110°W，地方时为 12:00';
                    els.btnPrev.disabled = false;
                    els.btnNext.disabled = false;
                    els.btnSummary.disabled = false;
                    break;
                    
                case 4:
                    show(els.eastHemisphere);
                    els.eastHemisphere.style.fill = '#0d47a1';
                    show(els.labelEastHemisphere);
                    show(els.terminatorMorning);
                    show(els.terminatorEvening);
                    show(els.labelMorning);
                    show(els.labelEvening);
                    show(els.sunPoint);
                    show(els.labelSunPoint);
                    show(els.labelSun110w);
                    show(els.labelSunTime);
                    els.sunPoint.setAttribute('cx', '50');
                    els.sunPoint.setAttribute('cy', '250');
                    show(els.point120e);
                    show(els.label120e);
                    show(els.calc1);
                    show(els.calc2);
                    show(els.calc3);
                    show(els.calc4);
                    show(els.calc5);
                    els.boundary20w.style.strokeWidth = '3';
                    els.boundary160e.style.strokeWidth = '3';
                    els.infoContent.textContent = '计算北京时间：110°W=12:00 → 120°E=次日 3:20';
                    els.btnPrev.disabled = false;
                    els.btnNext.disabled = false;
                    els.btnSummary.disabled = false;
                    break;
                    
                case 5:
                    show(els.eastHemisphere);
                    els.eastHemisphere.style.fill = '#0d47a1';
                    show(els.labelEastHemisphere);
                    show(els.terminatorMorning);
                    show(els.terminatorEvening);
                    show(els.labelMorning);
                    show(els.labelEvening);
                    show(els.sunPoint);
                    show(els.labelSunPoint);
                    show(els.labelSun110w);
                    show(els.labelSunTime);
                    els.sunPoint.setAttribute('cx', '50');
                    els.sunPoint.setAttribute('cy', '250');
                    show(els.point120e);
                    show(els.label120e);
                    hide(els.calc1);
                    hide(els.calc2);
                    hide(els.calc3);
                    hide(els.calc4);
                    hide(els.calc5);
                    show(els.summary1);
                    show(els.summary2);
                    show(els.summary3);
                    show(els.pointD);
                    show(els.labelD);
                    els.boundary20w.style.strokeWidth = '3';
                    els.boundary160e.style.strokeWidth = '3';
                    els.infoContent.textContent = '完整答案：';
                    els.btnPrev.disabled = false;
                    els.btnNext.disabled = true;
                    els.btnSummary.disabled = false;
                    break;
            }
        }
        
        // 按钮事件绑定
        els.btnPrev.addEventListener('click', function() {
            if (currentStep > 0) {
                currentStep--;
                updateDisplay();
            }
        });
        
        els.btnNext.addEventListener('click', function() {
            if (currentStep < currentStepMax) {
                currentStep++;
                updateDisplay();
            }
        });
        
        els.btnSummary.addEventListener('click', function() {
            currentStep = 5;
            updateDisplay();
        });
        
        // 初始化显示
        updateDisplay();
    })();
    </script>
