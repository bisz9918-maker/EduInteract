    <script>
    (function() {
        // Scene 1 实现：四棱锥 P-ABCD 空间结构
        // 分步序列：0-初始线框 → 1-平面 PAB 高亮 → 2-平面 PDC 高亮 → 3-两平面同时显示
        
        const currentStep = { value: 0 };
        const totalSteps = 4;
        
        // DOM 元素引用
        const polyPab = document.getElementById('scene1-poly-pab');
        const polyPdc = document.getElementById('scene1-poly-pdc');
        const btnPrev = document.getElementById('scene1-btn-prev');
        const btnNext = document.getElementById('scene1-btn-next');
        const stepIndicator = document.getElementById('scene1-step-indicator');
        
        // 更新场景显示状态
        function updateScene() {
            // 更新平面填充显示状态（opacity 控制，配合 CSS transition 实现渐变）
            polyPab.style.opacity = currentStep.value >= 1 ? '0.2' : '0';
            polyPdc.style.opacity = currentStep.value >= 2 ? '0.2' : '0';
            
            // 更新按钮状态
            btnPrev.disabled = currentStep.value === 0;
            btnNext.disabled = currentStep.value === totalSteps - 1;
            
            // 更新步骤指示器
            stepIndicator.textContent = '步骤 ' + (currentStep.value + 1) + '/' + totalSteps;
        }
        
        // 上一步按钮事件
        btnPrev.addEventListener('click', function() {
            if (currentStep.value > 0) {
                currentStep.value--;
                updateScene();
            }
        });
        
        // 下一步按钮事件
        btnNext.addEventListener('click', function() {
            if (currentStep.value < totalSteps - 1) {
                currentStep.value++;
                updateScene();
            }
        });
        
        // 初始化场景
        updateScene();
    })();
    </script>
