    <script>
    (function() {
        // Scene 1: 圆中切线、割线与角平分线的角度关系推导
        // 布局坐标表（参考）：
        // | 元素 | 位置 | 备注 |
        // | 圆 O | 圆心 (450,300)，半径 120 | 中下部 |
        // | 点 P | (200,100) | 圆外左上 |
        // | 点 E | (330,300) | 切点，圆左侧 |
        // | 点 B | (480,220) | 割线与圆交点（近 P） |
        // | 点 A | (570,300) | 割线与圆交点（远 P） |
        // | 点 C | (400,300) | AE 与角平分线交点 |
        // | 点 D | (440,260) | BE 与角平分线交点 |
        // | 点 O | (450,300) | 圆心 |
        
        const scene1 = {
            currentStep: 0,
            totalSteps: 4,
            
            // 元素引用
            elements: {
                // 高亮区域
                highlightPEB: document.getElementById('scene1-highlight-PEB'),
                highlightPAE: document.getElementById('scene1-highlight-PAE'),
                highlightAngleAPE: document.getElementById('scene1-highlight-angle-APE'),
                // 角度弧线
                arcPCE: document.getElementById('scene1-arc-PCE'),
                // 角度文字
                textQuestion: document.getElementById('scene1-text-question'),
                // 推导文字
                textChordTangent: document.getElementById('scene1-text-chord-tangent'),
                textBisector: document.getElementById('scene1-text-bisector'),
                textConclusion: document.getElementById('scene1-text-conclusion'),
                // 按钮和指示器
                btnPrev: document.getElementById('scene1-btn-prev'),
                btnNext: document.getElementById('scene1-btn-next'),
                stepIndicator: document.getElementById('scene1-step-indicator')
            },
            
            // 步骤配置
            steps: [
                {
                    name: '步骤 1/4',
                    description: '基础图形与已知条件',
                    show: {
                        highlightPEB: false,
                        highlightPAE: false,
                        highlightAngleAPE: false,
                        arcPCE: false,
                        textQuestion: false,
                        textChordTangent: false,
                        textBisector: false,
                        textConclusion: false
                    }
                },
                {
                    name: '步骤 2/4',
                    description: '弦切角定理',
                    show: {
                        highlightPEB: true,
                        highlightPAE: true,
                        highlightAngleAPE: false,
                        arcPCE: false,
                        textQuestion: false,
                        textChordTangent: true,
                        textBisector: false,
                        textConclusion: false
                    }
                },
                {
                    name: '步骤 3/4',
                    description: '角平分线性质',
                    show: {
                        highlightPEB: true,
                        highlightPAE: true,
                        highlightAngleAPE: true,
                        arcPCE: false,
                        textQuestion: false,
                        textChordTangent: true,
                        textBisector: true,
                        textConclusion: false
                    }
                },
                {
                    name: '步骤 4/4',
                    description: '角度计算结论',
                    show: {
                        highlightPEB: true,
                        highlightPAE: true,
                        highlightAngleAPE: true,
                        arcPCE: true,
                        textQuestion: true,
                        textChordTangent: true,
                        textBisector: true,
                        textConclusion: true
                    }
                }
            ],
            
            // 更新显示状态
            updateDisplay: function() {
                const step = this.steps[this.currentStep];
                const el = this.elements;
                
                // 更新高亮区域
                el.highlightPEB.style.opacity = step.show.highlightPEB ? '1' : '0';
                el.highlightPAE.style.opacity = step.show.highlightPAE ? '1' : '0';
                el.highlightAngleAPE.style.opacity = step.show.highlightAngleAPE ? '1' : '0';
                
                // 更新角度弧线
                el.arcPCE.style.opacity = step.show.arcPCE ? '1' : '0';
                
                // 更新角度文字
                el.textQuestion.style.opacity = step.show.textQuestion ? '1' : '0';
                
                // 更新推导文字
                el.textChordTangent.style.opacity = step.show.textChordTangent ? '1' : '0';
                el.textBisector.style.opacity = step.show.textBisector ? '1' : '0';
                el.textConclusion.style.opacity = step.show.textConclusion ? '1' : '0';
                
                // 更新按钮状态
                el.btnPrev.disabled = (this.currentStep === 0);
                el.btnNext.disabled = (this.currentStep === this.totalSteps - 1);
                
                // 更新步骤指示器
                el.stepIndicator.textContent = step.name;
            },
            
            // 上一步
            prevStep: function() {
                if (this.currentStep > 0) {
                    this.currentStep--;
                    this.updateDisplay();
                }
            },
            
            // 下一步
            nextStep: function() {
                if (this.currentStep < this.totalSteps - 1) {
                    this.currentStep++;
                    this.updateDisplay();
                }
            },
            
            // 初始化
            init: function() {
                this.elements.btnPrev.addEventListener('click', () => this.prevStep());
                this.elements.btnNext.addEventListener('click', () => this.nextStep());
                this.updateDisplay();
            }
        };
        
        // 启动场景
        scene1.init();
    })();
    </script>
