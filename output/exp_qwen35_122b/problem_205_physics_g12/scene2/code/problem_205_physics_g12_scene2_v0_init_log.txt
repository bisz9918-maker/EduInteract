    <script>
    (function() {
        var currentStep = 0;
        var totalSteps = 4;
        
        var gravityGroup = document.getElementById('scene2-gravity-group');
        var normalGroup = document.getElementById('scene2-normal-group');
        var centripetalGroup = document.getElementById('scene2-centripetal-group');
        var velocityGroup = document.getElementById('scene2-velocity-group');
        var formulaGroup = document.getElementById('scene2-formula-group');
        var formula1 = document.getElementById('scene2-formula-1');
        var formula2 = document.getElementById('scene2-formula-2');
        var btnPrev = document.getElementById('scene2-btn-prev');
        var btnNext = document.getElementById('scene2-btn-next');
        var stepIndicator = document.getElementById('scene2-step-indicator');
        
        function show(el) {
            if (el) {
                el.style.display = 'inline';
                el.style.opacity = '1';
            }
        }
        
        function hide(el) {
            if (el) {
                el.style.opacity = '0';
                setTimeout(function() {
                    if (el) {
                        el.style.display = 'none';
                    }
                }, 300);
            }
        }
        
        function updateScene() {
            hide(gravityGroup);
            hide(normalGroup);
            hide(centripetalGroup);
            hide(velocityGroup);
            hide(formulaGroup);
            
            if (currentStep >= 1) {
                show(gravityGroup);
                show(normalGroup);
            }
            
            if (currentStep >= 2) {
                show(centripetalGroup);
                show(formulaGroup);
                formula1.textContent = 'mg + N = F_n';
                formula2.textContent = '';
            }
            
            if (currentStep >= 3) {
                hide(normalGroup);
                var centripetalLine = document.getElementById('scene2-centripetal-line');
                var centripetalArrow = document.getElementById('scene2-centripetal-arrow');
                if (centripetalLine) {
                    centripetalLine.style.strokeDasharray = 'none';
                    centripetalLine.style.strokeWidth = '3';
                }
                if (centripetalArrow) {
                    centripetalArrow.style.fill = '#c62828';
                }
                show(velocityGroup);
                formula1.textContent = 'mg = m·v_L²/R';
                formula2.textContent = 'v_L ≥ √(gR)';
            }
            
            btnPrev.disabled = (currentStep === 0);
            btnNext.disabled = (currentStep === totalSteps - 1);
            
            var displayStep = Math.min(currentStep, 3);
            stepIndicator.textContent = '步骤 ' + displayStep + '/3';
        }
        
        function prevStep() {
            if (currentStep > 0) {
                currentStep--;
                updateScene();
            }
        }
        
        function nextStep() {
            if (currentStep < totalSteps - 1) {
                currentStep++;
                updateScene();
            }
        }
        
        btnPrev.addEventListener('click', prevStep);
        btnNext.addEventListener('click', nextStep);
        
        updateScene();
    })();
    </script>
