    <script>
    (function() {
        // Scene 1: 垂直平分线性质与角度关系推导
        // 布局坐标表 (调整后):
        // | 元素 | 类型 | 位置/范围 | 大小 | 标签偏移方向 | 备注 |
        // |------|------|-----------|------|-------------|------|
        // | 三角形 ABC 边 | 线段 | A(400,100)-B(720,400)-C(80,400)-A | - | - | 黑色 2px |
        // | 点 A | 顶点 | (400,100) | - | 上方 (390,85) | 顶点标签 |
        // | 点 B | 顶点 | (720,400) | - | 右下方 (735,415) | 顶点标签 |
        // | 点 C | 顶点 | (80,400) | - | 左下方 (65,415) | 顶点标签 |
        // | 点 E | BC 上点 | (240,400) | - | 下方 (235,420) | BC 边上的点 |
        // | 点 F | BC 上点 | (480,400) | - | 下方 (475,420) | BC 边上的点 |
        // | 点 G | AC 上点 | (240,250) | - | 右上方 (255,235) | AC 中点 |
        // | 点 H | AB 上点 | (560,250) | - | 左上方 (545,235) | AB 中点 |
        // | ∠BAC 标签 | text | (600,140) | - | - | 移到右侧空白区域 |
        // | ∠EAF 标签 | text | (600,170) | - | - | 移到右侧空白区域 |
        
        const TOTAL_STEPS = 5;
        let currentStep = 0;
        
        // 工具函数：显示元素
        function show(el) {
            if (el) {
                el.style.display = 'block';
                el.style.opacity = '0';
                setTimeout(() => {
                    el.style.opacity = '1';
                }, 10);
            }
        }
        
        // 工具函数：隐藏元素
        function hide(el) {
            if (el) {
                el.style.opacity = '0';
                setTimeout(() => {
                    el.style.display = 'none';
                }, 300);
            }
        }
        
        // 更新步骤显示
        function updateStep() {
            // 重置所有步骤相关元素的显示状态
            const stepElements = {
                // 步骤 0: 基础图形和已知条件
                step0: [
                    'scene1-line-AB', 'scene1-line-AC', 'scene1-line-BC',
                    'scene1-line-AE', 'scene1-line-AF',
                    'scene1-arc-BAC', 'scene1-arc-EAF', 'scene1-arc-B', 'scene1-arc-C',
                    'scene1-label-A', 'scene1-label-B', 'scene1-label-C',
                    'scene1-label-E', 'scene1-label-F',
                    'scene1-text-BAC', 'scene1-text-EAF-Q'
                ],
                // 步骤 1: 垂直平分线性质
                step1: [
                    'scene1-line-EG', 'scene1-line-FH',
                    'scene1-right-angle-G', 'scene1-right-angle-H',
                    'scene1-mark-EA', 'scene1-mark-EC', 'scene1-mark-FA', 'scene1-mark-FB',
                    'scene1-label-G', 'scene1-label-H',
                    'scene1-text-EA-EC', 'scene1-text-FA-FB'
                ],
                // 步骤 2: 等腰三角形底角
                step2: [
                    'scene1-text-angle-C', 'scene1-text-angle-B'
                ],
                // 步骤 3: 推导公式
                step3: [
                    'scene1-text-derive1', 'scene1-text-derive2'
                ],
                // 步骤 4: 结论
                step4: [
                    'scene1-fill-EAF', 'scene1-text-conclusion',
                    'scene1-text-options', 'scene1-text-option-B'
                ]
            };
            
            // 根据当前步骤显示对应元素
            for (let step = 0; step < TOTAL_STEPS; step++) {
                const elements = stepElements['step' + step];
                elements.forEach(id => {
                    const el = document.getElementById(id);
                    if (el) {
                        if (step <= currentStep) {
                            show(el);
                        } else {
                            hide(el);
                        }
                    }
                });
            }
            
            // 步骤 4 时隐藏问号标签
            if (currentStep >= 4) {
                hide(document.getElementById('scene1-text-EAF-Q'));
            } else {
                show(document.getElementById('scene1-text-EAF-Q'));
            }
            
            // 更新按钮状态
            const prevBtn = document.getElementById('scene1-prevBtn');
            const nextBtn = document.getElementById('scene1-nextBtn');
            const stepIndicator = document.getElementById('scene1-stepIndicator');
            
            prevBtn.disabled = (currentStep === 0);
            nextBtn.disabled = (currentStep === TOTAL_STEPS - 1);
            stepIndicator.textContent = '步骤 ' + (currentStep + 1) + '/' + TOTAL_STEPS;
        }
        
        // 事件绑定
        document.getElementById('scene1-prevBtn').addEventListener('click', function() {
            if (currentStep > 0) {
                currentStep--;
                updateStep();
            }
        });
        
        document.getElementById('scene1-nextBtn').addEventListener('click', function() {
            if (currentStep < TOTAL_STEPS - 1) {
                currentStep++;
                updateStep();
            }
        });
        
        // 初始化
        updateStep();
    })();
    </script>
