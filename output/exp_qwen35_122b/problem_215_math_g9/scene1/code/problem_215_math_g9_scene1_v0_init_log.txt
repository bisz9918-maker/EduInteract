    <script>
    (function() {
        const currentStep = { value: 0 };
        const maxStep = 4;
        
        // 工具函数：显示元素
        function show(elId) {
            const el = document.getElementById(elId);
            if (el) {
                el.style.opacity = '1';
                el.style.display = 'block';
                el.classList.add('scene1-fade-in');
            }
        }
        
        // 工具函数：隐藏元素
        function hide(elId) {
            const el = document.getElementById(elId);
            if (el) {
                el.style.opacity = '0';
                setTimeout(() => {
                    if (el.style.opacity === '0') {
                        el.style.display = 'none';
                    }
                }, 300);
            }
        }
        
        // 工具函数：闪烁效果
        function flash(elId, className) {
            const el = document.getElementById(elId);
            if (el) {
                el.classList.remove(className);
                void el.offsetWidth; // 触发重绘
                el.classList.add(className);
            }
        }
        
        // 更新场景状态
        function updateScene() {
            const step = currentStep.value;
            
            // 步骤 0: 基础图形（始终显示）
            show('scene1-line-a');
            show('scene1-square');
            show('scene1-bf-seg');
            show('scene1-de-seg');
            show('scene1-label-a');
            show('scene1-label-b');
            show('scene1-label-c');
            show('scene1-label-d');
            show('scene1-label-e');
            show('scene1-label-f');
            show('scene1-label-line-a');
            
            // 隐藏所有动态元素
            hide('scene1-vert-f');
            hide('scene1-vert-e');
            hide('scene1-label-de');
            hide('scene1-label-bf');
            hide('scene1-arc-baf');
            hide('scene1-arc-dae');
            hide('scene1-text-angle');
            hide('scene1-tri-baf');
            hide('scene1-tri-ade');
            hide('scene1-text-congruence');
            hide('scene1-text-aas');
            hide('scene1-label-af');
            hide('scene1-label-ae');
            hide('scene1-seg-ef');
            hide('scene1-label-ef');
            hide('scene1-text-result');
            
            // 步骤 1: 标注已知条件
            if (step >= 1) {
                show('scene1-vert-f');
                show('scene1-vert-e');
                show('scene1-label-de');
                show('scene1-label-bf');
                flash('scene1-de-seg', 'scene1-flash-blue');
                flash('scene1-bf-seg', 'scene1-flash-red');
            }
            
            // 步骤 2: 角度分析
            if (step >= 2) {
                show('scene1-arc-baf');
                show('scene1-arc-dae');
                show('scene1-text-angle');
                setTimeout(() => {
                    flash('scene1-arc-baf', 'scene1-flash-red');
                    flash('scene1-arc-dae', 'scene1-flash-blue');
                }, 300);
            }
            
            // 步骤 3: 证明全等
            if (step >= 3) {
                show('scene1-tri-baf');
                show('scene1-tri-ade');
                show('scene1-text-congruence');
                show('scene1-text-aas');
            }
            
            // 步骤 4: 得出结论
            if (step >= 4) {
                show('scene1-label-af');
                show('scene1-label-ae');
                show('scene1-seg-ef');
                show('scene1-label-ef');
                show('scene1-text-result');
            }
            
            // 更新按钮状态
            const prevBtn = document.getElementById('scene1-prev-btn');
            const nextBtn = document.getElementById('scene1-next-btn');
            
            prevBtn.disabled = (step === 0);
            
            if (step === maxStep) {
                nextBtn.textContent = '完成 ✓';
                nextBtn.disabled = true;
            } else {
                nextBtn.textContent = '下一步 ▶';
                nextBtn.disabled = false;
            }
        }
        
        // 初始化
        updateScene();
        
        // 事件绑定
        document.getElementById('scene1-prev-btn').addEventListener('click', () => {
            if (currentStep.value > 0) {
                currentStep.value--;
                updateScene();
            }
        });
        
        document.getElementById('scene1-next-btn').addEventListener('click', () => {
            if (currentStep.value < maxStep) {
                currentStep.value++;
                updateScene();
            }
        });
    })();
    </script>
