    <script>
    (function() {
        // Scene 2 实现
        // 布局坐标表对照：
        // EF 线段：(100,380)-(700,380)
        // C 点：(450,380), F 点：(700,380), E 点：(100,380)
        // D 点：(520,220), A 点：(520,120), B 点：(200,150)
        
        const currentStepMax = 5;
        let currentStep = 0;
        
        // 工具函数：显示元素
        function show(elId) {
            const el = document.getElementById(elId);
            if (el) {
                el.style.opacity = '1';
                el.style.display = 'block';
            }
        }
        
        // 工具函数：隐藏元素
        function hide(elId) {
            const el = document.getElementById(elId);
            if (el) {
                el.style.opacity = '0';
            }
        }
        
        // 更新按钮状态
        function updateButtons() {
            document.getElementById('scene2-prev-btn').disabled = (currentStep === 0);
            document.getElementById('scene2-next-btn').disabled = (currentStep === currentStepMax);
        }
        
        // 步骤 0：初始状态 - 只显示基础图形
        function renderStep0() {
            // 显示所有基础线段和点
            const baseElements = [
                'line-ef', 'line-ac', 'line-ab', 'line-bc', 'line-de', 'line-df',
                'point-e', 'point-c', 'point-f', 'point-d', 'point-a', 'point-b',
                'label-e', 'label-c', 'label-f', 'label-d', 'label-a', 'label-b'
            ];
            baseElements.forEach(id => show('scene2-' + id));
            
            // 隐藏所有角度弧线、标签、计算框
            const hiddenElements = [
                'arc-acb', 'label-acb',
                'arc-bce', 'label-bce', 'arc-ace', 'label-ace',
                'arc-acf', 'label-acf',
                'arc-dcf', 'label-dcf', 'arc-f', 'label-f', 'arc-cdf', 'label-cdf',
                'calc-box-1', 'calc-text-1', 'calc-text-1b',
                'calc-box-2', 'calc-text-2', 'calc-text-2b',
                'calc-box-3', 'calc-text-3', 'calc-text-3b',
                'calc-box-4', 'calc-text-4', 'calc-text-4b',
                'answer-box', 'answer-text',
                'line-ef-dashed'
            ];
            hiddenElements.forEach(id => hide('scene2-' + id));
            
            // 重置高亮
            document.getElementById('scene2-line-ab').classList.remove('scene2-highlight-blue', 'scene2-highlight-red', 'scene2-highlight-green');
            document.getElementById('scene2-line-bc').classList.remove('scene2-highlight-blue', 'scene2-highlight-red', 'scene2-highlight-green');
            document.getElementById('scene2-line-ac').classList.remove('scene2-highlight-blue', 'scene2-highlight-red', 'scene2-highlight-green');
            document.getElementById('scene2-line-df').classList.remove('scene2-highlight-green');
            document.getElementById('scene2-line-ef').classList.remove('scene2-highlight-green');
            
            // 重置∠CDF 标签大小
            document.getElementById('scene2-label-cdf').setAttribute('font-size', '13');
            document.getElementById('scene2-label-cdf').removeAttribute('font-weight');
        }
        
        // 步骤 1：等腰直角三角形底角推导
        function renderStep1() {
            renderStep0();
            // 显示∠ACB 相关
            show('scene2-arc-acb');
            show('scene2-label-acb');
            show('scene2-calc-box-1');
            show('scene2-calc-text-1');
            show('scene2-calc-text-1b');
            // 高亮△ABC
            document.getElementById('scene2-line-ab').classList.add('scene2-highlight-blue');
            document.getElementById('scene2-line-bc').classList.add('scene2-highlight-blue');
            document.getElementById('scene2-line-ac').classList.add('scene2-highlight-blue');
        }
        
        // 步骤 2：角度叠加计算
        function renderStep2() {
            renderStep1();
            // 显示∠BCE、∠ACE 相关
            show('scene2-arc-bce');
            show('scene2-label-bce');
            show('scene2-arc-ace');
            show('scene2-label-ace');
            show('scene2-calc-box-2');
            show('scene2-calc-text-2');
            show('scene2-calc-text-2b');
            // 高亮 BC 线段
            document.getElementById('scene2-line-bc').classList.add('scene2-highlight-red');
        }
        
        // 步骤 3：平角关系应用
        function renderStep3() {
            renderStep2();
            // 显示∠ACF 相关
            show('scene2-arc-acf');
            show('scene2-label-acf');
            show('scene2-calc-box-3');
            show('scene2-calc-text-3');
            show('scene2-calc-text-3b');
            // 显示辅助虚线
            show('scene2-line-ef-dashed');
        }
        
        // 步骤 4：三角形内角和定理
        function renderStep4() {
            renderStep3();
            // 显示∠DCF、∠F、∠CDF 相关
            show('scene2-arc-dcf');
            show('scene2-label-dcf');
            show('scene2-arc-f');
            show('scene2-label-f');
            show('scene2-arc-cdf');
            show('scene2-label-cdf');
            show('scene2-calc-box-4');
            show('scene2-calc-text-4');
            show('scene2-calc-text-4b');
            // 高亮△CDF
            document.getElementById('scene2-line-ac').classList.add('scene2-highlight-green');
            document.getElementById('scene2-line-df').classList.add('scene2-highlight-green');
            document.getElementById('scene2-line-ef').classList.add('scene2-highlight-green');
        }
        
        // 步骤 5：最终答案
        function renderStep5() {
            renderStep4();
            // 显示答案框
            show('scene2-answer-box');
            show('scene2-answer-text');
            // 放大∠CDF 标签
            document.getElementById('scene2-label-cdf').setAttribute('font-size', '18');
            document.getElementById('scene2-label-cdf').setAttribute('font-weight', 'bold');
        }
        
        // 渲染当前步骤
        function render() {
            switch(currentStep) {
                case 0: renderStep0(); break;
                case 1: renderStep1(); break;
                case 2: renderStep2(); break;
                case 3: renderStep3(); break;
                case 4: renderStep4(); break;
                case 5: renderStep5(); break;
            }
            updateButtons();
        }
        
        // 事件绑定
        document.getElementById('scene2-prev-btn').addEventListener('click', function() {
            if (currentStep > 0) {
                currentStep--;
                render();
            }
        });
        
        document.getElementById('scene2-next-btn').addEventListener('click', function() {
            if (currentStep < currentStepMax) {
                currentStep++;
                render();
            }
        });
        
        // 初始化
        render();
    })();
    </script>
