    <script>
    (function() {
        // Scene 1 实现
        // 布局坐标表对照：
        // 经线 37.5°：(350,80)-(350,380)
        // 经线 22.5°：(550,80)-(550,380)
        // 纬线 23.5°N：(50,120)-(750,120)
        // 纬线 0°：(50,230)-(750,230)
        // 纬线 23.5°S：(50,340)-(750,340)
        // 晨昏线：(260,80)-(520,380)
        // 黑夜区域：(50,80)-(260,80)-(520,380)-(50,380)
        // 甲点：(350,230)
        // 乙点：(550,340)
        
        var currentStep = 0;
        var maxStep = 5;
        
        // 工具函数：显示元素
        function show(el) {
            if (el) {
                el.style.display = 'block';
                el.style.opacity = '1';
            }
        }
        
        // 工具函数：隐藏元素
        function hide(el) {
            if (el) {
                el.style.display = 'none';
                el.style.opacity = '0';
            }
        }
        
        // 更新可见性
        function updateVisibility() {
            var elements = document.querySelectorAll('[data-step]');
            elements.forEach(function(el) {
                var stepAttr = el.getAttribute('data-step');
                if (stepAttr === 'all') {
                    show(el);
                } else {
                    var step = parseInt(stepAttr);
                    if (step <= currentStep) {
                        show(el);
                    } else {
                        hide(el);
                    }
                }
            });
            
            // 更新按钮状态
            var prevBtn = document.getElementById('scene1-prev-btn');
            var nextBtn = document.getElementById('scene1-next-btn');
            
            prevBtn.disabled = (currentStep === 0);
            
            if (currentStep === maxStep) {
                nextBtn.textContent = '完成';
                nextBtn.disabled = true;
            } else {
                nextBtn.textContent = '下一步 ▶';
                nextBtn.disabled = false;
            }
        }
        
        // 上一步
        function prevStep() {
            if (currentStep > 0) {
                currentStep--;
                updateVisibility();
            }
        }
        
        // 下一步
        function nextStep() {
            if (currentStep < maxStep) {
                currentStep++;
                updateVisibility();
            }
        }
        
        // 绑定事件
        document.getElementById('scene1-prev-btn').addEventListener('click', prevStep);
        document.getElementById('scene1-next-btn').addEventListener('click', nextStep);
        
        // 初始化
        updateVisibility();
    })();
    </script>
