    <script>
    (function() {
        // Scene 2 实现
        // 布局坐标表：
        // | 元素 | 位置 | 备注 |
        // | 背景网格 | (100,80)-(700,400) | 600×320 |
        // | 经线 170°E | (150,80)-(150,400) | 实线 |
        // | 经线 180° | (400,80)-(400,400) | 实线 |
        // | 经线 170°W | (650,80)-(650,400) | 实线 |
        // | 纬线 20°N | (100,320)-(700,320) | 实线 |
        // | 纬线 30°N | (100,200)-(700,200) | 实线 |
        // | 纬线 40°N | (100,100)-(700,100) | 实线 |
        // | B 点 | (150,320) | r=8, 绿色 |
        // | D 点 | (650,320) | r=8, 绿色 |
        // | B 到 D 路径 | (150,320)→(400,220)→(650,320) | 贝塞尔曲线 |
        // | 东北箭头 | (225,290)-(300,260) | 路径前半段 |
        // | 东南箭头 | (500,260)-(575,290) | 路径后半段 |

        const TOTAL_STEPS = 6;
        let currentStep = 1;

        // 元素显示控制
        function showElement(id) {
            const el = document.getElementById(id);
            if (el) {
                el.style.display = 'block';
                el.style.opacity = '1';
            }
        }

        function hideElement(id) {
            const el = document.getElementById(id);
            if (el) {
                el.style.opacity = '0';
                setTimeout(() => {
                    if (el) el.style.display = 'none';
                }, 300);
            }
        }

        function showElements(ids) {
            ids.forEach(id => showElement(id));
        }

        function hideElements(ids) {
            ids.forEach(id => hideElement(id));
        }

        // 步骤元素集合
        const stepElements = {
            1: [], // 初始状态：B、D 点及所有经纬度标签已显示
            2: ['scene2-path-BD', 'scene2-path-peak', 'scene2-label-longitude-change'],
            3: ['scene2-arrow-northeast', 'scene2-label-northeast'],
            4: ['scene2-arrow-southeast', 'scene2-label-southeast'],
            5: ['scene2-label-latitude-change'],
            6: ['scene2-conclusion']
        };

        // 更新场景显示
        function updateScene() {
            // 隐藏步骤 2-6 的所有元素
            for (let step = 2; step <= TOTAL_STEPS; step++) {
                hideElements(stepElements[step]);
            }

            // 显示当前步骤及之前的所有元素
            for (let step = 1; step <= currentStep; step++) {
                showElements(stepElements[step]);
            }

            // 更新步骤指示器
            document.getElementById('scene2-step-indicator').textContent = `步骤 ${currentStep}/${TOTAL_STEPS}`;

            // 更新按钮状态
            document.getElementById('scene2-btn-prev').disabled = (currentStep === 1);
            document.getElementById('scene2-btn-next').disabled = (currentStep === TOTAL_STEPS);
        }

        // 事件绑定
        document.getElementById('scene2-btn-prev').addEventListener('click', function() {
            if (currentStep > 1) {
                currentStep--;
                updateScene();
            }
        });

        document.getElementById('scene2-btn-next').addEventListener('click', function() {
            if (currentStep < TOTAL_STEPS) {
                currentStep++;
                updateScene();
            }
        });

        // 初始化
        updateScene();
    })();
    </script>
