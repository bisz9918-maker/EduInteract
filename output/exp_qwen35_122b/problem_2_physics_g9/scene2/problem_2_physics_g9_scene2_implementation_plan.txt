- JS 状态切换逻辑：
  ```javascript
  (function() {
    var currentState = 0;
    var states = [
      {
        text: "状态 1：木块受力分析 — 木块受到重力 G（竖直向下）、木板支持力 F_N（竖直向上）、对木板的压力 F_压（竖直向下）",
        balanceBox: false,
        actionReactionBox: false
      },
      {
        text: "状态 2：平衡力 — 重力 G 与支持力 F_N 是一对平衡力，它们作用在同一物体（木块）上，大小相等、方向相反、作用在同一直线上",
        balanceBox: true,
        actionReactionBox: false
      },
      {
        text: "状态 3：作用力与反作用力 — 支持力 F_N 与压力 F_压是一对作用力与反作用力，它们作用在不同物体上（F_N 作用在木块，F_压作用在木板），大小相等、方向相反、作用在同一直线上",
        balanceBox: false,
        actionReactionBox: true
      }
    ];
    
    function switchState(newState) {
      currentState = newState;
      var state = states[currentState];
      
      // 更新说明文字
      document.getElementById('scene2-statusText').textContent = state.text;
      
      // 更新配对框显示
      document.getElementById('scene2-balanceBox').style.display = state.balanceBox ? 'block' : 'none';
      document.getElementById('scene2-balanceLabel').style.display = state.balanceBox ? 'block' : 'none';
      document.getElementById('scene2-actionReactionBox').style.display = state.actionReactionBox ? 'block' : 'none';
      document.getElementById('scene2-actionReactionLabel').style.display = state.actionReactionBox ? 'block' : 'none';
      
      // 更新按钮高亮
      for (var i = 1; i <= 3; i++) {
        var btn = document.getElementById('scene2-btn-state' + i);
        if (btn) {
          btn.classList.toggle('active', i - 1 === currentState);
        }
      }
    }
    
    // 绑定按钮事件
    document.getElementById('scene2-btn-state1').addEventListener('click', function() { switchState(0); });
    document.getElementById('scene2-btn-state2').addEventListener('click', function() { switchState(1); });
    document.getElementById('scene2-btn-state3').addEventListener('click', function() { switchState(2); });
    
    // 初始化状态
    switchState(0);
  })();
  ```
