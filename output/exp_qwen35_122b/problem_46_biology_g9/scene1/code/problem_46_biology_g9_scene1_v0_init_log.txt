    <script>
    (function() {
        // Scene 1 实现 - DNA 分子结构详解
        
        const scene1Svg = document.getElementById('scene1-svg');
        const scene1PrevBtn = document.getElementById('scene1-prev-btn');
        const scene1NextBtn = document.getElementById('scene1-next-btn');
        
        let currentStep = 0;
        const totalSteps = 4;
        
        // 所有氢键线 ID 列表
        const hydrogenBondIds = [
            'scene1-hbond-1a', 'scene1-hbond-1b', 'scene1-hbond-1c',
            'scene1-hbond-2a', 'scene1-hbond-2b',
            'scene1-hbond-3a', 'scene1-hbond-3b', 'scene1-hbond-3c',
            'scene1-hbond-4a', 'scene1-hbond-4b', 'scene1-hbond-4c',
            'scene1-hbond-5a', 'scene1-hbond-5b', 'scene1-hbond-5c',
            'scene1-hbond-6a', 'scene1-hbond-6b',
            'scene1-hbond-7a', 'scene1-hbond-7b',
            'scene1-hbond-8a', 'scene1-hbond-8b'
        ];
        
        // C-G 碱基对（3 条氢键）：1, 3, 4, 5
        const cgBondIds = [
            'scene1-hbond-1a', 'scene1-hbond-1b', 'scene1-hbond-1c',
            'scene1-hbond-3a', 'scene1-hbond-3b', 'scene1-hbond-3c',
            'scene1-hbond-4a', 'scene1-hbond-4b', 'scene1-hbond-4c',
            'scene1-hbond-5a', 'scene1-hbond-5b', 'scene1-hbond-5c'
        ];
        
        // A-T 碱基对（2 条氢键）：2, 6, 7, 8
        const atBondIds = [
            'scene1-hbond-2a', 'scene1-hbond-2b',
            'scene1-hbond-6a', 'scene1-hbond-6b',
            'scene1-hbond-7a', 'scene1-hbond-7b',
            'scene1-hbond-8a', 'scene1-hbond-8b'
        ];
        
        // 工具函数：显示元素
        function show(element) {
            if (element) {
                element.style.display = 'block';
                element.style.opacity = '1';
                element.classList.remove('scene1-hidden');
                element.classList.add('scene1-visible');
            }
        }
        
        // 工具函数：隐藏元素
        function hide(element) {
            if (element) {
                element.style.display = 'none';
                element.style.opacity = '0';
                element.classList.remove('scene1-visible');
                element.classList.add('scene1-hidden');
            }
        }
        
        // 工具函数：重置所有高亮
        function resetHighlights() {
            // 重置标注①
            const ann1Arrow = document.getElementById('scene1-annotation-1-arrow');
            const ann1Head = document.getElementById('scene1-annotation-1-head');
            const ann1Text = document.getElementById('scene1-annotation-1-text');
            const ann1Extra = document.getElementById('scene1-annotation-1-extra');
            
            ann1Arrow.setAttribute('stroke', '#888888');
            ann1Head.setAttribute('fill', '#888888');
            ann1Text.setAttribute('fill', '#888888');
            ann1Text.classList.remove('highlight-blue-text');
            hide(ann1Extra);
            
            // 重置标注②
            const ann2Arrow = document.getElementById('scene1-annotation-2-arrow');
            const ann2Head = document.getElementById('scene1-annotation-2-head');
            const ann2Text = document.getElementById('scene1-annotation-2-text');
            const ann2Extra = document.getElementById('scene1-annotation-2-extra');
            
            ann2Arrow.setAttribute('stroke', '#888888');
            ann2Head.setAttribute('fill', '#888888');
            ann2Text.setAttribute('fill', '#888888');
            ann2Text.classList.remove('highlight-red-text');
            hide(ann2Extra);
            
            // 重置所有氢键线为灰色
            hydrogenBondIds.forEach(id => {
                const el = document.getElementById(id);
                if (el) {
                    el.setAttribute('stroke', '#888888');
                    el.setAttribute('stroke-width', '1.5');
                    el.classList.remove('highlight-red', 'highlight-green');
                }
            });
            
            // 隐藏配对规则文字
            hide(document.getElementById('scene1-pairing-rule-title'));
            hide(document.getElementById('scene1-pairing-rule-1'));
            hide(document.getElementById('scene1-pairing-rule-2'));
            
            // 隐藏计算框和计算文字
            hide(document.getElementById('scene1-calc-box'));
            hide(document.getElementById('scene1-calc-text-1'));
            hide(document.getElementById('scene1-calc-text-2'));
            hide(document.getElementById('scene1-calc-text-3'));
        }
        
        // 渲染当前步骤
        function renderStep() {
            // 重置所有高亮
            resetHighlights();
            
            // 更新按钮状态
            scene1PrevBtn.disabled = (currentStep === 0);
            scene1NextBtn.disabled = (currentStep === totalSteps);
            
            switch(currentStep) {
                case 0:
                    // 初始状态：所有元素灰色显示
                    break;
                    
                case 1:
                    // 高亮磷酸二酯键（蓝色）
                    const ann1Arrow1 = document.getElementById('scene1-annotation-1-arrow');
                    const ann1Head1 = document.getElementById('scene1-annotation-1-head');
                    const ann1Text1 = document.getElementById('scene1-annotation-1-text');
                    const ann1Extra1 = document.getElementById('scene1-annotation-1-extra');
                    
                    ann1Arrow1.setAttribute('stroke', '#1565c0');
                    ann1Arrow1.setAttribute('stroke-width', '3');
                    ann1Head1.setAttribute('fill', '#1565c0');
                    ann1Text1.setAttribute('fill', '#1565c0');
                    ann1Text1.classList.add('highlight-blue-text');
                    show(ann1Extra1);
                    break;
                    
                case 2:
                    // 高亮氢键（红色）
                    const ann2Arrow2 = document.getElementById('scene1-annotation-2-arrow');
                    const ann2Head2 = document.getElementById('scene1-annotation-2-head');
                    const ann2Text2 = document.getElementById('scene1-annotation-2-text');
                    const ann2Extra2 = document.getElementById('scene1-annotation-2-extra');
                    
                    ann2Arrow2.setAttribute('stroke', '#c62828');
                    ann2Arrow2.setAttribute('stroke-width', '3');
                    ann2Head2.setAttribute('fill', '#c62828');
                    ann2Text2.setAttribute('fill', '#c62828');
                    ann2Text2.classList.add('highlight-red-text');
                    show(ann2Extra2);
                    
                    // 所有氢键线变红加粗
                    hydrogenBondIds.forEach(id => {
                        const el = document.getElementById(id);
                        if (el) {
                            el.setAttribute('stroke', '#c62828');
                            el.setAttribute('stroke-width', '2.5');
                            el.classList.add('highlight-red');
                        }
                    });
                    break;
                    
                case 3:
                    // 显示碱基配对规则（绿色）
                    hydrogenBondIds.forEach(id => {
                        const el = document.getElementById(id);
                        if (el) {
                            el.setAttribute('stroke', '#2e7d32');
                            el.setAttribute('stroke-width', '2.5');
                            el.classList.add('highlight-green');
                        }
                    });
                    
                    show(document.getElementById('scene1-pairing-rule-title'));
                    show(document.getElementById('scene1-pairing-rule-1'));
                    show(document.getElementById('scene1-pairing-rule-2'));
                    break;
                    
                case 4:
                    // 显示碱基比例计算
                    show(document.getElementById('scene1-calc-box'));
                    show(document.getElementById('scene1-calc-text-1'));
                    show(document.getElementById('scene1-calc-text-2'));
                    show(document.getElementById('scene1-calc-text-3'));
                    break;
            }
        }
        
        // 事件绑定
        scene1PrevBtn.addEventListener('click', function() {
            if (currentStep > 0) {
                currentStep--;
                renderStep();
            }
        });
        
        scene1NextBtn.addEventListener('click', function() {
            if (currentStep < totalSteps) {
                currentStep++;
                renderStep();
            }
        });
        
        // 初始化渲染
        renderStep();
    })();
    </script>
