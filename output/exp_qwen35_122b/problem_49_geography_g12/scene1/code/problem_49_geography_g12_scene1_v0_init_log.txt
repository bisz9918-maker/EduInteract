    <script>
    (function() {
        // Scene 1: 南极俯视图中的晨昏线与日期分界线
        // 布局坐标表（实现对照依据）：
        // 画布尺寸：800×500px，中心点 O 位于 (400,350)
        // 中心点 O: (400,350), r=6
        // 70° 纬线：圆心 (400,350), r=120, 上半圆
        // 外圈纬线：圆心 (400,350), r=200, 上半圆
        // 经线 B: (400,350)-(227,250), 角度 150°
        // 切点 M: (296,310), 在 70° 纬线上，角度 150°
        // 端点 A: (650,220)
        // 晨昏线 MA: 贝塞尔曲线，M(296,310) 到 A(650,220), 控制点 (480,200)
        // 阴影扇形：从经线 B(150°) 到水平向左 (180°), r=200
        
        const scene1 = {
            currentStep: 0,
            totalSteps: 5,
            
            // 元素引用
            elements: {
                sectorShadow: document.getElementById('scene1-sector-shadow'),
                arcOuter: document.getElementById('scene1-arc-outer'),
                arc70: document.getElementById('scene1-arc-70'),
                lineHoriz: document.getElementById('scene1-line-horiz'),
                lineB: document.getElementById('scene1-line-B'),
                curveMA: document.getElementById('scene1-curve-MA'),
                centerO: document.getElementById('scene1-center-O'),
                pointM: document.getElementById('scene1-point-M'),
                pointA: document.getElementById('scene1-point-A'),
                labelO: document.getElementById('scene1-label-O'),
                label70: document.getElementById('scene1-label-70'),
                labelB: document.getElementById('scene1-label-B'),
                labelM: document.getElementById('scene1-label-M'),
                labelA: document.getElementById('scene1-label-A'),
                labelShadow: document.getElementById('scene1-label-shadow'),
                btnPrev: document.getElementById('scene1-btn-prev'),
                btnNext: document.getElementById('scene1-btn-next'),
                stepIndicator: document.getElementById('scene1-step-indicator'),
                summaryText: document.getElementById('scene1-summary-text')
            },
            
            // 显示/隐藏工具函数
            show: function(el) {
                if (el) {
                    el.style.display = 'block';
                    el.style.opacity = '1';
                }
            },
            
            hide: function(el) {
                if (el) {
                    el.style.display = 'none';
                    el.style.opacity = '0';
                }
            },
            
            // 更新步骤显示
            updateStep: function() {
                const el = this.elements;
                
                // 步骤 0: 基础框架
                if (this.currentStep >= 0) {
                    this.show(el.arcOuter);
                    this.show(el.arc70);
                    this.show(el.lineHoriz);
                    this.show(el.centerO);
                    this.show(el.labelO);
                    this.show(el.label70);
                }
                
                // 步骤 1: 添加经线与切点
                if (this.currentStep >= 1) {
                    this.show(el.lineB);
                    this.show(el.pointM);
                    this.show(el.labelB);
                    this.show(el.labelM);
                }
                
                // 步骤 2: 添加晨昏线
                if (this.currentStep >= 2) {
                    this.show(el.curveMA);
                    this.show(el.pointA);
                    this.show(el.labelA);
                }
                
                // 步骤 3: 添加阴影日期区域
                if (this.currentStep >= 3) {
                    this.show(el.sectorShadow);
                    this.show(el.labelShadow);
                }
                
                // 步骤 4: 总结标注
                if (this.currentStep >= 4) {
                    el.summaryText.innerHTML = '<strong>M 为切点：</strong>70°S 纬线出现极昼/极夜 | <strong>阴影区：</strong>日期分界线（180°经线或 0 时经线）';
                    el.summaryText.style.display = 'block';
                }
                
                // 更新按钮状态
                el.btnPrev.disabled = (this.currentStep === 0);
                el.btnNext.disabled = (this.currentStep === this.totalSteps - 1);
                
                // 更新步骤指示器
                el.stepIndicator.textContent = '步骤 ' + (this.currentStep + 1) + '/' + this.totalSteps;
            },
            
            // 初始化
            init: function() {
                const self = this;
                
                // 绑定按钮事件
                el.btnPrev.addEventListener('click', function() {
                    if (self.currentStep > 0) {
                        self.currentStep--;
                        self.updateStep();
                    }
                });
                
                el.btnNext.addEventListener('click', function() {
                    if (self.currentStep < self.totalSteps - 1) {
                        self.currentStep++;
                        self.updateStep();
                    }
                });
                
                // 初始显示
                this.updateStep();
            }
        };
        
        // 启动场景
        scene1.init();
    })();
    </script>
