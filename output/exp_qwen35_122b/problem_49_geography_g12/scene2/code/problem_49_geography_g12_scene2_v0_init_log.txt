    <script>
    (function() {
        // Scene 2 实现 - 太阳直射点坐标推导
        const scene2 = {
            currentStep: 0,
            totalSteps: 5,
            
            // 步骤元素配置
            stepElements: {
                0: {
                    show: ['scene2-outer-circle', 'scene2-point-o', 'scene2-point-m', 'scene2-line-ma', 'scene2-label-o', 'scene2-label-m', 'scene2-label-ma'],
                    hide: ['scene2-arc-70s', 'scene2-line-ob', 'scene2-line-oc', 'scene2-point-p', 'scene2-latitude-group', 'scene2-longitude-group', 'scene2-final-group', 'scene2-label-70s', 'scene2-label-ob', 'scene2-label-oc', 'scene2-label-p']
                },
                1: {
                    show: ['scene2-outer-circle', 'scene2-point-o', 'scene2-point-m', 'scene2-line-ma', 'scene2-arc-70s', 'scene2-label-o', 'scene2-label-m', 'scene2-label-ma', 'scene2-label-70s', 'scene2-latitude-group'],
                    hide: ['scene2-line-ob', 'scene2-line-oc', 'scene2-point-p', 'scene2-longitude-group', 'scene2-final-group', 'scene2-label-ob', 'scene2-label-oc', 'scene2-label-p']
                },
                2: {
                    show: ['scene2-outer-circle', 'scene2-point-o', 'scene2-point-m', 'scene2-line-ma', 'scene2-arc-70s', 'scene2-line-ob', 'scene2-line-oc', 'scene2-label-o', 'scene2-label-m', 'scene2-label-ma', 'scene2-label-70s', 'scene2-label-ob', 'scene2-label-oc', 'scene2-latitude-group', 'scene2-longitude-group'],
                    hide: ['scene2-point-p', 'scene2-final-group', 'scene2-label-p']
                },
                3: {
                    show: ['scene2-outer-circle', 'scene2-point-o', 'scene2-point-m', 'scene2-line-ma', 'scene2-arc-70s', 'scene2-line-ob', 'scene2-line-oc', 'scene2-point-p', 'scene2-label-o', 'scene2-label-m', 'scene2-label-ma', 'scene2-label-70s', 'scene2-label-ob', 'scene2-label-oc', 'scene2-label-p', 'scene2-latitude-group', 'scene2-longitude-group'],
                    hide: ['scene2-final-group']
                },
                4: {
                    show: ['scene2-outer-circle', 'scene2-point-o', 'scene2-point-m', 'scene2-line-ma', 'scene2-arc-70s', 'scene2-line-ob', 'scene2-line-oc', 'scene2-point-p', 'scene2-label-o', 'scene2-label-m', 'scene2-label-ma', 'scene2-label-70s', 'scene2-label-ob', 'scene2-label-oc', 'scene2-label-p', 'scene2-latitude-group', 'scene2-longitude-group', 'scene2-final-group'],
                    hide: []
                }
            },
            
            // 初始化
            init: function() {
                this.updateScene();
                this.bindEvents();
            },
            
            // 更新场景显示
            updateScene: function() {
                const config = this.stepElements[this.currentStep];
                
                // 显示元素
                config.show.forEach(id => {
                    const el = document.getElementById(id);
                    if (el) {
                        el.style.opacity = '1';
                        el.style.display = 'block';
                    }
                });
                
                // 隐藏元素
                config.hide.forEach(id => {
                    const el = document.getElementById(id);
                    if (el) {
                        el.style.opacity = '0';
                        if (el.tagName === 'G' || el.tagName === 'svg' || el.tagName === 'SVG') {
                            el.style.display = 'none';
                        }
                    }
                });
                
                // 更新步骤指示器
                document.getElementById('scene2-step-indicator').textContent = `步骤 ${this.currentStep + 1}/${this.totalSteps}`;
                
                // 更新按钮状态
                document.getElementById('scene2-btn-prev').disabled = (this.currentStep === 0);
                document.getElementById('scene2-btn-next').disabled = (this.currentStep === this.totalSteps - 1);
            },
            
            // 绑定事件
            bindEvents: function() {
                const prevBtn = document.getElementById('scene2-btn-prev');
                const nextBtn = document.getElementById('scene2-btn-next');
                
                prevBtn.addEventListener('click', () => {
                    if (this.currentStep > 0) {
                        this.currentStep--;
                        this.updateScene();
                    }
                });
                
                nextBtn.addEventListener('click', () => {
                    if (this.currentStep < this.totalSteps - 1) {
                        this.currentStep++;
                        this.updateScene();
                    }
                });
            }
        };
        
        // 启动场景
        scene2.init();
    })();
    </script>
