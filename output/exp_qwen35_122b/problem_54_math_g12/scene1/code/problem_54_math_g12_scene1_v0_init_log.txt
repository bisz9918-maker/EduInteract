    <script>
    (function() {
        // Scene 1: 长方体表面最短路径问题 - 初始条件展示
        // 布局坐标表（实现对照依据）：
        // | 元素 | 类型 | 位置/范围 | 大小 | 标签偏移方向 | 备注 |
        // |------|------|-----------|------|-------------|------|
        // | 长方体前下左 A | 顶点 | (150,380) | - | 左下 | 起点 |
        // | 长方体前下右 C | 顶点 | (350,380) | - | 右下 | |
        // | 长方体后下右 D | 顶点 | (420,350) | - | 右下 | |
        // | 长方体后下左 | 顶点 | (220,350) | - | 左下 | 隐藏点 |
        // | 长方体前上左 | 顶点 | (150,150) | - | 左上 | |
        // | 长方体前上右 | 顶点 | (350,150) | - | 右上 | |
        // | 长方体后上右 B | 顶点 | (420,120) | - | 右上 | |
        // | 长方体后上左 E | 顶点 | (220,120) | - | 左上 | |
        // | 点 F | 点 | (420,200) | 6 | - | 在棱 BD 上 |
        // | 标签 A | text | (135,400) | 20×16 | 左下 | 顶点标签 |
        // | 标签 C | text | (345,400) | 20×16 | 右下 | 顶点标签 |
        // | 标签 D | text | (445,365) | 20×16 | 右下 | 顶点标签（调整后） |
        // | 标签 F | text | (435,200) | 20×16 | 右侧 | 目标点标签 |
        // | 标签 B | text | (435,115) | 20×16 | 右上 | 顶点标签 |
        // | 标签 E | text | (240,140) | 20×16 | 左上 | 顶点标签 |
        // | 标注 AC=5cm | text | (250,410) | 60×16 | 下方 | 绿色 |
        // | 标注 CD=3cm | text | (370,385) | 60×16 | 下方 | 绿色（调整后） |
        // | 标注 DF=6cm | text | (440,275) | 60×16 | 右侧 | 绿色 |
        
        const scene1 = {
            currentStep: 0,
            totalSteps: 4,
            
            // 获取 DOM 元素
            getEl: function(id) {
                return document.getElementById(id);
            },
            
            // 显示元素（同时设置 display 和 opacity）
            show: function(el) {
                if (el) {
                    el.style.visibility = 'visible';
                    el.style.opacity = '1';
                }
            },
            
            // 隐藏元素
            hide: function(el) {
                if (el) {
                    el.style.visibility = 'hidden';
                    el.style.opacity = '0';
                }
            },
            
            // 更新步骤显示
            updateStep: function() {
                const stepIndicator = this.getEl('scene1-step-indicator');
                const prevBtn = this.getEl('scene1-prev-btn');
                const nextBtn = this.getEl('scene1-next-btn');
                
                // 更新步骤指示器
                stepIndicator.textContent = '步骤 ' + this.currentStep + '/' + this.totalSteps;
                
                // 更新按钮状态
                prevBtn.disabled = (this.currentStep === 0);
                nextBtn.disabled = (this.currentStep === this.totalSteps);
                
                // 根据当前步骤显示对应元素
                this.renderStep();
            },
            
            // 渲染当前步骤
            renderStep: function() {
                const step = this.currentStep;
                
                // 步骤 1: 长方体线框
                const visibleEdges = [
                    'scene1-棱-AC', 'scene1-棱-CD', 'scene1-棱-DF', 'scene1-棱-FB',
                    'scene1-棱-BB\'', 'scene1-棱-AA\'', 'scene1-棱-A\'B\'', 'scene1-棱-B\'E', 'scene1-棱-EA',
                    'scene1-棱-EE\'', 'scene1-棱-ED', 'scene1-棱-A\'E'
                ];
                
                // 步骤 2: 顶点标签
                const vertexLabels = [
                    'scene1-标签-A', 'scene1-标签-C', 'scene1-标签-D',
                    'scene1-标签-F', 'scene1-标签-B', 'scene1-标签-E'
                ];
                
                // 步骤 3: 已知长度标注
                const lengthLabels = [
                    'scene1-标注-AC', 'scene1-标注-CD', 'scene1-标注-DF'
                ];
                
                // 步骤 4: 顶点圆点和高亮外圈
                const vertexPoints = [
                    'scene1-点-A', 'scene1-点-C', 'scene1-点-D',
                    'scene1-点-F', 'scene1-点-B', 'scene1-点-E'
                ];
                const highlightCircles = ['scene1-高亮-A', 'scene1-高亮-F'];
                
                // 根据步骤显示元素
                if (step >= 1) {
                    visibleEdges.forEach(id => this.show(this.getEl(id)));
                } else {
                    visibleEdges.forEach(id => this.hide(this.getEl(id)));
                }
                
                if (step >= 2) {
                    vertexLabels.forEach(id => this.show(this.getEl(id)));
                } else {
                    vertexLabels.forEach(id => this.hide(this.getEl(id)));
                }
                
                if (step >= 3) {
                    lengthLabels.forEach(id => this.show(this.getEl(id)));
                } else {
                    lengthLabels.forEach(id => this.hide(this.getEl(id)));
                }
                
                if (step >= 4) {
                    vertexPoints.forEach(id => this.show(this.getEl(id)));
                    highlightCircles.forEach(id => this.show(this.getEl(id)));
                } else {
                    vertexPoints.forEach(id => this.hide(this.getEl(id)));
                    highlightCircles.forEach(id => this.hide(this.getEl(id)));
                }
            },
            
            // 下一步
            next: function() {
                if (this.currentStep < this.totalSteps) {
                    this.currentStep++;
                    this.updateStep();
                }
            },
            
            // 上一步
            prev: function() {
                if (this.currentStep > 0) {
                    this.currentStep--;
                    this.updateStep();
                }
            },
            
            // 初始化
            init: function() {
                const prevBtn = this.getEl('scene1-prev-btn');
                const nextBtn = this.getEl('scene1-next-btn');
                
                prevBtn.addEventListener('click', this.prev.bind(this));
                nextBtn.addEventListener('click', this.next.bind(this));
                
                // 初始渲染
                this.updateStep();
            }
        };
        
        // 启动场景
        scene1.init();
    })();
    </script>
