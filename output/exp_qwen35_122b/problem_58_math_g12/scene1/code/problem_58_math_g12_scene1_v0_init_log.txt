    <script>
    (function() {
        // Scene 1: 圆周角定理与圆面积计算
        // 布局坐标表：
        // | 元素 | 位置/范围 | 备注 |
        // |------|-----------|------|
        // | 圆 O | 圆心 (400,250)，半径 150 | 居中 |
        // | 点 O | (400,250) | 圆心 |
        // | 点 A | (530,175) | 圆周右侧偏上 |
        // | 点 B | (475,380) | 圆周下方偏右 |
        // | 点 C | (374,102) | 圆周上方偏左 |
        // | 弦 AB | (530,175)-(475,380) | - |
        // | 弦 BC | (475,380)-(374,102) | - |
        // | 弦 AC | (530,175)-(374,102) | - |
        // | 半径 OA | (400,250)-(530,175) | 步骤 1 出现 |
        // | 半径 OB | (400,250)-(475,380) | 步骤 1 出现 |
        
        const TOTAL_STEPS = 5;
        let currentStep = 0;
        
        // 获取 DOM 元素
        const prevBtn = document.getElementById('scene1-prev-btn');
        const nextBtn = document.getElementById('scene1-next-btn');
        
        // 步骤元素组
        const step1Group = document.getElementById('scene1-group-step1');
        const step1Labels = document.getElementById('scene1-group-step1-labels');
        const step2Group = document.getElementById('scene1-group-step2');
        const step2Labels = document.getElementById('scene1-group-step2-labels');
        const step3Group = document.getElementById('scene1-group-step3');
        const step4Group = document.getElementById('scene1-group-step4');
        
        // 公式文本元素
        const formulaText = document.getElementById('scene1-text-formula');
        const resultText = document.getElementById('scene1-text-result');
        
        // 更新场景显示
        function updateScene() {
            // 更新按钮状态
            prevBtn.disabled = (currentStep === 0);
            nextBtn.textContent = (currentStep === TOTAL_STEPS - 1) ? '完成' : '下一步 ▶';
            nextBtn.disabled = (currentStep === TOTAL_STEPS - 1);
            
            // 步骤 0：初始状态（已显示圆、点、弦、∠ACB 标签）
            // 步骤 1：显示半径 OA、OB + R 标签
            if (currentStep >= 1) {
                step1Group.style.display = 'block';
                step1Labels.style.display = 'block';
                setTimeout(() => {
                    step1Group.classList.add('fade-in');
                    step1Labels.classList.add('fade-in');
                }, 10);
            } else {
                step1Group.style.display = 'none';
                step1Labels.style.display = 'none';
                step1Group.classList.remove('fade-in');
                step1Labels.classList.remove('fade-in');
            }
            
            // 步骤 2：显示圆心角弧 + ∠AOB 标签 + 定理文本
            if (currentStep >= 2) {
                step2Group.style.display = 'block';
                step2Labels.style.display = 'block';
                setTimeout(() => {
                    step2Group.classList.add('fade-in');
                    step2Labels.classList.add('fade-in');
                }, 10);
            } else {
                step2Group.style.display = 'none';
                step2Labels.style.display = 'none';
                step2Group.classList.remove('fade-in');
                step2Labels.classList.remove('fade-in');
            }
            
            // 步骤 3：显示正弦定理公式
            if (currentStep >= 3) {
                step3Group.style.display = 'block';
                formulaText.textContent = '正弦定理：\\(\\frac{AB}{\\sin\\angle AOB} = 2R\\)';
                setTimeout(() => {
                    step3Group.classList.add('fade-in');
                }, 10);
                // 触发 MathJax 重新渲染
                if (window.MathJax) {
                    MathJax.typesetPromise();
                }
            } else {
                step3Group.style.display = 'none';
                step3Group.classList.remove('fade-in');
            }
            
            // 步骤 4：显示面积结果
            if (currentStep >= 4) {
                step4Group.style.display = 'block';
                formulaText.textContent = '代入：\\(\\frac{4}{\\sin 74°} = 2R \\Rightarrow R = \\frac{2}{\\sin 74°} \\approx 2.08\\)';
                resultText.textContent = '面积：\\(S = \\pi R^2 = \\frac{4\\pi}{\\sin^2 74°} \\approx 13.6\\)';
                setTimeout(() => {
                    step4Group.classList.add('fade-in');
                }, 10);
                // 触发 MathJax 重新渲染
                if (window.MathJax) {
                    MathJax.typesetPromise();
                }
            } else {
                step4Group.style.display = 'none';
                step4Group.classList.remove('fade-in');
            }
        }
        
        // 事件绑定
        prevBtn.addEventListener('click', function() {
            if (currentStep > 0) {
                currentStep--;
                updateScene();
            }
        });
        
        nextBtn.addEventListener('click', function() {
            if (currentStep < TOTAL_STEPS - 1) {
                currentStep++;
                updateScene();
            }
        });
        
        // 初始化
        updateScene();
    })();
    </script>
