        // 工具函数：显示元素
        function show(elId) {
            const el = document.getElementById(elId);
            if (el) {
                el.classList.remove('scene1-hidden');
                el.style.opacity = '0';
                requestAnimationFrame(() => {
                    el.style.opacity = '1';
                });
            }
        }

        // 工具函数：隐藏元素
        function hide(elId) {
            const el = document.getElementById(elId);
            if (el) {
                el.style.opacity = '0';
                setTimeout(() => {
                    el.classList.add('scene1-hidden');
                    el.style.opacity = '1';
                }, 300);
            }
        }

        // 更新场景显示
        function updateScene() {
            const config = stepConfig[currentStep];

            // 先隐藏所有元素
            config.hide.forEach(hide);

            // 再显示需要的元素
            config.show.forEach(show);

            // 更新按钮状态
            document.getElementById('scene1-btn-prev').disabled = (currentStep === 0);
            document.getElementById('scene1-btn-next').disabled = (currentStep === TOTAL_STEPS - 1);

            // 更新步骤指示器
            document.getElementById('scene1-step-indicator').textContent = `步骤 ${currentStep + 1}/${TOTAL_STEPS}`;
        }

        // 事件绑定
        document.getElementById('scene1-btn-prev').addEventListener('click', () => {
            if (currentStep > 0) {
                currentStep--;
                updateScene();
            }

            if (currentStep < TOTAL_STEPS - 1) {
                currentStep++;
                updateScene();
            }

        updateScene();
