    <script>
    (function() {
        // Scene 1 实现
        // 布局坐标表对照：
        // | 元素 | 位置 | 备注 |
        // | O(原点) | (400, 250) | 画布中心 |
        // | F(焦点) | (520, 250) | x 轴正半轴 |
        // | M(切点) | (455, 195) | 圆上第一象限 |
        // | P(交点) | (475, 145) | 双曲线右支上 |
        // | 圆半径 | 70px | 中心 (400, 250) |
        // | 双曲线顶点 | 左 (340,250), 右 (460,250) |
        
        const scene1 = {
            currentStep: 0,
            totalSteps: 5,
            
            // 元素可见性控制
            show: function(el) {
                if (el) {
                    el.style.display = '';
                    el.style.opacity = '1';
                }
            },
            hide: function(el) {
                if (el) {
                    el.style.opacity = '0';
                    setTimeout(() => {
                        if (el) el.style.display = 'none';
                    }, 300);
                }
            },
            
            // 更新步骤显示
            updateStep: function() {
                const step = this.currentStep;
                
                // 步骤 0: 初始状态 - 坐标系、双曲线、圆、原点 O
                // 步骤 1: 添加焦点 F
                // 步骤 2: 添加切线 l、切点 M、直角标记、线段 OM
                // 步骤 3: 添加交点 P、线段 PF、中点标记
                // 步骤 4: 结论步骤 - 高亮关键关系
                
                // 基础元素（始终显示）
                this.show(document.getElementById('scene1-axis-x'));
                this.show(document.getElementById('scene1-axis-y'));
                this.show(document.getElementById('scene1-arrow-x'));
                this.show(document.getElementById('scene1-arrow-y'));
                this.show(document.getElementById('scene1-label-x'));
                this.show(document.getElementById('scene1-label-y'));
                this.show(document.getElementById('scene1-label-O'));
                
                // 双曲线和圆（步骤 0 显示）
                if (step >= 0) {
                    this.show(document.getElementById('scene1-hyperbola-left'));
                    this.show(document.getElementById('scene1-hyperbola-right'));
                    this.show(document.getElementById('scene1-circle'));
                    this.show(document.getElementById('scene1-point-O'));
                }
                
                // 焦点 F（步骤 1 显示）
                if (step >= 1) {
                    this.show(document.getElementById('scene1-point-F'));
                    this.show(document.getElementById('scene1-label-F'));
                }
                
                // 切线 l、切点 M、直角标记、线段 OM（步骤 2 显示）
                if (step >= 2) {
                    this.show(document.getElementById('scene1-line-l'));
                    this.show(document.getElementById('scene1-point-M'));
                    this.show(document.getElementById('scene1-label-M'));
                    this.show(document.getElementById('scene1-segment-OM'));
                    this.show(document.getElementById('scene1-right-angle'));
                    this.show(document.getElementById('scene1-label-l'));
                }
                
                // 交点 P、线段 PF、中点标记（步骤 3 显示）
                if (step >= 3) {
                    this.show(document.getElementById('scene1-point-P'));
                    this.show(document.getElementById('scene1-label-P'));
                    this.show(document.getElementById('scene1-segment-PF'));
                    this.show(document.getElementById('scene1-midpoint-mark'));
                }
                
                // 图例（步骤 2 显示）
                if (step >= 2) {
                    this.show(document.getElementById('scene1-legend-hyperbola'));
                    this.show(document.getElementById('scene1-legend-circle'));
                    this.show(document.getElementById('scene1-legend-line'));
                }
                
                // 结论步骤高亮（步骤 4）
                if (step >= 4) {
                    const rightAngle = document.getElementById('scene1-right-angle');
                    const midpointMark = document.getElementById('scene1-midpoint-mark');
                    rightAngle.classList.add('scene1-highlight');
                    midpointMark.classList.add('scene1-highlight');
                } else {
                    const rightAngle = document.getElementById('scene1-right-angle');
                    const midpointMark = document.getElementById('scene1-midpoint-mark');
                    rightAngle.classList.remove('scene1-highlight');
                    midpointMark.classList.remove('scene1-highlight');
                }
                
                // 更新步骤指示器和按钮状态
                document.getElementById('scene1-step-indicator').textContent = '步骤 ' + (step + 1) + '/5';
                document.getElementById('scene1-btn-prev').disabled = (step === 0);
                document.getElementById('scene1-btn-next').disabled = (step === this.totalSteps - 1);
            },
            
            // 初始化
            init: function() {
                const btnPrev = document.getElementById('scene1-btn-prev');
                const btnNext = document.getElementById('scene1-btn-next');
                
                btnPrev.addEventListener('click', () => {
                    if (this.currentStep > 0) {
                        this.currentStep--;
                        this.updateStep();
                    }
                });
                
                btnNext.addEventListener('click', () => {
                    if (this.currentStep < this.totalSteps - 1) {
                        this.currentStep++;
                        this.updateStep();
                    }
                });
                
                // 初始显示步骤 0
                this.updateStep();
            }
        };
        
        // 页面加载完成后初始化
        if (document.readyState === 'loading') {
            document.addEventListener('DOMContentLoaded', () => scene1.init());
        } else {
            scene1.init();
        }
    })();
    </script>
