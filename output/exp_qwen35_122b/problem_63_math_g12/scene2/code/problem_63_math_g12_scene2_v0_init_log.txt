    <script>
    (function() {
        // Scene 2 实现
        // 坐标映射：原点 (400,280)，x 轴向右，y 轴向上，1 单位=60px
        
        const scene2 = {
            currentStep: 0,
            totalSteps: 6,
            
            // 元素显示/隐藏配置
            elements: {
                // 步骤 0: 基础△OMF
                step0: {
                    show: ['scene2-axis-x', 'scene2-axis-y', 'scene2-point-O', 'scene2-point-F', 'scene2-point-M',
                           'scene2-line-OM', 'scene2-line-MF', 'scene2-line-OF', 'scene2-right-angle', 'scene2-right-angle-text',
                           'scene2-label-O', 'scene2-label-F', 'scene2-label-M'],
                    hide: ['scene2-point-P', 'scene2-label-P', 'scene2-line-PM', 'scene2-line-OP',
                           'scene2-label-b', 'scene2-label-c', 'scene2-label-a',
                           'scene2-mark-pm-mf', 'scene2-mark-mf-pm', 'scene2-mark-op-of1', 'scene2-mark-op-of2',
                           'scene2-mark-of-op1', 'scene2-mark-of-op2',
                           'scene2-text-pythagoras', 'scene2-text-pf', 'scene2-text-op',
                           'scene2-conclusion']
                },
                // 步骤 1: 标注 b, c
                step1: {
                    show: ['scene2-label-b', 'scene2-label-c'],
                    hide: []
                },
                // 步骤 2: 标注 a，勾股定理
                step2: {
                    show: ['scene2-label-a', 'scene2-text-pythagoras'],
                    hide: []
                },
                // 步骤 3: 点 P，中点关系
                step3: {
                    show: ['scene2-point-P', 'scene2-label-P', 'scene2-line-PM', 'scene2-line-OP',
                           'scene2-mark-pm-mf', 'scene2-mark-mf-pm', 'scene2-text-pf'],
                    hide: []
                },
                // 步骤 4: OP=OF
                step4: {
                    show: ['scene2-mark-op-of1', 'scene2-mark-op-of2', 'scene2-mark-of-op1', 'scene2-mark-of-op2',
                           'scene2-text-op'],
                    hide: []
                },
                // 步骤 5: 结论
                step5: {
                    show: ['scene2-conclusion'],
                    hide: []
                }
            },
            
            // 工具函数：显示元素
            showElement: function(id) {
                const el = document.getElementById(id);
                if (el) {
                    el.style.display = 'block';
                    el.style.opacity = '0';
                    setTimeout(() => {
                        el.style.transition = 'opacity 0.3s';
                        el.style.opacity = '1';
                    }, 10);
                }
            },
            
            // 工具函数：隐藏元素
            hideElement: function(id) {
                const el = document.getElementById(id);
                if (el) {
                    el.style.transition = 'opacity 0.3s';
                    el.style.opacity = '0';
                    setTimeout(() => {
                        el.style.display = 'none';
                    }, 300);
                }
            },
            
            // 更新步骤
            updateStep: function() {
                const stepConfig = this.elements['step' + this.currentStep];
                if (!stepConfig) return;
                
                // 显示新元素
                stepConfig.show.forEach(id => this.showElement(id));
                
                // 更新步骤指示器
                document.getElementById('scene2-step-text').textContent = '步骤 ' + (this.currentStep + 1) + '/' + this.totalSteps;
                
                // 更新按钮状态
                document.getElementById('scene2-btn-prev').disabled = (this.currentStep === 0);
                document.getElementById('scene2-btn-next').disabled = (this.currentStep === this.totalSteps - 1);
                if (this.currentStep === this.totalSteps - 1) {
                    document.getElementById('scene2-btn-next').textContent = '完成';
                } else {
                    document.getElementById('scene2-btn-next').textContent = '下一步 ▶';
                }
            },
            
            // 初始化
            init: function() {
                // 初始显示步骤 0 的元素
                this.elements.step0.show.forEach(id => {
                    const el = document.getElementById(id);
                    if (el) {
                        el.style.display = 'block';
                        el.style.opacity = '1';
                    }
                });
                
                // 绑定按钮事件
                document.getElementById('scene2-btn-prev').addEventListener('click', () => {
                    if (this.currentStep > 0) {
                        this.currentStep--;
                        this.updateStep();
                    }
                });
                
                document.getElementById('scene2-btn-next').addEventListener('click', () => {
                    if (this.currentStep < this.totalSteps - 1) {
                        this.currentStep++;
                        this.updateStep();
                    }
                });
                
                // 更新初始状态
                this.updateStep();
            }
        };
        
        // 启动
        scene2.init();
    })();
    </script>
