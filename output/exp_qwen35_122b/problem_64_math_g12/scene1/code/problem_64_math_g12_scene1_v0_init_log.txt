    <script>
    (function() {
        // Scene 1 实现
        // 布局坐标表：
        // 底面 ABCD: A(350,320)-B(220,380)-C(350,400)-D(480,380)
        // 上底面 A₁B₁C₁D₁: A₁(350,140)-B₁(220,200)-C₁(350,220)-D₁(480,200)
        // 侧棱: AA₁(350,320)-(350,140) 虚线，BB₁(220,380)-(220,200) 实线，CC₁(350,400)-(350,220) 实线，DD₁(480,380)-(480,200) 实线
        // 对角线 AC: (350,320)-(350,400)
        // 对角线 BD: (220,380)-(480,380)
        // 对角线 B₁D₁: (220,200)-(480,200)
        
        const scene1 = {
            currentStep: 0,
            totalSteps: 5,
            
            // 元素 ID 映射
            elements: {
                baseFill: 'scene1-base-fill',
                topFill: 'scene1-top-fill',
                lineAc: 'scene1-line-ac',
                lineBd: 'scene1-line-bd',
                lineBd1: 'scene1-line-bd1',
                projB1B: 'scene1-proj-b1b',
                projD1D: 'scene1-proj-d1d',
                rightAngle: 'scene1-right-angle',
                labelAc: 'scene1-label-ac',
                labelBd: 'scene1-label-bd',
                labelBd1: 'scene1-label-bd1',
                explanation1: 'scene1-explanation-1',
                explanation2: 'scene1-explanation-2',
                explanation3: 'scene1-explanation-3',
                conclusion: 'scene1-conclusion'
            },
            
            // 步骤配置
            steps: [
                // Step 0: 基础结构
                {
                    baseFill: 'hidden',
                    topFill: 'hidden',
                    lineAc: 'hidden',
                    lineBd: 'hidden',
                    lineBd1: 'hidden',
                    projB1B: 'hidden',
                    projD1D: 'hidden',
                    rightAngle: 'hidden',
                    labelAc: 'hidden',
                    labelBd: 'hidden',
                    labelBd1: 'hidden',
                    explanation1: 'hidden',
                    explanation2: 'hidden',
                    explanation3: 'hidden',
                    conclusion: 'hidden'
                },
                // Step 1: 高亮 AC 和 B₁D₁
                {
                    baseFill: 'hidden',
                    topFill: 'hidden',
                    lineAc: 'visible',
                    lineBd: 'hidden',
                    lineBd1: 'visible',
                    projB1B: 'hidden',
                    projD1D: 'hidden',
                    rightAngle: 'hidden',
                    labelAc: 'visible',
                    labelBd: 'hidden',
                    labelBd1: 'visible',
                    explanation1: 'visible',
                    explanation2: 'hidden',
                    explanation3: 'hidden',
                    conclusion: 'hidden'
                },
                // Step 2: 高亮 BD，显示投影线
                {
                    baseFill: 'visible',
                    topFill: 'visible',
                    lineAc: 'visible',
                    lineBd: 'visible',
                    lineBd1: 'visible',
                    projB1B: 'visible',
                    projD1D: 'visible',
                    rightAngle: 'hidden',
                    labelAc: 'visible',
                    labelBd: 'visible',
                    labelBd1: 'visible',
                    explanation1: 'visible',
                    explanation2: 'visible',
                    explanation3: 'hidden',
                    conclusion: 'hidden'
                },
                // Step 3: 说明空间垂直转化为平面垂直
                {
                    baseFill: 'visible',
                    topFill: 'visible',
                    lineAc: 'visible',
                    lineBd: 'visible',
                    lineBd1: 'visible',
                    projB1B: 'visible',
                    projD1D: 'visible',
                    rightAngle: 'visible',
                    labelAc: 'visible',
                    labelBd: 'visible',
                    labelBd1: 'visible',
                    explanation1: 'visible',
                    explanation2: 'visible',
                    explanation3: 'visible',
                    conclusion: 'hidden'
                },
                // Step 4: 结论
                {
                    baseFill: 'visible',
                    topFill: 'visible',
                    lineAc: 'visible',
                    lineBd: 'visible',
                    lineBd1: 'visible',
                    projB1B: 'visible',
                    projD1D: 'visible',
                    rightAngle: 'visible',
                    labelAc: 'visible',
                    labelBd: 'visible',
                    labelBd1: 'visible',
                    explanation1: 'visible',
                    explanation2: 'visible',
                    explanation3: 'visible',
                    conclusion: 'visible'
                }
            ],
            
            // 初始化
            init: function() {
                this.updateScene();
                this.bindEvents();
            },
            
            // 更新场景显示
            updateScene: function() {
                const step = this.steps[this.currentStep];
                
                // 更新所有元素显示状态
                for (const key in this.elements) {
                    const elementId = this.elements[key];
                    const element = document.getElementById(elementId);
                    if (element) {
                        if (step[key] === 'visible') {
                            element.classList.remove('scene1-hidden');
                        } else {
                            element.classList.add('scene1-hidden');
                        }
                    }
                }
                
                // 更新按钮状态
                const prevBtn = document.getElementById('scene1-btn-prev');
                const nextBtn = document.getElementById('scene1-btn-next');
                
                prevBtn.disabled = (this.currentStep === 0);
                
                if (this.currentStep === this.totalSteps - 1) {
                    nextBtn.textContent = '完成';
                    nextBtn.disabled = true;
                } else {
                    nextBtn.textContent = '下一步 ▶';
                    nextBtn.disabled = false;
                }
            },
            
            // 绑定事件
            bindEvents: function() {
                const prevBtn = document.getElementById('scene1-btn-prev');
                const nextBtn = document.getElementById('scene1-btn-next');
                
                prevBtn.addEventListener('click', () => {
                    if (this.currentStep > 0) {
                        this.currentStep--;
                        this.updateScene();
                    }
                });
                
                nextBtn.addEventListener('click', () => {
                    if (this.currentStep < this.totalSteps - 1) {
                        this.currentStep++;
                        this.updateScene();
                    }
                });
            }
        };
        
        // 启动场景
        scene1.init();
    })();
    </script>
