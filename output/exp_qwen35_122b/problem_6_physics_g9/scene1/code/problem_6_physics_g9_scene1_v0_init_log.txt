    <script>
    (function() {
        // Scene 1 实现 - 电磁铁电路装置
        // 布局坐标表对照（已调整）：
        // | 元素 | 类型 | 位置/范围 | 大小 | 标签偏移方向 | 备注 |
        // | 电路图区域 | 背景框 | (20,60)-(450,420) | 430×360 | - | 左侧 60% 区域 |
        // | 右侧装置区域 | 背景框 | (480,60)-(780,420) | 300×360 | - | 右侧 40% 区域 |
        // | 电源 | 电池符号 | (220,420)-(260,440) | 40×20 | 下方 | 长线正极在左 |
        // | 开关 | 开关符号 | (280,420)-(320,440) | 40×20 | 下方 | 闭合状态 |
        // | 滑动变阻器 | 矩形 + 箭头 | (80,415)-(180,425) | 100×10 | 下方 | 滑片箭头向上 |
        // | 电流表 | 圆圈 +A | (90,330) | 30×30 | 左侧 | 圆圈内"A" |
        // | 定值电阻 | 矩形框 | (100,120)-(200,150) | 100×30 | 上方 | 水平放置 |
        // | 电磁铁线圈 | 螺线管 | (380,320)-(430,410) | 50×90 | 右侧 | 竖直绕线 |
        // | 弹簧测力计外壳 | 矩形 | (580,90)-(620,180) | 40×90 | 上方 | 带刻度 |
        // | 弹簧测力计弹簧 | 锯齿线 | (600,90)-(600,160) | 4×70 | - | 内部弹簧 |
        // | 挂钩 | 曲线 | (600,160)-(600,190) | 4×30 | - | 向下延伸 |
        // | 铁球 | 圆形 | (600,210) | 30×30 | 右侧 | 标注"铁球" |
        
        const currentStepMax = 5;
        let currentStep = 0;
        
        // 元素显示配置：每个步骤新增的元素
        const stepElements = {
            1: [
                'source', 'switch', 'ammeter',
                'wire1', 'wire3', 'wire4', 'wire4b', 'wire5',
                'label-source', 'label-switch', 'label-ammeter'
            ],
            2: [
                'rheostat', 'resistor',
                'wire2', 'wire6',
                'label-rheostat', 'label-resistor'
            ],
            3: [
                'electromagnet',
                'wire7', 'wire8', 'wire9', 'wire9b', 'wire9c',
                'label-electromagnet'
            ],
            4: [
                'hanging-point', 'spring-scale', 'spring', 'hook', 'ball',
                'label-spring', 'label-ball'
            ],
            5: [
                'current-arrow1', 'current-arrow2', 'current-arrow3',
                'current-arrow4', 'current-arrow5', 'current-arrow6', 'current-arrow7'
            ]
        };
        
        // 已激活元件（蓝高亮）：步骤 2 后电源、开关、电流表变蓝；步骤 3 后加变阻器；步骤 4 后加电阻；步骤 5 后加电磁铁
        const blueHighlightSteps = {
            'source': 2,
            'switch': 2,
            'ammeter': 2,
            'rheostat': 3,
            'resistor': 4,
            'electromagnet': 5
        };
        
        function getElement(id) {
            return document.getElementById('scene1-' + id);
        }
        
        function showElement(id) {
            const el = getElement(id);
            if (el) {
                el.style.opacity = '1';
                el.style.display = 'block';
            }
        }
        
        function hideElement(id) {
            const el = getElement(id);
            if (el) {
                el.style.opacity = '0';
                if (el.tagName === 'line' || el.tagName === 'polyline' || el.tagName === 'path' || el.tagName === 'rect' || el.tagName === 'circle') {
                    el.style.display = 'none';
                }
            }
        }
        
        function addBlueHighlight(id) {
            const el = getElement(id);
            if (el) {
                // 为组内所有图形元素添加蓝高亮
                const elements = el.querySelectorAll('rect, circle, line, polyline, path');
                elements.forEach(elem => {
                    elem.classList.add('scene1-blue-highlight');
                });
            }
        }
        
        function removeBlueHighlight(id) {
            const el = getElement(id);
            if (el) {
                const elements = el.querySelectorAll('.scene1-blue-highlight');
                elements.forEach(elem => {
                    elem.classList.remove('scene1-blue-highlight');
                });
            }
        }
        
        function updateScene() {
            // 重置所有元素
            const allElements = document.querySelectorAll('.scene1-element, .scene1-wire, .scene1-label, .scene1-current');
            allElements.forEach(el => {
                el.style.opacity = '0';
                if (el.tagName !== 'text') {
                    el.style.display = 'none';
                }
            });
            
            // 移除所有蓝高亮
            const allBlue = document.querySelectorAll('.scene1-blue-highlight');
            allBlue.forEach(el => el.classList.remove('scene1-blue-highlight'));
            
            // 显示当前步骤及之前步骤的元素
            for (let s = 1; s <= currentStep; s++) {
                if (stepElements[s]) {
                    stepElements[s].forEach(id => {
                        showElement(id);
                    });
                }
            }
            
            // 添加蓝高亮（根据当前步骤）
            for (const [id, step] of Object.entries(blueHighlightSteps)) {
                if (currentStep >= step) {
                    addBlueHighlight(id);
                }
            }
            
            // 更新按钮状态
            const prevBtn = document.getElementById('scene1-btn-prev');
            const nextBtn = document.getElementById('scene1-btn-next');
            prevBtn.disabled = (currentStep === 0);
            nextBtn.disabled = (currentStep === currentStepMax);
            
            // 更新步骤指示器
            document.getElementById('scene1-step-indicator').textContent = '步骤 ' + currentStep + '/' + currentStepMax;
        }
        
        // 事件绑定
        document.getElementById('scene1-btn-prev').addEventListener('click', function() {
            if (currentStep > 0) {
                currentStep--;
                updateScene();
            }
        });
        
        document.getElementById('scene1-btn-next').addEventListener('click', function() {
            if (currentStep < currentStepMax) {
                currentStep++;
                updateScene();
            }
        });
        
        // 初始化
        updateScene();
    })();
    </script>
