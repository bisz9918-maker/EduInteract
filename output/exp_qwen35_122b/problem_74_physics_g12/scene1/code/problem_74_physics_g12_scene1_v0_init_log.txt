    <script>
    (function() {
        // Scene 1 实现
        // 布局坐标表对照：
        // | 元素 | 类型 | 位置/范围 | 大小 | 标签偏移方向 | 备注 |
        // | 顶部支架 | 线段 | (200,40)-(600,40) | 400×3 | - | 带斜线阴影表示固定 |
        // | 左滑轮 | 圆 | (300,70) | r=25 | - | 左侧滑轮 |
        // | 右滑轮 | 圆 | (500,70) | r=25 | - | 右侧滑轮 |
        // | 左绳段 1 | 线段 | (300,95)-(300,220) | 4×125 | - | 左滑轮到线圈 |
        // | 右绳段 1 | 线段 | (500,95)-(500,150) | 4×55 | - | 右滑轮到砝码 |
        // | 磁场区域 | 矩形 | (220,120)-(480,200) | 260×80 | - | 虚线边界，内部×符号 |
        // | 线圈 abcd | 矩形 | (260,200)-(440,380) | 180×180 | - | 正方形，蓝色边框 |
        // | 砝码 m₂ | 矩形 | (470,150)-(530,210) | 60×60 | - | 红色填充 |
        
        const scene1 = {
            currentStep: 0,
            totalSteps: 5,
            elements: {}
        };
        
        // 获取 DOM 元素
        function initElements() {
            scene1.elements = {
                svg: document.getElementById('scene1-svg'),
                prevBtn: document.getElementById('scene1-btn-prev'),
                nextBtn: document.getElementById('scene1-btn-next'),
                stepIndicator: document.getElementById('scene1-step-indicator'),
                bracketLine: document.getElementById('scene1-bracket-line'),
                pulleyLeft: document.getElementById('scene1-pulley-left'),
                pulleyRight: document.getElementById('scene1-pulley-right'),
                ropeLeft: document.getElementById('scene1-rope-left'),
                ropeRight: document.getElementById('scene1-rope-right'),
                magnetRect: document.getElementById('scene1-magnet-rect'),
                magnetCrosses: document.getElementById('scene1-magnet-crosses'),
                coilRect: document.getElementById('scene1-coil-rect'),
                weightRect: document.getElementById('scene1-weight-rect'),
                hLine: document.getElementById('scene1-h-line'),
                hArrowTop: document.getElementById('scene1-h-arrow-top'),
                hArrowBottom: document.getElementById('scene1-h-arrow-bottom'),
                lLine: document.getElementById('scene1-l-line'),
                lArrowTop: document.getElementById('scene1-l-arrow-top'),
                lArrowBottom: document.getElementById('scene1-l-arrow-bottom'),
                coilLabels: document.getElementById('scene1-coil-labels'),
                weightLabels: document.getElementById('scene1-weight-labels'),
                magnetLabels: document.getElementById('scene1-magnet-labels'),
                processLabels: document.getElementById('scene1-process-labels')
            };
        }
        
        // 显示元素（同时设置 display 和 opacity）
        function show(el) {
            if (el) {
                el.style.display = 'block';
                el.style.opacity = '1';
            }
        }
        
        // 隐藏元素（同时设置 display 和 opacity）
        function hide(el) {
            if (el) {
                el.style.display = 'none';
                el.style.opacity = '0';
            }
        }
        
        // 更新步骤显示
        function updateStep() {
            const el = scene1.elements;
            
            // 重置所有标签
            hide(el.coilLabels);
            hide(el.weightLabels);
            hide(el.magnetLabels);
            hide(el.processLabels);
            hide(el.magnetCrosses);
            
            // 重置线圈和砝码样式
            el.coilRect.setAttribute('stroke', '#333');
            el.coilRect.setAttribute('stroke-width', '2');
            el.weightRect.setAttribute('fill-opacity', '0');
            
            // 根据步骤显示对应元素
            switch(scene1.currentStep) {
                case 0:
                    // 初始状态：仅显示基础框架
                    break;
                case 1:
                    // 高亮线圈
                    el.coilRect.setAttribute('stroke', '#1565c0');
                    el.coilRect.setAttribute('stroke-width', '3');
                    show(el.coilLabels);
                    break;
                case 2:
                    // 高亮砝码
                    el.weightRect.setAttribute('fill-opacity', '0.2');
                    show(el.weightLabels);
                    break;
                case 3:
                    // 高亮磁场区域
                    show(el.magnetCrosses);
                    show(el.magnetLabels);
                    break;
                case 4:
                    // 标注物理过程
                    show(el.processLabels);
                    break;
            }
            
            // 更新按钮状态
            el.prevBtn.disabled = (scene1.currentStep === 0);
            el.nextBtn.disabled = (scene1.currentStep === scene1.totalSteps - 1);
            
            // 更新步骤指示器
            el.stepIndicator.textContent = '步骤 ' + (scene1.currentStep + 1) + '/' + scene1.totalSteps;
        }
        
        // 上一步
        function prevStep() {
            if (scene1.currentStep > 0) {
                scene1.currentStep--;
                updateStep();
            }
        }
        
        // 下一步
        function nextStep() {
            if (scene1.currentStep < scene1.totalSteps - 1) {
                scene1.currentStep++;
                updateStep();
            }
        }
        
        // 初始化
        function init() {
            initElements();
            
            // 绑定事件
            scene1.elements.prevBtn.addEventListener('click', prevStep);
            scene1.elements.nextBtn.addEventListener('click', nextStep);
            
            // 初始显示
            updateStep();
        }
        
        // 页面加载完成后初始化
        if (document.readyState === 'loading') {
            document.addEventListener('DOMContentLoaded', init);
        } else {
            init();
        }
    })();
    </script>
