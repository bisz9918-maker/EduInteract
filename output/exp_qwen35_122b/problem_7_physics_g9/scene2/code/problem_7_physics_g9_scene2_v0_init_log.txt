    <script>
    (function() {
        // Scene 2 实现
        // 布局坐标表对照：
        // 乙物块：(180,150)-(230,220) | 50×70
        // 丙物块：(530,150)-(580,200) | 50×50
        // G_乙箭头：(205,185)-(205,260) 红色向下
        // F_浮乙箭头：(205,185)-(205,115) 蓝色向上
        // T 箭头：(205,185)-(205,115) 绿色向上
        // G_丙箭头：(555,175)-(555,250) 红色向下
        // N 箭头：(555,175)-(555,105) 黑色向上

        const TOTAL_STEPS = 4;
        let currentStep = 0;

        // 获取 DOM 元素
        const elements = {
            prevBtn: document.getElementById('scene2-prevBtn'),
            nextBtn: document.getElementById('scene2-nextBtn'),
            // 乙的受力
            arrowGyi: document.getElementById('scene2-arrow-Gyi'),
            labelGyi: document.getElementById('scene2-label-Gyi'),
            arrowFbuoyYi: document.getElementById('scene2-arrow-Fbuoy-yi'),
            labelFbuoyYi: document.getElementById('scene2-label-Fbuoy-yi'),
            arrowT: document.getElementById('scene2-arrow-T'),
            labelT: document.getElementById('scene2-label-T'),
            // 丙的受力
            arrowGbing: document.getElementById('scene2-arrow-Gbing'),
            labelGbing: document.getElementById('scene2-label-Gbing'),
            arrowN: document.getElementById('scene2-arrow-N'),
            labelN: document.getElementById('scene2-label-N'),
            // 公式
            formula1: document.getElementById('scene2-formula1'),
            formula2: document.getElementById('scene2-formula2'),
            formula3: document.getElementById('scene2-formula3'),
            // 密度结论
            density: document.getElementById('scene2-density'),
            // 结论
            conclusion1: document.getElementById('scene2-conclusion1'),
            conclusion2: document.getElementById('scene2-conclusion2'),
            // 选项 C 标注
            optionCText: document.getElementById('scene2-option-c-text'),
            // 答案
            answer: document.getElementById('scene2-answer')
        };

        // 显示元素
        function show(el) {
            if (el) {
                el.style.opacity = '1';
                el.style.display = 'block';
                el.classList.add('scene2-fade-in');
            }
        }

        // 隐藏元素
        function hide(el) {
            if (el) {
                el.style.opacity = '0';
                el.style.display = 'none';
                el.classList.remove('scene2-fade-in');
            }
        }

        // 更新步骤显示状态
        function updateStep() {
            // 步骤 0：初始状态 - 只显示基础重力
            // 步骤 1：显示乙的浮力和拉力
            // 步骤 2：显示密度结论和选项 C 错误
            // 步骤 3：显示质量关系和最终答案

            // 重置所有动态元素
            hide(elements.arrowFbuoyYi);
            hide(elements.labelFbuoyYi);
            hide(elements.arrowT);
            hide(elements.labelT);
            hide(elements.formula1);
            hide(elements.formula2);
            hide(elements.formula3);
            hide(elements.density);
            hide(elements.conclusion1);
            hide(elements.conclusion2);
            hide(elements.optionCText);
            hide(elements.answer);
            hide(elements.arrowN);
            hide(elements.labelN);

            if (currentStep >= 1) {
                show(elements.arrowFbuoyYi);
                show(elements.labelFbuoyYi);
                show(elements.arrowT);
                show(elements.labelT);
                show(elements.formula1);
            }

            if (currentStep >= 2) {
                show(elements.density);
                show(elements.conclusion1);
                show(elements.optionCText);
            }

            if (currentStep >= 3) {
                show(elements.formula2);
                show(elements.formula3);
                show(elements.conclusion2);
                show(elements.answer);
                show(elements.arrowN);
                show(elements.labelN);
            }

            // 更新按钮状态
            elements.prevBtn.disabled = (currentStep === 0);
            elements.nextBtn.disabled = (currentStep === TOTAL_STEPS - 1);
        }

        // 上一步按钮点击
        elements.prevBtn.addEventListener('click', function() {
            if (currentStep > 0) {
                currentStep--;
                updateStep();
            }
        });

        // 下一步按钮点击
        elements.nextBtn.addEventListener('click', function() {
            if (currentStep < TOTAL_STEPS - 1) {
                currentStep++;
                updateStep();
            }
        });

        // 初始化
        updateStep();
    })();
    </script>
