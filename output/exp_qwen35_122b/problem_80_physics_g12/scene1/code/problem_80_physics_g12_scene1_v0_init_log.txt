    <script>
    (function() {
        // Scene 1: 单摆摆动 + 平抛运动全过程
        // 布局坐标表对照：
        // | 元素 | 类型 | 位置/范围 | 大小 | 标签偏移方向 | 备注 |
        // |------|------|-----------|------|-------------|------|
        // | 岸 | 垂直线 | (50,200)-(50,400) | 2×200 | - | 左侧边界 |
        // | 水面 | 水平线 | (50,400)-(750,400) | 700×2 | - | y=400 |
        // | 浮台 | 矩形 | (550,395)-(720,405) | 170×10 | 上方 | 水面上的黑色矩形 |
        // | 悬挂点 O | 实心圆 | (300,80) | 8 | 右上方 (310,70) | 固定点 |
        // | 初始绳 | 线段 | (300,80)-(166,181) | - | - | 蓝色，初始位置 |
        // | 最低点绳 | 线段 | (300,80)-(300,248) | - | - | 黑色，最低点位置 |
        // | 初始质点 m | 实心圆 | (166,181) | 12 | 左下方 (140,190) | 蓝色 |
        // | 最低点质点 | 实心圆 | (300,248) | 12 | 右下方 (320,260) | 红色 |
        // | 摆动轨迹 | 圆弧 | 圆心 (300,80)，半径 168，角度 -53°到 0° | - | - | 蓝色虚线，透明度 30% |
        // | 平抛轨迹 | 抛物线 | (300,248) 到 (650,400) | - | - | 绿色虚线 |
        // | 速度矢量 v | 箭头 | (300,248)-(380,248) | 80×4 | 上方 (340,235) | 红色 |
        // | 角度α弧线 | 圆弧 | 圆心 (300,80)，半径 40，角度 -53°到 0° | - | 右侧 (340,120) | 蓝色虚线 |
        // | 绳长 l 标注线 | 垂直线 | (160,80)-(160,248) | 2×168 | 左侧 (130,164) | 灰色虚线（已左移避免重叠） |
        // | 悬挂高度 H 标注线 | 垂直线 | (720,80)-(720,400) | 2×320 | 右侧 (730,240) | 灰色虚线（已左移避免越界） |
        // | 平抛高度 h₂标注线 | 垂直线 | (720,248)-(720,400) | 2×152 | 右侧 (730,324) | 灰色虚线（已左移避免越界） |
        // | 水平位移 d 标注线 | 水平线 | (300,420)-(650,420) | 350×2 | 下方 (475,435) | 灰色虚线 |
        
        // 状态变量
        var currentStep = 0;
        var totalSteps = 4;
        
        // 工具函数：显示元素
        function show(el) {
            if (el) {
                el.style.display = 'block';
                el.style.opacity = '1';
            }
        }
        
        // 工具函数：隐藏元素
        function hide(el) {
            if (el) {
                el.style.display = 'none';
                el.style.opacity = '0';
            }
        }
        
        // 更新场景显示
        function updateScene() {
            // 步骤 0（初始状态）：岸、水面、浮台、悬挂点 O、初始绳、初始质点 m、角度α弧线
            // 步骤 1（摆动过程）：新增摆动轨迹弧线、h₁标注、重力势能→动能文字
            // 步骤 2（最低点受力）：新增最低点绳、最低点质点、速度矢量 v、受力公式文字
            // 步骤 3（平抛运动）：新增平抛轨迹抛物线、h₂标注、d 标注、水平匀速 + 竖直自由落体文字
            
            // 步骤 0 基础元素（始终显示）
            show(document.getElementById('scene1-shore'));
            show(document.getElementById('scene1-water'));
            show(document.getElementById('scene1-platform'));
            show(document.getElementById('scene1-platform-label'));
            show(document.getElementById('scene1-pivot'));
            show(document.getElementById('scene1-pivot-label'));
            show(document.getElementById('scene1-rope-init'));
            show(document.getElementById('scene1-mass-init'));
            show(document.getElementById('scene1-mass-init-label'));
            show(document.getElementById('scene1-angle-arc'));
            show(document.getElementById('scene1-angle-label'));
            show(document.getElementById('scene1-l-line'));
            show(document.getElementById('scene1-l-label'));
            show(document.getElementById('scene1-H-line'));
            show(document.getElementById('scene1-H-label'));
            
            // 步骤 1 新增元素
            if (currentStep >= 1) {
                show(document.getElementById('scene1-swing-path'));
                show(document.getElementById('scene1-h1-label'));
                show(document.getElementById('scene1-energy-text'));
            } else {
                hide(document.getElementById('scene1-swing-path'));
                hide(document.getElementById('scene1-h1-label'));
                hide(document.getElementById('scene1-energy-text'));
            }
            
            // 步骤 2 新增元素
            if (currentStep >= 2) {
                show(document.getElementById('scene1-rope-low'));
                show(document.getElementById('scene1-mass-low'));
                show(document.getElementById('scene1-velocity-line'));
                show(document.getElementById('scene1-velocity-arrow'));
                show(document.getElementById('scene1-velocity-label'));
                show(document.getElementById('scene1-force-text'));
            } else {
                hide(document.getElementById('scene1-rope-low'));
                hide(document.getElementById('scene1-mass-low'));
                hide(document.getElementById('scene1-velocity-line'));
                hide(document.getElementById('scene1-velocity-arrow'));
                hide(document.getElementById('scene1-velocity-label'));
                hide(document.getElementById('scene1-force-text'));
            }
            
            // 步骤 3 新增元素
            if (currentStep >= 3) {
                show(document.getElementById('scene1-projectile-path'));
                show(document.getElementById('scene1-h2-line'));
                show(document.getElementById('scene1-h2-label'));
                show(document.getElementById('scene1-d-line'));
                show(document.getElementById('scene1-d-label'));
                show(document.getElementById('scene1-projectile-text'));
            } else {
                hide(document.getElementById('scene1-projectile-path'));
                hide(document.getElementById('scene1-h2-line'));
                hide(document.getElementById('scene1-h2-label'));
                hide(document.getElementById('scene1-d-line'));
                hide(document.getElementById('scene1-d-label'));
                hide(document.getElementById('scene1-projectile-text'));
            }
            
            // 更新步骤指示器
            document.getElementById('scene1-step-indicator').textContent = '步骤 ' + (currentStep + 1) + '/' + totalSteps;
            
            // 更新按钮状态
            document.getElementById('scene1-btn-prev').disabled = (currentStep === 0);
            document.getElementById('scene1-btn-next').disabled = (currentStep === totalSteps - 1);
        }
        
        // 事件绑定
        document.getElementById('scene1-btn-prev').addEventListener('click', function() {
            if (currentStep > 0) {
                currentStep--;
                updateScene();
            }
        });
        
        document.getElementById('scene1-btn-next').addEventListener('click', function() {
            if (currentStep < totalSteps - 1) {
                currentStep++;
                updateScene();
            }
        });
        
        // 初始化场景
        updateScene();
    })();
    </script>
