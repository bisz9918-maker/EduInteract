    <script>
    (function() {
        // Scene 2 实现 - 悬线断开后的运动过程与时间关联
        
        // 配置数据
        const config = {
            // 物体 A 位置（矩形左上角）
            aPositions: {
                initial: { x: 150, y: 150 },
                equilibrium: { x: 200, y: 150 },
                compressed: { x: 30, y: 150 }
            },
            // 物体 B 位置（矩形左上角）
            bPositions: {
                initial: { x: 630, y: 100 },
                middle: { x: 630, y: 200 },
                ground: { x: 630, y: 310 }
            },
            // 弹簧右端点 x 坐标（左端固定在 x=20）
            springEndX: {
                initial: 150,
                equilibrium: 200,
                compressed: 80
            },
            // 步骤定义
            steps: [
                {
                    name: "步骤 1/4",
                    description: "悬线断开瞬间",
                    aPos: 'initial',
                    bPos: 'initial',
                    springEnd: 'initial',
                    showInitialLine: true,
                    showEquilibriumLine: false,
                    showCompressedLine: false,
                    showAmpArrow: false,
                    showHeightArrow: true,
                    highlightTime: 0,
                    showT1Label: false,
                    showT2Label: false,
                    showSummary: false,
                    showSyncArrow: false
                },
                {
                    name: "步骤 2/4",
                    description: "A 运动至平衡位置，B 下落",
                    aPos: 'equilibrium',
                    bPos: 'middle',
                    springEnd: 'equilibrium',
                    showInitialLine: false,
                    showEquilibriumLine: true,
                    showCompressedLine: false,
                    showAmpArrow: false,
                    showHeightArrow: true,
                    highlightTime: 1,
                    showT1Label: true,
                    showT2Label: false,
                    showSummary: false,
                    showSyncArrow: false
                },
                {
                    name: "步骤 3/4",
                    description: "A 压缩弹簧到最短，B 落地",
                    aPos: 'compressed',
                    bPos: 'ground',
                    springEnd: 'compressed',
                    showInitialLine: false,
                    showEquilibriumLine: false,
                    showCompressedLine: true,
                    showAmpArrow: true,
                    showHeightArrow: true,
                    highlightTime: 2,
                    showT1Label: true,
                    showT2Label: true,
                    showSummary: false,
                    showSyncArrow: false
                },
                {
                    name: "步骤 4/4",
                    description: "时间关系总结",
                    aPos: 'compressed',
                    bPos: 'ground',
                    springEnd: 'compressed',
                    showInitialLine: true,
                    showEquilibriumLine: true,
                    showCompressedLine: true,
                    showAmpArrow: true,
                    showHeightArrow: true,
                    highlightTime: 2,
                    showT1Label: true,
                    showT2Label: true,
                    showSummary: true,
                    showSyncArrow: true
                }
            ]
        };
        
        // 当前步骤
        let currentStep = 0;
        
        // 获取 DOM 元素
        const svg = document.getElementById('scene2-svg');
        const objA = document.getElementById('scene2-obj-a');
        const objB = document.getElementById('scene2-obj-b');
        const spring = document.getElementById('scene2-spring');
        const btnPrev = document.getElementById('scene2-btn-prev');
        const btnNext = document.getElementById('scene2-btn-next');
        const stepIndicator = document.getElementById('scene2-step-indicator');
        
        // 位置线元素
        const lineInitial = document.getElementById('scene2-line-initial');
        const lineEquilibrium = document.getElementById('scene2-line-equilibrium');
        const lineCompressed = document.getElementById('scene2-line-compressed');
        
        // 标签元素
        const labelInitial = document.getElementById('scene2-label-initial');
        const labelEquilibrium = document.getElementById('scene2-label-equilibrium');
        const labelCompressed = document.getElementById('scene2-label-compressed');
        const labelAmplitude = document.getElementById('scene2-label-amplitude');
        const labelH = document.getElementById('scene2-label-h');
        const labelT1 = document.getElementById('scene2-label-t1');
        const labelT2 = document.getElementById('scene2-label-t2');
        
        // 振幅箭头
        const ampArrow = document.getElementById('scene2-amp-arrow');
        const ampArrowLeft = document.getElementById('scene2-amp-arrow-left');
        const ampArrowLeft2 = document.getElementById('scene2-amp-arrow-left2');
        const ampArrowRight = document.getElementById('scene2-amp-arrow-right');
        const ampArrowRight2 = document.getElementById('scene2-amp-arrow-right2');
        
        // 高度箭头
        const heightArrow = document.getElementById('scene2-height-arrow');
        const heightArrowTop = document.getElementById('scene2-height-arrow-top');
        const heightArrowBottom = document.getElementById('scene2-height-arrow-bottom');
        
        // 时间标记
        const timeDot0 = document.getElementById('scene2-time-dot-0');
        const timeDot1 = document.getElementById('scene2-time-dot-1');
        const timeDot2 = document.getElementById('scene2-time-dot-2');
        
        // 总结文字和同步箭头
        const summaryText = document.getElementById('scene2-summary-text');
        const syncArrow = document.getElementById('scene2-sync-arrow');
        
        // 绘制弹簧（锯齿线）
        function drawSpring(startX, endX, y, segments = 8) {
            const height = 4;
            const segmentWidth = (endX - startX) / segments;
            let path = `M ${startX} ${y - height/2}`;
            
            for (let i = 1; i <= segments; i++) {
                const x = startX + i * segmentWidth;
                if (i === segments) {
                    path += ` L ${x} ${y - height/2}`;
                } else {
                    path += ` L ${x} ${y + (i % 2 === 0 ? -height/2 : height/2)}`;
                }
            }
            
            return path;
        }
        
        // 更新弹簧显示
        function updateSpring(endX) {
            const startX = 20;
            const y = 160;
            spring.setAttribute('d', drawSpring(startX, endX, y));
        }
        
        // 显示/隐藏元素工具函数
        function show(el) {
            el.style.display = 'block';
            el.style.opacity = '1';
        }
        
        function hide(el) {
            el.style.display = 'none';
            el.style.opacity = '0';
        }
        
        function showSvg(el) {
            el.setAttribute('opacity', '1');
        }
        
        function hideSvg(el) {
            el.setAttribute('opacity', '0');
        }
        
        // 更新场景状态
        function updateScene(stepIndex, animate = true) {
            const step = config.steps[stepIndex];
            
            if (animate) {
                // 动画过渡
                animateTransition(stepIndex);
            } else {
                // 直接更新
                applyStep(step);
            }
            
            // 更新按钮状态
            btnPrev.disabled = stepIndex === 0;
            btnNext.disabled = stepIndex === config.steps.length - 1;
            
            // 更新步骤指示器
            stepIndicator.textContent = step.name;
        }
        
        // 应用步骤配置
        function applyStep(step) {
            // 更新物体 A 位置
            const aPos = config.aPositions[step.aPos];
            objA.setAttribute('x', aPos.x);
            objA.setAttribute('y', aPos.y);
            
            // 更新物体 B 位置
            const bPos = config.bPositions[step.bPos];
            objB.setAttribute('x', bPos.x);
            objB.setAttribute('y', bPos.y);
            
            // 更新弹簧
            updateSpring(config.springEndX[step.springEnd]);
            
            // 更新位置线
            step.showInitialLine ? showSvg(lineInitial) : hideSvg(lineInitial);
            step.showEquilibriumLine ? showSvg(lineEquilibrium) : hideSvg(lineEquilibrium);
            step.showCompressedLine ? showSvg(lineCompressed) : hideSvg(lineCompressed);
            
            // 更新位置线标签
            step.showInitialLine ? showSvg(labelInitial) : hideSvg(labelInitial);
            step.showEquilibriumLine ? showSvg(labelEquilibrium) : hideSvg(labelEquilibrium);
            step.showCompressedLine ? showSvg(labelCompressed) : hideSvg(labelCompressed);
            
            // 更新振幅标注
            step.showAmpArrow ? showSvg(ampArrow) : hideSvg(ampArrow);
            step.showAmpArrow ? showSvg(ampArrowLeft) : hideSvg(ampArrowLeft);
            step.showAmpArrow ? showSvg(ampArrowLeft2) : hideSvg(ampArrowLeft2);
            step.showAmpArrow ? showSvg(ampArrowRight) : hideSvg(ampArrowRight);
            step.showAmpArrow ? showSvg(ampArrowRight2) : hideSvg(ampArrowRight2);
            step.showAmpArrow ? showSvg(labelAmplitude) : hideSvg(labelAmplitude);
            
            // 更新高度标注
            step.showHeightArrow ? showSvg(heightArrow) : hideSvg(heightArrow);
            step.showHeightArrow ? showSvg(heightArrowTop) : hideSvg(heightArrowTop);
            step.showHeightArrow ? showSvg(heightArrowBottom) : hideSvg(heightArrowBottom);
            step.showHeightArrow ? showSvg(labelH) : hideSvg(labelH);
            
            // 更新时间标记高亮
            timeDot0.setAttribute('opacity', step.highlightTime === 0 ? '1' : '0.3');
            timeDot1.setAttribute('opacity', step.highlightTime === 1 ? '1' : '0.3');
            timeDot2.setAttribute('opacity', step.highlightTime === 2 ? '1' : '0.3');
            
            // 更新时间标签
            step.showT1Label ? showSvg(labelT1) : hideSvg(labelT1);
            step.showT2Label ? showSvg(labelT2) : hideSvg(labelT2);
            
            // 更新总结文字和同步箭头
            step.showSummary ? showSvg(summaryText) : hideSvg(summaryText);
            step.showSyncArrow ? showSvg(syncArrow) : hideSvg(syncArrow);
        }
        
        // 动画过渡
        function animateTransition(targetStep) {
            const startStep = currentStep;
            const startStepConfig = config.steps[startStep];
            const endStepConfig = config.steps[targetStep];
            
            const startPosA = config.aPositions[startStepConfig.aPos];
            const endPosA = config.aPositions[endStepConfig.aPos];
            const startPosB = config.bPositions[startStepConfig.bPos];
            const endPosB = config.bPositions[endStepConfig.bPos];
            const startSpringEnd = config.springEndX[startStepConfig.springEnd];
            const endSpringEnd = config.springEndX[endStepConfig.springEnd];
            
            const duration = 300;
            const startTime = performance.now();
            
            function animate(currentTime) {
                const elapsed = currentTime - startTime;
                const progress = Math.min(elapsed / duration, 1);
                
                // 线性插值
                const curX = startPosA.x + (endPosA.x - startPosA.x) * progress;
                const curY = startPosA.y + (endPosA.y - startPosA.y) * progress;
                objA.setAttribute('x', curX);
                objA.setAttribute('y', curY);
                
                const curBX = startPosB.x + (endPosB.x - startPosB.x) * progress;
                const curBY = startPosB.y + (endPosB.y - startPosB.y) * progress;
                objB.setAttribute('x', curBX);
                objB.setAttribute('y', curBY);
                
                const curSpringEnd = startSpringEnd + (endSpringEnd - startSpringEnd) * progress;
                updateSpring(curSpringEnd);
                
                if (progress < 1) {
                    requestAnimationFrame(animate);
                } else {
                    // 动画结束，应用完整配置
                    applyStep(endStepConfig);
                }
            }
            
            requestAnimationFrame(animate);
        }
        
        // 事件绑定
        btnPrev.addEventListener('click', function() {
            if (currentStep > 0) {
                currentStep--;
                updateScene(currentStep);
            }
        });
        
        btnNext.addEventListener('click', function() {
            if (currentStep < config.steps.length - 1) {
                currentStep++;
                updateScene(currentStep);
            }
        });
        
        // 初始化场景
        updateScene(currentStep, false);
    })();
    </script>
