    <script>
    (function() {
        // Scene 2 实现 - 第一阶段能量转化分析
        // 坐标表对照：
        // 斜面底边：(100,420)-(300,420)
        // 斜面斜边：(100,420)-(300,280)
        // 斜面竖边：(300,280)-(300,420)
        // A 初始：(300,280), B 初始：(200,350), C 初始：(100,420)
        // A 末态：(300,420), B 末态：(250,315), C 末态：(150,385)

        var currentStep = 0;
        var totalSteps = 4;

        // 元素分组 ID 列表
        var stepGroups = [
            'scene2-step0',
            'scene2-step1',
            'scene2-step2',
            'scene2-step3'
        ];

        // 按钮元素
        var prevBtn = document.getElementById('scene2-prev-btn');
        var nextBtn = document.getElementById('scene2-next-btn');

        // 显示/隐藏工具函数
        function show(el) {
            if (el) {
                el.style.display = 'block';
                el.style.opacity = '1';
            }
        }

        function hide(el) {
            if (el) {
                el.style.display = 'none';
                el.style.opacity = '0';
            }
        }

        // 更新显示状态
        function updateDisplay() {
            // 隐藏所有步骤组
            for (var i = 0; i < totalSteps; i++) {
                var group = document.getElementById(stepGroups[i]);
                if (group) {
                    hide(group);
                }
            }

            // 显示当前步骤及之前的所有步骤
            for (var i = 0; i <= currentStep; i++) {
                var group = document.getElementById(stepGroups[i]);
                if (group) {
                    show(group);
                }
            }

            // 更新按钮状态
            if (prevBtn) {
                prevBtn.disabled = (currentStep === 0);
            }
            if (nextBtn) {
                nextBtn.disabled = (currentStep === totalSteps - 1);
            }
        }

        // 事件绑定
        if (prevBtn) {
            prevBtn.addEventListener('click', function() {
                if (currentStep > 0) {
                    currentStep--;
                    updateDisplay();
                }
            });
        }

        if (nextBtn) {
            nextBtn.addEventListener('click', function() {
                if (currentStep < totalSteps - 1) {
                    currentStep++;
                    updateDisplay();
                }
            });
        }

        // 初始化显示
        updateDisplay();
    })();
    </script>
