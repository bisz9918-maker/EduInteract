    <script>
    (function() {
        // Scene 1 实现：子弹射入滑块瞬间的动量守恒
        // 布局坐标表（参考）：
        // | 元素 | 类型 | 位置/范围 | 大小 | 标签偏移方向 | 备注 |
        // |------|------|-----------|------|-------------|------|
        // | 地面 | 线段 | (50,380)-(750,380) | 700×2 | - | 水平地面线 |
        // | 地面阴影 | 斜线 | (50,380)-(750,420) | 700×40 | - | 斜线阴影表示光滑水平面 |
        // | 小车底板 | 矩形 | (100,320)-(500,380) | 400×60 | - | 小车主体 |
        // | 小车左挡板 | 矩形 | (100,260)-(130,320) | 30×60 | - | 弹簧固定端 |
        // | 小车左轮 | 圆 | (160,380) | r=20 | - | 轮子 |
        // | 小车右轮 | 圆 | (440,380) | r=20 | - | 轮子 |
        // | 弹簧 | 折线 | (130,290)-(400,290) | 270×20 | - | 原长状态，8 个波峰 |
        // | 滑块 | 矩形 | (400,300)-(480,340) | 80×40 | - | 位于小车上表面 |
        // | 子弹 | 矩形 | (620,310)-(660,330) | 40×20 | - | 从右侧入射 |
        // | M 标签 | text | (70,280) | ~30×16 | 左方 | 小车质量标注 |
        // | m 标签 | text | (420,285) | ~30×16 | 上方 | 滑块质量标注 |
        // | m₀标签 | text | (640,275) | ~35×16 | 上方 | 子弹质量标注 |
        // | v₁箭头 | 箭头 | (280,240)-(340,240) | 60×4 | - | 向右，小车 + 滑块速度 |
        // | v₁标签 | text | (310,220) | ~60×16 | 上方 | v₁=10m/s |
        // | v₀箭头 | 箭头 | (640,250)-(580,250) | 60×4 | - | 向左，子弹速度 |
        // | v₀标签 | text | (610,230) | ~60×16 | 上方 | v₀=50m/s |
        // | v₂箭头 | 箭头 | (440,240)-(500,240) | 60×4 | - | 向右，共同速度（步骤 2 显示）|
        // | v₂标签 | text | (470,220) | ~60×16 | 上方 | v₂=4m/s（步骤 2 显示）|
        // | 弹簧原长标注 | text | (265,310) | ~50×16 | 下方 | 原长（步骤 0 显示）|
        
        // 状态变量
        var currentStep = 1;
        var totalSteps = 4;
        
        // 元素引用
        var bullet = document.getElementById('scene1-bullet');
        var v0Group = document.getElementById('scene1-v0-group');
        var v2Group = document.getElementById('scene1-v2-group');
        var v2Label = document.getElementById('scene1-v2-label');
        var springLabel = document.getElementById('scene1-spring-label');
        var contactHighlight = document.getElementById('scene1-contact-highlight');
        var equationGroup = document.getElementById('scene1-equation-group');
        var block = document.getElementById('scene1-block');
        var statusEl = document.getElementById('scene1-status');
        var btnPrev = document.getElementById('scene1-btn-prev');
        var btnNext = document.getElementById('scene1-btn-next');
        
        // 工具函数：显示元素
        function show(el) {
            el.style.display = 'block';
            el.style.opacity = '1';
        }
        
        // 工具函数：隐藏元素
        function hide(el) {
            el.style.display = 'none';
            el.style.opacity = '0';
        }
        
        // 更新场景状态
        function updateScene() {
            // 步骤 1：初始状态
            if (currentStep === 1) {
                // 子弹在右侧远处
                bullet.setAttribute('x', '620');
                hide(v2Group);
                hide(springLabel);
                hide(contactHighlight);
                hide(equationGroup);
                block.setAttribute('fill', '#e3f2fd');
                v2Label.textContent = 'v₂=?';
            }
            // 步骤 2：子弹接触滑块
            else if (currentStep === 2) {
                // 子弹移动至紧贴滑块右侧
                bullet.setAttribute('x', '480');
                // v₀箭头跟随子弹
                document.getElementById('scene1-v0-arrow').setAttribute('x1', '520');
                document.getElementById('scene1-v0-arrow').setAttribute('x2', '460');
                document.getElementById('scene1-v0-label').setAttribute('x', '490');
                hide(v2Group);
                hide(springLabel);
                show(contactHighlight);
                hide(equationGroup);
                block.setAttribute('fill', '#e3f2fd');
            }
            // 步骤 3：子弹嵌入滑块，获得共同速度
            else if (currentStep === 3) {
                // 子弹与滑块合并（视觉上子弹嵌入）
                bullet.setAttribute('x', '400');
                // 滑块向右移动一小段
                block.setAttribute('x', '420');
                // v₀箭头消失
                hide(v0Group);
                // v₂箭头显示
                show(v2Group);
                hide(springLabel);
                hide(contactHighlight);
                hide(equationGroup);
                // 滑块填充色变为绿色调
                block.setAttribute('fill', '#e8f5e9');
            }
            // 步骤 4：标注动量守恒方程
            else if (currentStep === 4) {
                // 保持步骤 3 的状态
                bullet.setAttribute('x', '400');
                block.setAttribute('x', '420');
                hide(v0Group);
                show(v2Group);
                show(equationGroup);
                block.setAttribute('fill', '#e8f5e9');
            }
            
            // 更新状态指示器
            statusEl.textContent = '步骤 ' + currentStep + '/' + totalSteps;
            
            // 更新按钮状态
            btnPrev.disabled = (currentStep === 1);
            btnNext.disabled = (currentStep === totalSteps);
        }
        
        // 按钮事件
        btnPrev.addEventListener('click', function() {
            if (currentStep > 1) {
                currentStep--;
                updateScene();
            }
        });
        
        btnNext.addEventListener('click', function() {
            if (currentStep < totalSteps) {
                currentStep++;
                updateScene();
            }
        });
        
        // 初始化场景
        updateScene();
    })();
    </script>
