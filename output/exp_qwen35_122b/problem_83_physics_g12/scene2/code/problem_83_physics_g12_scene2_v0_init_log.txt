    <script>
    (function() {
        // Scene 2 实现
        // 布局坐标表对照：
        // | 元素 | 位置/范围 | 备注 |
        // | 地面 | (50,350)-(750,380) | 带阴影斜线 |
        // | 小车底板 | (100,280)-(500,350) | 蓝色填充 |
        // | 小车左挡板 | (100,220)-(120,280) | 与底板连接 |
        // | 小车右挡板 | (480,280)-(500,300) | 矮挡板 |
        // | 左轮 | (150,350) r=20 | 黑色描边 |
        // | 右轮 | (450,350) r=20 | 黑色描边 |
        // | 滑块 | (350,220)-(470,280) | 绿色填充，内含子弹标记 |
        // | 子弹标记 | (420,250) r=8 | 红色填充 |
        // | 弹簧（原长） | (120,250)-(350,250) | 步骤 0-1 显示 |
        // | 弹簧（压缩） | (120,250)-(300,250) | 步骤 2-3 显示 |
        // | v₁箭头 | (250,180)-(300,180) | 蓝色，标注 v₁=10m/s |
        // | v₂箭头 | (400,160)-(450,160) | 绿色，标注 v₂=4m/s |
        // | v₃箭头 | (300,140)-(350,140) | 紫色，标注 v₃=? |
        // | M 标签 | (80,260) | M=2kg（调整 x 坐标避免边缘裁剪） |
        // | m+m₀标签 | (480,240) | m+m₀=2.0kg |
        // | d 标注线 | (120,300)-(350,300) | 虚线，步骤 2-3 显示 |
        // | d 标签 | (235,340) | d=0.50m（调整 y 坐标避免与小车重叠） |
        // | 能量框 | (560,120)-(760,340) | 步骤 3 显示 |

        const TOTAL_STEPS = 4;
        let currentStep = 0;

        // 工具函数：显示元素
        function show(el) {
            if (el) {
                el.style.display = 'block';
                el.style.opacity = '1';
                if (el.setAttribute) {
                    el.setAttribute('opacity', '1');
                }
            }
        }

        // 工具函数：隐藏元素
        function hide(el) {
            if (el) {
                el.style.display = 'none';
                el.style.opacity = '0';
                if (el.setAttribute) {
                    el.setAttribute('opacity', '0');
                }
            }
        }

        // 获取元素
        function getEl(id) {
            return document.getElementById(id);
        }

        // 更新步骤显示
        function renderStep(step) {
            // 步骤 0：初始状态（子弹射入后，滑块 v₂=4m/s，小车 v₁=10m/s，弹簧原长）
            // 步骤 1：弹簧开始压缩
            // 步骤 2：弹簧压缩到最短，三者共速
            // 步骤 3：能量转化关系

            // 重置所有元素
            const springOriginal = getEl('scene2-spring-original');
            const springCompressed = getEl('scene2-spring-compressed');
            const v1Group = getEl('scene2-v1-group');
            const v2Group = getEl('scene2-v2-group');
            const v3Group = getEl('scene2-v3-group');
            const dGroup = getEl('scene2-d-group');
            const energyGroup = getEl('scene2-energy-group');
            const block = getEl('scene2-block');
            const bullet = getEl('scene2-bullet');

            // 默认状态
            show(springOriginal);
            hide(springCompressed);
            show(v1Group);
            show(v2Group);
            hide(v3Group);
            hide(dGroup);
            hide(energyGroup);

            // 滑块和子弹初始位置
            block.setAttribute('x', '350');
            block.setAttribute('width', '120');
            bullet.setAttribute('cx', '420');

            if (step === 0) {
                // 步骤 0：初始状态
                getEl('scene2-v3-label').textContent = 'v₃=?';
                getEl('scene2-step-indicator').textContent = '步骤 0/4';
            } else if (step === 1) {
                // 步骤 1：弹簧开始压缩，显示 v₃
                show(v3Group);
                getEl('scene2-v3-label').textContent = 'v₃=?';
                getEl('scene2-step-indicator').textContent = '步骤 1/4';
            } else if (step === 2) {
                // 步骤 2：弹簧压缩到最短，三者共速
                hide(springOriginal);
                show(springCompressed);
                hide(v1Group);
                hide(v2Group);
                show(v3Group);
                show(dGroup);
                getEl('scene2-v3-label').textContent = 'v₃=共速';
                // 滑块左移
                block.setAttribute('x', '300');
                block.setAttribute('width', '120');
                bullet.setAttribute('cx', '370');
                getEl('scene2-step-indicator').textContent = '步骤 2/4';
            } else if (step === 3) {
                // 步骤 3：能量转化关系
                hide(springOriginal);
                show(springCompressed);
                hide(v1Group);
                hide(v2Group);
                show(v3Group);
                show(dGroup);
                show(energyGroup);
                getEl('scene2-v3-label').textContent = 'v₃=共速';
                // 滑块左移
                block.setAttribute('x', '300');
                block.setAttribute('width', '120');
                bullet.setAttribute('cx', '370');
                getEl('scene2-step-indicator').textContent = '步骤 3/4';
            }

            // 更新按钮状态
            const btnPrev = getEl('scene2-btn-prev');
            const btnNext = getEl('scene2-btn-next');
            btnPrev.disabled = (step === 0);
            btnNext.disabled = (step === TOTAL_STEPS - 1);
        }

        // 事件绑定
        getEl('scene2-btn-prev').addEventListener('click', function() {
            if (currentStep > 0) {
                currentStep--;
                renderStep(currentStep);
            }
        });

        getEl('scene2-btn-next').addEventListener('click', function() {
            if (currentStep < TOTAL_STEPS - 1) {
                currentStep++;
                renderStep(currentStep);
            }
        });

        // 初始化
        renderStep(0);
    })();
    </script>
