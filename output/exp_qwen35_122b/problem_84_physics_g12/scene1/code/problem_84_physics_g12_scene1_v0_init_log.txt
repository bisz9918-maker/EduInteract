    <script>
    (function() {
        // Scene 1 - U 形框架电磁感应装置与受力分析
        // 布局坐标表（实现对照依据）：
        // | 元素 | 类型 | 位置/范围 | 大小 | 标签偏移方向 | 备注 |
        // |------|------|-----------|------|-------------|------|
        // | 上导轨 | 线段 | (150,200)-(650,200) | 500×2 | - | 水平导轨 |
        // | 下导轨 | 线段 | (150,300)-(650,300) | 500×2 | - | 水平导轨 |
        // | 电阻 R | 矩形 | (630,190)-(670,310) | 40×120 | 右侧 | 右端电阻 |
        // | 金属棒 ab | 线段 | (250,300)-(350,200) | ~141×3 | - | 斜置金属棒 |
        // | 点 a | 点 | (350,200) | 4 | 右上 | 金属棒上端 |
        // | 点 b | 点 | (250,300) | 4 | 左下 | 金属棒下端 |
        // | 磁场箭头 1-5 | 箭头 | x=200,300,400,500,600, y=180→150 | 20×3 | 上方 | B 方向 |
        // | 外力 F 箭头 | 箭头 | (300,250)→(180,250) | 120×4 | 左侧 | 蓝色主动力 |
        // | 摩擦力 f 箭头 | 箭头 | (300,250)→(400,250) | 100×3 | 右侧 | 红色阻力 |
        // | 安培力 FA 箭头 | 箭头 | (300,270)→(420,270) | 120×3 | 右侧 | 红色阻力，与 f 错开 |
        // | 标签 L=1m | text | (400,320) | ~50×16 | 下方 | 框架宽度 |
        // | 标签 R=0.8Ω | text | (680,250) | ~60×16 | 右侧 | 电阻值 |
        // | 标签 r=0.2Ω | text | (385,205) | ~60×16 | 右上 | 金属棒电阻 |
        // | 标签 B=1T | text | (400,140) | ~50×16 | 上方 | 磁感应强度 |
        // | 标签 F=3N | text | (110,235) | ~50×16 | 左上 | 外力 |
        // | 标签 f=μmg | text | (420,270) | ~70×16 | 右上 | 摩擦力 |
        // | 标签 FA | text | (440,290) | ~40×16 | 右上 | 安培力 |
        // | 标签 m=0.2kg | text | (180,340) | ~80×16 | 左下 | 质量 |
        // | 标签 μ=0.5 | text | (260,340) | ~60×16 | 下方 | 动摩擦因数 |
        // | 平衡条件文字 | text | (400,400) | ~200×20 | - | 绿色高亮 |
        
        // 状态变量
        var currentStep = 0;
        var totalSteps = 6; // 0-5 共 6 个步骤
        
        // 元素分组（按步骤）
        var stepElements = {
            0: [], // 初始状态：基础结构（已默认显示）
            1: ['scene1-magnetic-field'], // 步骤 1：磁场
            2: ['scene1-force-F'], // 步骤 2：外力 F
            3: ['scene1-force-f'], // 步骤 3：摩擦力 f
            4: ['scene1-force-FA'], // 步骤 4：安培力 FA
            5: ['scene1-equation'] // 步骤 5：平衡条件
        };
        
        // 工具函数：显示元素（同时处理 display 和 opacity）
        function showElement(elementId) {
            var el = document.getElementById(elementId);
            if (el) {
                el.style.display = 'block';
                if (el.tagName === 'g') {
                    el.style.opacity = '1';
                    var children = el.querySelectorAll('*');
                    for (var i = 0; i < children.length; i++) {
                        children[i].style.opacity = '1';
                    }
                } else {
                    el.style.opacity = '1';
                }
            }
        }
        
        // 工具函数：隐藏元素
        function hideElement(elementId) {
            var el = document.getElementById(elementId);
            if (el) {
                el.style.display = 'none';
                el.style.opacity = '0';
            }
        }
        
        // 更新当前步骤的显示状态
        function updateStepDisplay() {
            // 隐藏所有步骤元素
            for (var step = 1; step < totalSteps; step++) {
                var elements = stepElements[step] || [];
                for (var i = 0; i < elements.length; i++) {
                    hideElement(elements[i]);
                }
            }
            
            // 显示当前步骤及之前的所有元素
            for (var s = 1; s <= currentStep; s++) {
                var elements = stepElements[s] || [];
                for (var j = 0; j < elements.length; j++) {
                    showElement(elements[j]);
                }
            }
            
            // 步骤 5 时，添加力箭头闪烁动画
            if (currentStep === 5) {
                var forceLines = ['scene1-force-F-line', 'scene1-force-f-line', 'scene1-force-FA-line'];
                for (var k = 0; k < forceLines.length; k++) {
                    var line = document.getElementById(forceLines[k]);
                    if (line) {
                        line.classList.remove('scene1-flash');
                        void line.offsetWidth;
                        line.classList.add('scene1-flash');
                    }
                }
            }
            
            // 更新按钮状态
            var prevBtn = document.getElementById('scene1-btn-prev');
            var nextBtn = document.getElementById('scene1-btn-next');
            prevBtn.disabled = (currentStep === 0);
            nextBtn.disabled = (currentStep === totalSteps - 1);
        }
        
        // 下一步
        function nextStep() {
            if (currentStep < totalSteps - 1) {
                currentStep++;
                updateStepDisplay();
            }
        }
        
        // 上一步
        function prevStep() {
            if (currentStep > 0) {
                currentStep--;
                updateStepDisplay();
            }
        }
        
        // 重置
        function resetStep() {
            currentStep = 0;
            updateStepDisplay();
        }
        
        // 初始化
        function init() {
            document.getElementById('scene1-btn-next').addEventListener('click', nextStep);
            document.getElementById('scene1-btn-prev').addEventListener('click', prevStep);
            document.getElementById('scene1-btn-reset').addEventListener('click', resetStep);
            updateStepDisplay();
        }
        
        if (document.readyState === 'loading') {
            document.addEventListener('DOMContentLoaded', init);
        } else {
            init();
        }
    })();
    </script>
