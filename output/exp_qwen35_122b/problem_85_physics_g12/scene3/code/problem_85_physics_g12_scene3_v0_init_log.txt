    <script>
    (function() {
        // Scene 3 实现 - 活塞 a 上升等温压缩过程

        // 常量定义
        const H = 280;           // 气缸高度 (px) - 从 y=90 到 y=370
        const INITIAL_OXYGEN_HEIGHT = 75;  // 初始氧气体积高度 (H/4 ≈ 70px，取 75px)
        const MAX_DELTA_H = H / 16;        // 最大上升距离 (H/16 = 17.5px)
        const PISTON_A_INITIAL_Y = 90 + INITIAL_OXYGEN_HEIGHT;  // 活塞 a 初始 y 坐标 (165)
        const SLIDER_LEFT = 250;
        const SLIDER_RIGHT = 550;
        const SLIDER_WIDTH = SLIDER_RIGHT - SLIDER_LEFT;  // 300px

        // 状态变量
        let delta_h = 0;  // 当前上升距离 (0 到 MAX_DELTA_H)
        let isDragging = false;

        // DOM 元素
        const svg = document.getElementById('scene3-svg');
        const pistonA = document.getElementById('scene3-piston-a');
        const oxygenArea = document.getElementById('scene3-oxygen-area');
        const nitrogenArea = document.getElementById('scene3-nitrogen-area');
        const sliderHandle = document.getElementById('scene3-slider-handle');
        const sliderTrack = document.getElementById('scene3-slider-track');
        const deltaHDisplay = document.getElementById('scene3-delta-h-display');
        const valPOx = document.getElementById('scene3-val-p-ox');
        const barPOx = document.getElementById('scene3-bar-p-ox');
        const valVOx = document.getElementById('scene3-val-v-ox');
        const barVOx = document.getElementById('scene3-bar-v-ox');
        const valTOx = document.getElementById('scene3-val-t-ox');
        const barTOx = document.getElementById('scene3-bar-t-ox');

        // 初始化
        function init() {
            updateDisplay();
            setupSlider();
        }

        // 更新所有显示
        function updateDisplay() {
            // 计算当前状态
            const currentOxygenHeight = INITIAL_OXYGEN_HEIGHT - delta_h;
            const volumeRatio = currentOxygenHeight / INITIAL_OXYGEN_HEIGHT;  // V/V₀
            const pressureRatio = INITIAL_OXYGEN_HEIGHT / currentOxygenHeight;  // P/P₀ (玻意耳定律)

            // 更新活塞 a 位置
            const pistonAY = PISTON_A_INITIAL_Y - delta_h;
            pistonA.setAttribute('y1', pistonAY);
            pistonA.setAttribute('y2', pistonAY);

            // 更新氧气区域
            oxygenArea.setAttribute('y', 90);
            oxygenArea.setAttribute('height', currentOxygenHeight);

            // 更新氮气区域 (活塞 a 下方 + 连通管 + 活塞 b 下方)
            const nitrogenPoints = [
                150, pistonAY,      // 活塞 a 左端
                350, pistonAY,      // 活塞 a 右端
                350, 370,           // 气缸 A 右下角
                354, 370,           // 连通管右下
                354, 340,           // 连通管右上
                500, 340,           // 连通管左上
                500, 370,           // 气缸 B 左下角
                600, 370,           // 气缸 B 右下角
                600, 94,            // 活塞 b 右端
                500, 94,            // 活塞 b 左端
                500, 340,           // 气缸 B 左侧中间
                354, 340,           // 连通管左侧中间
                354, 370,           // 连通管左下
                150, 370            // 气缸 A 左下角
            ];
            nitrogenArea.setAttribute('points', nitrogenPoints.join(' '));

            // 更新滑块位置
            const sliderX = SLIDER_LEFT + (delta_h / MAX_DELTA_H) * SLIDER_WIDTH;
            sliderHandle.setAttribute('cx', sliderX);

            // 更新 Δh 显示
            const deltaHFraction = (delta_h / H).toFixed(4);
            deltaHDisplay.textContent = `Δh = ${delta_h.toFixed(2)} px`;

            // 更新状态面板
            valPOx.textContent = `${pressureRatio.toFixed(2)} P₀`;
            barPOx.setAttribute('width', Math.min(60, pressureRatio * 60));

            valVOx.textContent = `${volumeRatio.toFixed(2)} V₀`;
            barVOx.setAttribute('width', Math.min(60, volumeRatio * 60));

            // 温度恒定
            valTOx.textContent = '280 K';
            barTOx.setAttribute('width', 60);
        }

        // 滑块交互设置
        function setupSlider() {
            // 鼠标事件
            sliderHandle.addEventListener('mousedown', onSliderMouseDown);
            document.addEventListener('mousemove', onSliderMouseMove);
            document.addEventListener('mouseup', onSliderMouseUp);

            // 触摸事件
            sliderHandle.addEventListener('touchstart', onSliderTouchStart, { passive: false });
            document.addEventListener('touchmove', onSliderTouchMove, { passive: false });
            document.addEventListener('touchend', onSliderTouchEnd);

            // 滑块轨道点击
            sliderTrack.addEventListener('click', onSliderTrackClick);
        }

        function onSliderMouseDown(e) {
            isDragging = true;
            e.preventDefault();
        }

        function onSliderMouseMove(e) {
            if (!isDragging) return;
            updateSliderFromEvent(e);
        }

        function onSliderMouseUp() {
            isDragging = false;
        }

        function onSliderTouchStart(e) {
            isDragging = true;
            e.preventDefault();
        }

        function onSliderTouchMove(e) {
            if (!isDragging) return;
            e.preventDefault();
            updateSliderFromEvent(e.touches[0]);
        }

        function onSliderTouchEnd() {
            isDragging = false;
        }

        function onSliderTrackClick(e) {
            updateSliderFromEvent(e);
        }

        function updateSliderFromEvent(e) {
            const rect = sliderTrack.getBoundingClientRect();
            let x = e.clientX - rect.left;

            // 限制在轨道范围内
            x = Math.max(0, Math.min(SLIDER_WIDTH, x));

            // 计算 delta_h
            delta_h = (x / SLIDER_WIDTH) * MAX_DELTA_H;
            delta_h = Math.max(0, Math.min(MAX_DELTA_H, delta_h));

            updateDisplay();
        }

        // 启动
        init();
    })();
    </script>
