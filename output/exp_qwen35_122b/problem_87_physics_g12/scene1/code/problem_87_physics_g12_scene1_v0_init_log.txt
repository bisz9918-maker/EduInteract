    <script>
    (function() {
        // Scene 1 实现
        // 布局坐标表对照（斜面整体上移 60px，底部留出方程框空间）：
        // | 元素 | 位置/范围 | 备注 |
        // | 斜面底边 | (150,320)-(650,320) | 水平底边（上移 60px） |
        // | 斜面斜边 | (150,320)-(650,80) | 倾角 37°（上移 60px） |
        // | A 球 | (500,160) r=15 | 上方小球（上移 40px） |
        // | B 球 | (320,260) r=15 | 下方小球（上移 40px） |
        // | L 虚线 | (320,260)-(500,160) | 两球距离 |
        // | 37°标签 | (175,310) | 角度值（上移 60px） |
        // | E 标签 | (755,50) | 电场标注（上移 10px） |

        const TOTAL_STEPS = 5;
        let currentStep = 0;

        // 元素引用
        const ballA = document.getElementById('scene1-ball-a');
        const ballB = document.getElementById('scene1-ball-b');
        const forcesA = document.getElementById('scene1-forces-a');
        const forcesB = document.getElementById('scene1-forces-b');
        const acceleration = document.getElementById('scene1-acceleration');
        const equation1 = document.getElementById('scene1-equation1');
        const equation2 = document.getElementById('scene1-equation2');
        const stepIndicator = document.getElementById('scene1-step-indicator');
        const btnPrev = document.getElementById('scene1-btn-prev');
        const btnNext = document.getElementById('scene1-btn-next');

        // 工具函数：显示元素
        function show(el) {
            if (el) {
                el.classList.remove('hidden');
                el.classList.add('visible');
                el.style.opacity = '1';
            }
        }

        // 工具函数：隐藏元素
        function hide(el) {
            if (el) {
                el.classList.remove('visible');
                el.classList.add('hidden');
                el.style.opacity = '0';
            }
        }

        // 更新场景状态
        function updateScene() {
            // 重置所有动态元素
            hide(forcesA);
            hide(forcesB);
            hide(acceleration);
            hide(equation1);
            hide(equation2);
            
            // 恢复球体默认填充
            ballA.style.fill = 'rgba(0,0,0,0.2)';
            ballB.style.fill = 'rgba(0,0,0,0.2)';

            // 根据当前步骤显示对应元素
            switch(currentStep) {
                case 0: // 初始状态 - 仅显示基础图
                    break;
                    
                case 1: // A 球受力分析
                    show(forcesA);
                    ballA.style.fill = 'rgba(21,101,192,0.4)';
                    break;
                    
                case 2: // A 球平衡方程
                    show(forcesA);
                    show(equation1);
                    ballA.style.fill = 'rgba(21,101,192,0.4)';
                    break;
                    
                case 3: // B 球受力分析
                    show(forcesB);
                    ballB.style.fill = 'rgba(21,101,192,0.4)';
                    break;
                    
                case 4: // B 球加速度分析
                    show(forcesB);
                    show(acceleration);
                    show(equation2);
                    ballB.style.fill = 'rgba(21,101,192,0.4)';
                    break;
            }

            // 更新步骤指示器
            stepIndicator.textContent = `步骤 ${currentStep + 1}/${TOTAL_STEPS}`;

            // 更新按钮状态
            btnPrev.disabled = currentStep === 0;
            btnNext.disabled = currentStep === TOTAL_STEPS - 1;
        }

        // 切换步骤
        function changeStep(delta) {
            const newStep = currentStep + delta;
            if (newStep >= 0 && newStep < TOTAL_STEPS) {
                currentStep = newStep;
                updateScene();
            }
        }

        // 事件绑定
        btnPrev.addEventListener('click', () => changeStep(-1));
        btnNext.addEventListener('click', () => changeStep(1));

        // 初始化
        updateScene();
    })();
    </script>
