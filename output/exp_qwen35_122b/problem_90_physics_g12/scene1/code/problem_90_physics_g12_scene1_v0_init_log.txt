    <script>
    (function() {
        // Scene 1 实现
        // 坐标表对照：
        // 图甲玻璃管：x=100-140, y=80-430
        // 图乙玻璃管：x=350-390, y=80-430
        // 图丙玻璃管：x=600-640, y=80-430
        // 比例：5.65 px/cm (L=62cm→350px, l₁=35cm→198px, l₂=40cm→226px, h=5cm→28px)

        const currentStep = { value: 0 }; // 0=甲，1=乙，2=丙
        const totalSteps = 3;

        const highlightBox = document.getElementById('scene1-highlight');
        const prevBtn = document.getElementById('scene1-prev-btn');
        const nextBtn = document.getElementById('scene1-next-btn');

        // 高亮框位置配置
        const highlightPositions = [
            { x: 90, y: 70, width: 60, height: 370 },  // 图甲
            { x: 340, y: 70, width: 60, height: 370 }, // 图乙
            { x: 590, y: 70, width: 60, height: 370 }  // 图丙
        ];

        function updateHighlight() {
            const pos = highlightPositions[currentStep.value];
            highlightBox.setAttribute('x', pos.x);
            highlightBox.setAttribute('y', pos.y);
            highlightBox.setAttribute('width', pos.width);
            highlightBox.setAttribute('height', pos.height);
            highlightBox.style.opacity = '1';

            // 更新按钮状态
            prevBtn.disabled = (currentStep.value === 0);
            nextBtn.disabled = (currentStep.value === totalSteps - 1);
        }

        function goToStep(step) {
            if (step < 0 || step >= totalSteps) return;
            currentStep.value = step;
            updateHighlight();
        }

        prevBtn.addEventListener('click', () => {
            goToStep(currentStep.value - 1);
        });

        nextBtn.addEventListener('click', () => {
            goToStep(currentStep.value + 1);
        });

        // 初始化
        updateHighlight();
    })();
    </script>
