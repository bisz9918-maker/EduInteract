    <script>
    (function() {
        // Scene 1 实现
        // 布局坐标表对照：
        // 海岸线：(100,280)-(700,280)
        // 港口 A：(150,280), 港口 B：(350,280), 港口 C：(550,280)
        // 甲船箭头：(180,230)-(520,230), 标签：(350,210)
        // 乙船箭头：(320,330)-(180,330), 标签：(250,365) - 调整位置避免拥挤
        // AB 距离线：(150,310)-(350,310), 标注：(250,325)
        // BC 距离线：(350,310)-(550,310), 标注：(450,325)
        // 标题：(400,40)
        
        const TOTAL_STEPS = 4;
        let currentStep = 0;
        
        // 元素 ID 映射
        const elements = {
            labels: ['scene1-label-A', 'scene1-label-B', 'scene1-label-C'],
            arrows: ['scene1-arrow-甲船', 'scene1-label-甲船', 'scene1-arrow-乙船', 'scene1-label-乙船'],
            distances: ['scene1-dist-AB', 'scene1-label-dist-AB', 'scene1-dist-BC', 'scene1-label-dist-BC']
        };
        
        // 按钮和指示器
        const btnPrev = document.getElementById('scene1-btn-prev');
        const btnNext = document.getElementById('scene1-btn-next');
        const stepIndicator = document.getElementById('scene1-step-indicator');
        
        // 更新元素可见性
        function updateVisibility() {
            // 步骤 0: 只显示海岸线和港口（无标签）
            // 步骤 1: 显示港口标签 + 甲船箭头和标签
            // 步骤 2: 显示乙船箭头和标签
            // 步骤 3: 显示距离线和标注
            
            // 港口标签
            elements.labels.forEach((id, idx) => {
                const el = document.getElementById(id);
                if (currentStep >= 1) {
                    el.style.display = 'block';
                    el.classList.add('scene1-fade-in');
                } else {
                    el.style.display = 'none';
                    el.classList.remove('scene1-fade-in');
                }
            });
            
            // 甲船相关
            if (currentStep >= 1) {
                document.getElementById('scene1-arrow-甲船').style.display = 'block';
                document.getElementById('scene1-label-甲船').style.display = 'block';
                document.getElementById('scene1-arrow-甲船').classList.add('scene1-fade-in');
                document.getElementById('scene1-label-甲船').classList.add('scene1-fade-in');
            } else {
                document.getElementById('scene1-arrow-甲船').style.display = 'none';
                document.getElementById('scene1-label-甲船').style.display = 'none';
            }
            
            // 乙船相关
            if (currentStep >= 2) {
                document.getElementById('scene1-arrow-乙船').style.display = 'block';
                document.getElementById('scene1-label-乙船').style.display = 'block';
                document.getElementById('scene1-arrow-乙船').classList.add('scene1-fade-in');
                document.getElementById('scene1-label-乙船').classList.add('scene1-fade-in');
            } else {
                document.getElementById('scene1-arrow-乙船').style.display = 'none';
                document.getElementById('scene1-label-乙船').style.display = 'none';
            }
            
            // 距离标注
            if (currentStep >= 3) {
                elements.distances.forEach(id => {
                    const el = document.getElementById(id);
                    el.style.display = 'block';
                    el.classList.add('scene1-fade-in');
                });
            } else {
                elements.distances.forEach(id => {
                    const el = document.getElementById(id);
                    el.style.display = 'none';
                    el.classList.remove('scene1-fade-in');
                });
            }
            
            // 更新按钮状态和指示器
            btnPrev.disabled = (currentStep === 0);
            btnNext.disabled = (currentStep === TOTAL_STEPS - 1);
            stepIndicator.textContent = `步骤 ${currentStep + 1}/${TOTAL_STEPS}`;
        }
        
        // 按钮事件
        btnPrev.addEventListener('click', () => {
            if (currentStep > 0) {
                currentStep--;
                updateVisibility();
            }
        });
        
        btnNext.addEventListener('click', () => {
            if (currentStep < TOTAL_STEPS - 1) {
                currentStep++;
                updateVisibility();
            }
        });
        
        // 初始化
        updateVisibility();
    })();
    </script>
