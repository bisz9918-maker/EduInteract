    <script>
    (function() {
        // Scene 1: 电路结构总览与开关位置识别
        // 布局坐标表对照：
        // | 元素 | 位置/范围 | 备注 |
        // | 电源 | (200,420)-(600,420) | 底部水平 |
        // | L1 | 圆心 (275,200) | 左侧灯泡 |
        // | L2 | 圆心 (425,200) | 右侧灯泡 |
        // | S2 | (250,120)-(450,120) | 顶部开关 |
        // | S1 公共端 | (550,380) | 右下角 |
        // | S1 触点 a | (600,380) | 右侧触点 |
        // | S1 触点 b | (450,280) | 中间触点 |
        
        const currentStepMax = 4;
        let currentStep = 0;
        
        // 元素引用
        const elements = {
            // 导线
            wireLeft: document.getElementById('scene1-wire-left'),
            wireL1Left: document.getElementById('scene1-wire-l1-left'),
            wireL1Right: document.getElementById('scene1-wire-l1-right'),
            wireL2Left: document.getElementById('scene1-wire-l2-left'),
            wireL2Right: document.getElementById('scene1-wire-l2-right'),
            wireS2Left: document.getElementById('scene1-wire-s2-left'),
            wireS2Right: document.getElementById('scene1-wire-s2-right'),
            wireS2: document.getElementById('scene1-wire-s2'),
            wireS1A: document.getElementById('scene1-wire-s1-a'),
            wireS1B: document.getElementById('scene1-wire-s1-b'),
            wireRight: document.getElementById('scene1-wire-right'),
            // 电源
            batteryPos: document.getElementById('scene1-battery-pos'),
            batteryNeg: document.getElementById('scene1-battery-neg'),
            // 灯泡
            lamp1: document.getElementById('scene1-lamp1'),
            lamp1Cross1: document.getElementById('scene1-lamp1-cross1'),
            lamp1Cross2: document.getElementById('scene1-lamp1-cross2'),
            lamp2: document.getElementById('scene1-lamp2'),
            lamp2Cross1: document.getElementById('scene1-lamp2-cross1'),
            lamp2Cross2: document.getElementById('scene1-lamp2-cross2'),
            // 开关 S2
            switch2Left: document.getElementById('scene1-switch2-left'),
            switch2Right: document.getElementById('scene1-switch2-right'),
            switch2Arm: document.getElementById('scene1-switch2-arm'),
            // 开关 S1
            switch1Common: document.getElementById('scene1-switch1-common'),
            switch1A: document.getElementById('scene1-switch1-a'),
            switch1B: document.getElementById('scene1-switch1-b'),
            switch1Arm: document.getElementById('scene1-switch1-arm'),
            // 节点
            nodeL1Left: document.getElementById('scene1-node-l1-left'),
            nodeL1L2: document.getElementById('scene1-node-l1-l2'),
            nodeL2Right: document.getElementById('scene1-node-l2-right'),
            nodeS2Left: document.getElementById('scene1-node-s2-left'),
            nodeS2Right: document.getElementById('scene1-node-s2-right'),
            // 触点高亮
            contactA: document.getElementById('scene1-contact-a'),
            contactB: document.getElementById('scene1-contact-b'),
            // 标签
            labelBattery: document.getElementById('scene1-label-battery'),
            labelL1: document.getElementById('scene1-label-l1'),
            labelL2: document.getElementById('scene1-label-l2'),
            labelS1: document.getElementById('scene1-label-s1'),
            labelS2: document.getElementById('scene1-label-s2'),
            labelA: document.getElementById('scene1-label-a'),
            labelB: document.getElementById('scene1-label-b'),
            // 参数
            paramL1: document.getElementById('scene1-param-l1'),
            paramL2: document.getElementById('scene1-param-l2'),
            // 控件
            btnPrev: document.getElementById('scene1-btn-prev'),
            btnNext: document.getElementById('scene1-btn-next'),
            stepIndicator: document.getElementById('scene1-step-indicator')
        };
        
        // 工具函数：设置元素为灰色（默认状态）
        function setGray(el) {
            if (!el) return;
            el.classList.remove('highlight', 'active');
        }
        
        // 工具函数：设置元素为蓝色高亮
        function setHighlight(el) {
            if (!el) return;
            el.classList.remove('active');
            el.classList.add('highlight');
        }
        
        // 工具函数：设置元素为黑色激活
        function setActive(el) {
            if (!el) return;
            el.classList.remove('highlight');
            el.classList.add('active');
        }
        
        // 更新步骤显示
        function updateStepDisplay() {
            elements.stepIndicator.textContent = `步骤 ${currentStep + 1}/5`;
            elements.btnPrev.disabled = currentStep === 0;
            elements.btnNext.disabled = currentStep === currentStepMax;
        }
        
        // 更新高亮状态
        function updateHighlight() {
            // 重置所有元素为灰色
            Object.values(elements).forEach(el => {
                if (el && el.classList) {
                    el.classList.remove('highlight', 'active', 'visible');
                }
            });
            
            switch(currentStep) {
                case 0: // 初始状态：全部灰色
                    break;
                    
                case 1: // 高亮电源和主回路
                    setActive(elements.batteryPos);
                    setActive(elements.batteryNeg);
                    setHighlight(elements.labelBattery);
                    setHighlight(elements.wireLeft);
                    setHighlight(elements.wireRight);
                    break;
                    
                case 2: // 高亮 S1 及 a、b 触点
                    setActive(elements.batteryPos);
                    setActive(elements.batteryNeg);
                    setHighlight(elements.labelBattery);
                    setHighlight(elements.wireLeft);
                    setHighlight(elements.wireRight);
                    setHighlight(elements.switch1Common);
                    setHighlight(elements.switch1A);
                    setHighlight(elements.switch1B);
                    setHighlight(elements.switch1Arm);
                    setHighlight(elements.wireS1A);
                    setHighlight(elements.wireS1B);
                    setHighlight(elements.labelS1);
                    setHighlight(elements.labelA);
                    setHighlight(elements.labelB);
                    elements.contactA.classList.add('visible');
                    elements.contactB.classList.add('visible');
                    break;
                    
                case 3: // 高亮 S2 支路
                    setActive(elements.batteryPos);
                    setActive(elements.batteryNeg);
                    setHighlight(elements.labelBattery);
                    setHighlight(elements.wireLeft);
                    setHighlight(elements.wireRight);
                    setHighlight(elements.switch1Common);
                    setHighlight(elements.switch1A);
                    setHighlight(elements.switch1B);
                    setHighlight(elements.switch1Arm);
                    setHighlight(elements.wireS1A);
                    setHighlight(elements.wireS1B);
                    setHighlight(elements.labelS1);
                    setHighlight(elements.labelA);
                    setHighlight(elements.labelB);
                    elements.contactA.classList.add('visible');
                    elements.contactB.classList.add('visible');
                    setHighlight(elements.wireS2);
                    setHighlight(elements.wireS2Left);
                    setHighlight(elements.wireS2Right);
                    setHighlight(elements.switch2Left);
                    setHighlight(elements.switch2Right);
                    setHighlight(elements.switch2Arm);
                    setHighlight(elements.labelS2);
                    setHighlight(elements.nodeS2Left);
                    setHighlight(elements.nodeS2Right);
                    break;
                    
                case 4: // 标注 L1、L2 参数
                    setActive(elements.batteryPos);
                    setActive(elements.batteryNeg);
                    setHighlight(elements.labelBattery);
                    setHighlight(elements.wireLeft);
                    setHighlight(elements.wireRight);
                    setHighlight(elements.switch1Common);
                    setHighlight(elements.switch1A);
                    setHighlight(elements.switch1B);
                    setHighlight(elements.switch1Arm);
                    setHighlight(elements.wireS1A);
                    setHighlight(elements.wireS1B);
                    setHighlight(elements.labelS1);
                    setHighlight(elements.labelA);
                    setHighlight(elements.labelB);
                    elements.contactA.classList.add('visible');
                    elements.contactB.classList.add('visible');
                    setHighlight(elements.wireS2);
                    setHighlight(elements.wireS2Left);
                    setHighlight(elements.wireS2Right);
                    setHighlight(elements.switch2Left);
                    setHighlight(elements.switch2Right);
                    setHighlight(elements.switch2Arm);
                    setHighlight(elements.labelS2);
                    setHighlight(elements.nodeS2Left);
                    setHighlight(elements.nodeS2Right);
                    setHighlight(elements.lamp1);
                    setHighlight(elements.lamp1Cross1);
                    setHighlight(elements.lamp1Cross2);
                    setHighlight(elements.lamp2);
                    setHighlight(elements.lamp2Cross1);
                    setHighlight(elements.lamp2Cross2);
                    setHighlight(elements.labelL1);
                    setHighlight(elements.labelL2);
                    setHighlight(elements.wireL1Left);
                    setHighlight(elements.wireL1Right);
                    setHighlight(elements.wireL2Left);
                    setHighlight(elements.wireL2Right);
                    setHighlight(elements.nodeL1Left);
                    setHighlight(elements.nodeL1L2);
                    setHighlight(elements.nodeL2Right);
                    elements.paramL1.classList.add('visible');
                    elements.paramL2.classList.add('visible');
                    break;
            }
            
            updateStepDisplay();
        }
        
        // 按钮事件
        elements.btnPrev.addEventListener('click', function() {
            if (currentStep > 0) {
                currentStep--;
                updateHighlight();
            }
        });
        
        elements.btnNext.addEventListener('click', function() {
            if (currentStep < currentStepMax) {
                currentStep++;
                updateHighlight();
            }
        });
        
        // 初始化
        updateHighlight();
    })();
    </script>
