            document.getElementById('scene2-fig3-arc-bottom')
        ];
        const fig3Angles = [
            document.getElementById('scene2-fig3-angle-top'),
            document.getElementById('scene2-fig3-angle-bottom')
        ];
        const fig3Type = document.getElementById('scene2-fig3-type');
        const fig3Judge = document.getElementById('scene2-fig3-judge');
        const fig3Conclusion = document.getElementById('scene2-fig3-conclusion');

        const fig4Triangles = [
            document.getElementById('scene2-fig4-triangle-top'),
            document.getElementById('scene2-fig4-triangle-bottom')
        ];
        const fig4Arcs = [
            document.getElementById('scene2-fig4-arc-top'),
            document.getElementById('scene2-fig4-arc-bottom')
        ];
        const fig4Angles = [
            document.getElementById('scene2-fig4-angle-top'),
            document.getElementById('scene2-fig4-angle-bottom')
        ];
        const fig4Type = document.getElementById('scene2-fig4-type');
        const fig4Judge = document.getElementById('scene2-fig4-judge');
        const fig4Conclusion = document.getElementById('scene2-fig4-conclusion');

        // 工具函数：显示/隐藏元素
        function show(el) {
            if (el) {
                el.classList.remove('scene2-hidden');
                el.classList.add('scene2-visible');
            }
        }

        function hide(el) {
            if (el) {
                el.classList.remove('scene2-visible');
                el.classList.add('scene2-hidden');
            }
        }

        // 更新按钮状态
        function updateButtons() {
            if (currentStep === 0) {
                btnPrev.classList.add('scene2-btn-disabled');
                btnPrevText.textContent = '◀ 上一步';
            } else {
                btnPrev.classList.remove('scene2-btn-disabled');
                btnPrevText.textContent = '◀ 上一步';
            }

            if (currentStep === totalSteps) {
                btnNext.classList.add('scene2-btn-disabled');
                btnNextText.textContent = '已完成';
            } else {
                btnNext.classList.remove('scene2-btn-disabled');
                btnNextText.textContent = '下一步 ▶';
            }

            stepIndicator.textContent = '步骤 ' + currentStep + '/' + totalSteps;
        }

        // 更新场景状态
        function updateScene() {
            // 重置所有高亮和标注
            fig1Triangles.forEach(t => {
                t.classList.remove('scene2-highlight-red', 'scene2-highlight-green');
                t.setAttribute('class', 'scene2-triangle');
            });
            fig2Triangles.forEach(t => {
                t.classList.remove('scene2-highlight-red', 'scene2-highlight-green');
                t.setAttribute('class', 'scene2-triangle');
            });
            fig3Triangles.forEach(t => {
                t.classList.remove('scene2-highlight-red', 'scene2-highlight-green');
                t.setAttribute('class', 'scene2-triangle');
            });
            fig4Triangles.forEach(t => {
                t.classList.remove('scene2-highlight-red', 'scene2-highlight-green');
                t.setAttribute('class', 'scene2-triangle');
            });

            // 隐藏所有标注
            [fig1Arcs, fig1Angles, fig2Arcs, fig2Angles, fig3Arcs, fig3Angles, fig4Arcs, fig4Angles].flat().forEach(hide);
            [fig1Type, fig1Judge, fig1Conclusion, fig2Type, fig2Judge, fig2Conclusion].forEach(hide);
            [fig3Type, fig3Judge, fig3Conclusion, fig4Type, fig4Judge, fig4Conclusion].forEach(hide);

            // 步骤 0：初始状态 - 只显示原图
            if (currentStep === 0) {
                // 所有标注已隐藏
            }
            // 步骤 1：高亮图①②的三角形
            else if (currentStep === 1) {
                fig1Triangles.forEach(t => t.setAttribute('class', 'scene2-triangle scene2-highlight-red'));
                fig2Triangles.forEach(t => t.setAttribute('class', 'scene2-triangle scene2-highlight-red'));
                fig1Arcs.forEach(show);
                fig1Angles.forEach(show);
                fig2Arcs.forEach(show);
                fig2Angles.forEach(show);
                show(fig1Type);
                show(fig1Judge);
                show(fig2Type);
                show(fig2Judge);
            }
            // 步骤 2：显示图①②结论
            else if (currentStep === 2) {
                fig1Triangles.forEach(t => t.setAttribute('class', 'scene2-triangle scene2-highlight-red'));
                fig2Triangles.forEach(t => t.setAttribute('class', 'scene2-triangle scene2-highlight-red'));
                fig1Arcs.forEach(show);
                fig1Angles.forEach(show);
                fig2Arcs.forEach(show);
                fig2Angles.forEach(show);
                show(fig1Type);
                show(fig1Judge);
                show(fig1Conclusion);
                show(fig2Type);
                show(fig2Judge);
                show(fig2Conclusion);
            }
            // 步骤 3：高亮图③④的三角形
            else if (currentStep === 3) {
                fig1Triangles.forEach(t => t.setAttribute('class', 'scene2-triangle scene2-highlight-red'));
                fig2Triangles.forEach(t => t.setAttribute('class', 'scene2-triangle scene2-highlight-red'));
                fig3Triangles.forEach(t => t.setAttribute('class', 'scene2-triangle scene2-highlight-green'));
                fig4Triangles.forEach(t => t.setAttribute('class', 'scene2-triangle scene2-highlight-green'));
                fig1Arcs.forEach(show);
                fig1Angles.forEach(show);
                fig2Arcs.forEach(show);
                fig2Angles.forEach(show);
                fig3Arcs.forEach(show);
                fig3Angles.forEach(show);
                fig4Arcs.forEach(show);
                fig4Angles.forEach(show);
                show(fig1Type);
                show(fig1Judge);
                show(fig1Conclusion);
                show(fig2Type);
                show(fig2Judge);
                show(fig2Conclusion);
                show(fig3Type);
                show(fig3Judge);
                show(fig3Conclusion);
                show(fig4Type);
                show(fig4Judge);
            }
            // 步骤 4：显示图④结论
            else if (currentStep === 4) {
                fig1Triangles.forEach(t => t.setAttribute('class', 'scene2-triangle scene2-highlight-red'));
                fig2Triangles.forEach(t => t.setAttribute('class', 'scene2-triangle scene2-highlight-red'));
                fig3Triangles.forEach(t => t.setAttribute('class', 'scene2-triangle scene2-highlight-green'));
                fig4Triangles.forEach(t => t.setAttribute('class', 'scene2-triangle scene2-highlight-green'));
                fig1Arcs.forEach(show);
                fig1Angles.forEach(show);
                fig2Arcs.forEach(show);
                fig2Angles.forEach(show);
                fig3Arcs.forEach(show);
                fig3Angles.forEach(show);
                fig4Arcs.forEach(show);
                fig4Angles.forEach(show);
                show(fig1Type);
                show(fig1Judge);
                show(fig1Conclusion);
                show(fig2Type);
                show(fig2Judge);
                show(fig2Conclusion);
                show(fig3Type);
                show(fig3Judge);
                show(fig3Conclusion);
                show(fig4Type);
                show(fig4Judge);
                show(fig4Conclusion);
            }

            updateButtons();
        }

        // 按钮点击事件
        btnPrev.addEventListener('click', function() {
            if (currentStep > 0) {
                currentStep--;
                updateScene();
            }
        });

        btnNext.addEventListener('click', function() {
            if (currentStep < totalSteps) {
                currentStep++;
                updateScene();
            }
        });

        // 初始化
        updateButtons();
    })();
    </script>
