    <script>
    (function() {
        // Scene 1 实现：正方形与菱形重叠关系及阴影面积计算
        
        // 布局坐标表（实现对照依据）：
        // | 元素 | 类型 | 位置/范围 | 大小 | 标签偏移方向 | 备注 |
        // |------|------|-----------|------|-------------|------|
        // | 正方形 ABCD | 矩形 | (200,200)-(350,350) | 150×150 | - | 黑色实线 2px |
        // | 菱形 PQCB | 四边形 | P(290,230)-Q(440,230)-C(350,350)-B(200,350) | - | - | 黑色实线 2px |
        // | 阴影区域 | 多边形 | A(200,200)-D(350,200)-E(350,230)-P(290,230)-B(200,350) | - | - | 灰色填充 #888888 30% 透明度 |
        // | 辅助线 PH | 虚线 | P(290,230)-H(290,350) | 120×1 | - | 灰色虚线 #888888 1px |
        // | 点 A 标签 | text | (185,195) | ~14×16 | 左上 | 正方形左上顶点 |
        // | 点 B 标签 | text | (185,365) | ~14×16 | 左下 | 正方形左下顶点 |
        // | 点 C 标签 | text | (365,365) | ~14×16 | 右下 | 正方形右下顶点 |
        // | 点 D 标签 | text | (365,195) | ~14×16 | 右上 | 正方形右上顶点 |
        // | 点 P 标签 | text | (290,215) | ~14×16 | 上方 | 菱形顶点，正方形内 |
        // | 点 Q 标签 | text | (455,230) | ~14×16 | 右侧 | 菱形右顶点 |
        // | 点 E 标签 | text | (365,225) | ~14×16 | 右侧 | PQ 与 CD 交点 |
        // | 点 H 标签 | text | (290,365) | ~14×16 | 下方 | P 到 BC 垂足 |
        // | 边长 5 标注 | text | (275,380) | ~14×16 | 下方 | BC 边中点下方，蓝色 |
        // | 高 4 标注 | text | (305,290) | ~14×16 | 右侧 | PH 辅助线右侧，蓝色 |
        // | 阴影面积=7.5 | text | (120,280) | ~70×16 | 左侧 | 画布左侧空白区，红色 |
        // | 直角符号 | 小正方形 | (285,345) | 8×8 | - | H 点处表示垂直 |
        
        // 状态机：currentStep (0-4)
        let currentStep = 0;
        const totalSteps = 4;
        
        // 获取 DOM 元素
        const stepText = document.getElementById('scene1-step-text');
        const btnPrevBg = document.getElementById('scene1-btn-prev-bg');
        const btnPrevText = document.getElementById('scene1-btn-prev-text');
        const btnNextBg = document.getElementById('scene1-btn-next-bg');
        const btnNextText = document.getElementById('scene1-btn-next-text');
        
        // 各步骤需要显示的元素
        const stepElements = {
            0: [], // 初始状态：只显示基础图形和点标签
            1: ['scene1-label-5-bottom', 'scene1-label-5-left'], // 显示边长 5
            2: ['scene1-helper-ph', 'scene1-right-angle', 'scene1-label-4', 'scene1-label-E', 'scene1-label-H'], // 显示辅助线和高 4
            3: ['scene1-shadow'], // 显示阴影区域
            4: ['scene1-result'] // 显示结果
        };
        
        // 工具函数：显示元素
        function showElement(id) {
            const el = document.getElementById(id);
            if (el) {
                el.classList.remove('scene1-hidden');
                if (el.tagName === 'text' || el.tagName === 'tspan') {
                    el.classList.add('scene1-visible-inline');
                } else {
                    el.classList.add('scene1-visible');
                }
            }
        }
        
        // 工具函数：隐藏元素
        function hideElement(id) {
            const el = document.getElementById(id);
            if (el) {
                el.classList.add('scene1-hidden');
                el.classList.remove('scene1-visible');
                el.classList.remove('scene1-visible-inline');
            }
        }
        
        // 更新显示状态
        function updateDisplay() {
            // 更新步骤指示器
            stepText.textContent = `步骤 ${currentStep}/${totalSteps}`;
            
            // 更新按钮状态
            if (currentStep === 0) {
                btnPrevBg.setAttribute('fill', '#cccccc');
                btnPrevText.setAttribute('fill', '#999999');
            } else {
                btnPrevBg.setAttribute('fill', '#e0e0e0');
                btnPrevText.setAttribute('fill', '#333333');
            }
            
            if (currentStep === totalSteps) {
                btnNextBg.setAttribute('fill', '#cccccc');
                btnNextText.setAttribute('fill', '#999999');
            } else {
                btnNextBg.setAttribute('fill', '#e0e0e0');
                btnNextText.setAttribute('fill', '#333333');
            }
            
            // 根据当前步骤显示/隐藏元素
            for (let step = 1; step <= totalSteps; step++) {
                const elements = stepElements[step] || [];
                elements.forEach(id => {
                    if (step <= currentStep) {
                        showElement(id);
                    } else {
                        hideElement(id);
                    }
                });
            }
        }
        
        // 点击事件处理
        function handleClick(evt) {
            const rect = evt.target.getBoundingClientRect();
            const x = evt.clientX - rect.left;
            const y = evt.clientY - rect.top;
            
            // 检查是否点击了上一步按钮
            if (x >= 330 && x <= 380 && y >= 455 && y <= 480) {
                if (currentStep > 0) {
                    currentStep--;
                    updateDisplay();
                }
            }
            // 检查是否点击了下一步按钮
            else if (x >= 420 && x <= 470 && y >= 455 && y <= 480) {
                if (currentStep < totalSteps) {
                    currentStep++;
                    updateDisplay();
                }
            }
        }
        
        // 绑定点击事件到 SVG
        const svg = document.getElementById('scene1-svg');
        svg.addEventListener('click', handleClick);
        
        // 初始化显示
        updateDisplay();
        
    })();
    </script>
