    <script>
    (function() {
        // Scene 2: 全等三角形证明（结论① △AED≌△AEF）
        
        // 状态配置（步骤 0-5）
        const steps = [
            {
                // 步骤 0：初始状态
                highlights: [],
                arcs: [],
                annotations: [],
                congruent: false,
                sas: false,
                deef: false,
                status: "观察△AED 和△AEF，准备证明全等"
            },
            {
                // 步骤 1：AD=AF 已知
                highlights: ['scene2-highlight-ad', 'scene2-highlight-af'],
                arcs: [],
                annotations: ['scene2-anno-adaf', 'scene2-anno-adaf-small'],
                congruent: false,
                sas: false,
                deef: false,
                status: "条件 1：AD=AF（已知条件）"
            },
            {
                // 步骤 2：AE=AE 公共边
                highlights: ['scene2-highlight-ad', 'scene2-highlight-af', 'scene2-highlight-ae'],
                arcs: [],
                annotations: ['scene2-anno-adaf', 'scene2-anno-adaf-small', 'scene2-anno-aeae', 'scene2-anno-aeae-small'],
                congruent: false,
                sas: false,
                deef: false,
                status: "条件 2：AE=AE（公共边）"
            },
            {
                // 步骤 3：角度相等
                highlights: ['scene2-highlight-ad', 'scene2-highlight-af', 'scene2-highlight-ae'],
                arcs: ['scene2-arc-dae', 'scene2-arc-fae'],
                annotations: ['scene2-anno-adaf', 'scene2-anno-adaf-small', 'scene2-anno-aeae', 'scene2-anno-aeae-small', 'scene2-anno-angle', 'scene2-anno-angle-small'],
                congruent: false,
                sas: false,
                deef: false,
                status: "条件 3：∠DAE=∠FAE=45°（由∠DAF=90°，∠DAE=45°推导）"
            },
            {
                // 步骤 4：SAS 全等结论
                highlights: ['scene2-highlight-ad', 'scene2-highlight-af', 'scene2-highlight-ae'],
                arcs: ['scene2-arc-dae', 'scene2-arc-fae'],
                annotations: ['scene2-anno-adaf', 'scene2-anno-adaf-small', 'scene2-anno-aeae', 'scene2-anno-aeae-small', 'scene2-anno-angle', 'scene2-anno-angle-small'],
                congruent: true,
                sas: true,
                deef: false,
                status: "根据 SAS 判定，△AED≌△AEF"
            },
            {
                // 步骤 5：对应边相等
                highlights: ['scene2-highlight-ad', 'scene2-highlight-af', 'scene2-highlight-ae', 'scene2-highlight-de', 'scene2-highlight-ef'],
                arcs: ['scene2-arc-dae', 'scene2-arc-fae'],
                annotations: ['scene2-anno-adaf', 'scene2-anno-adaf-small', 'scene2-anno-aeae', 'scene2-anno-aeae-small', 'scene2-anno-angle', 'scene2-anno-angle-small', 'scene2-anno-deef'],
                congruent: true,
                sas: true,
                deef: true,
                status: "全等三角形对应边相等：DE=EF（结论①得证）"
            }
        ];
        
        let currentStep = 0;
        
        // 获取元素
        const btnPrev = document.getElementById('scene2-btn-prev');
        const btnNext = document.getElementById('scene2-btn-next');
        const statusText = document.getElementById('scene2-status-text');
        
        // 辅助函数：显示/隐藏元素
        function showElement(id) {
            const el = document.getElementById(id);
            if (el) {
                el.style.opacity = '1';
                el.style.display = '';
            }
        }
        
        function hideElement(id) {
            const el = document.getElementById(id);
            if (el) {
                el.style.opacity = '0';
                el.style.display = 'none';
            }
        }
        
        // 渲染当前步骤
        function renderStep(stepIndex) {
            const step = steps[stepIndex];
            
            // 重置所有高亮
            ['scene2-highlight-ad', 'scene2-highlight-af', 'scene2-highlight-ae', 'scene2-highlight-de', 'scene2-highlight-ef'].forEach(id => {
                hideElement(id);
            });
            
            // 应用当前步骤的高亮
            step.highlights.forEach(id => {
                showElement(id);
            });
            
            // 重置所有弧线
            ['scene2-arc-dae', 'scene2-arc-fae'].forEach(id => {
                hideElement(id);
            });
            
            // 应用当前步骤的弧线
            step.arcs.forEach(id => {
                showElement(id);
            });
            
            // 重置所有标注
            ['scene2-anno-adaf', 'scene2-anno-adaf-small', 'scene2-anno-aeae', 'scene2-anno-aeae-small', 
             'scene2-anno-angle', 'scene2-anno-angle-small', 'scene2-anno-deef'].forEach(id => {
                hideElement(id);
            });
            
            // 应用当前步骤的标注
            step.annotations.forEach(id => {
                showElement(id);
            });
            
            // 全等结论和 SAS
            if (step.congruent) {
                showElement('scene2-anno-congruent');
            } else {
                hideElement('scene2-anno-congruent');
            }
            
            if (step.sas) {
                showElement('scene2-anno-sas');
            } else {
                hideElement('scene2-anno-sas');
            }
            
            // 更新说明文字
            statusText.textContent = step.status;
            
            // 更新按钮状态
            btnPrev.disabled = (stepIndex === 0);
            btnNext.disabled = (stepIndex === steps.length - 1);
        }
        
        // 事件绑定
        btnPrev.addEventListener('click', function() {
            if (currentStep > 0) {
                currentStep--;
                renderStep(currentStep);
            }
        });
        
        btnNext.addEventListener('click', function() {
            if (currentStep < steps.length - 1) {
                currentStep++;
                renderStep(currentStep);
            }
        });
        
        // 初始化
        renderStep(0);
    })();
    </script>
