    <script>
    (function() {
        // Scene 1 实现：圆台容器形状对比与液体密度分析

        // 布局坐标表（实现对照依据）：
        // | 元素 | 位置/范围 | 备注 |
        // |------|-----------|------|
        // | 左容器轮廓 | x=100-200, y=220-380 | 倒放圆台（上宽下窄） |
        // | 右容器轮廓 | x=500-600, y=220-380 | 正放圆台（上窄下宽） |
        // | 左液体区域 | x=110-190, y=250-380 | 灰色填充+透明度 30% |
        // | 右液体区域 | x=510-590, y=250-380 | 灰色填充+透明度 30% |
        // | 深度标注线 h | x=350, y=250→380 | 红色 |
        // | 桌面基线 | y=380, x=50-650 | 灰色 |
        // | 速度箭头 v | x=120-170, y=200 | 标注 v |
        // | 速度箭头 2v | x=520-570, y=200 | 标注 2v |
        // | 拉力箭头 F₁ | x=200-260, y=300 | 标注 F₁ |
        // | 拉力箭头 F₂ | x=600-660, y=300 | 标注 F₂ |
        // | 标签 ρ₁ | (130, 420) | 左容器下方 |
        // | 标签 ρ₂ | (550, 420) | 右容器下方 |
        // | 标签 V₁ | (150, 235) | 左液面上方 |
        // | 标签 V₂ | (550, 235) | 右液面上方 |
        // | 标签 h | (365, 315) | 深度线右侧 |
        // | 标签 m₁=m₂ | (400, 450) | 底部中央 |
        // | 体积公式框 | (280,100)-(420,140) | 步骤 2 显示 |
        // | 密度结论框 | (280,150)-(420,190) | 步骤 3 显示 |

        // 状态管理
        let currentStep = 0;
        const maxStep = 3;

        // DOM 元素引用
        const btnPrev = document.getElementById('scene1-btn-prev');
        const btnNext = document.getElementById('scene1-btn-next');
        const stepIndicator = document.getElementById('scene1-step-indicator');

        // 步骤相关元素
        const gridLeft = document.getElementById('scene1-grid-left');
        const gridRight = document.getElementById('scene1-grid-right');
        const formulaGroup = document.getElementById('scene1-formula-group');
        const conclusionGroup = document.getElementById('scene1-conclusion-group');

        // 工具函数：显示元素
        function showElement(el) {
            el.classList.remove('scene1-hidden');
            el.classList.add('scene1-visible');
            el.style.display = 'block';
            el.style.opacity = '1';
        }

        // 工具函数：隐藏元素
        function hideElement(el) {
            el.classList.remove('scene1-visible');
            el.classList.add('scene1-hidden');
            el.style.display = 'none';
            el.style.opacity = '0';
        }

        // 更新步骤显示
        function updateStepDisplay() {
            stepIndicator.textContent = `步骤 ${currentStep}/${maxStep}`;

            // 更新按钮状态
            btnPrev.disabled = (currentStep === 0);
            btnNext.disabled = (currentStep === maxStep);

            // 根据步骤显示/隐藏元素
            // 步骤 1：显示网格
            if (currentStep >= 1) {
                showElement(gridLeft);
                showElement(gridRight);
                // 动画效果
                gridLeft.style.transition = 'opacity 0.5s ease-in-out';
                gridRight.style.transition = 'opacity 0.5s ease-in-out';
                setTimeout(() => {
                    gridLeft.style.opacity = '1';
                    gridRight.style.opacity = '1';
                }, 10);
            } else {
                hideElement(gridLeft);
                hideElement(gridRight);
            }

            // 步骤 2：显示体积公式框
            if (currentStep >= 2) {
                showElement(formulaGroup);
                formulaGroup.style.transition = 'opacity 0.5s ease-in-out';
                setTimeout(() => {
                    formulaGroup.style.opacity = '1';
                }, 10);
            } else {
                hideElement(formulaGroup);
            }

            // 步骤 3：显示密度结论框
            if (currentStep >= 3) {
                showElement(conclusionGroup);
                conclusionGroup.style.transition = 'opacity 0.5s ease-in-out';
                setTimeout(() => {
                    conclusionGroup.style.opacity = '1';
                }, 10);
            } else {
                hideElement(conclusionGroup);
            }
        }

        // 下一步
        function nextStep() {
            if (currentStep < maxStep) {
                currentStep++;
                updateStepDisplay();
            }
        }

        // 上一步
        function prevStep() {
            if (currentStep > 0) {
                currentStep--;
                updateStepDisplay();
            }
        }

        // 事件绑定
        btnNext.addEventListener('click', nextStep);
        btnPrev.addEventListener('click', prevStep);

        // 初始化显示
        updateStepDisplay();

    })();
    </script>
