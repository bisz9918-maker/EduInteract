            updateButtons();

            // 更新步骤指示器
            stepIndicator.textContent = `步骤 ${step}/${totalSteps}`;
        }

        // 更新按钮状态
        function updateButtons() {
            if (currentStep === 0) {
                prevBtn.classList.remove('scene2-btn-active');
                prevBtn.classList.add('scene2-btn-disabled');
                prevBtn.disabled = true;
            } else {
                prevBtn.classList.remove('scene2-btn-disabled');
                prevBtn.classList.add('scene2-btn-active');
                prevBtn.disabled = false;
            }

            if (currentStep === totalSteps) {
                nextBtn.classList.remove('scene2-btn-active');
                nextBtn.classList.add('scene2-btn-disabled');
                nextBtn.disabled = true;
            } else {
                nextBtn.classList.remove('scene2-btn-disabled');
                nextBtn.classList.add('scene2-btn-active');
                nextBtn.disabled = false;
            }
        }

        // 上一步
        function prevStep() {
            if (currentStep > 0) {
                currentStep--;
                renderStep(currentStep);
            }
        }

        // 下一步
        function nextStep() {
            if (currentStep < totalSteps) {
                currentStep++;
                renderStep(currentStep);
            }
        }

        // 事件绑定
        prevBtn.addEventListener('click', prevStep);
        nextBtn.addEventListener('click', nextStep);

        // 初始化
        renderStep(0);

    })();
    </script>
</body>
