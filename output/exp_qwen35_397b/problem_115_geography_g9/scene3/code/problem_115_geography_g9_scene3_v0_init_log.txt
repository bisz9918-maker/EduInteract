    <script>
    (function() {
        // Scene 3 实现：东西半球划分与五带范围
        
        // 状态管理
        let currentStep = 0;
        const totalSteps = 5;
        
        // 工具函数：显示元素
        function show(el) {
            if (el) {
                el.classList.remove('scene3-hidden');
                el.classList.add('scene3-visible');
                el.style.display = 'block';
                el.style.opacity = '1';
            }
        }
        
        // 工具函数：隐藏元素
        function hide(el) {
            if (el) {
                el.classList.remove('scene3-visible');
                el.classList.add('scene3-hidden');
                el.style.display = 'none';
                el.style.opacity = '0';
            }
        }
        
        // 获取所有需要控制的元素
        const elements = {
            // 填充区域
            tropicZone: document.getElementById('scene3-tropic-zone'),
            northTempZone: document.getElementById('scene3-north-temp-zone'),
            southTempZone: document.getElementById('scene3-south-temp-zone'),
            
            // 分界线
            hemisphereLine: document.getElementById('scene3-hemisphere-line'),
            
            // 乙点高亮
            pointYiHighlight: document.getElementById('scene3-point-yi-highlight'),
            
            // 右侧说明
            hemisphereInfo: document.getElementById('scene3-hemisphere-info'),
            hemisphereInfo2: document.getElementById('scene3-hemisphere-info2'),
            tropicInfo: document.getElementById('scene3-tropic-info'),
            tropicInfo2: document.getElementById('scene3-tropic-info2'),
            northTempInfo: document.getElementById('scene3-north-temp-info'),
            northTempInfo2: document.getElementById('scene3-north-temp-info2'),
            
            // 条件说明
            conditionTitle: document.getElementById('scene3-condition-title'),
            condition1: document.getElementById('scene3-condition-1'),
            condition2: document.getElementById('scene3-condition-2'),
            condition3: document.getElementById('scene3-condition-3'),
            condition4: document.getElementById('scene3-condition-4'),
            
            // 答案
            answerText: document.getElementById('scene3-answer-text')
        };
        
        // 按钮和指示器
        const btnPrev = document.getElementById('scene3-btn-prev');
        const btnNext = document.getElementById('scene3-btn-next');
        const stepIndicator = document.getElementById('scene3-step-indicator');
        
        // 更新步骤显示
        function updateStepIndicator() {
            stepIndicator.textContent = `步骤 ${currentStep}/${totalSteps}`;
            btnPrev.disabled = (currentStep === 0);
            btnNext.disabled = (currentStep === totalSteps);
            if (currentStep === totalSteps) {
                btnNext.textContent = '已完成';
            } else {
                btnNext.textContent = '下一步 ▶';
            }
        }
        
        // 重置所有动态元素到初始状态
        function resetAll() {
            // 隐藏所有填充区域
            hide(elements.tropicZone);
            hide(elements.northTempZone);
            hide(elements.southTempZone);
            
            // 隐藏分界线
            hide(elements.hemisphereLine);
            
            // 隐藏乙点高亮
            hide(elements.pointYiHighlight);
            
            // 隐藏所有说明文字
            hide(elements.hemisphereInfo);
            hide(elements.hemisphereInfo2);
            hide(elements.tropicInfo);
            hide(elements.tropicInfo2);
            hide(elements.northTempInfo);
            hide(elements.northTempInfo2);
            hide(elements.conditionTitle);
            hide(elements.condition1);
            hide(elements.condition2);
            hide(elements.condition3);
            hide(elements.condition4);
            hide(elements.answerText);
        }
        
        // 根据当前步骤更新显示
        function updateDisplay() {
            resetAll();
            
            if (currentStep >= 1) {
                // 步骤 1：显示东西半球分界线
                show(elements.hemisphereLine);
                show(elements.hemisphereInfo);
                show(elements.hemisphereInfo2);
            }
            
            if (currentStep >= 2) {
                // 步骤 2：显示热带范围
                elements.tropicZone.classList.add('scene3-fade-in');
                show(elements.tropicZone);
                show(elements.tropicInfo);
                show(elements.tropicInfo2);
            }
            
            if (currentStep >= 3) {
                // 步骤 3：显示北温带范围
                elements.northTempZone.classList.add('scene3-fade-in');
                show(elements.northTempZone);
                show(elements.northTempInfo);
                show(elements.northTempInfo2);
            }
            
            if (currentStep >= 4) {
                // 步骤 4：显示综合判断条件
                show(elements.conditionTitle);
                show(elements.condition1);
                show(elements.condition2);
                show(elements.condition3);
                show(elements.condition4);
            }
            
            if (currentStep >= 5) {
                // 步骤 5：高亮乙点并显示答案
                show(elements.pointYiHighlight);
                show(elements.answerText);
            }
            
            updateStepIndicator();
        }
        
        // 上一步按钮点击处理
        btnPrev.addEventListener('click', function() {
            if (currentStep > 0) {
                currentStep--;
                updateDisplay();
            }
        });
        
        // 下一步按钮点击处理
        btnNext.addEventListener('click', function() {
            if (currentStep < totalSteps) {
                currentStep++;
                updateDisplay();
            }
        });
        
        // 初始化显示
        updateStepIndicator();
        
    })();
    </script>
