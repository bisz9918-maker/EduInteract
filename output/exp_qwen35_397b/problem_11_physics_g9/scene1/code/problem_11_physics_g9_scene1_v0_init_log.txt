    <script>
    (function() {
        // Scene 1 实现
        const currentStep = { value: 0 };
        const totalSteps = 6;

        // 获取 DOM 元素
        const btnPrev = document.getElementById('scene1-btn-prev');
        const btnNext = document.getElementById('scene1-btn-next');
        const stepIndicator = document.getElementById('scene1-step-indicator');

        // 动态元素
        const slideArrow = document.getElementById('scene1-slide-arrow');
        const rDecrease = document.getElementById('scene1-r-decrease');
        const iIncrease = document.getElementById('scene1-i-increase');
        const uDecrease = document.getElementById('scene1-u-decrease');
        const magneticLines = document.getElementById('scene1-magnetic-lines');
        const repelArrow = document.getElementById('scene1-repel-arrow');
        const conclusion = document.getElementById('scene1-conclusion');
        const currentArrows = document.getElementById('scene1-current-arrows');
        const coilN = document.getElementById('scene1-coil-n');
        const springCoil = document.getElementById('scene1-spring-coil');
        const vNeedle = document.getElementById('scene1-v-needle');
        const aNeedle = document.getElementById('scene1-a-needle');
        const rElement = document.getElementById('scene1-r');
        const vElement = document.getElementById('scene1-v');
        const aElement = document.getElementById('scene1-a');

        // 显示/隐藏工具函数
        function show(el) {
            if (el) {
                el.style.display = '';
                el.style.opacity = '1';
            }
        }

        function hide(el) {
            if (el) {
                el.style.display = 'none';
                el.style.opacity = '0';
            }
        }

        // 更新按钮状态
        function updateButtons() {
            btnPrev.disabled = currentStep.value === 0;
            btnNext.disabled = currentStep.value === totalSteps - 1;
            btnPrev.style.opacity = btnPrev.disabled ? '0.5' : '1';
            btnNext.style.opacity = btnNext.disabled ? '0.5' : '1';
        }

        // 更新步骤指示器
        function updateIndicator() {
            stepIndicator.textContent = `步骤 ${currentStep.value + 1}/${totalSteps}`;
        }

        // 重置所有动态元素
        function resetAll() {
            hide(slideArrow);
            hide(rDecrease);
            hide(iIncrease);
            hide(uDecrease);
            hide(magneticLines);
            hide(repelArrow);
            hide(conclusion);
            hide(currentArrows);
            hide(coilN);
            
            // 重置电表指针
            vNeedle.style.transform = 'rotate(0deg)';
            aNeedle.style.transform = 'rotate(0deg)';
            
            // 重置高亮
            rElement.querySelector('rect').classList.remove('highlight-blue');
            vElement.querySelector('circle').classList.remove('highlight-red');
            aElement.querySelector('circle').classList.remove('highlight-green');
            
            // 重置弹簧
            springCoil.setAttribute('d', 'M550,50 L545,60 L555,70 L545,80 L555,90 L545,100 L555,110 L550,120');
        }

        // 渲染当前步骤
        function renderStep(step) {
            resetAll();

            switch(step) {
                case 0:
                    // 初始状态：只显示基础电路
                    break;

                case 1:
                    // 步骤 1：高亮滑动变阻器，显示滑片移动箭头
                    show(slideArrow);
                    rElement.querySelector('rect').classList.add('highlight-blue');
                    break;

                case 2:
                    // 步骤 2：显示电阻减小
                    show(slideArrow);
                    show(rDecrease);
                    rElement.querySelector('rect').classList.add('highlight-blue');
                    break;

                case 3:
                    // 步骤 3：显示电流增大
                    show(slideArrow);
                    show(rDecrease);
                    show(iIncrease);
                    show(currentArrows);
                    rElement.querySelector('rect').classList.add('highlight-blue');
                    aElement.querySelector('circle').classList.add('highlight-green');
                    aNeedle.style.transform = 'rotate(15deg)';
                    break;

                case 4:
                    // 步骤 4：显示电压减小
                    show(slideArrow);
                    show(rDecrease);
                    show(iIncrease);
                    show(uDecrease);
                    show(currentArrows);
                    rElement.querySelector('rect').classList.add('highlight-blue');
                    aElement.querySelector('circle').classList.add('highlight-green');
                    vElement.querySelector('circle').classList.add('highlight-red');
                    aNeedle.style.transform = 'rotate(15deg)';
                    vNeedle.style.transform = 'rotate(-10deg)';
                    break;

                case 5:
                    // 步骤 5：显示电磁效应和结论
                    show(slideArrow);
                    show(rDecrease);
                    show(iIncrease);
                    show(uDecrease);
                    show(currentArrows);
                    show(magneticLines);
                    show(repelArrow);
                    show(conclusion);
                    show(coilN);
                    rElement.querySelector('rect').classList.add('highlight-blue');
                    aElement.querySelector('circle').classList.add('highlight-green');
                    vElement.querySelector('circle').classList.add('highlight-red');
                    aNeedle.style.transform = 'rotate(15deg)';
                    vNeedle.style.transform = 'rotate(-10deg)';
                    // 弹簧缩短（向上移动）
                    springCoil.setAttribute('d', 'M550,50 L545,58 L555,66 L545,74 L555,82 L545,90 L555,98 L550,106');
                    break;
            }

            updateButtons();
            updateIndicator();
        }

        // 按钮事件
        btnPrev.addEventListener('click', function() {
            if (currentStep.value > 0) {
                currentStep.value--;
                renderStep(currentStep.value);
            }
        });

        btnNext.addEventListener('click', function() {
            if (currentStep.value < totalSteps - 1) {
                currentStep.value++;
                renderStep(currentStep.value);
            }
        });

        // 初始化
        renderStep(0);
    })();
    </script>
