    <script>
    (function() {
        // Scene 2 实现
        let currentState = 0;
        const totalSteps = 4;

        const btnPrev = document.getElementById('scene2-btn-prev');
        const btnNext = document.getElementById('scene2-btn-next');
        const stepIndicator = document.getElementById('scene2-step-indicator');

        function showStep(step) {
            // 隐藏所有步骤
            for (let i = 0; i < totalSteps; i++) {
                const stepEl = document.getElementById('scene2-step' + i);
                if (stepEl) {
                    stepEl.classList.remove('scene2-visible');
                    stepEl.classList.add('scene2-hidden');
                }
            }
            // 显示当前步骤
            const currentStepEl = document.getElementById('scene2-step' + step);
            if (currentStepEl) {
                currentStepEl.classList.remove('scene2-hidden');
                currentStepEl.classList.add('scene2-visible');
            }
            // 更新按钮状态
            btnPrev.disabled = (step === 0);
            btnNext.disabled = (step === totalSteps - 1);
            // 更新步骤指示器
            stepIndicator.textContent = '步骤 ' + (step + 1) + '/' + totalSteps;
        }

        btnPrev.addEventListener('click', function() {
            if (currentState > 0) {
                currentState--;
                showStep(currentState);
            }
        });

        btnNext.addEventListener('click', function() {
            if (currentState < totalSteps - 1) {
                currentState++;
                showStep(currentState);
            }
        });

        // 初始化
        showStep(currentState);
    })();
    </script>
