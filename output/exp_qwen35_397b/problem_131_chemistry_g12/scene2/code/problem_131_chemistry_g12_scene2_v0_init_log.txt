    <script>
    (function() {
        // Scene 2 实现：四电极的电极反应式及产物分析
        
        const currentState = { step: 0 };
        const totalSteps = 5;
        
        // 更新步骤指示器
        function updateStepIndicator() {
            document.getElementById('scene2-step-indicator').textContent = `步骤 ${currentState.step}/${totalSteps}`;
            document.getElementById('scene2-step-indicator-html').textContent = `步骤 ${currentState.step}/${totalSteps}`;
            
            // 更新按钮状态
            document.getElementById('scene2-prev-btn').disabled = currentState.step === 0;
            document.getElementById('scene2-next-btn').disabled = currentState.step === totalSteps;
        }
        
        // 隐藏所有步骤相关元素
        function hideAllStepElements() {
            // 隐藏所有反应式组
            document.getElementById('scene2-zn-reaction-group').classList.add('scene2-hidden');
            document.getElementById('scene2-c-reaction-group').classList.add('scene2-hidden');
            document.getElementById('scene2-pt-reaction-group').classList.add('scene2-hidden');
            document.getElementById('scene2-cu-reaction-group').classList.add('scene2-hidden');
            document.getElementById('scene2-options-group').classList.add('scene2-hidden');
            
            // 恢复所有电极默认样式
            ['pt', 'cu', 'zn', 'c'].forEach(el => {
                const electrode = document.getElementById(`scene2-${el}-electrode`);
                electrode.setAttribute('stroke', '#000000');
                electrode.setAttribute('stroke-width', '2');
            });
            
            // 开关状态
            document.getElementById('scene2-switch-open').classList.remove('scene2-hidden');
            document.getElementById('scene2-switch-closed').classList.add('scene2-hidden');
        }
        
        // 更新场景状态
        function updateScene() {
            hideAllStepElements();
            
            const step = currentState.step;
            
            if (step === 0) {
                // 初始状态：只显示基础装置
            } else if (step === 1) {
                // Zn 电极反应
                document.getElementById('scene2-zn-reaction-group').classList.remove('scene2-hidden');
                document.getElementById('scene2-zn-reaction-group').classList.add('scene2-visible');
                const znElectrode = document.getElementById('scene2-zn-electrode');
                znElectrode.setAttribute('stroke', '#1565c0');
                znElectrode.setAttribute('stroke-width', '4');
                document.getElementById('scene2-switch-open').classList.add('scene2-hidden');
                document.getElementById('scene2-switch-closed').classList.remove('scene2-hidden');
            } else if (step === 2) {
                // C 电极反应
                document.getElementById('scene2-c-reaction-group').classList.remove('scene2-hidden');
                document.getElementById('scene2-c-reaction-group').classList.add('scene2-visible');
                const cElectrode = document.getElementById('scene2-c-electrode');
                cElectrode.setAttribute('stroke', '#1565c0');
                cElectrode.setAttribute('stroke-width', '4');
                document.getElementById('scene2-switch-open').classList.add('scene2-hidden');
                document.getElementById('scene2-switch-closed').classList.remove('scene2-hidden');
            } else if (step === 3) {
                // Pt 电极反应
                document.getElementById('scene2-pt-reaction-group').classList.remove('scene2-hidden');
                document.getElementById('scene2-pt-reaction-group').classList.add('scene2-visible');
                const ptElectrode = document.getElementById('scene2-pt-electrode');
                ptElectrode.setAttribute('stroke', '#1565c0');
                ptElectrode.setAttribute('stroke-width', '4');
                document.getElementById('scene2-switch-open').classList.add('scene2-hidden');
                document.getElementById('scene2-switch-closed').classList.remove('scene2-hidden');
            } else if (step === 4) {
                // Cu 电极反应
                document.getElementById('scene2-cu-reaction-group').classList.remove('scene2-hidden');
                document.getElementById('scene2-cu-reaction-group').classList.add('scene2-visible');
                const cuElectrode = document.getElementById('scene2-cu-electrode');
                cuElectrode.setAttribute('stroke', '#1565c0');
                cuElectrode.setAttribute('stroke-width', '4');
                document.getElementById('scene2-switch-open').classList.add('scene2-hidden');
                document.getElementById('scene2-switch-closed').classList.remove('scene2-hidden');
            } else if (step === 5) {
                // 选项判断汇总
                document.getElementById('scene2-options-group').classList.remove('scene2-hidden');
                document.getElementById('scene2-options-group').classList.add('scene2-visible');
                document.getElementById('scene2-switch-open').classList.add('scene2-hidden');
                document.getElementById('scene2-switch-closed').classList.remove('scene2-hidden');
            }
            
            updateStepIndicator();
        }
        
        // 按钮事件绑定
        document.getElementById('scene2-prev-btn').addEventListener('click', function() {
            if (currentState.step > 0) {
                currentState.step--;
                updateScene();
            }
        });
        
        document.getElementById('scene2-next-btn').addEventListener('click', function() {
            if (currentState.step < totalSteps) {
                currentState.step++;
                updateScene();
            }
        });
        
        // 初始化
        updateStepIndicator();
    })();
    </script>
