    <script>
    (function() {
        // Scene 1 实现：活塞位置变化与气体体积关系

        // 状态管理
        let currentState = 0;
        const totalStates = 4;

        // DOM 元素引用
        const btnPrev = document.getElementById('scene1-btn-prev');
        const btnNext = document.getElementById('scene1-btn-next');
        const stepIndicator = document.getElementById('scene1-step-indicator');

        // SVG 元素引用
        const reactionText = document.getElementById('scene1-reaction');
        const pistonInitial = document.getElementById('scene1-piston-initial');
        const pistonFinal = document.getElementById('scene1-piston-final');
        const leftArrow = document.getElementById('scene1-left-arrow');
        const rightArrow = document.getElementById('scene1-right-arrow');
        const mixLabel = document.getElementById('scene1-mix-label');
        const remainingLabel = document.getElementById('scene1-remaining-label');
        const vol1 = document.getElementById('scene1-vol-1');
        const vol3 = document.getElementById('scene1-vol-3');
        const vol2Left = document.getElementById('scene1-vol-2-left');
        const vol2Right = document.getElementById('scene1-vol-2-right');
        const pistonLabel = document.getElementById('scene1-piston-label');
        const centerLabel = document.getElementById('scene1-center-label');
        const rightZone = document.getElementById('scene1-right-zone');

        // 工具函数：显示元素
        function show(el) {
            if (el) {
                el.classList.remove('scene1-svg-hidden');
                el.classList.remove('scene1-hidden');
                el.style.display = 'block';
                el.style.opacity = '1';
            }
        }

        // 工具函数：隐藏元素
        function hide(el) {
            if (el) {
                el.classList.add('scene1-svg-hidden');
                el.classList.add('scene1-hidden');
                el.style.display = 'none';
                el.style.opacity = '0';
            }
        }

        // 更新状态显示
        function updateState() {
            // 隐藏所有动态元素
            hide(reactionText);
            hide(leftArrow);
            hide(rightArrow);
            hide(pistonFinal);
            hide(remainingLabel);
            hide(vol2Left);
            hide(vol2Right);
            hide(centerLabel);
            show(pistonInitial);
            show(mixLabel);
            show(vol1);
            show(vol3);
            show(pistonLabel);

            // 右区填充恢复红色
            rightZone.setAttribute('fill', '#ffcdd2');

            // 根据当前状态显示相应元素
            switch (currentState) {
                case 0:
                    // 初始状态：只显示基础元素
                    break;

                case 1:
                    // 点燃瞬间：显示反应方程式和左弹箭头
                    show(reactionText);
                    show(leftArrow);
                    break;

                case 2:
                    // 恢复温度：显示右滑箭头，右区变蓝色
                    show(reactionText);
                    show(rightArrow);
                    rightZone.setAttribute('fill', '#bbdefb');
                    break;

                case 3:
                    // 最终状态：活塞在中央，左右体积相等
                    hide(pistonInitial);
                    show(pistonFinal);
                    hide(mixLabel);
                    show(remainingLabel);
                    hide(vol1);
                    hide(vol3);
                    show(vol2Left);
                    show(vol2Right);
                    hide(pistonLabel);
                    show(centerLabel);
                    rightZone.setAttribute('fill', '#bbdefb');
                    // 调整右区位置（从 x=242 移到 x=402）
                    rightZone.setAttribute('x', '402');
                    rightZone.setAttribute('width', '316');
                    break;
            }

            // 更新按钮状态
            btnPrev.disabled = (currentState === 0);
            btnNext.disabled = (currentState === totalStates - 1);

            // 更新步骤指示器
            stepIndicator.textContent = `步骤 ${currentState + 1}/${totalStates}`;
        }

        // 事件处理：上一步
        btnPrev.addEventListener('click', function() {
            if (currentState > 0) {
                currentState--;
                updateState();
            }
        });

        // 事件处理：下一步
        btnNext.addEventListener('click', function() {
            if (currentState < totalStates - 1) {
                currentState++;
                updateState();
            }
        });

        // 初始化
        updateState();

    })();
    </script>
