    <script>
    (function() {
        // Scene 1: 溶液导电性微观原理
        // 布局坐标表参考：
        // | 元素 | 位置 | 备注 |
        // |------|------|------|
        // | 电池组 | (60,40)-(220,85) | 两节电池串联 |
        // | 灯泡 | (110,205) r=22 | 可变化亮度 |
        // | 烧杯 | (530,170)-(710,370) | 梯形轮廓 |
        // | 电极A(阳极) | (550,180)-(575,360) | 灰色 |
        // | 电极B(阴极) | (665,180)-(690,360) | 灰色 |
        // | 阳离子 | 红色圆形 d=12 | 带+号 |
        // | 阴离子 | 蓝色圆形 d=12 | 带-号 |
        // | 控件栏 | 底部居中 | 上一步/下一步按钮 |

        const STATE = {
            currentStep: 0,
            totalSteps: 4
        };

        // 每步需要显示的元素 ID 列表
        const STEP_ELEMENTS = {
            0: {
                show: [],
                hide: ['scene1-battery-plus', 'scene1-battery-minus', 'scene1-label-anode', 'scene1-label-cathode',
                       'scene1-cation-1', 'scene1-cation-2', 'scene1-cation-3', 'scene1-cation-4',
                       'scene1-anion-1', 'scene1-anion-2', 'scene1-anion-3', 'scene1-anion-4',
                       'scene1-arrow-cation', 'scene1-arrow-anion', 'scene1-label-cation', 'scene1-label-anion',
                       'scene1-label-current', 'scene1-arrow-current', 'scene1-wire1-flow', 'scene1-wire2-flow', 'scene1-wire3-flow']
            },
            1: {
                show: ['scene1-battery-plus', 'scene1-battery-minus', 'scene1-label-anode', 'scene1-label-cathode'],
                hide: ['scene1-cation-1', 'scene1-cation-2', 'scene1-cation-3', 'scene1-cation-4',
                       'scene1-anion-1', 'scene1-anion-2', 'scene1-anion-3', 'scene1-anion-4',
                       'scene1-arrow-cation', 'scene1-arrow-anion', 'scene1-label-cation', 'scene1-label-anion',
                       'scene1-label-current', 'scene1-arrow-current', 'scene1-wire1-flow', 'scene1-wire2-flow', 'scene1-wire3-flow']
            },
            2: {
                show: ['scene1-cation-1', 'scene1-cation-2', 'scene1-cation-3', 'scene1-cation-4',
                       'scene1-anion-1', 'scene1-anion-2', 'scene1-anion-3', 'scene1-anion-4',
                       'scene1-arrow-cation', 'scene1-arrow-anion', 'scene1-label-cation', 'scene1-label-anion'],
                hide: ['scene1-label-current', 'scene1-arrow-current', 'scene1-wire1-flow', 'scene1-wire2-flow', 'scene1-wire3-flow']
            },
            3: {
                show: ['scene1-label-current', 'scene1-arrow-current', 'scene1-wire1-flow', 'scene1-wire2-flow', 'scene1-wire3-flow'],
                hide: []
            }
        };

        // 工具函数：显示元素
        function showElement(id) {
            const el = document.getElementById(id);
            if (el) {
                el.style.opacity = '1';
                if (el.tagName === 'path' && id.includes('-flow')) {
                    el.style.display = 'block';
                }
            }
        }

        // 工具函数：隐藏元素
        function hideElement(id) {
            const el = document.getElementById(id);
            if (el) {
                el.style.opacity = '0';
                if (el.tagName === 'path' && id.includes('-flow')) {
                    el.style.display = 'none';
                }
            }
        }

        // 更新灯泡状态
        function updateBulb(step) {
            const bulb = document.getElementById('scene1-bulb');
            if (step >= 3) {
                // 发光状态
                bulb.setAttribute('fill', '#ffd54f');
                bulb.classList.add('scene1-bulb-glow');
            } else {
                // 灰色状态
                bulb.setAttribute('fill', '#cccccc');
                bulb.classList.remove('scene1-bulb-glow');
            }
        }

        // 更新箭头样式（步骤3变实线）
        function updateArrows(step) {
            const arrowCation = document.getElementById('scene1-arrow-cation');
            const arrowAnion = document.getElementById('scene1-arrow-anion');
            if (step >= 3) {
                arrowCation.setAttribute('stroke', '#333333');
                arrowCation.setAttribute('stroke-dasharray', 'none');
                arrowCation.setAttribute('marker-end', 'url(#scene1-arrowhead-dark)');
                arrowAnion.setAttribute('stroke', '#333333');
                arrowAnion.setAttribute('stroke-dasharray', 'none');
                arrowAnion.setAttribute('marker-end', 'url(#scene1-arrowhead-dark)');
            } else if (step >= 2) {
                arrowCation.setAttribute('stroke', '#888888');
                arrowCation.setAttribute('stroke-dasharray', '5,3');
                arrowCation.setAttribute('marker-end', 'url(#scene1-arrowhead)');
                arrowAnion.setAttribute('stroke', '#888888');
                arrowAnion.setAttribute('stroke-dasharray', '5,3');
                arrowAnion.setAttribute('marker-end', 'url(#scene1-arrowhead)');
            }
        }

        // 更新步骤显示
        function updateStepDisplay() {
            const step = STATE.currentStep;

            // 隐藏所有需要隐藏的元素
            STEP_ELEMENTS[step].hide.forEach(id => hideElement(id));

            // 显示所有需要显示的元素
            STEP_ELEMENTS[step].show.forEach(id => showElement(id));

            // 更新灯泡状态
            updateBulb(step);

            // 更新箭头样式
            updateArrows(step);

            // 更新步骤指示器
            const indicator = document.getElementById('scene1-step-indicator');
            indicator.textContent = `步骤 ${step + 1}/${STATE.totalSteps}`;

            // 更新按钮状态
            const btnPrev = document.getElementById('scene1-btn-prev');
            const btnNext = document.getElementById('scene1-btn-next');
            btnPrev.disabled = (step === 0);
            btnNext.disabled = (step === STATE.totalSteps - 1);
            btnNext.textContent = (step === STATE.totalSteps - 1) ? '完成' : '下一步 ▶';
        }

        // 初始化
        function init() {
            const btnPrev = document.getElementById('scene1-btn-prev');
            const btnNext = document.getElementById('scene1-btn-next');

            btnPrev.addEventListener('click', function() {
                if (STATE.currentStep > 0) {
                    STATE.currentStep--;
                    updateStepDisplay();
                }
            });

            btnNext.addEventListener('click', function() {
                if (STATE.currentStep < STATE.totalSteps - 1) {
                    STATE.currentStep++;
                    updateStepDisplay();
                }
            });

            // 初始状态
            updateStepDisplay();
        }

        // 启动
        init();
    })();
    </script>
