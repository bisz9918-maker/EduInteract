    (function() {
        // Scene 3 实现 - 多状态切换交互
        
        let currentState = 'none';
        
        // 获取所有元素
        const cardA = document.getElementById('scene3-card-a-bg');
        const cardB = document.getElementById('scene3-card-b-bg');
        const cardC = document.getElementById('scene3-card-c-bg');
        const cardD = document.getElementById('scene3-card-d-bg');
        
        const analysisA = document.getElementById('scene3-analysis-a');
        const analysisB = document.getElementById('scene3-analysis-b');
        const analysisC = document.getElementById('scene3-analysis-c');
        const analysisD = document.getElementById('scene3-analysis-d');
        
        const markA = document.getElementById('scene3-mark-a');
        const markB = document.getElementById('scene3-mark-b');
        const markC = document.getElementById('scene3-mark-c');
        const markD = document.getElementById('scene3-mark-d');
        
        const btnA = document.getElementById('scene3-btn-a');
        const btnB = document.getElementById('scene3-btn-b');
        const btnC = document.getElementById('scene3-btn-c');
        const btnD = document.getElementById('scene3-btn-d');
        
        const btnAText = document.getElementById('scene3-btn-a-text');
        const btnBText = document.getElementById('scene3-btn-b-text');
        const btnCText = document.getElementById('scene3-btn-c-text');
        const btnDText = document.getElementById('scene3-btn-d-text');
        
        // 分子图高亮元素
        const bondC3C4 = document.getElementById('scene3-bond-c3-c4');
        const bondC7C8 = document.getElementById('scene3-bond-c7-c8');
        const bondC9X = document.getElementById('scene3-bond-c9-x');
        const atomX3 = document.getElementById('scene3-atom-x3');
        const atomC9 = document.getElementById('scene3-atom-c9');
        
        // 重置所有卡片为初始状态
        function resetAll() {
            // 重置卡片样式
            cardA.setAttribute('class', 'scene3-card');
            cardB.setAttribute('class', 'scene3-card');
            cardC.setAttribute('class', 'scene3-card');
            cardD.setAttribute('class', 'scene3-card');
            
            // 隐藏分析文字和标记
            analysisA.style.display = 'none';
            analysisB.style.display = 'none';
            analysisC.style.display = 'none';
            analysisD.style.display = 'none';
            
            markA.style.display = 'none';
            markB.style.display = 'none';
            markC.style.display = 'none';
            markD.style.display = 'none';
            
            // 重置按钮样式
            btnA.setAttribute('class', 'scene3-btn');
            btnB.setAttribute('class', 'scene3-btn');
            btnC.setAttribute('class', 'scene3-btn');
            btnD.setAttribute('class', 'scene3-btn');
            
            btnAText.setAttribute('fill', '#1565c0');
            btnBText.setAttribute('fill', '#1565c0');
            btnCText.setAttribute('fill', '#1565c0');
            btnDText.setAttribute('fill', '#1565c0');
            
            // 重置分子图高亮
            bondC3C4.setAttribute('class', 'scene3-bond');
            bondC7C8.setAttribute('class', 'scene3-bond');
            bondC9X.setAttribute('class', 'scene3-bond');
            atomX3.setAttribute('class', 'scene3-atom-x');
            atomC9.setAttribute('class', 'scene3-atom-c');
        }
        
        // 激活选项A
        function activateA() {
            resetAll();
            cardA.setAttribute('class', 'scene3-card scene3-card-a-active');
            analysisA.style.display = 'block';
            markA.style.display = 'block';
            btnA.setAttribute('class', 'scene3-btn scene3-btn-active');
            btnAText.setAttribute('fill', '#ffffff');
            // 高亮③处原子
            atomX3.setAttribute('class', 'scene3-atom-x scene3-atom-highlight');
            currentState = 'A';
        }
        
        // 激活选项B
        function activateB() {
            resetAll();
            cardB.setAttribute('class', 'scene3-card scene3-card-b-active');
            analysisB.style.display = 'block';
            markB.style.display = 'block';
            btnB.setAttribute('class', 'scene3-btn scene3-btn-active');
            btnBText.setAttribute('fill', '#ffffff');
            // 高亮①处化学键 (C3-C4)
            bondC3C4.setAttribute('class', 'scene3-bond scene3-bond-highlight');
            currentState = 'B';
        }
        
        // 激活选项C
        function activateC() {
            resetAll();
            cardC.setAttribute('class', 'scene3-card scene3-card-c-active');
            analysisC.style.display = 'block';
            markC.style.display = 'block';
            btnC.setAttribute('class', 'scene3-btn scene3-btn-active');
            btnCText.setAttribute('fill', '#ffffff');
            // 高亮②处化学键 (C7-C8)
            bondC7C8.setAttribute('class', 'scene3-bond scene3-bond-highlight');
            currentState = 'C';
        }
        
        // 激活选项D
        function activateD() {
            resetAll();
            cardD.setAttribute('class', 'scene3-card scene3-card-d-active');
            analysisD.style.display = 'block';
            markD.style.display = 'block';
            btnD.setAttribute('class', 'scene3-btn scene3-btn-active');
            btnDText.setAttribute('fill', '#ffffff');
            // 高亮③处原子和C9-C8键
            atomX3.setAttribute('class', 'scene3-atom-x scene3-atom-highlight');
            bondC9X.setAttribute('class', 'scene3-bond scene3-bond-highlight');
            currentState = 'D';
        }
        
        // 绑定按钮点击事件
        btnA.addEventListener('click', activateA);
        btnB.addEventListener('click', activateB);
        btnC.addEventListener('click', activateC);
        btnD.addEventListener('click', activateD);
        
    })();
    </script>
