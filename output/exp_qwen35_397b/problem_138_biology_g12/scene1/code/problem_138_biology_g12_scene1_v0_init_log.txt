    <script>
    (function() {
        // Scene 1 实现：四种处理方法流程总览
        // 布局坐标表：
        // 植株甲外框：(50,40)-(190,160), 植株甲图标：中心(120,100)
        // 植株甲标签：(55,30), 植株甲基因型：(55,175)
        // 植株乙外框：(50,220)-(190,340), 植株乙图标：中心(120,280)
        // 植株乙标签：(55,210), 植株乙基因型：(55,355)
        // 处理①箭头：(190,100)-(320,100), 标签：(255,85)
        // 结果框 1：(330,75)-(470,115), 文本：(340,95)
        // 处理②箭头：(210,180)-(320,180), 标签：(255,165)
        // 结果框 2：(330,155)-(470,195), 文本：(340,175)
        // 处理③箭头：(190,280)-(320,280), 标签：(255,265)
        // 结果框 3：(330,255)-(470,295), 文本：(340,275)
        // 处理④箭头：(470,280)-(600,280), 标签：(535,265)
        // 结果框 4：(610,255)-(750,295), 文本：(620,275)
        // 概念框：(50,380)-(750,430), 文本：(60,405)
        // 按钮：上一步 (250,445)-(360,485), 下一步 (440,445)-(550,485)

        const steps = [
            {
                conceptText: "",
                highlightArrows: [],
                highlightFrames: [],
                highlightResults: []
            },
            {
                conceptText: "处理①：去雄后不授粉（或涂抹生长素）→ 促进子房发育为无种子豆荚（单性结实）",
                highlightArrows: ["scene1-arrow-1", "scene1-label-1"],
                highlightFrames: ["scene1-plant-a-frame"],
                highlightResults: ["scene1-result-1"]
            },
            {
                conceptText: "处理②：人工授粉杂交（植株甲♀×植株乙♂）→ 形成有种子豆荚，子代基因型为 AaBb",
                highlightArrows: ["scene1-arrow-2", "scene1-arrow-2a", "scene1-arrow-2b", "scene1-label-2"],
                highlightFrames: ["scene1-plant-a-frame", "scene1-plant-b-frame"],
                highlightResults: ["scene1-result-2"]
            },
            {
                conceptText: "处理③：花药离体培养 → 花粉（配子）经组织培养发育为单倍体植株（基因型 AB/Ab/aB/ab）",
                highlightArrows: ["scene1-arrow-3", "scene1-label-3"],
                highlightFrames: ["scene1-plant-b-frame"],
                highlightResults: ["scene1-result-3"]
            },
            {
                conceptText: "处理④：秋水仙素处理 → 抑制纺锤体形成，染色体数目加倍，单倍体→二倍体纯合子",
                highlightArrows: ["scene1-arrow-4", "scene1-label-4"],
                highlightFrames: [],
                highlightResults: ["scene1-result-3", "scene1-result-4"]
            }
        ];

        let currentStep = 0;

        const conceptText = document.getElementById("scene1-concept-text");
        const conceptBox = document.getElementById("scene1-concept-box");
        const btnPrev = document.getElementById("scene1-btn-prev");
        const btnNext = document.getElementById("scene1-btn-next");

        function resetHighlights() {
            const arrows = ["scene1-arrow-1", "scene1-arrow-2", "scene1-arrow-2a", "scene1-arrow-2b", 
                           "scene1-arrow-3", "scene1-arrow-4"];
            arrows.forEach(id => {
                const el = document.getElementById(id);
                if (el) {
                    el.classList.remove("arrow-line-highlight");
                    el.classList.add("arrow-line");
                    el.setAttribute("marker-end", "url(#scene1-arrowhead)");
                }
            });

            const labels = ["scene1-label-1", "scene1-label-2", "scene1-label-3", "scene1-label-4"];
            labels.forEach(id => {
                const el = document.getElementById(id);
                if (el) {
                    el.classList.remove("label-text-highlight");
                    el.classList.add("label-text");
                }
            });

            const frames = ["scene1-plant-a-frame", "scene1-plant-b-frame"];
            frames.forEach(id => {
                const el = document.getElementById(id);
                if (el) {
                    el.classList.remove("plant-frame-highlight");
                    el.classList.add("plant-frame");
                }
            });

            const results = ["scene1-result-1", "scene1-result-2", "scene1-result-3", "scene1-result-4"];
            results.forEach(id => {
                const el = document.getElementById(id);
                if (el) {
                    el.classList.remove("result-box-highlight");
                    el.classList.add("result-box");
                }
            });
        }

        function applyHighlights(stepIndex) {
            const step = steps[stepIndex];

            step.highlightArrows.forEach(id => {
                const el = document.getElementById(id);
                if (el) {
                    if (id.includes("arrow")) {
                        el.classList.remove("arrow-line");
                        el.classList.add("arrow-line-highlight");
                        el.setAttribute("marker-end", "url(#scene1-arrowhead-blue)");
                    } else if (id.includes("label")) {
                        el.classList.remove("label-text");
                        el.classList.add("label-text-highlight");
                    }
                }
            });

            step.highlightFrames.forEach(id => {
                const el = document.getElementById(id);
                if (el) {
                    el.classList.remove("plant-frame");
                    el.classList.add("plant-frame-highlight");
                }
            });

            step.highlightResults.forEach(id => {
                const el = document.getElementById(id);
                if (el) {
                    el.classList.remove("result-box");
                    el.classList.add("result-box-highlight");
                }
            });
        }

        function updateState() {
            resetHighlights();
            
            const step = steps[currentStep];
            conceptText.textContent = step.conceptText;
            
            if (currentStep === 0) {
                conceptBox.setAttribute("opacity", "0");
                conceptText.setAttribute("opacity", "0");
            } else {
                conceptBox.setAttribute("opacity", "1");
                conceptText.setAttribute("opacity", "1");
            }

            if (currentStep > 0) {
                applyHighlights(currentStep);
            }

            if (currentStep === 0) {
                btnPrev.classList.add("btn-disabled");
            } else {
                btnPrev.classList.remove("btn-disabled");
            }

            if (currentStep === 4) {
                btnNext.classList.add("btn-disabled");
            } else {
                btnNext.classList.remove("btn-disabled");
            }
        }

        btnPrev.addEventListener("click", function() {
            if (currentStep > 0) {
                currentStep--;
                updateState();
            }
        });

        btnNext.addEventListener("click", function() {
            if (currentStep < 4) {
                currentStep++;
                updateState();
            }
        });

        updateState();
    })();
    </script>
