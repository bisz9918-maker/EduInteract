    <script>
    (function() {
        // Scene 3 实现：单倍体育种流程与基因工程工具

        // 当前状态（1=单倍体育种，2=染色体加倍，3=基因工程）
        let currentState = 1;

        // 获取所有按钮和状态组
        const btn1 = document.getElementById('scene3-btn1-group');
        const btn2 = document.getElementById('scene3-btn2-group');
        const btn3 = document.getElementById('scene3-btn3-group');

        const state1Group = document.getElementById('scene3-state1');
        const state2Group = document.getElementById('scene3-state2');
        const state3Group = document.getElementById('scene3-state3');

        // 更新按钮高亮状态
        function updateButtonHighlight(activeState) {
            btn1.classList.remove('active');
            btn2.classList.remove('active');
            btn3.classList.remove('active');

            if (activeState === 1) btn1.classList.add('active');
            else if (activeState === 2) btn2.classList.add('active');
            else if (activeState === 3) btn3.classList.add('active');
        }

        // 切换状态
        function switchState(newState) {
            if (newState === currentState) return;

            // 隐藏所有状态组
            state1Group.classList.remove('scene3-visible');
            state1Group.classList.add('scene3-hidden');
            state2Group.classList.remove('scene3-visible');
            state2Group.classList.add('scene3-hidden');
            state3Group.classList.remove('scene3-visible');
            state3Group.classList.add('scene3-hidden');

            // 显示新状态组
            const newStateGroup = document.getElementById('scene3-state' + newState);
            newStateGroup.classList.remove('scene3-hidden');
            newStateGroup.classList.add('scene3-visible');

            // 更新按钮高亮
            updateButtonHighlight(newState);

            // 更新当前状态
            currentState = newState;
        }

        // 绑定按钮点击事件
        btn1.addEventListener('click', function() {
            switchState(1);
        });

        btn2.addEventListener('click', function() {
            switchState(2);
        });

        btn3.addEventListener('click', function() {
            switchState(3);
        });

        // 初始化：激活状态1
        updateButtonHighlight(1);
    })();
    </script>
