    <script>
    (function() {
        // Scene 1 实现：整体 (A+B) 水平受力分析

        let currentStep = 0;
        const totalSteps = 3;

        // 获取 DOM 元素
        const stepIndicator = document.getElementById('scene1-step-indicator');
        const btnPrev = document.getElementById('scene1-btn-prev-bg');
        const btnPrevText = document.getElementById('scene1-btn-prev-text');
        const btnNext = document.getElementById('scene1-btn-next-bg');
        const btnNextText = document.getElementById('scene1-btn-next-text');
        const hookLeft2 = document.getElementById('scene1-hook-left-2');

        // 步骤元素组
        const step1Elements = document.getElementById('scene1-step1-forces');
        const step2Elements = document.getElementById('scene1-step2-forces');
        const step3Elements = document.getElementById('scene1-step3-compare');
        const labelInitial = document.getElementById('scene1-label-initial');
        const labelChanged = document.getElementById('scene1-label-changed');

        // 显示/隐藏函数
        function showElement(el) {
            if (el) {
                el.style.display = 'block';
                el.style.opacity = '1';
                el.classList.add('scene1-visible');
                el.classList.remove('scene1-hidden');
            }
        }

        function hideElement(el) {
            if (el) {
                el.style.display = 'none';
                el.style.opacity = '0';
                el.classList.remove('scene1-visible');
                el.classList.add('scene1-hidden');
            }
        }

        // 更新步骤显示
        function updateStep() {
            // 隐藏所有步骤元素
            hideElement(step1Elements);
            hideElement(step2Elements);
            hideElement(step3Elements);
            hideElement(labelInitial);
            hideElement(labelChanged);

            // 根据当前步骤显示对应元素
            if (currentStep === 0) {
                // 步骤 0：只显示装置图，左侧 2 个钩码
                hookLeft2.style.display = 'block';
                stepIndicator.textContent = '步骤 0/3';
            } else if (currentStep === 1) {
                // 步骤 1：初始状态受力分析
                hookLeft2.style.display = 'block';
                showElement(step1Elements);
                showElement(labelInitial);
                stepIndicator.textContent = '步骤 1/3';
            } else if (currentStep === 2) {
                // 步骤 2：变化后受力分析，左侧 1 个钩码
                hookLeft2.style.display = 'none';
                showElement(step2Elements);
                showElement(labelChanged);
                stepIndicator.textContent = '步骤 2/3';
            } else if (currentStep === 3) {
                // 步骤 3：对比结论
                hookLeft2.style.display = 'none';
                showElement(step3Elements);
                stepIndicator.textContent = '步骤 3/3';
            }

            // 更新按钮状态
            if (currentStep === 0) {
                btnPrev.style.fill = '#cccccc';
                btnPrevText.style.fill = '#666666';
            } else {
                btnPrev.style.fill = '#1976d2';
                btnPrevText.style.fill = '#ffffff';
            }

            if (currentStep === totalSteps) {
                btnNext.style.fill = '#cccccc';
                btnNextText.style.fill = '#666666';
            } else {
                btnNext.style.fill = '#1976d2';
                btnNextText.style.fill = '#ffffff';
            }
        }

        // 下一步
        function stepForward() {
            if (currentStep < totalSteps) {
                currentStep++;
                updateStep();
            }
        }

        // 上一步
        function stepBackward() {
            if (currentStep > 0) {
                currentStep--;
                updateStep();
            }
        }

        // 绑定按钮事件
        btnNext.addEventListener('click', stepForward);
        btnPrev.addEventListener('click', stepBackward);

        // 初始化
        updateStep();
    })();
    </script>
