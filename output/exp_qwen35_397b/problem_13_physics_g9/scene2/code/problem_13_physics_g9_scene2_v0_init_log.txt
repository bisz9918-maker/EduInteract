    <script>
    (function() {
        // Scene 2: 木块 A 水平受力分析
        
        // 状态机
        let currentStep = 0;
        const totalSteps = 3;
        
        // 状态文本
        const statusTexts = [
            "木块 A 隔离体",
            "初始：左侧 2 个钩码，拉力=2G",
            "变化后：左侧 1 个钩码，拉力=G",
            "结论：f_AB' < f_AB，A、B 间静摩擦力减小"
        ];
        
        // 获取 DOM 元素
        const titleInitial = document.getElementById("scene2-title-initial");
        const titleChanged = document.getElementById("scene2-title-changed");
        const step1Group = document.getElementById("scene2-step1-group");
        const step2Group = document.getElementById("scene2-step2-group");
        const step3Group = document.getElementById("scene2-step3-group");
        const btnPrev = document.getElementById("scene2-btn-prev");
        const btnNext = document.getElementById("scene2-btn-next");
        const statusText = document.getElementById("scene2-status-text");
        
        // 显示/隐藏元素
        function showElement(el) {
            if (el) {
                el.style.display = "block";
                el.style.opacity = "1";
            }
        }
        
        function hideElement(el) {
            if (el) {
                el.style.display = "none";
                el.style.opacity = "0";
            }
        }
        
        // 渲染当前步骤
        function renderStep() {
            // 隐藏所有步骤元素
            hideElement(titleInitial);
            hideElement(titleChanged);
            hideElement(step1Group);
            hideElement(step2Group);
            hideElement(step3Group);
            
            // 根据当前步骤显示相应元素
            switch (currentStep) {
                case 0:
                    // 仅显示木块 A
                    statusText.textContent = statusTexts[0];
                    break;
                case 1:
                    // 初始状态受力分析
                    showElement(titleInitial);
                    showElement(step1Group);
                    statusText.textContent = statusTexts[1];
                    break;
                case 2:
                    // 变化后受力分析
                    showElement(titleChanged);
                    showElement(step2Group);
                    statusText.textContent = statusTexts[2];
                    break;
                case 3:
                    // 条形图对比
                    showElement(titleChanged);
                    showElement(step2Group);
                    showElement(step3Group);
                    statusText.textContent = statusTexts[3];
                    break;
            }
            
            // 更新按钮状态
            btnPrev.disabled = (currentStep === 0);
            btnNext.disabled = (currentStep === totalSteps);
        }
        
        // 按钮事件
        btnPrev.addEventListener("click", function() {
            if (currentStep > 0) {
                currentStep--;
                renderStep();
            }
        });
        
        btnNext.addEventListener("click", function() {
            if (currentStep < totalSteps) {
                currentStep++;
                renderStep();
            }
        });
        
        // 初始化
        renderStep();
    })();
    </script>
