    <script>
    (function() {
        // Scene 3 实现：选项判断多状态切换

        // 状态变量：0=初始，1=选项 A/B，2=选项 C，3=选项 D
        let currentState = 0;

        // 工具函数：显示元素
        function show(el) {
            if (el) {
                el.style.opacity = '1';
                el.style.display = '';
                if (el.classList) el.classList.remove('scene3-hidden');
            }
        }

        // 工具函数：隐藏元素
        function hide(el) {
            if (el) {
                el.style.opacity = '0';
                el.style.display = 'none';
                if (el.classList) el.classList.add('scene3-hidden');
            }
        }

        // 获取所有元素
        const elements = {
            // 判断标记
            markA: document.getElementById('scene3-mark-a'),
            markB: document.getElementById('scene3-mark-b'),
            markC: document.getElementById('scene3-mark-c'),
            markD: document.getElementById('scene3-mark-d'),
            // 结论区标题
            conclusionTitle: document.getElementById('scene3-conclusion-title'),
            // 结论 1（实验台摩擦力）
            conclusion1Bg: document.getElementById('scene3-conclusion-1-bg'),
            conclusion1Text: document.getElementById('scene3-conclusion-1-text'),
            conclusion1Formula: document.getElementById('scene3-conclusion-1-formula'),
            conclusion1Text2: document.getElementById('scene3-conclusion-1-text2'),
            // 结论 2（A、B 间摩擦力）
            conclusion2Bg: document.getElementById('scene3-conclusion-2-bg'),
            conclusion2Text: document.getElementById('scene3-conclusion-2-text'),
            conclusion2Formula: document.getElementById('scene3-conclusion-2-formula'),
            conclusion2Text2: document.getElementById('scene3-conclusion-2-text2'),
            // 高亮边框
            highlightA: document.getElementById('scene3-highlight-a'),
            highlightB: document.getElementById('scene3-highlight-b'),
            highlightC: document.getElementById('scene3-highlight-c'),
            highlightD: document.getElementById('scene3-highlight-d'),
            // 按钮
            btn1: document.getElementById('scene3-btn1-bg'),
            btn2: document.getElementById('scene3-btn2-bg'),
            btn3: document.getElementById('scene3-btn3-bg')
        };

        // 状态切换函数
        function updateState(newState) {
            currentState = newState;

            // 重置所有元素
            hide(elements.markA);
            hide(elements.markB);
            hide(elements.markC);
            hide(elements.markD);
            hide(elements.conclusionTitle);
            hide(elements.conclusion1Bg);
            hide(elements.conclusion1Text);
            hide(elements.conclusion1Formula);
            hide(elements.conclusion1Text2);
            hide(elements.conclusion2Bg);
            hide(elements.conclusion2Text);
            hide(elements.conclusion2Formula);
            hide(elements.conclusion2Text2);
            hide(elements.highlightA);
            hide(elements.highlightB);
            hide(elements.highlightC);
            hide(elements.highlightD);

            if (newState === 1) {
                // 状态 1：选项 A/B 分析
                show(elements.conclusionTitle);
                show(elements.conclusion1Bg);
                show(elements.conclusion1Text);
                show(elements.conclusion1Formula);
                show(elements.conclusion1Text2);
                // 显示 A 错误、B 正确标记
                show(elements.markA);
                show(elements.markB);
                hide(elements.markC);
                hide(elements.markD);
            } else if (newState === 2) {
                // 状态 2：选项 C 分析
                show(elements.conclusionTitle);
                show(elements.conclusion1Bg);
                show(elements.conclusion1Text);
                show(elements.conclusion1Formula);
                show(elements.conclusion1Text2);
                // 显示 A、C 错误，B 正确
                show(elements.markA);
                show(elements.markB);
                show(elements.markC);
                hide(elements.markD);
            } else if (newState === 3) {
                // 状态 3：选项 D 分析（完整结论）
                show(elements.conclusionTitle);
                show(elements.conclusion1Bg);
                show(elements.conclusion1Text);
                show(elements.conclusion1Formula);
                show(elements.conclusion1Text2);
                show(elements.conclusion2Bg);
                show(elements.conclusion2Text);
                show(elements.conclusion2Formula);
                show(elements.conclusion2Text2);
                // 显示所有判断标记
                show(elements.markA);
                show(elements.markB);
                show(elements.markC);
                show(elements.markD);
                // 显示高亮边框
                show(elements.highlightA);
                show(elements.highlightB);
                show(elements.highlightC);
                show(elements.highlightD);
            }
        }

        // 绑定按钮事件
        elements.btn1.addEventListener('click', function() {
            updateState(1);
        });

        elements.btn2.addEventListener('click', function() {
            updateState(2);
        });

        elements.btn3.addEventListener('click', function() {
            updateState(3);
        });

        // 初始状态：什么都不显示（只有选项框和文字）
        updateState(0);
    })();
    </script>
