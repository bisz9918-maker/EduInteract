    <script>
    (function() {
        // Scene 1 实现：基因对毛发颜色代谢途径的调控

        // 状态变量
        let currentStep = 0;
        const totalSteps = 5;

        // DOM 元素引用
        const btnPrev = document.getElementById('scene1-btn-prev');
        const btnNext = document.getElementById('scene1-btn-next');
        const stepIndicator = document.getElementById('scene1-step-indicator');

        // 元素显隐控制工具函数
        function showElement(el) {
            if (el) {
                el.classList.remove('hidden');
                el.classList.add('visible');
                el.style.display = 'block';
                el.style.opacity = '1';
            }
        }

        function hideElement(el) {
            if (el) {
                el.classList.remove('visible');
                el.classList.add('hidden');
                el.style.display = 'none';
                el.style.opacity = '0';
            }
        }

        // 更新显示状态
        function updateDisplay() {
            // 更新步骤指示器
            stepIndicator.textContent = '步骤 ' + (currentStep + 1) + '/' + totalSteps;

            // 更新按钮状态
            btnPrev.disabled = (currentStep === 0);
            btnNext.disabled = (currentStep === totalSteps - 1);

            // 根据当前步骤显示/隐藏元素组
            const groups = [
                document.getElementById('scene1-group-step0'),
                document.getElementById('scene1-group-step1'),
                document.getElementById('scene1-group-step2'),
                document.getElementById('scene1-group-step3'),
                document.getElementById('scene1-group-step4')
            ];

            // 步骤 0 显示 group-step0
            // 步骤 1 显示 group-step0 + group-step1
            // 步骤 2 显示 group-step0 + group-step1 + group-step2
            // 步骤 3 显示 group-step0 + group-step1 + group-step2 + group-step3
            // 步骤 4 显示所有组

            for (let i = 0; i < groups.length; i++) {
                if (i <= currentStep) {
                    showElement(groups[i]);
                } else {
                    hideElement(groups[i]);
                }
            }

            // 为箭头添加动画效果（步骤 2 和步骤 3）
            if (currentStep >= 2) {
                const arrow1 = document.getElementById('scene1-arrow-1');
                if (arrow1) {
                    arrow1.classList.remove('animate-arrow');
                    void arrow1.offsetWidth; // 触发重绘
                    arrow1.classList.add('animate-arrow');
                }
            }
            if (currentStep >= 3) {
                const arrow2 = document.getElementById('scene1-arrow-2');
                if (arrow2) {
                    arrow2.classList.remove('animate-arrow');
                    void arrow2.offsetWidth; // 触发重绘
                    arrow2.classList.add('animate-arrow');
                }
            }
        }

        // 按钮事件绑定
        btnPrev.addEventListener('click', function() {
            if (currentStep > 0) {
                currentStep--;
                updateDisplay();
            }
        });

        btnNext.addEventListener('click', function() {
            if (currentStep < totalSteps - 1) {
                currentStep++;
                updateDisplay();
            }
        });

        // 初始化显示
        updateDisplay();

    })();
    </script>
