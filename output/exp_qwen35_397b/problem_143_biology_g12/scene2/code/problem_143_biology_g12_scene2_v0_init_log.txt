    <script>
    (function() {
        // Scene 2: 干预前能量流动计算
        let currentState = 0;
        const totalSteps = 5;

        const btnPrev = document.getElementById('scene2-btn-prev');
        const btnNext = document.getElementById('scene2-btn-next');
        const stepIndicator = document.getElementById('scene2-step-indicator');

        // 元素分组
        const step1Elements = [
            'antelope-box', 'antelope-label', 'antelope-energy',
            'zebra-box', 'zebra-label', 'zebra-energy',
            'mouse-box', 'mouse-label', 'mouse-energy',
            'arrow-grass-antelope', 'arrow-grass-zebra', 'arrow-grass-mouse',
            'efficiency-1', 'efficiency-2', 'efficiency-3',
            'calc-1', 'calc-2', 'calc-3'
        ];

        const step2Elements = [
            'wilddog-box', 'wilddog-label', 'wilddog-energy',
            'snake-box', 'snake-label', 'snake-energy',
            'arrow-antelope-wilddog', 'arrow-zebra-wilddog', 'arrow-mouse-snake'
        ];

        const step3Elements = [
            'snakeater-box', 'snakeater-label', 'snakeater-energy',
            'arrow-snake-snakeater'
        ];

        const step4Elements = [
            'arrow-antelope-lion', 'arrow-zebra-lion', 'arrow-wilddog-lion', 'arrow-snakeater-lion',
            'lion-path-1', 'lion-path-2', 'lion-path-3', 'lion-path-4'
        ];

        const step5Elements = [
            'lion-box', 'lion-label', 'lion-energy',
            'summary-box', 'summary-label'
        ];

        function showElement(id) {
            const el = document.getElementById('scene2-' + id);
            if (el) {
                el.classList.remove('scene2-hidden');
                el.style.opacity = '1';
                el.style.display = '';
            }
        }

        function hideElement(id) {
            const el = document.getElementById('scene2-' + id);
            if (el) {
                el.classList.add('scene2-hidden');
                el.style.opacity = '0';
            }
        }

        function updateState() {
            // 隐藏所有步骤元素
            step1Elements.forEach(id => hideElement(id));
            step2Elements.forEach(id => hideElement(id));
            step3Elements.forEach(id => hideElement(id));
            step4Elements.forEach(id => hideElement(id));
            step5Elements.forEach(id => hideElement(id));

            // 根据当前状态显示相应元素
            if (currentState >= 1) {
                step1Elements.forEach(id => showElement(id));
            }
            if (currentState >= 2) {
                step2Elements.forEach(id => showElement(id));
            }
            if (currentState >= 3) {
                step3Elements.forEach(id => showElement(id));
            }
            if (currentState >= 4) {
                step4Elements.forEach(id => showElement(id));
            }
            if (currentState >= 5) {
                step5Elements.forEach(id => showElement(id));
                // 高亮狮子框
                document.getElementById('scene2-lion-box').classList.add('scene2-highlight');
            } else {
                document.getElementById('scene2-lion-box').classList.remove('scene2-highlight');
            }

            // 更新按钮状态
            btnPrev.disabled = currentState === 0;
            btnNext.disabled = currentState === totalSteps;
            stepIndicator.textContent = '步骤 ' + currentState + '/' + totalSteps;
        }

        btnPrev.addEventListener('click', function() {
            if (currentState > 0) {
                currentState--;
                updateState();
            }
        });

        btnNext.addEventListener('click', function() {
            if (currentState < totalSteps) {
                currentState++;
                updateState();
            }
        });

        // 初始化
        updateState();
    })();
    </script>
