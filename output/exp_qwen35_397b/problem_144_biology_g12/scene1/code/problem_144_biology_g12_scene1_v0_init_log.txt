    <script>
    (function() {
        // Scene 1: 果蝇眼型遗传杂交实验
        
        // 状态管理
        let currentStep = 0;
        const totalSteps = 5;
        
        // 获取 DOM 元素
        const btnPrev = document.getElementById('scene1-btn-prev');
        const btnNext = document.getElementById('scene1-btn-next');
        const stepIndicator = document.getElementById('scene1-step-indicator');
        
        // 元素分组 - 按步骤显示
        const stepElements = {
            0: {
                show: ['scene1-p-label', 'scene1-p-female-symbol', 'scene1-p-female-icon', 'scene1-p-cross', 'scene1-p-male-symbol', 'scene1-p-male-icon'],
                hide: ['scene1-arrow-p-f1', 'scene1-f1-label', 'scene1-f1-group', 'scene1-arrow-f1-f2', 'scene1-f1-mating-text', 'scene1-f2-label', 'scene1-f2-round-female', 'scene1-f2-round-female-label', 'scene1-f2-round-female-ratio', 'scene1-f2-round-female-genotype', 'scene1-ratio-sep1', 'scene1-f2-bar-female', 'scene1-f2-bar-female-label', 'scene1-f2-bar-female-ratio', 'scene1-f2-bar-female-genotype', 'scene1-ratio-sep2', 'scene1-f2-round-male', 'scene1-f2-round-male-label', 'scene1-f2-round-male-ratio', 'scene1-f2-round-male-genotype', 'scene1-ratio-sep3', 'scene1-f2-bar-male', 'scene1-f2-bar-male-label', 'scene1-f2-bar-male-ratio', 'scene1-f2-bar-male-genotype']
            },
            1: {
                show: ['scene1-arrow-p-f1', 'scene1-f1-label', 'scene1-f1-group'],
                hide: ['scene1-arrow-f1-f2', 'scene1-f1-mating-text', 'scene1-f2-label', 'scene1-f2-round-female', 'scene1-f2-round-female-label', 'scene1-f2-round-female-ratio', 'scene1-f2-round-female-genotype', 'scene1-ratio-sep1', 'scene1-f2-bar-female', 'scene1-f2-bar-female-label', 'scene1-f2-bar-female-ratio', 'scene1-f2-bar-female-genotype', 'scene1-ratio-sep2', 'scene1-f2-round-male', 'scene1-f2-round-male-label', 'scene1-f2-round-male-ratio', 'scene1-f2-round-male-genotype', 'scene1-ratio-sep3', 'scene1-f2-bar-male', 'scene1-f2-bar-male-label', 'scene1-f2-bar-male-ratio', 'scene1-f2-bar-male-genotype']
            },
            2: {
                show: ['scene1-arrow-f1-f2', 'scene1-f1-mating-text', 'scene1-f2-label', 'scene1-f2-round-female', 'scene1-f2-round-female-label', 'scene1-f2-bar-female', 'scene1-f2-bar-female-label', 'scene1-f2-round-male', 'scene1-f2-round-male-label', 'scene1-f2-bar-male', 'scene1-f2-bar-male-label'],
                hide: ['scene1-f2-round-female-ratio', 'scene1-f2-round-female-genotype', 'scene1-ratio-sep1', 'scene1-f2-bar-female-ratio', 'scene1-f2-bar-female-genotype', 'scene1-ratio-sep2', 'scene1-f2-round-male-ratio', 'scene1-f2-round-male-genotype', 'scene1-ratio-sep3', 'scene1-f2-bar-male-ratio', 'scene1-f2-bar-male-genotype']
            },
            3: {
                show: ['scene1-f2-round-female-ratio', 'scene1-ratio-sep1', 'scene1-f2-bar-female-ratio', 'scene1-ratio-sep2', 'scene1-f2-round-male-ratio', 'scene1-ratio-sep3', 'scene1-f2-bar-male-ratio'],
                hide: ['scene1-f2-round-female-genotype', 'scene1-f2-bar-female-genotype', 'scene1-f2-round-male-genotype', 'scene1-f2-bar-male-genotype']
            },
            4: {
                show: ['scene1-f2-round-female-genotype', 'scene1-f2-bar-female-genotype', 'scene1-f2-round-male-genotype', 'scene1-f2-bar-male-genotype'],
                hide: []
            }
        };
        
        // 工具函数：显示元素
        function showElement(id) {
            const el = document.getElementById(id);
            if (el) {
                el.classList.remove('scene1-hidden');
                el.classList.add('scene1-visible');
                if (el.tagName === 'text' || el.tagName === 'g') {
                    el.style.opacity = '1';
                }
            }
        }
        
        // 工具函数：隐藏元素
        function hideElement(id) {
            const el = document.getElementById(id);
            if (el) {
                el.classList.remove('scene1-visible');
                el.classList.add('scene1-hidden');
                if (el.tagName === 'text' || el.tagName === 'g') {
                    el.style.opacity = '0';
                }
            }
        }
        
        // 更新步骤显示
        function updateStep(step) {
            // 填充 F1 空白框 (步骤 4)
            const f1Text = document.getElementById('scene1-f1-text');
            if (step >= 4 && f1Text) {
                f1Text.textContent = '棒眼';
            } else if (f1Text) {
                f1Text.textContent = '';
            }
            
            // 根据当前步骤显示/隐藏元素
            for (let s = 0; s < totalSteps; s++) {
                const elements = stepElements[s];
                if (s <= step) {
                    // 显示此步骤及之前步骤应该显示的元素
                    elements.show.forEach(id => showElement(id));
                } else {
                    // 隐藏后续步骤的元素
                    elements.show.forEach(id => hideElement(id));
                }
            }
            
            // 更新步骤指示器
            stepIndicator.textContent = `步骤 ${step + 1}/${totalSteps}`;
            
            // 更新按钮状态
            updateButtons();
        }
        
        // 更新按钮状态
        function updateButtons() {
            const prevRect = btnPrev.querySelector('rect');
            const prevText = btnPrev.querySelector('text');
            const nextRect = btnNext.querySelector('rect');
            const nextText = btnNext.querySelector('text');
            
            if (currentStep === 0) {
                prevRect.setAttribute('fill', '#bdbdbd');
                prevText.setAttribute('fill', '#757575');
                btnPrev.classList.add('scene1-btn-disabled');
            } else {
                prevRect.setAttribute('fill', '#1976d2');
                prevText.setAttribute('fill', '#ffffff');
                btnPrev.classList.remove('scene1-btn-disabled');
            }
            
            if (currentStep === totalSteps - 1) {
                nextRect.setAttribute('fill', '#bdbdbd');
                nextText.setAttribute('fill', '#757575');
                btnNext.classList.add('scene1-btn-disabled');
            } else {
                nextRect.setAttribute('fill', '#1976d2');
                nextText.setAttribute('fill', '#ffffff');
                btnNext.classList.remove('scene1-btn-disabled');
            }
        }
        
        // 上一步按钮点击
        btnPrev.addEventListener('click', function() {
            if (currentStep > 0) {
                currentStep--;
                updateStep(currentStep);
            }
        });
        
        // 下一步按钮点击
        btnNext.addEventListener('click', function() {
            if (currentStep < totalSteps - 1) {
                currentStep++;
                updateStep(currentStep);
            }
        });
        
        // 初始化
        updateStep(currentStep);
        
    })();
    </script>
