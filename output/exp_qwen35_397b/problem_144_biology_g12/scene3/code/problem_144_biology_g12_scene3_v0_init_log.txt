    <script>
    (function() {
        // Scene 3 实现 - 验证实验设计

        // 布局坐标表（实现对照依据）：
        // | 元素 | 类型 | 位置/范围 | 大小 | 备注 |
        // |------|------|-----------|------|------|
        // | 标题区 | 文本 | (400,35) | - | 居中 |
        // | 实验一区域 | 容器 | (40,70)-(380,380) | 340×310 | 蓝色边框 |
        // | 实验二区域 | 容器 | (420,70)-(760,380) | 340×310 | 绿色边框 |
        // | 实验一标题 | text | (210,95) | - | 蓝色 |
        // | 实验二标题 | text | (590,95) | - | 绿色 |
        // | 实验一亲本♀ | 图标 | (120,160) | - | 棒眼雌蝇 |
        // | 实验一亲本♂ | 图标 | (300,160) | - | 圆眼雄蝇 |
        // | 实验二亲本♀ | 图标 | (500,160) | - | 圆眼雌蝇 |
        // | 实验二亲本♂ | 图标 | (680,160) | - | 棒眼雄蝇 |
        // | 实验一配子 | text | (120/300,240) | - | 灰色 |
        // | 实验二配子 | text | (500/680,240) | - | 灰色 |
        // | 实验一后代框 | 矩形 | (80,270)-(340,360) | 260×90 | - |
        // | 实验二后代框 | 矩形 | (460,270)-(720,360) | 260×90 | - |
        // | 总结区 | 文本框 | (40,400)-(760,440) | 720×40 | - |
        // | 控件栏 | 按钮组 | (250,455)-(550,490) | - | - |

        let currentStep = 0;
        const totalSteps = 5;

        // 工具函数：显示元素
        function show(el) {
            if (el) {
                el.classList.remove('hidden');
                el.style.display = '';
                el.style.opacity = '1';
            }
        }

        // 工具函数：隐藏元素
        function hide(el) {
            if (el) {
                el.classList.add('hidden');
                el.style.display = 'none';
                el.style.opacity = '0';
            }
        }

        // 获取所有需要控制的元素
        const elements = {
            exp1Area: document.getElementById('scene3-exp1-area'),
            exp1Title: document.getElementById('scene3-exp1-title'),
            exp1ComboLabel: document.getElementById('scene3-exp1-combo-label'),
            exp1Female: document.getElementById('scene3-exp1-female-group'),
            exp1Male: document.getElementById('scene3-exp1-male-group'),
            exp1Cross: document.getElementById('scene3-exp1-cross'),
            exp1Gametes: document.getElementById('scene3-exp1-gametes'),
            exp1OffspringBox: document.getElementById('scene3-exp1-offspring-box'),
            exp1Phenotype: document.getElementById('scene3-exp1-phenotype'),
            exp1Ratio: document.getElementById('scene3-exp1-ratio'),
            exp2Area: document.getElementById('scene3-exp2-area'),
            exp2Title: document.getElementById('scene3-exp2-title'),
            exp2ComboLabel: document.getElementById('scene3-exp2-combo-label'),
            exp2Female: document.getElementById('scene3-exp2-female-group'),
            exp2Male: document.getElementById('scene3-exp2-male-group'),
            exp2Cross: document.getElementById('scene3-exp2-cross'),
            exp2Gametes: document.getElementById('scene3-exp2-gametes'),
            exp2OffspringBox: document.getElementById('scene3-exp2-offspring-box'),
            exp2Phenotype: document.getElementById('scene3-exp2-phenotype'),
            exp2Ratio: document.getElementById('scene3-exp2-ratio'),
            summaryBox: document.getElementById('scene3-summary-box'),
            summaryText: document.getElementById('scene3-summary-text'),
            prevBtn: document.getElementById('scene3-prev-btn'),
            nextBtn: document.getElementById('scene3-next-btn'),
            stepIndicator: document.getElementById('scene3-step-indicator')
        };

        // 更新步骤显示
        function updateStep() {
            elements.stepIndicator.textContent = `步骤 ${currentStep}/${totalSteps}`;
            elements.prevBtn.disabled = currentStep === 0;
            elements.nextBtn.disabled = currentStep === totalSteps;

            // 根据当前步骤显示/隐藏元素
            switch(currentStep) {
                case 0:
                    // 初始状态：只显示标题和框架
                    hideAllExpElements();
                    elements.exp1Area.setAttribute('stroke-opacity', '0.5');
                    elements.exp1Area.setAttribute('fill-opacity', '0.1');
                    elements.exp2Area.setAttribute('stroke-opacity', '0.5');
                    elements.exp2Area.setAttribute('fill-opacity', '0.1');
                    elements.summaryBox.setAttribute('opacity', '0.5');
                    break;

                case 1:
                    // 实验一亲本
                    elements.exp1Area.setAttribute('stroke-opacity', '1');
                    elements.exp1Area.setAttribute('fill-opacity', '0.1');
                    show(elements.exp1Title);
                    show(elements.exp1ComboLabel);
                    show(elements.exp1Female);
                    show(elements.exp1Male);
                    show(elements.exp1Cross);
                    hide(elements.exp1Gametes);
                    hide(elements.exp1OffspringBox);
                    hide(elements.exp1Phenotype);
                    hide(elements.exp1Ratio);
                    break;

                case 2:
                    // 实验一完整
                    show(elements.exp1Gametes);
                    show(elements.exp1OffspringBox);
                    show(elements.exp1Phenotype);
                    show(elements.exp1Ratio);
                    break;

                case 3:
                    // 实验二亲本，实验一半透明
                    elements.exp1Area.setAttribute('stroke-opacity', '0.5');
                    elements.exp1Area.setAttribute('fill-opacity', '0.1');
                    elements.exp2Area.setAttribute('stroke-opacity', '1');
                    elements.exp2Area.setAttribute('fill-opacity', '0.1');
                    show(elements.exp2Title);
                    show(elements.exp2ComboLabel);
                    show(elements.exp2Female);
                    show(elements.exp2Male);
                    show(elements.exp2Cross);
                    hide(elements.exp2Gametes);
                    hide(elements.exp2OffspringBox);
                    hide(elements.exp2Phenotype);
                    hide(elements.exp2Ratio);
                    hide(elements.summaryText);
                    elements.summaryBox.setAttribute('opacity', '0.5');
                    break;

                case 4:
                    // 实验二完整 + 总结
                    show(elements.exp2Gametes);
                    show(elements.exp2OffspringBox);
                    show(elements.exp2Phenotype);
                    show(elements.exp2Ratio);
                    elements.summaryBox.setAttribute('opacity', '1');
                    show(elements.summaryText);
                    break;

                case 5:
                    // 最终状态：两个实验都高亮
                    elements.exp1Area.setAttribute('stroke-opacity', '1');
                    elements.exp2Area.setAttribute('stroke-opacity', '1');
                    break;
            }
        }

        // 隐藏所有实验元素
        function hideAllExpElements() {
            hide(elements.exp1Title);
            hide(elements.exp1ComboLabel);
            hide(elements.exp1Female);
            hide(elements.exp1Male);
            hide(elements.exp1Cross);
            hide(elements.exp1Gametes);
            hide(elements.exp1OffspringBox);
            hide(elements.exp1Phenotype);
            hide(elements.exp1Ratio);
            hide(elements.exp2Title);
            hide(elements.exp2ComboLabel);
            hide(elements.exp2Female);
            hide(elements.exp2Male);
            hide(elements.exp2Cross);
            hide(elements.exp2Gametes);
            hide(elements.exp2OffspringBox);
            hide(elements.exp2Phenotype);
            hide(elements.exp2Ratio);
            hide(elements.summaryText);
        }

        // 按钮事件
        elements.prevBtn.addEventListener('click', function() {
            if (currentStep > 0) {
                currentStep--;
                updateStep();
            }
        });

        elements.nextBtn.addEventListener('click', function() {
            if (currentStep < totalSteps) {
                currentStep++;
                updateStep();
            }
        });

        // 初始化
        updateStep();
    })();
    </script>
