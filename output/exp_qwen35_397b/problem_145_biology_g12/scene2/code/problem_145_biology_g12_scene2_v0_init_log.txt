    <script>
    (function() {
        // Scene 2: Ⅱ-5 性染色体异常成因分析
        // 布局坐标表对照：
        // 父方细胞(150,80) r=35 | 母方细胞(350,80) r=35
        // 减Ⅰ框(50,150)-(350,230) | 减Ⅱ框(400,150)-(750,230)
        // 正常配子 X(100,280) r=25 | Y(180,280) r=25
        // 异常配子 XY(350,280) r=28 | O(430,280) r=28
        // 卵细胞(250,360) r=25
        // 合子 1(350,400) r=30 | 合子 2(450,400) r=30 | 合子 3(550,400) r=30
        // 控件栏 y=450-490

        let currentStep = 0;
        const totalSteps = 5;

        // 每步可见元素 ID 列表
        const stepVisibility = {
            0: {
                show: ['scene2-father-group', 'scene2-mother-group'],
                hide: ['scene2-meiosis1-group', 'scene2-meiosis2-group', 
                       'scene2-normal-gametes-group', 'scene2-abnormal-gametes-group',
                       'scene2-egg-group', 'scene2-fertilization-arrows', 'scene2-zygotes-group']
            },
            1: {
                show: ['scene2-meiosis1-group', 'scene2-normal-gametes-group'],
                hide: ['scene2-meiosis2-group', 'scene2-abnormal-gametes-group',
                       'scene2-egg-group', 'scene2-fertilization-arrows', 'scene2-zygotes-group']
            },
            2: {
                show: ['scene2-meiosis2-group', 'scene2-abnormal-gametes-group'],
                hide: ['scene2-egg-group', 'scene2-fertilization-arrows', 'scene2-zygotes-group']
            },
            3: {
                show: ['scene2-egg-group', 'scene2-fertilization-arrows', 'scene2-zygote-1', 
                       'scene2-zygote-1-xb-red', 'scene2-zygote-1-label'],
                hide: ['scene2-zygote-2', 'scene2-zygote-2-x1', 'scene2-zygote-2-x2', 'scene2-zygote-2-y', 'scene2-zygote-2-label',
                       'scene2-zygote-3', 'scene2-zygote-3-x', 'scene2-zygote-3-y1', 'scene2-zygote-3-y2', 'scene2-zygote-3-label']
            },
            4: {
                show: ['scene2-zygote-2', 'scene2-zygote-2-x1', 'scene2-zygote-2-x2', 'scene2-zygote-2-y', 'scene2-zygote-2-label',
                       'scene2-zygote-3', 'scene2-zygote-3-x', 'scene2-zygote-3-y1', 'scene2-zygote-3-y2', 'scene2-zygote-3-label'],
                hide: []
            },
            5: {
                show: [],
                hide: [],
                highlight: ['scene2-zygote-1', 'scene2-zygote-2', 'scene2-zygote-3']
            }
        };

        // 工具函数：显示元素
        function showElement(id) {
            const el = document.getElementById(id);
            if (el) {
                el.classList.remove('scene2-hidden');
                el.classList.add('scene2-visible');
                if (el.tagName === 'circle' || el.tagName === 'rect' || el.tagName === 'line' || el.tagName === 'text') {
                    el.style.display = 'block';
                    el.style.opacity = '1';
                } else {
                    el.style.display = 'block';
                    el.style.opacity = '1';
                }
            }
        }

        // 工具函数：隐藏元素
        function hideElement(id) {
            const el = document.getElementById(id);
            if (el) {
                el.classList.remove('scene2-visible');
                el.classList.add('scene2-hidden');
                if (el.tagName === 'circle' || el.tagName === 'rect' || el.tagName === 'line' || el.tagName === 'text') {
                    el.style.display = 'none';
                    el.style.opacity = '0';
                } else {
                    el.style.display = 'none';
                    el.style.opacity = '0';
                }
            }
        }

        // 更新步骤显示
        function updateStep(step) {
            const config = stepVisibility[step];
            
            // 显示指定元素
            if (config.show) {
                config.show.forEach(id => showElement(id));
            }
            
            // 隐藏指定元素
            if (config.hide) {
                config.hide.forEach(id => hideElement(id));
            }

            // 处理高亮（步骤 5）
            const allZygotes = ['scene2-zygote-1', 'scene2-zygote-2', 'scene2-zygote-3'];
            allZygotes.forEach(id => {
                const el = document.getElementById(id);
                if (el) {
                    el.classList.remove('scene2-zygote-highlight');
                }
            });
            if (config.highlight) {
                config.highlight.forEach(id => {
                    const el = document.getElementById(id);
                    if (el) {
                        el.classList.add('scene2-zygote-highlight');
                    }
                });
            }

            // 更新步骤指示器
            document.getElementById('scene2-step-indicator').textContent = `步骤 ${step}/${totalSteps}`;

            // 更新按钮状态
            const prevBtn = document.getElementById('scene2-btn-prev');
            const nextBtn = document.getElementById('scene2-btn-next');
            prevBtn.disabled = (step === 0);
            nextBtn.disabled = (step === totalSteps);
            if (step === totalSteps) {
                nextBtn.textContent = '完成';
            } else {
                nextBtn.textContent = '下一步 ▶';
            }

            currentStep = step;
        }

        // 初始化：设置所有元素初始状态
        function init() {
            // 初始隐藏所有需要动态显示的元素
            const initiallyHidden = [
                'scene2-meiosis1-group', 'scene2-meiosis2-group',
                'scene2-normal-gametes-group', 'scene2-abnormal-gametes-group',
                'scene2-egg-group', 'scene2-fertilization-arrows', 'scene2-zygotes-group',
                'scene2-zygote-1', 'scene2-zygote-1-xb-red', 'scene2-zygote-1-label',
                'scene2-zygote-2', 'scene2-zygote-2-x1', 'scene2-zygote-2-x2', 'scene2-zygote-2-y', 'scene2-zygote-2-label',
                'scene2-zygote-3', 'scene2-zygote-3-x', 'scene2-zygote-3-y1', 'scene2-zygote-3-y2', 'scene2-zygote-3-label'
            ];
            initiallyHidden.forEach(id => hideElement(id));

            // 显示初始元素（父母细胞）
            showElement('scene2-father-group');
            showElement('scene2-mother-group');

            // 更新按钮状态
            document.getElementById('scene2-step-indicator').textContent = `步骤 0/${totalSteps}`;
            document.getElementById('scene2-btn-prev').disabled = true;
            document.getElementById('scene2-btn-next').disabled = false;
        }

        // 事件绑定
        document.getElementById('scene2-btn-prev').addEventListener('click', function() {
            if (currentStep > 0) {
                updateStep(currentStep - 1);
            }
        });

        document.getElementById('scene2-btn-next').addEventListener('click', function() {
            if (currentStep < totalSteps) {
                updateStep(currentStep + 1);
            }
        });

        // 页面加载完成后初始化
        window.addEventListener('DOMContentLoaded', init);
    })();
    </script>
