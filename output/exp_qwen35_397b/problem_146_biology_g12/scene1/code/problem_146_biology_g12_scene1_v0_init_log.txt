    <script>
    (function() {
        // Scene 1 实现：果蝇性染色体异常类型分步展示

        // 状态变量
        let currentStep = 0;
        const totalSteps = 5;

        // 每步需要显示的元素 ID 列表
        const stepElements = {
            0: ['scene1-cell-normal', 'scene1-autosome-1', 'scene1-chrom-x-normal', 'scene1-chrom-y-normal', 'scene1-label-normal', 'scene1-status-normal'],
            1: ['scene1-cell-xxy', 'scene1-autosome-2', 'scene1-chrom-x-xxy', 'scene1-chrom-y-xxy', 'scene1-label-xxy', 'scene1-status-xxy'],
            2: ['scene1-cell-xo', 'scene1-autosome-3', 'scene1-chrom-x-xo', 'scene1-label-xo', 'scene1-status-xo'],
            3: ['scene1-cell-xxx', 'scene1-autosome-4', 'scene1-chrom-x-xxx', 'scene1-label-xxx', 'scene1-status-xxx'],
            4: ['scene1-cell-oy', 'scene1-autosome-5', 'scene1-chrom-y-oy', 'scene1-label-oy', 'scene1-status-oy']
        };

        // 显示元素函数
        function showElement(id) {
            const el = document.getElementById(id);
            if (el) {
                el.classList.add('visible');
            }
        }

        // 隐藏元素函数
        function hideElement(id) {
            const el = document.getElementById(id);
            if (el) {
                el.classList.remove('visible');
            }
        }

        // 更新显示状态
        function updateDisplay() {
            // 隐藏所有元素
            for (let step = 0; step < totalSteps; step++) {
                stepElements[step].forEach(id => hideElement(id));
            }

            // 显示当前步骤及之前的所有元素
            for (let step = 0; step <= currentStep; step++) {
                stepElements[step].forEach(id => showElement(id));
            }

            // 更新步骤指示器
            document.getElementById('scene1-step-indicator').textContent = '步骤 ' + (currentStep + 1) + '/' + totalSteps;

            // 更新按钮状态
            const prevBtn = document.getElementById('scene1-btn-prev-bg');
            const nextBtn = document.getElementById('scene1-btn-next-bg');
            const prevText = document.getElementById('scene1-btn-prev-text');
            const nextText = document.getElementById('scene1-btn-next-text');

            if (currentStep === 0) {
                prevBtn.setAttribute('fill', '#e0e0e0');
                prevText.setAttribute('fill', '#999999');
            } else {
                prevBtn.setAttribute('fill', '#f5f5f5');
                prevText.setAttribute('fill', '#000000');
            }

            if (currentStep === totalSteps - 1) {
                nextBtn.setAttribute('fill', '#e0e0e0');
                nextText.setAttribute('fill', '#999999');
            } else {
                nextBtn.setAttribute('fill', '#f5f5f5');
                nextText.setAttribute('fill', '#000000');
            }
        }

        // 上一步
        function stepBack() {
            if (currentStep > 0) {
                currentStep--;
                updateDisplay();
            }
        }

        // 下一步
        function stepForward() {
            if (currentStep < totalSteps - 1) {
                currentStep++;
                updateDisplay();
            }
        }

        // 绑定按钮点击事件
        document.getElementById('scene1-btn-prev-bg').addEventListener('click', stepBack);
        document.getElementById('scene1-btn-next-bg').addEventListener('click', stepForward);

        // 初始化显示
        updateDisplay();
    })();
    </script>
