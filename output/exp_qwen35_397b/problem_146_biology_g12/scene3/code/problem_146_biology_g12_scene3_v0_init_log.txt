    <script>
    (function() {
        // Scene 3 实现
        let currentStep = 0;
        const totalSteps = 6;

        // 工具函数：显示/隐藏元素
        function showElement(id) {
            const el = document.getElementById(id);
            if (el) {
                el.classList.remove('scene3-hidden');
                el.classList.add('scene3-visible');
            }
        }

        function hideElement(id) {
            const el = document.getElementById(id);
            if (el) {
                el.classList.remove('scene3-visible');
                el.classList.add('scene3-hidden');
            }
        }

        // 更新步骤显示
        function updateStep() {
            // 隐藏所有步骤
            for (let i = 0; i < totalSteps; i++) {
                hideElement('scene3-step' + i);
            }

            // 显示当前步骤
            showElement('scene3-step' + currentStep);

            // 更新步骤指示器
            const indicator = document.getElementById('scene3-step-indicator');
            if (indicator) {
                indicator.textContent = '步骤 ' + (currentStep + 1) + '/' + totalSteps;
            }

            // 更新按钮状态
            const prevBtn = document.getElementById('scene3-prev-btn');
            const nextBtn = document.getElementById('scene3-next-btn');
            
            if (prevBtn) {
                prevBtn.style.opacity = currentStep === 0 ? '0.5' : '1';
                prevBtn.style.pointerEvents = currentStep === 0 ? 'none' : 'auto';
            }
            
            if (nextBtn) {
                nextBtn.style.opacity = currentStep === totalSteps - 1 ? '0.5' : '1';
                nextBtn.style.pointerEvents = currentStep === totalSteps - 1 ? 'none' : 'auto';
            }
        }

        // 事件处理
        document.getElementById('scene3-prev-btn').addEventListener('click', function() {
            if (currentStep > 0) {
                currentStep--;
                updateStep();
            }
        });

        document.getElementById('scene3-next-btn').addEventListener('click', function() {
            if (currentStep < totalSteps - 1) {
                currentStep++;
                updateStep();
            }
        });

        // 初始化
        updateStep();
    })();
    </script>
