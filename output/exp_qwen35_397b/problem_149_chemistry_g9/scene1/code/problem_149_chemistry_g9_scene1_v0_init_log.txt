    <script>
    (function() {
        // 状态配置
        const steps = [
            { // 步骤 0: 初始状态
                highlightBox: null,
                labels: []
            },
            { // 步骤 1: 高亮甲框中的 4 个 a₂分子
                highlightBox: { x: 210, y: 95, width: 190, height: 70 },
                labels: ['scene1-label-a2-count']
            },
            { // 步骤 2: 保持步骤 1，新增高亮 cb₂c 分子
                highlightBox: { x: 210, y: 95, width: 190, height: 110 },
                labels: ['scene1-label-a2-count', 'scene1-label-cb2c-count']
            },
            { // 步骤 3: 高亮乙框中的 c 原子
                highlightBox: null,
                labels: ['scene1-label-c-atom'],
                highlightElements: ['scene1-yi-c-1', 'scene1-yi-c-2']
            },
            { // 步骤 4: 保持步骤 3，新增高亮 b₂分子
                highlightBox: null,
                labels: ['scene1-label-c-atom', 'scene1-label-b2-mol'],
                highlightElements: ['scene1-yi-c-1', 'scene1-yi-c-2', 'scene1-yi-b2-1', 'scene1-yi-b2-2']
            },
            { // 步骤 5: 高亮乙框中的 a₂b 分子（生成物）
                highlightBox: { x: 515, y: 75, width: 160, height: 140 },
                labels: ['scene1-label-product'],
                highlightElements: ['scene1-yi-a2b-1', 'scene1-yi-a2b-2', 'scene1-yi-a2b-3', 'scene1-yi-a2b-4']
            },
            { // 步骤 6: 显示结论
                highlightBox: { x: 515, y: 75, width: 160, height: 140 },
                labels: ['scene1-label-product', 'scene1-conclusion'],
                highlightElements: ['scene1-yi-a2b-1', 'scene1-yi-a2b-2', 'scene1-yi-a2b-3', 'scene1-yi-a2b-4']
            }
        ];
        
        let currentState = 0;
        
        const highlightBox = document.getElementById('scene1-highlight-box');
        const stepIndicator = document.getElementById('scene1-step-indicator');
        const btnPrev = document.getElementById('scene1-btn-prev');
        const btnNext = document.getElementById('scene1-btn-next');
        
        // 所有动态标签
        const allLabels = [
            'scene1-label-a2-count',
            'scene1-label-cb2c-count',
            'scene1-label-c-atom',
            'scene1-label-b2-mol',
            'scene1-label-product',
            'scene1-conclusion'
        ];
        
        // 需要高亮的元素
        const allHighlightElements = [
            'scene1-yi-c-1', 'scene1-yi-c-2',
            'scene1-yi-b2-1', 'scene1-yi-b2-2',
            'scene1-yi-a2b-1', 'scene1-yi-a2b-2',
            'scene1-yi-a2b-3', 'scene1-yi-a2b-4'
        ];
        
        function hideElement(id) {
            const el = document.getElementById(id);
            if (el) {
                el.style.opacity = '0';
                el.style.display = 'none';
            }
        }
        
        function showElement(id) {
            const el = document.getElementById(id);
            if (el) {
                el.style.opacity = '1';
                el.style.display = '';
            }
        }
        
        function updateState() {
            const step = steps[currentState];
            
            // 更新步骤指示器
            stepIndicator.textContent = '步骤 ' + currentState + '/6';
            
            // 更新按钮状态
            btnPrev.disabled = (currentState === 0);
            btnNext.disabled = (currentState === 6);
            
            // 隐藏所有标签
            allLabels.forEach(id => hideElement(id));
            
            // 隐藏所有高亮元素样式
            allHighlightElements.forEach(id => {
                const el = document.getElementById(id);
                if (el) {
                    el.style.opacity = '1';
                    el.style.filter = '';
                }
            });
            
            // 隐藏高亮框
            highlightBox.style.opacity = '0';
            highlightBox.setAttribute('width', '0');
            highlightBox.setAttribute('height', '0');
            
            // 显示当前步骤的标签
            step.labels.forEach(id => showElement(id));
            
            // 显示高亮框
            if (step.highlightBox) {
                highlightBox.setAttribute('x', step.highlightBox.x);
                highlightBox.setAttribute('y', step.highlightBox.y);
                highlightBox.setAttribute('width', step.highlightBox.width);
                highlightBox.setAttribute('height', step.highlightBox.height);
                highlightBox.style.opacity = '1';
            }
            
            // 高亮特定元素
            if (step.highlightElements) {
                step.highlightElements.forEach(id => {
                    const el = document.getElementById(id);
                    if (el) {
                        el.style.filter = 'brightness(1.3)';
                    }
                });
            }
        }
        
        // 按钮事件绑定
        btnPrev.addEventListener('click', function() {
            if (currentState > 0) {
                currentState--;
                updateState();
            }
        });
        
        btnNext.addEventListener('click', function() {
            if (currentState < 6) {
                currentState++;
                updateState();
            }
        });
        
        // 初始化
        updateState();
    })();
    </script>
