    <script>
    (function() {
        // Scene 1: 测定空气中氧气含量实验装置
        // 布局坐标表参考：
        // 集气瓶：(100,180)-(260,400), 瓶口：(115,160)-(245,180), 橡皮塞：(110,145)-(245,160)
        // 刻度线：y=370/315/260/205, x=90-100
        // 燃烧匙：柄 (180,155)-(180,270), 斗 (180,285) 椭圆
        // 导管：(180,155)-(180,120)-(520,120)-(520,140)-(620,140)-(620,320)
        // 止水夹：(435,105)-(480,135)
        // 烧杯：(560,200)-(740,380) 梯形
        // 控件栏：y=440-500

        let currentState = 0;
        const totalSteps = 6;

        // 获取 DOM 元素
        const btnPrev = document.getElementById('scene1-btn-prev');
        const btnNext = document.getElementById('scene1-btn-next');
        const stepIndicator = document.getElementById('scene1-step-indicator');

        // 元素分组
        const elements = {
            scales: document.getElementById('scene1-scales'),
            waterBottle: document.getElementById('scene1-water-bottle'),
            spoonGroup: document.getElementById('scene1-spoon-group'),
            waterBeakerGroup: document.getElementById('scene1-water-beaker-group'),
            labelBottle: document.getElementById('scene1-label-bottle'),
            labelSpoon: document.getElementById('scene1-label-spoon'),
            labelPhosphorus: document.getElementById('scene1-label-phosphorus'),
            labelTube: document.getElementById('scene1-label-tube'),
            labelClamp: document.getElementById('scene1-label-clamp'),
            labelBeaker: document.getElementById('scene1-label-beaker'),
            labelWater: document.getElementById('scene1-label-water'),
            bottle: document.getElementById('scene1-bottle'),
            tube: document.getElementById('scene1-tube'),
            beaker: document.getElementById('scene1-beaker')
        };

        // 工具函数：显示元素
        function show(el) {
            if (el) {
                el.classList.remove('scene1-hidden');
                el.style.display = '';
                el.style.opacity = '1';
            }
        }

        // 工具函数：隐藏元素
        function hide(el) {
            if (el) {
                el.classList.add('scene1-hidden');
                el.style.display = 'none';
                el.style.opacity = '0';
            }
        }

        // 清除所有高亮
        function clearHighlights() {
            elements.bottle.classList.remove('scene1-highlight-bottle');
            elements.tube.classList.remove('scene1-highlight-tube');
            elements.beaker.classList.remove('scene1-highlight-beaker');
            const spoonHandle = document.getElementById('scene1-spoon-handle');
            const spoonBowl = document.getElementById('scene1-spoon-bowl');
            if (spoonHandle) spoonHandle.classList.remove('scene1-highlight-spoon');
            if (spoonBowl) spoonBowl.classList.remove('scene1-highlight-spoon');
        }

        // 更新显示状态
        function updateState() {
            // 清除所有高亮
            clearHighlights();

            // 根据当前状态显示/隐藏元素
            switch (currentState) {
                case 0:
                    // 步骤 0：基础装置轮廓
                    hide(elements.scales);
                    hide(elements.waterBottle);
                    hide(elements.spoonGroup);
                    hide(elements.waterBeakerGroup);
                    hide(elements.labelBottle);
                    hide(elements.labelSpoon);
                    hide(elements.labelPhosphorus);
                    hide(elements.labelTube);
                    hide(elements.labelClamp);
                    hide(elements.labelBeaker);
                    hide(elements.labelWater);
                    break;

                case 1:
                    // 步骤 1：标注集气瓶和刻度
                    show(elements.scales);
                    show(elements.waterBottle);
                    show(elements.labelBottle);
                    hide(elements.spoonGroup);
                    hide(elements.waterBeakerGroup);
                    hide(elements.labelSpoon);
                    hide(elements.labelPhosphorus);
                    hide(elements.labelTube);
                    hide(elements.labelClamp);
                    hide(elements.labelBeaker);
                    hide(elements.labelWater);
                    // 高亮集气瓶
                    elements.bottle.classList.add('scene1-highlight-bottle');
                    break;

                case 2:
                    // 步骤 2：标注燃烧匙和红磷
                    show(elements.scales);
                    show(elements.waterBottle);
                    show(elements.spoonGroup);
                    show(elements.labelBottle);
                    show(elements.labelSpoon);
                    show(elements.labelPhosphorus);
                    hide(elements.waterBeakerGroup);
                    hide(elements.labelTube);
                    hide(elements.labelClamp);
                    hide(elements.labelBeaker);
                    hide(elements.labelWater);
                    // 高亮燃烧匙和红磷
                    const spoonHandle = document.getElementById('scene1-spoon-handle');
                    const spoonBowl = document.getElementById('scene1-spoon-bowl');
                    if (spoonHandle) spoonHandle.classList.add('scene1-highlight-spoon');
                    if (spoonBowl) spoonBowl.classList.add('scene1-highlight-spoon');
                    break;

                case 3:
                    // 步骤 3：标注导管和止水夹
                    show(elements.scales);
                    show(elements.waterBottle);
                    show(elements.spoonGroup);
                    show(elements.labelBottle);
                    show(elements.labelSpoon);
                    show(elements.labelPhosphorus);
                    show(elements.labelTube);
                    show(elements.labelClamp);
                    hide(elements.waterBeakerGroup);
                    hide(elements.labelBeaker);
                    hide(elements.labelWater);
                    // 高亮导管
                    elements.tube.classList.add('scene1-highlight-tube');
                    break;

                case 4:
                    // 步骤 4：标注烧杯和水
                    show(elements.scales);
                    show(elements.waterBottle);
                    show(elements.spoonGroup);
                    show(elements.labelBottle);
                    show(elements.labelSpoon);
                    show(elements.labelPhosphorus);
                    show(elements.labelTube);
                    show(elements.labelClamp);
                    show(elements.waterBeakerGroup);
                    show(elements.labelBeaker);
                    show(elements.labelWater);
                    // 高亮烧杯
                    elements.beaker.classList.add('scene1-highlight-beaker');
                    break;

                case 5:
                    // 步骤 5：完整标注图（无高亮）
                    show(elements.scales);
                    show(elements.waterBottle);
                    show(elements.spoonGroup);
                    show(elements.waterBeakerGroup);
                    show(elements.labelBottle);
                    show(elements.labelSpoon);
                    show(elements.labelPhosphorus);
                    show(elements.labelTube);
                    show(elements.labelClamp);
                    show(elements.labelBeaker);
                    show(elements.labelWater);
                    // 无高亮
                    break;
            }

            // 更新按钮状态
            btnPrev.disabled = (currentState === 0);
            btnNext.disabled = (currentState === totalSteps - 1);

            // 更新步骤指示器
            stepIndicator.textContent = `步骤 ${currentState + 1}/${totalSteps}`;
        }

        // 事件监听
        btnPrev.addEventListener('click', function() {
            if (currentState > 0) {
                currentState--;
                updateState();
            }
        });

        btnNext.addEventListener('click', function() {
            if (currentState < totalSteps - 1) {
                currentState++;
                updateState();
            }
        });

        // 初始化
        updateState();
    })();
    </script>
