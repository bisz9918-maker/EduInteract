    <script>
    (function() {
        // Scene 2 实现 - 物质推断推理过程

        // 状态变量
        let currentStep = 0;
        const totalSteps = 6;

        // 每步配置
        const steps = [
            {
                // 步骤 0: 初始状态
                reasoning: "已知条件：\n• B 为最常见的酸 → H₂SO₄\n• D 为人体中含量最多的物质 → H₂O\n• G 为蓝色不溶物 → Cu(OH)₂↓",
                highlightBoxes: ['scene2-box-b', 'scene2-box-d1', 'scene2-box-g'],
                chemicalTexts: [],
                showAnswers: false
            },
            {
                // 步骤 1: 分析反应③
                reasoning: "分析反应③：E + F → G(↓) + ...\nG 为蓝色沉淀 Cu(OH)₂\n说明反应物中一个含 Cu²⁺，另一个含 OH⁻",
                highlightBoxes: ['scene2-box-e1', 'scene2-box-f', 'scene2-box-g', 'scene2-reaction-3'],
                chemicalTexts: [],
                showAnswers: false
            },
            {
                // 步骤 2: 推导 E
                reasoning: "由反应①：A + 酸 B → E + H₂O\nE 是铜盐，推测 E = CuSO₄",
                highlightBoxes: ['scene2-box-e1', 'scene2-reaction-1'],
                chemicalTexts: ['scene2-label-e1'],
                showAnswers: false
            },
            {
                // 步骤 3: 推导 F
                reasoning: "由反应③：CuSO₄ + F → Cu(OH)₂↓ + ...\nF 必须是碱，推测 F = NaOH",
                highlightBoxes: ['scene2-box-f', 'scene2-reaction-3'],
                chemicalTexts: ['scene2-label-f'],
                showAnswers: false
            },
            {
                // 步骤 4: 推导 A
                reasoning: "由反应①：A + H₂SO₄ → CuSO₄ + H₂O\nA 可能是 CuO（氧化物）或 Cu(OH)₂（碱）",
                highlightBoxes: ['scene2-box-a', 'scene2-reaction-1'],
                chemicalTexts: ['scene2-label-a'],
                showAnswers: false
            },
            {
                // 步骤 5: 推导 C
                reasoning: "由反应②：C + H₂SO₄ → Na₂SO₄ + H₂O + CO₂↑\nC 是碳酸盐，推测 C = Na₂CO₃",
                highlightBoxes: ['scene2-box-c', 'scene2-reaction-2'],
                chemicalTexts: ['scene2-label-c'],
                showAnswers: false
            },
            {
                // 步骤 6: 最终答案
                reasoning: "推断完成！\n\n反应方程式：\n① CuO + H₂SO₄ → CuSO₄ + H₂O\n② Na₂CO₃ + H₂SO₄ → Na₂SO₄ + H₂O + CO₂↑\n③ CuSO₄ + 2NaOH → Cu(OH)₂↓ + Na₂SO₄",
                highlightBoxes: [],
                chemicalTexts: ['scene2-label-e1', 'scene2-label-f', 'scene2-label-a', 'scene2-label-c'],
                showAnswers: true
            }
        ];

        // DOM 元素引用
        const reasoningText = document.getElementById('scene2-reasoning-text');
        const stepIndicator = document.getElementById('scene2-step-indicator');
        const btnPrev = document.getElementById('scene2-btn-prev-bg');
        const btnNext = document.getElementById('scene2-btn-next-bg');

        // 更新步骤显示
        function updateStep() {
            const step = steps[currentStep];

            // 更新推理文本
            reasoningText.textContent = step.reasoning;

            // 更新步骤指示器
            stepIndicator.textContent = `步骤 ${currentStep}/${totalSteps}`;

            // 清除所有高亮
            document.querySelectorAll('.scene2-highlight').forEach(el => {
                el.classList.remove('scene2-highlight');
            });

            // 应用新的高亮
            step.highlightBoxes.forEach(boxId => {
                const box = document.getElementById(boxId);
                if (box) {
                    box.classList.add('scene2-highlight');
                }
            });

            // 控制化学式文字显示
            document.querySelectorAll('.scene2-chemical-text').forEach(text => {
                text.classList.remove('visible');
            });
            step.chemicalTexts.forEach(textId => {
                const text = document.getElementById(textId);
                if (text) {
                    text.classList.add('visible');
                }
            });

            // 控制答案框显示
            const answerBoxes = ['scene2-answer-1', 'scene2-answer-2', 'scene2-answer-3'];
            const answerTexts = ['scene2-answer-text-1', 'scene2-answer-text-2', 'scene2-answer-text-3'];

            if (step.showAnswers) {
                answerBoxes.forEach(boxId => {
                    const box = document.getElementById(boxId);
                    if (box) box.style.opacity = '1';
                });
                answerTexts.forEach(textId => {
                    const text = document.getElementById(textId);
                    if (text) text.style.opacity = '1';
                });
            } else {
                answerBoxes.forEach(boxId => {
                    const box = document.getElementById(boxId);
                    if (box) box.style.opacity = '0';
                });
                answerTexts.forEach(textId => {
                    const text = document.getElementById(textId);
                    if (text) text.style.opacity = '0';
                });
            }

            // 更新按钮状态
            if (currentStep === 0) {
                btnPrev.style.fill = '#bdbdbd';
            } else {
                btnPrev.style.fill = '#1976d2';
            }

            if (currentStep === totalSteps) {
                btnNext.style.fill = '#bdbdbd';
            } else {
                btnNext.style.fill = '#1976d2';
            }
        }

        // 前进一步
        function stepForward() {
            if (currentStep < totalSteps) {
                currentStep++;
                updateStep();
            }
        }

        // 后退一步
        function stepBackward() {
            if (currentStep > 0) {
                currentStep--;
                updateStep();
            }
        }

        // 事件绑定
        btnNext.addEventListener('click', stepForward);
        btnPrev.addEventListener('click', stepBackward);

        // 初始化
        updateStep();
    })();
    </script>
