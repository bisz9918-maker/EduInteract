    (function() {
        // Scene 2 实现：关键难点突破——Fe₂O₃ 与酸反应的产物价态
        
        // 布局坐标表（实现对照依据）：
        // | 元素 | 类型 | 位置/范围 | 大小 | 备注 |
        // |------|------|-----------|------|------|
        // | 左侧标题 | text | (130,45) | - | "错误路径分析" |
        // | 错误方程式 | text | (130,80) | - | Fe₂O₃ + H₂SO₄ → FeSO₄ |
        // | 错误反应框 | rect | (30,100)-(230,170) | 200×70 | 红色边框 #c62828 |
        // | 错误标记 | path | (190-210,115-155) | 30×30 | 红色大叉 × |
        // | 中部标题 | text | (400,45) | - | "价态对比分析" |
        // | 价态数轴线 | line | (280,110)-(520,110) | 240×3 | 水平数轴 |
        // | 0/+2/+3 价刻度 | line | 300/400/500,105-115 | - | 数轴刻度 |
        // | Fe²⁺标记点 | circle | (400,90) | r=8 | 蓝色 #1565c0 |
        // | Fe³⁺标记点 | circle | (500,90) | r=8 | 蓝色 #1565c0 |
        // | 右侧标题 | text | (665,45) | - | "正确转化路径" |
        // | 正确路径框 1 | rect | (560,95)-(770,140) | 210×45 | 绿色边框 #2e7d32 |
        // | 正确路径框 2 | rect | (560,180)-(770,225) | 210×45 | 绿色边框 #2e7d32 |
        // | 底部说明 | text | (40,340)/(40,365) | - | 核心原理说明 |
        // | 状态按钮 | text+line | 底部三个 | - | 文字 + 下划线 |

        let currentState = 1;

        const btn1Text = document.getElementById('scene2-btn1-text');
        const btn1Underline = document.getElementById('scene2-btn1-underline');
        const btn2Text = document.getElementById('scene2-btn2-text');
        const btn2Underline = document.getElementById('scene2-btn2-underline');
        const btn3Text = document.getElementById('scene2-btn3-text');
        const btn3Underline = document.getElementById('scene2-btn3-underline');

        const groupError = document.getElementById('scene2-group-error');
        const groupValence = document.getElementById('scene2-group-valence');
        const groupCorrect = document.getElementById('scene2-group-correct');

        function updateButtons() {
            // 重置所有按钮为未激活状态
            btn1Text.setAttribute('fill', '#666');
            btn1Text.setAttribute('font-weight', 'normal');
            btn1Underline.setAttribute('stroke', '#ccc');
            
            btn2Text.setAttribute('fill', '#666');
            btn2Text.setAttribute('font-weight', 'normal');
            btn2Underline.setAttribute('stroke', '#ccc');
            
            btn3Text.setAttribute('fill', '#666');
            btn3Text.setAttribute('font-weight', 'normal');
            btn3Underline.setAttribute('stroke', '#ccc');

            // 设置当前激活按钮
            if (currentState === 1) {
                btn1Text.setAttribute('fill', '#1565c0');
                btn1Text.setAttribute('font-weight', 'bold');
                btn1Underline.setAttribute('stroke', '#1565c0');
            } else if (currentState === 2) {
                btn2Text.setAttribute('fill', '#1565c0');
                btn2Text.setAttribute('font-weight', 'bold');
                btn2Underline.setAttribute('stroke', '#1565c0');
            } else if (currentState === 3) {
                btn3Text.setAttribute('fill', '#1565c0');
                btn3Text.setAttribute('font-weight', 'bold');
                btn3Underline.setAttribute('stroke', '#1565c0');
            }
        }

        function updateState() {
            // 隐藏所有状态组
            groupError.classList.remove('active');
            groupValence.classList.remove('active');
            groupCorrect.classList.remove('active');

            // 显示当前状态组
            if (currentState === 1) {
                groupError.classList.add('active');
            } else if (currentState === 2) {
                groupValence.classList.add('active');
            } else if (currentState === 3) {
                groupCorrect.classList.add('active');
            }

            updateButtons();
        }

        // 按钮点击事件
        btn1Text.addEventListener('click', function() {
            currentState = 1;
            updateState();
        });
        btn2Text.addEventListener('click', function() {
            currentState = 2;
            updateState();
        });
        btn3Text.addEventListener('click', function() {
            currentState = 3;
            updateState();
        });

        // 初始化
        updateState();
    })();
    </script>
