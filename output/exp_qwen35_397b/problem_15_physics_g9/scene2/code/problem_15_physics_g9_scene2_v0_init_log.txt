    <script>
    (function() {
        // Scene 2: 图乙并排受力分析
        // 布局坐标表参考：
        // | 元素 | 位置 | 备注 |
        // |------|------|------|
        // | 桌面 | (80,350)-(720,350) | 水平主线 |
        // | 木块 A | (280,280) 100×70 | 左侧木块 |
        // | 木块 B | (380,280) 100×70 | 右侧木块 |
        // | 标签 A | (330,315) | 木块 A 中心 |
        // | 标签 B | (430,315) | 木块 B 中心 |
        // | 推力 F₂ | (180,315)-(280,315) | 蓝色箭头 |
        // | F₂=? 标签 | (230,295) | 推力上方 |
        // | 摩擦力 f₂ | (280,355)-(480,355) | 红色向左 |
        // | f₂ 标签 | (380,375) | 摩擦力下方 |
        // | 重力 G | (380,210)-(380,280) | 灰色虚线向下 |
        // | N=2mg 标签 | (405,250) | 重力右侧 |
        // | 支持力 N | (380,350)-(380,280) | 灰色虚线向上 |
        // | 速度 v₂ | (280,250)-(480,250) | 黑色向右 |
        // | v₂=2m/s | (380,230) | 速度上方 |

        let currentState = 0;
        const totalSteps = 3;

        // 各步骤应显示的元素（除了始终显示的桌面、木块、标签 A/B）
        const stepElements = {
            0: ['scene2-arrow-v', 'scene2-label-v'],
            1: ['scene2-arrow-v', 'scene2-label-v', 'scene2-arrow-g', 'scene2-arrow-n', 'scene2-label-n'],
            2: ['scene2-arrow-v', 'scene2-label-v', 'scene2-arrow-g', 'scene2-arrow-n', 'scene2-label-n', 'scene2-arrow-fri', 'scene2-label-fri'],
            3: ['scene2-arrow-v', 'scene2-label-v', 'scene2-arrow-g', 'scene2-arrow-n', 'scene2-label-n', 'scene2-arrow-fri', 'scene2-label-fri', 'scene2-arrow-f2', 'scene2-label-f2']
        };

        // 工具函数：显示元素
        function showElement(id) {
            const el = document.getElementById(id);
            if (el) {
                el.style.opacity = '1';
                el.style.display = '';
            }
        }

        // 工具函数：隐藏元素
        function hideElement(id) {
            const el = document.getElementById(id);
            if (el) {
                el.style.opacity = '0';
                el.style.display = 'none';
            }
        }

        // 更新步骤显示
        function updateStep() {
            // 隐藏所有可隐藏元素
            const allElements = ['scene2-arrow-v', 'scene2-label-v', 'scene2-arrow-g', 'scene2-arrow-n', 'scene2-label-n', 'scene2-arrow-fri', 'scene2-label-fri', 'scene2-arrow-f2', 'scene2-label-f2'];
            allElements.forEach(id => hideElement(id));

            // 显示当前步骤的元素
            stepElements[currentState].forEach(id => showElement(id));

            // 更新步骤指示器
            const indicator = document.getElementById('scene2-step-indicator');
            if (indicator) {
                indicator.textContent = '步骤 ' + currentState + '/' + totalSteps;
            }

            // 更新按钮状态
            const prevBtn = document.getElementById('scene2-btn-prev');
            const nextBtn = document.getElementById('scene2-btn-next');
            if (prevBtn) {
                prevBtn.disabled = (currentState === 0);
            }
            if (nextBtn) {
                nextBtn.disabled = (currentState === totalSteps);
            }

            // 步骤 3 时更新 F₂ 标签为 F₂=f₂
            const f2Label = document.getElementById('scene2-label-f2');
            if (f2Label) {
                if (currentState === 3) {
                    f2Label.textContent = 'F₂=f₂';
                } else {
                    f2Label.textContent = 'F₂=?';
                }
            }
        }

        // 按钮事件绑定
        document.getElementById('scene2-btn-prev').addEventListener('click', function() {
            if (currentState > 0) {
                currentState--;
                updateStep();
            }
        });

        document.getElementById('scene2-btn-next').addEventListener('click', function() {
            if (currentState < totalSteps) {
                currentState++;
                updateStep();
            }
        });

        // 初始化
        updateStep();
    })();
    </script>
