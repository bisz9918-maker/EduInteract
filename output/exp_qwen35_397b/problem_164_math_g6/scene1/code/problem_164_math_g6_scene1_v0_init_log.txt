    <script>
    (function() {
        // Scene 1 实现 - 梯形四棱柱的三视图

        const steps = [
            // 步骤 0: 仅显示立体图
            { show: [], hide: ['scene1-arrow-front', 'scene1-arrow-side', 'scene1-arrow-top', 'scene1-proj-front', 'scene1-proj-side', 'scene1-proj-top', 'scene1-front-view', 'scene1-side-view', 'scene1-top-view', 'scene1-highlight-front', 'scene1-highlight-side', 'scene1-highlight-top'] },
            // 步骤 1: 显示正面箭头、投影线、正面视图
            { show: ['scene1-arrow-front', 'scene1-proj-front', 'scene1-front-view'], hide: [] },
            // 步骤 2: 显示侧面箭头、投影线、侧面视图
            { show: ['scene1-arrow-side', 'scene1-proj-side', 'scene1-side-view'], hide: [] },
            // 步骤 3: 显示上面箭头、投影线、上面视图
            { show: ['scene1-arrow-top', 'scene1-proj-top', 'scene1-top-view'], hide: [] },
            // 步骤 4: 显示高亮轮廓
            { show: ['scene1-highlight-front', 'scene1-highlight-side', 'scene1-highlight-top'], hide: [] }
        ];

        let currentStep = 0;

        const prevBtn = document.getElementById('scene1-prev-btn');
        const nextBtn = document.getElementById('scene1-next-btn');
        const stepIndicator = document.getElementById('scene1-step-indicator');

        function updateButtons() {
            prevBtn.disabled = currentStep === 0;
            nextBtn.disabled = currentStep === 4;
            stepIndicator.textContent = `步骤 ${currentStep}/4`;
        }

        function showElement(id) {
            const el = document.getElementById(id);
            if (el) {
                el.style.opacity = '1';
            }
        }

        function hideElement(id) {
            const el = document.getElementById(id);
            if (el) {
                el.style.opacity = '0';
            }
        }

        function updateStep() {
            const step = steps[currentStep];
            step.show.forEach(id => showElement(id));
            step.hide.forEach(id => hideElement(id));
            updateButtons();
        }

        prevBtn.addEventListener('click', () => {
            if (currentStep > 0) {
                // 隐藏当前步骤显示的元素
                const prevStep = steps[currentStep];
                prevStep.show.forEach(id => hideElement(id));
                currentStep--;
                updateStep();
            }
        });

        nextBtn.addEventListener('click', () => {
            if (currentStep < 4) {
                currentStep++;
                updateStep();
            }
        });

        // 初始化
        updateButtons();
    })();
    </script>
