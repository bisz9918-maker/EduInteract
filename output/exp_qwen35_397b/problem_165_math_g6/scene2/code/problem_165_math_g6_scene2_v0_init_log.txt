    <script>
    (function() {
        // Scene 2 实现：地砖面积计算
        
        // 生成大地砖网格线 (20×20格，每格25px)
        const largeGridVlines = document.getElementById('scene2-large-grid-vlines');
        const largeGridHlines = document.getElementById('scene2-large-grid-hlines');
        for (let i = 1; i < 20; i++) {
            const x = 150 + i * 25;
            const vline = document.createElementNS('http://www.w3.org/2000/svg', 'line');
            vline.setAttribute('x1', x);
            vline.setAttribute('y1', 60);
            vline.setAttribute('x2', x);
            vline.setAttribute('y2', 310);
            vline.setAttribute('class', 'scene2-large-grid-line');
            largeGridVlines.appendChild(vline);
            
            const y = 60 + i * 25;
            const hline = document.createElementNS('http://www.w3.org/2000/svg', 'line');
            hline.setAttribute('x1', 150);
            hline.setAttribute('y1', y);
            hline.setAttribute('x2', 650);
            hline.setAttribute('y2', y);
            hline.setAttribute('class', 'scene2-large-grid-line');
            largeGridHlines.appendChild(hline);
        }
        
        // 生成小地砖网格线 (30×30格，每格约16.67px)
        const smallGridVlines = document.getElementById('scene2-small-grid-vlines');
        const smallGridHlines = document.getElementById('scene2-small-grid-hlines');
        const smallCellSize = 250 / 30;
        for (let i = 1; i < 30; i++) {
            const x = 150 + i * smallCellSize;
            const vline = document.createElementNS('http://www.w3.org/2000/svg', 'line');
            vline.setAttribute('x1', x);
            vline.setAttribute('y1', 60);
            vline.setAttribute('x2', x);
            vline.setAttribute('y2', 310);
            vline.setAttribute('class', 'scene2-small-grid-line');
            smallGridVlines.appendChild(vline);
            
            const y = 60 + i * smallCellSize;
            const hline = document.createElementNS('http://www.w3.org/2000/svg', 'line');
            hline.setAttribute('x1', 150);
            hline.setAttribute('y1', y);
            hline.setAttribute('x2', 650);
            hline.setAttribute('y2', y);
            hline.setAttribute('class', 'scene2-small-grid-line');
            smallGridHlines.appendChild(hline);
        }
        
        // 状态管理
        let currentStep = 0;
        const totalSteps = 3;
        
        // 元素引用
        const elements = {
            largeGrid: document.getElementById('scene2-large-grid'),
            smallGrid: document.getElementById('scene2-small-grid'),
            tileLargeSample: document.getElementById('scene2-tile-large-sample'),
            tileSmallSample: document.getElementById('scene2-tile-small-sample'),
            label3dm: document.getElementById('scene2-label-3dm'),
            label2dm: document.getElementById('scene2-label-2dm'),
            label400: document.getElementById('scene2-label-400'),
            label900: document.getElementById('scene2-label-900'),
            formula1: document.getElementById('scene2-formula1'),
            formula2: document.getElementById('scene2-formula2'),
            formula3: document.getElementById('scene2-formula3'),
            answerBox: document.getElementById('scene2-answer-box'),
            btnPrev: document.getElementById('scene2-btn-prev'),
            btnNext: document.getElementById('scene2-btn-next')
        };
        
        // 显示/隐藏工具函数
        function show(el) {
            if (el) {
                el.classList.remove('scene2-hidden');
                el.classList.add('scene2-visible');
                if (el.tagName === 'g' || el.tagName === 'rect' || el.tagName === 'text') {
                    el.style.display = '';
                    el.style.opacity = '1';
                }
            }
        }
        
        function hide(el) {
            if (el) {
                el.classList.remove('scene2-visible');
                el.classList.add('scene2-hidden');
                if (el.tagName === 'g' || el.tagName === 'rect' || el.tagName === 'text') {
                    el.style.display = '';
                    el.style.opacity = '0';
                }
            }
        }
        
        // 更新步骤显示
        function updateStep() {
            // 重置所有元素状态
            hide(elements.formula1);
            hide(elements.formula2);
            hide(elements.formula3);
            hide(elements.answerBox);
            hide(elements.tileSmallSample);
            hide(elements.label2dm);
            hide(elements.label900);
            
            show(elements.largeGrid);
            show(elements.tileLargeSample);
            show(elements.label3dm);
            show(elements.label400);
            hide(elements.smallGrid);
            
            // 根据当前步骤显示相应元素
            if (currentStep >= 1) {
                show(elements.formula1);
            }
            
            if (currentStep >= 2) {
                hide(elements.largeGrid);
                show(elements.smallGrid);
                show(elements.tileSmallSample);
                show(elements.label2dm);
                show(elements.formula2);
            }
            
            if (currentStep >= 3) {
                show(elements.formula3);
                show(elements.answerBox);
                hide(elements.label400);
                show(elements.label900);
            }
            
            // 更新按钮状态
            if (currentStep === 0) {
                elements.btnPrev.classList.add('scene2-btn-disabled');
            } else {
                elements.btnPrev.classList.remove('scene2-btn-disabled');
            }
            
            if (currentStep >= totalSteps) {
                elements.btnNext.classList.add('scene2-btn-disabled');
            } else {
                elements.btnNext.classList.remove('scene2-btn-disabled');
            }
        }
        
        // 按钮点击事件
        elements.btnPrev.addEventListener('click', function() {
            if (currentStep > 0) {
                currentStep--;
                updateStep();
            }
        });
        
        elements.btnNext.addEventListener('click', function() {
            if (currentStep < totalSteps) {
                currentStep++;
                updateStep();
            }
        });
        
        // 初始化
        updateStep();
    })();
    </script>
