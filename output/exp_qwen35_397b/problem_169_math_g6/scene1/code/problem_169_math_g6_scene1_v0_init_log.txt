    <script>
    (function() {
        // Scene 1: 方位角与距离 - 巡洋舰位置判断
        // 布局坐标表参考：
        // | 元素 | 类型 | 位置/范围 | 大小 | 标签偏移方向 | 备注 |
        // |------|------|-----------|------|-------------|------|
        // | 北方向轴 | 线段 | (400,70)-(400,200) | 4×130 | - | 垂直向上 |
        // | 南方向轴 | 线段 | (400,300)-(400,400) | 4×100 | - | 垂直向下 |
        // | 西方向轴 | 线段 | (150,250)-(350,250) | 200×4 | - | 水平向左 |
        // | 东方向轴 | 线段 | (450,250)-(650,250) | 200×4 | - | 水平向右 |
        // | 雷达站 | 实心圆 | (400,250) | r=6 | - | 坐标原点 |
        // | 巡洋舰 | 实心圆 | (503,127) | r=5 | - | 北偏东 40°方向 |
        // | 雷达站 - 巡洋舰连线 | 线段 | (400,250)-(503,127) | 约 160×110 | - | 蓝色高亮 |
        // | 40°角度弧 | 弧线 | 圆心 (400,250), r=50, 从 90°到 50° | 弧长≈35 | - | 红色高亮 |
        // | 40°标签 | text | (470,175) | ~35×16 | 右侧 | 角度数值 |
        // | 北标签 | text | (400,55) | ~20×16 | 上方 | 北方向 |
        // | 南标签 | text | (400,425) | ~20×16 | 下方 | 南方向 |
        // | 西标签 | text | (130,255) | ~20×16 | 左侧 | 西方向 |
        // | 东标签 | text | (670,255) | ~20×16 | 右侧 | 东方向 |
        // | 雷达站标签 | text | (415,270) | ~40×16 | 右下方 | 原点标注 |
        // | 巡洋舰标签 | text | (515,120) | ~40×16 | 右上方 | 目标点标注 |
        // | 比例尺线段 | 线段 | (550,350)-(650,350) | 100×3 | - | 绿色高亮 |
        // | 比例尺左刻度 | 线段 | (550,345)-(550,355) | 3×10 | - | 比例尺起点 |
        // | 比例尺右刻度 | 线段 | (650,345)-(650,355) | 3×10 | - | 比例尺终点 |
        // | 50km 标签 | text | (600,375) | ~40×16 | 下方 | 比例尺数值 |
        // | 选项 A 框 | 矩形 | (80,440)-(240,485) | 160×45 | - | 北偏东 40°方向 50km |
        // | 选项 B 框 | 矩形 | (280,440)-(440,485) | 160×45 | - | 北偏西 40°方向 50km |
        // | 选项 C 框 | 矩形 | (480,440)-(640,485) | 160×45 | - | 北偏东 40°方向 100km |
        
        const totalSteps = 5;
        let currentStep = 0;
        
        // 各步骤需要显示的元素
        const stepElements = {
            0: [ // 初始状态：基础坐标系
                'scene1-axis-north', 'scene1-axis-south', 'scene1-axis-west', 'scene1-axis-east',
                'scene1-label-north', 'scene1-label-south', 'scene1-label-west', 'scene1-label-east',
                'scene1-radar-station', 'scene1-label-radar'
            ],
            1: [ // 步骤 1：巡洋舰位置
                'scene1-cruiser', 'scene1-label-cruiser', 'scene1-connection-line'
            ],
            2: [ // 步骤 2：40°角度
                'scene1-angle-arc', 'scene1-angle-arrow', 'scene1-label-angle'
            ],
            3: [ // 步骤 3：比例尺
                'scene1-scale-bar', 'scene1-scale-tick-left', 'scene1-scale-tick-right', 'scene1-label-scale'
            ],
            4: [ // 步骤 4：选项与结论
                'scene1-option-a-box', 'scene1-option-a-text', 'scene1-option-a-text2',
                'scene1-option-b-box', 'scene1-option-b-text', 'scene1-option-b-text2',
                'scene1-option-c-box', 'scene1-option-c-text', 'scene1-option-c-text2',
                'scene1-conclusion'
            ]
        };
        
        // 工具函数：显示元素
        function showElement(id) {
            const el = document.getElementById(id);
            if (el) {
                el.classList.remove('scene1-hidden');
                el.classList.add('scene1-visible', 'scene1-fade-in');
            }
        }
        
        // 工具函数：隐藏元素
        function hideElement(id) {
            const el = document.getElementById(id);
            if (el) {
                el.classList.remove('scene1-visible', 'scene1-fade-in');
                el.classList.add('scene1-hidden');
            }
        }
        
        // 更新步骤显示
        function updateStep() {
            // 隐藏所有步骤元素
            for (let step = 1; step < totalSteps; step++) {
                stepElements[step].forEach(id => hideElement(id));
            }
            
            // 显示当前步骤及之前步骤的元素
            for (let step = 0; step <= currentStep; step++) {
                stepElements[step].forEach(id => showElement(id));
            }
            
            // 步骤 4 特殊处理：高亮选项 A
            if (currentStep >= 4) {
                document.getElementById('scene1-option-a-box').classList.add('highlight');
                document.getElementById('scene1-option-a-text').classList.add('highlight');
                document.getElementById('scene1-option-a-text2').classList.add('highlight');
            }
            
            // 更新按钮状态
            const prevBtn = document.getElementById('scene1-prev-btn');
            const nextBtn = document.getElementById('scene1-next-btn');
            const indicator = document.getElementById('scene1-step-indicator');
            
            prevBtn.disabled = (currentStep === 0);
            nextBtn.disabled = (currentStep >= totalSteps - 1);
            indicator.textContent = `步骤 ${currentStep + 1}/${totalSteps}`;
        }
        
        // 事件绑定
        document.getElementById('scene1-prev-btn').addEventListener('click', function() {
            if (currentStep > 0) {
                currentStep--;
                updateStep();
            }
        });
        
        document.getElementById('scene1-next-btn').addEventListener('click', function() {
            if (currentStep < totalSteps - 1) {
                currentStep++;
                updateStep();
            }
        });
        
        // 初始化
        updateStep();
    })();
    </script>
