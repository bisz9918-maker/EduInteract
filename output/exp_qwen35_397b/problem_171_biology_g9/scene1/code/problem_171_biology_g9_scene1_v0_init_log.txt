    <script>
    (function() {
        // Scene 1 实现
        let currentStep = 0;
        const totalSteps = 5;
        
        // 获取 DOM 元素
        const boxes = {
            a: document.getElementById('scene1-box-a'),
            b: document.getElementById('scene1-box-b'),
            c: document.getElementById('scene1-box-c'),
            d: document.getElementById('scene1-box-d'),
            e: document.getElementById('scene1-box-e')
        };
        
        const annotations = {
            a: document.getElementById('scene1-annotation-a'),
            b: document.getElementById('scene1-annotation-b'),
            c: document.getElementById('scene1-annotation-c'),
            d: document.getElementById('scene1-annotation-d'),
            e: document.getElementById('scene1-annotation-e')
        };
        
        const moleculeElements = {
            base: document.getElementById('scene1-molecule-base'),
            baseLabel: document.getElementById('scene1-molecule-base-label'),
            sugar: document.getElementById('scene1-molecule-sugar'),
            sugarLabel: document.getElementById('scene1-molecule-sugar-label'),
            phosphate: document.getElementById('scene1-molecule-phosphate'),
            phosphateLabel: document.getElementById('scene1-molecule-phosphate-label'),
            conn1: document.getElementById('scene1-conn-1'),
            conn2: document.getElementById('scene1-conn-2'),
            nucleotideBox: document.getElementById('scene1-nucleotide-box'),
            nucleotideText: document.getElementById('scene1-nucleotide-text')
        };
        
        const dnaGroup = document.getElementById('scene1-dna-group');
        const boxTextD = document.getElementById('scene1-box-text-d');
        const stepIndicator = document.getElementById('scene1-step-indicator');
        
        const btnPrevBg = document.getElementById('scene1-btn-prev-bg');
        const btnPrevText = document.getElementById('scene1-btn-prev-text');
        const btnNextBg = document.getElementById('scene1-btn-next-bg');
        const btnNextText = document.getElementById('scene1-btn-next-text');
        
        // 辅助函数：显示/隐藏元素
        function show(el) {
            if (el) {
                el.classList.remove('hidden');
                el.style.display = '';
                el.style.opacity = '1';
            }
        }
        
        function hide(el) {
            if (el) {
                el.classList.add('hidden');
                el.style.display = 'none';
                el.style.opacity = '0';
            }
        }
        
        // 更新框的状态
        function updateBoxStates() {
            // 重置所有框为默认状态
            ['a', 'b', 'c', 'd', 'e'].forEach(key => {
                boxes[key].classList.remove('box-current', 'box-completed');
                boxes[key].classList.add('box-default');
            });
            
            // 设置已完成步骤（绿色）
            for (let i = 0; i < currentStep; i++) {
                const keys = ['a', 'b', 'c', 'd', 'e'];
                if (keys[i]) {
                    boxes[keys[i]].classList.remove('box-default', 'box-current');
                    boxes[keys[i]].classList.add('box-completed');
                }
            }
            
            // 设置当前步骤（蓝色高亮）
            if (currentStep > 0 && currentStep <= 5) {
                const keys = ['a', 'b', 'c', 'd', 'e'];
                const currentKey = keys[currentStep - 1];
                if (currentKey && boxes[currentKey]) {
                    boxes[currentKey].classList.remove('box-default', 'box-completed');
                    boxes[currentKey].classList.add('box-current');
                }
            }
        }
        
        // 更新标注显示
        function updateAnnotations() {
            // 隐藏所有标注
            ['a', 'b', 'c', 'd', 'e'].forEach(key => {
                hide(annotations[key]);
            });
            
            // 显示当前步骤的标注
            if (currentStep > 0 && currentStep <= 5) {
                const keys = ['a', 'b', 'c', 'd', 'e'];
                const currentKey = keys[currentStep - 1];
                if (currentKey && annotations[currentKey]) {
                    show(annotations[currentKey]);
                }
            }
        }
        
        // 更新分子结构区
        function updateMoleculeArea() {
            // 隐藏所有分子元素
            hide(moleculeElements.base);
            hide(moleculeElements.baseLabel);
            hide(moleculeElements.sugar);
            hide(moleculeElements.sugarLabel);
            hide(moleculeElements.phosphate);
            hide(moleculeElements.phosphateLabel);
            hide(moleculeElements.conn1);
            hide(moleculeElements.conn2);
            hide(moleculeElements.nucleotideBox);
            hide(moleculeElements.nucleotideText);
            hide(dnaGroup);
            
            // 步骤 1：显示含氮碱基
            if (currentStep >= 1) {
                show(moleculeElements.base);
                show(moleculeElements.baseLabel);
            }
            
            // 步骤 2：显示脱氧核糖
            if (currentStep >= 2) {
                show(moleculeElements.sugar);
                show(moleculeElements.sugarLabel);
            }
            
            // 步骤 3：显示磷酸和连接线
            if (currentStep >= 3) {
                show(moleculeElements.phosphate);
                show(moleculeElements.phosphateLabel);
                show(moleculeElements.conn1);
                show(moleculeElements.conn2);
            }
            
            // 步骤 4：显示核苷酸标注框
            if (currentStep >= 4) {
                show(moleculeElements.nucleotideBox);
                show(moleculeElements.nucleotideText);
                boxTextD.textContent = '脱氧核苷酸';
            } else {
                boxTextD.textContent = '( )';
            }
            
            // 步骤 5：显示 DNA 双螺旋
            if (currentStep >= 5) {
                hide(moleculeElements.base);
                hide(moleculeElements.baseLabel);
                hide(moleculeElements.sugar);
                hide(moleculeElements.sugarLabel);
                hide(moleculeElements.phosphate);
                hide(moleculeElements.phosphateLabel);
                hide(moleculeElements.conn1);
                hide(moleculeElements.conn2);
                hide(moleculeElements.nucleotideBox);
                hide(moleculeElements.nucleotideText);
                show(dnaGroup);
            }
        }
        
        // 更新按钮状态
        function updateButtons() {
            if (currentStep <= 0) {
                btnPrevBg.classList.remove('btn-enabled');
                btnPrevBg.classList.add('btn-disabled');
                btnPrevText.style.fill = '#666666';
            } else {
                btnPrevBg.classList.remove('btn-disabled');
                btnPrevBg.classList.add('btn-enabled');
                btnPrevText.style.fill = '#ffffff';
            }
            
            if (currentStep >= totalSteps) {
                btnNextBg.classList.remove('btn-enabled');
                btnNextBg.classList.add('btn-disabled');
                btnNextText.style.fill = '#666666';
            } else {
                btnNextBg.classList.remove('btn-disabled');
                btnNextBg.classList.add('btn-enabled');
                btnNextText.style.fill = '#ffffff';
            }
            
            stepIndicator.textContent = `步骤 ${currentStep}/${totalSteps}`;
        }
        
        // 渲染当前状态
        function render() {
            updateBoxStates();
            updateAnnotations();
            updateMoleculeArea();
            updateButtons();
        }
        
        // 按钮事件处理
        btnPrevBg.addEventListener('click', function() {
            if (currentStep > 0) {
                currentStep--;
                render();
            }
        });
        
        btnNextBg.addEventListener('click', function() {
            if (currentStep < totalSteps) {
                currentStep++;
                render();
            }
        });
        
        // 添加按钮悬停效果
        btnPrevBg.addEventListener('mouseenter', function() {
            if (!btnPrevBg.classList.contains('btn-disabled')) {
                this.style.opacity = '0.8';
            }
        });
        btnPrevBg.addEventListener('mouseleave', function() {
            this.style.opacity = '1';
        });
        
        btnNextBg.addEventListener('mouseenter', function() {
            if (!btnNextBg.classList.contains('btn-disabled')) {
                this.style.opacity = '0.8';
            }
        });
        btnNextBg.addEventListener('mouseleave', function() {
            this.style.opacity = '1';
        });
        
        // 初始渲染
        render();
    })();
    </script>
