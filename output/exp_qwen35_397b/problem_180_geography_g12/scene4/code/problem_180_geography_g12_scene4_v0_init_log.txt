    <script>
    (function() {
        // Scene 4 实现：时间计算——经度差与地方时换算
        
        let currentState = 0;
        const totalSteps = 4;
        
        // 每步需要显示的元素
        const steps = {
            0: [],
            1: ['scene4-step-box-1', 'scene4-time-a'],
            2: ['scene4-step-box-2', 'scene4-longitude-c'],
            3: ['scene4-step-box-3', 'scene4-longitude-arrow'],
            4: ['scene4-step-box-4', 'scene4-beijing-longitude']
        };
        
        // 高亮点
        const pointHighlights = {
            1: { id: 'scene4-point-a', r: 6, fill: '#c62828' },
            2: { id: 'scene4-point-c', r: 6, fill: '#1565c0' }
        };
        
        const btnPrev = document.getElementById('scene4-btn-prev');
        const btnNext = document.getElementById('scene4-btn-next');
        const stepIndicator = document.getElementById('scene4-step-indicator');
        
        // 显示元素函数
        function showElement(id) {
            const el = document.getElementById(id);
            if (el) {
                el.style.display = '';
                el.style.opacity = '1';
                el.classList.add('scene4-visible');
            }
        }
        
        // 隐藏元素函数
        function hideElement(id) {
            const el = document.getElementById(id);
            if (el) {
                el.style.display = '';
                el.style.opacity = '0';
                el.classList.remove('scene4-visible');
            }
        }
        
        // 更新步骤显示
        function updateStep() {
            // 隐藏所有步骤元素
            for (let i = 1; i <= totalSteps; i++) {
                steps[i].forEach(id => hideElement(id));
            }
            
            // 显示当前步骤及之前所有步骤的元素
            for (let i = 1; i <= currentState; i++) {
                steps[i].forEach(id => showElement(id));
            }
            
            // 恢复点大小
            ['scene4-point-a', 'scene4-point-c'].forEach(id => {
                const el = document.getElementById(id);
                if (el) {
                    el.setAttribute('r', '4');
                    el.setAttribute('fill', '#000000');
                }
            });
            
            // 高亮当前步骤的点
            if (pointHighlights[currentState]) {
                const h = pointHighlights[currentState];
                const el = document.getElementById(h.id);
                if (el) {
                    el.setAttribute('r', h.r);
                    el.setAttribute('fill', h.fill);
                }
            }
            
            // 更新按钮状态
            btnPrev.disabled = currentState === 0;
            btnNext.disabled = currentState === totalSteps;
            
            // 更新指示器
            stepIndicator.textContent = `步骤 ${currentState}/${totalSteps}`;
        }
        
        // 按钮事件
        btnPrev.addEventListener('click', function() {
            if (currentState > 0) {
                currentState--;
                updateStep();
            }
        });
        
        btnNext.addEventListener('click', function() {
            if (currentState < totalSteps) {
                currentState++;
                updateStep();
            }
        });
        
        // 初始化
        updateStep();
    })();
    </script>
