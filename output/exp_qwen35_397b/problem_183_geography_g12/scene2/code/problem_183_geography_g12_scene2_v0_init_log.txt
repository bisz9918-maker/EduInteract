    <script>
    (function() {
        // Scene 2 实现 - 地球公转示意图
        
        // 状态变量
        let currentState = 0;
        const totalSteps = 4;
        
        // 获取 DOM 元素
        const btnPrev = document.getElementById('scene2-btn-prev');
        const btnNext = document.getElementById('scene2-btn-next');
        const stepIndicator = document.getElementById('scene2-step-indicator');
        
        // 元素引用
        const elements = {
            // 地球 C 高亮
            earthCCircle: document.getElementById('scene2-earth-c-circle'),
            // 昼半球/夜半球
            dayA: document.getElementById('scene2-day-a'),
            nightA: document.getElementById('scene2-night-a'),
            dayC: document.getElementById('scene2-day-c'),
            nightC: document.getElementById('scene2-night-c'),
            // 直射点
            sunpointA: document.getElementById('scene2-sunpoint-a'),
            sunpointB: document.getElementById('scene2-sunpoint-b'),
            sunpointC: document.getElementById('scene2-sunpoint-c'),
            sunpointD: document.getElementById('scene2-sunpoint-d'),
            // 极昼极夜
            polarDayC: document.getElementById('scene2-polar-day-c'),
            polarNightC: document.getElementById('scene2-polar-night-c'),
            // 太阳光线
            sunrayC: document.getElementById('scene2-sunray-c'),
            // 标注文本
            annotation1: document.getElementById('scene2-annotation-1'),
            annotation2: document.getElementById('scene2-annotation-2'),
            annotation3: document.getElementById('scene2-annotation-3'),
            // 纬度高亮
            latHighlight: document.getElementById('scene2-lat-highlight'),
            // 答案框
            answerBox: document.getElementById('scene2-answer-box'),
            answerText: document.getElementById('scene2-answer-text'),
            answerText2: document.getElementById('scene2-answer-text2')
        };
        
        // 工具函数：显示元素
        function show(el) {
            if (el) {
                el.classList.remove('scene2-hidden');
                el.classList.add('scene2-visible');
                el.style.display = 'block';
                el.style.opacity = '1';
            }
        }
        
        // 工具函数：隐藏元素
        function hide(el) {
            if (el) {
                el.classList.remove('scene2-visible');
                el.classList.add('scene2-hidden');
                el.style.display = 'none';
                el.style.opacity = '0';
            }
        }
        
        // 更新步骤状态
        function updateState() {
            // 更新按钮状态
            btnPrev.disabled = (currentState === 0);
            btnNext.disabled = (currentState === totalSteps);
            btnNext.textContent = (currentState === totalSteps) ? '完成' : '下一步 ▶';
            
            // 更新步骤指示
            stepIndicator.textContent = `步骤 ${currentState}/${totalSteps}`;
            
            // 重置所有动态元素
            resetElements();
            
            // 根据当前状态显示相应元素
            switch (currentState) {
                case 0:
                    // 初始状态：只显示基础元素
                    break;
                    
                case 1:
                    // 步骤 1：高亮地球 C，显示标注 1
                    highlightEarthC();
                    show(elements.annotation1);
                    show(elements.latHighlight);
                    show(elements.sunpointC);
                    show(elements.dayC);
                    show(elements.nightC);
                    break;
                    
                case 2:
                    // 步骤 2：显示极昼极夜
                    highlightEarthC();
                    show(elements.annotation1);
                    show(elements.annotation2);
                    show(elements.latHighlight);
                    show(elements.sunpointC);
                    show(elements.dayC);
                    show(elements.nightC);
                    show(elements.polarDayC);
                    show(elements.polarNightC);
                    break;
                    
                case 3:
                    // 步骤 3：显示太阳光线和正午太阳高度标注
                    highlightEarthC();
                    show(elements.annotation1);
                    show(elements.annotation2);
                    show(elements.annotation3);
                    show(elements.latHighlight);
                    show(elements.sunpointC);
                    show(elements.dayC);
                    show(elements.nightC);
                    show(elements.polarDayC);
                    show(elements.polarNightC);
                    show(elements.sunrayC);
                    break;
                    
                case 4:
                    // 步骤 4：显示答案
                    highlightEarthC();
                    show(elements.annotation1);
                    show(elements.annotation2);
                    show(elements.annotation3);
                    show(elements.latHighlight);
                    show(elements.sunpointC);
                    show(elements.dayC);
                    show(elements.nightC);
                    show(elements.polarDayC);
                    show(elements.polarNightC);
                    show(elements.sunrayC);
                    show(elements.answerBox);
                    show(elements.answerText);
                    show(elements.answerText2);
                    break;
            }
        }
        
        // 重置所有动态元素
        function resetElements() {
            // 地球 C 轮廓恢复
            elements.earthCCircle.setAttribute('stroke', '#000000');
            elements.earthCCircle.setAttribute('stroke-width', '1');
            
            // 隐藏所有动态元素
            hide(elements.annotation1);
            hide(elements.annotation2);
            hide(elements.annotation3);
            hide(elements.latHighlight);
            hide(elements.sunpointA);
            hide(elements.sunpointB);
            hide(elements.sunpointC);
            hide(elements.sunpointD);
            hide(elements.dayA);
            hide(elements.nightA);
            hide(elements.dayC);
            hide(elements.nightC);
            hide(elements.polarDayC);
            hide(elements.polarNightC);
            hide(elements.sunrayC);
            hide(elements.answerBox);
            hide(elements.answerText);
            hide(elements.answerText2);
        }
        
        // 高亮地球 C
        function highlightEarthC() {
            elements.earthCCircle.setAttribute('stroke', '#c62828');
            elements.earthCCircle.setAttribute('stroke-width', '3');
        }
        
        // 事件监听
        btnPrev.addEventListener('click', function() {
            if (currentState > 0) {
                currentState--;
                updateState();
            }
        });
        
        btnNext.addEventListener('click', function() {
            if (currentState < totalSteps) {
                currentState++;
                updateState();
            }
        });
        
        // 初始化
        updateState();
        
    })();
    </script>
