    <script>
    (function() {
        // Scene 1 实现
        // 布局坐标表（实现对照依据）：
        // | 元素 | 类型 | 位置/范围 | 大小 | 标签偏移方向 | 备注 |
        // |------|------|-----------|------|-------------|------|
        // | 地球球体 | 圆形 | 圆心(350,280) | r=70 | - | 蓝色渐变 #4fc3f7 |
        // | 自转轴 | 线段 | (290,160)-(410,400) | 长267px | 右上方 | 黑色实线，倾斜23.5° |
        // | 赤道平面 | 平行四边形 | 中心(350,280)，长轴400px | 400×120 | 左上方 | 半透明蓝色 #1565c0 opacity=0.2 |
        // | 黄道面 | 平行四边形 | 中心(350,280)，长轴500px | 500×100 | 右下方 | 半透明橙色 #ff9800 opacity=0.2 |
        // | 黄赤交角弧线 | 圆弧 | 圆心(350,280)，半径90px | 弧长对应23.5° | - | 红色加粗 #c62828 |
        // | 自转轴标签 | text | (420,170) | ~60×16 | 右侧 | 在自转轴上端右侧 |
        // | 赤道平面标签 | text | (180,240) | ~70×16 | 左侧 | 在赤道平面左边缘外侧 |
        // | 黄道面标签 | text | (520,345) | ~70×16 | 右侧 | 在黄道面右边缘外侧（向下调整避免重叠） |
        // | 黄赤交角数值 | text | (450,245) | ~50×16 | 右上方 | 在弧线外侧（向上调整避免重叠） |
        // | 标题 | text | (400,40) | ~200×20 | - | 居中顶部 |
        // | 控件栏 | 按钮组 | y=450~490 | 800×40 | - | 底部绝对定位 |

        let currentState = 0;
        const totalSteps = 4;

        // 每步需要显示的元素
        const stepElements = {
            0: ['scene1-earth', 'scene1-axis', 'scene1-title'],
            1: ['scene1-earth', 'scene1-axis', 'scene1-title', 'scene1-equator-plane', 'scene1-label-equator'],
            2: ['scene1-earth', 'scene1-axis', 'scene1-title', 'scene1-equator-plane', 'scene1-label-equator', 'scene1-ecliptic-plane', 'scene1-label-ecliptic'],
            3: ['scene1-earth', 'scene1-axis', 'scene1-title', 'scene1-equator-plane', 'scene1-label-equator', 'scene1-ecliptic-plane', 'scene1-label-ecliptic', 'scene1-angle-arc', 'scene1-label-angle', 'scene1-label-axis']
        };

        // 工具函数：显示元素
        function showElement(id) {
            const el = document.getElementById(id);
            if (el) {
                if (el.classList.contains('scene1-fade-in')) {
                    el.classList.add('scene1-visible');
                } else {
                    el.style.opacity = '1';
                }
            }
        }

        // 工具函数：隐藏元素
        function hideElement(id) {
            const el = document.getElementById(id);
            if (el) {
                if (el.classList.contains('scene1-fade-in')) {
                    el.classList.remove('scene1-visible');
                } else {
                    el.style.opacity = '0';
                }
            }
        }

        // 更新状态
        function updateState() {
            // 更新步骤指示器
            const indicator = document.getElementById('scene1-step-indicator');
            if (indicator) {
                indicator.textContent = '步骤 ' + (currentState + 1) + '/' + totalSteps;
            }

            // 更新按钮状态
            const prevBtn = document.getElementById('scene1-btn-prev');
            const nextBtn = document.getElementById('scene1-btn-next');
            if (prevBtn) {
                prevBtn.disabled = (currentState === 0);
            }
            if (nextBtn) {
                nextBtn.disabled = (currentState === totalSteps - 1);
            }

            // 根据当前步骤显示/隐藏元素
            const elementsToShow = stepElements[currentState];
            
            // 所有可能元素
            const allElements = [
                'scene1-earth', 'scene1-axis', 'scene1-title',
                'scene1-equator-plane', 'scene1-label-equator',
                'scene1-ecliptic-plane', 'scene1-label-ecliptic',
                'scene1-angle-arc', 'scene1-label-angle', 'scene1-label-axis'
            ];

            // 隐藏所有，然后显示需要的
            allElements.forEach(id => {
                if (elementsToShow.includes(id)) {
                    showElement(id);
                } else {
                    hideElement(id);
                }
            });

            // 步骤 1：自转轴高亮
            const axis = document.getElementById('scene1-axis');
            if (axis) {
                if (currentState >= 1) {
                    axis.setAttribute('stroke-width', '5');
                } else {
                    axis.setAttribute('stroke-width', '3');
                }
            }

            // 步骤 2：赤道平面边框高亮
            const equatorPlane = document.getElementById('scene1-equator-plane');
            if (equatorPlane) {
                if (currentState >= 2) {
                    equatorPlane.setAttribute('stroke', '#1565c0');
                    equatorPlane.setAttribute('stroke-width', '2');
                } else {
                    equatorPlane.removeAttribute('stroke');
                }
            }
        }

        // 按钮事件绑定
        const prevBtn = document.getElementById('scene1-btn-prev');
        const nextBtn = document.getElementById('scene1-btn-next');

        if (prevBtn) {
            prevBtn.addEventListener('click', function() {
                if (currentState > 0) {
                    currentState--;
                    updateState();
                }
            });
        }

        if (nextBtn) {
            nextBtn.addEventListener('click', function() {
                if (currentState < totalSteps - 1) {
                    currentState++;
                    updateState();
                }
            });
        }

        // 初始化
        updateState();
    })();
    </script>
