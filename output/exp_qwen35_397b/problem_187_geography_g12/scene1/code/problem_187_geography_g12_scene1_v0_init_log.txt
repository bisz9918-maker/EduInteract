    <script>
    (function() {
        // Scene 1: 太阳能热水器倾角与正午太阳高度角关系
        
        // 布局坐标表（实现对照依据）：
        // | 元素 | 类型 | 位置/范围 | 大小 | 标签偏移方向 | 备注 |
        // |------|------|-----------|------|-------------|------|
        // | 地面线 | 线段 | (50,350)-(450,350) | 400×4 | - | 水平加粗黑线 |
        // | 热水器面板 | 矩形 | (120,280)-(320,340) | 200×60，旋转 -25° | - | 蓝色填充 20% |
        // | 热水器支架 | 线段 | (140,320)-(140,350) | 4×30 | - | 支撑杆 |
        // | 太阳光线 | 箭头组 | (450,80)-(200,230) | 3 条平行虚线 | - | 红色虚线箭头 |
        // | 地平线参考 | 线段 | (500,350)-(750,350) | 250×2 | - | 右侧参考线 |
        // | 角度α弧线 | 圆弧 | (140,345) 半径 40 | 40px，-25°到 0° | - | 绿色弧线 |
        // | 角度 H 弧线 | 圆弧 | (525,315) 半径 35 | 35px | - | 绿色弧线 (右侧地平线上方) |
        // | 直角标记 | L 形 | (135,345) | 15×15 | - | 灰色直角符号 |
        // | α标签 | text | (200,365) | ~20×16 | 右侧 | 热水器倾角 |
        // | H 标签 | text | (505,305) | ~20×16 | 左上方 | 正午太阳高度角 |
        // | 90°标签 | text | (145,375) | ~30×16 | 下方 | 直角标注 |
        // | 公式标题 | text | (540,150) | - | - | 角度关系 |
        // | 公式文本 | text | (540,190) | - | - | α+H=90° |
        // | 公式推导 | text | (540,235) | - | - | α=90°-H |
        // | 步骤按钮组 | 按钮 | (250,460)-(550,490) | 300×30 | - | 底部控件栏 |
        
        let currentStep = 0;
        const totalSteps = 4;
        
        // 获取 DOM 元素
        const btnPrev = document.getElementById('scene1-btn-prev');
        const btnNext = document.getElementById('scene1-btn-next');
        const stepIndicator = document.getElementById('scene1-step-indicator');
        
        // 步骤相关元素
        const solarRays = document.getElementById('scene1-solar-rays');
        const horizonLine = document.getElementById('scene1-horizon-line');
        const angleH = document.getElementById('scene1-angle-h');
        const rightAngle = document.getElementById('scene1-right-angle');
        const formulaContent = document.getElementById('scene1-formula-content');
        
        // 显示元素工具函数
        function showElement(el) {
            el.style.display = 'block';
            el.style.opacity = '1';
            el.classList.add('fade-in');
        }
        
        // 隐藏元素工具函数
        function hideElement(el) {
            el.style.display = 'none';
            el.style.opacity = '0';
            el.classList.remove('fade-in');
        }
        
        // 更新步骤显示
        function updateStep() {
            // 隐藏所有步骤相关元素
            hideElement(solarRays);
            hideElement(horizonLine);
            hideElement(angleH);
            hideElement(rightAngle);
            hideElement(formulaContent);
            
            // 根据当前步骤显示相应元素
            if (currentStep >= 1) {
                showElement(solarRays);
                showElement(horizonLine);
                showElement(angleH);
            }
            if (currentStep >= 2) {
                showElement(rightAngle);
            }
            if (currentStep >= 3) {
                showElement(formulaContent);
            }
            
            // 更新步骤指示器
            stepIndicator.textContent = '步骤 ' + (currentStep + 1) + '/' + totalSteps;
            
            // 更新按钮状态
            btnPrev.disabled = (currentStep === 0);
            btnNext.disabled = (currentStep === totalSteps - 1);
        }
        
        // 上一步按钮点击事件
        btnPrev.addEventListener('click', function() {
            if (currentStep > 0) {
                currentStep--;
                updateStep();
            }
        });
        
        // 下一步按钮点击事件
        btnNext.addEventListener('click', function() {
            if (currentStep < totalSteps - 1) {
                currentStep++;
                updateStep();
            }
        });
        
        // 初始化
        updateStep();
        
    })();
    </script>
