    <script>
    (function() {
        // Scene 1: 四棱锥 P-ABCD 命题①分析
        // 布局坐标表：
        // P(400,80), A(200,350), B(350,400), C(600,380), D(450,320), E(750,460)

        let currentStep = 0;
        const totalSteps = 4;

        // 获取 DOM 元素
        const planePab = document.getElementById('scene1-plane-pab');
        const edgeCd = document.getElementById('scene1-edge-cd');
        const extAb = document.getElementById('scene1-ext-ab');
        const extDc = document.getElementById('scene1-ext-dc');
        const pointE = document.getElementById('scene1-point-e');
        const labelE = document.getElementById('scene1-label-e');
        const conclusion = document.getElementById('scene1-conclusion');
        const stepIndicator = document.getElementById('scene1-step-indicator');
        const btnPrevGroup = document.getElementById('scene1-btn-prev-group');
        const btnNextGroup = document.getElementById('scene1-btn-next-group');

        // 显示/隐藏工具函数
        function show(el) {
            el.style.display = '';
            el.style.opacity = '1';
        }

        function hide(el) {
            el.style.display = 'none';
            el.style.opacity = '0';
        }

        // 更新状态
        function updateState() {
            // 重置所有元素
            hide(planePab);
            edgeCd.classList.remove('highlighted');
            hide(extAb);
            hide(extDc);
            hide(pointE);
            hide(labelE);
            hide(conclusion);

            // 根据当前步骤显示相应元素
            if (currentStep >= 1) {
                show(planePab);
            }
            if (currentStep >= 2) {
                edgeCd.classList.add('highlighted');
            }
            if (currentStep >= 3) {
                show(extAb);
                show(extDc);
                show(pointE);
                show(labelE);
            }
            if (currentStep >= 4) {
                show(conclusion);
            }

            // 更新步骤指示器
            stepIndicator.textContent = `步骤 ${currentStep + 1}/${totalSteps}`;

            // 更新按钮状态
            const prevRect = btnPrevGroup.querySelector('rect');
            const prevText = btnPrevGroup.querySelector('text');
            const nextRect = btnNextGroup.querySelector('rect');
            const nextText = btnNextGroup.querySelector('text');

            if (currentStep === 0) {
                prevRect.setAttribute('fill', '#e0e0e0');
                prevText.setAttribute('fill', '#999');
            } else {
                prevRect.setAttribute('fill', '#f5f5f5');
                prevText.setAttribute('fill', '#333');
            }

            if (currentStep >= totalSteps) {
                nextRect.setAttribute('fill', '#e0e0e0');
                nextText.setAttribute('fill', '#999');
            } else {
                nextRect.setAttribute('fill', '#f5f5f5');
                nextText.setAttribute('fill', '#333');
            }
        }

        // 按钮点击事件
        btnPrevGroup.addEventListener('click', function() {
            if (currentStep > 0) {
                currentStep--;
                updateState();
            }
        });

        btnNextGroup.addEventListener('click', function() {
            if (currentStep < totalSteps) {
                currentStep++;
                updateState();
            }
        });

        // 初始状态
        updateState();
    })();
    </script>
