    <script>
    (function() {
        // Scene 2 实现
        let currentStep = 0;
        const totalSteps = 5;
        
        const stepElements = {
            1: ['scene2-line-ab'],
            2: ['scene2-triangle-abc', 'scene2-right-angle', 'scene2-label-right-angle'],
            3: ['scene2-label-ab', 'scene2-label-bc', 'scene2-label-ac'],
            4: ['scene2-label-equation'],
            5: ['scene2-label-result']
        };
        
        const btnPrev = document.getElementById('scene2-btn-prev');
        const btnNext = document.getElementById('scene2-btn-next');
        const stepIndicator = document.getElementById('scene2-step-indicator');
        
        function showElement(id) {
            const el = document.getElementById(id);
            if (el) {
                el.classList.remove('hidden');
                requestAnimationFrame(() => {
                    el.classList.add('visible');
                });
            }
        }
        
        function hideElement(id) {
            const el = document.getElementById(id);
            if (el) {
                el.classList.remove('visible');
                el.classList.add('hidden');
            }
        }
        
        function updateStep() {
            stepIndicator.textContent = `步骤 ${currentStep}/${totalSteps}`;
            btnPrev.disabled = (currentStep === 0);
            btnNext.disabled = (currentStep === totalSteps);
            
            for (let step = 1; step <= totalSteps; step++) {
                const elements = stepElements[step];
                if (elements) {
                    elements.forEach(id => {
                        if (step <= currentStep) {
                            showElement(id);
                        } else {
                            hideElement(id);
                        }
                    });
                }
            }
        }
        
        btnPrev.addEventListener('click', () => {
            if (currentStep > 0) {
                currentStep--;
                updateStep();
            }
        });
        
        btnNext.addEventListener('click', () => {
            if (currentStep < totalSteps) {
                currentStep++;
                updateStep();
            }
        });
        
        updateStep();
    })();
    </script>
