    <script>
    (function() {
        // Scene 2: 奇偶性对比验证
        const state = { current: 0 }; // 0: t>0, 1: t=0, 2: t<0
        
        // 元素引用
        const els = {
            lineL1: document.getElementById('scene2-line-l1'),
            lineL2: document.getElementById('scene2-line-l2'),
            lineZero: document.getElementById('scene2-line-zero'),
            labelL1: document.getElementById('scene2-label-l1'),
            labelL2: document.getElementById('scene2-label-l2'),
            pointM1: document.getElementById('scene2-point-m1'),
            pointN1: document.getElementById('scene2-point-n1'),
            pointM2: document.getElementById('scene2-point-m2'),
            pointN2: document.getElementById('scene2-point-n2'),
            labelM1: document.getElementById('scene2-label-m1'),
            labelN1: document.getElementById('scene2-label-n1'),
            labelM2: document.getElementById('scene2-label-m2'),
            labelN2: document.getElementById('scene2-label-n2'),
            triOM1N1: document.getElementById('scene2-tri-om1n1'),
            triOM2N2: document.getElementById('scene2-tri-om2n2'),
            symM1M2: document.getElementById('scene2-sym-m1m2'),
            symN1N2: document.getElementById('scene2-sym-n1n2'),
            areaST: document.getElementById('scene2-area-st'),
            areaSNT: document.getElementById('scene2-area-snt'),
            areaNote: document.getElementById('scene2-area-note'),
            conclusionText: document.getElementById('scene2-conclusion-text'),
            btnBg0: document.getElementById('scene2-btn-bg-0'),
            btnBg1: document.getElementById('scene2-btn-bg-1'),
            btnBg2: document.getElementById('scene2-btn-bg-2'),
            btnGroups: document.querySelectorAll('.scene2-btn-group')
        };

        // 计算三角形面积值
        const areaValue = 2100;

        // 显示/隐藏元素
        function show(el) {
            if (el) { el.style.display = ''; el.style.opacity = '1'; }
        }

        function hide(el) {
            if (el) { el.style.display = 'none'; el.style.opacity = '0'; }
        }

        // 更新按钮状态
        function updateButtons(activeIdx) {
            const btns = [els.btnBg0, els.btnBg1, els.btnBg2];
            btns.forEach((bg, i) => {
                if (i === activeIdx) {
                    bg.setAttribute('fill', i === 0 ? '#1976d2' : (i === 1 ? '#666' : '#2e7d32'));
                    bg.setAttribute('stroke', 'none');
                    bg.nextElementSibling.setAttribute('fill', '#fff');
                } else {
                    bg.setAttribute('fill', '#f5f5f5');
                    bg.setAttribute('stroke', '#333');
                    bg.setAttribute('stroke-width', '2');
                    bg.nextElementSibling.setAttribute('fill', '#333');
                }
            });
        }

        // 状态 0: t > 0
        function showState0() {
            show(els.lineL1); show(els.labelL1);
            show(els.pointM1); show(els.pointN1);
            show(els.labelM1); show(els.labelN1);
            show(els.triOM1N1);
            show(els.areaST); show(els.areaNote);
            
            hide(els.lineL2); hide(els.labelL2);
            hide(els.pointM2); hide(els.pointN2);
            hide(els.labelM2); hide(els.labelN2);
            hide(els.triOM2N2); hide(els.areaSNT);
            hide(els.symM1M2); hide(els.symN1N2);
            hide(els.lineZero);
            
            els.areaST.textContent = 'S(t) = ' + areaValue;
            els.conclusionText.textContent = '直线 L: y = kx + t (t > 0)';
            updateButtons(0);
        }

        // 状态 1: t = 0
        function showState1() {
            hide(els.lineL1); hide(els.lineL2);
            hide(els.labelL1); hide(els.labelL2);
            hide(els.pointM1); hide(els.pointN1);
            hide(els.pointM2); hide(els.pointN2);
            hide(els.labelM1); hide(els.labelN1);
            hide(els.labelM2); hide(els.labelN2);
            hide(els.triOM1N1); hide(els.triOM2N2);
            hide(els.areaST); hide(els.areaSNT); hide(els.areaNote);
            hide(els.symM1M2); hide(els.symN1N2);
            
            show(els.lineZero);
            
            els.conclusionText.textContent = '直线过原点，S(0) = 0';
            updateButtons(1);
        }

        // 状态 2: t < 0 (对比状态)
        function showState2() {
            show(els.lineL1); show(els.lineL2);
            show(els.labelL1); show(els.labelL2);
            show(els.pointM1); show(els.pointN1);
            show(els.pointM2); show(els.pointN2);
            show(els.labelM1); show(els.labelN1);
            show(els.labelM2); show(els.labelN2);
            show(els.triOM1N1); show(els.triOM2N2);
            show(els.areaST); show(els.areaSNT); show(els.areaNote);
            show(els.symM1M2); show(els.symN1N2);
            hide(els.lineZero);
            
            els.areaST.textContent = 'S(t) = ' + areaValue;
            els.areaSNT.textContent = 'S(-t) = ' + areaValue;
            els.conclusionText.textContent = 'S(t) = S(-t)  ∴  f(t) 是偶函数';
            updateButtons(2);
        }

        // 切换状态
        function setState(s) {
            state.current = s;
            if (s === 0) showState0();
            else if (s === 1) showState1();
            else showState2();
        }

        // 按钮点击事件
        els.btnGroups.forEach((grp, idx) => {
            grp.addEventListener('click', function() {
                if (idx !== state.current) setState(idx);
            });
        });

        // 初始化：状态 0
        setState(0);
    })();
    </script>
