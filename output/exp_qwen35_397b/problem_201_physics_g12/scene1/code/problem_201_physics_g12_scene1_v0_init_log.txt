    <script>
    (function() {
        // Scene 1 实现：带电粒子在复合场中的运动轨迹

        // 状态变量
        let currentStep = 0;
        const totalSteps = 5;

        // 每步应显示的元素配置
        // 步骤 0: 电场线 + A 点 + B 点 + 标签 A/B/E
        // 步骤 1: + 轨迹
        // 步骤 2: + v₀ 矢量
        // 步骤 3: + v_B 矢量
        // 步骤 4: + 坐标系
        const stepElements = {
            0: ['scene1-field-lines', 'scene1-label-e', 'scene1-point-a', 'scene1-point-b', 'scene1-label-a', 'scene1-label-b'],
            1: ['scene1-trajectory'],
            2: ['scene1-velocity-v0'],
            3: ['scene1-velocity-vb'],
            4: ['scene1-coordinates']
        };

        // 获取元素引用
        const btnPrev = document.getElementById('scene1-btn-prev');
        const btnNext = document.getElementById('scene1-btn-next');
        const stepIndicator = document.getElementById('scene1-step-indicator');

        // 显示元素函数
        function showElement(elementId) {
            const el = document.getElementById(elementId);
            if (el) {
                if (el.classList.contains('scene1-fade-in')) {
                    el.classList.add('scene1-visible');
                } else {
                    el.style.display = 'block';
                    el.style.opacity = '1';
                }
            }
        }

        // 隐藏元素函数
        function hideElement(elementId) {
            const el = document.getElementById(elementId);
            if (el) {
                if (el.classList.contains('scene1-fade-in')) {
                    el.classList.remove('scene1-visible');
                } else {
                    el.style.display = 'none';
                    el.style.opacity = '0';
                }
            }
        }

        // 更新显示状态
        function updateDisplay() {
            // 根据当前步骤显示/隐藏元素
            for (let step = 0; step < totalSteps; step++) {
                const elements = stepElements[step];
                if (step <= currentStep) {
                    elements.forEach(id => showElement(id));
                } else {
                    elements.forEach(id => hideElement(id));
                }
            }

            // 更新步骤指示器
            stepIndicator.textContent = `步骤 ${currentStep + 1}/${totalSteps}`;

            // 更新按钮状态
            btnPrev.disabled = currentStep === 0;
            btnNext.disabled = currentStep === totalSteps - 1;
        }

        // 上一步按钮事件
        btnPrev.addEventListener('click', function() {
            if (currentStep > 0) {
                currentStep--;
                updateDisplay();
            }
        });

        // 下一步按钮事件
        btnNext.addEventListener('click', function() {
            if (currentStep < totalSteps - 1) {
                currentStep++;
                updateDisplay();
            }
        });

        // 初始化显示
        updateDisplay();
    })();
    </script>
