    <script>
    (function() {
        // Scene 3 实现
        let currentStep = 0;
        const totalSteps = 5;

        const btnPrev = document.getElementById('scene3-btn-prev');
        const btnNext = document.getElementById('scene3-btn-next');
        const stepIndicator = document.getElementById('scene3-step-indicator');

        // 公式框元素
        const formulaBoxes = [
            document.getElementById('scene3-formula-box-1'),
            document.getElementById('scene3-formula-box-2'),
            document.getElementById('scene3-formula-box-3'),
            document.getElementById('scene3-formula-box-4'),
            document.getElementById('scene3-formula-box-5')
        ];

        // 轨迹元素
        const trajectory = document.getElementById('scene3-trajectory');

        // 更新步骤显示
        function updateStep() {
            // 更新步骤指示器
            stepIndicator.textContent = '步骤 ' + currentStep + '/' + totalSteps;

            // 更新按钮状态
            btnPrev.disabled = (currentStep === 0);
            btnNext.disabled = (currentStep === totalSteps);

            // 更新公式框可见性
            formulaBoxes.forEach(function(box, index) {
                if (index < currentStep) {
                    box.classList.add('scene3-visible');
                } else {
                    box.classList.remove('scene3-visible');
                }
            });

            // 步骤 5 时高亮轨迹
            if (currentStep === totalSteps) {
                trajectory.setAttribute('stroke-width', '3');
                trajectory.setAttribute('stroke', '#0d47a1');
            } else {
                trajectory.setAttribute('stroke-width', '2');
                trajectory.setAttribute('stroke', '#1565c0');
            }
        }

        // 按钮事件绑定
        btnPrev.addEventListener('click', function() {
            if (currentStep > 0) {
                currentStep--;
                updateStep();
            }
        });

        btnNext.addEventListener('click', function() {
            if (currentStep < totalSteps) {
                currentStep++;
                updateStep();
            }
        });

        // 初始化
        updateStep();
    })();
    </script>
