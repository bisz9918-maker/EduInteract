    <script>
    (function() {
        // Scene 1: 带电粒子在电场中的类平抛运动轨迹分析
        
        const TOTAL_STEPS = 5;
        let currentStep = 0;
        
        // 元素分组映射：每个步骤需要显示的元素
        const stepElements = {
            0: [], // 初始状态：只显示坐标轴、边界、A 点、C 点（这些元素默认显示）
            1: ['scene1-e-field-group', 'scene1-velocity-a-group'],
            2: ['scene1-trajectory-part1-group', 'scene1-midpoint-group'],
            3: ['scene1-trajectory-part2-group'],
            4: ['scene1-aprime-group', 'scene1-b-field-group']
        };
        
        // 显示元素
        function showElement(id) {
            const el = document.getElementById(id);
            if (el) {
                el.classList.add('visible');
                el.style.opacity = '1';
                el.style.display = '';
            }
        }
        
        // 隐藏元素
        function hideElement(id) {
            const el = document.getElementById(id);
            if (el) {
                el.classList.remove('visible');
                el.style.opacity = '0';
                el.style.display = 'none';
            }
        }
        
        // 更新显示状态
        function updateDisplay() {
            // 累积显示：步骤 0 到 currentStep 的所有元素
            for (let step = 0; step < TOTAL_STEPS; step++) {
                const elements = stepElements[step] || [];
                elements.forEach(id => {
                    if (step <= currentStep) {
                        showElement(id);
                    } else {
                        hideElement(id);
                    }
                });
            }
            
            // 更新步骤指示器
            const indicator = document.getElementById('scene1-step-indicator');
            if (indicator) {
                indicator.textContent = `步骤 ${currentStep + 1}/${TOTAL_STEPS}`;
            }
            
            // 更新按钮状态
            const btnPrev = document.getElementById('scene1-btn-prev');
            const btnNext = document.getElementById('scene1-btn-next');
            if (btnPrev) {
                btnPrev.disabled = (currentStep === 0);
            }
            if (btnNext) {
                btnNext.disabled = (currentStep === TOTAL_STEPS - 1);
            }
        }
        
        // 上一步
        function prevStep() {
            if (currentStep > 0) {
                currentStep--;
                updateDisplay();
            }
        }
        
        // 下一步
        function nextStep() {
            if (currentStep < TOTAL_STEPS - 1) {
                currentStep++;
                updateDisplay();
            }
        }
        
        // 初始化
        function init() {
            // 绑定事件
            const btnPrev = document.getElementById('scene1-btn-prev');
            const btnNext = document.getElementById('scene1-btn-next');
            
            if (btnPrev) {
                btnPrev.addEventListener('click', prevStep);
            }
            if (btnNext) {
                btnNext.addEventListener('click', nextStep);
            }
            
            // 初始显示
            updateDisplay();
        }
        
        // DOM 加载完成后初始化
        if (document.readyState === 'loading') {
            document.addEventListener('DOMContentLoaded', init);
        } else {
            init();
        }
    })();
    </script>
