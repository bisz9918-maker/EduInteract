    <script>
    (function() {
        // Scene 2: 带电粒子在磁场中的圆周运动及返回轨迹

        // 布局坐标表（实现对照依据）：
        // | 元素 | 位置/范围 | 备注 |
        // | x 轴 | (220,250)-(750,250) | 水平轴 |
        // | y 轴 | (520,70)-(520,430) | 垂直轴 |
        // | 原点 O | (520,250) | 坐标原点，标签 (555,285) |
        // | x=-2L 线 | (360,70)-(360,430) | 电场左边界 |
        // | 磁场区域 | (680,100)-(770,430) | ×填充，向右上移 |
        // | A'点 | (520,330) | (0,-L) 位置，标签 (475,355) |
        // | 圆心 C | (600,330) | 圆周运动圆心，标签 (590,450) |
        // | 半径 R | 65px | R=mv/qB，标签 (560,475) |
        // | 穿出点 P | (648,266) | 轨迹与 x 轴交点附近，标签 (655,245) |
        // | 返回点 F | (360,420) | 在 x=-2L 上，标签 (250,445) |
        // | v₀标签 | (525,275) | 移到箭头上方更远处 |
        // | 电场线（下）| y 从 300-400 | 缩短给轨迹留空间 |

        let currentState = 0;
        const totalSteps = 5;

        // 每步需要显示的元素 ID 列表
        const steps = {
            0: [], // 初始状态：只显示基础元素
            1: ['scene2-circle-center', 'scene2-label-center', 'scene2-radius-R', 'scene2-label-R'],
            2: ['scene2-arc-trajectory', 'scene2-angle-arc'],
            3: ['scene2-point-P', 'scene2-label-P', 'scene2-velocity-P'],
            4: ['scene2-return-path', 'scene2-point-F', 'scene2-label-F-coord']
        };

        // 显示元素
        function showElement(id) {
            const el = document.getElementById(id);
            if (el) {
                el.classList.remove('hidden');
                el.style.display = '';
                el.style.opacity = '1';
            }
        }

        // 隐藏元素
        function hideElement(id) {
            const el = document.getElementById(id);
            if (el) {
                el.classList.add('hidden');
                el.style.display = 'none';
                el.style.opacity = '0';
            }
        }

        // 更新显示状态
        function updateDisplay() {
            // 隐藏所有步骤元素
            for (let step = 1; step < totalSteps; step++) {
                if (steps[step]) {
                    steps[step].forEach(id => hideElement(id));
                }
            }

            // 显示当前步骤及之前步骤的元素
            for (let step = 1; step <= currentState; step++) {
                if (steps[step]) {
                    steps[step].forEach(id => showElement(id));
                }
            }

            // 更新步骤指示器
            const indicator = document.getElementById('scene2-step-indicator');
            if (indicator) {
                indicator.textContent = `步骤 ${currentState + 1}/${totalSteps}`;
            }

            // 更新按钮状态
            const prevBtn = document.getElementById('scene2-btn-prev');
            const nextBtn = document.getElementById('scene2-btn-next');
            if (prevBtn) {
                prevBtn.disabled = (currentState === 0);
            }
            if (nextBtn) {
                nextBtn.disabled = (currentState === totalSteps - 1);
            }
        }

        // 生成磁场区域×符号，避开轨迹区域和所有标签
        function generateCrosses() {
            const group = document.getElementById('scene2-magnetic-field-crosses');
            if (!group) return;

            const startX = 680;
            const endX = 770;
            const startY = 100;
            const endY = 430;
            const cols = 3;
            const rows = 5;
            const gapX = (endX - startX) / (cols - 1);
            const gapY = (endY - startY) / (rows - 1);

            // 需要避开的区域（轨迹和标签）
            const avoidZones = [
                { x: 555, y: 285, r: 40 },   // 原点 O 标签
                { x: 720, y: 240, r: 35 },   // x 轴标签
                { x: 530, y: 60, r: 30 },    // y 轴标签
                { x: 525, y: 275, r: 40 },   // v₀ 标签
                { x: 475, y: 355, r: 40 },   // A' 标签
                { x: 600, y: 330, r: 80 },   // 圆心及轨迹区域
                { x: 655, y: 245, r: 40 },   // P 点及标签
                { x: 590, y: 450, r: 45 },   // 圆心标签
                { x: 560, y: 475, r: 40 },   // R 标签
                { x: 250, y: 445, r: 80 }    // 返回点坐标标签
            ];

            for (let i = 0; i < rows; i++) {
                for (let j = 0; j < cols; j++) {
                    const x = startX + j * gapX;
                    const y = startY + i * gapY;
                    
                    // 检查是否在避开区域内
                    let shouldSkip = false;
                    for (const zone of avoidZones) {
                        const dist = Math.sqrt((x - zone.x) ** 2 + (y - zone.y) ** 2);
                        if (dist < zone.r) {
                            shouldSkip = true;
                            break;
                        }
                    }
                    
                    if (shouldSkip) {
                        continue;
                    }
                    
                    const cross = document.createElementNS('http://www.w3.org/2000/svg', 'text');
                    cross.setAttribute('x', x);
                    cross.setAttribute('y', y);
                    cross.setAttribute('font-size', '16');
                    cross.setAttribute('fill', '#000');
                    cross.setAttribute('text-anchor', 'middle');
                    cross.setAttribute('dominant-baseline', 'middle');
                    cross.textContent = '×';
                    group.appendChild(cross);
                }
            }
        }

        // 事件处理
        function setupEventListeners() {
            const prevBtn = document.getElementById('scene2-btn-prev');
            const nextBtn = document.getElementById('scene2-btn-next');

            if (prevBtn) {
                prevBtn.addEventListener('click', function() {
                    if (currentState > 0) {
                        currentState--;
                        updateDisplay();
                    }
                });
            }

            if (nextBtn) {
                nextBtn.addEventListener('click', function() {
                    if (currentState < totalSteps - 1) {
                        currentState++;
                        updateDisplay();
                    }
                });
            }
        }

        // 初始化
        function init() {
            generateCrosses();
            setupEventListeners();
            updateDisplay();
        }

        // 页面加载完成后初始化
        if (document.readyState === 'loading') {
            document.addEventListener('DOMContentLoaded', init);
        } else {
            init();
        }
    })();
    </script>
