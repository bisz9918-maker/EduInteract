    <script>
    (function() {
        // Scene 3 实现 - 带电粒子偏转相似三角形
        
        // 状态管理
        let currentStep = 0;
        const totalSteps = 5;
        
        // 元素引用
        const btnPrev = document.getElementById('scene3-btn-prev');
        const btnNext = document.getElementById('scene3-btn-next');
        const stepIndicator = document.getElementById('scene3-step-indicator-html');
        
        // 需要控制的元素
        const elements = {
            tri1: document.getElementById('scene3-tri1'),
            tri2: document.getElementById('scene3-tri2'),
            arrowY1: document.getElementById('scene3-arrow-y1'),
            arrowY2: document.getElementById('scene3-arrow-y2'),
            arrowDy: document.getElementById('scene3-arrow-dy'),
            labelY1: document.getElementById('scene3-label-y1'),
            labelY2: document.getElementById('scene3-label-y2'),
            labelDy: document.getElementById('scene3-label-dy'),
            formula: document.getElementById('scene3-formula'),
            formulaText: document.getElementById('scene3-formula-text'),
            trajectoryIn: document.getElementById('scene3-trajectory-in'),
            trajectoryOut: document.getElementById('scene3-trajectory-out')
        };
        
        // 工具函数：显示元素
        function show(el) {
            if (el) {
                el.classList.remove('hidden');
                el.style.opacity = '1';
                el.style.display = '';
            }
        }
        
        // 工具函数：隐藏元素
        function hide(el) {
            if (el) {
                el.classList.add('hidden');
                el.style.opacity = '0';
                el.style.display = 'none';
            }
        }
        
        // 更新步骤显示
        function updateStep() {
            stepIndicator.textContent = '步骤 ' + (currentStep + 1) + '/' + totalSteps;
            btnPrev.disabled = (currentStep === 0);
            btnNext.disabled = (currentStep === totalSteps - 1);
        }
        
        // 设置步骤状态
        function setStep(step) {
            currentStep = step;
            
            // 重置所有特殊元素
            hide(elements.tri1);
            hide(elements.tri2);
            hide(elements.arrowY1);
            hide(elements.arrowY2);
            hide(elements.arrowDy);
            hide(elements.labelY1);
            hide(elements.labelY2);
            hide(elements.labelDy);
            hide(elements.formula);
            
            // 重置轨迹样式
            elements.trajectoryIn.setAttribute('stroke-width', '3');
            elements.trajectoryOut.setAttribute('stroke-width', '3');
            
            switch(step) {
                case 0:
                    // 初始状态：只显示基础装置和轨迹
                    break;
                    
                case 1:
                    // 高亮电场内三角形
                    show(elements.tri1);
                    show(elements.arrowY1);
                    show(elements.labelY1);
                    elements.trajectoryIn.setAttribute('stroke-width', '4');
                    break;
                    
                case 2:
                    // 高亮电场外三角形
                    show(elements.tri1);
                    show(elements.arrowY1);
                    show(elements.labelY1);
                    show(elements.tri2);
                    show(elements.arrowDy);
                    show(elements.labelDy);
                    elements.trajectoryOut.setAttribute('stroke-width', '4');
                    break;
                    
                case 3:
                    // 显示相似关系
                    show(elements.tri1);
                    show(elements.tri2);
                    show(elements.arrowY1);
                    show(elements.arrowY2);
                    show(elements.arrowDy);
                    show(elements.labelY1);
                    show(elements.labelY2);
                    show(elements.labelDy);
                    show(elements.formula);
                    elements.formulaText.textContent = 'y₂/y₁ = (L+D)/L';
                    elements.trajectoryIn.setAttribute('stroke-width', '3');
                    elements.trajectoryOut.setAttribute('stroke-width', '3');
                    break;
                    
                case 4:
                    // 显示计算结果
                    show(elements.tri1);
                    show(elements.tri2);
                    show(elements.arrowY1);
                    show(elements.arrowY2);
                    show(elements.arrowDy);
                    show(elements.labelY1);
                    show(elements.labelY2);
                    show(elements.labelDy);
                    show(elements.formula);
                    elements.formulaText.textContent = 'y₂ = 7.5cm';
                    break;
            }
            
            updateStep();
        }
        
        // 事件绑定
        btnPrev.addEventListener('click', function() {
            if (currentStep > 0) {
                setStep(currentStep - 1);
            }
        });
        
        btnNext.addEventListener('click', function() {
            if (currentStep < totalSteps - 1) {
                setStep(currentStep + 1);
            }
        });
        
        // 初始化
        setStep(0);
        
    })();
    </script>
