    <script>
    (function() {
        // Scene 1 实现 - 旋转圆盘系统结构示意

        // 状态机
        let currentStep = 0;
        const totalSteps = 5;

        // 每步显示的元素 ID 列表
        const steps = [
            // 步骤 0: 圆盘 + 转轴 + O 标签
            ['scene1-disc', 'scene1-axis', 'scene1-axis-top', 'scene1-label-o'],
            // 步骤 1: + 物体 B + 标签 B + r_B 标注
            ['scene1-disc', 'scene1-axis', 'scene1-axis-top', 'scene1-label-o', 'scene1-obj-b', 'scene1-label-b', 'scene1-rb-line', 'scene1-rb-label'],
            // 步骤 2: + 物体 A + 标签 A + r_A 标注
            ['scene1-disc', 'scene1-axis', 'scene1-axis-top', 'scene1-label-o', 'scene1-obj-b', 'scene1-label-b', 'scene1-rb-line', 'scene1-rb-label', 'scene1-obj-a', 'scene1-label-a', 'scene1-ra-line', 'scene1-ra-label'],
            // 步骤 3: + OB 绳 + AB 绳
            ['scene1-disc', 'scene1-axis', 'scene1-axis-top', 'scene1-label-o', 'scene1-obj-b', 'scene1-label-b', 'scene1-rb-line', 'scene1-rb-label', 'scene1-obj-a', 'scene1-label-a', 'scene1-ra-line', 'scene1-ra-label', 'scene1-rope-ob', 'scene1-rope-ab'],
            // 步骤 4: + 转动箭头 + ω符号
            ['scene1-disc', 'scene1-axis', 'scene1-axis-top', 'scene1-label-o', 'scene1-obj-b', 'scene1-label-b', 'scene1-rb-line', 'scene1-rb-label', 'scene1-obj-a', 'scene1-label-a', 'scene1-ra-line', 'scene1-ra-label', 'scene1-rope-ob', 'scene1-rope-ab', 'scene1-rotation-arrow', 'scene1-omega']
        ];

        // 所有可控制的元素 ID
        const allElementIds = [
            'scene1-disc', 'scene1-axis', 'scene1-axis-top', 'scene1-label-o',
            'scene1-obj-b', 'scene1-label-b', 'scene1-rb-line', 'scene1-rb-label',
            'scene1-obj-a', 'scene1-label-a', 'scene1-ra-line', 'scene1-ra-label',
            'scene1-rope-ob', 'scene1-rope-ab',
            'scene1-rotation-arrow', 'scene1-omega'
        ];

        // 工具函数：显示元素
        function showElement(id) {
            const el = document.getElementById(id);
            if (el) {
                el.classList.remove('scene1-hidden');
                el.classList.add('scene1-visible');
            }
        }

        // 工具函数：隐藏元素
        function hideElement(id) {
            const el = document.getElementById(id);
            if (el) {
                el.classList.remove('scene1-visible');
                el.classList.add('scene1-hidden');
            }
        }

        // 渲染当前步骤
        function renderStep() {
            const currentElements = steps[currentStep];
            
            // 隐藏所有元素
            allElementIds.forEach(id => hideElement(id));
            
            // 显示当前步骤的元素
            currentElements.forEach(id => showElement(id));
            
            // 更新步骤指示器
            const indicator = document.getElementById('scene1-step-indicator');
            indicator.textContent = '步骤 ' + (currentStep + 1) + '/' + totalSteps;
            
            // 更新按钮状态
            const prevBtnBg = document.getElementById('scene1-btn-prev-bg');
            const nextBtnBg = document.getElementById('scene1-btn-next-bg');
            
            if (currentStep === 0) {
                prevBtnBg.setAttribute('fill', '#cccccc');
            } else {
                prevBtnBg.setAttribute('fill', '#1976d2');
            }
            
            if (currentStep === totalSteps - 1) {
                nextBtnBg.setAttribute('fill', '#cccccc');
            } else {
                nextBtnBg.setAttribute('fill', '#1976d2');
            }
        }

        // 上一步
        function prevStep() {
            if (currentStep > 0) {
                currentStep--;
                renderStep();
            }
        }

        // 下一步
        function nextStep() {
            if (currentStep < totalSteps - 1) {
                currentStep++;
                renderStep();
            }
        }

        // 按钮点击区域检测
        const controlsGroup = document.getElementById('scene1-controls');
        
        controlsGroup.addEventListener('click', function(e) {
            const rect = controlsGroup.getBoundingClientRect();
            const x = e.clientX - rect.left;
            const y = e.clientY - rect.top;
            
            // 上一步按钮区域 (280, 450) 100x35
            if (x >= 280 && x <= 380 && y >= 450 && y <= 485) {
                prevStep();
            }
            // 下一步按钮区域 (420, 450) 100x35
            else if (x >= 420 && x <= 520 && y >= 450 && y <= 485) {
                nextStep();
            }
        });

        // 初始化
        renderStep();
    })();
    </script>
