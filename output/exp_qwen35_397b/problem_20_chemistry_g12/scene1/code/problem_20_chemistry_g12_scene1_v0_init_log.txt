    <script>
    (function() {
        // Scene 1 实现：原电池工作原理

        // 状态管理
        let currentStep = 0;
        const totalSteps = 5;

        // 每步可见元素配置
        const stepConfig = {
            0: { // 初始状态：基础装置
                fade: [],
                slide: []
            },
            1: { // 步骤 1：电极极性
                fade: ['scene1-label-positive', 'scene1-label-negative'],
                slide: []
            },
            2: { // 步骤 2：电子流向
                fade: ['scene1-label-positive', 'scene1-label-negative', 'scene1-electron-group'],
                slide: []
            },
            3: { // 步骤 3：离子迁移
                fade: ['scene1-label-positive', 'scene1-label-negative', 'scene1-electron-group', 'scene1-ions-group'],
                slide: []
            },
            4: { // 步骤 4：电极反应式
                fade: ['scene1-label-positive', 'scene1-label-negative', 'scene1-electron-group', 'scene1-ions-group'],
                slide: ['scene1-reactions-group']
            }
        };

        // DOM 元素
        const btnPrev = document.getElementById('scene1-btn-prev');
        const btnNext = document.getElementById('scene1-btn-next');
        const stepIndicator = document.getElementById('scene1-step-indicator');

        // 更新视图函数
        function updateView() {
            // 隐藏所有可切换元素
            const allFadeElements = ['scene1-label-positive', 'scene1-label-negative', 'scene1-electron-group', 'scene1-ions-group'];
            const allSlideElements = ['scene1-reactions-group'];

            allFadeElements.forEach(id => {
                const el = document.getElementById(id);
                if (el) el.classList.remove('visible');
            });
            allSlideElements.forEach(id => {
                const el = document.getElementById(id);
                if (el) el.classList.remove('visible');
            });

            // 显示当前步骤及之前步骤的元素
            const config = stepConfig[currentStep];
            config.fade.forEach(id => {
                const el = document.getElementById(id);
                if (el) el.classList.add('visible');
            });
            config.slide.forEach(id => {
                const el = document.getElementById(id);
                if (el) el.classList.add('visible');
            });

            // 更新步骤指示器
            stepIndicator.textContent = `步骤 ${currentStep + 1}/${totalSteps}`;

            // 更新按钮状态
            btnPrev.disabled = currentStep === 0;
            btnNext.disabled = currentStep === totalSteps - 1;

            // 触发电子流向动画（步骤 2）
            if (currentStep >= 2) {
                const arrow = document.getElementById('scene1-electron-arrow');
                if (arrow) {
                    arrow.classList.remove('visible');
                    requestAnimationFrame(() => {
                        arrow.classList.add('visible');
                    });
                }
            }
        }

        // 按钮事件
        btnPrev.addEventListener('click', () => {
            if (currentStep > 0) {
                currentStep--;
                updateView();
            }
        });

        btnNext.addEventListener('click', () => {
            if (currentStep < totalSteps - 1) {
                currentStep++;
                updateView();
            }
        });

        // 初始化视图
        updateView();
    })();
    </script>
