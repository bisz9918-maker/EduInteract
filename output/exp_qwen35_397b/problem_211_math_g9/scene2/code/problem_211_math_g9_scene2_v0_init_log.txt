        const btnCText = document.getElementById('scene2-btn-c-text');

        function resetState() {
            // 重置按钮样式
            [btnA, btnB, btnC, btnD].forEach(btn => {
                btn.classList.remove('scene2-btn-active');
                btn.classList.add('scene2-btn-default');
            });
            [btnAText, btnBText, btnCText, btnDText].forEach(text => {
                text.classList.remove('scene2-btn-text-active');
                text.classList.add('scene2-btn-text-default');
            });

            aeLine.setAttribute('stroke', '#888888');
            aeLine.setAttribute('stroke-width', '2');

            cfLine.setAttribute('stroke-width', '2');

            angle2.setAttribute('stroke-width', '1.5');

                mark.classList.add('scene2-hidden');
                mark.classList.remove('scene2-visible');
            });

            judgeTitle.textContent = '判定类型';
            judgeType.textContent = '请选择选项';

            judgeText.textContent = '';
            judgeCondition.textContent = '';
        }

            btnA.classList.add('scene2-btn-active');
            btnAText.classList.remove('scene2-btn-text-default');
            btnAText.classList.add('scene2-btn-text-active');

            aeLine.setAttribute('stroke-width', '3');
            cfLine.setAttribute('stroke', '#1565c0');
            cfLine.setAttribute('stroke-width', '3');
            
            // 显示等长标记
            markAE.classList.remove('scene2-hidden');
            markAE.classList.add('scene2-visible');
            markCF.classList.remove('scene2-hidden');
            markCF.classList.add('scene2-visible');
            
            // 更新判定框
            judgeTitle.textContent = '判定类型';
            judgeType.textContent = 'SSA';
            judgeType.setAttribute('fill', '#c62828');

            judgeResult.setAttribute('fill', '#c62828');
            judgeText.textContent = '不能判定全等';
            judgeCondition.textContent = 'AE=CF, AB=CD, ∠ABE=∠CDF';
        }

        function showStateB() {
            resetState();
            btnB.classList.remove('scene2-btn-default');
            btnB.classList.add('scene2-btn-active');
            btnBText.classList.remove('scene2-btn-text-default');
            btnBText.classList.add('scene2-btn-text-active');

            markBE.classList.add('scene2-visible');

            judgeType.textContent = 'SAS';
            judgeType.setAttribute('fill', '#2e7d32');
            judgeResult.textContent = '✓';
            judgeResult.setAttribute('fill', '#2e7d32');
            judgeText.textContent = '可判定全等';
            judgeCondition.textContent = 'BE=FD, AB=CD, ∠ABE=∠CDF';
        }

        function showStateC() {
            resetState();
            btnC.classList.remove('scene2-btn-default');
            btnC.classList.add('scene2-btn-active');
            btnCText.classList.remove('scene2-btn-text-default');
            btnCText.classList.add('scene2-btn-text-active');

            markDE.classList.add('scene2-visible');

            judgeType.textContent = 'SAS';
            judgeType.setAttribute('fill', '#2e7d32');
            judgeResult.textContent = '✓';
            judgeResult.setAttribute('fill', '#2e7d32');

            judgeType.textContent = 'ASA';
            judgeType.setAttribute('fill', '#2e7d32');
            judgeResult.textContent = '✓';
            judgeResult.setAttribute('fill', '#2e7d32');
