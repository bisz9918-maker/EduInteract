    <script>
    (function() {
        // Scene 2 实现：三角形分类对比展示
        
        // 状态变量
        let currentState = 0; // 0=初始, 1=小三角, 2=大三角, 3=总计
        
        // 获取 DOM 元素
        const groupSmall = document.getElementById('scene2-group-small');
        const pathLarge = document.getElementById('scene2-path-large');
        const labelSmallCount = document.getElementById('scene2-label-small-count');
        const labelLargeCount = document.getElementById('scene2-label-large-count');
        const labelTotal = document.getElementById('scene2-label-total');
        const btnState1 = document.getElementById('scene2-btn-state1');
        const btnState2 = document.getElementById('scene2-btn-state2');
        const btnState3 = document.getElementById('scene2-btn-state3');
        
        // 工具函数：显示元素
        function show(el) {
            el.classList.remove('scene2-hidden');
            el.style.display = '';
            el.style.opacity = '1';
        }
        
        // 工具函数：隐藏元素
        function hide(el) {
            el.classList.add('scene2-hidden');
            el.style.display = 'none';
            el.style.opacity = '0';
        }
        
        // 工具函数：更新按钮状态
        function updateButtons(activeBtn) {
            btnState1.classList.remove('active');
            btnState2.classList.remove('active');
            btnState3.classList.remove('active');
            if (activeBtn) activeBtn.classList.add('active');
        }
        
        // 状态切换函数
        function setState(state) {
            currentState = state;
            
            // 重置所有元素
            hide(groupSmall);
            hide(pathLarge);
            hide(labelSmallCount);
            hide(labelLargeCount);
            hide(labelTotal);
            updateButtons(null);
            
            if (state === 1) {
                // 状态 1：显示小三角形
                show(groupSmall);
                show(labelSmallCount);
                updateButtons(btnState1);
            } else if (state === 2) {
                // 状态 2：显示大三角形
                show(pathLarge);
                show(labelLargeCount);
                updateButtons(btnState2);
            } else if (state === 3) {
                // 状态 3：显示全部
                show(groupSmall);
                show(pathLarge);
                show(labelTotal);
                updateButtons(btnState3);
            }
        }
        
        // 绑定按钮事件
        btnState1.addEventListener('click', function() { setState(1); });
        btnState2.addEventListener('click', function() { setState(2); });
        btnState3.addEventListener('click', function() { setState(3); });
        
        // 初始状态：显示框架和标题，小三角形半透明可见
        show(groupSmall);
        
    })();
    </script>
