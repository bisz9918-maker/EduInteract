    <script>
    (function() {
        // Scene 1 实现
        let currentStep = 0;
        const totalSteps = 5;
        
        // 每步的公式文本
        const formulas = [
            "1 节长度 = 2.5cm",
            "2 节长度 = 2.5 + 2.5 - 0.8 = 4.2cm",
            "3 节长度 = 2.5×3 - 0.8×2 = 5.9cm",
            "n 节长度 = 2.5n - 0.8(n-1) = 1.7n + 0.8",
            "100 节长度 = 1.7×100 + 0.8 = 170.8cm ✓"
        ];
        
        // 获取元素
        const step0Group = document.getElementById('scene1-step0-group');
        const step1Group = document.getElementById('scene1-step1-group');
        const step2Group = document.getElementById('scene1-step2-group');
        const step3Group = document.getElementById('scene1-step3-group');
        const formulaText = document.getElementById('scene1-formula-text');
        const stepIndicator = document.getElementById('scene1-step-indicator');
        const btnPrev = document.getElementById('scene1-btn-prev-bg');
        const btnNext = document.getElementById('scene1-btn-next-bg');
        
        // 显示/隐藏函数
        function showElement(el) {
            if (el) {
                el.classList.remove('hidden');
                el.style.opacity = '1';
                el.style.display = '';
            }
        }
        
        function hideElement(el) {
            if (el) {
                el.classList.add('hidden');
                el.style.opacity = '0';
            }
        }
        
        // 更新步骤显示
        function updateStep() {
            // 隐藏所有步骤组
            hideElement(step0Group);
            hideElement(step1Group);
            hideElement(step2Group);
            hideElement(step3Group);
            
            // 根据当前步骤显示对应组
            if (currentStep >= 1) showElement(step0Group);
            if (currentStep >= 2) showElement(step1Group);
            if (currentStep >= 3) showElement(step2Group);
            if (currentStep >= 4) showElement(step3Group);
            
            // 更新公式
            formulaText.textContent = formulas[currentStep];
            
            // 更新步骤指示器
            stepIndicator.textContent = '步骤 ' + (currentStep + 1) + '/' + totalSteps;
            
            // 更新按钮状态
            if (currentStep === 0) {
                btnPrev.style.opacity = '0.5';
                btnPrev.style.cursor = 'not-allowed';
            } else {
                btnPrev.style.opacity = '1';
                btnPrev.style.cursor = 'pointer';
            }
            
            if (currentStep === totalSteps - 1) {
                btnNext.style.opacity = '0.5';
                btnNext.style.cursor = 'not-allowed';
            } else {
                btnNext.style.opacity = '1';
                btnNext.style.cursor = 'pointer';
            }
        }
        
        // 按钮点击事件
        btnPrev.addEventListener('click', function() {
            if (currentStep > 0) {
                currentStep--;
                updateStep();
            }
        });
        
        btnNext.addEventListener('click', function() {
            if (currentStep < totalSteps - 1) {
                currentStep++;
                updateStep();
            }
        });
        
        // 初始化
        updateStep();
    })();
    </script>
