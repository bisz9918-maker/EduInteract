    <script>
    (function() {
        // Scene 2: ∠ABC=45°的几何验证
        // 布局坐标表：
        // x 轴：(200,250)-(600,250)
        // y 轴：(400,100)-(400,400)
        // 原点 O：(400,250)
        // 点 B：(600,250)
        // 点 C：(400,450)
        // 线段 OB：蓝色 #1565c0
        // 线段 OC：蓝色 #1565c0
        // 线段 BC：绿色 #2e7d32
        // 等长标记：OB 中点 (500,250) 垂直短线，OC 中点 (400,350) 水平短线
        // 直角符号：(400,250)-(425,275) 红色 #c62828
        // 角度弧线：以 B(600,250) 为圆心，r=40

        let currentStep = 0;
        const totalSteps = 5;

        const btnPrev = document.getElementById('scene2-btn-prev');
        const btnNext = document.getElementById('scene2-btn-next');
        const stepIndicator = document.getElementById('scene2-step-indicator');

        const step1Group = document.getElementById('scene2-step1');
        const step2Group = document.getElementById('scene2-step2');
        const step3Group = document.getElementById('scene2-step3');
        const step4Group = document.getElementById('scene2-step4');

        function updateDisplay() {
            // 更新步骤指示器
            stepIndicator.textContent = `步骤 ${currentStep + 1}/${totalSteps}`;

            // 更新按钮状态
            btnPrev.disabled = (currentStep === 0);
            btnNext.disabled = (currentStep === totalSteps - 1);

            // 控制各步骤元素的显隐
            if (currentStep >= 1) {
                step1Group.classList.remove('scene2-hidden');
                step1Group.classList.add('scene2-visible');
            } else {
                step1Group.classList.remove('scene2-visible');
                step1Group.classList.add('scene2-hidden');
            }

            if (currentStep >= 2) {
                step2Group.classList.remove('scene2-hidden');
                step2Group.classList.add('scene2-visible');
            } else {
                step2Group.classList.remove('scene2-visible');
                step2Group.classList.add('scene2-hidden');
            }

            if (currentStep >= 3) {
                step3Group.classList.remove('scene2-hidden');
                step3Group.classList.add('scene2-visible');
            } else {
                step3Group.classList.remove('scene2-visible');
                step3Group.classList.add('scene2-hidden');
            }

            if (currentStep >= 4) {
                step4Group.classList.remove('scene2-hidden');
                step4Group.classList.add('scene2-visible');
            } else {
                step4Group.classList.remove('scene2-visible');
                step4Group.classList.add('scene2-hidden');
            }
        }

        function nextStep() {
            if (currentStep < totalSteps - 1) {
                currentStep++;
                updateDisplay();
            }
        }

        function prevStep() {
            if (currentStep > 0) {
                currentStep--;
                updateDisplay();
            }
        }

        // 绑定事件
        btnPrev.addEventListener('click', prevStep);
        btnNext.addEventListener('click', nextStep);

        // 初始化显示
        updateDisplay();
    })();
    </script>
