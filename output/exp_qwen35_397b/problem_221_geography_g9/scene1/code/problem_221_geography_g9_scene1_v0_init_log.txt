    <script>
    (function() {
        // Scene 1 实现：昼夜分布与晨昏线综合分析

        // 状态管理
        let currentStep = 0;

        // 每步配置

                       'scene1-description-group'],
                hide: ['scene1-label-morning', 'scene1-label-6oclock', 'scene1-label-western',
                       'scene1-hemisphere-boundary', 'scene1-solar-meridian', 'scene1-label-solar'],
                description: ['原图展示某日部分', '地区昼夜分布。阴', '影表示黑夜，白色', '表示白昼。观察晨', '昏线走向及甲、乙', '两地位置。']
            },
            {
                // 步骤 1：高亮晨昏线，标注晨线
                show: ['scene1-frame', 'scene1-lat-235n', 'scene1-lat-0', 'scene1-lat-235s',

                       'scene1-point-a', 'scene1-point-b', 'scene1-point-a-group',
                       'scene1-label-235n', 'scene1-label-0', 'scene1-label-235s',
                       'scene1-label-375', 'scene1-label-225', 'scene1-label-a', 'scene1-label-b',
                       'scene1-description-group', 'scene1-label-morning'],
                hide: ['scene1-label-6oclock', 'scene1-label-western',
                       'scene1-hemisphere-boundary', 'scene1-solar-meridian', 'scene1-label-solar'],
                description: ['晨昏线是昼夜分界', '线。顺着地球自转', '方向（自西向东），', '由夜入昼为晨线，', '由昼入夜为昏线。', '图中晨昏线东侧为', '白昼，故为晨线。']
            },
            {
                // 步骤 2：叠加东西半球分界线
                show: ['scene1-frame', 'scene1-lat-235n', 'scene1-lat-0', 'scene1-lat-235s',
                       'scene1-lon-375w', 'scene1-lon-225w', 'scene1-terminator', 'scene1-shadow',
                       'scene1-point-a', 'scene1-point-b', 'scene1-point-a-group',
                       'scene1-label-235n', 'scene1-label-0', 'scene1-label-235s',
                       'scene1-label-375', 'scene1-label-225', 'scene1-label-a', 'scene1-label-b',
                       'scene1-description-group', 'scene1-label-morning', 'scene1-hemisphere-boundary', 'scene1-label-western'],
                hide: ['scene1-label-6oclock', 'scene1-solar-meridian', 'scene1-label-solar'],
                description: ['东西半球分界线为', '20°W 和 160°E。20°W', '以东、160°E 以西为', '东半球。图中阴影', '区（黑夜）主要位于', '20°W 以西，故主要', '位于西半球。']
            },
            {
                // 步骤 3：高亮甲地，说明晨线与赤道交点地方时为 6 时
                show: ['scene1-frame', 'scene1-lat-235n', 'scene1-lat-0', 'scene1-lat-235s',
                       'scene1-lon-375w', 'scene1-lon-225w', 'scene1-terminator', 'scene1-shadow',
                       'scene1-point-a', 'scene1-point-b', 'scene1-point-a-group',
                       'scene1-label-235n', 'scene1-label-0', 'scene1-label-235s',

            },

                // 步骤 4：展示直射点经线
                show: ['scene1-frame', 'scene1-lat-235n', 'scene1-lat-0', 'scene1-lat-235s',

                       'scene1-label-6oclock', 'scene1-solar-meridian', 'scene1-label-solar'],
                hide: [],
                description: ['北半球（23.5°N 以', '北）昼弧长于夜', '弧，说明北半球昼', '长夜短，太阳直射', '北半球。昼半球中', '央经线即为太阳直', '射点所在经线。']
            }
        ];

        // 获取元素辅助函数
        function getElement(id) {
            return document.getElementById(id);
        }

        // 显示元素
        function showElement(id) {
            const el = getElement(id);
            if (el) {
                el.style.opacity = '1';
                el.style.display = 'block';
            }
        }

        // 隐藏元素
        function hideElement(id) {
            const el = getElement(id);
            if (el) {
                el.style.opacity = '0';
            }
        }

        // 更新说明文本
        function updateDescription(lines) {
            const descEl = getElement('scene1-description');
            if (descEl) {
                descEl.innerHTML = '';
                lines.forEach((line, index) => {
                    const tspan = document.createElementNS('http://www.w3.org/2000/svg', 'tspan');
                    tspan.setAttribute('x', '430');
                    tspan.setAttribute('dy', index === 0 ? '0' : '22');
                    tspan.textContent = line;
                    descEl.appendChild(tspan);
                });
            }
        }

        // 更新步骤
        function updateStep() {
            const step = steps[currentStep];

            // 显示指定元素
            step.show.forEach(id => showElement(id));

            // 隐藏指定元素
            step.hide.forEach(id => hideElement(id));

            // 更新说明文本
            updateDescription(step.description);

            // 更新步骤指示器
            const indicator = getElement('scene1-step-indicator');
            if (indicator) {
                indicator.textContent = `步骤 ${currentStep + 1}/${totalSteps}`;
            }

            // 更新按钮状态
            const prevBtn = getElement('scene1-btn-prev');
            const nextBtn = getElement('scene1-btn-next');
            if (prevBtn) {
                prevBtn.disabled = currentStep === 0;
            }
            if (nextBtn) {
                nextBtn.disabled = currentStep === totalSteps - 1;
            }

            // 特殊动画处理
            if (currentStep === 1) {
                // 步骤 1：晨昏线闪烁
                const terminator = getElement('scene1-terminator');
                if (terminator) {
                    terminator.classList.add('scene1-pulse');
                    setTimeout(() => terminator.classList.remove('scene1-pulse'), 1600);
                }
                const morningLabel = getElement('scene1-label-morning');
                if (morningLabel) {
                    morningLabel.classList.add('scene1-fade-in');
                }
            }
            if (currentStep === 2) {
                // 步骤 2：东西半球分界线绘制动画
                const boundary = getElement('scene1-hemisphere-boundary');
                if (boundary) {
                    boundary.classList.add('scene1-draw-line');
                }
                const westernLabel = getElement('scene1-label-western');
                if (westernLabel) {
                    westernLabel.classList.add('scene1-fade-in');
                }
            }
            if (currentStep === 3) {
                // 步骤 3：甲点放大动画
                const pointAGroup = getElement('scene1-point-a-group');
                if (pointAGroup) {
                    pointAGroup.classList.add('scene1-scale');
                    setTimeout(() => pointAGroup.classList.remove('scene1-scale'), 600);
                }
                const oclockLabel = getElement('scene1-label-6oclock');
                if (oclockLabel) {
                    oclockLabel.classList.add('scene1-fade-in');
                }
            }
            if (currentStep === 4) {
                // 步骤 4：直射点经线绘制动画
                const solarMeridian = getElement('scene1-solar-meridian');
                if (solarMeridian) {
                    solarMeridian.classList.add('scene1-draw-line');
                }
                const solarLabel = getElement('scene1-label-solar');
                if (solarLabel) {
                    solarLabel.classList.add('scene1-fade-in');
                }
            }
        }

        // 初始化
        function init() {
            const prevBtn = getElement('scene1-btn-prev');
            const nextBtn = getElement('scene1-btn-next');

            if (prevBtn) {
                prevBtn.addEventListener('click', () => {
                    if (currentStep > 0) {
                        currentStep--;
                        updateStep();
                    }
                });
            }

            if (nextBtn) {
                nextBtn.addEventListener('click', () => {
                    if (currentStep < totalSteps - 1) {
                        currentStep++;
                        updateStep();
                    }
                });
            }

            // 初始状态
            updateStep();
        }

        // 页面加载完成后初始化
        if (document.readyState === 'loading') {
            document.addEventListener('DOMContentLoaded', init);
        } else {
            init();
        }
    })();
