    <script>
    (function() {
        // Scene 1 实现：晨昏线识别与昼夜半球划分

        let currentState = 0;
        const totalSteps = 4;

        const btnPrev = document.getElementById('scene1-btn-prev');
        const btnNext = document.getElementById('scene1-btn-next');
        const statusEl = document.getElementById('scene1-status');

        // 元素引用
        const 晨昏线 EDF = document.getElementById('scene1-晨昏线 -EDF');
        const 晨线 ED = document.getElementById('scene1-晨线 -ED');
        const 昏线 DF = document.getElementById('scene1-昏线 -DF');
        const 夜半球 Fill = document.getElementById('scene1-夜半球-fill');
        const 晨线标注 = document.getElementById('scene1-晨线标注');
        const 昏线标注 = document.getElementById('scene1-昏线标注');
        const 夜半球标签 = document.getElementById('scene1-标签 - 夜半球');
        const 昼半球标签 = document.getElementById('scene1-标签 - 昼半球');

        function updateScene() {
            // 更新状态显示
            statusEl.textContent = `步骤 ${currentState}/4`;

            // 更新按钮状态
            btnPrev.disabled = (currentState === 0);
            btnNext.disabled = (currentState === totalSteps);

            // 步骤 0: 初始状态 - 晨昏线细线
            // 步骤 1+: 晨昏线加粗
            if (currentState >= 1) {
                晨昏线 EDF.setAttribute('stroke', '#1565c0');
                晨昏线 EDF.setAttribute('stroke-width', '3');
            } else {
                晨昏线 EDF.setAttribute('stroke', '#cccccc');
                晨昏线 EDF.setAttribute('stroke-width', '1');
            }

            // 步骤 2+: 显示晨线/昏线标注
            if (currentState >= 2) {
                晨线 ED.style.display = 'block';
                昏线 DF.style.display = 'block';
                晨线标注.style.display = 'block';
                昏线标注.style.display = 'block';
            } else {
                晨线 ED.style.display = 'none';
                昏线 DF.style.display = 'none';
                晨线标注.style.display = 'none';
                昏线标注.style.display = 'none';
            }

            // 步骤 3+: 显示夜半球填充
            if (currentState >= 3) {
                夜半球 Fill.style.display = 'block';
            } else {
                夜半球 Fill.style.display = 'none';
            }

            // 步骤 4: 显示昼/夜半球文字标签
            if (currentState >= 4) {
                夜半球标签.style.opacity = '1';
                昼半球标签.style.opacity = '1';
            } else {
                夜半球标签.style.opacity = '0';
                昼半球标签.style.opacity = '0';
            }
        }

        btnPrev.addEventListener('click', function() {
            if (currentState > 0) {
                currentState--;
                updateScene();
            }
        });

        btnNext.addEventListener('click', function() {
            if (currentState < totalSteps) {
                currentState++;
                updateScene();
            }
        });

        // 初始化
        updateScene();
    })();
    </script>
