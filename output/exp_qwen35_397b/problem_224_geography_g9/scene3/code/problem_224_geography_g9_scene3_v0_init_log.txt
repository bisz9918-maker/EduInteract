    <script>
    (function() {
        // Scene 3 实现：各点地理参数比较分析
        
        let currentState = 0;
        
        // 获取 DOM 元素
        const lineBC = document.getElementById('scene3-line-bc');
        const lineBD = document.getElementById('scene3-line-bd');
        const lineCD = document.getElementById('scene3-line-cd');
        const sunheightD = document.getElementById('scene3-sunheight-d');
        const sunheightC = document.getElementById('scene3-sunheight-c');
        const conclusion = document.getElementById('scene3-conclusion');
        const btn1 = document.getElementById('scene3-btn-state1');
        const btn2 = document.getElementById('scene3-btn-state2');
        const btn3 = document.getElementById('scene3-btn-state3');
        
        // 隐藏所有连线和标注
        function hideAll() {
            lineBC.setAttribute('opacity', '0');
            lineBD.setAttribute('opacity', '0');
            lineCD.setAttribute('opacity', '0');
            sunheightD.setAttribute('opacity', '0');
            sunheightC.setAttribute('opacity', '0');
        }
        
        // 重置按钮状态
        function resetButtons() {
            btn1.classList.remove('active');
            btn2.classList.remove('active');
            btn3.classList.remove('active');
        }
        
        // 更新状态
        function updateState(state) {
            currentState = state;
            hideAll();
            resetButtons();
            
            if (state === 1) {
                // 正午太阳高度角
                lineBC.setAttribute('opacity', '1');
                sunheightD.setAttribute('opacity', '1');
                sunheightC.setAttribute('opacity', '1');
                conclusion.textContent = '正午太阳高度角相等：B 和 C（同纬度对称点）';
                btn1.classList.add('active');
            } else if (state === 2) {
                // 地方时
                lineBD.setAttribute('opacity', '1');
                conclusion.textContent = '地方时相等：B 和 D（同经度 45°W）';
                btn2.classList.add('active');
            } else if (state === 3) {
                // 线速度
                lineCD.setAttribute('opacity', '1');
                conclusion.textContent = '线速度相等：C 和 D（同纬度 66.5°N）';
                btn3.classList.add('active');
            } else {
                // 初始状态
                conclusion.textContent = '请选择比较类型';
            }
        }
        
        // 绑定按钮事件
        btn1.addEventListener('click', function() { updateState(1); });
        btn2.addEventListener('click', function() { updateState(2); });
        btn3.addEventListener('click', function() { updateState(3); });
        
    })();
    </script>
