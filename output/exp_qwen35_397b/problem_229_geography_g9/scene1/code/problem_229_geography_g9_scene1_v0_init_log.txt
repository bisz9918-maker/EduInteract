    <script>
    (function() {
        // Scene 1: 经纬网与点的经纬度判读
        // 布局坐标表参考：
        // | 元素 | 类型 | 位置/范围 | 大小 | 标签偏移方向 | 备注 |
        // |------|------|-----------|------|-------------|------|
        // | 经纬网外框 | 矩形 | (100,80)-(700,380) | 600×300 | - | 主图区 |
        // | 经线 1(140°E) | 弧线 | (140,80)-(140,380) | - | - | 最左侧经线 |
        // | 经线 2(160°E) | 弧线 | (240,80)-(240,380) | - | - | 左二经线 |
        // | 经线 3(180°) | 直线 | (340,80)-(340,380) | - | - | 中央经线 |
        // | 经线 4(160°W) | 弧线 | (440,80)-(440,380) | - | - | 右二经线 |
        // | 经线 5(140°W) | 弧线 | (540,80)-(540,380) | - | - | 最右侧经线 |
        // | 纬线 1(60°N) | 弧线 | (140,130)-(540,130) | - | - | 最上纬线 |
        // | 纬线 2(50°N) | 弧线 | (140,230)-(540,230) | - | - | 中间纬线 |
        // | 纬线 3(40°N) | 弧线 | (140,330)-(540,330) | - | - | 最下纬线 |
        // | 点 D 标记 | 实心圆 | (140,130) | r=5 | - | 140°E 与 60°N 交点 |
        // | 点 E 标记 | 实心圆 | (240,330) | r=5 | - | 160°E 与 40°N 交点 |
        // | 点 F 标记 | 实心圆 | (440,230) | r=5 | - | 160°W 与 50°N 交点 |
        
        let currentState = 0;
        const totalSteps = 5;
        
        // 获取 DOM 元素
        const meridians = [
            document.getElementById('scene1-meridian-1'),
            document.getElementById('scene1-meridian-2'),
            document.getElementById('scene1-meridian-3'),
            document.getElementById('scene1-meridian-4'),
            document.getElementById('scene1-meridian-5')
        ];
        
        const parallels = [
            document.getElementById('scene1-parallel-1'),
            document.getElementById('scene1-parallel-2'),
            document.getElementById('scene1-parallel-3')
        ];
        
        const coordBoxes = [
            document.getElementById('scene1-coord-box-d'),
            document.getElementById('scene1-coord-text-d'),
            document.getElementById('scene1-coord-box-e'),
            document.getElementById('scene1-coord-text-e'),
            document.getElementById('scene1-coord-box-f'),
            document.getElementById('scene1-coord-text-f')
        ];
        
        const meridianArrows = document.querySelectorAll('.scene1-direction-arrow[id^="scene1-arrow-meridian-"]');
        const meridianArrowLabel = document.getElementById('scene1-arrow-meridian-label');
        const parallelArrow = document.getElementById('scene1-arrow-parallel-1');
        const parallelArrowLabel = document.getElementById('scene1-arrow-parallel-label');
        
        const stepIndicator = document.getElementById('scene1-step-indicator');
        const btnPrev = document.getElementById('scene1-btn-prev');
        const btnPrevBg = document.getElementById('scene1-btn-prev-bg');
        const btnPrevText = document.getElementById('scene1-btn-prev-text');
        const btnNext = document.getElementById('scene1-btn-next');
        const btnNextBg = document.getElementById('scene1-btn-next-bg');
        const btnNextText = document.getElementById('scene1-btn-next-text');
        
        const points = [
            document.getElementById('scene1-point-d'),
            document.getElementById('scene1-point-e'),
            document.getElementById('scene1-point-f')
        ];
        
        // 更新显示状态
        function updateDisplay() {
            // 重置所有高亮
            meridians.forEach(m => m.classList.remove('highlight'));
            parallels.forEach(p => p.classList.remove('highlight'));
            
            // 隐藏所有箭头
            meridianArrows.forEach(a => a.classList.remove('visible'));
            meridianArrowLabel.style.opacity = '0';
            parallelArrow.classList.remove('visible');
            parallelArrowLabel.style.opacity = '0';
            
            // 隐藏所有坐标框
            coordBoxes.forEach(el => {
                el.classList.remove('visible');
                el.style.opacity = '0';
            });
            
            // 移除点闪烁效果
            points.forEach(p => p.classList.remove('scene1-blink'));
            
            // 根据当前状态设置显示
            switch (currentState) {
                case 0:
                    // 初始状态：只显示基础经纬网和点
                    break;
                    
                case 1:
                    // 步骤 1：经线高亮 + 方向箭头
                    meridians.forEach(m => m.classList.add('highlight'));
                    meridianArrows.forEach(a => a.classList.add('visible'));
                    meridianArrowLabel.style.opacity = '1';
                    break;
                    
                case 2:
                    // 步骤 2：纬线高亮 + 方向箭头
                    parallels.forEach(p => p.classList.add('highlight'));
                    parallelArrow.classList.add('visible');
                    parallelArrowLabel.style.opacity = '1';
                    break;
                    
                case 3:
                    // 步骤 3：显示 D 坐标框
                    document.getElementById('scene1-coord-box-d').classList.add('visible');
                    document.getElementById('scene1-coord-text-d').style.opacity = '1';
                    break;
                    
                case 4:
                    // 步骤 4：显示 D 和 E 坐标框
                    document.getElementById('scene1-coord-box-d').classList.add('visible');
                    document.getElementById('scene1-coord-text-d').style.opacity = '1';
                    document.getElementById('scene1-coord-box-e').classList.add('visible');
                    document.getElementById('scene1-coord-text-e').style.opacity = '1';
                    break;
                    
                case 5:
                    // 步骤 5：显示所有坐标框 + 点闪烁
                    document.getElementById('scene1-coord-box-d').classList.add('visible');
                    document.getElementById('scene1-coord-text-d').style.opacity = '1';
                    document.getElementById('scene1-coord-box-e').classList.add('visible');
                    document.getElementById('scene1-coord-text-e').style.opacity = '1';
                    document.getElementById('scene1-coord-box-f').classList.add('visible');
                    document.getElementById('scene1-coord-text-f').style.opacity = '1';
                    points.forEach(p => p.classList.add('scene1-blink'));
                    break;
            }
            
            // 更新状态指示器
            stepIndicator.textContent = `步骤 ${currentState}/${totalSteps}`;
            
            // 更新按钮状态
            if (currentState === 0) {
                btnPrev.classList.add('scene1-btn-disabled');
                btnPrevBg.setAttribute('fill', '#bdbdbd');
                btnPrevText.setAttribute('fill', '#9e9e9e');
            } else {
                btnPrev.classList.remove('scene1-btn-disabled');
                btnPrevBg.setAttribute('fill', '#e0e0e0');
                btnPrevText.setAttribute('fill', '#333333');
            }
            
            if (currentState === totalSteps) {
                btnNext.classList.add('scene1-btn-disabled');
                btnNextBg.setAttribute('fill', '#bdbdbd');
                btnNextText.setAttribute('fill', '#9e9e9e');
            } else {
                btnNext.classList.remove('scene1-btn-disabled');
                btnNextBg.setAttribute('fill', '#1565c0');
                btnNextText.setAttribute('fill', '#ffffff');
            }
        }
        
        // 按钮点击事件
        btnPrev.addEventListener('click', function() {
            if (currentState > 0 && !btnPrev.classList.contains('scene1-btn-disabled')) {
                currentState--;
                updateDisplay();
            }
        });
        
        btnNext.addEventListener('click', function() {
            if (currentState < totalSteps && !btnNext.classList.contains('scene1-btn-disabled')) {
                currentState++;
                updateDisplay();
            }
        });
        
        // 添加按钮悬停效果
        btnPrev.addEventListener('mouseenter', function() {
            if (!btnPrev.classList.contains('scene1-btn-disabled')) {
                btnPrevBg.setAttribute('fill', '#cfcfcf');
            }
        });
        btnPrev.addEventListener('mouseleave', function() {
            if (!btnPrev.classList.contains('scene1-btn-disabled')) {
                btnPrevBg.setAttribute('fill', '#e0e0e0');
            }
        });
        
        btnNext.addEventListener('mouseenter', function() {
            if (!btnNext.classList.contains('scene1-btn-disabled')) {
                btnNextBg.setAttribute('fill', '#1976d2');
            }
        });
        btnNext.addEventListener('mouseleave', function() {
            if (!btnNext.classList.contains('scene1-btn-disabled')) {
                btnNextBg.setAttribute('fill', '#1565c0');
            }
        });
        
        // 添加点击反馈
        btnPrev.addEventListener('mousedown', function() {
            if (!btnPrev.classList.contains('scene1-btn-disabled')) {
                btnPrevBg.setAttribute('fill', '#9e9e9e');
            }
        });
        btnPrev.addEventListener('mouseup', function() {
            if (!btnPrev.classList.contains('scene1-btn-disabled')) {
                btnPrevBg.setAttribute('fill', '#e0e0e0');
            }
        });
        
        btnNext.addEventListener('mousedown', function() {
            if (!btnNext.classList.contains('scene1-btn-disabled')) {
                btnNextBg.setAttribute('fill', '#0d47a1');
            }
        });
        btnNext.addEventListener('mouseup', function() {
            if (!btnNext.classList.contains('scene1-btn-disabled')) {
                btnNextBg.setAttribute('fill', '#1565c0');
            }
        });
        
        // 初始化显示
        updateDisplay();
        
    })();
    </script>
