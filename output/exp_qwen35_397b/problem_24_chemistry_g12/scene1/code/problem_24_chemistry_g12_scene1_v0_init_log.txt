    <script>
    (function() {
        // Scene 1 实现 - 三容器初始状态对比

        // 状态定义：每步显示的元素
        const steps = {
            0: {
                // 初始状态：仅显示容器框架、活塞轮廓、V₀标识、容器名称
                fills: [],
                tags: [],
                substances: [],
                pistonHighlight: false,
                voEmphasize: false,
                pistonLines: false
            },
            1: {
                // 甲容器：填充 + 物质 + 恒容标识
                fills: ['scene1-fill-jia'],
                tags: ['scene1-tag-rong-jia'],
                substances: ['scene1-substance-jia'],
                pistonHighlight: false,
                voEmphasize: false,
                pistonLines: false
            },
            2: {
                // 乙容器：填充 + 物质 + 恒压标识 + 活塞高亮
                fills: ['scene1-fill-jia', 'scene1-fill-yi'],
                tags: ['scene1-tag-rong-jia', 'scene1-tag-ya-yi'],
                substances: ['scene1-substance-jia', 'scene1-substance-yi'],
                pistonHighlight: true,
                voEmphasize: false,
                pistonLines: true
            },
            3: {
                // 丙容器：填充 + 物质 + 恒容标识
                fills: ['scene1-fill-jia', 'scene1-fill-yi', 'scene1-fill-bing'],
                tags: ['scene1-tag-rong-jia', 'scene1-tag-ya-yi', 'scene1-tag-rong-bing'],
                substances: ['scene1-substance-jia', 'scene1-substance-yi', 'scene1-substance-bing'],
                pistonHighlight: false,
                voEmphasize: false,
                pistonLines: true
            },
            4: {
                // 完成状态：所有元素 + V₀强调
                fills: ['scene1-fill-jia', 'scene1-fill-yi', 'scene1-fill-bing'],
                tags: ['scene1-tag-rong-jia', 'scene1-tag-ya-yi', 'scene1-tag-rong-bing'],
                substances: ['scene1-substance-jia', 'scene1-substance-yi', 'scene1-substance-bing'],
                pistonHighlight: false,
                voEmphasize: true,
                pistonLines: true
            }
        };

        let currentStep = 0;
        const totalSteps = 4;

        // 工具函数：显示元素
        function showElement(id) {
            const el = document.getElementById(id);
            if (el) {
                el.classList.add('visible');
            }
        }

        // 工具函数：隐藏元素
        function hideElement(id) {
            const el = document.getElementById(id);
            if (el) {
                el.classList.remove('visible');
            }
        }

        // 更新显示状态
        function updateDisplay(step) {
            const state = steps[step];

            // 更新填充区域
            ['scene1-fill-jia', 'scene1-fill-yi', 'scene1-fill-bing'].forEach(id => {
                if (state.fills.includes(id)) {
                    showElement(id);
                } else {
                    hideElement(id);
                }
            });

            // 更新标识牌
            ['scene1-tag-rong-jia', 'scene1-tag-ya-yi', 'scene1-tag-rong-bing'].forEach(id => {
                if (state.tags.includes(id)) {
                    showElement(id);
                } else {
                    hideElement(id);
                }
            });

            // 更新物质标签
            ['scene1-substance-jia', 'scene1-substance-yi', 'scene1-substance-bing'].forEach(id => {
                if (state.substances.includes(id)) {
                    showElement(id);
                } else {
                    hideElement(id);
                }
            });

            // 更新活塞高亮
            const piston = document.getElementById('scene1-piston-yi');
            if (piston) {
                if (state.pistonHighlight) {
                    piston.classList.add('highlight');
                } else {
                    piston.classList.remove('highlight');
                }
            }

            // 更新活塞纹理线
            const pistonLines = document.querySelectorAll('.scene1-piston-line');
            pistonLines.forEach(line => {
                if (state.pistonLines) {
                    line.classList.add('visible');
                } else {
                    line.classList.remove('visible');
                }
            });

            // 更新 V₀ 强调
            const voLine = document.getElementById('scene1-vo-line');
            if (voLine) {
                if (state.voEmphasize) {
                    voLine.classList.add('emphasize');
                } else {
                    voLine.classList.remove('emphasize');
                }
            }

            // 更新步骤指示器
            const indicator = document.getElementById('scene1-step-indicator');
            if (indicator) {
                indicator.textContent = `步骤 ${step}/${totalSteps}`;
            }

            // 更新按钮状态
            const prevBtn = document.getElementById('scene1-btn-prev-bg');
            const nextBtn = document.getElementById('scene1-btn-next-bg');
            const prevText = document.getElementById('scene1-btn-prev-text');
            const nextText = document.getElementById('scene1-btn-next-text');

            if (prevBtn && prevText) {
                if (step === 0) {
                    prevBtn.setAttribute('fill', '#bdbdbd');
                    prevText.setAttribute('fill', '#9e9e9e');
                } else {
                    prevBtn.setAttribute('fill', '#9e9e9e');
                    prevText.setAttribute('fill', '#ffffff');
                }
            }

            if (nextBtn && nextText) {
                if (step >= totalSteps) {
                    nextBtn.setAttribute('fill', '#90caf9');
                    nextText.setAttribute('fill', '#bbdefb');
                } else {
                    nextBtn.setAttribute('fill', '#1976d2');
                    nextText.setAttribute('fill', '#ffffff');
                }
            }
        }

        // 前进一步
        function stepForward() {
            if (currentStep < totalSteps) {
                currentStep++;
                updateDisplay(currentStep);
            }
        }

        // 后退一步
        function stepBack() {
            if (currentStep > 0) {
                currentStep--;
                updateDisplay(currentStep);
            }
        }

        // 绑定按钮点击事件
        function setupEventListeners() {
            const prevGroup = document.getElementById('scene1-btn-prev-bg').parentNode;
            const nextGroup = document.getElementById('scene1-btn-next-bg').parentNode;

            if (prevGroup) {
                prevGroup.style.cursor = 'pointer';
                prevGroup.addEventListener('click', function(e) {
                    e.preventDefault();
                    if (currentStep > 0) {
                        stepBack();
                    }
                });
            }

            if (nextGroup) {
                nextGroup.style.cursor = 'pointer';
                nextGroup.addEventListener('click', function(e) {
                    e.preventDefault();
                    if (currentStep < totalSteps) {
                        stepForward();
                    }
                });
            }
        }

        // 初始化
        function init() {
            updateDisplay(currentStep);
            setupEventListeners();
        }

        // 页面加载完成后初始化
        if (document.readyState === 'loading') {
            document.addEventListener('DOMContentLoaded', init);
        } else {
            init();
        }
    })();
    </script>
