        // 获取所有 SVG 元素 ID
        const allSvgIds = [];
        document.querySelectorAll('#scene2-svg [id]').forEach(el => {
            allSvgIds.push(el.id);
        });
        
        // 显示/隐藏元素函数
        function showElement(id) {
            const el = document.getElementById(id);
            if (el) {
                el.style.display = 'block';
                el.style.opacity = '1';
            }
        }
        
        function hideElement(id) {
            const el = document.getElementById(id);
            if (el) {
                el.style.display = 'none';
                el.style.opacity = '0';
            }
        }
        
        // 移除动画类
        function removeAnimationClass(id) {
            const el = document.getElementById(id);
            if (el) {
                el.classList.remove('scene2-flash', 'scene2-pulse');
            }
        }
        
        // 渲染当前步骤
        function renderStep() {
            // 隐藏所有元素
            allSvgIds.forEach(id => hideElement(id));
            
            // 显示当前步骤的元素
            const currentIds = steps[currentStep];
            currentIds.forEach(id => showElement(id));
            
            // 移除所有动画类
            allSvgIds.forEach(id => removeAnimationClass(id));
            
            // 添加步骤特定动画
            if (currentStep === 1) {
                // 步骤 1: P→F₁箭头闪烁
                document.getElementById('scene2-p-f1-arrow')?.classList.add('scene2-flash');
                document.getElementById('scene2-p-f1-arrowhead')?.classList.add('scene2-flash');
            } else if (currentStep === 2) {
                // 步骤 2: 配子箭头闪烁
                document.getElementById('scene2-f1-f2-arrow-f1')?.classList.add('scene2-flash');
                document.getElementById('scene2-f1-f2-arrow-f2')?.classList.add('scene2-flash');
                document.getElementById('scene2-f1-f2-arrow-m1')?.classList.add('scene2-flash');
                document.getElementById('scene2-f1-f2-arrow-m2')?.classList.add('scene2-flash');
            } else if (currentStep === 4) {
                // 步骤 4: 比例框脉冲
                document.getElementById('scene2-ratio-box')?.classList.add('scene2-pulse');
            }
            
            // 更新步骤指示器
            document.getElementById('scene2-step-indicator').textContent = `步骤 ${currentStep + 1}/${totalSteps}`;
            
            // 更新按钮状态
            document.getElementById('scene2-btn-prev').disabled = (currentStep === 0);
            document.getElementById('scene2-btn-next').disabled = (currentStep === totalSteps - 1);
        }
        
        // 更新步骤
        function updateStep(delta) {
            const newStep = currentStep + delta;
            if (newStep >= 0 && newStep < totalSteps) {
                currentStep = newStep;
                requestAnimationFrame(() => {
                    renderStep();
                });
            }
        }
        
        // 事件绑定
        document.getElementById('scene2-btn-prev').addEventListener('click', () => updateStep(-1));
        document.getElementById('scene2-btn-next').addEventListener('click', () => updateStep(1));
        
        // 初始化
        renderStep();
    })();
