    <script>
    (function() {
        // Scene 3 实现：测交实验方案对比与评价
        
        // 状态变量
        let currentStep = 0;
        const totalSteps = 4;
        
        // DOM 元素引用
        const btnPrev = document.getElementById('scene3-btn-prev');
        const btnNext = document.getElementById('scene3-btn-next');
        const stepIndicator = document.getElementById('scene3-step-indicator');
        
        // 元素分组
        const leftExperimentElements = [
            'scene3-left-box',
            'scene3-left-title',
            'scene3-left-p-label',
            'scene3-left-p-genotype',
            'scene3-left-gamete-label',
            'scene3-left-female-g1',
            'scene3-left-female-g2',
            'scene3-left-male-g1',
            'scene3-left-male-g2',
            'scene3-left-arrow1',
            'scene3-left-arrow2',
            'scene3-left-arrow3',
            'scene3-left-arrow4',
            'scene3-left-offspring1-geno',
            'scene3-left-offspring2-geno',
            'scene3-left-offspring3-geno',
            'scene3-left-offspring4-geno',
            'scene3-left-ratio'
        ];
        
        const rightExperimentElements = [
            'scene3-right-box',
            'scene3-right-title',
            'scene3-right-p-label',
            'scene3-right-p-genotype',
            'scene3-right-gamete-label',
            'scene3-right-female-g',
            'scene3-right-male-g1',
            'scene3-right-male-g2',
            'scene3-right-arrow1',
            'scene3-right-arrow2',
            'scene3-right-arrow3',
            'scene3-right-offspring1-geno',
            'scene3-right-offspring2-geno',
            'scene3-right-ratio'
        ];
        
        const problemBox = ['scene3-problem-box'];
        const conclusionBox = ['scene3-conclusion-box'];
        const comparisonTable = ['scene3-comparison-table'];
        
        // 工具函数：显示元素
        function showElements(elementIds) {
            elementIds.forEach(id => {
                const el = document.getElementById(id);
                if (el) {
                    el.classList.remove('scene3-hidden');
                    el.classList.add('scene3-visible');
                }
            });
        }
        
        // 工具函数：隐藏元素
        function hideElements(elementIds) {
            elementIds.forEach(id => {
                const el = document.getElementById(id);
                if (el) {
                    el.classList.remove('scene3-visible');
                    el.classList.add('scene3-hidden');
                }
            });
        }
        
        // 更新显示状态
        function updateDisplay() {
            // 步骤 0：只显示左侧实验
            // 步骤 1：左侧实验 + 问题标注
            // 步骤 2：左侧实验 + 问题标注 + 右侧实验 + 结论标注
            // 步骤 3：全部显示（包括对比表格）
            
            // 始终显示左侧实验
            showElements(leftExperimentElements);
            
            if (currentStep >= 1) {
                showElements(problemBox);
            } else {
                hideElements(problemBox);
            }
            
            if (currentStep >= 2) {
                showElements(rightExperimentElements);
                showElements(conclusionBox);
            } else {
                hideElements(rightExperimentElements);
                hideElements(conclusionBox);
            }
            
            if (currentStep >= 3) {
                showElements(comparisonTable);
            } else {
                hideElements(comparisonTable);
            }
            
            // 更新按钮状态
            btnPrev.disabled = (currentStep === 0);
            
            if (currentStep === totalSteps - 1) {
                btnNext.textContent = '完成';
                btnNext.disabled = true;
            } else {
                btnNext.textContent = '下一步 ▶';
                btnNext.disabled = false;
            }
            
            // 更新步骤指示器
            stepIndicator.textContent = '步骤 ' + (currentStep + 1) + '/' + totalSteps;
            
            // 更新左侧边框颜色（步骤 1 后变为红色）
            const leftBox = document.getElementById('scene3-left-box');
            if (currentStep >= 1) {
                leftBox.setAttribute('stroke', '#c62828');
            } else {
                leftBox.setAttribute('stroke', '#888888');
            }
        }
        
        // 事件处理：上一步
        btnPrev.addEventListener('click', function() {
            if (currentStep > 0) {
                currentStep--;
                updateDisplay();
            }
        });
        
        // 事件处理：下一步
        btnNext.addEventListener('click', function() {
            if (currentStep < totalSteps - 1) {
                currentStep++;
                updateDisplay();
            }
        });
        
        // 初始化显示
        updateDisplay();
        
    })();
    </script>
