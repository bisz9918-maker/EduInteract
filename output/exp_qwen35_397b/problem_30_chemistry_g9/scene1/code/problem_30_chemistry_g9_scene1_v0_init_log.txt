            <text id="scene1-material-m" x="240" y="425" text-anchor="middle" font-size="13px" fill="#2e7d32" style="display:none;">氢氧化钙/碳酸钠</text>

            <!-- 提示文字（初始隐藏） -->
            <text id="scene1-hint-y" x="400" y="185" text-anchor="middle" font-size="13px" fill="#ff9800" style="display:none;">4 条连线</text>

            <!-- 最终确定物质标签（初始隐藏） -->
            <text id="scene1-final-x" x="140" y="245" text-anchor="middle" font-size="13px" fill="#2e7d32" style="display:none;">碳酸钠</text>

        <div id="scene1-controls">
            <button id="scene1-btn-prev" class="scene1-btn scene1-btn-prev">◀ 上一步</button>
            <span id="scene1-step-indicator">步骤 0/5</span>
            <button id="scene1-btn-next" class="scene1-btn scene1-btn-next">下一步 ▶</button>
        </div>

    <script>
    (function() {
        // Scene 1 实现 - 物质推断关系网络图

        let currentStep = 0;
        const totalSteps = 5;

        // 获取所有元素
        const elements = {
            // 节点
            nodeX: document.getElementById('scene1-node-x'),
            nodeY: document.getElementById('scene1-node-y'),
            nodeM: document.getElementById('scene1-node-m'),
            nodeN: document.getElementById('scene1-node-n'),
            nodeFe: document.getElementById('scene1-node-fe'),
            // 连线
            lineXY: document.getElementById('scene1-line-x-y'),
            lineXM: document.getElementById('scene1-line-x-m'),
            lineYM: document.getElementById('scene1-line-y-m'),
            lineYFe: document.getElementById('scene1-line-y-fe'),
            lineYN: document.getElementById('scene1-line-y-n'),
            arrowNFe: document.getElementById('scene1-arrow-n-fe'),
            // 标签
            labelX: document.getElementById('scene1-label-x'),
            labelY: document.getElementById('scene1-label-y'),
            labelM: document.getElementById('scene1-label-m'),
            labelN: document.getElementById('scene1-label-n'),
            labelFe: document.getElementById('scene1-label-fe'),
            // 物质标签
            materialY: document.getElementById('scene1-material-y'),
            materialN: document.getElementById('scene1-material-n'),
            materialX: document.getElementById('scene1-material-x'),
            materialM: document.getElementById('scene1-material-m'),
            // 提示
            hintY: document.getElementById('scene1-hint-y'),
            // 最终标签
            finalX: document.getElementById('scene1-final-x'),
            finalM: document.getElementById('scene1-final-m'),
            // 控件
            btnPrev: document.getElementById('scene1-btn-prev'),
            btnNext: document.getElementById('scene1-btn-next'),
            stepIndicator: document.getElementById('scene1-step-indicator')
        };

        // 辅助函数：显示/隐藏元素
        function show(el) {
            if (el) {
                el.style.display = 'block';
                el.style.opacity = '1';
            }
        }

        function hide(el) {
            if (el) {
                el.style.display = 'none';
                el.style.opacity = '0';
            }
        }

        // 重置所有元素到默认状态
        function resetToDefault() {
            // 所有节点恢复蓝色描边
            [elements.nodeX, elements.nodeY, elements.nodeM, elements.nodeN, elements.nodeFe].forEach(node => {
                node.setAttribute('stroke', '#1565c0');
                node.setAttribute('stroke-width', '2');
            });
            // 所有连线恢复黑色
            [elements.lineXY, elements.lineXM, elements.lineYM, elements.lineYFe, elements.lineYN, elements.arrowNFe].forEach(line => {
                line.setAttribute('stroke', '#000000');
                line.setAttribute('stroke-width', '2');
            });
            // 隐藏所有物质标签和提示
            hide(elements.materialY);
            hide(elements.materialN);
            hide(elements.materialX);
            hide(elements.materialM);
            hide(elements.hintY);
            hide(elements.finalX);
            hide(elements.finalM);
        }

        // 更新步骤显示
        function updateStep() {
            resetToDefault();

            // 更新步骤指示器
            elements.stepIndicator.textContent = `步骤 ${currentStep}/5`;

            // 更新按钮状态
            elements.btnPrev.disabled = currentStep === 0;
            elements.btnNext.disabled = currentStep === totalSteps;
            elements.btnNext.textContent = currentStep === totalSteps ? '完成' : '下一步 ▶';

            // 根据步骤显示不同内容
            switch (currentStep) {
                case 0:
                    // 初始状态：只显示原题空图（已默认显示）
                    break;

                case 1:
                    // 高亮 Y 的连接数
                    elements.nodeY.setAttribute('stroke', '#ff9800');
                    elements.nodeY.setAttribute('stroke-width', '4');
                    elements.lineXY.setAttribute('stroke', '#ff9800');
                    elements.lineXY.setAttribute('stroke-width', '4');
                    elements.lineYM.setAttribute('stroke', '#ff9800');
                    elements.lineYM.setAttribute('stroke-width', '4');
                    elements.lineYFe.setAttribute('stroke', '#ff9800');
                    elements.lineYFe.setAttribute('stroke-width', '4');
                    elements.lineYN.setAttribute('stroke', '#ff9800');
                    elements.lineYN.setAttribute('stroke-width', '4');
                    show(elements.hintY);
                    break;

                case 2:
                    // 标注 Y=稀盐酸
                    elements.nodeY.setAttribute('stroke', '#ff9800');
                    elements.nodeY.setAttribute('stroke-width', '4');
                    elements.lineXY.setAttribute('stroke', '#ff9800');
                    elements.lineXY.setAttribute('stroke-width', '4');
                    elements.lineYM.setAttribute('stroke', '#ff9800');
                    elements.lineYM.setAttribute('stroke-width', '4');
                    elements.lineYFe.setAttribute('stroke', '#ff9800');
                    elements.lineYFe.setAttribute('stroke-width', '4');
                    elements.lineYN.setAttribute('stroke', '#ff9800');
                    elements.lineYN.setAttribute('stroke-width', '4');
                    show(elements.materialY);
                    break;

                case 3:
                    // 标注 N=氧化铁
                    show(elements.materialY);
                    elements.nodeN.setAttribute('stroke', '#2e7d32');
                    elements.nodeN.setAttribute('stroke-width', '3');
                    elements.arrowNFe.setAttribute('stroke', '#2e7d32');
                    elements.arrowNFe.setAttribute('stroke-width', '3');
                    show(elements.materialN);
                    break;

                case 4:
                    // 标注 X、M=碳酸钠/氢氧化钙
                    show(elements.materialY);
                    show(elements.materialN);
                    elements.nodeX.setAttribute('stroke', '#2e7d32');
                    elements.nodeX.setAttribute('stroke-width', '3');
                    elements.nodeM.setAttribute('stroke', '#2e7d32');
                    elements.nodeM.setAttribute('stroke-width', '3');
                    show(elements.materialX);
                    show(elements.materialM);
                    break;

                case 5:
                    // 完整标注所有物质（最终结论）
                    show(elements.materialY);
                    show(elements.materialN);
                    hide(elements.materialX);
                    hide(elements.materialM);
                    show(elements.finalX);
                    show(elements.finalM);
                    break;
            }
        }

        // 事件绑定
        elements.btnPrev.addEventListener('click', function() {
            if (currentStep > 0) {
                currentStep--;
                updateStep();
            }
        });

        elements.btnNext.addEventListener('click', function() {
            if (currentStep < totalSteps) {
                currentStep++;
                updateStep();
            }
        });

        // 初始化
        updateStep();
    })();
    </script>
