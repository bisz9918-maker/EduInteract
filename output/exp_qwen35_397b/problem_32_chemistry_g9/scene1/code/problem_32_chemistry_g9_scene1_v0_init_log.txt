    <script>
    (function() {
        // Scene 1 实现 - 化学实验装置分步展示
        
        // 布局坐标表参考 (来自 spec.json):
        // | 元素 | 类型 | 位置/范围 | 大小 | 标签偏移方向 | 备注 |
        // | A瓶轮廓 | 圆形+瓶颈 | 中心(200,130) | 半径55px | - | 左侧圆底烧瓶 |
        // | B瓶轮廓 | 圆形+瓶颈 | 中心(500,130) | 半径55px | - | 右侧圆底烧瓶 |
        // | 胶头滴管 | 组合图形 | (530,175)-(540,250) | - | 右侧 | B瓶右侧 |
        // | 导管(水平) | 线段 | (200,260)-(500,260) | 300×3 | - | 连接两瓶 |
        // | 导管(垂直) | 线段 | (350,260)-(350,360) | 3×100 | - | 中央向下 |
        // | 活塞a | 矩形 | (485,250)-(515,270) | 30×20 | 右侧 | B瓶下方导管 |
        // | 活塞b | 矩形 | (335,300)-(365,320) | 30×20 | 右侧 | 中央垂直导管 |
        // | 活塞c | 矩形 | (185,250)-(215,270) | 30×20 | 左侧 | A瓶下方导管 |
        // | 烧杯轮廓 | 梯形 | (290,360)-(410,450) | 120×90 | - | 下方中央 |
        // | 标签"A" | text | (200,55) | - | 上方 | A瓶正上方 |
        // | 标签"B" | text | (500,55) | - | 上方 | B瓶正上方 |
        // | 标签"HCl" | text | (200,135) | - | 居中 | A瓶内部中央 |
        // | 标签"NH₃" | text | (500,135) | - | 居中 | B瓶内部中央 |
        
        // 状态定义
        const steps = [
            {
                // 步骤 0: 初始状态 - 完整装置
                name: "初始状态",
                highlight: []
            },
            {
                // 步骤 1: A瓶及HCl气体高亮
                name: "A瓶 (HCl)",
                highlight: ['scene1-flask-a-body', 'scene1-flask-a-gas', 'scene1-flask-a-neck', 'scene1-label-a', 'scene1-label-hcl']
            },
            {
                // 步骤 2: B瓶、NH₃气体及胶头滴管高亮
                name: "B瓶 (NH₃) + 胶头滴管",
                highlight: ['scene1-flask-b-body', 'scene1-flask-b-gas', 'scene1-flask-b-neck', 'scene1-dropper-bulb', 'scene1-dropper-tube', 'scene1-dropper-tip', 'scene1-label-b', 'scene1-label-nh3', 'scene1-label-dropper-1', 'scene1-label-dropper-2', 'scene1-label-dropper-3']
            },
            {
                // 步骤 3: 三个活塞高亮
                name: "活塞 a, b, c",
                highlight: ['scene1-piston-a', 'scene1-piston-b', 'scene1-piston-c', 'scene1-label-piston-a', 'scene1-label-piston-b', 'scene1-label-piston-c']
            },
            {
                // 步骤 4: 烧杯及水高亮
                name: "烧杯 + 水",
                highlight: ['scene1-beaker-body', 'scene1-beaker-water', 'scene1-label-water']
            }
        ];
        
        let currentStep = 0;
        const totalSteps = steps.length;
        
        // DOM 元素
        const btnPrev = document.getElementById('scene1-btn-prev');
        const btnNext = document.getElementById('scene1-btn-next');
        const stepIndicator = document.getElementById('scene1-step-indicator');
        
        // 清除所有高亮
        function clearHighlights() {
            // 清除烧瓶高亮
            ['scene1-flask-a-body', 'scene1-flask-b-body', 'scene1-flask-a-neck', 'scene1-flask-b-neck'].forEach(id => {
                const el = document.getElementById(id);
                if (el) el.classList.remove('scene1-highlight-flask-a', 'scene1-highlight-flask-b');
            });
            // 清除气体高亮
            ['scene1-flask-a-gas', 'scene1-flask-b-gas'].forEach(id => {
                const el = document.getElementById(id);
                if (el) el.style.fillOpacity = '';
            });
            // 清除活塞高亮
            ['scene1-piston-a', 'scene1-piston-b', 'scene1-piston-c'].forEach(id => {
                const el = document.getElementById(id);
                if (el) el.classList.remove('scene1-highlight-piston');
            });
            // 清除烧杯高亮
            const beakerBody = document.getElementById('scene1-beaker-body');
            if (beakerBody) beakerBody.classList.remove('scene1-highlight-beaker');
            // 清除标签高亮
            document.querySelectorAll('.scene1-label, .scene1-label-small').forEach(el => {
                el.classList.remove('scene1-highlight-label');
            });
            // 清除胶头滴管高亮
            ['scene1-dropper-bulb', 'scene1-dropper-tube', 'scene1-dropper-tip'].forEach(id => {
                const el = document.getElementById(id);
                if (el) el.style.stroke = '';
            });
        }
        
        // 应用高亮
        function applyHighlight(stepIndex) {
            const step = steps[stepIndex];
            step.highlight.forEach(id => {
                const el = document.getElementById(id);
                if (!el) return;
                
                // A瓶高亮
                if (id.includes('flask-a')) {
                    el.classList.add('scene1-highlight-flask-a');
                    if (id.includes('gas')) {
                        el.style.fillOpacity = '0.4';
                    }
                }
                // B瓶高亮
                else if (id.includes('flask-b')) {
                    el.classList.add('scene1-highlight-flask-b');
                    if (id.includes('gas')) {
                        el.style.fillOpacity = '0.4';
                    }
                }
                // 胶头滴管高亮
                else if (id.includes('dropper')) {
                    el.style.stroke = '#2e7d32';
                    el.style.strokeWidth = '2px';
                }
                // 活塞高亮
                else if (id.includes('piston') && !id.includes('label')) {
                    el.classList.add('scene1-highlight-piston');
                }
                // 烧杯高亮
                else if (id.includes('beaker')) {
                    el.classList.add('scene1-highlight-beaker');
                    if (id.includes('water')) {
                        el.style.fillOpacity = '0.5';
                    }
                }
                // 标签高亮
                else if (id.includes('label')) {
                    el.classList.add('scene1-highlight-label');
                }
            });
        }
        
        // 更新状态
        function updateState() {
            clearHighlights();
            applyHighlight(currentStep);
            
            // 更新按钮状态
            btnPrev.disabled = currentStep === 0;
            btnNext.disabled = currentStep === totalSteps - 1;
            
            // 更新步骤指示器
            stepIndicator.textContent = `步骤 ${currentStep + 1}/${totalSteps}`;
        }
        
        // 事件绑定
        btnPrev.addEventListener('click', function() {
            if (currentStep > 0) {
                currentStep--;
                updateState();
            }
        });
        
        btnNext.addEventListener('click', function() {
            if (currentStep < totalSteps - 1) {
                currentStep++;
                updateState();
            }
        });
        
        // 初始化
        updateState();
    })();
    </script>
