    <script>
    (function() {
        // Scene 1 实现：平移法比较甲乙周长

        // 布局坐标表（对照依据）：
        // | 元素 | 类型 | 位置/范围 | 大小 | 备注 |
        // |------|------|-----------|------|------|
        // | 方格背景 | 网格 | (50,50)-(750,370) | 700×320 | 17 格×8 格，每格 40px |
        // | 甲图形 | 多边形 | 左侧区域 | 200×120 | 5 格×3 格带凹陷 |
        // | 乙图形 | 多边形 | 右侧区域 | 200×120 | 5 格×3 格带凸起 |
        // | 甲-标签 | text | (230,220) | ~30×20 | 图形内部中央 |
        // | 乙-标签 | text | (550,220) | ~30×20 | 图形内部中央 |
        // | 甲凹边 1 | 高亮线段 | (210,130)-(210,170) | 40×3 | 竖直凹边 |
        // | 甲凹边 2 | 高亮线段 | (250,130)-(250,170) | 40×3 | 竖直凹边 |
        // | 甲平移箭头 1 | 箭头 | (195,160)→(195,110) | 50px | 指示向上平移 |
        // | 甲平移箭头 2 | 箭头 | (265,160)→(265,110) | 50px | 指示向上平移 |
        // | 甲外接长方形 | 虚线框 | (130,130)-(330,290) | 200×160 | 绿色虚线 |
        // | 乙凸边 1 | 高亮线段 | (510,90)-(510,130) | 40×3 | 竖直凸边 |
        // | 乙凸边 2 | 高亮线段 | (570,90)-(570,130) | 40×3 | 竖直凸边 |
        // | 乙平移箭头 1 | 箭头 | (495,100)→(495,150) | 50px | 指示向下平移 |
        // | 乙平移箭头 2 | 箭头 | (585,100)→(585,150) | 50px | 指示向下平移 |
        // | 乙外接长方形 | 虚线框 | (450,90)-(650,290) | 200×200 | 绿色虚线 |
        // | 结论文本 | text | (400,350) | ~120×24 | 居中 |

        let currentStep = 0;
        const totalSteps = 5;

        // 获取 DOM 元素
        const jiaHighlight1 = document.getElementById('scene1-jia-highlight-1');
        const jiaHighlight2 = document.getElementById('scene1-jia-highlight-2');
        const jiaArrow1 = document.getElementById('scene1-jia-arrow-1');
        const jiaArrow2 = document.getElementById('scene1-jia-arrow-2');
        const jiaRect = document.getElementById('scene1-jia-rect');
        const yiHighlight1 = document.getElementById('scene1-yi-highlight-1');
        const yiHighlight2 = document.getElementById('scene1-yi-highlight-2');
        const yiArrow1 = document.getElementById('scene1-yi-arrow-1');
        const yiArrow2 = document.getElementById('scene1-yi-arrow-2');
        const yiRect = document.getElementById('scene1-yi-rect');
        const conclusion = document.getElementById('scene1-conclusion');
        const stepIndicator = document.getElementById('scene1-step-indicator');
        const btnPrevBg = document.getElementById('scene1-btn-prev-bg');
        const btnPrevText = document.getElementById('scene1-btn-prev-text');
        const btnNextBg = document.getElementById('scene1-btn-next-bg');
        const btnNextText = document.getElementById('scene1-btn-next-text');

        // 更新按钮状态
        function updateButtons() {
            if (currentStep === 0) {
                btnPrevBg.setAttribute('fill', '#f5f5f5');
                btnPrevBg.setAttribute('stroke', '#cccccc');
                btnPrevText.setAttribute('fill', '#999999');
            } else {
                btnPrevBg.setAttribute('fill', '#ffffff');
                btnPrevBg.setAttribute('stroke', '#1976d2');
                btnPrevText.setAttribute('fill', '#1976d2');
            }

            if (currentStep === totalSteps) {
                btnNextBg.setAttribute('fill', '#f5f5f5');
                btnNextBg.setAttribute('stroke', '#cccccc');
                btnNextText.setAttribute('fill', '#999999');
            } else {
                btnNextBg.setAttribute('fill', '#ffffff');
                btnNextBg.setAttribute('stroke', '#1976d2');
                btnNextText.setAttribute('fill', '#1976d2');
            }

            stepIndicator.textContent = '步骤 ' + (currentStep + 1) + '/' + totalSteps;
        }

        // 更新步骤显示
        function updateStep() {
            // 重置所有元素为隐藏
            jiaHighlight1.setAttribute('opacity', '0');
            jiaHighlight2.setAttribute('opacity', '0');
            jiaArrow1.setAttribute('opacity', '0');
            jiaArrow2.setAttribute('opacity', '0');
            jiaRect.setAttribute('opacity', '0');
            yiHighlight1.setAttribute('opacity', '0');
            yiHighlight2.setAttribute('opacity', '0');
            yiArrow1.setAttribute('opacity', '0');
            yiArrow2.setAttribute('opacity', '0');
            yiRect.setAttribute('opacity', '0');
            conclusion.setAttribute('opacity', '0');

            // 根据当前步骤显示相应元素
            if (currentStep >= 1) {
                // 步骤 1：甲凹边高亮 + 平移箭头
                jiaHighlight1.setAttribute('opacity', '1');
                jiaHighlight2.setAttribute('opacity', '1');
                jiaArrow1.setAttribute('opacity', '1');
                jiaArrow2.setAttribute('opacity', '1');
            }

            if (currentStep >= 2) {
                // 步骤 2：甲外接长方形
                jiaRect.setAttribute('opacity', '1');
            }

            if (currentStep >= 3) {
                // 步骤 3：乙凸边高亮 + 平移箭头（隐藏甲相关）
                jiaHighlight1.setAttribute('opacity', '0');
                jiaHighlight2.setAttribute('opacity', '0');
                jiaArrow1.setAttribute('opacity', '0');
                jiaArrow2.setAttribute('opacity', '0');
                jiaRect.setAttribute('opacity', '0');

                yiHighlight1.setAttribute('opacity', '1');
                yiHighlight2.setAttribute('opacity', '1');
                yiArrow1.setAttribute('opacity', '1');
                yiArrow2.setAttribute('opacity', '1');
            }

            if (currentStep >= 4) {
                // 步骤 4：乙外接长方形
                yiRect.setAttribute('opacity', '1');
            }

            if (currentStep >= 5) {
                // 步骤 5：同时显示两个长方形，显示结论
                jiaRect.setAttribute('opacity', '1');
                yiRect.setAttribute('opacity', '1');
                yiHighlight1.setAttribute('opacity', '0');
                yiHighlight2.setAttribute('opacity', '0');
                yiArrow1.setAttribute('opacity', '0');
                yiArrow2.setAttribute('opacity', '0');
                conclusion.setAttribute('opacity', '1');
            }

            updateButtons();
        }

        // 按钮点击事件
        btnPrevBg.addEventListener('click', function() {
            if (currentStep > 0) {
                currentStep--;
                updateStep();
            }
        });

        btnNextBg.addEventListener('click', function() {
            if (currentStep < totalSteps) {
                currentStep++;
                updateStep();
            }
        });

        // 鼠标悬停效果
        btnPrevBg.addEventListener('mouseenter', function() {
            if (currentStep > 0) {
                btnPrevBg.setAttribute('fill', '#e3f2fd');
            }
        });
        btnPrevBg.addEventListener('mouseleave', function() {
            if (currentStep > 0) {
                btnPrevBg.setAttribute('fill', '#ffffff');
            }
        });

        btnNextBg.addEventListener('mouseenter', function() {
            if (currentStep < totalSteps) {
                btnNextBg.setAttribute('fill', '#e3f2fd');
            }
        });
        btnNextBg.addEventListener('mouseleave', function() {
            if (currentStep < totalSteps) {
                btnNextBg.setAttribute('fill', '#ffffff');
            }
        });

        // 初始化
        updateStep();
    })();
    </script>
