    <script>
    (function() {
        // Scene 2: 距离 a 与影长 L 的关系
        // 布局坐标表（实现对照依据）:
        // | 元素 | 类型 | 位置/范围 | 大小 | 标签偏移方向 | 备注 |
        // |------|------|-----------|------|-------------|------|
        // | 标题 | text | (400,35) | ~240×24 | - | 居中 |
        // | 地面 | 线段 | (60,400)-(640,400) | 580×3 | - | #888888 |
        // | 路灯杆 | 矩形 | (85,120)-(95,400) | 10×280 | - | #000000 |
        // | 光源 | 圆形 | (90,100) | r=18 | - | #ffc107 |
        // | 物体 | 梯形 | 初始 (300,340)-(340,400) | 40×60 | - | #1565c0 |
        // | 光线 | 线段 | (90,100)→物体顶部→地面交点 | 动态 | - | #c62828 |
        // | 影子区域 | 多边形 | 物体底部到光线与地面交点 | 动态 | - | #2e7d32 |
        // | 距离 a 标注 | 双向箭头 | (95,415) 到物体底部 x | 动态 | 下方 | #333333 |
        // | 影长 L 标注 | 双向箭头 | 物体底部 x 到影子末端 | 动态 | 下方 | #2e7d32 |
        // | 数值面板 | 矩形 | (660,80)-(780,280) | 120×200 | - | #f5f5f5 |
        // | 滑块 | 轨道+手柄 | (150,465)-(650,475) | 500×10 | - | #cccccc |
        
        // 常量定义
        const LIGHT_X = 90, LIGHT_Y = 100;
        const GROUND_Y = 400;
        const POLE_X = 90;
        const OBJECT_HEIGHT = 60;
        const OBJECT_TOP_Y = GROUND_Y - OBJECT_HEIGHT; // 340
        const OBJECT_WIDTH_BOTTOM = 40;
        const OBJECT_WIDTH_TOP = 20;
        const LIGHT_HEIGHT = GROUND_Y - LIGHT_Y; // 300
        
        // 滑块范围
        const SLIDER_MIN_X = 150;
        const SLIDER_MAX_X = 650;
        const OBJECT_MIN_X = 150;
        const OBJECT_MAX_X = 550;
        
        // 状态变量
        let sliderX = 350;
        let objectX = 300; // 物体左下角 x 坐标
        let isDragging = false;
        let records = [];
        
        // DOM 元素引用
        const svg = document.getElementById('scene2-svg');
        const objectEl = document.getElementById('scene2-object');
        const rayEl = document.getElementById('scene2-ray');
        const shadowEl = document.getElementById('scene2-shadow');
        const arrowAEl = document.getElementById('scene2-arrow-a');
        const arrowARightEl = document.getElementById('scene2-arrow-a-right');
        const labelAEl = document.getElementById('scene2-label-a');
        const arrowLEl = document.getElementById('scene2-arrow-l');
        const arrowLRightEl = document.getElementById('scene2-arrow-l-right');
        const labelLEl = document.getElementById('scene2-label-l');
        const panelAEl = document.getElementById('scene2-panel-a');
        const panelLEl = document.getElementById('scene2-panel-l');
        const sliderHandleEl = document.getElementById('scene2-slider-handle');
        const btnRecordGroup = document.getElementById('scene2-btn-record');
        const btnRecordBg = document.getElementById('scene2-btn-record-bg');
        const recordListEl = document.getElementById('scene2-record-list');
        
        // 生成地面阴影线
        function createGroundHatch() {
            const hatchGroup = document.getElementById('scene2-ground-hatch');
            hatchGroup.innerHTML = '';
            for (let x = 60; x <= 640; x += 10) {
                const line = document.createElementNS('http://www.w3.org/2000/svg', 'line');
                line.setAttribute('x1', x);
                line.setAttribute('y1', 400);
                line.setAttribute('x2', x - 5);
                line.setAttribute('y2', 410);
                hatchGroup.appendChild(line);
            }
        }
        
        // 计算影子长度（相似三角形原理）
        // 路灯高度 H = 300, 物体高度 h = 60
        // 物体到光源水平距离 d = objectX + 20 - 90 (物体中心到光源)
        // 影子长度 L = d * h / (H - h) = d * 60 / 240 = d / 4
        function calculateShadow(objectBottomLeftX) {
            const objectCenterX = objectBottomLeftX + OBJECT_WIDTH_BOTTOM / 2;
            const d = objectCenterX - LIGHT_X; // 物体到光源的水平距离
            const L = d * OBJECT_HEIGHT / (LIGHT_HEIGHT - OBJECT_HEIGHT); // 影子长度
            const shadowEndX = objectBottomLeftX + OBJECT_WIDTH_BOTTOM + L;
            return { L, shadowEndX };
        }
        
        // 更新所有视觉元素
        function updateScene() {
            const { L, shadowEndX } = calculateShadow(objectX);
            
            // 更新物体位置（梯形）
            const objectPoints = `${objectX + OBJECT_WIDTH_TOP/2},${OBJECT_TOP_Y} ${objectX + OBJECT_WIDTH_BOTTOM - OBJECT_WIDTH_TOP/2},${OBJECT_TOP_Y} ${objectX + OBJECT_WIDTH_BOTTOM},${GROUND_Y} ${objectX},${GROUND_Y}`;
            objectEl.setAttribute('points', objectPoints);
            
            // 更新光线（光源 → 物体顶部中点 → 地面交点）
            const objectTopCenterX = objectX + OBJECT_WIDTH_BOTTOM / 2;
            const rayPoints = `${LIGHT_X},${LIGHT_Y} ${objectTopCenterX},${OBJECT_TOP_Y} ${shadowEndX},${GROUND_Y}`;
            rayEl.setAttribute('points', rayPoints);
            
            // 更新影子区域（三角形：物体左下、物体右下、影子末端）
            const shadowPoints = `${objectX},${GROUND_Y} ${objectX + OBJECT_WIDTH_BOTTOM},${GROUND_Y} ${shadowEndX},${GROUND_Y}`;
            shadowEl.setAttribute('points', shadowPoints);
            
            // 更新距离 a 标注
            const a = (objectX - POLE_X) / 100; // 转换为米（假设 100px = 1m）
            arrowAEl.setAttribute('x2', objectX);
            arrowARightEl.setAttribute('points', `${objectX},415 ${objectX - 5},412 ${objectX - 5},418`);
            const aMidX = (POLE_X + objectX) / 2;
            labelAEl.setAttribute('x', aMidX);
            labelAEl.textContent = `a=${a.toFixed(1)}`;
            
            // 更新影长 L 标注
            arrowLEl.setAttribute('x1', objectX + OBJECT_WIDTH_BOTTOM);
            arrowLEl.setAttribute('x2', shadowEndX);
            arrowLRightEl.setAttribute('points', `${shadowEndX},450 ${shadowEndX - 5},447 ${shadowEndX - 5},453`);
            const lMidX = (objectX + OBJECT_WIDTH_BOTTOM + shadowEndX) / 2;
            labelLEl.setAttribute('x', lMidX);
            labelLEl.textContent = `L=${(L/100).toFixed(1)}`;
            
            // 更新数值面板
            panelAEl.textContent = `a = ${a.toFixed(1)} m`;
            panelLEl.textContent = `L = ${(L/100).toFixed(1)} m`;
            
            // 更新滑块手柄位置
            sliderHandleEl.setAttribute('cx', sliderX);
        }
        
        // 滑块 X 到物体 X 的映射
        function sliderToObjectX(sx) {
            const ratio = (sx - SLIDER_MIN_X) / (SLIDER_MAX_X - SLIDER_MIN_X);
            return OBJECT_MIN_X + ratio * (OBJECT_MAX_X - OBJECT_MIN_X);
        }
        
        // 物体 X 到滑块 X 的映射
        function objectToSliderX(ox) {
            const ratio = (ox - OBJECT_MIN_X) / (OBJECT_MAX_X - OBJECT_MIN_X);
            return SLIDER_MIN_X + ratio * (SLIDER_MAX_X - SLIDER_MIN_X);
        }
        
        // 鼠标/触摸事件处理
        function handleStart(e) {
            const rect = svg.getBoundingClientRect();
            const clientX = e.touches ? e.touches[0].clientX : e.clientX;
            const x = clientX - rect.left;
            
            // 检查是否点击在滑块手柄上
            const handleCx = parseFloat(sliderHandleEl.getAttribute('cx'));
            const dist = Math.sqrt((x - handleCx) ** 2 + (470 - 470) ** 2);
            if (dist < 20) {
                isDragging = true;
                e.preventDefault();
            }
        }
        
        function handleMove(e) {
            if (!isDragging) return;
            
            const rect = svg.getBoundingClientRect();
            const clientX = e.touches ? e.touches[0].clientX : e.clientX;
            let x = clientX - rect.left;
            
            // 限制在滑块范围内
            x = Math.max(SLIDER_MIN_X, Math.min(SLIDER_MAX_X, x));
            sliderX = x;
            objectX = sliderToObjectX(sliderX);
            
            updateScene();
            e.preventDefault();
        }
        
        function handleEnd() {
            isDragging = false;
        }
        
        // 记录功能
        function addRecord() {
            const { L } = calculateShadow(objectX);
            const a = (objectX - POLE_X) / 100;
            
            records.unshift({ a: a.toFixed(1), L: (L/100).toFixed(1) });
            if (records.length > 5) records.pop();
            
            // 更新记录列表显示
            let text = '';
            records.forEach((r, i) => {
                text += `• a=${r.a}, L=${r.L}`;
                if (i < records.length - 1) text += '\n';
            });
            recordListEl.textContent = text;
        }
        
        // 按钮点击效果
        function handleRecordClick() {
            addRecord();
            // 按钮按下效果
            btnRecordBg.setAttribute('fill', '#0d47a1');
            setTimeout(() => {
                btnRecordBg.setAttribute('fill', '#1565c0');
            }, 150);
        }
        
        // 初始化
        function init() {
            createGroundHatch();
            updateScene();
            
            // 事件监听
            svg.addEventListener('mousedown', handleStart);
            svg.addEventListener('mousemove', handleMove);
            svg.addEventListener('mouseup', handleEnd);
            svg.addEventListener('mouseleave', handleEnd);
            
            svg.addEventListener('touchstart', handleStart, { passive: false });
            svg.addEventListener('touchmove', handleMove, { passive: false });
            svg.addEventListener('touchend', handleEnd);
            
            // 记录按钮点击
            btnRecordGroup.addEventListener('click', handleRecordClick);
        }
        
        init();
    })();
    </script>
