    <script>
    (function() {
        // Scene 1: 人体血流模型 - 甲→乙→丙结构示意
        
        // 状态定义：0=初始, 1=高亮甲, 2=高亮乙, 3=高亮丙, 4=显示血流方向
        let currentState = 0;
        const totalSteps = 5;
        
        // 获取 DOM 元素
        const highlightA = document.getElementById('scene1-highlight-a');
        const highlightB = document.getElementById('scene1-highlight-b');
        const highlightC = document.getElementById('scene1-highlight-c');
        const flowLabel = document.getElementById('scene1-flow-label');
        const btnPrev = document.getElementById('scene1-btn-prev');
        const btnNext = document.getElementById('scene1-btn-next');
        const stepIndicator = document.getElementById('scene1-step-indicator');
        
        // 隐藏所有高亮层和标注
        function hideAllHighlights() {
            highlightA.classList.remove('visible');
            highlightB.classList.remove('visible');
            highlightC.classList.remove('visible');
            flowLabel.classList.remove('visible');
        }
        
        // 更新状态显示
        function updateState() {
            hideAllHighlights();
            
            switch (currentState) {
                case 0:
                    // 初始状态：只显示基础结构
                    break;
                case 1:
                    // 步骤 1：高亮甲
                    highlightA.classList.add('visible');
                    break;
                case 2:
                    // 步骤 2：高亮乙
                    highlightB.classList.add('visible');
                    break;
                case 3:
                    // 步骤 3：高亮丙
                    highlightC.classList.add('visible');
                    break;
                case 4:
                    // 步骤 4：显示血流方向
                    flowLabel.classList.add('visible');
                    break;
            }
            
            // 更新步骤指示器
            stepIndicator.textContent = '步骤 ' + (currentState + 1) + '/' + totalSteps;
            
            // 更新按钮状态
            btnPrev.disabled = (currentState === 0);
            btnNext.disabled = (currentState === totalSteps - 1);
        }
        
        // 上一步按钮点击
        btnPrev.addEventListener('click', function() {
            if (currentState > 0) {
                currentState--;
                updateState();
            }
        });
        
        // 下一步按钮点击
        btnNext.addEventListener('click', function() {
            if (currentState < totalSteps - 1) {
                currentState++;
                updateState();
            }
        });
        
        // 初始化
        updateState();
    })();
    </script>
