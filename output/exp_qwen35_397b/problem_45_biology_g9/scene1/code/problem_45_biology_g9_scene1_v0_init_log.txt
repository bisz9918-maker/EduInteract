    <script>
    (function() {
        // Scene 1 实现：血液流经器官示意图
        
        // 状态变量
        let currentState = 0;
        const totalSteps = 5;
        
        // DOM 元素引用
        const btnPrev = document.getElementById('scene1-btn-prev');
        const btnNext = document.getElementById('scene1-btn-next');
        const stepIndicator = document.getElementById('scene1-step-indicator');
        
        // 可显隐的元素组
        const annotationA = document.getElementById('scene1-annotation-a');
        const annotationB = document.getElementById('scene1-annotation-b');
        const annotationC = document.getElementById('scene1-annotation-c');
        const arrowA = document.getElementById('scene1-arrow-a');
        const arrowC = document.getElementById('scene1-arrow-c');
        
        // 标签元素
        const labelA = document.getElementById('scene1-label-a');
        const labelB = document.getElementById('scene1-label-b');
        const labelC = document.getElementById('scene1-label-c');
        
        // 工具函数：显示元素
        function show(el) {
            el.classList.remove('scene1-hidden');
            el.classList.add('scene1-visible');
            el.style.display = 'block';
            el.style.opacity = '1';
        }
        
        // 工具函数：隐藏元素
        function hide(el) {
            el.classList.remove('scene1-visible');
            el.classList.add('scene1-hidden');
            el.style.display = 'none';
            el.style.opacity = '0';
        }
        
        // 更新状态显示
        function updateState() {
            // 重置所有标注和箭头
            hide(annotationA);
            hide(annotationB);
            hide(annotationC);
            hide(arrowA);
            hide(arrowC);
            
            // 重置标签颜色
            labelA.setAttribute('fill', '#000000');
            labelB.setAttribute('fill', '#000000');
            labelC.setAttribute('fill', '#000000');
            
            // 根据当前状态显示相应元素
            if (currentState >= 1) {
                show(annotationA);
                labelA.setAttribute('fill', '#1565c0');
            }
            if (currentState >= 2) {
                show(annotationB);
                labelB.setAttribute('fill', '#1565c0');
            }
            if (currentState >= 3) {
                show(annotationC);
                labelC.setAttribute('fill', '#1565c0');
            }
            if (currentState >= 4) {
                show(arrowA);
                show(arrowC);
            }
            
            // 更新步骤指示器
            stepIndicator.textContent = `步骤 ${currentState + 1}/${totalSteps}`;
            
            // 更新按钮状态
            btnPrev.disabled = (currentState === 0);
            btnNext.disabled = (currentState === totalSteps - 1);
        }
        
        // 事件处理：上一步
        btnPrev.addEventListener('click', function() {
            if (currentState > 0) {
                currentState--;
                updateState();
            }
        });
        
        // 事件处理：下一步
        btnNext.addEventListener('click', function() {
            if (currentState < totalSteps - 1) {
                currentState++;
                updateState();
            }
        });
        
        // 初始化
        updateState();
    })();
    </script>
