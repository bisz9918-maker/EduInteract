    <script>
    (function() {
        // Scene 3: 组织细胞处血液气体交换 - 正确答案验证

        // 状态变量
        let currentStep = 0;
        const totalSteps = 5;

        // 获取 DOM 元素
        const vesselA = document.getElementById('scene3-vessel-a');
        const vesselC = document.getElementById('scene3-vessel-c');
        const o2Arrow = document.getElementById('scene3-o2-arrow');
        const o2Label = document.getElementById('scene3-label-o2');
        const co2Arrow = document.getElementById('scene3-co2-arrow');
        const co2Label = document.getElementById('scene3-label-co2');
        const arterialLabel = document.getElementById('scene3-label-arterial');
        const venousLabel = document.getElementById('scene3-label-venous');
        const conclusionText = document.getElementById('scene3-conclusion-text');
        const conclusionCheck = document.getElementById('scene3-conclusion-check');
        const btnPrev = document.getElementById('scene3-btn-prev');
        const btnNext = document.getElementById('scene3-btn-next');
        const btnNextText = document.getElementById('scene3-btn-next-text');
        const stepIndicator = document.getElementById('scene3-step-indicator');

        // 工具函数：显示元素
        function show(el) {
            el.classList.remove('scene3-hidden');
            el.classList.add('scene3-visible');
        }

        // 工具函数：隐藏元素
        function hide(el) {
            el.classList.remove('scene3-visible');
            el.classList.add('scene3-hidden');
        }

        // 更新步骤状态
        function updateStep() {
            // 更新步骤指示器
            stepIndicator.textContent = '步骤 ' + (currentStep + 1) + '/' + totalSteps;

            // 步骤 0：初始状态 - 只显示基础图示
            if (currentStep === 0) {
                hide(arterialLabel);
                hide(venousLabel);
                hide(o2Arrow);
                hide(o2Label);
                hide(co2Arrow);
                hide(co2Label);
                hide(conclusionText);
                hide(conclusionCheck);
                vesselA.setAttribute('fill', 'none');
                vesselC.setAttribute('fill', 'none');
                btnPrev.classList.add('scene3-btn-disabled');
                btnNext.classList.remove('scene3-btn-disabled');
                btnNextText.textContent = '下一步 ▶';
            }
            // 步骤 1：高亮 a 为动脉血
            else if (currentStep === 1) {
                show(arterialLabel);
                hide(venousLabel);
                hide(o2Arrow);
                hide(o2Label);
                hide(co2Arrow);
                hide(co2Label);
                hide(conclusionText);
                hide(conclusionCheck);
                vesselA.setAttribute('fill', 'rgba(198, 40, 40, 0.3)');
                vesselC.setAttribute('fill', 'none');
                btnPrev.classList.remove('scene3-btn-disabled');
                btnNext.classList.remove('scene3-btn-disabled');
                btnNextText.textContent = '下一步 ▶';
            }
            // 步骤 2：展示气体交换
            else if (currentStep === 2) {
                show(arterialLabel);
                hide(venousLabel);
                show(o2Arrow);
                show(o2Label);
                show(co2Arrow);
                show(co2Label);
                hide(conclusionText);
                hide(conclusionCheck);
                vesselA.setAttribute('fill', 'rgba(198, 40, 40, 0.3)');
                vesselC.setAttribute('fill', 'none');
                btnPrev.classList.remove('scene3-btn-disabled');
                btnNext.classList.remove('scene3-btn-disabled');
                btnNextText.textContent = '下一步 ▶';
            }
            // 步骤 3：高亮 c 为静脉血
            else if (currentStep === 3) {
                show(arterialLabel);
                show(venousLabel);
                show(o2Arrow);
                show(o2Label);
                show(co2Arrow);
                show(co2Label);
                hide(conclusionText);
                hide(conclusionCheck);
                vesselA.setAttribute('fill', 'rgba(198, 40, 40, 0.3)');
                vesselC.setAttribute('fill', 'rgba(21, 101, 192, 0.3)');
                btnPrev.classList.remove('scene3-btn-disabled');
                btnNext.classList.remove('scene3-btn-disabled');
                btnNextText.textContent = '下一步 ▶';
            }
            // 步骤 4：显示结论
            else if (currentStep === 4) {
                show(arterialLabel);
                show(venousLabel);
                show(o2Arrow);
                show(o2Label);
                show(co2Arrow);
                show(co2Label);
                show(conclusionText);
                show(conclusionCheck);
                vesselA.setAttribute('fill', 'rgba(198, 40, 40, 0.3)');
                vesselC.setAttribute('fill', 'rgba(21, 101, 192, 0.3)');
                btnPrev.classList.remove('scene3-btn-disabled');
                btnNext.classList.add('scene3-btn-disabled');
                btnNextText.textContent = '完成';
            }
        }

        // 按钮点击事件
        btnPrev.addEventListener('click', function() {
            if (currentStep > 0) {
                currentStep--;
                updateStep();
            }
        });

        btnNext.addEventListener('click', function() {
            if (currentStep < totalSteps - 1) {
                currentStep++;
                updateStep();
            }
        });

        // 键盘事件支持
        document.addEventListener('keydown', function(e) {
            if (e.key === 'ArrowLeft' && currentStep > 0) {
                currentStep--;
                updateStep();
            } else if (e.key === 'ArrowRight' && currentStep < totalSteps - 1) {
                currentStep++;
                updateStep();
            }
        });

        // 初始化
        updateStep();
    })();
    </script>
