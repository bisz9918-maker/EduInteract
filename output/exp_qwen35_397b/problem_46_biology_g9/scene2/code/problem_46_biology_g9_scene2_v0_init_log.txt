    <script>
    (function() {
        // Scene 2: DNA 半保留复制 - 同位素标记传递

        // 布局坐标表（实现对照依据）：
        // | 元素 | 类型 | 位置/范围 | 大小 | 备注 |
        // |------|------|-----------|------|------|
        // | 亲代 DNA | 双链矩形 | (80,40)-(200,60) | 120×20 | 两条深蓝色链 |
        // | 亲代标签 | text | (140,25) | ~50×14 | 上方 |
        // | 第 1 代 DNA-左 | 双链矩形 | (40,130)-(120,150) | 80×20 | 上链深蓝下链浅绿 |
        // | 第 1 代 DNA-右 | 双链矩形 | (160,130)-(240,150) | 80×20 | 上链浅绿下链深蓝 |
        // | 第 1 代标签 | text | (140,115) | ~50×14 | 上方 |
        // | 第 2 代 DNA-1~4 | 双链矩形 | (10,240)-(280,260) | 60×20 each | 4 个 DNA |
        // | 第 2 代标签 | text | (140,225) | ~50×14 | 上方 |
        // | 第 3 代 DNA-1~8 | 双链矩形 | (5,360)-(320,380) | 35×20 each | 8 个 DNA |
        // | 第 3 代标签 | text | (140,345) | ~50×14 | 上方 |
        // | 统计框 | 矩形 | (520,100)-(780,280) | 260×180 | 右侧统计区域 |
        // | 图例框 | 矩形 | (520,300)-(780,380) | 260×80 | 右下方图例 |
        // | 按钮组 | 按钮 | (280,450)-(520,485) | 240×35 | 上一步/下一步 |

        let currentStep = 0;
        const totalSteps = 4;

        // 每步需要显示的元素
        const stepElements = {
            0: ['scene2-parent-group'],
            1: ['scene2-parent-group', 'scene2-gen1-group', 'scene2-conn-p-g1-l', 'scene2-conn-p-g1-r'],
            2: ['scene2-parent-group', 'scene2-gen1-group', 'scene2-gen2-group', 
                'scene2-conn-p-g1-l', 'scene2-conn-p-g1-r',
                'scene2-conn-g1-g2-1', 'scene2-conn-g1-g2-2', 'scene2-conn-g1-g2-3', 'scene2-conn-g1-g2-4'],
            3: ['scene2-parent-group', 'scene2-gen1-group', 'scene2-gen2-group', 'scene2-gen3-group',
                'scene2-conn-p-g1-l', 'scene2-conn-p-g1-r',
                'scene2-conn-g1-g2-1', 'scene2-conn-g1-g2-2', 'scene2-conn-g1-g2-3', 'scene2-conn-g1-g2-4',
                'scene2-conn-g2-g3-1', 'scene2-conn-g2-g3-2', 'scene2-conn-g2-g3-3', 'scene2-conn-g2-g3-4',
                'scene2-conn-g2-g3-5', 'scene2-conn-g2-g3-6', 'scene2-conn-g2-g3-7', 'scene2-conn-g2-g3-8'],
            4: ['scene2-parent-group', 'scene2-gen1-group', 'scene2-gen2-group', 'scene2-gen3-group', 'scene2-stats-group',
                'scene2-conn-p-g1-l', 'scene2-conn-p-g1-r',
                'scene2-conn-g1-g2-1', 'scene2-conn-g1-g2-2', 'scene2-conn-g1-g2-3', 'scene2-conn-g1-g2-4',
                'scene2-conn-g2-g3-1', 'scene2-conn-g2-g3-2', 'scene2-conn-g2-g3-3', 'scene2-conn-g2-g3-4',
                'scene2-conn-g2-g3-5', 'scene2-conn-g2-g3-6', 'scene2-conn-g2-g3-7', 'scene2-conn-g2-g3-8']
        };

        // 工具函数：显示元素
        function showElement(elementId) {
            const el = document.getElementById(elementId);
            if (el) {
                el.style.display = '';
                el.style.opacity = '1';
                if (el.classList.contains('scene2-fade-in')) {
                    el.classList.remove('scene2-fade-in');
                    void el.offsetWidth; // 触发重绘
                    el.classList.add('scene2-fade-in');
                }
            }
        }

        // 工具函数：隐藏元素
        function hideElement(elementId) {
            const el = document.getElementById(elementId);
            if (el) {
                el.style.display = 'none';
                el.style.opacity = '0';
            }
        }

        // 更新步骤显示
        function updateStepDisplay() {
            // 隐藏所有可能显示的元素
            const allElements = [
                'scene2-parent-group', 'scene2-gen1-group', 'scene2-gen2-group', 
                'scene2-gen3-group', 'scene2-stats-group',
                'scene2-conn-p-g1-l', 'scene2-conn-p-g1-r',
                'scene2-conn-g1-g2-1', 'scene2-conn-g1-g2-2', 'scene2-conn-g1-g2-3', 'scene2-conn-g1-g2-4',
                'scene2-conn-g2-g3-1', 'scene2-conn-g2-g3-2', 'scene2-conn-g2-g3-3', 'scene2-conn-g2-g3-4',
                'scene2-conn-g2-g3-5', 'scene2-conn-g2-g3-6', 'scene2-conn-g2-g3-7', 'scene2-conn-g2-g3-8'
            ];
            allElements.forEach(id => hideElement(id));

            // 显示当前步骤需要的元素
            const elementsToShow = stepElements[currentStep];
            elementsToShow.forEach(id => showElement(id));

            // 更新步骤指示器
            const indicator = document.getElementById('scene2-step-indicator');
            if (indicator) {
                indicator.textContent = `步骤 ${currentStep}/${totalSteps}`;
            }

            // 更新按钮状态
            const prevBtn = document.getElementById('scene2-btn-prev');
            const nextBtn = document.getElementById('scene2-btn-next');
            if (prevBtn) {
                prevBtn.disabled = currentStep === 0;
            }
            if (nextBtn) {
                nextBtn.disabled = currentStep === totalSteps;
                if (currentStep === totalSteps) {
                    nextBtn.textContent = '已完成';
                } else {
                    nextBtn.textContent = '下一步 ▶';
                }
            }
        }

        // 事件处理：上一步
        document.getElementById('scene2-btn-prev').addEventListener('click', function() {
            if (currentStep > 0) {
                currentStep--;
                updateStepDisplay();
            }
        });

        // 事件处理：下一步
        document.getElementById('scene2-btn-next').addEventListener('click', function() {
            if (currentStep < totalSteps) {
                currentStep++;
                updateStepDisplay();
            }
        });

        // 初始化显示
        updateStepDisplay();

    })();
    </script>
