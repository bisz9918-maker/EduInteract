    <script>
    (function() {
        // Scene 1: 大圆航线图示

        let currentStep = 0;
        const totalSteps = 4;

        // 元素引用
        const btnPrev = document.getElementById('scene1-btn-prev');
        const btnNext = document.getElementById('scene1-btn-next');
        const stepIndicator = document.getElementById('scene1-step-indicator');

        // 各步骤需要显示的元素
        const stepElements = {
            0: [], // 初始状态：只显示基础经纬网和点
            1: ['scene1-path-latitude'], // 步骤1：显示纬线路径
            2: ['scene1-path-greatcircle'], // 步骤2：显示大圆航线
            3: ['scene1-arrow-1', 'scene1-arrow-2', 'scene1-direction-label'] // 步骤3：显示方向箭头
        };

        // 显示元素
        function showElement(id) {
            const el = document.getElementById(id);
            if (el) {
                el.style.opacity = '1';
                el.style.display = 'block';
            }
        }

        // 隐藏元素
        function hideElement(id) {
            const el = document.getElementById(id);
            if (el) {
                el.style.opacity = '0';
                el.style.display = 'none';
            }
        }

        // 更新显示状态
        function updateDisplay() {
            // 隐藏所有步骤元素
            for (let step = 1; step < totalSteps; step++) {
                stepElements[step].forEach(id => hideElement(id));
            }

            // 显示当前步骤及之前步骤的元素
            for (let step = 1; step <= currentStep; step++) {
                // 步骤2时隐藏步骤1的纬线路径
                if (step === 1 && currentStep >= 2) {
                    hideElement('scene1-path-latitude');
                } else {
                    stepElements[step].forEach(id => showElement(id));
                }
            }

            // 更新按钮状态
            btnPrev.disabled = currentStep === 0;
            btnNext.disabled = currentStep === totalSteps - 1;

            // 更新步骤指示器
            stepIndicator.textContent = `步骤 ${currentStep + 1}/${totalSteps}`;
        }

        // 上一步
        btnPrev.addEventListener('click', function() {
            if (currentStep > 0) {
                currentStep--;
                updateDisplay();
            }
        });

        // 下一步
        btnNext.addEventListener('click', function() {
            if (currentStep < totalSteps - 1) {
                currentStep++;
                updateDisplay();
            }
        });

        // 初始化
        updateDisplay();
    })();
    </script>
