    <script>
    (function() {
        // Scene 4: 正午太阳高度角与地方时计算演示
        // 布局坐标表:
        // | 元素 | 位置/范围 | 备注 |
        // |------|-----------|------|
        // | 地球截面圆 | 圆心 (180,200), r=100 | 左侧主图 |
        // | 地心 O | (180,200) | 标注"O" |
        // | 赤道线 | (80,200)-(280,200) | 水平虚线 |
        // | 当地纬度点 P | (230,150) 对应 30.5°N | 动态 |
        // | 直射点 S | (200,180) 对应 20°N | 固定 |
        // | 经度数轴 | (450,200)-(750,200) | 右侧主图 |
        // | 纬度滑块 | (100,440)-(350,455) | 范围 -90° 到 90° |
        // | 经度滑块 | (450,440)-(700,455) | 范围 -180° 到 180° |
        // | 公式区 | (400,50)-(780,100) | H = 90° - |φ - δ| |
        // | 结果区 | (400,110)-(780,160) | 显示计算结果 |
        
        const state = {
            latitude: 30.5,      // 当地纬度，范围 -90 到 90
            longitude: 114.5,    // 当地经度，范围 -180 到 180
            directLatitude: 20   // 太阳直射点纬度，固定 20°N
        };
        
        // 滑块配置
        const latitudeSlider = {
            trackX: 100, trackY: 447, trackWidth: 250,
            min: -90, max: 90
        };
        
        const longitudeSlider = {
            trackX: 450, trackY: 447, trackWidth: 250,
            min: -180, max: 180
        };
        
        // DOM 元素
        const pointP = document.getElementById('scene4-point-p');
        const labelP = document.getElementById('scene4-label-p');
        const pointLongitude = document.getElementById('scene4-point-longitude');
        const labelLongitude = document.getElementById('scene4-label-longitude');
        const handleLatitude = document.getElementById('scene4-handle-latitude');
        const handleLongitude = document.getElementById('scene4-handle-longitude');
        const latitudeValue = document.getElementById('scene4-latitude-value');
        const longitudeValue = document.getElementById('scene4-longitude-value');
        const resultDisplay = document.getElementById('scene4-result-display');
        const timeDisplay = document.getElementById('scene4-time-display');
        const angleArc = document.getElementById('scene4-angle-arc');
        const labelH = document.getElementById('scene4-label-h');
        
        // 坐标计算函数
        function latitudeToY(lat) {
            // 纬度转换为地球截面圆上的 Y 坐标
            // 90°N -> y=100, 0° -> y=200, 90°S -> y=300
            return 200 - Math.sin(lat * Math.PI / 180) * 100;
        }
        
        function latitudeToX(lat) {
            // 纬度转换为地球截面圆上的 X 坐标（假设在右侧经线上）
            return 180 + Math.cos(lat * Math.PI / 180) * 100;
        }
        
        function longitudeToX(lon) {
            // 经度转换为数轴上的 X 坐标
            // -180° -> x=450, 0° -> x=600, 180° -> x=750
            return 450 + (lon + 180) / 360 * 300;
        }
        
        function latToSliderX(lat) {
            // 纬度转换为滑块手柄 X 坐标
            return latitudeSlider.trackX + (lat - latitudeSlider.min) / (latitudeSlider.max - latitudeSlider.min) * latitudeSlider.trackWidth;
        }
        
        function sliderXToLat(x) {
            // 滑块手柄 X 坐标转换为纬度
            const clampedX = Math.max(latitudeSlider.trackX, Math.min(latitudeSlider.trackX + latitudeSlider.trackWidth, x));
            return latitudeSlider.min + (clampedX - latitudeSlider.trackX) / latitudeSlider.trackWidth * (latitudeSlider.max - latitudeSlider.min);
        }
        
        function lonToSliderX(lon) {
            // 经度转换为滑块手柄 X 坐标
            return longitudeSlider.trackX + (lon - longitudeSlider.min) / (longitudeSlider.max - longitudeSlider.min) * longitudeSlider.trackWidth;
        }
        
        function sliderXToLon(x) {
            // 滑块手柄 X 坐标转换为经度
            const clampedX = Math.max(longitudeSlider.trackX, Math.min(longitudeSlider.trackX + longitudeSlider.trackWidth, x));
            return longitudeSlider.min + (clampedX - longitudeSlider.trackX) / longitudeSlider.trackWidth * (longitudeSlider.max - longitudeSlider.min);
        }
        
        // 计算正午太阳高度角 H = 90° - |φ - δ|
        function calculateSolarAltitude(lat, directLat) {
            return 90 - Math.abs(lat - directLat);
        }
        
        // 计算地方时（基于经度差，15°=1 小时）
        function calculateLocalTime(lon) {
            // 以 0°经线为基准（12 时为正午）
            // 每 15°经度差 = 1 小时时间差
            const timeDiff = lon / 15;
            let localTime = 12 + timeDiff;
            // 规范化到 0-24 范围
            while (localTime < 0) localTime += 24;
            while (localTime >= 24) localTime -= 24;
            return localTime;
        }
        
        // 格式化纬度显示
        function formatLatitude(lat) {
            if (lat > 0) return lat.toFixed(1) + '°N';
            if (lat < 0) return Math.abs(lat).toFixed(1) + '°S';
            return '0°';
        }
        
        // 格式化经度显示
        function formatLongitude(lon) {
            if (lon > 0) return lon.toFixed(1) + '°E';
            if (lon < 0) return Math.abs(lon).toFixed(1) + '°W';
            return '0°';
        }
        
        // 格式化时间显示
        function formatTime(hours) {
            const h = Math.floor(hours);
            const m = Math.round((hours - h) * 60);
            if (m === 0) return h + '时';
            return h + '时' + m + '分';
        }
        
        // 更新显示
        function updateDisplay() {
            // 更新地球截面图上的 P 点位置
            const pX = latitudeToX(state.latitude);
            const pY = latitudeToY(state.latitude);
            pointP.setAttribute('cx', pX);
            pointP.setAttribute('cy', pY);
            labelP.setAttribute('x', pX + 10);
            labelP.setAttribute('y', pY - 5);
            
            // 更新经度数轴上的标记点
            const lonX = longitudeToX(state.longitude);
            pointLongitude.setAttribute('cx', lonX);
            labelLongitude.setAttribute('x', lonX);
            labelLongitude.textContent = formatLongitude(state.longitude);
            
            // 更新滑块手柄位置
            handleLatitude.setAttribute('cx', latToSliderX(state.latitude));
            handleLongitude.setAttribute('cx', lonToSliderX(state.longitude));
            
            // 更新数值标签
            latitudeValue.textContent = formatLatitude(state.latitude);
            longitudeValue.textContent = formatLongitude(state.longitude);
            
            // 计算并更新结果
            const H = calculateSolarAltitude(state.latitude, state.directLatitude);
            const localTime = calculateLocalTime(state.longitude);
            
            resultDisplay.textContent = '正午太阳高度角：' + H.toFixed(1) + '°';
            
            if (H < 0) {
                timeDisplay.textContent = '地方时：' + formatTime(localTime) + '（极夜，无日照）';
            } else {
                timeDisplay.textContent = '地方时：' + formatTime(localTime);
            }
            
            // 更新太阳高度角弧线
            updateAngleArc(pX, pY, H);
        }
        
        // 更新角度弧线
        function updateAngleArc(pX, pY, H) {
            // 简化：弧线位置相对固定，只更新标签
            labelH.textContent = 'H=' + H.toFixed(0) + '°';
        }
        
        // 拖拽处理
        let draggingLatitude = false;
        let draggingLongitude = false;
        
        function handleMouseDown(e) {
            const rect = document.getElementById('scene4-svg').getBoundingClientRect();
            const x = e.clientX - rect.left;
            const y = e.clientY - rect.top;
            
            // 检查是否点击纬度滑块手柄
            const latHandleX = parseFloat(handleLatitude.getAttribute('cx'));
            const latHandleY = parseFloat(handleLatitude.getAttribute('cy'));
            if (Math.sqrt((x - latHandleX) ** 2 + (y - latHandleY) ** 2) < 20) {
                draggingLatitude = true;
                return;
            }
            
            // 检查是否点击经度滑块手柄
            const lonHandleX = parseFloat(handleLongitude.getAttribute('cx'));
            const lonHandleY = parseFloat(handleLongitude.getAttribute('cy'));
            if (Math.sqrt((x - lonHandleX) ** 2 + (y - lonHandleY) ** 2) < 20) {
                draggingLongitude = true;
                return;
            }
        }
        
        function handleMouseMove(e) {
            if (!draggingLatitude && !draggingLongitude) return;
            
            const rect = document.getElementById('scene4-svg').getBoundingClientRect();
            const x = e.clientX - rect.left;
            
            if (draggingLatitude) {
                state.latitude = sliderXToLat(x);
            } else if (draggingLongitude) {
                state.longitude = sliderXToLon(x);
            }
            
            updateDisplay();
        }
        
        function handleMouseUp() {
            draggingLatitude = false;
            draggingLongitude = false;
        }
        
        // 触摸事件支持
        function handleTouchStart(e) {
            e.preventDefault();
            const touch = e.touches[0];
            const rect = document.getElementById('scene4-svg').getBoundingClientRect();
            const x = touch.clientX - rect.left;
            const y = touch.clientY - rect.top;
            
            const latHandleX = parseFloat(handleLatitude.getAttribute('cx'));
            const latHandleY = parseFloat(handleLatitude.getAttribute('cy'));
            if (Math.sqrt((x - latHandleX) ** 2 + (y - latHandleY) ** 2) < 25) {
                draggingLatitude = true;
                return;
            }
            
            const lonHandleX = parseFloat(handleLongitude.getAttribute('cx'));
            const lonHandleY = parseFloat(handleLongitude.getAttribute('cy'));
            if (Math.sqrt((x - lonHandleX) ** 2 + (y - lonHandleY) ** 2) < 25) {
                draggingLongitude = true;
            }
        }
        
        function handleTouchMove(e) {
            if (!draggingLatitude && !draggingLongitude) return;
            e.preventDefault();
            
            const touch = e.touches[0];
            const rect = document.getElementById('scene4-svg').getBoundingClientRect();
            const x = touch.clientX - rect.left;
            
            if (draggingLatitude) {
                state.latitude = sliderXToLat(x);
            } else if (draggingLongitude) {
                state.longitude = sliderXToLon(x);
            }
            
            updateDisplay();
        }
        
        function handleTouchEnd() {
            draggingLatitude = false;
            draggingLongitude = false;
        }
        
        // 事件绑定
        const svg = document.getElementById('scene4-svg');
        svg.addEventListener('mousedown', handleMouseDown);
        svg.addEventListener('mousemove', handleMouseMove);
        svg.addEventListener('mouseup', handleMouseUp);
        svg.addEventListener('mouseleave', handleMouseUp);
        
        svg.addEventListener('touchstart', handleTouchStart, { passive: false });
        svg.addEventListener('touchmove', handleTouchMove, { passive: false });
        svg.addEventListener('touchend', handleTouchEnd);
        
        // 初始化显示
        updateDisplay();
    })();
    </script>
