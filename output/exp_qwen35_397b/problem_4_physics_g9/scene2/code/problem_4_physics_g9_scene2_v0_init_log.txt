    <script>
    (function() {
        // Scene 2 实现 - 电路修改方案对比
        
        const currentState = { value: 0 };
        const wireB = document.getElementById('scene2-wire-b');
        const wireM = document.getElementById('scene2-wire-M');
        const bulbL1Glow = document.getElementById('scene2-bulb-L1-glow');
        const bulbL2Glow = document.getElementById('scene2-bulb-L2-glow');
        const currentArrows = document.getElementById('scene2-current-arrows');
        const statusLabel = document.getElementById('scene2-status-label');
        const btnOriginal = document.getElementById('scene2-btn-original');
        const btnSeries = document.getElementById('scene2-btn-series');
        const btnParallel = document.getElementById('scene2-btn-parallel');

        function updateState(state) {
            currentState.value = state;
            
            // 重置所有状态
            wireB.setAttribute('stroke', '#000000');
            wireB.setAttribute('stroke-width', '2');
            wireB.removeAttribute('stroke-dasharray');
            wireM.setAttribute('d', 'M 180 310 Q 350 300 620 250');
            bulbL1Glow.setAttribute('fill', 'none');
            bulbL2Glow.setAttribute('fill', 'none');
            currentArrows.style.opacity = '0';
            
            // 按钮高亮
            btnOriginal.setAttribute('fill', '#1565c0');
            btnSeries.setAttribute('fill', '#1565c0');
            btnParallel.setAttribute('fill', '#1565c0');
            
            if (state === 0) {
                // 原电路
                statusLabel.textContent = '原电路（短路）';
                btnOriginal.setAttribute('fill', '#0d47a1');
            } else if (state === 1) {
                // 方案 C - 串联
                wireB.setAttribute('stroke', '#888888');
                wireB.setAttribute('stroke-dasharray', '5,5');
                bulbL1Glow.setAttribute('fill', '#c8e6c9');
                bulbL2Glow.setAttribute('fill', '#c8e6c9');
                currentArrows.style.opacity = '1';
                statusLabel.textContent = '串联电路';
                btnSeries.setAttribute('fill', '#0d47a1');
            } else if (state === 2) {
                // 方案 D - 并联
                wireM.setAttribute('d', 'M 180 310 Q 350 300 520 250');
                bulbL1Glow.setAttribute('fill', '#c8e6c9');
                bulbL2Glow.setAttribute('fill', '#c8e6c9');
                currentArrows.style.opacity = '1';
                statusLabel.textContent = '并联电路';
                btnParallel.setAttribute('fill', '#0d47a1');
            }
        }

        // 按钮事件绑定
        btnOriginal.addEventListener('click', () => updateState(0));
        btnSeries.addEventListener('click', () => updateState(1));
        btnParallel.addEventListener('click', () => updateState(2));

        // 初始状态
        updateState(0);
    })();
    </script>
