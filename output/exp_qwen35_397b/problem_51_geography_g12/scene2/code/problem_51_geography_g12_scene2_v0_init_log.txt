    <script>
    (function() {
        // Scene 2 实现：时间计算与日期范围示意图
        
        // 布局坐标表（实现对照依据）：
        // | 元素 | 类型 | 位置/范围 | 大小 | 标签偏移方向 | 备注 |
        // |------|------|-----------|------|-------------|------|
        // | 地球轮廓圆 | circle | 圆心 (400,250) | r=180 | - | 居中，黑色 2px 实线 |
        // | 0°经线 | line | (400,250)-(220,250) | 180×0 | 左侧外 | 红色#c62828 3px，0 时经线 |
        // | 90°E 经线 | line | (400,250)-(400,430) | 0×180 | 下方外 | 黑色 1px |
        // | 120°E 经线 | line | (400,250)-(310,404) | 斜线 | 左下方外 | 黑色 1px |
        // | 180°经线 | line | (400,250)-(580,250) | 180×0 | 右侧外 | 蓝色#1565c0 3px，日界线 |
        // | 新一天扇形 | path | 从 0°经线逆时针到 180°经线 | 270°扇形 | - | 绿色#2e7d32 透明度 20% |
        // | 0°经线标签 | text | (200,245) | ~60×16 | 左侧 | "0°(0 时)" |
        // | 90°E 标签 | text | (395,455) | ~50×16 | 下方 | "90°E(6 时)" |
        // | 120°E 标签 | text | (290,425) | ~60×16 | 左下方 | "120°E(8 时)" |
        // | 180°经线标签 | text | (590,245) | ~70×16 | 右侧 | "180°(日界线)" |
        // | 北极点 O | text | (400,235) | ~20×16 | 上方 | "O(北极)" |
        // | 北京时间框 | rect+text | (550,380) | 140×50 | - | 白色背景，黑色边框 |
        // | 新一天占比文本 | text | (400,100) | ~120×16 | 上方 | "新一天范围：3/4" |
        
        // 状态机：每步显示的元素 ID 列表
        const steps = [
            // 步骤 0：地球轮廓圆、0°经线、北极点
            ['scene2-circle-earth', 'scene2-line-0deg', 'scene2-label-pole', 'scene2-line-45deg', 'scene2-line-135deg'],
            // 步骤 1：显示 0°经线标签
            ['scene2-circle-earth', 'scene2-line-0deg', 'scene2-label-pole', 'scene2-line-45deg', 'scene2-line-135deg', 'scene2-label-0deg'],
            // 步骤 2：显示 90°E 和 120°E 经线及标签
            ['scene2-circle-earth', 'scene2-line-0deg', 'scene2-label-pole', 'scene2-line-45deg', 'scene2-line-135deg', 'scene2-label-0deg', 'scene2-line-90deg', 'scene2-line-120deg', 'scene2-label-90deg', 'scene2-label-120deg', 'scene2-arc-90deg', 'scene2-arc-30deg'],
            // 步骤 3：显示 180°经线及标签
            ['scene2-circle-earth', 'scene2-line-0deg', 'scene2-label-pole', 'scene2-line-45deg', 'scene2-line-135deg', 'scene2-label-0deg', 'scene2-line-90deg', 'scene2-line-120deg', 'scene2-label-90deg', 'scene2-label-120deg', 'scene2-arc-90deg', 'scene2-arc-30deg', 'scene2-line-180deg', 'scene2-label-180deg', 'scene2-arc-60deg'],
            // 步骤 4：显示新一天扇形、占比文本、北京时间框
            ['scene2-circle-earth', 'scene2-line-0deg', 'scene2-label-pole', 'scene2-line-45deg', 'scene2-line-135deg', 'scene2-label-0deg', 'scene2-line-90deg', 'scene2-line-120deg', 'scene2-label-90deg', 'scene2-label-120deg', 'scene2-arc-90deg', 'scene2-arc-30deg', 'scene2-line-180deg', 'scene2-label-180deg', 'scene2-arc-60deg', 'scene2-sector-newday', 'scene2-label-ratio', 'scene2-rect-beijing', 'scene2-label-beijing', 'scene2-label-beijing-time']
        ];
        
        let currentState = 0;
        
        // 工具函数：显示元素
        function showElement(id) {
            const el = document.getElementById(id);
            if (el) {
                el.style.display = '';
                if (el.classList.contains('fade-in')) {
                    requestAnimationFrame(() => {
                        el.classList.add('visible');
                    });
                }
            }
        }
        
        // 工具函数：隐藏元素
        function hideElement(id) {
            const el = document.getElementById(id);
            if (el) {
                if (el.classList.contains('fade-in')) {
                    el.classList.remove('visible');
                    setTimeout(() => {
                        el.style.display = 'none';
                    }, 300);
                } else {
                    el.style.display = 'none';
                }
            }
        }
        
        // 更新状态
        function updateState() {
            const currentIds = steps[currentState];
            const prevIds = currentState > 0 ? steps[currentState - 1] : [];
            const nextIds = currentState < steps.length - 1 ? steps[currentState + 1] : [];
            
            // 收集所有需要显示的元素 ID
            const allIds = new Set();
            steps.forEach(step => step.forEach(id => allIds.add(id)));
            
            // 更新每个元素的可见性
            allIds.forEach(id => {
                if (currentIds.includes(id)) {
                    showElement(id);
                } else {
                    hideElement(id);
                }
            });
            
            // 更新步骤指示器
            document.getElementById('scene2-step-indicator').textContent = `步骤 ${currentState + 1}/5`;
            
            // 更新按钮状态
            document.getElementById('scene2-btn-prev').disabled = currentState === 0;
            document.getElementById('scene2-btn-next').disabled = currentState === steps.length - 1;
        }
        
        // 按钮事件绑定
        document.getElementById('scene2-btn-prev').addEventListener('click', () => {
            if (currentState > 0) {
                currentState--;
                updateState();
            }
        });
        
        document.getElementById('scene2-btn-next').addEventListener('click', () => {
            if (currentState < steps.length - 1) {
                currentState++;
                updateState();
            }
        });
        
        // 初始化
        updateState();
    })();
    </script>
