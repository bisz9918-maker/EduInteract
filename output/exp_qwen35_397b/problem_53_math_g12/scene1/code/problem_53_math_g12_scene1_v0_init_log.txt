    <script>
    (function() {
        // Scene 1: 四棱锥中线面平行关系及比例推导
        // 布局坐标表 (实现对照依据):
        // | 元素 | 类型 | 位置/范围 | 大小 | 标签偏移方向 | 备注 |
        // |------|------|-----------|------|-------------|------|
        // | 顶点 P | 点 | (400, 70) | - | 上方 10px | 四棱锥顶点 |
        // | 顶点 A | 点 | (140, 360) | - | 左下 8px | 底面左前 |
        // | 顶点 B | 点 | (460, 390) | - | 右下 8px | 底面右前 |
        // | 顶点 C | 点 | (620, 300) | - | 右侧 10px | 底面右后 |
        // | 顶点 D | 点 | (240, 280) | - | 左下 8px | 底面左后（被遮挡） |
        // | 点 M | 圆点 | (514, 348) | r=4 #c62828 | 下方 10px | AC 上，CM:MA=1:4 |
        // | 点 N | 圆点 | (514, 118) | r=4 #c62828 | 上方 10px | PC 上 |

        let currentStep = 0;
        const totalSteps = 6;

        // 获取 DOM 元素
        const btnPrev = document.getElementById('scene1-btn-prev');
        const btnNext = document.getElementById('scene1-btn-next');
        const stepIndicator = document.getElementById('scene1-step-indicator');

        // 需要控制的元素
        const planePAC = document.getElementById('scene1-plane-pac');
        const planePAD = document.getElementById('scene1-plane-pad');
        const parallelSymbol = document.getElementById('scene1-parallel-symbol');
        const parallelText = document.getElementById('scene1-parallel-text');
        const ratioMark = document.getElementById('scene1-ratio-mark');
        const conclusion = document.getElementById('scene1-conclusion');
        const step2Hint = document.getElementById('scene1-step2-hint');

        // 更新显示状态
        function updateDisplay() {
            // 重置所有特殊元素
            planePAC.setAttribute('fill-opacity', '0');
            planePAD.setAttribute('fill-opacity', '0');
            parallelSymbol.setAttribute('opacity', '0');
            parallelText.setAttribute('opacity', '0');
            ratioMark.setAttribute('opacity', '0');
            conclusion.setAttribute('opacity', '0');
            step2Hint.setAttribute('opacity', '0');

            // 根据当前步骤显示相应元素
            switch (currentStep) {
                case 0:
                    // 初始状态：只显示基础图形
                    break;
                case 1:
                    // 步骤 1：高亮平面 PAC
                    planePAC.setAttribute('fill-opacity', '0.15');
                    break;
                case 2:
                    // 步骤 2：高亮平面 PAD，显示提示
                    planePAD.setAttribute('fill-opacity', '0.15');
                    step2Hint.setAttribute('opacity', '1');
                    break;
                case 3:
                    // 步骤 3：显示平行符号
                    planePAD.setAttribute('fill-opacity', '0.15');
                    step2Hint.setAttribute('opacity', '1');
                    parallelSymbol.setAttribute('opacity', '1');
                    parallelText.setAttribute('opacity', '1');
                    break;
                case 4:
                    // 步骤 4：显示比例标记
                    planePAD.setAttribute('fill-opacity', '0.15');
                    step2Hint.setAttribute('opacity', '1');
                    parallelSymbol.setAttribute('opacity', '1');
                    parallelText.setAttribute('opacity', '1');
                    ratioMark.setAttribute('opacity', '1');
                    break;
                case 5:
                    // 步骤 5：显示结论
                    planePAD.setAttribute('fill-opacity', '0.15');
                    step2Hint.setAttribute('opacity', '1');
                    parallelSymbol.setAttribute('opacity', '1');
                    parallelText.setAttribute('opacity', '1');
                    ratioMark.setAttribute('opacity', '1');
                    conclusion.setAttribute('opacity', '1');
                    break;
            }

            // 更新步骤指示器
            stepIndicator.textContent = `步骤 ${currentStep + 1}/${totalSteps}`;

            // 更新按钮状态
            btnPrev.disabled = currentStep === 0;
            btnNext.disabled = currentStep === totalSteps - 1;
        }

        // 按钮事件绑定
        btnPrev.addEventListener('click', function() {
            if (currentStep > 0) {
                currentStep--;
                updateDisplay();
            }
        });

        btnNext.addEventListener('click', function() {
            if (currentStep < totalSteps - 1) {
                currentStep++;
                updateDisplay();
            }
        });

        // 初始化显示
        updateDisplay();
    })();
    </script>
