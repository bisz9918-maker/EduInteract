    <script>
    (function() {
        // Scene 2 实现 - 抛物线几何问题
        
        const STATE = {
            currentStep: 0,
            totalSteps: 6
        };
        
        // 元素可见性配置（每步显示哪些元素）
        const STEP_CONFIG = {
            0: { // 初始状态
                visible: ['scene2-tri-ofa', 'scene2-line-af', 'scene2-ang-60-arc', 'scene2-ang-60', 'scene2-label-of']
            },
            1: { // 显示 AF 长度标注
                visible: ['scene2-tri-ofa', 'scene2-line-af', 'scene2-ang-60-arc', 'scene2-ang-60', 'scene2-label-of', 'scene2-label-af']
            },
            2: { // 显示 FB 和 AB 标注
                visible: ['scene2-tri-ofa', 'scene2-line-af', 'scene2-line-fb', 'scene2-ang-60-arc', 'scene2-ang-60', 'scene2-label-of', 'scene2-label-fb', 'scene2-label-af', 'scene2-label-ab']
            },
            3: { // 显示 A 点坐标和∠OFA
                visible: ['scene2-tri-ofa', 'scene2-line-af', 'scene2-line-fb', 'scene2-line-an', 'scene2-ang-60-arc', 'scene2-ang-60', 'scene2-ang-120-arc', 'scene2-ang-120', 'scene2-label-of', 'scene2-label-fb', 'scene2-label-af', 'scene2-label-ab', 'scene2-label-a-coord']
            },
            4: { // 显示三角形 ONB
                visible: ['scene2-tri-ofa', 'scene2-tri-onb', 'scene2-line-af', 'scene2-line-fb', 'scene2-line-an', 'scene2-line-on', 'scene2-line-nb', 'scene2-line-ob', 'scene2-ang-60-arc', 'scene2-ang-60', 'scene2-ang-120-arc', 'scene2-ang-120', 'scene2-label-of', 'scene2-label-fb', 'scene2-label-af', 'scene2-label-ab', 'scene2-label-a-coord']
            },
            5: { // 显示最终结论
                visible: ['scene2-tri-ofa', 'scene2-tri-onb', 'scene2-line-af', 'scene2-line-fb', 'scene2-line-an', 'scene2-line-on', 'scene2-line-nb', 'scene2-line-ob', 'scene2-ang-60-arc', 'scene2-ang-60', 'scene2-ang-120-arc', 'scene2-ang-120', 'scene2-ang-30-arc', 'scene2-ang-30', 'scene2-label-of', 'scene2-label-fb', 'scene2-label-af', 'scene2-label-ab', 'scene2-label-a-coord', 'scene2-conclusion-box', 'scene2-conclusion-text']
            }
        };
        
        function showElement(id) {
            const el = document.getElementById(id);
            if (el) {
                el.classList.remove('scene2-hidden');
                el.classList.add('scene2-visible');
            }
        }
        
        function hideElement(id) {
            const el = document.getElementById(id);
            if (el) {
                el.classList.remove('scene2-visible');
                el.classList.add('scene2-hidden');
            }
        }
        
        function updateStep(step) {
            STATE.currentStep = step;
            
            // 更新步骤指示器
            const indicator = document.getElementById('scene2-step-indicator-html');
            if (indicator) {
                indicator.textContent = `步骤 ${step + 1}/${STATE.totalSteps}`;
            }
            
            // 获取当前步骤配置
            const config = STEP_CONFIG[step];
            
            // 所有需要控制的元素
            const allElements = [
                'scene2-tri-ofa', 'scene2-tri-onb', 'scene2-line-af', 'scene2-line-fb',
                'scene2-line-an', 'scene2-line-on', 'scene2-line-nb', 'scene2-line-ob',
                'scene2-ang-60-arc', 'scene2-ang-60', 'scene2-ang-120-arc', 'scene2-ang-120',
                'scene2-ang-30-arc', 'scene2-ang-30', 'scene2-label-of', 'scene2-label-fb',
                'scene2-label-af', 'scene2-label-ab', 'scene2-label-a-coord',
                'scene2-conclusion-box', 'scene2-conclusion-text'
            ];
            
            // 先隐藏所有元素
            allElements.forEach(id => hideElement(id));
            
            // 显示当前步骤需要的元素
            if (config && config.visible) {
                config.visible.forEach(id => showElement(id));
            }
            
            // 更新按钮状态
            const btnPrev = document.getElementById('scene2-btn-prev');
            const btnNext = document.getElementById('scene2-btn-next');
            
            if (btnPrev) {
                btnPrev.disabled = (step === 0);
            }
            if (btnNext) {
                btnNext.disabled = (step === STATE.totalSteps - 1);
            }
        }
        
        // 初始化
        function init() {
            const btnPrev = document.getElementById('scene2-btn-prev');
            const btnNext = document.getElementById('scene2-btn-next');
            
            if (btnPrev) {
                btnPrev.addEventListener('click', () => {
                    if (STATE.currentStep > 0) {
                        updateStep(STATE.currentStep - 1);
                    }
                });
            }
            
            if (btnNext) {
                btnNext.addEventListener('click', () => {
                    if (STATE.currentStep < STATE.totalSteps - 1) {
                        updateStep(STATE.currentStep + 1);
                    }
                });
            }
            
            // 初始化显示步骤 0
            updateStep(0);
        }
        
        // DOM 加载完成后初始化
        if (document.readyState === 'loading') {
            document.addEventListener('DOMContentLoaded', init);
        } else {
            init();
        }
    })();
    </script>
