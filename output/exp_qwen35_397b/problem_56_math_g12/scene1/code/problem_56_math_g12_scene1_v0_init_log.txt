    <script>
    (function() {
        // Scene 1 实现：圆与正六边形嵌套构造
        
        // 状态管理
        let currentStep = 0;
        const totalSteps = 5;
        
        // 每步可见元素配置
        const stepConfig = [
            // 步骤0: 初始状态 - 圆0、六边形0、半径R₀、标签R₀=1
            ['scene1-circle0', 'scene1-hex0', 'scene1-radius-r0', 'scene1-label-r0'],
            // 步骤1: 显示边心距R₁、30°角弧、直角符号、30°标签
            ['scene1-circle0', 'scene1-hex0', 'scene1-radius-r0', 'scene1-label-r0',
             'scene1-apothem-r1', 'scene1-angle-arc', 'scene1-right-angle', 'scene1-label-angle'],
            // 步骤2: 显示圆1、R₁标签
            ['scene1-circle0', 'scene1-hex0', 'scene1-radius-r0', 'scene1-label-r0',
             'scene1-apothem-r1', 'scene1-angle-arc', 'scene1-right-angle', 'scene1-label-angle',
             'scene1-circle1', 'scene1-label-r1'],
            // 步骤3: 显示六边形1、圆2
            ['scene1-circle0', 'scene1-hex0', 'scene1-radius-r0', 'scene1-label-r0',
             'scene1-apothem-r1', 'scene1-angle-arc', 'scene1-right-angle', 'scene1-label-angle',
             'scene1-circle1', 'scene1-label-r1',
             'scene1-hex1', 'scene1-circle2'],
            // 步骤4: 显示递推公式
            ['scene1-circle0', 'scene1-hex0', 'scene1-radius-r0', 'scene1-label-r0',
             'scene1-apothem-r1', 'scene1-angle-arc', 'scene1-right-angle', 'scene1-label-angle',
             'scene1-circle1', 'scene1-label-r1',
             'scene1-hex1', 'scene1-circle2',
             'scene1-formula', 'scene1-label-sqrt']
        ];
        
        // 获取DOM元素
        const btnPrev = document.getElementById('scene1-btn-prev');
        const btnNext = document.getElementById('scene1-btn-next');
        const stepIndicator = document.getElementById('scene1-step-indicator');
        
        // 显示/隐藏元素工具函数
        function showElement(id) {
            const el = document.getElementById(id);
            if (el) {
                el.classList.remove('scene1-hidden');
                el.style.display = 'block';
                el.style.opacity = '1';
            }
        }
        
        function hideElement(id) {
            const el = document.getElementById(id);
            if (el) {
                el.classList.add('scene1-hidden');
                el.style.display = 'none';
                el.style.opacity = '0';
            }
        }
        
        // 更新步骤显示
        function updateStep() {
            // 隐藏所有元素
            const allElements = stepConfig.flat();
            const uniqueElements = [...new Set(allElements)];
            uniqueElements.forEach(id => hideElement(id));
            
            // 显示当前步骤的元素
            const visibleElements = stepConfig[currentStep];
            visibleElements.forEach(id => showElement(id));
            
            // 更新步骤指示器
            stepIndicator.textContent = `步骤 ${currentStep + 1}/${totalSteps}`;
            
            // 更新按钮状态
            btnPrev.disabled = currentStep === 0;
            btnNext.disabled = currentStep === totalSteps - 1;
        }
        
        // 按钮事件绑定
        btnPrev.addEventListener('click', function() {
            if (currentStep > 0) {
                currentStep--;
                updateStep();
            }
        });
        
        btnNext.addEventListener('click', function() {
            if (currentStep < totalSteps - 1) {
                currentStep++;
                updateStep();
            }
        });
        
        // 初始化
        updateStep();
    })();
    </script>
