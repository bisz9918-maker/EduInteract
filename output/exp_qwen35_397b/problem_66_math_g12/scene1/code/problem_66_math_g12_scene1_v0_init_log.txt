    <script>
    (function() {
        // Scene 1 实现：正三角形绕中位线旋转
        
        // 状态管理
        let currentStep = 0;
        const totalSteps = 5;
        
        // DOM 元素引用
        const btnPrev = document.getElementById('scene1-btn-prev');
        const btnNext = document.getElementById('scene1-btn-next');
        const stepIndicator = document.getElementById('scene1-step-indicator');
        
        // 左侧平面图形元素
        const polygonAed = document.getElementById('scene1-polygon-aed');
        const arcTrajectory = document.getElementById('scene1-arc-trajectory');
        const labelTheta = document.getElementById('scene1-label-theta');
        const labelRotateHint = document.getElementById('scene1-label-rotate-hint');
        
        // 右侧立体图形组
        const groupSolid = document.getElementById('scene1-group-solid');
        const polygonFed = document.getElementById('scene1-polygon-fed');
        const lineFd = document.getElementById('scene1-line-fd');
        const lineFe = document.getElementById('scene1-line-fe');
        const lineDeRight = document.getElementById('scene1-line-de-right');
        const lineAD = document.getElementById('scene1-line-a-d');
        const lineAE = document.getElementById('scene1-line-a-e');
        const lineAF = document.getElementById('scene1-line-a-f');
        const lineAG = document.getElementById('scene1-line-a-g');
        const pointAPrime = document.getElementById('scene1-point-a-prime');
        const pointDRight = document.getElementById('scene1-point-d-right');
        const pointERight = document.getElementById('scene1-point-e-right');
        const pointFRight = document.getElementById('scene1-point-f-right');
        const pointGRight = document.getElementById('scene1-point-g-right');
        const labelAPrime = document.getElementById('scene1-label-a-prime');
        const labelDRight = document.getElementById('scene1-label-d-right');
        const labelERight = document.getElementById('scene1-label-e-right');
        const labelFRight = document.getElementById('scene1-label-f-right');
        const labelGRight = document.getElementById('scene1-label-g-right');
        
        // 工具函数：显示元素
        function show(el) {
            if (el) {
                el.classList.remove('scene1-hidden');
                el.classList.add('scene1-visible');
                el.style.display = 'block';
                el.style.opacity = '1';
            }
        }
        
        // 工具函数：隐藏元素
        function hide(el) {
            if (el) {
                el.classList.remove('scene1-visible');
                el.classList.add('scene1-hidden');
                el.style.display = 'none';
                el.style.opacity = '0';
            }
        }
        
        // 更新显示状态
        function updateDisplay() {
            // 更新步骤指示器
            stepIndicator.textContent = '步骤 ' + (currentStep + 1) + '/' + totalSteps;
            
            // 更新按钮状态
            btnPrev.disabled = (currentStep === 0);
            btnNext.disabled = (currentStep === totalSteps - 1);
            
            // 根据步骤显示不同元素
            switch (currentStep) {
                case 0:
                    // 初始状态：只显示左侧平面图形基础结构
                    hide(polygonAed);
                    hide(arcTrajectory);
                    hide(labelTheta);
                    hide(labelRotateHint);
                    hide(groupSolid);
                    break;
                    
                case 1:
                    // 步骤 1：高亮△AED，显示提示文字
                    show(polygonAed);
                    polygonAed.setAttribute('fill-opacity', '0.2');
                    show(labelRotateHint);
                    hide(arcTrajectory);
                    hide(labelTheta);
                    hide(groupSolid);
                    break;
                    
                case 2:
                    // 步骤 2：显示旋转轨迹圆弧
                    show(polygonAed);
                    polygonAed.setAttribute('fill-opacity', '0.2');
                    show(labelRotateHint);
                    show(arcTrajectory);
                    arcTrajectory.style.strokeDashoffset = '0';
                    show(labelTheta);
                    hide(groupSolid);
                    break;
                    
                case 3:
                    // 步骤 3：显示 A'点和部分立体结构
                    show(polygonAed);
                    polygonAed.setAttribute('fill-opacity', '0.2');
                    show(labelRotateHint);
                    show(arcTrajectory);
                    arcTrajectory.style.strokeDashoffset = '0';
                    show(labelTheta);
                    show(groupSolid);
                    
                    // 显示 A'点、A'D、A'E、A'G
                    show(pointAPrime);
                    show(labelAPrime);
                    show(lineAD);
                    show(lineAE);
                    show(lineAG);
                    show(pointDRight);
                    show(pointERight);
                    show(pointGRight);
                    show(labelDRight);
                    show(labelERight);
                    show(labelGRight);
                    
                    // 隐藏底面和 A'F
                    hide(polygonFed);
                    hide(lineFd);
                    hide(lineFe);
                    hide(lineDeRight);
                    hide(lineAF);
                    hide(pointFRight);
                    hide(labelFRight);
                    break;
                    
                case 4:
                    // 步骤 4：显示完整三棱锥
                    show(polygonAed);
                    polygonAed.setAttribute('fill-opacity', '0.2');
                    show(labelRotateHint);
                    show(arcTrajectory);
                    arcTrajectory.style.strokeDashoffset = '0';
                    show(labelTheta);
                    show(groupSolid);
                    
                    // 显示所有立体元素
                    show(pointAPrime);
                    show(labelAPrime);
                    show(lineAD);
                    show(lineAE);
                    show(lineAF);
                    show(lineAG);
                    show(pointDRight);
                    show(pointERight);
                    show(pointFRight);
                    show(pointGRight);
                    show(labelDRight);
                    show(labelERight);
                    show(labelFRight);
                    show(labelGRight);
                    show(lineFd);
                    show(lineFe);
                    show(lineDeRight);
                    
                    // 显示底面填充
                    show(polygonFed);
                    polygonFed.setAttribute('fill-opacity', '0.15');
                    break;
            }
        }
        
        // 事件处理：上一步
        btnPrev.addEventListener('click', function() {
            if (currentStep > 0) {
                currentStep--;
                updateDisplay();
            }
        });
        
        // 事件处理：下一步
        btnNext.addEventListener('click', function() {
            if (currentStep < totalSteps - 1) {
                currentStep++;
                updateDisplay();
            }
        });
        
        // 初始化显示
        updateDisplay();
        
    })();
    </script>
