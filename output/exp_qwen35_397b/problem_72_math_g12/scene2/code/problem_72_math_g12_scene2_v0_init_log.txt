    <script>
    (function() {
        // Scene 2: 四棱锥表面展开图
        
        let currentStep = 0;
        const totalSteps = 5;
        
        // 工具函数：显示元素
        function show(el) {
            if (el) {
                el.classList.remove('scene2-hidden');
                el.classList.add('scene2-visible');
                el.style.display = 'block';
                el.style.opacity = '1';
            }
        }
        
        // 工具函数：隐藏元素
        function hide(el) {
            if (el) {
                el.classList.remove('scene2-visible');
                el.classList.add('scene2-hidden');
                el.style.display = 'none';
                el.style.opacity = '0';
            }
        }
        
        // 获取元素
        const baseRect = document.getElementById('scene2-base-rect');
        const baseLabel = document.getElementById('scene2-base-label');
        const baseLen = document.getElementById('scene2-base-len');
        const baseWidth = document.getElementById('scene2-base-width');
        const baseArea = document.getElementById('scene2-base-area');
        
        const frontGroup = document.getElementById('scene2-front-group');
        const frontLabel = document.getElementById('scene2-front-label');
        const frontSlant = document.getElementById('scene2-front-slant');
        const frontArea = document.getElementById('scene2-front-area');
        
        const backGroup = document.getElementById('scene2-back-group');
        const backLabel = document.getElementById('scene2-back-label');
        const backSlant = document.getElementById('scene2-back-slant');
        const backArea = document.getElementById('scene2-back-area');
        
        const leftGroup = document.getElementById('scene2-left-group');
        const leftLabel = document.getElementById('scene2-left-label');
        const leftSlant = document.getElementById('scene2-left-slant');
        const leftArea = document.getElementById('scene2-left-area');
        
        const rightGroup = document.getElementById('scene2-right-group');
        const rightLabel = document.getElementById('scene2-right-label');
        const rightSlant = document.getElementById('scene2-right-slant');
        const rightArea = document.getElementById('scene2-right-area');
        
        const summaryGroup = document.getElementById('scene2-summary-group');
        
        const stepIndicator = document.getElementById('scene2-step-indicator');
        const btnPrev = document.getElementById('scene2-btn-prev');
        const btnNext = document.getElementById('scene2-btn-next');
        
        // 更新显示状态
        function updateDisplay() {
            stepIndicator.textContent = `步骤 ${currentStep}/5`;
            
            // 按钮状态
            btnPrev.disabled = (currentStep === 0);
            btnNext.disabled = (currentStep === totalSteps);
            
            // 底面始终显示
            show(baseRect);
            show(baseLabel);
            show(baseLen);
            show(baseWidth);
            show(baseArea);
            
            // 步骤 1: 前侧面
            if (currentStep >= 1) {
                show(frontGroup);
                show(frontLabel);
                show(frontSlant);
                show(frontArea);
            } else {
                hide(frontGroup);
                hide(frontLabel);
                hide(frontSlant);
                hide(frontArea);
            }
            
            // 步骤 2: 后侧面
            if (currentStep >= 2) {
                show(backGroup);
                show(backLabel);
                show(backSlant);
                show(backArea);
            } else {
                hide(backGroup);
                hide(backLabel);
                hide(backSlant);
                hide(backArea);
            }
            
            // 步骤 3: 左侧面
            if (currentStep >= 3) {
                show(leftGroup);
                show(leftLabel);
                show(leftSlant);
                show(leftArea);
            } else {
                hide(leftGroup);
                hide(leftLabel);
                hide(leftSlant);
                hide(leftArea);
            }
            
            // 步骤 4: 右侧面
            if (currentStep >= 4) {
                show(rightGroup);
                show(rightLabel);
                show(rightSlant);
                show(rightArea);
            } else {
                hide(rightGroup);
                hide(rightLabel);
                hide(rightSlant);
                hide(rightArea);
            }
            
            // 步骤 5: 总面积汇总
            if (currentStep >= 5) {
                show(summaryGroup);
            } else {
                hide(summaryGroup);
            }
        }
        
        // 下一步
        btnNext.addEventListener('click', function() {
            if (currentStep < totalSteps) {
                currentStep++;
                updateDisplay();
            }
        });
        
        // 上一步
        btnPrev.addEventListener('click', function() {
            if (currentStep > 0) {
                currentStep--;
                updateDisplay();
            }
        });
        
        // 初始化显示
        updateDisplay();
        
    })();
    </script>
