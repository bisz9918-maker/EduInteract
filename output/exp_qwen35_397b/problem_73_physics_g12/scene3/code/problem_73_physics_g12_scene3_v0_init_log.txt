    <script>
    (function() {
        // Scene 3 实现：能量转化关系与电量计算
        
        let currentState = 0;
        const totalStates = 4;
        
        // 获取 DOM 元素
        const btnPrev = document.getElementById('scene3-btn-prev');
        const btnNext = document.getElementById('scene3-btn-next');
        
        const boxPotential = document.getElementById('scene3-box-potential');
        const boxKinetic = document.getElementById('scene3-box-kinetic');
        const boxElectric = document.getElementById('scene3-box-electric');
        
        const formulaEnergy = document.getElementById('scene3-formula-energy');
        const formulaGeometry = document.getElementById('scene3-formula-geometry');
        const formulaCharge = document.getElementById('scene3-formula-charge');
        const step1 = document.getElementById('scene3-step1');
        const step2 = document.getElementById('scene3-step2');
        
        // 显示/隐藏辅助函数
        function showElement(el) {
            el.style.display = 'block';
            el.style.opacity = '1';
        }
        
        function hideElement(el) {
            el.style.display = 'none';
            el.style.opacity = '0';
        }
        
        // 重置所有高亮
        function resetHighlights() {
            boxPotential.classList.remove('scene3-highlight');
            boxKinetic.classList.remove('scene3-highlight');
            boxElectric.classList.remove('scene3-highlight');
        }
        
        // 设置状态
        function setState(state) {
            currentState = state;
            
            // 重置所有公式和步骤
            hideElement(formulaEnergy);
            hideElement(formulaGeometry);
            hideElement(formulaCharge);
            hideElement(step1);
            hideElement(step2);
            resetHighlights();
            
            // 根据状态显示相应元素
            if (state >= 1) {
                showElement(formulaEnergy);
                showElement(formulaGeometry);
                boxPotential.classList.add('scene3-highlight');
            }
            
            if (state >= 2) {
                showElement(formulaCharge);
                boxElectric.classList.add('scene3-highlight');
            }
            
            if (state >= 3) {
                showElement(step1);
                showElement(step2);
                // 所有公式高亮
                formulaEnergy.classList.add('scene3-formula-highlight');
                formulaGeometry.classList.add('scene3-formula-highlight');
                formulaCharge.classList.add('scene3-formula-highlight');
            }
            
            // 更新按钮状态
            updateButtons();
            
            // 重新渲染 MathJax
            if (window.MathJax && window.MathJax.typesetPromise) {
                MathJax.typesetPromise();
            }
        }
        
        // 更新按钮状态
        function updateButtons() {
            if (currentState === 0) {
                btnPrev.classList.add('scene3-btn-disabled');
            } else {
                btnPrev.classList.remove('scene3-btn-disabled');
            }
            
            if (currentState === totalStates - 1) {
                btnNext.classList.add('scene3-btn-disabled');
            } else {
                btnNext.classList.remove('scene3-btn-disabled');
            }
        }
        
        // 事件绑定
        btnPrev.addEventListener('click', function() {
            if (currentState > 0) {
                setState(currentState - 1);
            }
        });
        
        btnNext.addEventListener('click', function() {
            if (currentState < totalStates - 1) {
                setState(currentState + 1);
            }
        });
        
        // 初始化
        setState(0);
        
    })();
    </script>
