    <script>
    (function() {
        // 坐标表参考:
        // 悬挂点 O: (250, 60)
        // 初始质点: (130, 150)
        // 最低点质点: (250, 210)
        // 水面: y=345
        // 高度 H 标注线: x=320, y=60~345
        // 高度 h 标注线: x=270, y=150~210
        
        let currentStep = 0;
        const totalSteps = 3;
        
        // 步骤定义
        const steps = [
            {
                // 步骤 0: 初始状态
                show: ['scene1-suspend-point', 'scene1-rope-initial', 'scene1-mass-initial', 
                       'scene1-vertical-ref', 'scene1-angle-arc', 'scene1-water-surface', 
                       'scene1-shore', 'scene1-platform', 'scene1-height-H',
                       'scene1-label-O', 'scene1-label-l', 'scene1-label-alpha', 
                       'scene1-label-H', 'scene1-label-m'],
                hide: ['scene1-rope-lowest', 'scene1-mass-lowest', 'scene1-trajectory', 
                       'scene1-height-h', 'scene1-h-arrow-top', 'scene1-h-arrow-bottom',
                       'scene1-velocity-v', 'scene1-v-arrow', 'scene1-label-h', 
                       'scene1-label-v', 'scene1-label-formula']
            },
            {
                // 步骤 1: 显示摆动轨迹和最低点
                show: ['scene1-suspend-point', 'scene1-rope-initial', 'scene1-mass-initial', 
                       'scene1-vertical-ref', 'scene1-angle-arc', 'scene1-water-surface', 
                       'scene1-shore', 'scene1-platform', 'scene1-height-H',
                       'scene1-label-O', 'scene1-label-l', 'scene1-label-alpha', 
                       'scene1-label-H', 'scene1-label-m',
                       'scene1-rope-lowest', 'scene1-mass-lowest', 'scene1-trajectory'],
                hide: ['scene1-height-h', 'scene1-h-arrow-top', 'scene1-h-arrow-bottom',
                       'scene1-velocity-v', 'scene1-v-arrow', 'scene1-label-h', 
                       'scene1-label-v', 'scene1-label-formula']
            },
            {
                // 步骤 2: 显示高度差 h
                show: ['scene1-suspend-point', 'scene1-rope-initial', 'scene1-mass-initial', 
                       'scene1-vertical-ref', 'scene1-angle-arc', 'scene1-water-surface', 
                       'scene1-shore', 'scene1-platform', 'scene1-height-H',
                       'scene1-label-O', 'scene1-label-l', 'scene1-label-alpha', 
                       'scene1-label-H', 'scene1-label-m',
                       'scene1-rope-lowest', 'scene1-mass-lowest', 'scene1-trajectory',
                       'scene1-height-h', 'scene1-h-arrow-top', 'scene1-h-arrow-bottom', 'scene1-label-h'],
                hide: ['scene1-velocity-v', 'scene1-v-arrow', 'scene1-label-v', 'scene1-label-formula']
            },
            {
                // 步骤 3: 显示速度和机械能守恒公式
                show: ['scene1-suspend-point', 'scene1-rope-initial', 'scene1-mass-initial', 
                       'scene1-vertical-ref', 'scene1-angle-arc', 'scene1-water-surface', 
                       'scene1-shore', 'scene1-platform', 'scene1-height-H',
                       'scene1-label-O', 'scene1-label-l', 'scene1-label-alpha', 
                       'scene1-label-H', 'scene1-label-m',
                       'scene1-rope-lowest', 'scene1-mass-lowest', 'scene1-trajectory',
                       'scene1-height-h', 'scene1-h-arrow-top', 'scene1-h-arrow-bottom', 'scene1-label-h',
                       'scene1-velocity-v', 'scene1-v-arrow', 'scene1-label-v', 'scene1-label-formula'],
                hide: []
            }
        ];
        
        function showElement(id) {
            const el = document.getElementById(id);
            if (el) {
                el.style.opacity = '1';
                if (el.tagName === 'line' || el.tagName === 'circle' || el.tagName === 'path' || 
                    el.tagName === 'polygon' || el.tagName === 'rect' || el.tagName === 'text') {
                    el.removeAttribute('opacity');
                } else {
                    el.style.display = 'block';
                }
            }
        }
        
        function hideElement(id) {
            const el = document.getElementById(id);
            if (el) {
                if (el.tagName === 'line' || el.tagName === 'circle' || el.tagName === 'path' || 
                    el.tagName === 'polygon' || el.tagName === 'rect' || el.tagName === 'text') {
                    el.setAttribute('opacity', '0');
                } else {
                    el.style.display = 'none';
                }
            }
        }
        
        function updateStep() {
            const step = steps[currentStep];
            
            // 显示该步骤应显示的元素
            step.show.forEach(id => showElement(id));
            
            // 隐藏该步骤应隐藏的元素
            step.hide.forEach(id => hideElement(id));
            
            // 更新步骤指示器
            document.getElementById('scene1-step-indicator').textContent = `步骤 ${currentStep}/${totalSteps}`;
            
            // 更新按钮状态
            document.getElementById('scene1-btn-prev').disabled = currentStep === 0;
            document.getElementById('scene1-btn-next').disabled = currentStep === totalSteps;
        }
        
        function stepForward() {
            if (currentStep < totalSteps) {
                currentStep++;
                updateStep();
            }
        }
        
        function stepBackward() {
            if (currentStep > 0) {
                currentStep--;
                updateStep();
            }
        }
        
        // 绑定事件
        document.getElementById('scene1-btn-next').addEventListener('click', stepForward);
        document.getElementById('scene1-btn-prev').addEventListener('click', stepBackward);
        
        // 初始化
        updateStep();
    })();
    </script>
