    <script>
    (function() {
        // Scene 2: 最低点受力分析与向心力公式
        let currentStep = 0;
        const totalSteps = 4;

        // 元素引用
        const gravityGroup = document.getElementById('scene2-gravity-group');
        const tensionGroup = document.getElementById('scene2-tension-group');
        const centripetalGroup = document.getElementById('scene2-centripetal-group');
        const calculationGroup = document.getElementById('scene2-calculation-group');
        const stepIndicator = document.getElementById('scene2-step-indicator');
        const btnPrevBg = document.getElementById('scene2-btn-prev-bg');
        const btnPrevText = document.getElementById('scene2-btn-prev-text');
        const btnNextBg = document.getElementById('scene2-btn-next-bg');
        const btnNextText = document.getElementById('scene2-btn-next-text');
        const tensionArrow = document.getElementById('scene2-arrow-tension');

        // 工具函数：显示/隐藏元素
        function show(el) {
            el.classList.remove('hidden');
            el.style.display = '';
            el.style.opacity = '1';
        }

        function hide(el) {
            el.classList.add('hidden');
            el.style.display = 'none';
            el.style.opacity = '0';
        }

        // 更新步骤显示
        function updateStep() {
            // 根据当前步骤显示/隐藏对应组
            if (currentStep >= 1) {
                show(gravityGroup);
            } else {
                hide(gravityGroup);
            }

            if (currentStep >= 2) {
                show(tensionGroup);
            } else {
                hide(tensionGroup);
            }

            if (currentStep >= 3) {
                show(centripetalGroup);
            } else {
                hide(centripetalGroup);
            }

            if (currentStep >= 4) {
                show(calculationGroup);
                // 加粗拉力箭头
                tensionArrow.setAttribute('stroke-width', '5');
            } else {
                hide(calculationGroup);
                tensionArrow.setAttribute('stroke-width', '3');
            }

            // 更新步骤指示器
            stepIndicator.textContent = '步骤 ' + currentStep + '/' + totalSteps;

            // 更新按钮状态
            if (currentStep === 0) {
                btnPrevBg.setAttribute('fill', '#e0e0e0');
                btnPrevText.setAttribute('fill', '#999');
            } else {
                btnPrevBg.setAttribute('fill', '#f5f5f5');
                btnPrevText.setAttribute('fill', '#333');
            }

            if (currentStep === totalSteps) {
                btnNextBg.setAttribute('fill', '#e0e0e0');
                btnNextText.setAttribute('fill', '#999');
            } else {
                btnNextBg.setAttribute('fill', '#f5f5f5');
                btnNextText.setAttribute('fill', '#333');
            }
        }

        // 按钮点击事件
        btnPrevBg.addEventListener('click', function() {
            if (currentStep > 0) {
                currentStep--;
                updateStep();
            }
        });

        btnNextBg.addEventListener('click', function() {
            if (currentStep < totalSteps) {
                currentStep++;
                updateStep();
            }
        });

        // 添加点击反馈效果
        btnPrevBg.style.cursor = 'pointer';
        btnNextBg.style.cursor = 'pointer';

        btnPrevBg.addEventListener('mouseenter', function() {
            if (currentStep > 0) {
                this.setAttribute('fill', '#e0e0e0');
            }
        });
        btnPrevBg.addEventListener('mouseleave', function() {
            if (currentStep > 0) {
                this.setAttribute('fill', '#f5f5f5');
            }
        });

        btnNextBg.addEventListener('mouseenter', function() {
            if (currentStep < totalSteps) {
                this.setAttribute('fill', '#e0e0e0');
            }
        });
        btnNextBg.addEventListener('mouseleave', function() {
            if (currentStep < totalSteps) {
                this.setAttribute('fill', '#f5f5f5');
            }
        });

        // 初始化
        updateStep();
    })();
    </script>
