    <script>
    (function() {
        // Scene 1: 三球连接系统运动过程
        
        // 布局坐标表（实现对照依据）：
        // 画布尺寸：800×500px，L=100px 作为绘图单位
        // 斜面三角形顶点：(120,400), (466,400), (466,200)
        // 初始位置：A(323,200), B(236,250), C(150,300)
        // 步骤 1：A(466,300), B(323,200), C(236,250)
        // 步骤 2：A(466,300) 灰色，B(466,300), C(323,200)
        // 步骤 3：A(466,300) 灰色，B(466,300) 灰色，C(466,400)
        
        // 各步骤三球坐标
        const stepPositions = [
            // 步骤 0: 初始状态
            { a: { x: 323, y: 200 }, b: { x: 236, y: 250 }, c: { x: 150, y: 300 } },
            // 步骤 1: A 刚要落地
            { a: { x: 466, y: 300 }, b: { x: 323, y: 200 }, c: { x: 236, y: 250 } },
            // 步骤 2: B 刚要落地
            { a: { x: 466, y: 300 }, b: { x: 466, y: 300 }, c: { x: 323, y: 200 } },
            // 步骤 3: C 刚要落地
            { a: { x: 466, y: 300 }, b: { x: 466, y: 300 }, c: { x: 466, y: 400 } }
        ];
        
        // 各步骤速度标签
        const velocityLabels = ['', 'v₁', 'v₂', 'v₃'];
        
        // 当前步骤状态
        let currentStep = 0;
        
        // DOM 元素引用
        const ballA = document.getElementById('scene1-ball-a');
        const ballB = document.getElementById('scene1-ball-b');
        const ballC = document.getElementById('scene1-ball-c');
        const labelA = document.getElementById('scene1-label-a');
        const labelB = document.getElementById('scene1-label-b');
        const labelC = document.getElementById('scene1-label-c');
        const massA = document.getElementById('scene1-mass-a');
        const massB = document.getElementById('scene1-mass-b');
        const massC = document.getElementById('scene1-mass-c');
        const ropeAB = document.getElementById('scene1-rope-ab');
        const ropeBC = document.getElementById('scene1-rope-bc');
        const trajectoryA = document.getElementById('scene1-trajectory-a');
        const trajectoryB = document.getElementById('scene1-trajectory-b');
        const trajectoryC = document.getElementById('scene1-trajectory-c');
        const velocityGroup = document.getElementById('scene1-velocity-group');
        const velocityArrow = document.getElementById('scene1-velocity-arrow');
        const velocityLabel = document.getElementById('scene1-velocity-label');
        const btnPrevBg = document.getElementById('scene1-btn-prev-bg');
        const btnPrevText = document.getElementById('scene1-btn-prev-text');
        const btnNextBg = document.getElementById('scene1-btn-next-bg');
        const btnNextText = document.getElementById('scene1-btn-next-text');
        
        // 阶段指示器
        const indicators = [
            document.getElementById('scene1-indicator-0'),
            document.getElementById('scene1-indicator-1'),
            document.getElementById('scene1-indicator-2'),
            document.getElementById('scene1-indicator-3')
        ];
        
        // 更新按钮状态
        function updateButtons() {
            if (currentStep === 0) {
                btnPrevBg.setAttribute('fill', '#cccccc');
                btnPrevText.setAttribute('fill', '#666666');
            } else {
                btnPrevBg.setAttribute('fill', '#1565c0');
                btnPrevText.setAttribute('fill', '#ffffff');
            }
            
            if (currentStep === 3) {
                btnNextBg.setAttribute('fill', '#cccccc');
                btnNextText.setAttribute('fill', '#666666');
            } else {
                btnNextBg.setAttribute('fill', '#1565c0');
                btnNextText.setAttribute('fill', '#ffffff');
            }
        }
        
        // 更新阶段指示器
        function updateIndicators() {
            const colors = ['#1565c0', '#2e7d32', '#c62828', '#888888'];
            indicators.forEach((ind, i) => {
                if (i === currentStep) {
                    ind.setAttribute('fill', colors[i]);
                    ind.setAttribute('stroke', '#000000');
                    ind.setAttribute('stroke-width', '2');
                } else if (i < currentStep) {
                    ind.setAttribute('fill', colors[i]);
                    ind.setAttribute('stroke', '#000000');
                    ind.setAttribute('stroke-width', '1');
                } else {
                    ind.setAttribute('fill', '#cccccc');
                    ind.setAttribute('stroke', '#000000');
                    ind.setAttribute('stroke-width', '1');
                }
            });
        }
        
        // 渲染当前步骤
        function renderStep() {
            const pos = stepPositions[currentStep];
            
            // 更新球位置
            ballA.setAttribute('cx', pos.a.x);
            ballA.setAttribute('cy', pos.a.y);
            ballB.setAttribute('cx', pos.b.x);
            ballB.setAttribute('cy', pos.b.y);
            ballC.setAttribute('cx', pos.c.x);
            ballC.setAttribute('cy', pos.c.y);
            
            // 更新标签位置（跟随球）
            labelA.setAttribute('x', pos.a.x + 12);
            labelA.setAttribute('y', pos.a.y - 5);
            massA.setAttribute('x', pos.a.x + 12);
            massA.setAttribute('y', pos.a.y + 18);
            
            labelB.setAttribute('x', pos.b.x + 12);
            labelB.setAttribute('y', pos.b.y - 5);
            massB.setAttribute('x', pos.b.x + 12);
            massB.setAttribute('y', pos.b.y + 18);
            
            labelC.setAttribute('x', pos.c.x + 12);
            labelC.setAttribute('y', pos.c.y - 5);
            massC.setAttribute('x', pos.c.x + 12);
            massC.setAttribute('y', pos.c.y + 18);
            
            // 更新绳子位置
            ropeAB.setAttribute('x1', pos.a.x);
            ropeAB.setAttribute('y1', pos.a.y);
            ropeAB.setAttribute('x2', pos.b.x);
            ropeAB.setAttribute('y2', pos.b.y);
            
            ropeBC.setAttribute('x1', pos.b.x);
            ropeBC.setAttribute('y1', pos.b.y);
            ropeBC.setAttribute('x2', pos.c.x);
            ropeBC.setAttribute('y2', pos.c.y);
            
            // 更新轨迹显示
            trajectoryA.setAttribute('opacity', currentStep >= 1 ? '0.6' : '0');
            trajectoryB.setAttribute('opacity', currentStep >= 2 ? '0.6' : '0');
            trajectoryC.setAttribute('opacity', currentStep >= 3 ? '0.6' : '0');
            
            // 更新球的高亮和状态
            // 步骤 0: 正常显示
            // 步骤 1: A 高亮（蓝色加粗）
            // 步骤 2: A 灰色，B 高亮（绿色加粗）
            // 步骤 3: A、B 灰色，C 高亮（红色加粗）
            
            if (currentStep === 0) {
                ballA.setAttribute('fill', '#1565c0');
                ballA.setAttribute('stroke', '#000000');
                ballA.setAttribute('stroke-width', '1.5');
                ballB.setAttribute('fill', '#2e7d32');
                ballB.setAttribute('stroke', '#000000');
                ballB.setAttribute('stroke-width', '1.5');
                ballC.setAttribute('fill', '#c62828');
                ballC.setAttribute('stroke', '#000000');
                ballC.setAttribute('stroke-width', '1.5');
                velocityGroup.setAttribute('opacity', '0');
            } else if (currentStep === 1) {
                ballA.setAttribute('fill', '#1565c0');
                ballA.setAttribute('stroke', '#0d47a1');
                ballA.setAttribute('stroke-width', '3');
                ballB.setAttribute('fill', '#2e7d32');
                ballB.setAttribute('stroke', '#000000');
                ballB.setAttribute('stroke-width', '1.5');
                ballC.setAttribute('fill', '#c62828');
                ballC.setAttribute('stroke', '#000000');
                ballC.setAttribute('stroke-width', '1.5');
                
                // 显示速度箭头 v₁
                velocityArrow.setAttribute('stroke', '#1565c0');
                velocityArrow.setAttribute('marker-end', 'url(#scene1-arrowhead-blue)');
                velocityLabel.textContent = 'v₁';
                velocityLabel.setAttribute('fill', '#1565c0');
                velocityGroup.setAttribute('opacity', '1');
            } else if (currentStep === 2) {
                ballA.setAttribute('fill', '#90a4ae');
                ballA.setAttribute('stroke', '#666666');
                ballA.setAttribute('stroke-width', '1.5');
                ballB.setAttribute('fill', '#2e7d32');
                ballB.setAttribute('stroke', '#1b5e20');
                ballB.setAttribute('stroke-width', '3');
                ballC.setAttribute('fill', '#c62828');
                ballC.setAttribute('stroke', '#000000');
                ballC.setAttribute('stroke-width', '1.5');
                
                // 显示速度箭头 v₂
                velocityArrow.setAttribute('stroke', '#2e7d32');
                velocityArrow.setAttribute('marker-end', 'url(#scene1-arrowhead-green)');
                velocityLabel.textContent = 'v₂';
                velocityLabel.setAttribute('fill', '#2e7d32');
                velocityGroup.setAttribute('opacity', '1');
            } else if (currentStep === 3) {
                ballA.setAttribute('fill', '#90a4ae');
                ballA.setAttribute('stroke', '#666666');
                ballA.setAttribute('stroke-width', '1.5');
                ballB.setAttribute('fill', '#90a4ae');
                ballB.setAttribute('stroke', '#666666');
                ballB.setAttribute('stroke-width', '1.5');
                ballC.setAttribute('fill', '#c62828');
                ballC.setAttribute('stroke', '#b71c1c');
                ballC.setAttribute('stroke-width', '3');
                
                // 显示速度箭头 v₃
                velocityArrow.setAttribute('stroke', '#c62828');
                velocityArrow.setAttribute('marker-end', 'url(#scene1-arrowhead-red)');
                velocityLabel.textContent = 'v₃';
                velocityLabel.setAttribute('fill', '#c62828');
                velocityGroup.setAttribute('opacity', '1');
            }
            
            // 更新按钮状态
            updateButtons();
            
            // 更新指示器
            updateIndicators();
        }
        
        // 上一步
        function prevStep() {
            if (currentStep > 0) {
                currentStep--;
                renderStep();
            }
        }
        
        // 下一步
        function nextStep() {
            if (currentStep < 3) {
                currentStep++;
                renderStep();
            }
        }
        
        // 绑定按钮点击事件
        btnPrevBg.addEventListener('click', prevStep);
        btnPrevText.addEventListener('click', prevStep);
        btnNextBg.addEventListener('click', nextStep);
        btnNextText.addEventListener('click', nextStep);
        
        // 初始渲染
        renderStep();
        
    })();
    </script>
</body>
