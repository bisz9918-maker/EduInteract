    <script>
    (function() {
        // Scene 1 实现 - 物理过程三状态示意图

        // 状态变量
        let currentStep = 0;
        const totalSteps = 2;

        // 获取 DOM 元素
        const btnPrev = document.getElementById('scene1-btn-prev');
        const btnNext = document.getElementById('scene1-btn-next');
        const statusText = document.getElementById('scene1-status-text');

        // 需要控制显示的元素组
        const bulletGroup = document.getElementById('scene1-bullet-group');
        const arrowV2Group = document.getElementById('scene1-arrow-v2-group');
        const arrowV3Group = document.getElementById('scene1-arrow-v3-group');
        const compressionGroup = document.getElementById('scene1-compression-group');
        const block = document.getElementById('scene1-block');
        const spring = document.getElementById('scene1-spring');

        // 显示/隐藏工具函数
        function show(el) {
            el.style.display = '';
            el.style.opacity = '1';
        }

        function hide(el) {
            el.style.display = 'none';
            el.style.opacity = '0';
        }

        // 更新状态显示
        function updateState() {
            // 更新状态文本
            statusText.textContent = `步骤 ${currentStep}/${totalSteps}`;

            // 更新按钮状态
            btnPrev.disabled = (currentStep === 0);
            btnNext.disabled = (currentStep === totalSteps);

            // 根据步骤显示/隐藏元素
            if (currentStep === 0) {
                // 步骤 0：初始状态
                show(bulletGroup);
                hide(arrowV2Group);
                hide(arrowV3Group);
                hide(compressionGroup);
                // 滑块恢复原位置和颜色
                block.setAttribute('x', '520');
                block.setAttribute('fill', 'rgba(21,101,192,0.2)');
                // 弹簧恢复原长
                spring.setAttribute('d', 'M130,365 L140,365 L145,355 L155,375 L165,355 L175,375 L185,355 L195,375 L205,355 L215,375 L225,355 L235,375 L245,355 L255,375 L265,355 L275,375 L285,355 L295,375 L305,355 L315,375 L325,355 L335,375 L345,355 L355,375 L365,355 L375,375 L385,355 L395,375 L405,355 L415,375 L425,355 L435,375 L445,355 L455,375 L465,355 L475,375 L485,355 L495,375 L505,355 L515,365 L520,365');
            } else if (currentStep === 1) {
                // 步骤 1：子弹射入滑块后
                hide(bulletGroup);
                show(arrowV2Group);
                hide(arrowV3Group);
                hide(compressionGroup);
                // 滑块颜色加深表示合并
                block.setAttribute('fill', 'rgba(21,101,192,0.6)');
                block.setAttribute('x', '520');
                // 弹簧保持原长
                spring.setAttribute('d', 'M130,365 L140,365 L145,355 L155,375 L165,355 L175,375 L185,355 L195,375 L205,355 L215,375 L225,355 L235,375 L245,355 L255,375 L265,355 L275,375 L285,355 L295,375 L305,355 L315,375 L325,355 L335,375 L345,355 L355,375 L365,355 L375,375 L385,355 L395,375 L405,355 L415,375 L425,355 L435,375 L445,355 L455,375 L465,355 L475,375 L485,355 L495,375 L505,355 L515,365 L520,365');
            } else if (currentStep === 2) {
                // 步骤 2：弹簧压缩最短
                hide(bulletGroup);
                hide(arrowV2Group);
                show(arrowV3Group);
                show(compressionGroup);
                // 滑块左移
                block.setAttribute('x', '480');
                block.setAttribute('fill', 'rgba(21,101,192,0.6)');
                // 弹簧缩短至压缩状态 (130,365)-(480,365)
                spring.setAttribute('d', 'M130,365 L140,365 L145,355 L155,375 L165,355 L175,375 L185,355 L195,375 L205,355 L215,375 L225,355 L235,375 L245,355 L255,375 L265,355 L275,375 L285,355 L295,375 L305,355 L315,375 L325,355 L335,375 L345,355 L355,375 L365,355 L375,375 L385,355 L395,375 L405,355 L415,375 L425,355 L435,375 L445,355 L455,375 L465,355 L475,375 L480,365');
            }
        }

        // 按钮事件处理
        btnPrev.addEventListener('click', function() {
            if (currentStep > 0) {
                currentStep--;
                updateState();
            }
        });

        btnNext.addEventListener('click', function() {
            if (currentStep < totalSteps) {
                currentStep++;
                updateState();
            }
        });

        // 初始化状态
        updateState();
    })();
    </script>
