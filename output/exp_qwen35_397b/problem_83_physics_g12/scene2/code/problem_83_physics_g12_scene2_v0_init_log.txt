    <script>
    (function() {
        // Scene 2 实现：子弹射入滑块—动量守恒分析
        
        let currentStep = 0;
        const totalSteps = 3;

        const btnPrev = document.getElementById('scene2-btn-prev');
        const btnNext = document.getElementById('scene2-btn-next');
        const stepIndicator = document.getElementById('scene2-step-indicator');

        // 元素引用
        const beforeCollision = document.getElementById('scene2-before-collision');
        const afterCollision = document.getElementById('scene2-after-collision');
        const equalsSign = document.getElementById('scene2-equals-sign');
        const eqEquals = document.getElementById('scene2-eq-equals');
        const eqValues = document.getElementById('scene2-eq-values');

        function updateDisplay() {
            // 更新步骤指示器
            stepIndicator.textContent = `步骤 ${currentStep + 1}/${totalSteps}`;

            // 更新按钮状态
            btnPrev.disabled = currentStep === 0;
            btnNext.disabled = currentStep === totalSteps - 1;

            // 根据步骤显示/隐藏元素
            if (currentStep === 0) {
                // 步骤0：只显示碰撞前
                beforeCollision.classList.remove('scene2-hidden');
                beforeCollision.classList.add('scene2-visible');
                afterCollision.classList.add('scene2-hidden');
                afterCollision.classList.remove('scene2-visible');
                equalsSign.classList.add('scene2-hidden');
                eqEquals.classList.add('scene2-hidden');
                eqValues.classList.add('scene2-hidden');
            } else if (currentStep === 1) {
                // 步骤1：显示碰撞前和碰撞后
                beforeCollision.classList.remove('scene2-hidden');
                beforeCollision.classList.add('scene2-visible');
                afterCollision.classList.remove('scene2-hidden');
                afterCollision.classList.add('scene2-visible');
                equalsSign.classList.add('scene2-hidden');
                eqEquals.classList.add('scene2-hidden');
                eqValues.classList.add('scene2-hidden');
            } else if (currentStep === 2) {
                // 步骤2：显示完整动量守恒方程
                beforeCollision.classList.remove('scene2-hidden');
                beforeCollision.classList.add('scene2-visible');
                afterCollision.classList.remove('scene2-hidden');
                afterCollision.classList.add('scene2-visible');
                equalsSign.classList.remove('scene2-hidden');
                equalsSign.classList.add('scene2-visible');
                eqEquals.classList.remove('scene2-hidden');
                eqEquals.classList.add('scene2-visible');
                eqValues.classList.remove('scene2-hidden');
                eqValues.classList.add('scene2-visible');
            }
        }

        btnPrev.addEventListener('click', function() {
            if (currentStep > 0) {
                currentStep--;
                updateDisplay();
            }
        });

        btnNext.addEventListener('click', function() {
            if (currentStep < totalSteps - 1) {
                currentStep++;
                updateDisplay();
            }
        });

        // 初始化显示
        updateDisplay();
    })();
    </script>
