    <script>
    (function() {
        // Scene 1: A 球受力分析——四力平衡的直观呈现
        // 布局坐标表（实现对照依据）：
        // | 元素 | 类型 | 位置/范围 | 大小 | 标签偏移方向 | 备注 |
        // |------|------|-----------|------|-------------|------|
        // | 斜面 | 多边形 | (80,420)-(720,140) | 底边 640×高 280 | - | 倾角 37°，黑色轮廓 |
        // | 37°角弧 | 弧线 | 圆心 (100,400), r=25 | 弧长≈20 | 内侧 | 斜面左下角 |
        // | 37°标签 | text | (115,390) | ~30×16 | 角弧内侧 | 斜面倾角 |
        // | 电场线 1-6 | 线段 + 箭头 | y=80,130,180,230,280,330 | 700×2 | 右端箭头 | 水平向右 |
        // | E 标签 | text | (765,75) | ~20×16 | 右侧 | 电场方向标识 |
        // | A 球 | 圆 | 圆心 (550,220) | r=15 | - | 斜面上方，白底黑边 |
        // | A 标签 | text | (572,202) | ~15×16 | 右上 | A 球外侧 |
        // | B 球 | 圆 | 圆心 (350,300) | r=15 | - | A 球左下方，白底黑边 |
        // | B 标签 | text | (332,282) | ~15×16 | 左上 | B 球外侧 |
        // | L 标注线 | 双向箭头 | (368,292)-(532,228) | ~175×3 | 上方 15px | A、B 沿斜面距离 |
        // | L 标签 | text | (450,245) | ~15×16 | 上方 | L 标注中点上方 |
        // | 重力 mg | 箭头 | (550,220)-(550,285) | 65×3 | 末端下方 | 竖直向下，红色 |
        // | mg 标签 | text | (555,298) | ~30×16 | 下方 | 重力标识 |
        // | 支持力 N | 箭头 | (550,220)-(513,170) | ~62×3 | 末端左上 | 垂直斜面向上，蓝色 |
        // | N 标签 | text | (502,158) | ~15×16 | 左上 | 支持力标识 |
        // | 电场力 F_E | 箭头 | (550,220)-(485,220) | 65×3 | 末端左侧 | 水平向左，绿色 |
        // | F_E 标签 | text | (455,200) | ~55×16 | 上方 | F_E=qE |
        // | 库仑力 F_C | 箭头 | (550,220)-(584,195) | ~42×3 | 末端右上 | 沿斜面向上，绿色 |
        // | F_C 标签 | text | (590,182) | ~65×16 | 上方 | F_C=kq²/L² |
        // | 平衡标注 | text | (400,360) | ~120×16 | - | 步骤 5 显示"合力为零" |
        // | 控件栏 | 按钮组 | y=450~490 | 800×40 | 底部居中 | 上一步/下一步按钮 |

        let currentStep = 0;
        const totalSteps = 6;

        // 每步应显示的元素（按步骤索引 0-5）
        // 步骤 0: 初始状态（斜面、电场、两球、L 标注）
        // 步骤 1: 显示重力 mg
        // 步骤 2: 显示支持力 N
        // 步骤 3: 显示电场力 F_E
        // 步骤 4: 显示库仑力 F_C
        // 步骤 5: 显示平衡标注，所有力高亮
        const steps = [
            [],  // 步骤 0: 无额外力
            ['scene1-force-mg-line', 'scene1-force-mg-label'],  // 步骤 1: 重力
            ['scene1-force-mg-line', 'scene1-force-mg-label', 'scene1-force-n-line', 'scene1-force-n-label'],  // 步骤 2: + 支持力
            ['scene1-force-mg-line', 'scene1-force-mg-label', 'scene1-force-n-line', 'scene1-force-n-label', 'scene1-force-fe-line', 'scene1-force-fe-label', 'scene1-force-fe-sub', 'scene1-force-fe-eq'],  // 步骤 3: + 电场力
            ['scene1-force-mg-line', 'scene1-force-mg-label', 'scene1-force-n-line', 'scene1-force-n-label', 'scene1-force-fe-line', 'scene1-force-fe-label', 'scene1-force-fe-sub', 'scene1-force-fe-eq', 'scene1-force-fc-line', 'scene1-force-fc-label', 'scene1-force-fc-sub', 'scene1-force-fc-eq'],  // 步骤 4: + 库仑力
            ['scene1-force-mg-line', 'scene1-force-mg-label', 'scene1-force-n-line', 'scene1-force-n-label', 'scene1-force-fe-line', 'scene1-force-fe-label', 'scene1-force-fe-sub', 'scene1-force-fe-eq', 'scene1-force-fc-line', 'scene1-force-fc-label', 'scene1-force-fc-sub', 'scene1-force-fc-eq']  // 步骤 5: 全部力 + 平衡标注
        ];

        // 获取所有力相关元素
        const allForceElements = [
            'scene1-force-mg-line', 'scene1-force-mg-label',
            'scene1-force-n-line', 'scene1-force-n-label',
            'scene1-force-fe-line', 'scene1-force-fe-label', 'scene1-force-fe-sub', 'scene1-force-fe-eq',
            'scene1-force-fc-line', 'scene1-force-fc-label', 'scene1-force-fc-sub', 'scene1-force-fc-eq'
        ];

        function updateDisplay() {
            // 更新步骤指示器
            document.getElementById('scene1-step-indicator').textContent = `步骤 ${currentStep + 1}/${totalSteps}`;

            // 更新按钮状态
            document.getElementById('scene1-btn-prev').disabled = (currentStep === 0);
            document.getElementById('scene1-btn-next').disabled = (currentStep === totalSteps - 1);

            // 获取当前步骤应显示的元素
            const visibleElements = steps[currentStep];

            // 隐藏所有力元素
            allForceElements.forEach(id => {
                const el = document.getElementById(id);
                if (el) {
                    el.classList.remove('visible');
                }
            });

            // 显示当前步骤的力元素
            visibleElements.forEach(id => {
                const el = document.getElementById(id);
                if (el) {
                    el.classList.add('visible');
                }
            });

            // 平衡标注只在步骤 5 显示
            const balanceLabel = document.getElementById('scene1-balance-label');
            if (currentStep === 5) {
                balanceLabel.classList.add('visible');
            } else {
                balanceLabel.classList.remove('visible');
            }
        }

        // 按钮事件绑定
        document.getElementById('scene1-btn-prev').addEventListener('click', function() {
            if (currentStep > 0) {
                currentStep--;
                updateDisplay();
            }
        });

        document.getElementById('scene1-btn-next').addEventListener('click', function() {
            if (currentStep < totalSteps - 1) {
                currentStep++;
                updateDisplay();
            }
        });

        // 初始化显示
        updateDisplay();
    })();
    </script>
