    <script>
    (function() {
        // Scene 3: B 球释放瞬间受力分析
        
        // 状态变量
        let currentStep = 0;
        const totalSteps = 6;
        
        // 每步需要显示的元素 ID 数组
        const stepElements = {
            0: ['scene3-label-release'],
            1: ['scene3-force-mg-group', 'scene3-label-mg'],
            2: ['scene3-force-n-group', 'scene3-label-n'],
            3: ['scene3-force-fe-group', 'scene3-label-fe'],
            4: ['scene3-force-fc-group', 'scene3-label-fc'],
            5: ['scene3-force-net-group', 'scene3-label-net', 'scene3-accel-a-group', 'scene3-label-a-accel'],
            6: ['scene3-formula-box']
        };
        
        // 显示/隐藏元素工具函数
        function showElement(id) {
            const el = document.getElementById(id);
            if (el) {
                if (el.classList.contains('scene3-force-element')) {
                    el.classList.add('visible');
                } else if (id === 'scene3-formula-box') {
                    el.classList.add('visible');
                } else {
                    el.style.opacity = '1';
                }
            }
        }
        
        function hideElement(id) {
            const el = document.getElementById(id);
            if (el) {
                if (el.classList.contains('scene3-force-element')) {
                    el.classList.remove('visible');
                } else if (id === 'scene3-formula-box') {
                    el.classList.remove('visible');
                } else {
                    el.style.opacity = '0';
                }
            }
        }
        
        // 更新显示状态
        function updateDisplay() {
            // 更新步骤指示器
            document.getElementById('scene3-step-indicator').textContent = '步骤 ' + currentStep + '/' + totalSteps;
            
            // 更新按钮状态
            document.getElementById('scene3-btn-prev').disabled = (currentStep === 0);
            document.getElementById('scene3-btn-next').disabled = (currentStep === totalSteps);
            
            // 步骤 0 特殊处理：释放瞬间标签
            const releaseLabel = document.getElementById('scene3-label-release');
            if (currentStep === 0) {
                releaseLabel.style.opacity = '1';
            } else {
                releaseLabel.style.opacity = '0';
            }
            
            // 根据当前步骤显示/隐藏元素
            for (let step = 1; step <= totalSteps; step++) {
                const elements = stepElements[step];
                if (elements) {
                    elements.forEach(function(id) {
                        if (step <= currentStep) {
                            showElement(id);
                        } else {
                            hideElement(id);
                        }
                    });
                }
            }
        }
        
        // 按钮事件处理
        document.getElementById('scene3-btn-prev').addEventListener('click', function() {
            if (currentStep > 0) {
                currentStep--;
                updateDisplay();
            }
        });
        
        document.getElementById('scene3-btn-next').addEventListener('click', function() {
            if (currentStep < totalSteps) {
                currentStep++;
                updateDisplay();
            }
        });
        
        // 初始化显示
        updateDisplay();
        
    })();
    </script>
