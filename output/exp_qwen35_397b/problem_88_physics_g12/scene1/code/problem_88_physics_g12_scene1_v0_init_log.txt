    <script>
    (function() {
        // Scene 1 实现 - 圆弧轨道电磁感应装置

        // 状态管理
        let currentState = 0;
        const totalSteps = 5;

        // 元素引用
        const btnPrev = document.getElementById('scene1-btn-prev');
        const btnNext = document.getElementById('scene1-btn-next');
        const statusDisplay = document.getElementById('scene1-status');

        // 各步骤需要显示的元素组
        const stepElements = {
            0: {
                show: ['scene1-orbit-left', 'scene1-orbit-right', 'scene1-top-connection', 
                       'scene1-bottom-connection', 'scene1-center-o',
                       'scene1-label-o', 'scene1-label-a', 'scene1-label-b', 
                       'scene1-label-c', 'scene1-label-d'],
                hide: ['scene1-field-arrows', 'scene1-dimension-lines', 'scene1-rod-cd',
                       'scene1-rod-ab', 'scene1-resistor-group', 'scene1-trajectory']
            },
            1: {
                show: ['scene1-field-arrows', 'scene1-label-B'],
                hide: []
            },
            2: {
                show: ['scene1-dimension-lines', 'scene1-label-r-h', 'scene1-label-r-v', 'scene1-label-L'],
                hide: []
            },
            3: {
                show: ['scene1-rod-cd'],
                hide: []
            },
            4: {
                show: ['scene1-rod-ab', 'scene1-resistor-group', 'scene1-label-R'],
                hide: []
            },
            5: {
                show: ['scene1-trajectory'],
                hide: []
            }
        };

        // 显示元素函数
        function showElement(elementId) {
            const el = document.getElementById(elementId);
            if (el) {
                el.style.display = '';
                if (el.classList.contains('scene1-fade-in')) {
                    // 使用 requestAnimationFrame 确保 transition 生效
                    requestAnimationFrame(() => {
                        el.classList.add('visible');
                    });
                }
            }
        }

        // 隐藏元素函数
        function hideElement(elementId) {
            const el = document.getElementById(elementId);
            if (el) {
                el.style.display = 'none';
                el.classList.remove('visible');
            }
        }

        // 更新状态显示
        function updateState() {
            // 更新按钮状态
            btnPrev.disabled = (currentState === 0);
            btnNext.disabled = (currentState >= totalSteps);
            btnNext.textContent = (currentState >= totalSteps) ? '已完成' : '下一步 ▶';

            // 更新状态指示器
            statusDisplay.textContent = `步骤 ${currentState}/${totalSteps}`;

            // 根据当前状态显示/隐藏元素
            for (let step = 0; step <= currentState; step++) {
                const elements = stepElements[step];
                if (elements) {
                    elements.show.forEach(id => showElement(id));
                    // 只在最后一步隐藏需要隐藏的元素
                    if (step === currentState) {
                        elements.hide.forEach(id => hideElement(id));
                    }
                }
            }

            // 隐藏后续步骤的元素
            for (let step = currentState + 1; step <= totalSteps; step++) {
                const elements = stepElements[step];
                if (elements) {
                    elements.show.forEach(id => hideElement(id));
                }
            }

            // 特殊处理：步骤3给金属棒cd添加脉冲效果
            const rodCd = document.getElementById('scene1-rod-cd');
            if (rodCd) {
                if (currentState === 3) {
                    rodCd.classList.add('scene1-pulse');
                } else {
                    rodCd.classList.remove('scene1-pulse');
                }
            }
        }

        // 初始化所有元素为隐藏状态（除了基础轨道）
        function initialize() {
            // 初始只显示步骤0的元素
            const initialShow = stepElements[0].show;
            const initialHide = [...stepElements[0].hide];

            // 添加后续步骤需要显示的元素到隐藏列表
            for (let step = 1; step <= totalSteps; step++) {
                if (stepElements[step]) {
                    stepElements[step].show.forEach(id => {
                        if (!initialHide.includes(id)) {
                            initialHide.push(id);
                        }
                    });
                }
            }

            // 显示初始元素
            initialShow.forEach(id => {
                const el = document.getElementById(id);
                if (el) {
                    el.style.display = '';
                    if (el.classList.contains('scene1-fade-in')) {
                        el.classList.add('visible');
                    }
                }
            });

            // 隐藏其他元素
            initialHide.forEach(id => hideElement(id));

            // 绑定按钮事件
            btnPrev.addEventListener('click', () => {
                if (currentState > 0) {
                    currentState--;
                    updateState();
                }
            });

            btnNext.addEventListener('click', () => {
                if (currentState < totalSteps) {
                    currentState++;
                    updateState();
                }
            });

            // 初始更新
            updateState();
        }

        // 页面加载完成后初始化
        if (document.readyState === 'loading') {
            document.addEventListener('DOMContentLoaded', initialize);
        } else {
            initialize();
        }
    })();
    </script>
