    <script>
    (function() {
        // Scene 2: 金属棒速度分解与有效切割速度分析
        // 布局坐标表参考：
        // | 元素 | 类型 | 位置/范围 | 大小 | 标签偏移方向 | 备注 |
        // |------|------|-----------|------|-------------|------|
        // | 圆弧轨道 | 弧线 | 圆心(180,320)，半径180px，从270°到0° | 线宽2px | - | 四分之一圆，黑色实线 |
        // | 水平基准线 | 线段 | (180,320)-(360,320) | 180×2 | - | 灰色虚线，参考用 |
        // | 竖直基准线 | 线段 | (180,140)-(180,320) | 2×180 | - | 灰色虚线，参考用 |
        // | 金属棒 | 短线段 | 随θ变化，中心在圆弧上 | 长40px，宽3px | - | 红色实线，垂直于半径方向 |
        // | 速度矢量 v₀ | 箭头 | 从金属棒中心沿切线方向 | 长60px，线宽2px | 箭头末端右上 | 蓝色实线箭头 |
        // | 水平分量 vx | 箭头 | 从金属棒中心水平向右 | 长50px，线宽2px | 箭头下方 | 绿色实线箭头 |
        // | 竖直分量 vy | 箭头 | 从金属棒中心竖直向上 | 长40px，线宽1px | 箭头右侧 | 灰色虚线箭头 |
        // | 磁场 B | 箭头组 | 3个竖直向上箭头，x=100,140,400 | 长50px，线宽1px | 箭头右侧 | 灰色虚线箭头 |
        // | 角度θ弧线 | 弧线 | 圆心(180,320)，半径40px | 线宽1px | 弧线外侧 | 灰色细实线 |
        // | 角度θ标签 | text | (210,300) | ~30×16 | - | 显示θ值 |
        // | 公式显示区 | text | (550,180) | ~150×60 | - | 显示 E=BLvₓ |
        // | 滑块控件 | input range | (250,455)-(550,475) | 300×20 | - | 角度θ从0°到90° |
        // | 角度数值显示 | text | (570,460) | ~50×20 | - | 显示当前θ值 |

        const CX = 180;  // 圆心x
        const CY = 320;  // 圆心y
        const R = 180;   // 圆弧半径

        // 获取 DOM 元素
        const slider = document.getElementById('scene2-slider');
        const sliderValue = document.getElementById('scene2-slider-value');
        const rod = document.getElementById('scene2-rod');
        const arrowV0 = document.getElementById('scene2-arrow-v0');
        const labelV0 = document.getElementById('scene2-label-v0');
        const arrowVx = document.getElementById('scene2-arrow-vx');
        const labelVx = document.getElementById('scene2-label-vx');
        const arrowVy = document.getElementById('scene2-arrow-vy');
        const labelVy = document.getElementById('scene2-label-vy');
        const angleArc = document.getElementById('scene2-angle-arc');
        const labelTheta = document.getElementById('scene2-label-theta');
        const formulaValue = document.getElementById('scene2-formula-value');

        // 更新函数
        function update(thetaDeg) {
            const theta = thetaDeg * Math.PI / 180;  // 转换为弧度

            // 金属棒中心坐标
            const rodCx = CX + R * Math.sin(theta);
            const rodCy = CY - R * Math.cos(theta);

            // 金属棒方向：垂直于半径，即切线方向
            // 半径方向角度 = theta (从竖直向上顺时针)
            // 切线方向角度 = theta (从水平向右逆时针)
            const rodLength = 40;
            const rodAngle = theta;  // 切线方向与水平的夹角
            const rodDx = rodLength * Math.cos(rodAngle);
            const rodDy = -rodLength * Math.sin(rodAngle);

            // 设置金属棒位置（中心在圆弧上，方向垂直于半径）
            rod.setAttribute('x1', rodCx - rodDx / 2);
            rod.setAttribute('y1', rodCy - rodDy / 2);
            rod.setAttribute('x2', rodCx + rodDx / 2);
            rod.setAttribute('y2', rodCy + rodDy / 2);

            // 速度矢量 v₀：沿切线方向，长度60px
            const v0Length = 60;
            const v0Dx = v0Length * Math.cos(theta);
            const v0Dy = -v0Length * Math.sin(theta);
            arrowV0.setAttribute('x1', rodCx);
            arrowV0.setAttribute('y1', rodCy);
            arrowV0.setAttribute('x2', rodCx + v0Dx);
            arrowV0.setAttribute('y2', rodCy + v0Dy);

            // v₀ 标签位置：箭头末端右上方
            labelV0.setAttribute('x', rodCx + v0Dx + 10);
            labelV0.setAttribute('y', rodCy + v0Dy - 15);

            // 水平分量 vx：水平向右，长度 = 60*cos(θ)
            const vxLength = v0Length * Math.cos(theta);
            arrowVx.setAttribute('x1', rodCx);
            arrowVx.setAttribute('y1', rodCy);
            arrowVx.setAttribute('x2', rodCx + vxLength);
            arrowVx.setAttribute('y2', rodCy);

            // vx 标签位置：箭头下方
            labelVx.setAttribute('x', rodCx + vxLength / 2);
            labelVx.setAttribute('y', rodCy + 20);

            // 竖直分量 vy：竖直向上，长度 = 60*sin(θ)
            const vyLength = v0Length * Math.sin(theta);
            arrowVy.setAttribute('x1', rodCx);
            arrowVy.setAttribute('y1', rodCy);
            arrowVy.setAttribute('x2', rodCx);
            arrowVy.setAttribute('y2', rodCy - vyLength);

            // vy 标签位置：箭头右侧
            labelVy.setAttribute('x', rodCx + 15);
            labelVy.setAttribute('y', rodCy - vyLength / 2);

            // 角度θ弧线：从270°到(270°+θ)
            const angleArcRadius = 40;
            const startAngle = 270 * Math.PI / 180;
            const endAngle = (270 + thetaDeg) * Math.PI / 180;
            const arcStartX = CX + angleArcRadius * Math.cos(startAngle);
            const arcStartY = CY + angleArcRadius * Math.sin(startAngle);
            const arcEndX = CX + angleArcRadius * Math.cos(endAngle);
            const arcEndY = CY + angleArcRadius * Math.sin(endAngle);
            const arcPath = `M ${CX} ${CY + angleArcRadius} A ${angleArcRadius} ${angleArcRadius} 0 ${thetaDeg > 0 ? '1' : '0'} 1 ${arcEndX} ${arcEndY}`;
            angleArc.setAttribute('d', arcPath);

            // 角度θ标签位置
            const thetaLabelR = 50;
            const thetaLabelAngle = (270 + thetaDeg / 2) * Math.PI / 180;
            labelTheta.setAttribute('x', CX + thetaLabelR * Math.cos(thetaLabelAngle) + 10);
            labelTheta.setAttribute('y', CY + thetaLabelR * Math.sin(thetaLabelAngle));

            // 更新滑块数值显示
            sliderValue.textContent = `θ = ${thetaDeg}°`;

            // 更新公式显示（假设 B=1, L=1, v₀=1 用于演示）
            const E = Math.cos(theta);
            formulaValue.textContent = `E = BLv₀cosθ = ${E.toFixed(2)} BLv₀`;
        }

        // 事件监听
        slider.addEventListener('input', function() {
            update(parseInt(this.value));
        });

        // 初始化
        update(0);
    })();
    </script>
