    <script>
    (function() {
        // Scene 2 实现：状态丙的压强平衡与活塞位置计算
        
        // 布局坐标表（对照依据）：
        // 玻璃管轮廓: (370,80)-(430,430), 60×350
        // 上端气体: (372,82)-(428,280), 56×198
        // 水银柱: (372,280)-(428,308), 56×28
        // 下端气体: (372,308)-(428,422), 56×114
        // 活塞: (365,422)-(435,422)
        // 尺寸标注线: x=450
        // 压强标注: x=340 (P₀在 x=30)
        // 方程区: y=475 居中
        
        let currentStep = 0;
        const totalSteps = 5;
        
        // 每步的可见元素配置
        const stepConfig = [
            // 步骤 0：初始状态 - 仅显示基础图形
            {
                visible: ['scene2-tube', 'scene2-tube-top', 'scene2-tube-bottom', 
                          'scene2-gas-upper', 'scene2-mercury', 'scene2-gas-lower', 'scene2-piston'],
                equation: ''
            },
            // 步骤 1：标注上端气体
            {
                visible: ['scene2-tube', 'scene2-tube-top', 'scene2-tube-bottom',
                          'scene2-gas-upper', 'scene2-mercury', 'scene2-gas-lower', 'scene2-piston',
                          'scene2-dim-l1', 'scene2-pressure-p3'],
                equation: '上端气体：l₁ = 35cm, 压强 P₃'
            },
            // 步骤 2：标注水银柱压强关系
            {
                visible: ['scene2-tube', 'scene2-tube-top', 'scene2-tube-bottom',
                          'scene2-gas-upper', 'scene2-mercury', 'scene2-gas-lower', 'scene2-piston',
                          'scene2-dim-l1', 'scene2-pressure-p3', 'scene2-dim-h', 'scene2-pressure-p4'],
                equation: '压强平衡：P₃ = P₄ + 5cmHg'
            },
            // 步骤 3：标注下端气体
            {
                visible: ['scene2-tube', 'scene2-tube-top', 'scene2-tube-bottom',
                          'scene2-gas-upper', 'scene2-mercury', 'scene2-gas-lower', 'scene2-piston',
                          'scene2-dim-l1', 'scene2-pressure-p3', 'scene2-dim-h', 'scene2-pressure-p4',
                          'scene2-dim-l4'],
                equation: '下端气体：压强 P₄, 长度 l₄ = ?'
            },
            // 步骤 4：标注活塞位置与几何约束
            {
                visible: ['scene2-tube', 'scene2-tube-top', 'scene2-tube-bottom',
                          'scene2-gas-upper', 'scene2-mercury', 'scene2-gas-lower', 'scene2-piston',
                          'scene2-dim-l1', 'scene2-pressure-p3', 'scene2-dim-h', 'scene2-pressure-p4',
                          'scene2-dim-l4', 'scene2-dim-d', 'scene2-pressure-p0'],
                equation: '几何约束：62 = 35 + 5 + l₄ + d'
            },
            // 步骤 5：展示完整方程组和解
            {
                visible: ['scene2-tube', 'scene2-tube-top', 'scene2-tube-bottom',
                          'scene2-gas-upper', 'scene2-mercury', 'scene2-gas-lower', 'scene2-piston',
                          'scene2-dim-l1', 'scene2-pressure-p3', 'scene2-dim-h', 'scene2-pressure-p4',
                          'scene2-dim-l4', 'scene2-dim-d', 'scene2-pressure-p0'],
                equation: 'P₃=80cmHg → P₄=75cmHg → l₄=20.53cm → d=1.47cm'
            }
        ];
        
        // 更新元素可见性
        function updateVisibility() {
            const config = stepConfig[currentStep];
            
            // 获取所有元素
            const allElements = [
                'scene2-tube', 'scene2-tube-top', 'scene2-tube-bottom',
                'scene2-gas-upper', 'scene2-mercury', 'scene2-gas-lower', 'scene2-piston',
                'scene2-dim-l1', 'scene2-dim-h', 'scene2-dim-l4', 'scene2-dim-d',
                'scene2-pressure-p3', 'scene2-pressure-p4', 'scene2-pressure-p0',
                'scene2-equation-text'
            ];
            
            // 更新每个元素的可见性
            allElements.forEach(id => {
                const el = document.getElementById(id);
                if (el) {
                    if (config.visible.includes(id)) {
                        el.classList.remove('scene2-hidden');
                        el.classList.add('scene2-visible');
                    } else {
                        el.classList.remove('scene2-visible');
                        el.classList.add('scene2-hidden');
                    }
                }
            });
            
            // 更新方程文本
            const eqText = document.getElementById('scene2-equation-text');
            if (eqText) {
                eqText.textContent = config.equation;
            }
            
            // 更新步骤指示器
            const indicator = document.getElementById('scene2-step-indicator');
            if (indicator) {
                indicator.textContent = `步骤 ${currentStep}/${totalSteps}`;
            }
            
            // 更新按钮状态
            const prevBtn = document.getElementById('scene2-btn-prev');
            const nextBtn = document.getElementById('scene2-btn-next');
            if (prevBtn) prevBtn.disabled = currentStep === 0;
            if (nextBtn) nextBtn.disabled = currentStep === totalSteps;
        }
        
        // 步骤切换
        function goToStep(step) {
            if (step < 0 || step > totalSteps) return;
            currentStep = step;
            updateVisibility();
        }
        
        // 事件绑定
        document.getElementById('scene2-btn-prev').addEventListener('click', () => {
            goToStep(currentStep - 1);
        });
        
        document.getElementById('scene2-btn-next').addEventListener('click', () => {
            goToStep(currentStep + 1);
        });
        
        // 初始化
        updateVisibility();
    })();
    </script>
