    <script>
    (function() {
        // Scene 2: 平行板电场中类平抛运动的速度矢量分解
        // 布局坐标表（实现对照依据）：
        // | 元素 | 类型 | 位置/范围 | 大小 | 标签偏移方向 | 备注 |
        // |------|------|-----------|------|-------------|------|
        // | 坐标轴 x | 虚线 | (150,250)-(550,250) | 400×1 | - | 水平轴，灰色 |
        // | 坐标轴 y | 虚线 | (350,100)-(350,400) | 1×300 | - | 竖直轴，灰色 |
        // | 原点 O' | 圆点 | (350,250) | r=4 | 左下 | 矢量起点 |
        // | v₀矢量 | 箭头 | (350,250)-(500,250) | 150×3 | 上方 | 蓝色，水平向右 |
        // | v_y 矢量 | 箭头 | (350,250)-(350,370) | 3×120 | 右侧 | 绿色，竖直向下 |
        // | v 合矢量 | 箭头 | (350,250)-(500,370) | 150×120 | 上方 | 红色，斜向右下 |
        // | 平行四边形辅助线 1 | 虚线 | (500,250)-(500,370) | 1×120 | - | 灰色，竖直 |
        // | 平行四边形辅助线 2 | 虚线 | (350,370)-(500,370) | 150×1 | - | 灰色，水平 |
        // | 偏转角θ弧 | 圆弧 | 圆心 (350,250), r=50 | 弧长~40 | 弧外侧 | 红色，从 x 轴到 v |
        // | θ标签 | text | (390,285) | ~30×16 | - | 红色，在弧外侧 |
        // | v₀标签 | text | (425,230) | ~40×16 | 上方 | 蓝色 |
        // | v_y 标签 | text | (360,310) | ~30×16 | 右侧 | 绿色 |
        // | v 标签 | text | (440,340) | ~25×16 | 右下 | 红色 |
        // | 数值框 1 | 矩形 | (580,150)-(760,200) | 180×50 | - | v₀数值 |
        // | 数值框 2 | 矩形 | (580,210)-(760,260) | 180×50 | - | v_y 数值 |
        // | 数值框 3 | 矩形 | (580,270)-(760,320) | 180×50 | - | v 数值 |
        // | 数值框 4 | 矩形 | (580,330)-(760,380) | 180×50 | - | θ数值 |
        // | tanθ公式 | text | (590,400) | ~150×16 | - | tanθ=v_y/v₀ |
        // | 控件栏 | 按钮组 | (250,450)-(550,490) | 300×40 | - | 上一步/下一步按钮 |

        let currentStep = 0;
        const totalSteps = 5;

        // 获取所有元素
        const elements = {
            // 始终显示
            axisX: document.getElementById('scene2-axis-x'),
            axisY: document.getElementById('scene2-axis-y'),
            origin: document.getElementById('scene2-origin'),
            originLabel: document.getElementById('scene2-origin-label'),
            // 步骤 0: v₀
            vecV0: document.getElementById('scene2-vec-v0'),
            labelV0: document.getElementById('scene2-label-v0'),
            box1: document.getElementById('scene2-box1'),
            box1Text: document.getElementById('scene2-box1-text'),
            // 步骤 1: v_y
            vecVy: document.getElementById('scene2-vec-vy'),
            labelVy: document.getElementById('scene2-label-vy'),
            box2: document.getElementById('scene2-box2'),
            box2Text: document.getElementById('scene2-box2-text'),
            // 步骤 2: v 和平行四边形
            vecV: document.getElementById('scene2-vec-v'),
            labelV: document.getElementById('scene2-label-v'),
            paraLine1: document.getElementById('scene2-para-line1'),
            paraLine2: document.getElementById('scene2-para-line2'),
            // 步骤 3: θ
            angleArc: document.getElementById('scene2-angle-arc'),
            labelTheta: document.getElementById('scene2-label-theta'),
            formula: document.getElementById('scene2-formula'),
            // 步骤 4: 完整数值
            box3: document.getElementById('scene2-box3'),
            box3Text: document.getElementById('scene2-box3-text'),
            box4: document.getElementById('scene2-box4'),
            box4Text: document.getElementById('scene2-box4-text')
        };

        const btnPrev = document.getElementById('scene2-btn-prev');
        const btnNext = document.getElementById('scene2-btn-next');
        const stepIndicator = document.getElementById('scene2-step-indicator');

        // 辅助函数：显示元素
        function show(el) {
            if (el) {
                el.style.display = 'block';
                el.style.opacity = '1';
            }
        }

        // 辅助函数：隐藏元素
        function hide(el) {
            if (el) {
                el.style.display = 'none';
                el.style.opacity = '0';
            }
        }

        // 更新显示状态
        function updateDisplay() {
            // 始终显示：坐标轴、原点
            show(elements.axisX);
            show(elements.axisY);
            show(elements.origin);
            show(elements.originLabel);

            // 步骤 0: 只显示 v₀
            if (currentStep >= 0) {
                show(elements.vecV0);
                show(elements.labelV0);
                show(elements.box1);
                show(elements.box1Text);
            } else {
                hide(elements.vecV0);
                hide(elements.labelV0);
                hide(elements.box1);
                hide(elements.box1Text);
            }

            // 步骤 1: 显示 v_y
            if (currentStep >= 1) {
                show(elements.vecVy);
                show(elements.labelVy);
                show(elements.box2);
                show(elements.box2Text);
            } else {
                hide(elements.vecVy);
                hide(elements.labelVy);
                hide(elements.box2);
                hide(elements.box2Text);
            }

            // 步骤 2: 显示 v 和平行四边形
            if (currentStep >= 2) {
                show(elements.vecV);
                show(elements.labelV);
                show(elements.paraLine1);
                show(elements.paraLine2);
            } else {
                hide(elements.vecV);
                hide(elements.labelV);
                hide(elements.paraLine1);
                hide(elements.paraLine2);
            }

            // 步骤 3: 显示θ
            if (currentStep >= 3) {
                show(elements.angleArc);
                show(elements.labelTheta);
                show(elements.formula);
            } else {
                hide(elements.angleArc);
                hide(elements.labelTheta);
                hide(elements.formula);
            }

            // 步骤 4: 显示完整数值
            if (currentStep >= 4) {
                show(elements.box3);
                show(elements.box3Text);
                show(elements.box4);
                show(elements.box4Text);
                // 更新数值
                elements.box3Text.textContent = 'v ≈ 2.5×10⁶ m/s';
                elements.box4Text.textContent = 'θ ≈ 37°';
            } else {
                hide(elements.box3);
                hide(elements.box3Text);
                hide(elements.box4);
                hide(elements.box4Text);
            }

            // 更新按钮状态
            btnPrev.disabled = (currentStep === 0);
            btnNext.disabled = (currentStep === totalSteps - 1);

            // 更新步骤指示器
            stepIndicator.textContent = '步骤 ' + (currentStep + 1) + '/' + totalSteps;
        }

        // 按钮事件
        btnPrev.addEventListener('click', function() {
            if (currentStep > 0) {
                currentStep--;
                updateDisplay();
            }
        });

        btnNext.addEventListener('click', function() {
            if (currentStep < totalSteps - 1) {
                currentStep++;
                updateDisplay();
            }
        });

        // 初始化显示
        updateDisplay();
    })();
    </script>
