    <script>
    (function() {
        // ===== Scene 2: 功率计算与亮度对比 =====
        
        // 状态变量
        const state = {
            voltage: 3.0,  // 电源电压 0-6V
            r1: 6.0,       // R₁阻值 0-10Ω
            r2: 3.0        // R₂阻值 0-5Ω
        };

        // 滑块配置
        const sliderConfig = {
            voltage: { min: 0, max: 6, step: 0.1, initial: 3.0, trackX: 40, trackWidth: 220 },
            r1: { min: 0, max: 10, step: 0.1, initial: 6.0, trackX: 300, trackWidth: 220 },
            r2: { min: 0, max: 5, step: 0.1, initial: 3.0, trackX: 560, trackWidth: 200 }
        };

        // DOM 元素引用
        const elements = {
            voltageHandle: document.getElementById('scene2-slider-voltage-handle'),
            voltageValue: document.getElementById('scene2-value-voltage'),
            r1Handle: document.getElementById('scene2-slider-r1-handle'),
            r1Value: document.getElementById('scene2-value-r1'),
            r2Handle: document.getElementById('scene2-slider-r2-handle'),
            r2Value: document.getElementById('scene2-value-r2'),
            resultP1: document.getElementById('scene2-result-p1'),
            resultP2: document.getElementById('scene2-result-p2'),
            barL1: document.getElementById('scene2-bar-l1'),
            barL2: document.getElementById('scene2-bar-l2'),
            conclusion: document.getElementById('scene2-conclusion'),
            lampL1: document.getElementById('scene2-lamp-l1'),
            lampL2: document.getElementById('scene2-lamp-l2'),
            lampL1Glow: document.getElementById('scene2-lamp-l1-glow'),
            lampL2Glow: document.getElementById('scene2-lamp-l2-glow')
        };

        // 计算功率
        function calculatePower() {
            const u = state.voltage;
            const r1 = Math.max(state.r1, 0.1);  // 避免除零
            const r2 = Math.max(state.r2, 0.1);
            
            const p1 = (u * u) / r1;
            const p2 = (u * u) / r2;
            
            return { p1, p2 };
        }

        // 更新显示
        function updateDisplay() {
            const { p1, p2 } = calculatePower();
            
            // 更新数值显示
            elements.voltageValue.textContent = state.voltage.toFixed(1) + 'V';
            elements.r1Value.textContent = state.r1.toFixed(1) + 'Ω';
            elements.r2Value.textContent = state.r2.toFixed(1) + 'Ω';
            
            // 更新公式结果
            elements.resultP1.textContent = p1.toFixed(2) + 'W';
            elements.resultP2.textContent = p2.toFixed(2) + 'W';
            
            // 更新柱状图高度 (最大 6W 对应 120px)
            const maxPower = 6;
            const chartHeight = 120;
            const baseY = 410;
            
            const h1 = Math.min((p1 / maxPower) * chartHeight, chartHeight);
            const h2 = Math.min((p2 / maxPower) * chartHeight, chartHeight);
            
            elements.barL1.setAttribute('height', h1.toFixed(1));
            elements.barL1.setAttribute('y', (baseY - h1).toFixed(1));
            
            elements.barL2.setAttribute('height', h2.toFixed(1));
            elements.barL2.setAttribute('y', (baseY - h2).toFixed(1));
            
            // 更新结论
            if (p1 > p2 + 0.01) {
                elements.conclusion.textContent = 'L₁更亮';
            } else if (p2 > p1 + 0.01) {
                elements.conclusion.textContent = 'L₂更亮';
            } else {
                elements.conclusion.textContent = '一样亮';
            }
            
            // 更新灯泡亮度（根据功率）
            const maxBrightness = 0.8;
            const brightness1 = Math.min((p1 / 6) * maxBrightness, maxBrightness);
            const brightness2 = Math.min((p2 / 6) * maxBrightness, maxBrightness);
            
            elements.lampL1Glow.style.opacity = brightness1.toFixed(2);
            elements.lampL2Glow.style.opacity = brightness2.toFixed(2);
        }

        // 滑块拖拽处理
        function setupSlider(handle, valueKey, config) {
            let isDragging = false;
            
            handle.addEventListener('mousedown', function(e) {
                isDragging = true;
                e.preventDefault();
            });
            
            document.addEventListener('mousemove', function(e) {
                if (!isDragging) return;
                
                const svg = document.getElementById('scene2-svg');
                const rect = svg.getBoundingClientRect();
                const mouseX = e.clientX - rect.left;
                
                // 计算滑块位置（限制在轨道范围内）
                let newX = mouseX;
                const minX = config.trackX;
                const maxX = config.trackX + config.trackWidth;
                
                if (newX < minX) newX = minX;
                if (newX > maxX) newX = maxX;
                
                // 更新手柄位置
                handle.setAttribute('cx', newX);
                
                // 计算对应的值
                const ratio = (newX - minX) / config.trackWidth;
                const value = config.min + ratio * (config.max - config.min);
                
                // 四舍五入到步长
                const steppedValue = Math.round(value / config.step) * config.step;
                state[valueKey] = Math.max(config.min, Math.min(config.max, steppedValue));
                
                updateDisplay();
            });
            
            document.addEventListener('mouseup', function() {
                isDragging = false;
            });
            
            // 触摸支持
            handle.addEventListener('touchstart', function(e) {
                isDragging = true;
                e.preventDefault();
            });
            
            document.addEventListener('touchmove', function(e) {
                if (!isDragging) return;
                
                const svg = document.getElementById('scene2-svg');
                const rect = svg.getBoundingClientRect();
                const touchX = e.touches[0].clientX - rect.left;
                
                let newX = touchX;
                const minX = config.trackX;
                const maxX = config.trackX + config.trackWidth;
                
                if (newX < minX) newX = minX;
                if (newX > maxX) newX = maxX;
                
                handle.setAttribute('cx', newX);
                
                const ratio = (newX - minX) / config.trackWidth;
                const value = config.min + ratio * (config.max - config.min);
                const steppedValue = Math.round(value / config.step) * config.step;
                state[valueKey] = Math.max(config.min, Math.min(config.max, steppedValue));
                
                updateDisplay();
                e.preventDefault();
            });
            
            document.addEventListener('touchend', function() {
                isDragging = false;
            });
        }

        // 初始化滑块
        setupSlider(elements.voltageHandle, 'voltage', sliderConfig.voltage);
        setupSlider(elements.r1Handle, 'r1', sliderConfig.r1);
        setupSlider(elements.r2Handle, 'r2', sliderConfig.r2);

        // 初始显示
        updateDisplay();

    })();
    </script>
