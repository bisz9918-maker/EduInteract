    <script>
    (function() {
        'use strict';

        var currentStep = 0;
        var maxStep = 4;

        // DOM references
        var btnPrev = document.getElementById('scene1-btn-prev');
        var btnNext = document.getElementById('scene1-btn-next');
        var stepIndicator = document.getElementById('scene1-step-indicator');
        var formulas = document.getElementById('scene1-formulas');
        var h2Extra1 = document.getElementById('scene1-h2-extra-1');
        var h2Extra2 = document.getElementById('scene1-h2-extra-2');
        var equation = document.getElementById('scene1-equation');
        var eqStep2 = document.getElementById('scene1-eq-step2');
        var eqStep3 = document.getElementById('scene1-eq-step3');
        var eqStep4 = document.getElementById('scene1-eq-step4');

        function updateUI() {
            // Button states
            btnPrev.disabled = (currentStep === 0);
            btnNext.disabled = (currentStep === maxStep);
            stepIndicator.textContent = '步骤 ' + currentStep + ' / ' + maxStep;

            // Step 1+: show formulas
            if (currentStep >= 1) {
                formulas.style.opacity = '1';
            } else {
                formulas.style.opacity = '0';
            }

            // Step 3+: show extra H2 molecules
            if (currentStep >= 3) {
                h2Extra1.style.opacity = '1';
                h2Extra2.style.opacity = '1';
            } else {
                h2Extra1.style.opacity = '0';
                h2Extra2.style.opacity = '0';
            }

            // Step 2+: show equation group
            if (currentStep >= 2) {
                equation.style.opacity = '1';
            } else {
                equation.style.opacity = '0';
            }

            // Equation step variants
            if (currentStep === 2) {
                eqStep2.style.opacity = '1';
                eqStep3.style.opacity = '0';
                eqStep4.style.opacity = '0';
            } else if (currentStep === 3) {
                eqStep2.style.opacity = '0';
                eqStep3.style.opacity = '1';
                eqStep4.style.opacity = '0';
            } else if (currentStep === 4) {
                eqStep2.style.opacity = '0';
                eqStep3.style.opacity = '0';
                eqStep4.style.opacity = '1';
            } else {
                eqStep2.style.opacity = '0';
                eqStep3.style.opacity = '0';
                eqStep4.style.opacity = '0';
            }
        }

        function nextStep() {
            if (currentStep < maxStep) {
                currentStep++;
                updateUI();
            }
        }

        function prevStep() {
            if (currentStep > 0) {
                currentStep--;
                updateUI();
            }
        }

        btnNext.addEventListener('click', nextStep);
        btnPrev.addEventListener('click', prevStep);

        // Initialize
        updateUI();

    })();
    </script>
