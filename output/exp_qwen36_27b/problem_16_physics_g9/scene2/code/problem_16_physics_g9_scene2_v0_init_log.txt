    <script>
    (function() {
        'use strict';

        var currentStep = 0;
        var totalSteps = 5;

        function setClass(el, className) {
            el.setAttribute('class', className);
        }

        function showEl(el) {
            el.style.opacity = '1';
            el.style.display = '';
        }
        function hideEl(el) {
            el.style.opacity = '0';
            el.style.display = 'none';
        }

        function updateArrows(step) {
            for (var i = 0; i < 5; i++) {
                var arrow = document.getElementById('scene2-arrow-' + i);
                if (i < step) {
                    setClass(arrow, 'arrow-active');
                    arrow.setAttribute('marker-end', 'url(#scene2-arrowhead)');
                } else {
                    setClass(arrow, 'arrow-inactive');
                    arrow.setAttribute('marker-end', 'url(#scene2-arrowhead-gray)');
                }
            }
        }

        function updateNodes(step) {
            for (var i = 0; i <= 5; i++) {
                var node = document.getElementById('scene2-node-' + i);
                var textEl = document.getElementById('scene2-node-' + i + '-text');
                var textEl2 = document.getElementById('scene2-node-' + i + '-text2');

                if (i < step) {
                    setClass(node, 'node-done');
                    setClass(textEl, 'node-text-done');
                    if (textEl2) setClass(textEl2, 'node-text-done');
                } else if (i === step) {
                    setClass(node, 'node-active');
                    setClass(textEl, 'node-text-active');
                    if (textEl2) setClass(textEl2, 'node-text-active');
                } else {
                    setClass(node, 'node-inactive');
                    setClass(textEl, 'node-text-inactive');
                    if (textEl2) setClass(textEl2, 'node-text-inactive');
                }
            }
        }

        function updateOptions(step) {
            if (step < 5) {
                for (var i = 0; i < 4; i++) {
                    var optId = ['a', 'b', 'c', 'd'][i];
                    setClass(document.getElementById('scene2-option-' + optId), 'option-inactive');
                    setClass(document.getElementById('scene2-option-' + optId + '-text'), 'option-text-inactive');
                    hideEl(document.getElementById('scene2-mark-' + optId));
                }
            } else {
                // A: 错误
                setClass(document.getElementById('scene2-option-a'), 'option-wrong');
                setClass(document.getElementById('scene2-option-a-text'), 'option-text-wrong');
                showEl(document.getElementById('scene2-mark-a'));

                // B: 错误
                setClass(document.getElementById('scene2-option-b'), 'option-wrong');
                setClass(document.getElementById('scene2-option-b-text'), 'option-text-wrong');
                showEl(document.getElementById('scene2-mark-b'));

                // C: 正确
                setClass(document.getElementById('scene2-option-c'), 'option-correct');
                setClass(document.getElementById('scene2-option-c-text'), 'option-text-correct');
                showEl(document.getElementById('scene2-mark-c'));

                // D: 错误
                setClass(document.getElementById('scene2-option-d'), 'option-wrong');
                setClass(document.getElementById('scene2-option-d-text'), 'option-text-wrong');
                showEl(document.getElementById('scene2-mark-d'));
            }
        }

        function updateConclusion(step) {
            var box = document.getElementById('scene2-conclusion-box');
            var text = document.getElementById('scene2-conclusion-text');

            if (step < 5) {
                setClass(box, 'conclusion-inactive');
                setClass(text, 'conclusion-text-inactive');
                hideEl(text);
            } else {
                setClass(box, 'conclusion-active');
                setClass(text, 'conclusion-text-active');
                showEl(text);
            }
        }

        function updateButtons(step) {
            var prevBg = document.getElementById('scene2-prev-btn-bg');
            var prevText = document.getElementById('scene2-prev-btn-text');
            var nextBg = document.getElementById('scene2-next-btn-bg');
            var nextText = document.getElementById('scene2-next-btn-text');

            if (step <= 0) {
                setClass(prevBg, 'btn-disabled');
                setClass(prevText, 'btn-text-disabled');
            } else {
                setClass(prevBg, 'btn-active');
                setClass(prevText, 'btn-text-active');
            }

            if (step >= totalSteps) {
                setClass(nextBg, 'btn-disabled');
                setClass(nextText, 'btn-text-disabled');
            } else {
                setClass(nextBg, 'btn-active');
                setClass(nextText, 'btn-text-active');
            }
        }

        function updateStep(step) {
            currentStep = step;
            updateNodes(step);
            updateArrows(step);
            updateOptions(step);
            updateConclusion(step);
            updateButtons(step);
        }

        // 事件绑定 - 绑定到 rect 上，因为 text 有 pointer-events: none
        document.getElementById('scene2-prev-btn-bg').addEventListener('click', function() {
            if (currentStep > 0) {
                updateStep(currentStep - 1);
            }
        });

        document.getElementById('scene2-next-btn-bg').addEventListener('click', function() {
            if (currentStep < totalSteps) {
                updateStep(currentStep + 1);
            }
        });

        // 初始化
        updateStep(0);

    })();
    </script>
