    <script>
    (function() { 'use strict';

        // ===== 物理模型参数 =====
        // 铅笔芯总长 L=24cm，对应490px，比例 k=490/24≈20.417px/cm
        var L = 24;           // 总长度 cm
        var k = 490 / 24;     // px/cm ≈ 20.417
        var A_x = 100;        // A端x坐标
        var B_x = 590;        // B端x坐标
        var U_factor = 3.6;   // I = U_factor / x_cm

        // ===== 量程参数 =====
        var ranges = {
            '0.6': { maxI: 0.6, c_cm: 6, d_cm: 18, label: '0-0.6A', scope_cm: 12.0 },
            '3':   { maxI: 3.0, c_cm: 1.2, d_cm: 22.8, label: '0-3A', scope_cm: 21.6 }
        };

        // ===== 状态变量 =====
        var currentRange = '0.6';
        var pPos = 345;       // 滑片P的x坐标，初始中点

        // ===== DOM元素引用 =====
        var svg = document.getElementById('scene1-svg');
        var sliderP = document.getElementById('scene1-slider-p');
        var labelP = document.getElementById('scene1-label-p');
        var apSegment = document.getElementById('scene1-ap-segment');
        var pbSegment = document.getElementById('scene1-pb-segment');
        var pointC = document.getElementById('scene1-point-c');
        var pointD = document.getElementById('scene1-point-d');
        var labelC = document.getElementById('scene1-label-c');
        var labelD = document.getElementById('scene1-label-d');
        var rangeLine = document.getElementById('scene1-range-line');
        var rangeLineLeft = document.getElementById('scene1-range-line-left');
        var rangeLineRight = document.getElementById('scene1-range-line-right');
        var switchToP = document.getElementById('scene1-switch-to-p');
        var ammeterA1 = document.getElementById('scene1-ammeter-a1');
        var ammeterA2 = document.getElementById('scene1-ammeter-a2');
        var displayI1 = document.getElementById('scene1-display-i1');
        var displayI2 = document.getElementById('scene1-display-i2');
        var displayRatio = document.getElementById('scene1-display-ratio');
        var displayRange = document.getElementById('scene1-display-range');
        var displayScope = document.getElementById('scene1-display-scope');
        var warning = document.getElementById('scene1-warning');
        var btnRange06 = document.getElementById('scene1-btn-range06');
        var btnRange3 = document.getElementById('scene1-btn-range3');

        // ===== 计算函数 =====
        function calcCurrents(px) {
            var x_cm = (px - A_x) / k;
            var x1_cm = x_cm;          // AP段长度
            var x2_cm = L - x_cm;      // PB段长度
            // 防止除零
            if (x1_cm < 0.01) x1_cm = 0.01;
            if (x2_cm < 0.01) x2_cm = 0.01;
            var i1 = U_factor / x1_cm;
            var i2 = U_factor / x2_cm;
            return { i1: i1, i2: i2, x1_cm: x1_cm, x2_cm: x2_cm };
        }

        function getCdPositions(rangeKey) {
            var r = ranges[rangeKey];
            var c_x = A_x + r.c_cm * k;
            var d_x = A_x + r.d_cm * k;
            return { c_x: c_x, d_x: d_x };
        }

        // ===== 更新显示 =====
        function updateDisplay() {
            var r = ranges[currentRange];
            var cd = getCdPositions(currentRange);
            var currents = calcCurrents(pPos);

            // 更新AP/PB段高亮
            apSegment.setAttribute('x', A_x);
            apSegment.setAttribute('width', Math.max(0, pPos - A_x));
            pbSegment.setAttribute('x', pPos);
            pbSegment.setAttribute('width', Math.max(0, B_x - pPos));

            // 更新滑片P位置
            var poly = sliderP.querySelector('polygon');
            poly.setAttribute('points', pPos + ',370 ' + (pPos - 8) + ',350 ' + (pPos + 8) + ',350');
            var stemLine = sliderP.querySelector('line');
            stemLine.setAttribute('x1', pPos);
            stemLine.setAttribute('x2', pPos);
            labelP.setAttribute('x', pPos);

            // 更新开关到滑片的垂直导线
            switchToP.setAttribute('x1', pPos);
            switchToP.setAttribute('x2', pPos);

            // 更新c/d点位置
            pointC.setAttribute('cx', cd.c_x);
            pointD.setAttribute('cx', cd.d_x);
            labelC.setAttribute('x', cd.c_x - 5);
            labelD.setAttribute('x', cd.d_x - 5);

            // 更新cd范围标记线
            rangeLine.setAttribute('x1', cd.c_x);
            rangeLine.setAttribute('x2', cd.d_x);
            rangeLineLeft.setAttribute('x1', cd.c_x);
            rangeLineLeft.setAttribute('x2', cd.c_x);
            rangeLineRight.setAttribute('x1', cd.d_x);
            rangeLineRight.setAttribute('x2', cd.d_x);

            // 更新数据面板
            displayI1.textContent = 'I₁ = ' + currents.i1.toFixed(2) + 'A';
            displayI2.textContent = 'I₂ = ' + currents.i2.toFixed(2) + 'A';

            // 比值
            var ratio;
            if (currents.i1 >= currents.i2) {
                ratio = (currents.i1 / currents.i2);
                displayRatio.textContent = 'I₁:I₂ = ' + ratio.toFixed(2) + ':1';
            } else {
                ratio = (currents.i2 / currents.i1);
                displayRatio.textContent = 'I₁:I₂ = 1:' + ratio.toFixed(2);
            }

            displayRange.textContent = '量程：' + r.label;
            displayScope.textContent = '范围：' + r.scope_cm.toFixed(1) + 'cm';

            // 超限检测
            var overLimit = false;
            if (currents.i1 > r.maxI || currents.i2 > r.maxI) {
                overLimit = true;
            }

            // 警告显示
            if (overLimit) {
                warning.setAttribute('opacity', '1');
            } else {
                warning.setAttribute('opacity', '0');
            }

            // 电流表超限高亮
            if (currents.i1 > r.maxI) {
                ammeterA1.setAttribute('stroke', '#d32f2f');
                ammeterA1.setAttribute('stroke-width', '3');
            } else {
                ammeterA1.setAttribute('stroke', '#000');
                ammeterA1.setAttribute('stroke-width', '2');
            }
            if (currents.i2 > r.maxI) {
                ammeterA2.setAttribute('stroke', '#d32f2f');
                ammeterA2.setAttribute('stroke-width', '3');
            } else {
                ammeterA2.setAttribute('stroke', '#000');
                ammeterA2.setAttribute('stroke-width', '2');
            }
        }

        // ===== 拖拽实现 =====
        var isDragging = false;

        function getMouseX(e) {
            var rect = svg.getBoundingClientRect();
            var scaleX = 800 / rect.width;
            if (e.touches) {
                return (e.touches[0].clientX - rect.left) * scaleX;
            }
            return (e.clientX - rect.left) * scaleX;
        }

        function onDragStart(e) {
            e.preventDefault();
            isDragging = true;
            sliderP.style.cursor = 'grabbing';
        }

        function onDragMove(e) {
            if (!isDragging) return;
            e.preventDefault();
            var mx = getMouseX(e);
            // 限制在铅笔芯范围内
            pPos = Math.max(A_x, Math.min(B_x, mx));
            updateDisplay();
        }

        function onDragEnd(e) {
            isDragging = false;
            sliderP.style.cursor = 'grab';
        }

        // 鼠标事件
        sliderP.addEventListener('mousedown', onDragStart);
        svg.addEventListener('mousemove', onDragMove);
        document.addEventListener('mouseup', onDragEnd);

        // 触摸事件
        sliderP.addEventListener('touchstart', onDragStart, { passive: false });
        svg.addEventListener('touchmove', onDragMove, { passive: false });
        document.addEventListener('touchend', onDragEnd);

        // ===== 量程切换 =====
        function switchRange(rangeKey) {
            currentRange = rangeKey;
            if (rangeKey === '0.6') {
                btnRange06.classList.add('active');
                btnRange3.classList.remove('active');
            } else {
                btnRange3.classList.add('active');
                btnRange06.classList.remove('active');
            }
            updateDisplay();
        }

        btnRange06.addEventListener('click', function() { switchRange('0.6'); });
        btnRange3.addEventListener('click', function() { switchRange('3'); });

        // ===== 初始化 =====
        updateDisplay();

    })();
    </script>
