    <script>
    (function() {
        'use strict';

        let currentStep = 0;
        const totalSteps = 5;

        // Element references (all prefixed with scene2-)
        var els = {
            lineOA: document.getElementById('scene2-line-oa'),
            labelRn: document.getElementById('scene2-label-rn'),
            lineOM: document.getElementById('scene2-line-om'),
            labelRn1: document.getElementById('scene2-label-rn1'),
            angleArc: document.getElementById('scene2-angle-arc'),
            label30: document.getElementById('scene2-label-30'),
            lineAM: document.getElementById('scene2-line-am'),
            rightAngle: document.getElementById('scene2-right-angle'),
            recurrence: document.getElementById('scene2-recurrence-overlay'),
            circleInner: document.getElementById('scene2-circle-inner'),
            formulaBox: document.getElementById('scene2-formula-overlay'),
            btnPrev: document.getElementById('scene2-btn-prev'),
            btnNext: document.getElementById('scene2-btn-next'),
            stepIndicator: document.getElementById('scene2-step-indicator'),
        };

        // Step definitions: which elements to show at each step
        var stepElements = {
            1: ['lineOA', 'labelRn'],
            2: ['lineOM', 'labelRn1', 'angleArc', 'label30'],
            3: ['lineAM', 'rightAngle', 'recurrence'],
            4: ['circleInner'],
            5: ['formulaBox'],
        };

        // Elements that need stroke-dashoffset animation
        var lineLengths = {
            lineOA: 160,
            lineOM: 138.6,
            circleInner: 871.5,
        };

        function showEl(name) {
            var el = els[name];
            if (!el) return;
            el.style.opacity = '1';
            el.classList.remove('scene2-hidden');
            el.classList.add('scene2-visible');
            if (lineLengths[name] !== undefined) {
                el.style.strokeDashoffset = '0';
            }
        }

        function hideEl(name) {
            var el = els[name];
            if (!el) return;
            el.style.opacity = '0';
            el.classList.remove('scene2-visible');
            el.classList.add('scene2-hidden');
            if (lineLengths[name] !== undefined) {
                el.style.strokeDashoffset = String(lineLengths[name]);
            }
        }

        function updateStep(step) {
            currentStep = step;

            // Update button states
            els.btnPrev.disabled = (step === 0);
            els.btnNext.disabled = (step === totalSteps);
            els.stepIndicator.textContent = step + '/' + totalSteps;

            // Update element visibility for each step
            for (var s = 1; s <= totalSteps; s++) {
                var shouldShow = (s <= step);
                stepElements[s].forEach(function(name) {
                    // Special case: hide recurrence relation at step 5
                    if (name === 'recurrence' && step >= 5) {
                        hideEl(name);
                    } else if (shouldShow) {
                        showEl(name);
                    } else {
                        hideEl(name);
                    }
                });
            }

            // Re-typeset MathJax for newly visible formulas
            if (window.MathJax && MathJax.typesetPromise) {
                MathJax.typesetPromise().catch(function() {});
            }
        }

        // Event listeners
        els.btnPrev.addEventListener('click', function() {
            if (currentStep > 0) updateStep(currentStep - 1);
        });

        els.btnNext.addEventListener('click', function() {
            if (currentStep < totalSteps) updateStep(currentStep + 1);
        });

        // Initialize at step 0
        updateStep(0);
    })();
    </script>
