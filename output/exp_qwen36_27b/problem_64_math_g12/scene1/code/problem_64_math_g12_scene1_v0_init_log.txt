    <script>
    (function() { 'use strict';

        var currentStep = 0;
        var maxStep = 5;

        // DOM references
        var btnPrev = document.getElementById('scene1-btn-prev');
        var btnNext = document.getElementById('scene1-btn-next');
        var indicator = document.getElementById('scene1-step-indicator');

        // Step groups
        var stepGroups = [
            null, // step 0 has no extra group
            document.getElementById('scene1-step1-group'),
            document.getElementById('scene1-step2-group'),
            document.getElementById('scene1-step3-group'),
            document.getElementById('scene1-step4-group'),
            document.getElementById('scene1-step5-group')
        ];

        function showGroup(el) {
            el.style.display = 'block';
            el.classList.add('scene1-active');
            // Force reflow then add fadein class for transition
            requestAnimationFrame(function() {
                requestAnimationFrame(function() {
                    el.classList.add('scene1-fadein');
                });
            });
        }

        function hideGroup(el) {
            el.classList.remove('scene1-fadein');
            // Wait for transition to finish, then hide
            setTimeout(function() {
                el.style.display = 'none';
                el.classList.remove('scene1-active');
            }, 50);
        }

        function updateDisplay() {
            // Update step groups visibility
            for (var i = 1; i <= maxStep; i++) {
                if (stepGroups[i]) {
                    if (i <= currentStep) {
                        showGroup(stepGroups[i]);
                    } else {
                        hideGroup(stepGroups[i]);
                    }
                }
            }

            // Button states
            btnPrev.disabled = (currentStep <= 0);
            btnNext.disabled = (currentStep >= maxStep);

            // Step indicator
            indicator.textContent = '步骤 ' + currentStep + ' / ' + maxStep;

            // Step 5 flash animation: fade out green flash lines after a moment
            if (currentStep === 5) {
                var acFlash = document.getElementById('scene1-ac-flash');
                var b1d1Flash = document.getElementById('scene1-b1d1-flash');
                acFlash.style.transition = 'none';
                b1d1Flash.style.transition = 'none';
                acFlash.style.opacity = '0.7';
                b1d1Flash.style.opacity = '0.7';
                setTimeout(function() {
                    acFlash.style.transition = 'opacity 1.2s ease';
                    b1d1Flash.style.transition = 'opacity 1.2s ease';
                    acFlash.style.opacity = '0';
                    b1d1Flash.style.opacity = '0';
                }, 1500);
            }
        }

        function nextStep() {
            if (currentStep < maxStep) {
                currentStep++;
                updateDisplay();
            }
        }

        function prevStep() {
            if (currentStep > 0) {
                currentStep--;
                updateDisplay();
            }
        }

        btnNext.addEventListener('click', nextStep);
        btnPrev.addEventListener('click', prevStep);

        // Initial display
        updateDisplay();

    })();
    </script>
