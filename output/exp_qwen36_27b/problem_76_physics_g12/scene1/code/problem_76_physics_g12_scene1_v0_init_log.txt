    <script>
    (function() {
        var currentStep = 0;
        var totalSteps = 4;
        var cPositions = [180, 300, 340, 420, 490];
        var bOffsets = [0, 0, 0, 0, 70];

        var groupC = document.getElementById('scene1-group-c');
        var groupB = document.getElementById('scene1-group-b');
        var arrowV0 = document.getElementById('scene1-arrow-v0');
        var arrowVC = document.getElementById('scene1-arrow-vc');
        var arrowVA = document.getElementById('scene1-arrow-va');
        var arrowVBC = document.getElementById('scene1-arrow-vbc');
        var frictionC = document.getElementById('scene1-friction-c');
        var frictionAB = document.getElementById('scene1-friction-ab');
        var frictionB = document.getElementById('scene1-friction-b');
        var btnPrev = document.getElementById('scene1-btn-prev');
        var btnNext = document.getElementById('scene1-btn-next');
        var stepIndicator = document.getElementById('scene1-step-indicator');

        function showEl(el) { el.style.display = ''; el.style.opacity = '1'; }
        function hideEl(el) { el.style.display = 'none'; el.style.opacity = '0'; }

        function updateScene(step) {
            var cX = cPositions[step];
            var bOff = bOffsets[step];
            var cDx = cX - 180;
            groupC.setAttribute('transform', 'translate(' + cDx + ',0)');
            groupB.setAttribute('transform', 'translate(' + bOff + ',0)');

            hideEl(arrowV0); hideEl(arrowVC); hideEl(arrowVA);
            hideEl(arrowVBC); hideEl(frictionC); hideEl(frictionAB); hideEl(frictionB);

            switch(step) {
                case 0: showEl(arrowV0); break;
                case 1: showEl(arrowVC); showEl(frictionC); showEl(frictionAB); break;
                case 2: showEl(arrowVC); showEl(arrowVA); break;
                case 3: showEl(arrowVC); showEl(frictionC); showEl(frictionB); break;
                case 4: showEl(arrowVA); showEl(arrowVBC); break;
            }

            stepIndicator.textContent = '步骤 ' + step + '/4';
            btnPrev.disabled = (step === 0);
            btnNext.disabled = (step === totalSteps);
        }

        btnNext.addEventListener('click', function() {
            if (currentStep < totalSteps) { currentStep++; updateScene(currentStep); }
        });
        btnPrev.addEventListener('click', function() {
            if (currentStep > 0) { currentStep--; updateScene(currentStep); }
        });

        updateScene(0);
    })();
    </script>
