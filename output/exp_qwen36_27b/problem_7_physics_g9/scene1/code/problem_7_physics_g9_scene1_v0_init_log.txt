    <script>
    (function() {
        'use strict';

        var currentStep = 0;
        var maxStep = 4;

        var btnPrev = document.getElementById('scene1-btn-prev');
        var btnNext = document.getElementById('scene1-btn-next');
        var waterlineLeft = document.getElementById('scene1-waterline-left');
        var waterlineRight = document.getElementById('scene1-waterline-right');
        var blockYi = document.getElementById('scene1-block-yi');
        var blockBing = document.getElementById('scene1-block-bing');

        var step1Elements = [
            document.getElementById('scene1-label-water-left'),
            document.getElementById('scene1-label-water-right')
        ];
        var step2Elements = [
            document.getElementById('scene1-highlight-jia-top-left'),
            document.getElementById('scene1-highlight-jia-top-right'),
            document.getElementById('scene1-label-jia-flat')
        ];
        var step3Elements = [
            document.getElementById('scene1-label-yi-submerged'),
            document.getElementById('scene1-label-bing-above')
        ];
        var step4Elements = [
            document.getElementById('scene1-conclusion-text')
        ];

        function showElement(el) {
            el.style.display = '';
            el.classList.add('visible');
        }
        function hideElement(el) {
            el.classList.remove('visible');
            el.style.display = 'none';
        }

        function updateDisplay() {
            btnPrev.disabled = (currentStep === 0);
            btnNext.disabled = (currentStep === maxStep);

            if (currentStep >= 1) {
                waterlineLeft.setAttribute('stroke', '#1565c0');
                waterlineLeft.setAttribute('stroke-width', '3');
                waterlineRight.setAttribute('stroke', '#1565c0');
                waterlineRight.setAttribute('stroke-width', '3');
                step1Elements.forEach(showElement);
            } else {
                waterlineLeft.setAttribute('stroke', '#000000');
                waterlineLeft.setAttribute('stroke-width', '2');
                waterlineRight.setAttribute('stroke', '#000000');
                waterlineRight.setAttribute('stroke-width', '2');
                step1Elements.forEach(hideElement);
            }

            if (currentStep >= 2) {
                step2Elements.forEach(showElement);
            } else {
                step2Elements.forEach(hideElement);
            }

            if (currentStep >= 3) {
                blockYi.setAttribute('stroke', '#c62828');
                blockYi.setAttribute('stroke-width', '3');
                blockBing.setAttribute('stroke', '#c62828');
                blockBing.setAttribute('stroke-width', '3');
                step3Elements.forEach(showElement);
            } else {
                blockYi.setAttribute('stroke', '#000000');
                blockYi.setAttribute('stroke-width', '2');
                blockBing.setAttribute('stroke', '#000000');
                blockBing.setAttribute('stroke-width', '2');
                step3Elements.forEach(hideElement);
            }

            if (currentStep >= 4) {
                step4Elements.forEach(showElement);
            } else {
                step4Elements.forEach(hideElement);
            }
        }

        btnPrev.addEventListener('click', function() {
            if (currentStep > 0) { currentStep--; updateDisplay(); }
        });
        btnNext.addEventListener('click', function() {
            if (currentStep < maxStep) { currentStep++; updateDisplay(); }
        });

        // 初始化：所有步骤元素隐藏
        step1Elements.concat(step2Elements, step3Elements, step4Elements).forEach(function(el) {
            el.style.display = 'none';
            el.classList.remove('visible');
        });
        updateDisplay();
    })();
    </script>
