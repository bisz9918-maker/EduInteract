    <script>
    (function() { 'use strict';
        var currentStep = 0;
        var totalSteps = 3;
        var btnPrev = document.getElementById('scene2-btn-prev');
        var btnNext = document.getElementById('scene2-btn-next');
        var stepIndicator = document.getElementById('scene2-step-indicator');
        var step1Els = document.querySelectorAll('.step1-el');
        var step2Els = document.querySelectorAll('.step2-el');
        function updateStep() {
            btnPrev.disabled = (currentStep === 0);
            btnNext.disabled = (currentStep >= totalSteps - 1);
            stepIndicator.textContent = '步骤 ' + (currentStep + 1) + ' / ' + totalSteps;
            step1Els.forEach(function(el) {
                if (currentStep >= 1) { el.classList.add('visible'); }
                else { el.classList.remove('visible'); }
            });
            step2Els.forEach(function(el) {
                if (currentStep >= 2) { el.classList.add('visible'); }
                else { el.classList.remove('visible'); }
            });
        }
        btnNext.addEventListener('click', function() {
            if (currentStep < totalSteps - 1) { currentStep++; updateStep(); }
        });
        btnPrev.addEventListener('click', function() {
            if (currentStep > 0) { currentStep--; updateStep(); }
        });
        updateStep();
    })();
    </script>
