    <script>
    (function() {
        'use strict';
        var currentStep = 0;

        function updateScene() {
            // 更新步骤指示
            document.getElementById('scene1-step-indicator').textContent = '步骤 ' + currentStep + '/3';

            // 更新按钮状态
            var prevBtn = document.getElementById('scene1-btn-prev');
            var nextBtn = document.getElementById('scene1-btn-next');
            prevBtn.disabled = (currentStep === 0);
            nextBtn.disabled = (currentStep === 3);

            // 更新各步骤元素显隐
            for (var i = 1; i <= 3; i++) {
                var group = document.getElementById('scene1-step' + i + '-group');
                if (i <= currentStep) {
                    group.style.display = '';
                    group.style.opacity = '1';
                } else {
                    group.style.opacity = '0';
                    (function(g) {
                        setTimeout(function() {
                            if (g.style.opacity === '0') {
                                g.style.display = 'none';
                            }
                        }, 500);
                    })(group);
                }
            }
        }

        // 按钮事件
        document.getElementById('scene1-btn-prev').addEventListener('click', function() {
            if (currentStep > 0) {
                currentStep--;
                updateScene();
            }
        });

        document.getElementById('scene1-btn-next').addEventListener('click', function() {
            if (currentStep < 3) {
                currentStep++;
                updateScene();
            }
        });

        // 初始化
        updateScene();
    })();
    </script>
